Implementation details
- We will allocate output as torch.zeros((batch_seq_len, hidden_size), dtype=torch.bfloat16, device=device) to ensure zero initialization.
- Triton scatter kernel:
  - One program per token.
  - A while loop that iterates over hidden_size in chunks of BLOCK_SIZE columns. For each chunk, load the corresponding slice from expert_outputs[i, :] and atomic_add into out[token_indices[i], col:col+BLOCK_SIZE].
  - Use masks for the tail when hidden_size is not a multiple of BLOCK_SIZE.
  - Keep BLOCK_SIZE as a compile-time constant (e.g., 128 or 256). For the provided configurations, hidden_size is often 256, so BLOCK_SIZE=256 works well and simplifies masks. If other configs appear with different hidden_size, we can generalize by iterating over chunks.

- Define Triton scatter-add kernel: scatter_add_kernel

```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_kernel(
    out_ptr,                       # *fp16
    exp_ptr,                       # *fp16
    token_indices_ptr,             # *int32
    batch_seq_len: tl.constexpr,   # int
    hidden_size: tl.constexpr,     # int
    n_tokens: tl.constexpr,        # int
    stride_row: tl.constexpr,      # int (out.stride(0))
    stride_col: tl.constexpr,      # int (out.stride(1))
    exp_stride_row: tl.constexpr,  # int (exp.stride(0))
    exp_stride_col: tl.constexpr,  # int (exp.stride(1))
    BLOCK_SIZE: tl.constexpr       # int, e.g., 256
):
    pid = tl.program_id(0)  # token id
    # bounds check: skip if pid >= n_tokens
    if pid >= n_tokens:
        return

    # Load token index (int32)
    idx = tl.load(token_indices_ptr + pid)
    # If idx is out of range, skip (defensive, though inputs are valid)
    if idx < 0 or idx >= batch_seq_len:
        return

    # Loop over hidden columns in chunks of BLOCK_SIZE
    col = 0
    while col < hidden_size:
        cols = col + tl.arange(0, BLOCK_SIZE)
        mask = cols < hidden_size

        # Load expert slice for this token
        src = tl.load(
            exp_ptr + pid * exp_stride_row + cols * exp_stride_col,
            mask=mask,
            other=0.0
        )

        # Compute destination pointers for this row and columns
        dest_ptrs = out_ptr + idx * stride_row + cols * stride_col

        # Atomic add into output
        tl.atomic_add(dest_ptrs, src, mask=mask)

        col += BLOCK_SIZE
```

import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_kernel(
    out_ptr,                       # *fp16
    exp_ptr,                       # *fp16
    token_indices_ptr,             # *int32
    batch_seq_len: tl.constexpr,   # int
    hidden_size: tl.constexpr,     # int
    n_tokens: tl.constexpr,        # int
    stride_row: tl.constexpr,      # int (out.stride(0))
    stride_col: tl.constexpr,      # int (out.stride(1))
    exp_stride_row: tl.constexpr,  # int (exp.stride(0))
    exp_stride_col: tl.constexpr,  # int (exp.stride(1))
    BLOCK_SIZE: tl.constexpr       # int, e.g., 256
):
    pid = tl.program_id(0)  # token id
    if pid >= n_tokens:
        return

    # Load token index (int32)
    idx = tl.load(token_indices_ptr + pid)
    if idx < 0 or idx >= batch_seq_len:
        return

    # Iterate over hidden columns in chunks
    col = 0
    while col < hidden_size:
        cols = col + tl.arange(0, BLOCK_SIZE)
        mask = cols < hidden_size

        # Load expert slice for this token
        src = tl.load(
            exp_ptr + pid * exp_stride_row + cols * exp_stride_col,
            mask=mask,
            other=0.0
        )

        # Destination pointers
        dest_ptrs = out_ptr + idx * stride_row + cols * stride_col

        # Atomic add
        tl.atomic_add(dest_ptrs, src, mask=mask)

        col += BLOCK_SIZE


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states, expert_outputs, token_indices):
        # Shapes
        batch_size = axes_and_scalars["batch_size"]
        seq_len = axes_and_scalars["seq_len"]
        hidden_size = axes_and_scalars["hidden_size"]
        batch_seq_len = batch_size * seq_len
        n_tokens = expert_outputs.shape[0]

        # Output buffer: zero-initialized for correct accumulation
        # Note: device comes from final_hidden_states
        out = torch.zeros((batch_seq_len, hidden_size), dtype=torch.bfloat16, device=final_hidden_states.device)

        # Ensure inputs are contiguous
        expert_outputs = expert_outputs.contiguous()
        token_indices_i32 = token_indices.to(torch.int32).contiguous()

        # Strides (in elements)
        stride_row = out.stride(0)
        stride_col = out.stride(1)
        exp_stride_row = expert_outputs.stride(0)
        exp_stride_col = expert_outputs.stride(1)

        # Launch Triton kernel: one program per token
        grid = (n_tokens,)
        scatter_add_kernel[grid](
            out, expert_outputs, token_indices_i32,
            batch_seq_len=batch_seq_len,
            hidden_size=hidden_size,
            n_tokens=n_tokens,
            stride_row=stride_row,
            stride_col=stride_col,
            exp_stride_row=exp_stride_row,
            exp_stride_col=exp_stride_col,
            BLOCK_SIZE=256,  # tuned for hidden_size=256; generalizes via while loop
            num_warps=4
        )

        return out

Notes
- The Triton kernel performs the scatter-add with atomic adds. It handles duplicates correctly and iterates over hidden columns in chunks. With hidden_size=256 (as in the provided configs), BLOCK_SIZE=256 makes the loop trivial and avoids tail masks.
- We use torch.zeros for output initialization to ensure correctness. The heavy computation is done by the Triton kernel, and no torch tensor is modified by PyTorch operations in the hot path beyond trivial conversions.
- To optimize further (after correctness is established), you can:
  - Support arbitrary hidden_size by keeping BLOCK_SIZE at 128 or 256 and letting the while loop iterate over all chunks; this already generalizes.
  - Tune num_warps based on GPU. For 256 columns, 4 warps is a reasonable default. For larger hidden sizes, 8 warps can help.
