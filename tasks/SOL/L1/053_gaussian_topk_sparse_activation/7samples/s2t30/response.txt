Thank you for the detailed feedback. The last change introduced a “fast path” in the reduction kernel that assumed N <= BLOCK_SIZE. That simplification breaks for larger N (e.g., intermediate_size=12288), causing incorrect results and the reported runtime errors. To restore correctness and prevent recurrence, I will:

- Revert to a robust, mask-based chunked reduction over N that works for any N. No fast path assumptions.
- Keep inv_norm_cdf(target_sparsity) in a Triton scalar kernel using the Abramowitz & Stegun approximation. Ensure no torch operations on tensors in the host code.
- Keep the 2D tiled gating kernel: grid over rows and tiles of N, each program handles one tile. This approach was previously validated to be correct across all workloads.
- Tune BLOCK_SIZE and num_warps for good performance while maintaining correctness. We’ll use BLOCK_SIZE_RS=1024 for reduction and BLOCK_SIZE_GT=1024 for gating, which balances vectorization and parallelism for common sizes and avoids the pitfalls of per-row single-pass specialization.

Forward remains Triton-only:
- No torch.mean, torch.std, torch.special, torch.tensor, or elementwise math on tensors in the host code. Only allocation, dtype casting, and kernel launches.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def reduce_mean_std_2d(x_ptr, mean_ptr, std_ptr, N, BLOCK_SIZE: tl.constexpr):
    """
    Triton reduction: compute per-row mean and population std (unbiased=False) across last dim.
    x_ptr: pointer to input flattened as [rows, N] (row-major), contiguous
    mean_ptr/std_ptr: per-row outputs [rows], float32
    N: number of columns (int32)
    """
    row_id = tl.program_id(axis=0)
    sum_val = 0.0
    sum_sq = 0.0

    start = 0
    while start < N:
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < N
        # Load a chunk of the row
        x = tl.load(x_ptr + row_id * N + offs, mask=mask, other=0.0)
        # Accumulate sum and sum of squares across the chunk
        sum_val += tl.sum(x)
        sum_sq += tl.sum(x * x)
        start += BLOCK_SIZE

    mean = sum_val / N
    var = sum_sq / N - mean * mean  # population variance (unbiased=False)
    std = tl.sqrt(var)
    tl.store(mean_ptr + row_id, mean)
    tl.store(std_ptr + row_id, std)


@triton.jit
def compute_inv_ndtri_scalar(inv_ptr, p, BLOCK_SIZE: tl.constexpr):
    """
    Triton scalar kernel: compute inverse standard normal CDF (quantile) for p in (0,1).
    Uses Abramowitz & Stegun approximation (formula 26.2.23).
    Writes result to inv_ptr[0] as float32.
    """
    # p is a scalar float argument; no need to load
    # Constants for the approximation
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p_high = 1.0 - p_low

    # Lower region
    if p < p_low:
        q = tl.sqrt(-2.0 * tl.log(p))
        result = (((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6) / \
                 ((((d1*q + d2)*q + d3)*q + d4)*q + 1.0)
        tl.store(inv_ptr, result)
        return

    # Central region
    if p <= p_high:
        q = p - 0.5
        r = q * q
        result = (((((a1*r + a2)*r + a3)*r + a4)*r + a5)*r + a6) * q / \
                 (((((b1*r + b2)*r + b3)*r + b4)*r + b5)*r + 1.0)
        tl.store(inv_ptr, result)
        return

    # Upper region
    q = tl.sqrt(-2.0 * tl.log(1.0 - p))
    result = -(((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6) / \
             ((((d1*q + d2)*q + d3)*q + d4)*q + 1.0)
    tl.store(inv_ptr, result)
    return


@triton.jit
def gate_rows_2d(x_ptr, mean_ptr, std_ptr, inv_ptr, out_ptr, rows, N, BLOCK_SIZE: tl.constexpr):
    """
    2D Triton kernel: elementwise gating per row over tiles of N.
    For each element x[i], y[i] = max(0, x[i] - (mean[row] + std[row] * inv_ptr[0])).
    Grid: (rows, num_tiles), where num_tiles = ceil_div(N, BLOCK_SIZE).
    x_ptr: [rows, N], mean_ptr/std_ptr: [rows], out_ptr: [rows, N]
    """
    row = tl.program_id(axis=0)
    tile = tl.program_id(axis=1)
    start = tile * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    # Load per-row stats
    mean_row = tl.load(mean_ptr + row)
    std_row = tl.load(std_ptr + row)
    inv_cdf = tl.load(inv_ptr)  # scalar

    # Compute threshold per element and apply gating
    threshold = mean_row + std_row * inv_cdf
    x = tl.load(x_ptr + row * N + offs, mask=mask, other=0.0)
    y = x - threshold
    y = tl.maximum(y, 0.0)  # ReLU
    tl.store(out_ptr + row * N + offs, y, mask=mask)


def run(inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
    """
    Triton-optimized Gaussian-based top-k sparse activation.
    - Compute per-row mean and std (population) across last dim (feature dimension).
    - Compute inv_norm_cdf(target_sparsity) using Triton scalar kernel.
    - Apply y = max(0, x - (mean + std * inv_cdf)) via Triton 2D kernel.
    Returns: tensor with same shape as inputs, dtype bfloat16.
    """
    # Early return if no sparsity requested
    if target_sparsity == 0.0:
        return inputs

    # Compute in float32 for stability
    x = inputs
    x_f32 = x.to(torch.float32)

    # Flatten to 2D [rows, N] where N = intermediate_size
    N = x_f32.shape[-1]
    rows = x_f32.numel() // N
    x_2d = x_f32.view(rows, N)

    # 1) Triton reduction to compute per-row mean and std (unbiased=False)
    mean = torch.empty(rows, device=x_f32.device, dtype=torch.float32)
    std = torch.empty(rows, device=x_f32.device, dtype=torch.float32)
    BLOCK_SIZE_RS = 1024
    reduce_mean_std_2d[(rows,)](x_2d, mean, std, N, BLOCK_SIZE=BLOCK_SIZE_RS, num_warps=4)

    # 2) Triton scalar kernel: inv_norm_cdf(target_sparsity)
    inv_cdf_buf = torch.empty(1, device=x_f32.device, dtype=torch.float32)
    compute_inv_ndtri_scalar[(1,)](inv_cdf_buf, float(target_sparsity), BLOCK_SIZE=1)

    # 3) Triton 2D kernel: elementwise gating over tiles of N
    out_2d = torch.empty((rows, N), device=x_f32.device, dtype=torch.float32)
    BLOCK_SIZE_GT = 1024
    num_tiles = (N + BLOCK_SIZE_GT - 1) // BLOCK_SIZE_GT
    grid = (rows, num_tiles)
    gate_rows_2d[grid](x_2d, mean, std, inv_cdf_buf, out_2d, rows, N, BLOCK_SIZE=BLOCK_SIZE_GT, num_warps=4)

    # Reshape back to original and cast to bfloat16
    out = out_2d.view(*x.shape).to(torch.bfloat16)
    return out
```

- Kept the Triton scalar kernel for inv_norm_cdf, avoiding torch operations on tensors.
- Retained the 2D tiled gating kernel with grid parallelism over tiles, which has proven correct and fast.
