import torch
import triton
import triton.language as tl


@triton.jit
def _rowwise_stats_kernel(X, MEANS, STDs, F, BLOCK: tl.constexpr):
    """
    Compute per-row mean and population std across last dimension (features).
    X: [B*S, F] input (float32), flattened rows
    MEANS: [B*S] output means (float32)
    STDs: [B*S] output stds (float32)
    """
    pid = tl.program_id(axis=0)
    row = pid
    start = row * F

    offs = tl.arange(0, BLOCK)
    sum_val = 0.0
    sum_sq = 0.0

    # Loop over feature chunks
    for k in range(0, F, BLOCK):
        idx = k + offs
        mask = idx < F
        # X is [B*S, F] contiguous: pointer offset = start + idx
        x = tl.load(X + start + idx, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    # mean = sum / F, population std = sqrt(sum_sq/F - mean^2)
    mean = sum_val / F
    var = sum_sq / F - mean * mean
    std = tl.sqrt(var)

    # Write results
    tl.store(MEANS + row, mean)
    tl.store(STDs + row, std)


@triton.jit
def _sparsify_relu_kernel(X, MEANS, STDs, ICDF, OUT, F, BLOCK: tl.constexpr):
    """
    Apply sparsification with ReLU: OUT = max(0, X - (MEANS + STDs * ICDF))
    X: [B*S, F] input (float32)
    MEANS, STDs: [B*S] scalars
    ICDF: 1-element tensor with inverse normal CDF of target_sparsity
    OUT: [B*S, F] output (float32)
    """
    pid = tl.program_id(axis=0)
    row = pid
    start_in = row * F
    start_out = row * F

    mean = tl.load(MEANS + row)
    std = tl.load(STDs + row)
    # Load icdf scalar
    icdf = tl.load(ICDF)  # 1-element tensor
    thresh = mean + std * icdf
    # Loop over feature chunks
    for k in range(0, F, BLOCK):
        idx = k + tl.arange(0, BLOCK)
        mask = idx < F
        x = tl.load(X + start_in + idx, mask=mask, other=0.0)
        y = x - thresh
        y = tl.maximum(y, 0.0)  # ReLU
        tl.store(OUT + start_out + idx, y, mask=mask)


def run(inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
    """
    Triton-based Gaussian-based top-k sparse activation:
      - Compute per-(batch, seq) mean and std of input features (fp32).
      - Compute icdf(target_sparsity) via Triton (Abramowitz & Stegun 5.2.23 central region).
      - Apply sparsification: max(0, input - (mean + std * icdf)).

    Args:
        inputs: Input tensor of shape [batch_size, seq_len, intermediate_size]
        target_sparsity: Float in [0, 1] indicating target sparsity level.

    Returns:
        Sparsified tensor of same shape as input, in bfloat16.
    """
    B, S, F = inputs.shape
    device = inputs.device
    dtype = inputs.dtype

    # Flatten rows for simpler indexing
    x_rows = inputs.reshape(B * S, F).to(torch.float32)
    out_rows = torch.empty((B * S, F), dtype=torch.float32, device=device)

    # Allocate per-row means and stds
    means = torch.empty((B * S,), dtype=torch.float32, device=device)
    stds = torch.empty((B * S,), dtype=torch.float32, device=device)

    # Choose BLOCK and launch parameters based on F
    if F >= 2048:
        BLOCK_STATS = 2048
        num_warps_stats = 8
        num_stages_stats = 3
    else:
        BLOCK_STATS = 1024
        num_warps_stats = 4
        num_stages_stats = 2

    # Launch rowwise stats kernel: one program per row
    grid_stats = (B * S,)
    _rowwise_stats_kernel[grid_stats](
        x_rows, means, stds, F, BLOCK=BLOCK_STATS, num_warps=num_warps_stats, num_stages=num_stages_stats
    )

    # Compute icdf for target_sparsity using Triton (single scalar)
    icdf = torch.empty((), dtype=torch.float32, device=device)
    p_dev = torch.empty((), dtype=torch.float32, device=device)
    p_dev.fill_(float(target_sparsity))
    _icdf_ndtri_kernel[(1,)](
        p_dev, icdf, BLOCK=1, num_warps=1, num_stages=1
    )

    # Choose BLOCK for sparsification
    BLOCK_SP = 2048 if F >= 2048 else 1024
    num_warps_sp = 8 if BLOCK_SP == 2048 else 4
    num_stages_sp = 3 if BLOCK_SP == 2048 else 2

    # Launch sparsify + ReLU kernel: one program per row
    _sparsify_relu_kernel[grid_stats](
        x_rows, means, stds, icdf, out_rows, F, BLOCK=BLOCK_SP, num_warps=num_warps_sp, num_stages=num_stages_sp
