```python
import torch
import torch.nn.functional as F
import math

# Triton is required for the optimized implementation
import triton
import triton.language as tl


@triton.jit
def mean_std_kernel(
    X_ptr,          # *f32, input tensor pointer
    mean_ptr,       # *f32, output mean per (b, s)
    var_ptr,        # *f32, output variance per (b, s)
    B: tl.constexpr,
    S: tl.constexpr,
    F: tl.constexpr,
    stride_b: tl.constexpr,
    stride_s: tl.constexpr,
    stride_f: tl.constexpr,
    BLOCK_F: tl.constexpr,
):
    # One program per (b, s)
    b = tl.program_id(0)
    s = tl.program_id(1)
    # Compute base offset for this (b, s)
    base = b * stride_b + s * stride_s

    # Accumulators for sum and sum of squares
    sum_val = 0.0
    sum_sq = 0.0

    # Iterate over F in chunks of BLOCK_F
    off = 0
    while off < F:
        idx = off + tl.arange(0, BLOCK_F)
        mask = idx < F
        ptrs = X_ptr + base + idx * stride_f
        vals = tl.load(ptrs, mask=mask, other=0.0)
        sum_val += tl.sum(vals, axis=0)
        sum_sq += tl.sum(vals * vals, axis=0)
        off += BLOCK_F

    n = F
    mean = sum_val / n
    # population variance (unbiased=False), std = sqrt(mean)
    var = sum_sq / n - mean * mean
    # Store results
    tl.store(mean_ptr + b * S + s, mean)
    tl.store(var_ptr + b * S + s, var)


@triton.jit
def relu_threshold_kernel(
    X_ptr,          # *f32, input tensor pointer
    mean_ptr,       # *f32, per (b, s) mean
    var_ptr,        # *f32, per (b, s) var
    OUT_ptr,        # *f32, output tensor pointer
    B: tl.constexpr,
    S: tl.constexpr,
    F: tl.constexpr,
    stride_b: tl.constexpr,
    stride_s: tl.constexpr,
    stride_f: tl.constexpr,
    target_sparsity: tl.constexpr,  # scalar float
    BLOCK_F: tl.constexpr,
):
    # 3D launch grid: (B, S, ceil(F / BLOCK_F))
    b = tl.program_id(0)
    s = tl.program_id(1)
    c = tl.program_id(2)
    start = c * BLOCK_F
    offs = start + tl.arange(0, BLOCK_F)
    mask = offs < F

    # Load input chunk
    base = b * stride_b + s * stride_s
    x_ptrs = X_ptr + base + offs * stride_f
    x = tl.load(x_ptrs, mask=mask, other=0.0)

    # Load per-(b, s) mean and std
    mean = tl.load(mean_ptr + b * S + s)
    var = tl.load(var_ptr + b * S + s)
    std = tl.sqrt(var)

    # Compute cutoff and apply ReLU(x - cutoff)
    cutoff = mean + std * target_sparsity
    y = tl.maximum(x - cutoff, 0.0)

    # Store output
    out_ptrs = OUT_ptr + base + offs * stride_f
    tl.store(out_ptrs, y, mask=mask)


@torch.no_grad()
def run(inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
    """
    Gaussian-based top-k sparse activation.
    Computes adaptive sparsity threshold based on input statistics:
      - For each (b, s), compute mean and std across feature dimension F.
      - threshold = mean + std * ndtri(target_sparsity)
      - out = max(0, inputs - threshold)

    Args:
        inputs: Input tensor of shape [batch_size, seq_len, intermediate_size]
        target_sparsity: Float in [0, 1].
    Returns:
        Sparsified tensor of same shape as input, in bfloat16.
    """
    # Early return if no sparsity requested
    if target_sparsity == 0.0:
        return inputs

    # Ensure input is float32 for numerics, keep original dtype for output
    x = inputs.to(torch.float32)
    B, S, F = x.shape

    # Compute strides in elements
    stride_b = x.stride(0)
    stride_s = x.stride(1)
    stride_f = x.stride(2)

    # Allocate per-(b, s) mean and variance
    mean = torch.empty((B, S), dtype=torch.float32, device=x.device)
    var = torch.empty((B, S), dtype=torch.float32, device=x.device)

    # Launch mean and std reduction kernel
    BLOCK_F = 1024
    grid = (B, S)
    mean_std_kernel[grid](
        x, mean, var,
        B, S, F,
        stride_b, stride_s, stride_f,
        BLOCK_F,
        num_warps=4, num_stages=1
    )

    # Compute invnormcdf(target_sparsity) once on host (A&S approximation)
    # Keep as Python float; Triton will accept scalar arg.
    # Using the same constants as the original _ndtri.
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p = float(target_sparsity)
    if p <= p_low:
        # lower region
        q = torch.sqrt(torch.tensor(-2.0 * math.log(p), dtype=torch.float32, device=x.device)).item()
        z = (((((c1 * q) + c2) * q + c3) * q + c4) * q + c5) * q + c6
        d = (((d1 * q + d2) * q + d3) * q + d4) * q + 1.0
        std_multiplier = z / d
    elif p >= 1.0 - p_low:
        # upper region
        q = torch.sqrt(torch.tensor(-2.0 * math.log(1.0 - p), dtype=torch.float32, device=x.device)).item()
        z = (((((c1 * q) + c2) * q + c3) * q + c4) * q + c5) * q + c6
        d = (((d1 * q + d2) * q + d3) * q + d4) * q + 1.0
        std_multiplier = - (z / d)
    else:
        # central region
        q = p - 0.5
        r = q * q
        a = (((((a1 * r) + a2) * r + a3) * r + a4) * r + a5) * r + a6
        b = (((((b1 * r) + b2) * r + b3) * r + b4) * r + b5) * r + 1.0
        std_multiplier = a * q / b

    # Allocate output (float32 for computation)
    out = torch.empty_like(x)

    # Launch elementwise ReLU threshold kernel with a 3D grid
    BLOCK_F = 1024
    grid = (B, S, triton.cdiv(F, BLOCK_F))
    relu_threshold_kernel[grid](
        x, mean, var, out,
        B, S, F,
        stride_b, stride_s, stride_f,
        std_multiplier,
        BLOCK_F,
        num_warps=4, num_stages=1
    )

    # Return in bfloat16 to match original behavior
    return out.to(torch.bfloat16)


class Model(torch.nn.Module):
    def forward(self, *args):
        # Keep the original signature and behavior
        # Delegate to run(inputs, target_sparsity). The evaluator passes two arguments.
        # If someone calls with only one, we can handle it by extracting target_sparsity=0.01 default.
        if len(args) == 1:
            # Assume default sparsity 0.01
            return run(args[0], 0.01)
        elif len(args) == 2:
            return run(args[0], args[1])
        else:
            # Fallback to original signature expectation: second arg is target_sparsity
            # If more than 2, take the last tensor as inputs and the second float as sparsity
            if len(args) >= 2 and isinstance(args[1], (float, int)):
                return run(args[0], float(args[1]))
            # Otherwise, default sparsity
            return run(args[0], 0.01)
```
