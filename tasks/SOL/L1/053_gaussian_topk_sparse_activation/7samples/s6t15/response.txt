Observations and plan:
- The main computation is two passes over the feature dimension per (batch, seq) row:
  1) Rowwise reduction to compute mean and population std (unbiased=False), done in _rowwise_stats_kernel.
  2) Elementwise sparsification using ReLU(input - (mean + std * icdf)), done in _sparsify_relu_kernel.

- The icdf computation is a single scalar per call using the Abramowitz & Stegun 5.2.23 central-region formula with a5 negative. This is negligible overhead compared to the rowwise kernels, and we keep it inside a Triton kernel to satisfy the “Triton-only” constraint.

Performance improvements:
- Continue using one program per (batch, seq) row, which scales well and keeps code simple.
- Choose larger BLOCK size (2048) when F >= 2048 to reduce loop iterations and improve memory throughput. Keep 1024 as a safe default otherwise. Both are compile-time constexpr values in Triton. Use num_warps=8 for BLOCK=2048 and num_warps=4 for 1024, with num_stages=2 for better pipelining on memory-bound loops.
- Keep masks in loads/stores to avoid out-of-bounds access and ensure correctness for any F. This prevents illegal memory access and runtime errors.

- Avoid any torch operations in forward except for creating a 1-element device tensor for the probability and the output buffer. All computations occur in Triton kernels.

Final code (ModelNew entry point):

```python
import torch
import triton
import triton.language as tl


@triton.jit
def _rowwise_stats_kernel(X, B, S, F, MEANS, STDs, BLOCK: tl.constexpr):
    """
    Compute per-(batch, seq) row mean and population std across last dimension (features).
    X: [B, S, F] input (float32)
    MEANS: [B*S] output means (float32)
    STDs: [B*S] output stds (float32)
    """
    pid = tl.program_id(axis=0)
    # Map program id to (b, s)
    b = pid // S
    s = pid % S

    # Compute base pointer offset for this row
    base = (b * S + s) * F
    # Accumulators
    sum_val = 0.0
    sum_sq = 0.0

    # Loop over features in chunks of BLOCK
    for off in range(0, F, BLOCK):
        idx = off + tl.arange(0, BLOCK)
        mask = idx < F
        x = tl.load(X + base + idx, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    n = F
    mean = sum_val / n
    # Population std (unbiased=False): sqrt(E[x^2] - (E[x])^2)
    var = sum_sq / n - mean * mean
    std = tl.sqrt(var)

    # Store results for this row
    tl.store(MEANS + pid, mean)
    tl.store(STDs + pid, std)


@triton.jit
def _sparsify_relu_kernel(X, MEANS, STDs, ICDF, OUT, B, S, F, BLOCK: tl.constexpr):
    """
    Apply adaptive threshold sparsification per row:
      threshold = mean + std * ICDF
      OUT = max(0, X - threshold)
    X: [B, S, F] input (float32)
    MEANS: [B*S] per-row means (float32)
    STDs: [B*S] per-row stds (float32)
    ICDF: [1] scalar inverse CDF (float32)
    OUT: [B, S, F] output (float32)
    """
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S
    base = (b * S + s) * F

    mean = tl.load(MEANS + pid)
    std = tl.load(STDs + pid)
    threshold = mean + std * tl.load(ICDF)

    for off in range(0, F, BLOCK):
        idx = off + tl.arange(0, BLOCK)
        mask = idx < F
        x = tl.load(X + base + idx, mask=mask, other=0.0)
        y = x - threshold
        y = tl.maximum(y, 0.0)  # ReLU
        tl.store(OUT + base + idx, y, mask=mask)


@triton.jit
def _icdf_ndtri_kernel(P, ICDF, BLOCK: tl.constexpr):
    """
    Compute inverse normal CDF for scalar probability P using Abramowitz & Stegun 5.2.23 central region.
    P: [1] device tensor containing probability (float32)
    ICDF: [1] device tensor to store icdf (float32)
    """
    # Central region: q = p - 0.5
    p = tl.load(P)
    q = p - 0.5
    r = q * q

    # Coefficients for numerator polynomial
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01  # negative as per A&S
    a6 = 2.506628277459239e+00

    # Coefficients for denominator polynomial
    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    poly_num = (((((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6)
    poly_den = (((((b1 * r + b2) * r + b3) * r + b4) * r + b5) * r + 1.0)
    icdf = poly_num * q / poly_den

    # Store result
    tl.store(ICDF, icdf)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        """
        Triton-optimized Gaussian-based top-k sparse activation.

        Computes adaptive sparsity threshold based on input statistics:
        1) Compute mean and std of input across feature dimension
        2) threshold = mean + std * norm.icdf(target_sparsity)
        3) Apply ReLU(input - threshold) to create sparse activations

        Returns:
            Sparsified tensor of same shape as input.
        """
        # If no sparsity requested, return as-is
        if target_sparsity == 0.0:
            return x

        # Compute in float32 for numerical stability
        x_fp32 = x.to(torch.float32)

        B, S, F = x_fp32.shape

        # Allocate per-row means and stds
        means = torch.empty((B * S,), dtype=torch.float32, device=x.device)
        stds = torch.empty((B * S,), dtype=torch.float32, device=x.device)

        # Output buffer (float32 for compute; convert at the end)
        out = torch.empty((B, S, F), dtype=torch.float32, device=x.device)

        # Choose BLOCK based on F (constexpr constants)
        if F >= 2048:
            BLOCK_STATS = 2048
            num_warps_stats = 8
            BLOCK_SP = 2048
            num_warps_sp = 8
        else:
            BLOCK_STATS = 1024
            num_warps_stats = 4
            BLOCK_SP = 1024
            num_warps_sp = 4

        # Launch rowwise stats kernel
        grid = (B * S,)
        _rowwise_stats_kernel[grid](
            x_fp32, B, S, F, means, stds, BLOCK=BLOCK_STATS, num_warps=num_warps_stats, num_stages=2
        )

        # Compute icdf for target_sparsity using Triton (single scalar)
        icdf = torch.empty((), dtype=torch.float32, device=x.device)
        p_dev = torch.empty((), dtype=torch.float32, device=x.device)
        p_dev.fill_(float(target_sparsity))
        _icdf_ndtri_kernel[(1,)](
            p_dev, icdf, BLOCK=1, num_warps=1, num_stages=1
        )

        # Launch sparsify + ReLU kernel
        _sparsify_relu_kernel[grid](
            x_fp32, means, stds, icdf, out, B, S, F, BLOCK=BLOCK_SP, num_warps=num_warps_sp, num_stages=2
        )

        # Return in bfloat16, matching original behavior
        return out.to(torch.bfloat16)
```
