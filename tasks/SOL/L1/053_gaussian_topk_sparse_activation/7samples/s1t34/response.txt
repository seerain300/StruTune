import torch
import triton
import triton.language as tl


# Kernel 1: compute per-row mean and population std over last dim N.
# X: [rows, N] where rows = batch_size * seq_len, N = intermediate_size.
# MEAN: fp32 [rows], STD: fp32 [rows]
@triton.jit
def row_stats_kernel(
    X_ptr,           # *f32, contiguous [rows, N]
    MEAN_ptr,        # *f32, [rows]
    STD_ptr,         # *f32, [rows]
    rows: tl.constexpr,
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    row_id = tl.program_id(0)
    sum_val = 0.0
    sum_sq = 0.0
    # Iterate over columns in chunks of BLOCK
    for col_start in range(0, N, BLOCK):
        offs = col_start + tl.arange(0, BLOCK)
        mask = offs < N
        idx = row_id * N + offs
        x = tl.load(X_ptr + idx, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_val / N
    var = sum_sq / N - mean * mean
    std = tl.sqrt(var)
    # Store results
    tl.store(MEAN_ptr + row_id, mean)
    tl.store(STD_ptr + row_id, std)


# Kernel 2: scalar inverse normal CDF using Abramowitz & Stegun 5.2.23
@triton.jit
def ndtri_kernel(
    p_ptr,           # *f32, [1] input probability
    q_ptr,           # *f32, [1] output z-score
):
    p = tl.load(p_ptr)  # scalar
    # constants
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p_high = 1.0 - p_low

    # Lower region
    q_lower = (((((c1 * tl.sqrt(-2.0 * tl.log(p))) + c2) * tl.sqrt(-2.0 * tl.log(p)) +
                 c3) * tl.sqrt(-2.0 * tl.log(p)) + c4) * tl.sqrt(-2.0 * tl.log(p)) +
               c5) * tl.sqrt(-2.0 * tl.log(p)) + c6) / \
              (((((d1 * tl.sqrt(-2.0 * tl.log(p))) + d2) * tl.sqrt(-2.0 * tl.log(p)) +
                 d3) * tl.sqrt(-2.0 * tl.log(p)) + d4) * tl.sqrt(-2.0 * tl.log(p)) + 1.0)

    # Central region
    q_mid = (((((a1 * (p - 0.5) * (p - 0.5) + a2) * (p - 0.5) * (p - 0.5) + a3) * (p - 0.5) * (p - 0.5) +
                a4) * (p - 0.5) * (p - 0.5) + a5) * (p - 0.5) * (p - 0.5) + a6) * (p - 0.5) * (p - 0.5)) / \
            (((((b1 * (p - 0.5) * (p - 0.5) + b2) * (p - 0.5) * (p - 0.5) + b3) * (p - 0.5) * (p - 0.5) +
               b4) * (p - 0.5) * (p - 0.5) + b5) * (p - 0.5) * (p - 0.5) + 1.0)

    # Upper region
    q_upper = -(((((c1 * tl.sqrt(-2.0 * tl.log(1.0 - p))) + c2) * tl.sqrt(-2.0 * tl.log(1.0 - p)) +
                  c3) * tl.sqrt(-2.0 * tl.log(1.0 - p)) + c4) * tl.sqrt(-2.0 * tl.log(1.0 - p)) +
                c5) * tl.sqrt(-2.0 * tl.log(1.0 - p)) + c6) / \
              (((((d1 * tl.sqrt(-2.0 * tl.log(1.0 - p))) + d2) * tl.sqrt(-2.0 * tl.log(1.0 - p)) +
                 d3) * tl.sqrt(-2.0 * tl.log(1.0 - p)) + d4) * tl.sqrt(-2.0 * tl.log(1.0 - p)) + 1.0)

    # Select region
    q_val = tl.where(p < p_low, q_lower, 0.0)
    q_val = tl.where((p >= p_low) & (p <= p_high), q_mid, q_val)
    q_val = tl.where(p > p_high, q_upper, q_val)

    tl.store(q_ptr, q_val)


# Kernel 3: compute threshold per row: mean[row] + std[row] * multiplier
@triton.jit
def threshold_vec_kernel(
    MEAN_ptr,        # *f32, [rows]
    STD_ptr,         # *f32, [rows]
    MULTIPLIER_ptr,  # *f32, [1] scalar
    THRESH_ptr,      # *f32, [rows]
    rows: tl.constexpr,
):
    row_id = tl.program_id(0)
    mean = tl.load(MEAN_ptr + row_id)
    std = tl.load(STD_ptr + row_id)
    mul = tl.load(MULTIPLIER_ptr)
    thr = mean + std * mul
    tl.store(THRESH_ptr + row_id, thr)


# Kernel 4: elementwise ReLU(x - threshold[row]) over rows
@triton.jit
def relu_threshold_kernel(
    X_ptr,           # *f32, contiguous [rows, N] flattened
    THRESH_ptr,      # *f32, [rows]
    OUT_ptr,         # *f32, flattened [rows, N]
    rows: tl.constexpr,
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    row_id = tl.program_id(0)
    base = row_id * N
    # Load threshold for this row
    thr = tl.load(THRESH_ptr + row_id)
    for col_start in range(0, N, BLOCK):
        offs = col_start + tl.arange(0, BLOCK)
        mask = offs < N
        idx = base + offs
        x = tl.load(X_ptr + idx, mask=mask, other=0.0)
        y = x - thr
        y = tl.where(y > 0.0, y, 0.0)  # ReLU
        tl.store(OUT_ptr + idx, y, mask=mask)


def run(inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
    """
    Triton-optimized version of the original function:
    - Compute per-row mean/std in Triton.
    - Compute multiplier via Triton (A&S approximation).
    - Compute per-row threshold in Triton.
    - Apply ReLU(x - threshold[row]) in Triton.
    - Return bf16 to match original behavior.
    """
    # Ensure input is float32 for computation
    inputs_f32 = inputs.to(torch.float32)
    # Flatten to [rows, N]
    B, S, N = inputs_f32.shape
    rows = B * S
    x2d = inputs_f32.view(rows * N)

    # 1) Compute per-row mean and std in Triton
    mean = torch.empty(rows, device=inputs.device, dtype=torch.float32)
    std = torch.empty(rows, device=inputs.device, dtype=torch.float32)
    grid_stats = (rows,)
    row_stats_kernel[grid_stats](
        x2d, mean, std,
        rows=rows, N=N,
        BLOCK=1024,
        num_warps=8,
    )

    # 2) Compute multiplier = inverse normal CDF of target_sparsity in Triton
    p = torch.empty(1, device=inputs.device, dtype=torch.float32)
    q = torch.empty(1, device=inputs.device, dtype=torch.float32)
    p.fill_(float(target_sparsity))
    ndtri_kernel[(1,)](p, q)

    # 3) Compute threshold per row (fp32)
    threshold = torch.empty(rows, device=inputs.device, dtype=torch.float32)
    threshold_vec_kernel[grid_stats](mean, std, q, threshold, rows)

    # 4) Elementwise ReLU against per-row threshold in Triton
    OUT = torch.empty(rows * N, device=inputs.device, dtype=torch.float32)
    relu_grid = (rows,)
    relu_threshold_kernel[relu_grid](
        x2d, threshold, OUT,
        rows=rows, N=N,
        BLOCK=1024,
        num_warps=4,
    )

    # 5) Reshape and cast to bfloat16 to match original output
    return OUT.view(B, S, N).to(torch.bfloat16)


class ModelNew(torch.nn.Module):
