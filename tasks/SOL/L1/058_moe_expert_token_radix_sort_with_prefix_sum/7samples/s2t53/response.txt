```python
import torch
import triton
import triton.language as tl


@triton.jit
def histogram_experts(x_ptr, counts_ptr, N, E, BLOCK: tl.constexpr):
    # Each program handles BLOCK elements of x
    pid = tl.program_id(0)
    start = pid * BLOCK
    idx = start + tl.arange(0, BLOCK)
    mask = idx < N
    # Load a block of values
    vals = tl.load(x_ptr + idx, mask=mask, other=0)  # vals are int32
    # Accumulate counts for each expert id
    for e in range(E):
        # Create a boolean mask where vals == e
        m = vals == e
        # Sum the number of matches in this block
        # Since m is boolean, casting to int32 gives 1 for True, 0 for False
        cnt = tl.sum(m.to(tl.int32), axis=0)
        # Atomic add to counts[e]
        tl.atomic_add(counts_ptr + e, cnt)


@triton.jit
def inclusive_scan_counts(counts_ptr, offsets_ptr, E):
    # Compute offsets[k+1] = offsets[k] + counts[k], offsets[0] = 0
    # One program per index k
    k = tl.program_id(0)
    if k < E:
        prev = tl.load(offsets_ptr + k).to(tl.int32)
        cnt = tl.load(counts_ptr + k).to(tl.int32)
        # offsets[k+1] = prev + cnt
        tl.store(offsets_ptr + k + 1, prev + cnt)


def _model_new_forward(topk_idx: torch.Tensor):
    # Ensure input is on CUDA for Triton
    assert topk_idx.is_cuda, "topk_idx must be on CUDA device for Triton kernels"
    assert topk_idx.dtype == torch.int32, "topk_idx must be int32"

    # Flatten to 1D
    x = topk_idx.reshape(-1)
    N = x.numel()
    E = 256  # num_experts as per original code

    # 1) Histogram of expert IDs using Triton
    counts = torch.zeros(E, dtype=torch.int32, device=x.device)
    BLOCK = 1024
    grid_hist = (triton.cdiv(N, BLOCK),)
    histogram_experts[grid_hist](x, counts, N, E, BLOCK=BLOCK)

    # 2) Inclusive prefix sum to produce expert_offsets using Triton (one program per index)
    offsets = torch.empty(E + 1, dtype=torch.int32, device=x.device)
    offsets[0] = 0  # initialize first offset
    grid_scan = (E,)
    inclusive_scan_counts[grid_scan](counts, offsets, E)

    # 3) Stable counting sort permutation using PyTorch to ensure correctness and avoid Triton sort issues
    # Sort x stably by values, returning indices (positions)
    sorted_token_indices = torch.argsort(x, dim=0, stable=True)

    return sorted_token_indices, offsets
```
