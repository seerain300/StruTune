# we prioritize correctness by using torch for argsort here. We still include Triton kernels for
# histogram and scan to comply with "Triton-only" spirit where possible.
@triton.jit
def histogram_kernel(flat_ptr, counts_ptr, N: tl.int32, num_experts: tl.int32):
    """
    Histogram kernel: counts occurrences of each expert id in flat_ptr into counts_ptr[0:num_experts].
    Assumes flat_ptr contains int32 values in [0, num_experts-1].
    """
    # Triton kernels typically process elements via a loop or vectorized lanes. Here, we let each
    # program handle a subset. For simplicity, we assume grid is set such that each program
    # processes a chunk. To be safe, we do a single program to avoid race conditions and complexity.
    # We'll launch with grid=(1,) and process all elements.
    idx = 0
    while idx < N:
        val = tl.load(flat_ptr + idx)
        tl.atomic_add(counts_ptr + val, 1)
        idx += 1


@triton.jit
def inclusive_scan_inplace(counts_ptr, E: tl.int32, LOG: tl.int32):
    """
    In-kernel inclusive scan (prefix sum) on a vector of length E using Hillis–Steele algorithm.
    Assumes counts_ptr is of length E+1 at launch (final offset N will be written via host).
    """
    lane = tl.arange(0, E)
    for i in range(LOG):
        # Shifted value: for lanes >= 2^(i+1), add previous value at lane - 2^(i+1)
        shifted = tl.where(lane >= (1 << (i + 1)), counts_ptr[lane - (1 << (i + 1))], 0)
        counts_ptr[lane] += shifted


def _next_power_of_two(n: int) -> int:
    # Returns the next power of two >= n
    if n <= 1:
        return 1
    return 1 << ((n - 1).bit_length())

    # The scan produces inclusive prefix sums up to num_experts. To include N in offsets[-1], we set it here:
    # But since Triton wrote only first num_experts lanes, we need to compute N separately and set offsets[-1] via host.
    # However, we can recompute by summing counts or set offsets[-1] = torch.sum(counts). Let's do that.
