import torch
import triton
import triton.language as tl


@triton.jit
def stable_bitonic_argsort_inplace(vals_ptr, idx_ptr, N: tl.int32, BLOCK: tl.int32, LOG: tl.int32):
    # Stable bitonic argsort over first N elements of vals_ptr, writes sorted indices into idx_ptr.
    # We pad to BLOCK (next power-of-two >= N) and use sentinel MAX_INT for padded lanes so they sort to the end.
    MAX_INT = (1 << 31) - 1

    # Initialize idx_ptr[0..BLOCK-1] with positions 0..BLOCK-1
    for i in range(BLOCK):
        tl.store(idx_ptr + i, i)

    # We will implement bitonic sort using nested loops and pairwise compare-and-swap on idx_ptr,
    # loading values from vals_ptr via the indices.
    # The outer loop k iterates over sequence sizes: 2, 4, 8, ..., BLOCK.
    # The inner loop j iterates over subsequence sizes: k/2, k/4, ..., 1.
    # For each pair (p, q), where q = p ^ step, we determine direction based on (p & k) == 0.
    # Stable tie-break: for equal values, swap if idx_p > idx_q (original index determines order).

    # Unrolled outer loops using LOG and BLOCK
    # Note: Triton allows Python loops with compile-time constants; here LOG and BLOCK are tl.constexpr from launch.
    for k in range(2, BLOCK + 1, 2):
        step = k // 2
        # We can unroll with compile-time loops using LOG but more simply do:
        # For each step, process all pairs once.
        # Triton does not support arbitrary nested loops cleanly here; implement with dynamic while-like behavior using masks.
        # A simpler approach is to write the full bitonic network using tl.arange and masks. Triton supports elementwise operations.
        # We'll set up indices and perform compare-and-swap on idx_ptr.

        # Create index vector lanes
        i = tl.arange(0, BLOCK)
        # For each step, we process pairs (i, i ^ step)
        # Determine direction per position: if (i & k) == 0, ascending; else descending.
        # We need a vectorized compare-and-swap. Triton can do pairwise updates using vector masks.

        # We will implement the network with vectorized pairwise updates:
        # For each step, compute partner = i ^ step; for ascending direction, swap if (vals[i] > vals[partner]) or (vals[i] == vals[partner] and idx[i] > idx[partner]).
        # For descending direction, swap if (vals[i] < vals[partner]) or (vals[i] == vals[partner] and idx[i] > idx[partner]).

        # Note: Triton does not allow direct dynamic indexing with another vector; we need to use memory addresses.
        # We can load/store using idx_ptr values as addresses:
        # idx_i = tl.load(idx_ptr + i), idx_partner = tl.load(idx_ptr + partner)
        # val_i = tl.load(vals_ptr + idx_i), val_partner = tl.load(vals_ptr + idx_partner)

        # Determine partner and direction
        partner = i ^ step
        # Direction mask: ascending when (i & k) == 0, descending otherwise
        dir_asc = (i & k) == 0

        # Load values and indices for current and partner positions
        idx_i = tl.load(idx_ptr + i)
        idx_partner = tl.load(idx_ptr + partner)
        val_i = tl.load(vals_ptr + idx_i)
        val_partner = tl.load(vals_ptr + idx_partner)

        # Stable compare-and-swap:
        # Ascending: swap if val_i > val_partner OR (val_i == val_partner AND idx_i > idx_partner)
        # Descending: swap if val_i < val_partner OR (val_i == val_partner AND idx_i > idx_partner)
        # Since we can't do both in one, we use dir_asc to select the condition.
        asc_cond = (val_i > val_partner) | ((val_i == val_partner) & (idx_i > idx_partner))
        desc_cond = (val_i < val_partner) | ((val_i == val_partner) & (idx_i > idx_partner))
        cond = tl.where(dir_asc, asc_cond, desc_cond)

        # If cond, swap idx_ptr[i] and idx_ptr[partner]
        # We'll perform the swap by writing tmp = idx_ptr[i], idx_ptr[i] = idx_ptr[partner], idx_ptr[partner] = tmp
        # Triton supports elementwise masked loads/stores. We can store new values back.

        # Compute new values for i and partner positions
        new_i = tl.where(cond, idx_partner, idx_i)
        new_partner = tl.where(cond, idx_i, idx_partner)

        # We need to store them back; Triton allows masked store. We can store new_i to idx_ptr[i] and new_partner to idx_ptr[partner]
        # But we must avoid double writes or race conditions. Triton executes elementwise; it is safe.
        tl.store(idx_ptr + i, new_i)
        tl.store(idx_ptr + partner, new_partner)

    # After sorting, idx_ptr[0..N-1] holds the argsort indices for the original N elements.
    sorted_indices = idx_ptr[:N]
    return sorted_indices


@triton.jit
def histogram_kernel(flat_ptr, counts_ptr, N: tl.int32, num_experts: tl.int32):
    # Count occurrences per expert index into counts_ptr[0..num_experts-1].
    # We process the flat array and atomic_add 1 for each element.
    # Launch over a single program instance; process in chunks.
    # Note: Triton requires compile-time loops. We use a while loop with a static chunk size.
    CHUNK = 1024
    start = 0
    while start < N:
        offs = start + tl.arange(0, CHUNK)
        mask = offs < N
        vals = tl.load(flat_ptr + offs, mask=mask, other=0)
        # Atomic add 1 for each valid value
        for j in range(CHUNK):
            val = vals[j]
            is_valid = mask[j]
            if is_valid:
                tl.atomic_add(counts_ptr + val, 1)
        start += CHUNK


def next_power_of_two(x: int) -> int:
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()
