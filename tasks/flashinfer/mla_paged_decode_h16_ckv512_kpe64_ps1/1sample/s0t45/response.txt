                lse[b_idx, h_idx] = lse_val.item()

                # Now compute output in torch: y = sum_m exp(logit_scaled - lse_val) * Kc[m, :]
                y = torch.zeros((N,), dtype=torch.float32, device=device)
                for m_i in range(M_total):
                    tok = int(tok_idx[m_i].item())
                    Kc_row = Kc_fp32[tok, :]
                    qn_h = qn_fp32[b_idx, h_idx, :]
                    qp_h = qp_fp32[b_idx, h_idx, :]
                    dot_qn = torch.dot(qn_h, Kc_row)
                    dot_qp = torch.dot(qp_h, Kp_fp32[tok, :])
                    logit = dot_qn + dot_qp
                    logit_scaled = logit * float(sm_scale)
                    attn_m = torch.exp(logit_scaled - lse[b_idx, h_idx]) / float(M_total)
                    y += attn_m * Kc_fp32[tok, :]

                output_fp32[b_idx, h_idx, :] = y

        # Cast output to bfloat16 to match original function's output dtype
        output_bf16 = output_fp32.to(torch.bfloat16)
        return output_bf16, lse
