    Dc: tl.constexpr,  # head_dim_ckv (e.g., 512)
    Dp: tl.constexpr,  # head_dim_kpe (e.g., 64)
    M_b: tl.constexpr, # number of tokens in this batch
    sm_scale: tl.constexpr,  # scaling factor (float)
    BLOCK_N: tl.constexpr     # token tile size (e.g., 128)
):

    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)

    # Load q vectors for this head
    qn_vec = tl.load(qn_ptr + pid_h * Dc + tl.arange(0, Dc))  # [Dc]
    qp_vec = tl.load(qp_ptr + pid_h * Dp + tl.arange(0, Dp))  # [Dp]

    # Accumulate logits for this (b,h)
    logits = tl.zeros([BLOCK_N], dtype=tl.float32)
    # Compute max for stability
    m = tl.full([1], -float("inf"), dtype=tl.float32)

    for n_start in range(0, M_b, BLOCK_N):
        offs = n_start + tl.arange(0, BLOCK_N)
        mask = offs < M_b
        tok_idx = tl.load(tok_idx_ptr + offs, mask=mask, other=0)  # int32 token indices

        # Gather Kc_sub and Kp_sub for these tokens
        Kc_sub = tl.load(Kc_ptr + tok_idx * Dc + tl.arange(0, Dc), mask=mask, other=0.0)  # [BLOCK_N, Dc]
