Algorithm recap
- Inputs: topk_idx with shape (batch_size, seq_len, num_experts_per_tok), dtype int32, on CUDA.
- Flatten to 1D x of length N = batch_size * seq_len * num_experts_per_tok.
- All computation is done in Triton:
  1) Histogram: counts[e] = number of elements equal to e in x (use atomic_add).
  2) Prefix sum: offsets[1..E] = inclusive prefix sums of counts (offsets[0] = 0).
  3) Stable counting sort: produce sorted_token_indices permutation, so that output[i] = original position pos where topk_idx[pos] = e_sorted[i]. We maintain starts[e] = offsets[e] and write out[starts[id]] = pos, then starts[id] += 1. This is stable because we iterate positions in increasing order.

Plan for correctness
- Keep the Triton kernels simple:
  - histogram_experts: one program processes BLOCK elements with masked loads and atomic_add to counts.
  - inclusive_scan_counts: one program computes offsets via a sequential loop over E.
  - stable_counting_sort: one program processes positions sequentially (pos = 0..N-1), ensuring stability.

Final code
```python
import torch
import triton
import triton.language as tl


@triton.jit
def histogram_experts(x_ptr, counts_ptr, N, E, BLOCK: tl.constexpr):
    """
    Compute per-expert histogram of x_ptr[0:N] into counts_ptr[0:E].
    Each program processes BLOCK elements; masked loads ensure no OOB.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < N
    vals = tl.load(x_ptr + offs, mask=mask, other=0)
    # Atomically increment counts for valid elements; Triton supports int32 atomics
    for i in range(BLOCK):
        if mask[i]:
            idx = vals[i].to(tl.int32)
            # idx is in [0, E), already ensured by caller
            tl.atomic_add(counts_ptr + idx, 1)


@triton.jit
def inclusive_scan_counts(counts_ptr, offsets_ptr, E, BLOCK: tl.constexpr):
    """
    Compute inclusive prefix sums of counts_ptr[0:E] into offsets_ptr[0:E+1].
    offsets_ptr[0] is set by host to 0; we write offsets[k+1] = offsets[k] + counts[k].
    One program with sequential loop over E.
    """
    # Single program; iterate and update
    # offsets_ptr[0] already initialized on host
    total = 0
    for k in range(E):
        total += tl.load(counts_ptr + k)
        tl.store(offsets_ptr + k + 1, total)


@triton.jit
def stable_counting_sort(x_ptr, starts_ptr, out_ptr, N, E, BLOCK: tl.constexpr):
    """
    Stable counting sort. For each position pos in [0, N):
      id = x[pos]
      out[starts[id]] = pos
      starts[id] += 1
    out is the permutation of original indices. starts is exclusive prefix sums per expert.
    One program processes positions sequentially.
    """
    for pos in range(0, N, BLOCK):
        for i in range(BLOCK):
            idx = pos + i
            if idx < N:
                id = tl.load(x_ptr + idx).to(tl.int32)
                tgt = tl.load(starts_ptr + id)
                tl.store(out_ptr + tgt, idx)
                tl.store(starts_ptr + id, tgt + 1)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.num_experts = 256  # consistent with original code

    def forward(self, topk_idx: torch.Tensor):
        """
        topk_idx: (batch_size, seq_len, num_experts_per_tok), int32, on CUDA
        Returns:
          sorted_token_indices: int32 of shape (N,)
          expert_offsets: int32 of shape (E+1,)
        """
        # Use input exactly as provided; do not generate or alter with torch ops.
        assert topk_idx.dtype == torch.int32, "topk_idx must be int32"
        assert topk_idx.is_cuda, "topk_idx must be on CUDA device for Triton kernels"

        # Flatten to 1D for processing
        x = topk_idx.reshape(-1)
        N = x.numel()
        E = self.num_experts

        # 1) Histogram of expert IDs using Triton
        counts = torch.zeros(E, dtype=torch.int32, device=x.device)
        BLOCK_HIST = 1024
        grid_hist = (triton.cdiv(N, BLOCK_HIST),)
        histogram_experts[grid_hist](x, counts, N, E, BLOCK=BLOCK_HIST)

        # 2) Inclusive prefix sum to produce expert_offsets using Triton
        offsets = torch.empty(E + 1, dtype=torch.int32, device=x.device)
        offsets[0] = 0
        grid_scan = (1,)
        inclusive_scan_counts[grid_scan](counts, offsets, E, BLOCK=1)

        # 3) Stable counting sort using Triton to produce sorted_token_indices (permutation)
        out = torch.empty(N, dtype=torch.int32, device=x.device)
        starts = offsets.clone()  # exclusive prefix sums per expert
        grid_sort = (1,)
        stable_counting_sort[grid_sort](x, starts, out, N, E, BLOCK=1024)

        # Return sorted_token_indices (int32 permutation) and expert_offsets (int32, length E+1)
        return out, offsets
```
