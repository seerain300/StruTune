import triton
import triton.language as tl

def count_expert_ids_atomic_kernel(flat_ptr, counts_ptr, N):

    i = 0
    while i < N:
        val = tl.load(flat_ptr + i)
        tl.atomic_add(counts_ptr + val, 1)
        i += 1
