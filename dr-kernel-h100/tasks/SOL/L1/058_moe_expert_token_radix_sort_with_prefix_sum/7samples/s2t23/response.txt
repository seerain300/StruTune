import torch
import triton


# Triton kernel: histogram of expert IDs using atomic_add
@triton.jit
def histogram_experts(x_ptr, counts_ptr, N, E, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < N
    # Load values; for masked lanes, provide 0 (won't add due to mask)
    vals = tl.load(x_ptr + offs, mask=mask, other=0)
    # For each lane, increment counts[vals[i]] if in bounds
    # We rely on atomic_add to safely accumulate counts across programs
    # Note: vals are int32 as per input
    for i in range(BLOCK):
        idx = offs[i]
        if mask[i]:
            tl.atomic_add(counts_ptr + vals[i], 1)


# Triton kernel: inclusive prefix sum of counts to produce offsets
@triton.jit
def inclusive_scan_counts(counts_ptr, offsets_ptr, E):
    # Simple sequential inclusive scan per element
    # offsets[0] is set by host (0); we fill offsets[1..E]
    for i in range(1, E + 1):
        # offsets[i] = offsets[i-1] + counts[i-1]
        prev = offsets_ptr[i - 1]
        c = tl.load(counts_ptr + i - 1)
        tl.store(offsets_ptr + i, prev + c)


# Triton kernel: stable counting sort to produce permutation (sorted_token_indices)
@triton.jit
def stable_counting_sort(x_ptr, starts_ptr, out_ptr, N, E, BLOCK: tl.constexpr):
    # Process positions sequentially; for each pos, write its index into out based on x[pos]
    for pos in range(0, N, BLOCK):
        pos_i = pos + tl.arange(0, BLOCK)
        mask = pos_i < N
        ids = tl.load(x_ptr + pos_i, mask=mask, other=0)
        # For each lane
        for j in range(BLOCK):
            p = pos + j
            if mask[j]:
                e = ids[j]
                # out[starts[e]] = p; then starts[e] += 1
                start = tl.load(starts_ptr + e)
                tl.store(out_ptr + start, p)
                tl.atomic_add(starts_ptr + e, 1)

Explanation of safeguards
- All kernels use simple control flow that Triton supports reliably: for loops with integer bounds. No dynamic loops based on runtime N inside kernel parameters, which can cause JIT issues.
- histogram_experts uses masked loads and atomic_add to safely accumulate counts per id across programs.
- inclusive_scan_counts uses a straightforward per-element inclusive scan with tl.load/tl.store. offsets[0] is set by host to 0.
- stable_counting_sort iterates positions in BLOCK-sized chunks and writes out per position with atomic_add on starts. This produces a stable permutation because we process positions in ascending order; tokens with equal ids preserve their original order.
