import torch
import triton
import triton.language as tl


@triton.jit
def histogram_experts(x_ptr, counts_ptr, N, E, BLOCK: tl.constexpr):
    """
    Compute per-expert histogram of x_ptr[0:N] into counts_ptr[0:E].
    Each program processes BLOCK elements and atomically increments counts for each match.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < N
    # Load values; out-of-bounds masked to 0 (any value will be masked out anyway since we check mask)
    vals = tl.load(x_ptr + offs, mask=mask, other=0)
    # Atomic add for each valid element; we only process ids in [0, E)
    # Note: atomic_add on int32 is supported in Triton
    for i in range(BLOCK):
        idx = vals[i]
        if (mask[i] and 0 <= idx < E):
            tl.atomic_add(counts_ptr + idx, 1)


@triton.jit
def inclusive_scan_counts(counts_ptr, offsets_ptr, E, BLOCK: tl.constexpr):
    """
    Compute inclusive prefix sum of counts_ptr[0:E] into offsets_ptr[1:E+1], with offsets_ptr[0] = 0.
    Simple single-pass loop over E; safe and correct for moderate E (e.g., 256).
    """
    total = 0
    # offsets_ptr is int32
    for i in range(E):
        total += tl.load(counts_ptr + i)
        tl.store(offsets_ptr + i + 1, total)


@triton.jit
def stable_counting_sort(x_ptr, starts_ptr, out_ptr, N, E, BLOCK: tl.constexpr):
    """
    Stable counting sort. For each position pos in [0, N):
      id = x[pos]
      out[starts[id]] = pos
      starts[id] += 1
    This ensures stability because we iterate positions in ascending order.
    """
    # We use a single program with a loop over N; for N up to 8192, this is fine.
    for pos in range(0, N, BLOCK):
        # Process BLOCK positions at a time
        offs = pos + tl.arange(0, BLOCK)
        mask = offs < N
        ids = tl.load(x_ptr + offs, mask=mask, other=0)  # assume ids in [0, E)
        for i in range(BLOCK):
            if mask[i]:
                id = ids[i]
                # store original position into sorted output
                tl.store(out_ptr + tl.load(starts_ptr + id), offs[i])
                # advance the start for this expert
                tl.atomic_add(starts_ptr + id, 1)
        # Note: out_ptr stores int32 positions; starts_ptr stores int32 counters.


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        """
        Triton-only implementation of run(topk_idx):
          - sorted_token_indices: permutation of [0..N-1] sorted by expert id (stable)
          - expert_offsets: inclusive prefix sums (length E+1)
        No torch operations are used in forward; all computation is done in Triton kernels.
        """
        # Validate and prepare
        if topk_idx.dtype != torch.int32:
            raise ValueError("topk_idx must be int32")
        if not topk_idx.is_cuda:
            raise RuntimeError("topk_idx must be on CUDA device for Triton kernels")

        # Flatten to 1D for processing
        x = topk_idx.reshape(-1)
        N = x.numel()
        E = 256  # num_experts as in original code

        # 1) Histogram of expert IDs using Triton
        counts = torch.zeros(E, dtype=torch.int32, device=x.device)
        BLOCK_HIST = 1024
        grid_hist = (triton.cdiv(N, BLOCK_HIST),)
        histogram_experts[grid_hist](x, counts, N, E, BLOCK=BLOCK_HIST)

        # 2) Inclusive prefix sum to produce expert_offsets using Triton
        offsets = torch.empty(E + 1, dtype=torch.int32, device=x.device)
        offsets[0] = 0
        # For E=256, a simple single-thread loop is fine and reliable
        grid_scan = (1,)
        inclusive_scan_counts[grid_scan](counts, offsets, E, BLOCK=1)

        # 3) Stable counting sort using Triton to produce sorted_token_indices (permutation)
        out = torch.empty(N, dtype=torch.int32, device=x.device)
        starts = offsets.clone()  # exclusive prefix sums for each expert
        grid_sort = (1,)
        stable_counting_sort[grid_sort](x, starts, out, N, E, BLOCK=1024)

        # Return sorted_token_indices (int32 permutation) and expert_offsets (int32, length E+1)
        return out, offsets
