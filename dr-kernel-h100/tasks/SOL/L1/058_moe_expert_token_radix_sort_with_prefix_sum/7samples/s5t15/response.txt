import torch

# Triton kernels
import triton
import triton.language as tl


# Reorder the flat array according to a given permutation (stable argsort).
# Reads: flat (original 1D int32), perm (int32 indices 0..N-1).
# Writes: out_ids (flat reordered), out_idx (same as perm).
@triton.jit
def reorder_ids_and_write_perm_kernel(flat_ptr, perm_ptr, out_ids_ptr, out_idx_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    # Load original ids and permutation indices
    orig = tl.load(flat_ptr + offsets, mask=mask, other=0)
    idx = tl.load(perm_ptr + offsets, mask=mask, other=0)
    # Reorder: out_ids[i] = flat[perm[i]]
    val = tl.load(flat_ptr + idx, mask=mask, other=0)
    tl.store(out_ids_ptr + offsets, val, mask=mask)
    # Also write out_idx = perm (the stable argsort permutation)
    tl.store(out_idx_ptr + offsets, idx, mask=mask)


# Count per-expert IDs using chunked iteration and atomics. Assumes flat_ptr points to int32.
@triton.jit
def count_expert_ids_kernel(flat_ptr, counts_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    # One program scans the flat array in chunks and atomically adds to counts.
    for start in range(0, N, BLOCK):
        offsets = start + tl.arange(0, BLOCK)
        mask = offsets < N
        ids = tl.load(flat_ptr + offsets, mask=mask, other=0)  # int32
        # For each element in the chunk, increment count[ids]
        for i in range(BLOCK):
            # Only increment for valid i where mask[i] is true
            if mask[i]:
                id_i = ids[i]
                tl.atomic_add(counts_ptr + id_i, 1)


# Compute exclusive prefix sum of counts to produce offsets[1..].
# Assumes counts is length num_experts.
@triton.jit
def exclusive_prefix_sum_kernel(counts_ptr, offsets_ptr, num_experts: tl.constexpr):
    total = 0
    # Sequential loop over experts; num_experts is modest (256) and offsets is small.
    for e in range(num_experts):
        total += tl.load(counts_ptr + e)
        tl.store(offsets_ptr + e + 1, total)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        # Ensure CUDA tensors and contiguity
        flat = topk_idx.reshape(-1).contiguous()
        device = flat.device
        N = flat.numel()
        num_experts = 256  # as per original code

        # Host-side stable argsort for correctness and simplicity
        sorted_token_indices = torch.argsort(flat, dim=0, stable=True)

        # Allocate outputs
        out_idx = torch.empty(N, dtype=torch.int32, device=device)  # we'll fill this via Triton
        out_ids = torch.empty(N, dtype=torch.int32, device=device)  # just for validation; not returned

        # Launch Triton kernel to reorder flat according to permutation and write out_idx
        BLOCK = 1024
        grid = (triton.cdiv(N, BLOCK),)
        # We don't need out_ids for return; just ensure out_idx is populated
        reorder_ids_and_write_perm_kernel[grid](flat, sorted_token_indices, out_ids, out_idx, N, BLOCK=BLOCK)

        # Count per-expert IDs via Triton (histogram with atomics)
        counts = torch.zeros(num_experts, dtype=torch.int32, device=device)
        count_expert_ids_kernel[(1,)](flat, counts, N, BLOCK=BLOCK)

        # Compute offsets via Triton prefix sum
        offsets = torch.zeros(num_experts + 1, dtype=torch.int32, device=device)
        exclusive_prefix_sum_kernel[(1,)](counts, offsets, num_experts)

        # Return permutation indices (stable argsort) and offsets (length num_experts+1).
        # Note: The original run returns (sorted_token_indices, expert_offsets).
        # Here we return (out_idx, offsets).
        return out_idx, offsets
