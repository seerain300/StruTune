# Triton stable argsort: given values of length N, returns indices (0..N-1) sorted ascending.
# We emulate stable=True by tie-breaking on original index.
@triton.jit
def stable_argsort(vals_ptr, out_idx_ptr, N: tl.int32, BLOCK: tl.constexpr, LOG: tl.constexpr):
    # One program handles the entire array of lanes [0..BLOCK-1]
    pid = tl.program_id(axis=0)
    if pid != 0:
        return  # only one program executes

    idx = tl.arange(0, BLOCK)  # lane indices
    # Prepare values: copy first N, pad with large sentinel so they sort to the end
    MAX_INT = (1 << 31) - 1
    vals = tl.load(vals_ptr + idx, mask=idx < N, other=MAX_INT)

    # Original indices for tie-breaking
    orig = idx

    # Bitonic sort network with stable tie-breaking
    size = 2
    while size <= BLOCK:
        stride = size // 2
        while stride > 0:
            partner = idx ^ stride
            a = vals
            b = tl.load(vals_ptr + partner)
            oa = orig
            ob = partner  # use partner as original index for 'b'

            # Stable compare: swap if (a > b) OR (a == b AND oa > ob)
            cmp_gt = a > b
            cmp_eq = a == b
            cmp_stable = (cmp_gt | (cmp_eq & (oa > ob)))
            swap = ~cmp_stable

            # Apply swap
            new_a = tl.where(swap, b, a)
            new_b = tl.where(swap, a, b)
            # Write back sorted pairs for current lane and its partner
            tl.store(vals_ptr + idx, new_a, mask=True)
            tl.store(vals_ptr + partner, new_b, mask=True)

            # We also need to permute 'orig' accordingly. Maintain 'orig' as the original position.
            # However, Triton doesn't support multi-output, so we recompute 'orig' after each pass by reordering.
            # Instead, we keep 'orig' as idx throughout since tie-breaking uses (value, idx) inside the network.

    Triton stable argsort (ascending). Writes sorted indices into out_idx_ptr[0..N-1].
    Pads to BLOCK (next power of two >= N), uses sentinel for padded lanes, and tie-breaks by original index.
    """

    if pid != 0:
        return

    idx = tl.arange(0, BLOCK)
    MAX_INT = (1 << 31) - 1
    vals = tl.load(vals_ptr + idx, mask=idx < N, other=MAX_INT)

    # We need to produce out_idx: for each original position i, output its sorted index.
    # Implement by running bitonic network on pairs (value, position) and tracking positions.
    # Triton does not support returning multiple arrays; we instead maintain 'out_idx' by assigning
    # positions to corresponding values at the end of sorting.

    # Initialize out_idx as arange
    out_idx = idx

    # Bitonic sort network: sort pairs (vals, out_idx) ascending by vals (stable tie-break by out_idx)
    size = 2
    while size <= BLOCK:
        stride = size // 2
        while stride > 0:
            partner = idx ^ stride
            a = vals
            b = tl.load(vals_ptr + partner)
            oa = out_idx
            ob = tl.load(out_idx_ptr + partner, mask=True)

            # but we can't read partner beyond N; so we only operate on valid lanes.
            # We keep 'other' as MAX_INT for vals and avoid partner ops for lanes >= N.

            cmp_eq = a == b
            cmp_stable = cmp_gt | (cmp_eq & (oa > ob))
            swap = ~cmp_stable

            new_a = tl.where(swap, b, a)
            new_b = tl.where(swap, a, b)

            # Apply updates only for lanes < N
            tl.store(vals_ptr + idx, new_a, mask=idx < N)
            tl.store(out_idx_ptr + idx, tl.where(swap, partner, oa), mask=idx < N)

            stride //= 2
        size *= 2
