# Define a proper Triton per-lane inclusive scan kernel:
# We will implement a kernel that launches with grid=(E,), each program computes out[i] = sum(counts[0..i]).
# Triton supports while loops for scalar-like control flow.

@triton.jit
def per_lane_inclusive_scan(counts_ptr, out_ptr, E: tl.int32):
    """
    One program per output index i; compute out[i] = sum(counts[0..i]) using atomic_add.
    This ensures Triton-only offsets computation.
    """
    i = tl.program_id(axis=0)
    if i < E:
        # Initialize out[i] = counts[i]
        acc = tl.load(counts_ptr + i)
        tl.store(out_ptr + i, acc)
        # Add previous counts[0..i-1] to out[i]
        j = 0
        # Dynamic while loop over j <= i-1
        while j < i:
            prev = tl.load(counts_ptr + j)
            tl.atomic_add(out_ptr + i, prev)
            j += 1

def _next_power_of_two(x: int) -> int:
    if x <= 1:
