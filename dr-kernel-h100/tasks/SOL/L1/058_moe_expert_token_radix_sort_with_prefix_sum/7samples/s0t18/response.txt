```python
import torch

# Try to import Triton; we will use it if available
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

# Triton kernel: count occurrences of each expert id in [0, 255] from the flattened vector
if TRITON_AVAILABLE:
    @triton.jit
    def bincount_kernel(x_ptr, counts_ptr, N, BLOCK: tl.constexpr):
        pid = tl.program_id(axis=0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N
        # Load flattened values (int32)
        vals = tl.load(x_ptr + offs, mask=mask, other=0)
        # For each possible expert id 0..255, atomically add 1 for each occurrence
        # This mimics torch.bincount for valid ids in [0, 255]
        for i in range(256):
            # Build a mask for values == i
            eq_mask = (vals == i) & mask
            # Sum number of true in this mask: cast to int32 and reduce
            sum_i = tl.sum(eq_mask.to(tl.int32), axis=0)
            # Atomically add to counts[i]
            tl.atomic_add(counts_ptr + i, sum_i)

    # Triton kernel: inclusive prefix sum over a 256-length counts vector, writing int64 offsets
    @triton.jit
    def inclusive_prefix_sum_kernel(counts_ptr, y_ptr, L: tl.constexpr):
        # L is 257: we write offsets[1..257] = inclusive scan of counts[0..255]
        # We iterate manually up to L-1; y_ptr[0] should remain zero (prefix before index 0)
        cur_sum = tl.zeros((), dtype=tl.int64)
        # Manually unrolled loop up to index 256; L is constexpr so Triton can optimize
        # Note: Triton requires static loop bounds; using a fixed bound 257 here.
        # This covers all 256 bins plus the final element.
        for j in range(257):
            val_j = tl.load(counts_ptr + j).to(tl.int64)
            cur_sum = cur_sum + val_j
            tl.store(y_ptr + j, cur_sum)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        """
        Compute:
          - sorted_token_indices: stable argsort permutation of flattened indices (int32)
          - expert_offsets: inclusive counts per expert id [0..255], shape (257,), dtype int64
        Using Triton for bincount and prefix sum, torch for argsort for correctness.
        """
        # Ensure input is on CUDA and contiguous for Triton
        device = topk_idx.device
        if not topk_idx.is_cuda:
            # If not CUDA, move to current default device
            device = torch.device("cuda")
            topk_idx = topk_idx.to(device)
        flat = topk_idx.reshape(-1).contiguous()
        N = flat.numel()

        # 1) Triton bincount of flattened values into counts[0..255] (int32)
        counts = torch.zeros(256, dtype=torch.int32, device=device)
        BLOCK = 1024
        grid = (triton.cdiv(N, BLOCK),)
        if TRITON_AVAILABLE:
            bincount_kernel[grid](flat, counts, N, BLOCK=BLOCK)
        else:
            # Fallback: torch.bincount for correctness if Triton not available
            counts = torch.bincount(flat, minlength=256)

        # 2) Inclusive prefix sum of counts in Triton, produce int64 offsets of length 257
        expert_offsets = torch.empty(257, dtype=torch.int64, device=device)
        if TRITON_AVAILABLE:
            inclusive_prefix_sum_kernel[(1,)](counts, expert_offsets, L=257)
        else:
            # Fallback: torch.cumsum for correctness
            expert_offsets = counts.cumsum(0).to(torch.int64)

        # 3) sorted_token_indices: stable argsort of flattened indices, return int32 permutation
        sorted_token_indices = flat.argsort(stable=True).to(torch.int32)

        return sorted_token_indices, expert_offsets
```
