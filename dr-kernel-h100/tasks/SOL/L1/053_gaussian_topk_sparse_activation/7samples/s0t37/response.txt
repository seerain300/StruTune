  - The gating kernel expects x, mean, std, and out to have consistent dtypes (fp32 for computation). If any of these are inadvertently cast to bfloat16 or fp16 at the time of the kernel call, arithmetic can go awry.
  - Our host code used x_f32 = x_contig.to(torch.float32) and created out_f32 as empty_like(x_f32), which is correct. However, in one earlier version we used out_f32 = torch.empty_like(x), which could mismatch dtypes and strides. We must ensure out_f32 is created as empty_like(x_f32).

```python
import torch
import triton
import triton.language as tl


@triton.jit
def compute_row_stats_kernel(x_ptr, mean_ptr, std_ptr, total_rows, K, BLOCK_SIZE: tl.constexpr):
    """
    Compute per-row (over last dim K) mean and std for each row (flattened [B*S]).
    x_ptr: flattened input pointer, stride per row is K.
    mean_ptr, std_ptr: shape [total_rows], fp32.
    """
    row_id = tl.program_id(0)
    offs = 0
    total = 0.0
    total2 = 0.0

    for start in range(0, K, BLOCK_SIZE):
        idx = start + tl.arange(0, BLOCK_SIZE)
        mask = idx < K
        # Row linear index is row_id * K + idx
        ptr = x_ptr + row_id * K + idx
        x = tl.load(ptr, mask=mask, other=0.0)
        x = x.to(tl.float32)
        total += tl.sum(x, axis=0)
        total2 += tl.sum(x * x, axis=0)

    mean = total / K
    # population std (unbiased=False), matching torch.std(..., unbiased=False)
    var = total2 / K - mean * mean
    std = tl.sqrt(var)
    tl.store(mean_ptr + row_id, mean)
    tl.store(std_ptr + row_id, std)


@triton.jit
def compute_ndtri_kernel(z_buf_ptr, target_sparsity, a1, a2, a3, a4, a5, a6,
                         b1, b2, b3, b4, b5, c1, c2, c3, c4, c5, c6, d1, d2, d3, d4,
                         p_low, p_high, BLOCK_SIZE: tl.constexpr):
    """
    Compute inverse standard normal CDF (0 < p < 1) using A&S 5.2.23 approximation.
    Writes result to z_buf_ptr[0] as fp32.
    """
    p = target_sparsity  # scalar passed from host
    # lower region
    q = tl.sqrt(-2.0 * tl.log(p))
    z_low = (((((c1 * q + c2) * q + c3) * q + c4) * q + c5) * q + c6) / \
            ((((d1 * q + d2) * q + d3) * q + d4) * q + 1.0)
    # central region
    q2 = p - 0.5
    r = q2 * q2
    poly = (((((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6)
    denom = (((((b1 * r + b2) * r + b3) * r + b4) * r + b5) * r + 1.0)
    z_mid = poly * q2 / denom
    # upper region
    q3 = tl.sqrt(-2.0 * tl.log(1.0 - p))
    z_up = -(((((c1 * q3 + c2) * q3 + c3) * q3 + c4) * q3 + c5) * q3 + c6) / \
           ((((d1 * q3 + d2) * q3 + d3) * q3 + d4) * q3 + 1.0)
    # choose region by threshold
    mask_low = p < p_low
    mask_up = p > p_high
    z = tl.where(mask_low, z_low, z_mid)
    z = tl.where(mask_up, z_up, z)
    tl.store(z_buf_ptr, z)


@triton.jit
def apply_gating_2d_kernel(x_ptr, mean_ptr, std_ptr, z_ptr, out_ptr,
                           total_rows, K, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise gating: out[r, k] = relu(x[r, k] - (mean[r] + std[r] * z))
    Flattened layout: row r corresponds to index r * K + k for k in [0, K).
    """
    row = tl.program_id(0)
    col_block = tl.program_id(1)
    offs = col_block * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < K
    # load scalar threshold for this row
    mean = tl.load(mean_ptr + row)
    std = tl.load(std_ptr + row)
    z = tl.load(z_ptr)  # scalar
    threshold = mean + std * z
    # load x and compute out
    x = tl.load(x_ptr + row * K + offs, mask=mask, other=0.0)
    x = x.to(tl.float32)
    out = x - threshold
    out = tl.maximum(out, 0.0)  # relu
    tl.store(out_ptr + row * K + offs, out, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # constants for A&S 5.2.23 approximation
        self.a1 = -3.969683028665376e+01
        self.a2 = 2.209460984245205e+02
        self.a3 = -2.759285104469687e+02
        self.a4 = 1.383577518672690e+02
        self.a5 = -3.066479806614716e+01
        self.a6 = 2.506628277459239e+00

        self.b1 = -5.447609879822406e+01
        self.b2 = 1.615858368580409e+02
        self.b3 = -1.556989798598866e+02
        self.b4 = 6.680131188771972e+01
        self.b5 = -1.328068155288572e+01

        self.c1 = -7.784894002430293e-03
        self.c2 = -3.223964580411365e-01
        self.c3 = -2.400758277161838e+00
        self.c4 = -2.549732539343734e+00
        self.c5 = 4.374664141464968e+00
        self.c6 = 2.938163982698783e+00

        self.d1 = 7.784695709041462e-03
        self.d2 = 3.224671290700398e-01
        self.d3 = 2.445134137142996e+00
        self.d4 = 3.754408661907416e+00

        self.p_low = 0.02425

    def forward(self, x: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        """
        Triton-only implementation:
        - Compute per-row mean and std across last dim (K).
        - Compute z = _ndtri(target_sparsity) in Triton.
        - Apply gating: out = relu(x - (mean + std * z)), cast to bfloat16.
        """
        assert x.is_cuda, "Input must be on CUDA for Triton kernels."

        # Flatten rows: total_rows = B * S
        B, S, K = x.shape
        total_rows = B * S

        # Ensure contiguous input along last dim
        x_contig = x.contiguous()

        # 1) Compute per-row mean and std in fp32
        mean = torch.empty(total_rows, dtype=torch.float32, device=x.device)
        std = torch.empty(total_rows, dtype=torch.float32, device=x.device)
        BLOCK_SIZE_STATS = 2048
        grid_stats = (total_rows,)
        compute_row_stats_kernel[grid_stats](
            x_contig.view(-1), mean, std, total_rows, K,
            BLOCK_SIZE=BLOCK_SIZE_STATS,
            num_warps=4, num_stages=2
        )

        # 2) Compute z = inverse standard normal CDF in Triton (scalar)
        z_buf = torch.empty(1, dtype=torch.float32, device=x.device)
        p_high = 1.0 - self.p_low
        compute_ndtri_kernel[(1,)](
            z_buf, float(target_sparsity),
            self.a1, self.a2, self.a3, self.a4, self.a5, self.a6,
            self.b1, self.b2, self.b3, self.b4, self.b5,
            self.c1, self.c2, self.c3, self.c4, self.c5, self.c6,
            self.d1, self.d2, self.d3, self.d4,
            self.p_low, p_high,
            BLOCK_SIZE=1024,
            num_warps=1, num_stages=1
        )
        z = float(z_buf.item())

        # 3) Apply gating with 2D Triton kernel
        x_f32 = x_contig.to(torch.float32)
        out_f32 = torch.empty_like(x_f32)  # ensure dtype and shape match x_f32

        # Dynamic tuning for gating kernel (proven robust and fast)
        if K >= 8192:
            block_size_gate = 4096
            num_warps_gate = 8
        else:
            block_size_gate = 2048
            num_warps_gate = 4

        grid_gate = (total_rows, triton.cdiv(K, block_size_gate))
        apply_gating_2d_kernel[grid_gate](
            x_f32.view(-1), mean, std, z_buf, out_f32.view(-1),
            total_rows, K,
            BLOCK_SIZE=block_size_gate,
            num_warps=num_warps_gate,
            num_stages=2
        )

        # Cast back to bfloat16 to match original behavior
        return out_f32.to(torch.bfloat16)
```

- Kept per-row reduction in fp32 with masked, vectorized loads and fp32 accumulators; this matched torch.std(..., unbiased=False) and was correct.
