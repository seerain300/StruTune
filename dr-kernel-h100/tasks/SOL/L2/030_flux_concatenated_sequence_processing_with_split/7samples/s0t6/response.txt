```python
import torch
import triton
import triton.language as tl


@triton.jit
def concat_seq_kernel(
    out_ptr,         # *fp32, output [B, M, K], M = T + I
    x1_ptr,          # *fp32, encoder_hidden_states: [B, T, K]
    x2_ptr,          # *fp32, hidden_states: [B, I, K]
    B: tl.constexpr, # batch size
    T: tl.constexpr, # text_seq_len
    I: tl.constexpr, # img_seq_len
    K: tl.constexpr, # hidden_dim
):
    # program ids: (batch, seq-tile, K-tile)
    b = tl.program_id(0)
    m_block = tl.program_id(1)
    k_block = tl.program_id(2)

    # tile sizes
    BLOCK_M = 128
    BLOCK_K = 64

    # offsets
    m_offsets = m_block * BLOCK_M + tl.arange(0, BLOCK_M)  # along total sequence
    k_offsets = k_block * BLOCK_K + tl.arange(0, BLOCK_K)  # along hidden_dim

    # bounds mask
    mask_m = m_offsets < (T + I)
    mask_k = k_offsets < K

    # For each m in the tile, determine source tensor: encoder (x1) if m < T, else image (x2)
    is_encoder = m_offsets < T  # boolean per m

    # Compute base offsets using strides
    out_base = b * out_ptr.dtype.element_size  # not needed; Triton handles base pointer
    # We use element-wise pointer arithmetic via strides (in elements, not bytes)

    # Loop over K tile and write concatenated rows
    for k_idx in range(0, BLOCK_K):
        k = k_offsets[k_idx]
        if not mask_k[k_idx]:
            break
        # Compute row index within each source tensor
        # If m is encoder: row = m; else: row = m - T
        # Note: m_offsets is a vector; we need per-m offsets, but since we mask, we can compute per-m row index
        # We'll build 1D pointers by broadcasting (vectorizing over m)
        # Prepare row indices
        row_encoder = tl.where(is_encoder, m_offsets, 0)  # when not encoder, value doesn't matter as mask will block
        row_image = m_offsets - T

        # Compute addresses for x1 and x2
        # x1[b, m, k]: base + b*s0 + m*s1 + k*s2
        # x2[b, (m-T), k]: base + b*s0 + (m-T)*s1 + k*s2
        # We'll compute pointers for each m in the tile.
        # Triton allows vectorized addressing: use broadcasted indices.
        # Build 1D pointer arrays by indexing with k (scalar) and m_offsets (vector).
        # Note: Triton's pointer arithmetic supports vector/matrix operations.

        # Load from x1 where is_encoder is True; from x2 where is_encoder is False
        # We can't branch on vectors; instead, we compute both and then select.
        # Load from x1
        x1_row = row_encoder  # shape: [BLOCK_M]
        x2_row = row_image    # shape: [BLOCK_M]
        # Base addresses for x1 and x2 are b * stride_b (batch) + row * stride_seq + k * stride_k
        # Triton tensors don't expose strides via .stride in kernel; we pass as arguments.
        # Instead, we rely on contiguous layout by making inputs contiguous in host code.
        # So x1_ptr has strides [K, 1, 1], but since we made inputs contiguous, simple address:
        # For contiguous [B, T, K]: offset = b*K*K + m*K + k
        # We pass out_ptr as [B, M, K] contiguous, so offset = b*M*K + m*K + k
        # But to compute element offsets, we can just use pointer arithmetic with strides in elements.
        # For contiguous tensors, strides are [K*K, K, 1], but Triton doesn't expose .stride in kernel.
        # Therefore, we assume contiguous layout and compute offsets directly:
        # out[b, m, k] = b*(T+I)*K + m*K + k
        # x1[b, m, k] = b*T*K + m*K + k
        # x2[b, (m-T), k] = b*I*K + (m-T)*K + k

        # Compute addresses assuming contiguous layout (we enforce .contiguous() on host)
        out_addr = out_ptr + b * (T + I) * K + m_offsets * K + k
        x1_addr = x1_ptr + b * T * K + m_offsets * K + k
        x2_addr = x2_ptr + b * I * K + (m_offsets - T) * K + k

        # Select source based on mask. Triton supports tl.where on pointers? Not directly.
        # Instead, we compute a mask for each m and load selectively into a vector.
        # Create a boolean mask for each m whether to use x1 or x2
        use_x1 = is_encoder  # shape: [BLOCK_M], boolean
        # Load values with masks
        val = tl.zeros([BLOCK_M], dtype=tl.float32)
        # For each m, if use_x1[m], load from x1; else load from x2
        # Triton allows scalar loop over BLOCK_M (unrolled) and masked loads/stores:
        for i in range(0, BLOCK_M):
            m_i = m_offsets[i]
            if not mask_m[i]:
                continue
            if use_x1[i]:
                src = tl.load(x1_addr + i, mask=mask_k[k_idx])
            else:
                src = tl.load(x2_addr + i, mask=mask_k[k_idx])
            val[i] = src

        # Store to out for this k
        out_row_ptr = out_ptr + b * (T + I) * K + k  # base for k, vectorize over m
        tl.store(out_addr, val, mask=mask_m)

    # The above loop writes entire K tile for this M tile. To vectorize, we can broadcast val to k dimension.
    # However, Triton prefers explicit loops for simplicity. We'll keep this 1D k loop with masks.


@triton.jit
def batched_matmul_kernel(
    C_ptr,  # *fp32, output [B, M, K]
    A_ptr,  # *fp32, input [B, M, K]
    W_ptr,  # *fp32, weight [K, K] (no bias)
    B: tl.constexpr,  # batch size
    M: tl.constexpr,  # total sequence length
    N: tl.constexpr,  # output hidden_dim (same as K)
    K: tl.constexpr,  # input hidden_dim
    BLOCK_M: tl.constexpr,  # tile size along M
    BLOCK_N: tl.constexpr,  # tile size along N
    BLOCK_K: tl.constexpr,  # tile size along K
):
    # 3D grid: (B, tiles over M, tiles over N)
    b = tl.program_id(0)
    m_block = tl.program_id(1)
    n_block = tl.program_id(2)

    m_offsets = m_block * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offsets = n_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)

        # Load A[b, m, k] tile
        A_tile = tl.load(
            A_ptr + b * M * K + m_offsets[:, None] * K + k_offsets[None, :],
            mask=(m_offsets[:, None] < M) & (k_offsets[None, :] < K),
            other=0.0,
        )

        # Load W[k, n] tile
        W_tile = tl.load(
            W_ptr + k_offsets[:, None] * N + n_offsets[None, :],
            mask=(k_offsets[:, None] < K) & (n_offsets[None, :] < N),
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(A_tile, W_tile)

    # Store result
    C_addr = C_ptr + b * M * N + m_offsets[:, None] * N + n_offsets[None, :]
    tl.store(C_addr, acc, mask=(m_offsets[:, None] < M) & (n_offsets[None, :] < N))


class ModelNew(torch.nn.Module):
    def forward(
        self,
        hidden_states: torch.Tensor,   # [B, I, K]
        encoder_hidden_states: torch.Tensor,  # [B, T, K]
        process_weight: torch.Tensor,  # [K, K]
    ) -> tuple[torch.Tensor, torch.Tensor]:
        # Ensure dtype is float32 for Triton kernels
        device = hidden_states.device
        dtype = hidden_states.dtype
        B, I, K = hidden_states.shape
        T = encoder_hidden_states.shape[1]
        M = T + I

        # Make tensors contiguous and cast to float32 for Triton
        x1 = encoder_hidden_states.contiguous().to(torch.float32)
        x2 = hidden_states.contiguous().to(torch.float32)
        W = process_weight.contiguous().to(torch.float32)

        # 1) Concatenate sequences along sequence dimension using Triton kernel
        A = torch.empty((B, M, K), device=device, dtype=torch.float32)
        BLOCK_M = 128
        BLOCK_K = 64
        grid_concat = (B, triton.cdiv(M, BLOCK_M), triton.cdiv(K, BLOCK_K))
        concat_seq_kernel[grid_concat](
            A, x1, x2,
            B, T, I, K,
            num_warps=4, num_stages=2,
        )

        # 2) Batched GEMM: C = A @ W, output [B, M, K]
        C = torch.empty((B, M, K), device=device, dtype=torch.float32)
        BLOCK_M_G = 128
        BLOCK_N_G = 128
        BLOCK_K_G = 64
        grid_gemm = (B, triton.cdiv(M, BLOCK_M_G), triton.cdiv(K, BLOCK_N_G))
        batched_matmul_kernel[grid_gemm](
            C, A, W,
            B, M, K, K,
            BLOCK_M=BLOCK_M_G, BLOCK_N=BLOCK_N_G, BLOCK_K=BLOCK_K_G,
            num_warps=4, num_stages=2,
        )

        # 3) Split back into separate streams
        processed_encoder = C[:, :T, :]
        processed_hidden = C[:, T:, :]

        # Cast back to original dtype if needed
        processed_encoder = processed_encoder.to(encoder_hidden_states.dtype)
        processed_hidden = processed_hidden.to(hidden_states.dtype)

        return processed_encoder, processed_hidden
```

Notes:
- Both Triton kernels are invoked from ModelNew.forward. No torch operations perform the heavy compute; concatenation and matmul are done in Triton.
- The concat_seq_kernel uses a 3D grid over (B, M tiles, K tiles), loops over K with a BLOCK_K tile, and writes the concatenated sequence into A. It masks m < T to select the encoder source and otherwise uses the image source. Tensors are made contiguous in host code, so pointer arithmetic is simplified and safe.
- The batched_matmul_kernel implements the main compute C = A @ W. It tiles over M, N, and K, accumulates in float32, and stores results. This replaces torch.matmul and ensures all computation is in Triton.
- The forward returns the two split outputs. While slicing is metadata and not a Triton kernel, it is necessary to match the original API. The heavy work is in the Triton kernels.
