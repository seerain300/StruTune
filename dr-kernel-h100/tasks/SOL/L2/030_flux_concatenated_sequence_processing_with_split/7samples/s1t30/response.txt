Here is the complete Triton-only implementation with the entry point ModelNew. All heavy computation (concatenation, matrix multiplication, and split) is performed by Triton kernels. The forward method only allocates tensors, ensures contiguity, and launches Triton kernels. No torch.matmul or PyTorch tensor compute methods are used in the forward.

```python
import torch
import triton
import triton.language as tl

@triton.jit
def _concatenate_streams_kernel(
    E_ptr,        # encoder_hidden_states: [B, T, D]
    H_ptr,        # hidden_states: [B, I, D]
    C_ptr,        # concatenated output: [B, T+I, D]
    B, T, I, D,
    E_b_stride, E_t_stride, E_d_stride,
    H_b_stride, H_i_stride, H_d_stride,
    C_b_stride, C_seqlen_stride, C_d_stride,
):
    # 2D grid over (batch, position in T+I)
    pid_b = tl.program_id(0)
    pid_pos = tl.program_id(1)
    t_total = T + I
    # Determine which stream (encoder or hidden) and the exact position
    stream = pid_pos // t_total  # 0 => encoder rows, 1 => hidden rows
    pos = pid_pos % t_total
    if stream == 1:
        pos = pos - T  # map to hidden index when stream == 1
    src_ptr = E_ptr + pid_b * E_b_stride + pos * E_t_stride if stream == 0 else H_ptr + pid_b * H_b_stride + pos * H_i_stride
    dst_ptr = C_ptr + pid_b * C_b_stride + pid_pos * C_seqlen_stride

    # Copy D-length row
    for d in range(0, D):
        val = tl.load(src_ptr + d * E_d_stride)
        tl.store(dst_ptr + d * C_d_stride, val)


@triton.jit
def _batched_matmul_ct_wt_kernel(
    C_ptr,        # input X: [B, T+I, D]
    W_ptr,        # process_weight: [D, D], W^T will be [D, D] (same as W)
    Y_ptr,        # output Y: [B, T+I, D]
    B, M, D,      # M = T+I
    C_b_stride, C_m_stride, C_d_stride,
    W_k_stride, W_n_stride,  # W strides: (stride along K, stride along N) where W is [K, N] = [D, D]
    Y_b_stride, Y_m_stride, Y_d_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Grid: (B, tiles over M, tiles over N)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    m0 = pid_m * BLOCK_M
    n0 = pid_n * BLOCK_N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, D, BLOCK_K):
        # load X_sub: [BLOCK_M, BLOCK_K] from C[b, m:m+BLOCK_M, k:k+BLOCK_K]
        X_sub = tl.zeros((BLOCK_M, BLOCK_K), dtype=tl.float32)
        for mi in range(0, BLOCK_M):
            for ki in range(0, BLOCK_K):
                m_idx = m0 + mi
                k_idx = k0 + ki
                # mask for valid m and k
                mask_x = (m_idx < M) & (k_idx < D)
                X_sub[mi, ki] = tl.load(
                    C_ptr + pid_b * C_b_stride + m_idx * C_m_stride + k_idx * C_d_stride,
                    mask=mask_x,
                    other=0.0,
                )

        # load Wt_sub: [BLOCK_K, BLOCK_N] from W[k:k+BLOCK_K, n:n+BLOCK_N], note W is [D, D] and we index as W^T
        Wt_sub = tl.zeros((BLOCK_K, BLOCK_N), dtype=tl.float32)
        for ki in range(0, BLOCK_K):
            for ni in range(0, BLOCK_N):
                k_idx = k0 + ki
                n_idx = n0 + ni
                mask_w = (k_idx < D) & (n_idx < D)
                Wt_sub[ki, ni] = tl.load(
                    W_ptr + k_idx * W_k_stride + n_idx * W_n_stride,
                    mask=mask_w,
                    other=0.0,
                )

        # acc += X_sub @ Wt_sub
        for mi in range(0, BLOCK_M):
            for ki in range(0, BLOCK_K):
                row = X_sub[mi, ki]
                acc[mi, :] += row * Wt_sub[ki, :]

    # store acc to Y[b, m:m+BLOCK_M, n:n+BLOCK_N]
    for mi in range(0, BLOCK_M):
        m_idx = m0 + mi
        for ni in range(0, BLOCK_N):
            n_idx = n0 + ni
            mask_y = (m_idx < M) & (n_idx < D)
            tl.store(
                Y_ptr + pid_b * Y_b_stride + m_idx * Y_m_stride + n_idx * Y_d_stride,
                acc[mi, ni],
                mask=mask_y,
            )


@triton.jit
def _split_copy_kernel(
    Y_ptr,        # processed: [B, T+I, D]
    E_out_ptr,    # processed_encoder: [B, T, D]
    H_out_ptr,    # processed_hidden: [B, I, D]
    B, T, I, D,
    Y_b_stride, Y_m_stride, Y_d_stride,
    E_b_stride, E_m_stride, E_d_stride,
    H_b_stride, H_m_stride, H_d_stride,
    BLOCK_D: tl.constexpr,
):
    # 3D grid over (batch, stream-row index, tiles over D)
    pid_b = tl.program_id(0)
    pid_row = tl.program_id(1)
    pid_dtile = tl.program_id(2)
    d0 = pid_dtile * BLOCK_D
    stream = pid_row // (T + I)  # 0 => encoder, 1 => hidden
    pos = pid_row % (T + I)
    if stream == 1:
        pos = pos - T

    # copy a BLOCK_D tile along feature dimension
    for di in range(0, BLOCK_D):
        d_idx = d0 + di
        val = tl.load(Y_ptr + pid_b * Y_b_stride + pos * Y_m_stride + d_idx * Y_d_stride)
        if stream == 0:
            tl.store(E_out_ptr + pid_b * E_b_stride + pos * E_m_stride + d_idx * E_d_stride, val)
        else:
            # hidden row pos = pos - T
            h_pos = pos - T
            tl.store(H_out_ptr + pid_b * H_b_stride + h_pos * H_m_stride + d_idx * H_d_stride, val)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        """
        Triton-only implementation of the original run:
        - Concatenate encoder_hidden_states and hidden_states along sequence dimension.
        - Apply linear projection without bias: (concatenated @ process_weight.T).
        - Split back into separate encoder and image streams.
        """
        # Ensure dtype and contiguity
        dtype = hidden_states.dtype
        assert hidden_states.device.type == "cuda", "Inputs must be on CUDA device for Triton kernels."
        E = encoder_hidden_states.contiguous()
        H = hidden_states.contiguous()
        W = process_weight.contiguous()

        B, T, D = E.shape
        I = H.shape[1]
        M_total = T + I

        # 1) Concatenate streams into C: [B, T+I, D]
        C = torch.empty((B, M_total, D), device=E.device, dtype=torch.float32)

        grid_concat = (B, M_total)
        _concatenate_streams_kernel[grid_concat](
            E, H, C,
            B, T, I, D,
            E.stride(0), E.stride(1), E.stride(2),
            H.stride(0), H.stride(1), H.stride(2),
            C.stride(0), C.stride(1), C.stride(2),
            num_warps=4, num_stages=2,
        )

        # 2) Compute processed = C @ W^T using Triton GEMM (no bias)
        Y = torch.empty((B, M_total, D), device=E.device, dtype=torch.float32)

        # Grid over (B, tiles over M_total, tiles over D)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        grid_gemm = (B, triton.cdiv(M_total, BLOCK_M), triton.cdiv(D, BLOCK_N))
        _batched_matmul_ct_wt_kernel[grid_gemm](
            C, W, Y,
            B, M_total, D,
            C.stride(0), C.stride(1), C.stride(2),
            W.stride(0), W.stride(1),
            Y.stride(0), Y.stride(1), Y.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=2,
        )

        # 3) Split into encoder and hidden outputs
        processed_encoder = torch.empty((B, T, D), device=E.device, dtype=torch.float32)
        processed_hidden = torch.empty((B, I, D), device=E.device, dtype=torch.float32)

        BLOCK_D = 128
        grid_split = (B, M_total, triton.cdiv(D, BLOCK_D))
        _split_copy_kernel[grid_split](
            Y, processed_encoder, processed_hidden,
            B, T, I, D,
            Y.stride(0), Y.stride(1), Y.stride(2),
            processed_encoder.stride(0), processed_encoder.stride(1), processed_encoder.stride(2),
            processed_hidden.stride(0), processed_hidden.stride(1), processed_hidden.stride(2),
            BLOCK_D=BLOCK_D,
            num_warps=4, num_stages=2,
        )

        # Cast outputs back to original dtype if desired
        processed_encoder = processed_encoder.to(dtype)
        processed_hidden = processed_hidden.to(dtype)

        return processed_encoder, processed_hidden
```

Notes:
- All heavy work (concatenation, matrix multiplication, split) is done inside Triton kernels. The forward method does no torch.matmul or PyTorch tensor compute methods, satisfying the Triton-only requirement.
- The batched GEMM kernel computes Y[b, m, n] = sum_k C[b, m, k] * W[k, n], which matches processed = concatenated @ process_weight.T (no bias). We accumulate in float32 for numerical stability.
- The split kernel copies feature tiles into the two outputs based on stream index.
