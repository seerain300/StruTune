- Original operation:
  - Concatenate encoder_hidden_states [B, T, H] and hidden_states [B, I, H] along the sequence dimension to form concatenated [B, L=T+I, H].
  - Apply linear projection: processed = concatenated @ process_weight.T, where process_weight is [H, H], yielding [B, L, H].
  - Split: processed_encoder = processed[:, :T, :], processed_hidden = processed[:, T:, :].
- Goals:
  - Implement concatenation as a Triton kernel that writes the concatenated tensor directly from the source tensors, with stride-based addressing and masks. This avoids torch.cat and reduces complexity.
  - Implement matmul as a Triton kernel that computes processed = concatenated @ process_weight.T. We’ll use a 2D grid over rows (M=B*L) and columns (N=H), and loop over K=H in chunks (BLOCK_K). Use masks for non-divisible sizes. Accumulate in float32 for numerical stability; Triton will cast on store to the output dtype.
  - Implement splitting as Triton copy kernels that copy the first T rows and the next I rows from processed into the two outputs.
  - Ensure all kernels are actually launched from ModelNew.forward; no torch operations are used in the host code on tensors.

import torch
import triton
import triton.language as tl


@triton.jit
def concat_encoder_image_kernel(
    e_ptr,                # *const float: [B, T, H]
    i_ptr,                # *const float: [B, I, H]
    out_ptr,              # *float: [B, L, H], L = T + I
    B: tl.int32, T: tl.int32, I: tl.int32, H: tl.int32,
    e_s0, e_s1, e_s2,     # strides for encoder
    i_s0, i_s1, i_s2,     # strides for image
    o_s0, o_s1, o_s2,     # strides for output
    BLOCK_l: tl.constexpr, BLOCK_h: tl.constexpr,
):
    # 3D launch: (B, ceil((T+I)/BLOCK_l), ceil(H/BLOCK_h))
    pid_b = tl.program_id(0)
    pid_l = tl.program_id(1)
    pid_h = tl.program_id(2)

    l = pid_l * BLOCK_l + tl.arange(0, BLOCK_l)  # sequence indices in [0, T+I)
    h = pid_h * BLOCK_h + tl.arange(0, BLOCK_h)  # hidden dims

    # Bounds masks
    mask_l = l < (T + I)
    mask_h = h < H
    mask = mask_l[:, None] & mask_h[None, :]     # [BLOCK_l, BLOCK_h]

    # Compute source pointers
    # If l < T: take from encoder at (b, l, h); else: take from image at (b, l - T, h)
    is_encoder = l < T
    # For encoder
    e_offsets = pid_b * e_s0 + l[None, :] * e_s1 + h[:, None] * e_s2
    e_vals = tl.load(e_ptr + e_offsets, mask=mask & is_encoder[:, None], other=0.0)
    # For image
    i_offsets = pid_b * i_s0 + (l[None, :] - T) * i_s1 + h[:, None] * i_s2
    i_vals = tl.load(i_ptr + i_offsets, mask=mask & (~is_encoder)[:, None], other=0.0)

    out_vals = tl.where(is_encoder[:, None], e_vals, i_vals)

    # Store to output
    out_offsets = pid_b * o_s0 + l[None, :] * o_s1 + h[:, None] * o_s2
    tl.store(out_ptr + out_offsets, out_vals, mask=mask)


@triton.jit
def matmul_kernel(
    A_ptr,        # *const float: [B, L, H], input concatenated
    W_ptr,        # *const float: [H, H], process_weight (no bias)
    C_ptr,        # *float: [B, L, H], output processed
    B: tl.int32, L: tl.int32, H: tl.int32,
    a_s0, a_s1, a_s2,
    w_s0, w_s1, w_s2,
    c_s0, c_s1, c_s2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Grid over rows (M = B*L) and cols (N = H)
    pid_m = tl.program_id(0)  # over rows
    pid_n = tl.program_id(1)  # over cols

    m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]

    mask_m = m < (B * L)
    mask_n = n < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K = H
    for k0 in range(0, H, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)  # [BLOCK_K]
        mask_k = k < H

        # A tile: [BLOCK_M, BLOCK_K]
        a_offsets = m[:, None] * a_s0 + (m // L) * a_s1 + (m % L) * a_s2
        A_tile = tl.load(
            A_ptr + a_offsets,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # W^T tile: we need W[k, n] -> shape [BLOCK_K, BLOCK_N]
        w_offsets = k[:, None] * w_s0 + n[None, :] * w_s1
        Wt_tile = tl.load(
            W_ptr + w_offsets,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(A_tile, Wt_tile)

    # Write back to C: [BLOCK_M, BLOCK_N]
    c_offsets = (m[:, None] // L) * c_s0 + (m[:, None] % L) * c_s1 + n[None, :] * c_s2
    tl.store(
        C_ptr + c_offsets,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


@triton.jit
def copy_rows_kernel(
    src_ptr,            # *const float: [B, L, H]
    dst_ptr,            # *float: [B, ? , H]
    B: tl.int32, L: tl.int32, H: tl.int32,
    NUM_ROWS: tl.int32,      # number of rows to copy (T or I)
    ROW_START: tl.int32,     # starting row in src for copy
    src_s0, src_s1, src_s2,
    dst_s0, dst_s1, dst_s2,
    BLOCK_l: tl.constexpr, BLOCK_h: tl.constexpr,
):
    # 3D grid: (B, ceil(NUM_ROWS/BLOCK_l), ceil(H/BLOCK_h))
    pid_b = tl.program_id(0)
    pid_lr = tl.program_id(1)
    pid_h = tl.program_id(2)

    r = pid_lr * BLOCK_l + tl.arange(0, BLOCK_l)  # row indices in src to copy [0, NUM_ROWS)
    h = pid_h * BLOCK_h + tl.arange(0, BLOCK_h)

    mask_r = r < NUM_ROWS
    mask_h = h < H
    mask = mask_r[:, None] & mask_h[None, :]

    src_offsets = pid_b * src_s0 + (ROW_START + r[None, :]) * src_s1 + h[:, None] * src_s2
    vals = tl.load(src_ptr + src_offsets, mask=mask, other=0.0)

    dst_offsets = pid_b * dst_s0 + r[None, :] * dst_s1 + h[:, None] * dst_s2
    tl.store(dst_ptr + dst_offsets, vals, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        # Shapes
        B = hidden_states.shape[0]
        I = hidden_states.shape[1]
        T = encoder_hidden_states.shape[1]
        H = hidden_states.shape[2]
        L = T + I

        # Ensure inputs are contiguous and on same device
        device = hidden_states.device
        e = encoder_hidden_states.contiguous()
        i = hidden_states.contiguous()
        w = process_weight.contiguous()

        # 1) Concatenate along sequence dimension using Triton
        out = torch.empty((B, L, H), dtype=torch.float32, device=device)

        # Strides
        e_s0, e_s1, e_s2 = e.stride()
        i_s0, i_s1, i_s2 = i.stride()
        o_s0, o_s1, o_s2 = out.stride()

        # Launch concat kernel
        BLOCK_l = 64
        BLOCK_h = 64
        grid_concat = (B, triton.cdiv(L, BLOCK_l), triton.cdiv(H, BLOCK_h))
        concat_encoder_image_kernel[grid_concat](
            e, i, out,
            B, T, I, H,
            e_s0, e_s1, e_s2,
            i_s0, i_s1, i_s2,
            o_s0, o_s1, o_s2,
            BLOCK_l=BLOCK_l, BLOCK_h=BLOCK_h,
            num_warps=4, num_stages=2,
        )

        # 2) Matmul: processed = out @ w.T, out is [B, L, H], w is [H, H]
        processed = torch.empty((B, L, H), dtype=torch.float32, device=device)

        a_s0, a_s1, a_s2 = out.stride()
        w_s0, w_s1, w_s2 = w.stride()
        c_s0, c_s1, c_s2 = processed.stride()

        # Choose block sizes; these can be tuned
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

        grid_matmul = (triton.cdiv(B * L, BLOCK_M), triton.cdiv(H, BLOCK_N))
        matmul_kernel[grid_matmul](
            out, w, processed,
            B, L, H,
            a_s0, a_s1, a_s2,
            w_s0, w_s1, w_s2,
            c_s0, c_s1, c_s2,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=2,
        )

        # 3) Split into encoder and image streams using Triton copy kernels
        processed_encoder = torch.empty((B, T, H), dtype=torch.float32, device=device)
        processed_hidden = torch.empty((B, I, H), dtype=torch.float32, device=device)

        # For encoder: copy first T rows
        grid_enc = (B, triton.cdiv(T, 64), triton.cdiv(H, 64))
        copy_rows_kernel[grid_enc](
            processed, processed_encoder,
            B, L, H,
            NUM_ROWS=T,
            ROW_START=0,
            processed.stride(0), processed.stride(1), processed.stride(2),
            processed_encoder.stride(0), processed_encoder.stride(1), processed_encoder.stride(2),
            BLOCK_l=64, BLOCK_h=64,
            num_warps=4, num_stages=2,
        )

        # For image: copy next I rows
        grid_img = (B, triton.cdiv(I, 64), triton.cdiv(H, 64))
        copy_rows_kernel[grid_img](
            processed, processed_hidden,
            B, L, H,
            NUM_ROWS=I,
            ROW_START=T,
            processed.stride(0), processed.stride(1), processed.stride(2),
            processed_hidden.stride(0), processed_hidden.stride(1), processed_hidden.stride(2),
            BLOCK_l=64, BLOCK_h=64,
            num_warps=4, num_stages=2,
        )

        return processed_encoder, processed_hidden
