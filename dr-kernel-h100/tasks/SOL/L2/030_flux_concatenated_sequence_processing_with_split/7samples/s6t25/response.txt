import torch
import triton
import triton.language as tl


@triton.jit
def cat_linear_kernel(
    encoder_ptr,   # [B, T, H], float32
    hidden_ptr,    # [B, I, H], float32
    weight_ptr,    # [H, H], float32
    out_ptr,       # [B, T+I, H], float32
    B: tl.int32, T: tl.int32, I: tl.int32, H: tl.int32,
    stride_e_n: tl.int32, stride_e_s: tl.int32, stride_e_h: tl.int32,
    stride_h_n: tl.int32, stride_h_s: tl.int32, stride_h_h: tl.int32,
    stride_w_h: tl.int32, stride_w_k: tl.int32,
    stride_out_n: tl.int32, stride_out_s: tl.int32, stride_out_h: tl.int32,
    total_seq: tl.int32,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # Program ids: each program computes one (n, s) row of out
    pid_n = tl.program_id(0)
    pid_s = tl.program_id(1)

    # Output pointer for this (n, s)
    out_row_ptr = out_ptr + pid_n * stride_out_n + pid_s * stride_out_s

    # Accumulator for output vector
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # Determine if this s belongs to encoder or hidden stream
    is_encoder = pid_s < T

    # Loop over K (input features) in tiles of BLOCK_K
    k0 = 0
    while k0 < H:
        # Build index vector for k
        k_idx = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_idx < H

        # Load input vector element k from the appropriate source
        if is_encoder:
            src_ptr = encoder_ptr + pid_n * stride_e_n + pid_s * stride_e_s
            input_k = tl.load(src_ptr + k_idx * stride_e_h, mask=k_mask, other=0.0)
        else:
            src_ptr = hidden_ptr + pid_n * stride_h_n + (pid_s - T) * stride_h_s
            input_k = tl.load(src_ptr + k_idx * stride_h_h, mask=k_mask, other=0.0)

        # Accumulate acc[h] += sum_k input_k[k] * weight[k, h] for h in this tile
        h0 = 0
        while h0 < H:
            h_idx = h0 + tl.arange(0, BLOCK_H)
            h_mask = h_idx < H
            # Load weight slice [BLOCK_K, BLOCK_H]: weight[k, h]
            w_ptrs = weight_ptr + k_idx[:, None] * stride_w_h + h_idx[None, :] * stride_w_k
            w_load = tl.load(w_ptrs, mask=k_mask[:, None] & h_mask[None, :], other=0.0)
            # Accumulate: acc[h] += sum over k of input_k[k] * w_load[k, h]
            acc[h_idx] += tl.sum(w_load * input_k[:, None], axis=0)
            h0 += BLOCK_H

        k0 += BLOCK_K

    # Store the accumulated output vector for this (n, s)
    out_ptrs = out_row_ptr + tl.arange(0, BLOCK_H) * stride_out_h
    out_mask = tl.arange(0, BLOCK_H) < H
    tl.store(out_ptrs, acc, mask=out_mask)

        # Ensure tensors are on the same device and dtype; keep float32 for stability

        _, I, H2 = hidden_states.shape

        # Make inputs contiguous
        e = encoder_hidden_states.contiguous()

        w = process_weight.contiguous()

        # Allocate output [B, T+I, H] contiguous
        total_seq = T + I
        out = torch.empty((B, total_seq, H), dtype=torch.float32, device=device)

        # Strides
        stride_e_n, stride_e_s, stride_e_h = e.stride(0), e.stride(1), e.stride(2)
        stride_h_n, stride_h_s, stride_h_h = h.stride(0), h.stride(1), h.stride(2)
        stride_w_h, stride_w_k = w.stride(0), w.stride(1)  # w is [H, H]
        stride_out_n, stride_out_s, stride_out_h = out.stride(0), out.stride(1), out.stride(2)

        # Launch Triton kernel over (B, total_seq)
        # Tile sizes: choose 128 for H; masks handle tails
