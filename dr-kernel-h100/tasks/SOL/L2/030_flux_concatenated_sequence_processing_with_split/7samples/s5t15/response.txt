import torch
import triton
import triton.language as tl


@triton.jit
def concat_encoder_image_kernel(
    e_ptr,                # *const T: [B, T, H]
    i_ptr,                # *const T: [B, I, H]
    out_ptr,              # *T: [B, L, H] with L=T+I
    B: tl.int32, T: tl.int32, I: tl.int32, H: tl.int32,
    e_s0, e_s1, e_s2,     # strides for e_ptr (B, T, H)
    i_s0, i_s1, i_s2,     # strides for i_ptr (B, I, H)
    o_s0, o_s1, o_s2,     # strides for out_ptr (B, L, H)
    BLOCK_l: tl.constexpr, BLOCK_h: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_l = tl.program_id(1)
    pid_h = tl.program_id(2)

    b = pid_b
    l_start = pid_l * BLOCK_l
    h_start = pid_h * BLOCK_h

    l = l_start + tl.arange(0, BLOCK_l)   # [BLOCK_l]
    h = h_start + tl.arange(0, BLOCK_h)   # [BLOCK_h]

    # 2D grid: we'll broadcast to [BLOCK_l, BLOCK_h] with masks
    L = T + I

    # Masks for bounds
    mask_l = l < L
    mask_h = h < H
    mask_2d = mask_l[:, None] & mask_h[None, :]

    # Select source: if l < T -> encoder, else -> image at l - T
    use_encoder = l < T

    # Compute base pointers for each source
    e_ptrs = e_ptr + b * e_s0 + l[:, None] * e_s1 + h[None, :] * e_s2
    i_ptrs = i_ptr + b * i_s0 + (l[:, None] - T) * i_s1 + h[None, :] * i_s2

    # Select appropriate source based on use_encoder
    # We create a pointer that is e_ptrs where use_encoder is True, else i_ptrs
    # Triton allows masked load/store, but for assignment we need to select. Use where-like selection:
    out_ptrs = out_ptr + b * o_s0 + l[:, None] * o_s1 + h[None, :] * o_s2

    # We'll load with masks; Triton will handle broadcasting. For correctness, we do masked load/store.
    # To perform the selection, we can compute values from both and store based on use_encoder.
    # However, Triton doesn't allow Python-side if in kernel. Instead, we compute both and use masked load to fetch from correct source.

    # Load from encoder where use_encoder is True
    val_e = tl.load(e_ptrs, mask=mask_2d & use_encoder[:, None], other=0)
    # Load from image where use_encoder is False
    val_i = tl.load(i_ptrs, mask=mask_2d & (~use_encoder)[:, None], other=0)
    # Compose output: val_e where use_encoder, else val_i
    val = tl.where(use_encoder[:, None], val_e, val_i)

    # Store
    tl.store(out_ptrs, val, mask=mask_2d)


@triton.jit
def matmul_kernel(
    A_ptr, W_ptr, C_ptr,
    M: tl.int32, N: tl.int32, K: tl.int32,
    A_s0, A_s1, A_s2,
    W_s0, W_s1, W_s2,
    C_s0, C_s1, C_s2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile along M
    pid_n = tl.program_id(1)  # tile along N

    m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        k_ids = k + tl.arange(0, BLOCK_K)  # [BLOCK_K]

        # A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + m[:, None] * A_s0 + k_ids[None, :] * A_s2
        a_mask = (m[:, None] < M) & (k_ids[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # W^T tile: W is [K, N] where W^T is [N, K] for our use (process_weight.T with shape [H, H])
        # We need to load W^T[k, n] = W[n, k]
        w_ptrs = W_ptr + n[None, :] * W_s0 + k_ids[:, None] * W_s1
        w_mask = (n[None, :] < N) & (k_ids[:, None] < K)
        w = tl.load(w_ptrs, mask=w_mask, other=0.0)

        acc += tl.dot(a, w)

    # Store C[m, n] = acc
    c_ptrs = C_ptr + m[:, None] * C_s0 + n[None, :] * C_s1
    c_mask = (m[:, None] < M) & (n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


@triton.jit
def copy_rows_kernel(
    src_ptr, dst_ptr,
    B: tl.int32, L: tl.int32, H: tl.int32,
    src_s0, src_s1, src_s2,
    dst_s0, dst_s1, dst_s2,
    ROW_START: tl.int32,  # starting row index in src for copy
    BLOCK_l: tl.constexpr, BLOCK_h: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_l = tl.program_id(1)  # tile along rows
    pid_h = tl.program_id(2)  # tile along hidden dim

    l_start = pid_l * BLOCK_l
    h_start = pid_h * BLOCK_h

    l = l_start + tl.arange(0, BLOCK_l)   # rows to copy
    h = h_start + tl.arange(0, BLOCK_h)   # hidden dim

    mask_l = l < L
    mask_h = h < H
    mask_2d = mask_l[:, None] & mask_h[None, :]

    src_rows = ROW_START + l  # actual rows in src
    src_ptrs = src_ptr + pid_b * src_s0 + src_rows[:, None] * src_s1 + h[None, :] * src_s2
    dst_ptrs = dst_ptr + pid_b * dst_s0 + l[:, None] * dst_s1 + h[None, :] * dst_s2

    val = tl.load(src_ptrs, mask=mask_2d, other=0.0)
    tl.store(dst_ptrs, val, mask=mask_2d)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
                encoder_hidden_states: torch.Tensor,
                process_weight: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-optimized version of the given PyTorch run function.
        hidden_states: [B, I, H]
        encoder_hidden_states: [B, T, H]
        process_weight: [H, H]
        Returns (processed_encoder: [B, T, H], processed_hidden: [B, I, H])
        """
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, \
            "All tensors must be on CUDA device for Triton kernels."

        B, T, H = encoder_hidden_states.shape
        B2, I, H2 = hidden_states.shape
        assert B == B2, "Batch size mismatch between encoder and image tensors."
        assert H == H2, "Hidden dim mismatch."
        L = T + I
        W = process_weight
        assert W.shape[0] == H and W.shape[1] == H, "process_weight must be [H, H]."

        # 1) Concatenate encoder_hidden_states and hidden_states along sequence dimension using Triton
        concatenated = torch.empty((B, L, H), device=hidden_states.device, dtype=hidden_states.dtype)

        BLOCK_l = 64
        BLOCK_h = 64
        grid_concat = (B, triton.cdiv(L, BLOCK_l), triton.cdiv(H, BLOCK_h))
        concat_encoder_image_kernel[grid_concat](
            encoder_hidden_states, hidden_states, concatenated,
            B, T, I, H,
            encoder_hidden_states.stride(0), encoder_hidden_states.stride(1), encoder_hidden_states.stride(2),
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            BLOCK_l=BLOCK_l, BLOCK_h=BLOCK_h,
            num_warps=4, num_stages=2,
        )

        # 2) Matrix multiplication: processed = concatenated @ process_weight.T
        # Ensure process_weight is [H, H], which is already the case.
        processed = torch.empty((B, L, H), device=hidden_states.device, dtype=hidden_states.dtype)

        M = B * L
        N = H
        K = H

        # For matmul, Triton expects shapes A: [M, K], W: [K, N] (here W is process_weight.T)
        # We pass W as process_weight.T; but since process_weight is [H, H], W^T is [H, H].
        # To keep it general, we can call torch.transpose to get W.T explicitly.
        W_T = process_weight.transpose(0, 1).contiguous()  # [H, H] -> [H, H], but we make it contiguous.

        # Launch matmul kernel
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        grid_matmul = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
        matmul_kernel[grid_matmul](
            concatenated, W_T, processed,
            M, N, K,
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            W_T.stride(0), W_T.stride(1), W_T.stride(2),
            processed.stride(0), processed.stride(1), processed.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=2,
        )

        # 3) Split into encoder and hidden streams using Triton copy kernels
        processed_encoder = torch.empty((B, T, H), device=hidden_states.device, dtype=hidden_states.dtype)
        processed_hidden = torch.empty((B, I, H), device=hidden_states.device, dtype=hidden_states.dtype)

        # Copy first T rows
        grid_copy_encoder = (B, triton.cdiv(T, BLOCK_l), triton.cdiv(H, BLOCK_h))
        copy_rows_kernel[grid_copy_encoder](
            processed, processed_encoder,
            B, L, H,
            processed.stride(0), processed.stride(1), processed.stride(2),
            processed_encoder.stride(0), processed_encoder.stride(1), processed_encoder.stride(2),
            ROW_START=0, BLOCK_l=BLOCK_l, BLOCK_h=BLOCK_h,
            num_warps=4, num_stages=2,
        )

        # Copy next I rows
        grid_copy_hidden = (B, triton.cdiv(I, BLOCK_l), triton.cdiv(H, BLOCK_h))
        copy_rows_kernel[grid_copy_hidden](
            processed, processed_hidden,
            B, L, H,
            processed.stride(0), processed.stride(1), processed.stride(2),
            processed_hidden.stride(0), processed_hidden.stride(1), processed_hidden.stride(2),
            ROW_START=T, BLOCK_l=BLOCK_l, BLOCK_h=BLOCK_h,
            num_warps=4, num_stages=2,
        )

        return processed_encoder, processed_hidden

Explanation of changes and why this should work:
- Concatenation kernel now fully defines the selection logic with masks: we compute pointer grids for both encoder and image and perform masked loads based on l < T. This avoids any out-of-bounds access and ensures correct selection. We store the composed value into out with mask.
- The matmul kernel is a straightforward tiled GEMM with masks. We compute processed = concatenated @ process_weight.T. We pass process_weight.T as W_T with shape [H, H] (contiguous) and use BLOCK_M=64, BLOCK_N=64, BLOCK_K=64. The grid is (ceil(M/64), ceil(N/64)) = (ceil(B*(T+I)/64), ceil(H/64)). This keeps the kernel simple and robust. Note: the output processed uses the same dtype as concatenated; Triton’s tl.dot accumulates in float32, which matches typical hidden_dim dtype. If your model uses fp16/bf16, you may want to cast loads to float32 and cast stores back, but given the default float32 inputs, this should be fine.
- Splitting uses Triton copy_rows_kernel to copy the first T rows and the next I rows from processed to the two outputs. This avoids torch slicing and ensures Triton-only execution.

Performance considerations:
- This is a correctness-first implementation. For speed, you can:
  - Use larger BLOCK sizes (e.g., 128) and more warps (e.g., 8) tuned for your GPU.
  - Adopt a more advanced tiled matmul (B, M, K) x (K, N) with tl.dot optimized for your shapes.
  - Consider using fp32 accumulation for fp16/bf16 inputs and cast back at store time for better numerical stability.
