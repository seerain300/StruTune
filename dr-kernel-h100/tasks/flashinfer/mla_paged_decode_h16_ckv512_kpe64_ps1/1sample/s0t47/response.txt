import math
import torch

@torch.no_grad()
def run(q_nope, q_pe, ckv_cache, kpe_cache, kv_indptr, kv_indices, sm_scale):
    # Shapes
    batch_size, num_qo_heads, head_dim_ckv = q_nope.shape
    head_dim_kpe = q_pe.shape[-1]
    num_pages = ckv_cache.shape[0]
    _, _, kpe_dim = kpe_cache.shape

    # Cast to fp32 and ensure contiguous
    qn = q_nope.to(torch.float32).contiguous()  # [B, H, N]
    qp = q_pe.to(torch.float32).contiguous()    # [B, H, Kp_dim]

    Kc_all = ckv_cache.to(torch.float32).contiguous()   # [num_pages, N]
    Kp_all = kpe_cache.to(torch.float32).contiguous()   # [num_pages, Kp_dim]

    # Output and lse
    output = torch.empty(
        (batch_size, num_qo_heads, head_dim_ckv), dtype=torch.bfloat16, device=qn.device
    )
    lse = torch.full((batch_size, num_qo_heads), -float("inf"), dtype=torch.float32, device=qn.device)

    for b in range(batch_size):
        # Define the set of token indices for this batch element
        start = int(kv_indptr[b].item())
        end = int(kv_indptr[b + 1].item())
        M_total = end - start
        if M_total <= 0:
            lse[b] = -float("inf")
            output[b].zero_()
            continue

        tok_idx = kv_indices[start:end].to(torch.long)  # [M_total]
        # Gather Kc and Kp rows for these tokens
        Kc = Kc_all[tok_idx]  # [M_total, N]
        Kp = Kp_all[tok_idx]  # [M_total, Kp_dim]

        # Compute logits for each head
        for h in range(num_qo_heads):
            qnh = qn[b, h]  # [N]
            qph = qp[b, h]  # [Kp_dim]
            logits = qnh.unsqueeze(0) @ Kc.transpose(0, 1) + qph.unsqueeze(0) @ Kp.transpose(0, 1)  # [1, M_total]
            logits = logits.squeeze(0)  # [M_total]
            logits_scaled = logits * (float(sm_scale) if sm_scale is not None else 1.0)

            # LogSumExp over tokens, divide by ln(2) to match original
            row_max = torch.max(logits_scaled)
            sum_exp = torch.sum(torch.exp(logits_scaled - row_max))
            lse[b, h] = math.log(sum_exp) / math.log(2.0)

            # Compute attention weights and output
            # attn = softmax(logits_scaled, dim=0) scaled by M_total
            attn = torch.exp(logits_scaled - lse[b, h])  # [M_total]
            # Output is sum over tokens: attn[m] * Kc[m, :]
            out_vec = (attn.unsqueeze(1) * Kc).sum(dim=0)  # [N]
            output[b, h] = out_vec.to(torch.bfloat16)

    return output, lse
