  5) qn = q_nope[b, h, :] -> [Dc], qp = q_pe[b, h, :] -> [Dp].

  8) lse[b, h] = logsumexp(logits_scaled) / ln(2) (base-2 logsumexp).
  9) attn[b, h, t] = softmax(logits_scaled[h, t]).
  10) output[b, h, :] = attn[b, h, :] @ Kc[:, :] -> [Dc].
