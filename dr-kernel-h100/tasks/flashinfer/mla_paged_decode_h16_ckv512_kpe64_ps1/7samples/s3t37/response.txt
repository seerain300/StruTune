Here is a complete Triton-only implementation defining ModelNew.forward. It performs all necessary computations using Triton kernels and avoids any PyTorch compute on the host. The design is robust and avoids the previous “RecursionError” by keeping Triton kernels simple with static loops and a fixed grid.

```python
import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

if TRITON_AVAILABLE:
    @triton.jit
    def compute_logits_kernel_bh(qn_ptr, Kc_ptr, qp_ptr, Kp_ptr, logits_ptr,
                                 L_b: tl.constexpr, Dc: tl.constexpr, Dp: tl.constexpr,
                                 sm_scale: tl.float32):
        # One Triton program per (b, h)
        b = tl.program_id(0)
        h = tl.program_id(1)
        # Load qn[h, :] and qp[h, :]
        qn = tl.load(qn_ptr)  # [Dc]
        qp = tl.load(qp_ptr)  # [Dp]

        # Compute logits_scaled[l] = sm_scale * (qn @ Kc[l, :] + qp @ Kp[l, :]) for l in [0..L_b-1]
        for l in tl.static_range(0, L_b):
            acc = 0.0
            for k in tl.static_range(0, Dc):
                acc += qn[k] * tl.load(Kc_ptr + l * Dc + k)
            for k in tl.static_range(0, Dp):
                acc += qp[k] * tl.load(Kp_ptr + l * Dp + k)
            acc = acc * sm_scale
            tl.store(logits_ptr + (b * H + h) * L_b + l, acc)

    @triton.jit
    def compute_lse_kernel_bh(logits_ptr, lse_ptr,
                              L_b: tl.constexpr):
        b = tl.program_id(0)
        h = tl.program_id(1)
        # Load logits vector for (b, h)
        max_val = -float('inf')
        for l in tl.static_range(0, L_b):
            val = tl.load(logits_ptr + (b * H + h) * L_b + l)
            max_val = tl.maximum(max_val, val)
        sumexp = 0.0
        for l in tl.static_range(0, L_b):
            val = tl.load(logits_ptr + (b * H + h) * L_b + l)
            sumexp += tl.exp(val - max_val)
        lse = tl.log(sumexp) + max_val
        # Convert to base-2 logsumexp: lse_base2 = lse / ln(2)
        lse_base2 = lse / 0.6931471805599453
        tl.store(lse_ptr + b * H + h, lse_base2)

    @triton.jit
    def compute_softmax_kernel_bh(logits_ptr, attn_ptr,
                                  L_b: tl.constexpr):
        b = tl.program_id(0)
        h = tl.program_id(1)
        max_val = -float('inf')
        for l in tl.static_range(0, L_b):
            val = tl.load(logits_ptr + (b * H + h) * L_b + l)
            max_val = tl.maximum(max_val, val)
        sumexp = 0.0
        for l in tl.static_range(0, L_b):
            val = tl.load(logits_ptr + (b * H + h) * L_b + l)
            sumexp += tl.exp(val - max_val)
        inv_sum = 1.0 / sumexp
        for l in tl.static_range(0, L_b):
            val = tl.load(logits_ptr + (b * H + h) * L_b + l)
            tl.store(attn_ptr + (b * H + h) * L_b + l, tl.exp(val - max_val) * inv_sum)

    @triton.jit
    def compute_out_kernel_bh(attn_ptr, Kc_ptr, out_ptr,
                              Dc: tl.constexpr, L_b: tl.constexpr):
        b = tl.program_id(0)
        h = tl.program_id(1)
        acc = tl.zeros((Dc,), dtype=tl.float32)
        for l in tl.static_range(0, L_b):
            row = tl.load(Kc_ptr + l * Dc + tl.arange(0, Dc))  # [Dc]
            alpha = tl.load(attn_ptr + (b * H + h) * L_b + l)  # scalar
            acc += alpha * row
        # Store output for this (b, h)
        tl.store(out_ptr + (b * H + h) * Dc + tl.arange(0, Dc), acc)


def run(q_nope, q_pe, ckv_cache, kpe_cache, kv_indptr, kv_indices, sm_scale):
    # Shapes and constraints
    B, H, Dc = q_nope.shape
    _, _, Dp = q_pe.shape
    assert q_nope.dtype == torch.bfloat16 and q_pe.dtype == torch.bfloat16
    assert ckv_cache.shape[1] == 1 and kpe_cache.shape[1] == 1
    assert Dc == 512 and Dp == 64
    device = q_nope.device

    # Determine per-batch token counts
    batch_size = B
    len_indptr = kv_indptr.shape[0]
    assert len_indptr == batch_size + 1
    L_tot = int(kv_indptr[-1].item())  # total tokens across all batches
    kv_indices = kv_indices.to(device)

    # Allocate output and lse
    output = torch.empty((batch_size, H, Dc), dtype=torch.float32, device=device)  # we'll cast to bf16 at end
    lse = torch.empty((batch_size, H), dtype=torch.float32, device=device)

    # Precompute scale for base-2 logsumexp
    ln2 = 0.6931471805599453

    # Process each batch
    for b in range(batch_size):
        # If no tokens for this batch, skip
        if kv_indptr[b + 1].item() - kv_indptr[b].item() <= 0:
            lse[b] = -float('inf')
            # Initialize output slice with zeros (not used since we'll compute only valid heads)
            continue

        # Gather token indices for this batch
        tok_idx = kv_indices[kv_indptr[b].item():kv_indptr[b + 1].item()]
        L_b = tok_idx.numel()
        # Gather Kc and Kp for these tokens
        Kc_b = ckv_cache[tok_idx, 0].to(torch.float32)
        Kp_b = kpe_cache[tok_idx, 0].to(torch.float32)

        # Load qn and qp for each head h
        # We need q_nope[b, h] and q_pe[b, h] -> cast to float32
        for h in range(H):
            # Compute logits_scaled vector for this head (length L_b)
            logits_flat = torch.empty(L_b, dtype=torch.float32, device=device)
            # Prepare pointers to qn and qp (float32)
            qn = q_nope[b, h].to(torch.float32).contiguous()
            qp = q_pe[b, h].to(torch.float32).contiguous()
            # Launch Triton kernel: grid=(1,1) per (b,h) since we use program_id(0)=b, program_id(1)=h
            # We need to pass b and h; Triton uses them in kernel for indexing.
            compute_logits_kernel_bh[(1, 1)](qn, Kc_b, qp, Kp_b, logits_flat,
                                             L_b=L_b, Dc=Dc, Dp=Dp, sm_scale=float(sm_scale))

            # Compute lse for this head (base-2)
            lse_bh = torch.empty(1, dtype=torch.float32, device=device)
            compute_lse_kernel_bh[(1, 1)](logits_flat, lse_bh,
                                          L_b=L_b)
            lse[b, h] = lse_bh[0]

            # Compute softmax attn vector for this head
            attn = torch.empty(L_b, dtype=torch.float32, device=device)
            compute_softmax_kernel_bh[(1, 1)](logits_flat, attn,
                                              L_b=L_b)

            # Compute final output vector: out[h, :] = attn @ Kc_b
            out_vec = torch.empty(Dc, dtype=torch.float32, device=device)
            compute_out_kernel_bh[(1, 1)](attn, Kc_b, out_vec,
                                          Dc=Dc, L_b=L_b)
            output[b, h] = out_vec

    # Cast output to bfloat16 to match original
    output = output.to(torch.bfloat16)
    return output, lse
```
