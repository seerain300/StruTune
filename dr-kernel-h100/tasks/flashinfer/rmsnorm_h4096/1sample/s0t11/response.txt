```python
import torch
import triton
import triton.language as tl


# Reduction kernel: per-row inv_rms = rsqrt(mean(x^2) + EPS)
@triton.jit
def _row_rms_inv_kernel(
    hidden_ptr,           # *const fp16/bf16/fp32, shape [B, H]
    inv_rms_ptr,          # *float32, shape [B]
    B: tl.int32,
    H: tl.int32,
    EPS: tl.float32,
    hidden_stride_row: tl.int32,  # stride along rows (elements)
    hidden_stride_col: tl.int32,  # stride along cols (elements)
    BLOCK_SIZE: tl.constexpr,
):
    row = tl.program_id(0)
    if row >= B:
        return

    sum_sq = 0.0
    col = 0
    while col < H:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < H
        x = tl.load(hidden_ptr + row * hidden_stride_row + offs * hidden_stride_col, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_sq / H
    inv_rms = tl.rsqrt(mean + EPS)
    tl.store(inv_rms_ptr + row, inv_rms)


# Elementwise apply kernel: out[row, col] = hidden[row, col] * inv_rms[row] * weight[col]
@triton.jit
def _apply_two_scales_kernel(
    hidden_ptr,           # *fp16/bf16/fp32, shape [B, H]
    inv_rms_ptr,          # *float32, shape [B]
    weight_ptr,           # *float32, shape [H]
    out_ptr,              # *float32, shape [B, H]
    B: tl.int32,
    H: tl.int32,
    W: tl.int32,          # weight length (should equal H)
    hidden_stride_row: tl.int32,
    hidden_stride_col: tl.int32,
    out_stride_row: tl.int32,
    out_stride_col: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    row = tl.program_id(0)
    col_block = tl.program_id(1)
    if row >= B:
        return

    col_start = col_block * BLOCK_SIZE
    offs = col_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < H

    # Load row slice and corresponding weight slice
    x = tl.load(hidden_ptr + row * hidden_stride_row + offs * hidden_stride_col, mask=mask, other=0.0)
    # Compute in float32
    x = x.to(tl.float32)
    inv_r = tl.load(inv_rms_ptr + row)
    w = tl.load(weight_ptr + offs, mask=mask, other=1.0)

    y = x * inv_r * w  # elementwise multiply

    tl.store(out_ptr + row * out_stride_row + offs * out_stride_col, y, mask=mask)


def run_triton(hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    # Ensure inputs are on CUDA and contiguous
    hidden = hidden_states.contiguous()
    weight = weight.contiguous()

    B, H = hidden.shape
    # Constants
    EPS = 1e-5

    # Allocate per-row inv_rms in fp32
    inv_rms = torch.empty(B, dtype=torch.float32, device=hidden.device)

    # Launch reduction kernel: one program per row
    BLOCK_SIZE_R = 1024  # good default for 4096-wide rows
    grid_reduce = (B,)
    _row_rms_inv_kernel[grid_reduce](
        hidden, inv_rms,
        B, H, EPS,
        hidden.stride(0), hidden.stride(1),
        BLOCK_SIZE=BLOCK_SIZE_R,
        num_warps=8, num_stages=2
    )

    # Choose elementwise tile and warps based on H
    if H >= 4096:
        BLOCK_SIZE_E = 4096
        num_warps_e = 16
    elif H >= 2048:
        BLOCK_SIZE_E = 2048
        num_warps_e = 16
    else:
        BLOCK_SIZE_E = 1024
        num_warps_e = 8

    # Allocate output in fp32 for compute
    out_f32 = torch.empty((B, H), dtype=torch.float32, device=hidden.device)

    # Launch elementwise apply: 2D grid over rows and column blocks
    grid_apply = (B, triton.cdiv(H, BLOCK_SIZE_E))
    _apply_two_scales_kernel[grid_apply](
        hidden, inv_rms, weight.to(torch.float32), out_f32,
        B, H, H,  # W=H
        hidden.stride(0), hidden.stride(1),
        out_f32.stride(0), out_f32.stride(1),
        BLOCK_SIZE=BLOCK_SIZE_E,
        num_warps=num_warps_e, num_stages=2
    )

    # Cast back to original dtype
    return out_f32.to(hidden_states.dtype)
```

- The Triton kernels now have explicit, correct argument lists, including shape and strides. The previous CompilationError was due to missing parameters and referencing non-existent strides inside the kernel.
- Indexing uses passed-in strides (hidden.stride(0), hidden.stride(1), etc.), avoiding invalid pointer arithmetic. Masks ensure no out-of-bounds accesses.
- Reduction and elementwise kernels compute in float32, preserving numerical stability. The final result is cast to the original hidden_states dtype, matching the original PyTorch implementation.
