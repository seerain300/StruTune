import torch
import triton
import triton.language as tl


# Triton kernel: RMSNorm per row
# y = weight * x / sqrt(mean(x^2) + eps)
@triton.jit
def triton_rmsnorm(x_ptr, y_ptr, norm_weight_ptr, B, H, L, D, eps, stride_b, stride_h, stride_l, stride_d, BLOCK: tl.constexpr):
    row_id = tl.program_id(0)  # over B*H*L rows
    BL = H * L
    b = row_id // BL
    rem = row_id % BL
    h = rem // L
    l = rem % L

    base = b * stride_b + h * stride_h + l * stride_l

    # Compute sum of squares over D
    sum_sq = 0.0
    for start in range(0, D, BLOCK):
        idx = start + tl.arange(0, BLOCK)
        mask = idx < D
        x = tl.load(x_ptr + base + idx * stride_d, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)

    mean = sum_sq / D
    scale = tl.rsqrt(mean + eps)
    inv_scale = scale  # scalar
    for start in range(0, D, BLOCK):
        idx = start + tl.arange(0, BLOCK)
        mask = idx < D
        x = tl.load(x_ptr + base + idx * stride_d, mask=mask, other=0.0)
        w = tl.load(norm_weight_ptr + idx, mask=mask, other=0.0)
        y = (x.to(tl.float32) * inv_scale) * w.to(tl.float32)
        tl.store(y_ptr + base + idx * stride_d, y, mask=mask)


# Triton kernel: build cos/sin for rotation using small-angle approximations
# emb[:half] = pos * inv_freq; emb[half:] = emb[:half]
# cos[i] = 1 - emb[i]^2 / 2; sin[i] = emb[i] - emb[i]^3 / 6
@triton.jit
def triton_build_cos_sin(pos, inv_freq_ptr, cos_ptr, sin_ptr, half, BLOCK: tl.constexpr):
    # We produce vectors of length 2*half (head_dim)
    for start in range(0, 2 * half, BLOCK):
        idx = start + tl.arange(0, BLOCK)
        mask = idx < (2 * half)
        # For emb[:half], alpha = pos * inv_freq[i]
        alpha = pos * tl.load(inv_freq_ptr + (idx < half) * (idx - half))  # only meaningful for idx < half
        # We need to load inv_freq at positions idx % half. Triton doesn't support vectorized indexing; compute alpha via scalar broadcasting:
        # Since we iterate over idx, for idx < half: alpha = pos * inv_freq[idx]; for idx >= half: alpha = pos * inv_freq[idx-half]
        alpha = tl.where(idx < half, pos * tl.load(inv_freq_ptr + idx), pos * tl.load(inv_freq_ptr + (idx - half)))
        emb = alpha
        cos_val = 1.0 - emb * emb * 0.5
        sin_val = emb - emb * emb * emb * (1.0 / 6.0)
        tl.store(cos_ptr + idx, cos_val, mask=mask)
        tl.store(sin_ptr + idx, sin_val, mask=mask)


# Triton kernel: apply rotation to x using cos/sin (head_dim = 128)
@triton.jit
def triton_apply_rotation(x_ptr, y_ptr, cos_ptr, sin_ptr, D, stride_b, stride_h, stride_l, stride_d, BLOCK: tl.constexpr):
    row_id = tl.program_id(0)  # over B*H*L rows
    BL = H * L
    b = row_id // BL
    rem = row_id % BL
    h = rem // L
    l = rem % L

    base = b * stride_b + h * stride_h + l * stride_l

    half = D // 2
    # First half: y1 = x1 * cos - x2 * sin
    for i in range(0, half):
        idx1 = i
        idx2 = i + half
        x1 = tl.load(x_ptr + base + idx1 * stride_d, mask=(idx1 < half), other=0.0)
        x2 = tl.load(x_ptr + base + idx2 * stride_d, mask=(idx2 < half), other=0.0)
        c = tl.load(cos_ptr + idx1, mask=(idx1 < half), other=0.0)
        s = tl.load(sin_ptr + idx1, mask=(idx1 < half), other=0.0)
        x1_f32 = x1.to(tl.float32)
        x2_f32 = x2.to(tl.float32)
        c_f32 = c.to(tl.float32)
        s_f32 = s.to(tl.float32)
        y1 = x1_f32 * c_f32 - x2_f32 * s_f32
        tl.store(y_ptr + base + idx1 * stride_d, y1, mask=(idx1 < half))

    # Second half: y2 = x2 * cos + x1 * sin
    for i in range(0, half):
        idx1 = i
        idx2 = i + half
        x1 = tl.load(x_ptr + base + idx1 * stride_d, mask=(idx1 < half), other=0.0)
        x2 = tl.load(x_ptr + base + idx2 * stride_d, mask=(idx2 < half), other=0.0)
        c = tl.load(cos_ptr + idx1, mask=(idx1 < half), other=0.0)
        s = tl.load(sin_ptr + idx1, mask=(idx1 < half), other=0.0)
        x1_f32 = x1.to(tl.float32)
        x2_f32 = x2.to(tl.float32)
        c_f32 = c.to(tl.float32)
        s_f32 = s.to(tl.float32)
        y2 = x2_f32 * c_f32 + x1_f32 * s_f32
        tl.store(y_ptr + base + idx2 * stride_d, y2, mask=(idx2 < half))
