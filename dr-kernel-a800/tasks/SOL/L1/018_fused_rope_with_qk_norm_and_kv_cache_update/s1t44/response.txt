```python
import triton
import triton.language as tl


@triton.jit
def rmsnorm_row_kernel(
    x_ptr,         # *const T, input tensor
    w_ptr,         # *const T, weight (length D)
    y_ptr,         # *T, output tensor
    B,             # int32, number of rows (flattened)
    D,             # int32, head_dim
    eps,           # float32
    BLOCK: tl.constexpr,
):
    # Each program handles one row
    row = tl.program_id(0)
    if row >= B:
        return

    # Compute indices along head_dim
    idx = tl.arange(0, BLOCK)
    mask = idx < D

    # Load x row and weight, cast to fp32 for accumulation
    x = tl.load(x_ptr + row * D + idx, mask=mask, other=0.0)
    w = tl.load(w_ptr + idx, mask=mask, other=1.0)
    x32 = x.to(tl.float32)
    w32 = w.to(tl.float32)

    # Compute mean of squares
    sq = x32 * x32
    mean = tl.sum(sq, axis=0) / D

    # Compute scale
    inv_scale = 1.0 / tl.sqrt(mean + eps)

    # Normalize and apply weight, then cast back to original dtype
    y32 = w32 * x32 * inv_scale
    y = y32.to(x.dtype)

    # Store result
    tl.store(y_ptr + row * D + idx, y, mask=mask)

    def forward(self, query, key, value, position_ids, key_cache, value_cache, cache_position, q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps):
        # We implement only RMSNorm in Triton; avoid torch ops in host.
        # Compute shapes
        B, num_q_heads, seq_len, D = query.shape

        # Allocate outputs
        query_norm = torch.empty_like(query)
        key_norm = torch.empty_like(key)

        # Launch Triton kernel for query RMSNorm
        B_rows = B * num_q_heads * seq_len
        grid_query = (B_rows,)
        # Ensure contiguous along last dim for simple indexing
        x_q = query
        w_q = q_norm_weight
        y_q = query_norm
        rmsnorm_row_kernel[grid_query](
            x_q, w_q, y_q,
            B_rows, D, float(rms_norm_eps),
            BLOCK=D,
            num_warps=4,
        )

        # Launch Triton kernel for key RMSNorm
        Bk, num_kv_heads, _, _ = key.shape
        assert Bk == B and num_kv_heads == key.shape[1]  # sanity
        B_rows_k = B * num_kv_heads * seq_len
        grid_key = (B_rows_k,)
        x_k = key
        w_k = k_norm_weight
        y_k = key_norm
        rmsnorm_row_kernel[grid_key](
            x_k, w_k, y_k,
            B_rows_k, D, float(rms_norm_eps),
            BLOCK=D,
            num_warps=4,
        )

        # Return normalized query and key, and original caches to match original signature
        return query_norm, key_norm, key_cache, value_cache
```
