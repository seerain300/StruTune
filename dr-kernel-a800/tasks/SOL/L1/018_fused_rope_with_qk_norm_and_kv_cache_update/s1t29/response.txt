import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_row_kernel(
    x_ptr,  # *const T
    out_ptr,  # *mut T
    weight_ptr,  # *const T
    B: tl.constexpr,
    H: tl.constexpr,
    L: tl.constexpr,
    D: tl.constexpr,
    stride_b: tl.int32,
    stride_h: tl.int32,
    stride_l: tl.int32,
    stride_d: tl.int32,
    eps: tl.float32,
    BLOCK: tl.constexpr,
):
    # Each program handles one "row" = one (batch, head, position)
    row_id = tl.program_id(0)
    # Compute b, h, l from row_id
    # row_id ranges from 0 to B*H*L - 1
    l = row_id % L
    tmp = row_id // L
    h = tmp % H
    b = tmp // H

    base = b * stride_b + h * stride_h + l * stride_l

    # Accumulate sum of squares over D (head_dim)
    sum_sq = 0.0
    for idx in range(0, BLOCK):
        x_val = tl.load(x_ptr + base + idx * stride_d, mask=(idx < D), other=0.0)
        x_f32 = x_val.to(tl.float32)
        sum_sq += x_f32 * x_f32

    mean = sum_sq / D
    inv_scale = tl.rsqrt(mean + eps)  # 1/sqrt(mean + eps)

    # Apply RMSNorm and weight
    for idx in range(0, BLOCK):
        x_val = tl.load(x_ptr + base + idx * stride_d, mask=(idx < D), other=0.0)
        w_val = tl.load(weight_ptr + idx, mask=(idx < D), other=0.0)
        y = (x_val.to(tl.float32) * inv_scale) * w_val.to(tl.float32)
        tl.store(out_ptr + base + idx * stride_d, y, mask=(idx < D))

    """
    Compute y = weight * x / sqrt(mean(x^2) + eps) per row using Triton.
    x: [B, H, L, D], dtype bfloat16 or float32 (we convert to float32 for computation)
    weight: [D], same dtype as x (we convert to float32 for computation)
    Returns y with same shape/dtype as x.
    """

    assert x.is_contiguous(), "x must be contiguous for this kernel."
    assert weight.is_cuda, "weight must be CUDA tensor."
    assert weight.is_contiguous(), "weight must be contiguous."

    y = torch.empty_like(x)

    # Strides in elements for [B, H, L, D] contiguous layout
    stride_b = H * L * D
    stride_h = L * D
    stride_l = D
    stride_d = 1

    # One program per row
    grid = (B * H * L,)

    # Launch kernel; BLOCK must be >= D; D is known at shape time.
    rmsnorm_row_kernel[grid](
        x, y, weight,
        B, H, L, D,
        stride_b, stride_h, stride_l, stride_d,
        eps,
        BLOCK=D,  # use full head_dim
        num_warps=1,
        num_stages=1,
    )
    return y
