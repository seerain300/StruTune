Root-cause analysis and plan

- The original PyTorch forward does:
  1) RMSNorm on query and key along the last dimension D=128 (per [B, H, S] row), epsilon eps=1e-6. Compute in fp32, output in original dtype (bf16).
  2) Rotation using position_ids and inv_freq:
     - For each position s (per batch), angle = position_ids[b, s] * inv_freq where inv_freq length is D//2=64.
     - For query: cos_all = cos(angle) cat with itself to length D; sin_all = sin(angle) cat with itself.
     - For key: use sin_all (keys use sin-based rotation).
     - Rotate: y = x * cos - rotate_half(x) * sin, rotate_half(x) = [-x[..., D//2:], x[..., :D//2]].
  3) Cache update: write rotated keys into key_cache at cache_position indices for each batch and kv head; write original values into value_cache at the same indices.

- What went wrong previously:
  - The earlier Triton RMSNorm kernel processed only 128 columns per iteration without iterating over D, which caused incomplete computation on some shapes (e.g., B=1, S=1, H=8). This led to incorrect outputs.

  - Implement RMSNorm Triton kernel that loops over D in chunks of BLOCK_SIZE using a while loop. Use grid over (B, H, S) with BLOCK_SIZE=128. This ensures we cover all columns of the 4D tensor and prevents under-computation across different shapes.

import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_kernel(
    x_ptr, out_ptr, weight_ptr,
    B, H, S, D,
    x_stride_b, x_stride_h, x_stride_s, x_stride_d,
    out_stride_b, out_stride_h, out_stride_s, out_stride_d,
    eps,
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles one row: (b, h, s)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    # First pass: compute sum of squares across D
    sumsq = 0.0
    off_d = 0
    while off_d < D:
        offs = off_d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
        x_vals = tl.load(x_row_ptr + offs * x_stride_d, mask=mask, other=0.0)
        x_vals_fp32 = x_vals.to(tl.float32)
        sumsq += tl.sum(x_vals_fp32 * x_vals_fp32, axis=0)
        off_d += BLOCK_SIZE

    D_fp32 = tl.full((), D, tl.float32)
    mean = sumsq / D_fp32
    inv_rms = tl.rsqrt(mean + eps)

    # Second pass: write normalized and scaled output
    off_d = 0
    while off_d < D:
        offs = off_d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
        w_row_ptr = weight_ptr + offs
        x_vals = tl.load(x_row_ptr + offs * x_stride_d, mask=mask, other=0.0)
        w_vals = tl.load(w_row_ptr, mask=mask, other=1.0)  # weight; out-of-range masked
        x_vals_fp32 = x_vals.to(tl.float32)
        w_vals_fp32 = w_vals.to(tl.float32)

        out_row_ptr = out_ptr + b * out_stride_b + h * out_stride_h + s * out_stride_s
        tl.store(out_row_ptr + offs * out_stride_d, y_fp32, mask=mask)
        off_d += BLOCK_SIZE


@triton.jit
def rotation_kernel(
    x_ptr, out_ptr, pos_ptr, invfreq_ptr,
    B, H, S, D,
    x_stride_b, x_stride_h, x_stride_s, x_stride_d,
    out_stride_b, out_stride_h, out_stride_s, out_stride_d,
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles one row: (b, h, s)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    # Load position for this (b, s)
    pos = tl.load(pos_ptr + b * S + s)

    # Build cos_all and sin_all vectors in fp32
    cos_all = tl.zeros((D,), dtype=tl.float32)
    sin_all = tl.zeros((D,), dtype=tl.float32)

    # First half [0:D//2)
    half = D // 2
    off = 0
    while off < half:
        offs = off + tl.arange(0, BLOCK_SIZE)
        mask = offs < half
        angle = pos * tl.load(invfreq_ptr + offs, mask=mask, other=0.0)
        c = tl.cos(angle)  # cos(angle)
        s2 = tl.sin(angle) # sin(angle)
        # assign cos to first half, sin to first half (we'll expand to full D in next step)
        cos_all[offs] = c
        sin_all[offs] = s2
        off += BLOCK_SIZE

    # Expand to full D:
    # First half: cos_all[:half] is already filled; second half: copy first half
    # Second half: sin_all[:half] is already filled; first half: keep sin_all[:half]
    # But Triton doesn't support direct indexing like cos_all[offs] = value; we instead construct
    # full vectors using tl.concatenate. However, Triton requires loops or vectorized ops.
    # We'll construct cos_all_full and sin_all_full by expanding:
    # cos_all_full = concatenate([cos_all[:half], cos_all[:half]])
    # sin_all_full = concatenate([sin_all[:half], sin_all[:half]])
    # In Triton, we can only create vectors with tl.arange; to fill, we use two while loops.

    # Allocate full vectors via concatenation: Triton supports building via loops.
    # We'll fill using offs in [0, D) with two passes:
    off = 0
    while off < half:
        offs = off + tl.arange(0, BLOCK_SIZE)
        mask = offs < half
        cos_all[offs + half] = tl.load(cos_all + offs)  # copy first half to second half
        off += BLOCK_SIZE

    # For sin_all, second half is already sin_all[:half]; no need to copy.

    # Now apply rotation: out = x * cos_all - rotate_half(x) * sin_all
    off_d = 0
    while off_d < D:
        offs = off_d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
        out_row_ptr = out_ptr + b * out_stride_b + h * out_stride_h + s * out_stride_s
        x_vals = tl.load(x_row_ptr + offs * x_stride_d, mask=mask, other=0.0)
        x_fp32 = x_vals.to(tl.float32)
        # Build rotate_half(x)
        # For offs in [0, D), rotate_half(x)[offs] = -x[D//2 + offs - (D//2)] when offs >= D//2, else 0.
        # But simpler: we can compute using indexing: for i in [0..D-1], other_i = D-1-i
        # Here, since Triton doesn't support vectorized reverse indexing, we compute by splitting:
        # For first half, take x[offs]; for second half, take -x[offs-half].
        # However, Triton doesn't support dynamic indexing; we'll instead use a temporary vector
        # that holds x for both halves. Instead, we compute rotate_half via two loads:
        # Let half2 = D//2. For i in [0..half2-1], rh[i] = -x[half+i]; for i in [half2..D-1], rh[i] = x[i-half2].
        # Implement via two while loops:
        rh = tl.zeros((D,), dtype=tl.float32)
        # first half of rotated: -x[1st half]
        off_rh1 = 0
        while off_rh1 < half:
            idxs = off_rh1 + tl.arange(0, BLOCK_SIZE)
            mask_rh1 = idxs < half
            x1 = tl.load(x_row_ptr + idxs * x_stride_d, mask=mask_rh1, other=0.0)
            rh[idxs] = (-x1.to(tl.float32))
            off_rh1 += BLOCK_SIZE
        # second half of rotated: x[2nd half]
        off_rh2 = half
        while off_rh2 < D:
            idxs = off_rh2 + tl.arange(0, BLOCK_SIZE)
            mask_rh2 = idxs < D
            x2 = tl.load(x_row_ptr + (idxs - half) * x_stride_d, mask=mask_rh2, other=0.0)
            rh[idxs] = x2.to(tl.float32)
            off_rh2 += BLOCK_SIZE

        # Now out = x * cos_all - rh * sin_all
        cos_vec = tl.load(cos_all + offs, mask=mask, other=0.0)
        sin_vec = tl.load(sin_all + offs, mask=mask, other=0.0)
        out_vals_fp32 = x_fp32 * cos_vec - rh[offs] * sin_vec
        tl.store(out_row_ptr + offs * out_stride_d, out_vals_fp32, mask=mask)

    src_ptr, dst_ptr,
    B, H, S, D, L,
    src_stride_b, src_stride_h, src_stride_s, src_stride_d,
    dst_stride_b, dst_stride_h, dst_stride_s, dst_stride_d,
    cache_pos_ptr,
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles one row: (b, h, s)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    pos = tl.load(cache_pos_ptr + b * S + s)  # cache_position[s]

    # Loop over D to copy src row into dst at (b, h, pos)
    off_d = 0
    while off_d < D:
        offs = off_d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        src_row_ptr = src_ptr + b * src_stride_b + h * src_stride_h + s * src_stride_s
        dst_row_ptr = dst_ptr + b * dst_stride_b + h * dst_stride_h + pos * dst_stride_s
        vals = tl.load(src_row_ptr + offs * src_stride_d, mask=mask, other=0.0)
        tl.store(dst_row_ptr + offs * dst_stride_d, vals, mask=mask)
        off_d += BLOCK_SIZE
