```python
import torch
import triton
import triton.language as tl

# Triton kernel: RMSNorm over the last dimension (head_dim=128) for each row.
# Input X is [rows, 128] fp32, Out is [rows, 128] fp32.
@triton.jit
def rmsnorm_rows_kernel(X_ptr, Out_ptr, rows, head_dim, eps: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    sumsq = 0.0
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / head_dim
    r = tl.rsqrt(mean + eps)
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        y = x * r
        tl.store(Out_ptr + row_id * head_dim + offs, y, mask=mask)

# Triton kernel: compute cos and sin for first half (64 dims) given positions.
# Inputs:
#   pos_ptr: int64 array [rows]
#   inv_ptr: float32 array [64] (inv_freq[:64])
#   cos_out: float32 [rows, 64]
#   sin_out: float32 [rows, 64]
@triton.jit
def compute_cos_sin_first_half_kernel(pos_ptr, inv_ptr, cos_out, sin_out,
                                      rows: tl.constexpr, half_dim: tl.constexpr,
                                      BLOCK_SIZE: tl.constexpr):
    row = tl.program_id(0)
    if row >= rows:
        return
    for col in range(0, half_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < half_dim
        pos = tl.load(pos_ptr + row)
        angle = pos.to(tl.float32) * inv_ptr[offs]
        c = tl.cos(angle)
        s = tl.sin(angle)
        tl.store(cos_out + row * half_dim + offs, c, mask=mask)
        tl.store(sin_out + row * half_dim + offs, s, mask=mask)

# Triton kernel: apply full 128-dim rotation per row using cos/sin for first half.
# Inputs:
#   x_in: fp32 [rows, 128]
#   cos_ptr: fp32 [rows, 64] (first half)
#   sin_ptr: fp32 [rows, 64] (first half)
#   y_out: fp32 [rows, 128]
@triton.jit
def rotate_full_128_kernel(x_in, cos_ptr, sin_ptr, y_out,
                           rows: tl.constexpr):
    row = tl.program_id(0)
    if row >= rows:
        return
    # First half: 0..63
    for i in range(64):
        angle = cos_ptr[row, i]  # cos[i]
        sgn = sin_ptr[row, i]    # sin[i]
        x0 = tl.load(x_in + row * 128 + i)
        x64 = tl.load(x_in + row * 128 + i + 64)
        y0 = x0 * angle + x64 * sgn
        tl.store(y_out + row * 128 + i, y0)
    # Second half: 64..127, mapped from first half
    for i in range(64):
        angle = cos_ptr[row, i]  # cos[i]
        sgn = sin_ptr[row, i]    # sin[i]
        x64 = tl.load(x_in + row * 128 + i + 64)
        x0 = tl.load(x_in + row * 128 + i)
        y64 = x64 * angle - x0 * sgn
        tl.store(y_out + row * 128 + i + 64, y64)

def _launch_rmsnorm(x_fp32, out_fp32, rows, eps):
    head_dim = x_fp32.shape[1]
    grid = (rows,)
    rmsnorm_rows_kernel[grid](x_fp32, out_fp32, rows, head_dim, eps, BLOCK_SIZE=128, num_warps=4)

def _launch_compute_cos_sin(pos, inv_freq_half, cos_out, sin_out):
    rows = pos.numel()
    half_dim = inv_freq_half.numel()
    grid = (rows,)
    compute_cos_sin_first_half_kernel[grid](pos.to(torch.int64), inv_freq_half.to(torch.float32),
                                            cos_out, sin_out, rows, half_dim, BLOCK_SIZE=128, num_warps=4)

def _launch_rotate_full_128(x_fp32, cos_ptr, sin_ptr, y_fp32, rows):
    grid = (rows,)
    rotate_full_128_kernel[grid](x_fp32, cos_ptr, sin_ptr, y_fp32, rows, num_warps=4)

class ModelNew(torch.nn.Module):
    def forward(self, query, key, value, position_ids, key_cache, value_cache, cache_position, q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps):
        # Shapes:
        # query: [Bq, Hq, Tq, 128]
        # key:   [Bk, Hk, Tk, 128]
        # value: [B, S, D] (not used in this function; original code also didn't use it)
        Bq, Hq, Tq, Dq = query.shape
        Bk, Hk, Tk, Dk = key.shape
        assert Dq == 128 and Dk == 128, "head_dim must be 128"
        inv_freq = inv_freq.to(torch.float32)

        # 1) RMSNorm for query (fp32 compute), then rotation
        query_rows = Bq * Hq * Tq
        query_in_fp32 = query.to(torch.float32)                  # [rows_q, 128]
        query_norm_fp32 = torch.empty_like(query_in_fp32)        # [rows_q, 128]
        _launch_rmsnorm(query_in_fp32, query_norm_fp32, query_rows, rms_norm_eps)

        # 2) Compute cos/sin for query positions (first half 64)
        pos_q = position_ids[:, :Tq].reshape(-1).to(torch.int64) # [Bq*Tq]
        cos_q = torch.empty((pos_q.numel(), 64), dtype=torch.float32, device=query.device)
        sin_q = torch.empty((pos_q.numel(), 64), dtype=torch.float32, device=query.device)
        _launch_compute_cos_sin(pos_q, inv_freq[:64], cos_q, sin_q)

        # 3) Apply rotation to query_norm -> query_rotated (fp32 compute)
        query_rotated_fp32 = torch.empty_like(query_norm_fp32)
        _launch_rotate_full_128(query_norm_fp32, cos_q, sin_q, query_rotated_fp32, query_rows)

        # Cast to bf16 to match original return type for query_rotated
        query_rotated = query_rotated_fp32.to(torch.bfloat16).reshape(Bq, Hq, Tq, 128)

        # Repeat for key (key_cache and value_cache are returned unchanged)
        key_rows = Bk * Hk * Tk
        key_in_fp32 = key.to(torch.float32)                     # [rows_k, 128]
        key_norm_fp32 = torch.empty((key_rows, 128), dtype=torch.float32, device=key.device)
        _launch_rmsnorm(key_in_fp32, key_norm_fp32, key_rows, rms_norm_eps)

        pos_k = position_ids[:, :Tk].reshape(-1).to(torch.int64)  # [Bk*Tk]
        cos_k = torch.empty((pos_k.numel(), 64), dtype=torch.float32, device=key.device)
        sin_k = torch.empty((pos_k.numel(), 64), dtype=torch.float32, device=key.device)
        _launch_compute_cos_sin(pos_k, inv_freq[:64], cos_k, sin_k)

        key_rotated_fp32 = torch.empty((key_rows, 128), dtype=torch.float32, device=key.device)
        _launch_rotate_full_128(key_norm_fp32, cos_k, sin_k, key_rotated_fp32, key_rows)

        key_rotated = key_rotated_fp32.to(torch.bfloat16).reshape(Bk, Hk, Tk, 128)

        # Return as original: (query_rotated, key_rotated, key_cache, value_cache)
        return query_rotated, key_rotated, key_cache, value_cache
```
