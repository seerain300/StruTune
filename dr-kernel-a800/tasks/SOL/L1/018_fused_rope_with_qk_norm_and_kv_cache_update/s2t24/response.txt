import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel_4d(X_ptr, Y_ptr, M, D, eps, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: RMSNorm across last dimension D for a 4D tensor of shape (M, D),
    where M = B * N * S. Each program handles one row (one token).
    Y_ptr must be of shape (M, D). X_ptr and Y_ptr can represent slices of a
    larger 4D tensor via linear indexing: row_id * D + d.
    """
    row_id = tl.program_id(axis=0)
    if row_id >= M:
        return
    # Compute sum of squares in fp32
    sum_sq = 0.0
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)
    # Write normalized output
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        y = (x_f32 / r).to(x.dtype)
        tl.store(Y_ptr + row_id * D + offs, y, mask=mask)

    cos_ptr, sin_ptr,

    """

    b = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    if b >= B or n >= N:
        return

    # Iterate over tokens s in this batch
    for s in range(0, S):
        # Load normalized key row and value row
        k_ptr = key_norm_ptr + b * (N * S * D) + n * (S * D) + s * D
        v_ptr = value_ptr + b * (N * S * D) + n * (S * D) + s * D

        # Load vectors for this token s (broadcast across D chunked)
        # We'll do rotation in chunks of BLOCK_SIZE over D=128
        D_half = D // 2  # 64
        # Load first half of key row
        offs1 = tl.arange(0, BLOCK_SIZE)
        mask1 = offs1 < D_half
        k1 = tl.load(k_ptr + offs1, mask=mask1, other=0.0)  # [BLOCK_SIZE]
        # Load second half of key row
        offs2 = tl.arange(0, BLOCK_SIZE) + D_half
        mask2 = offs2 < D
        k2 = tl.load(k_ptr + offs2, mask=mask2, other=0.0)  # [BLOCK_SIZE]

        # Load cos and sin for this s across D
        # cos_ptr[b, s, :] and sin_ptr[b, s, :]
        # For simplicity, treat as 1D over D chunks
        # Load first half cos/sin
        cos1 = tl.load(cos_ptr + b * (S * D) + s * D + offs1, mask=mask1, other=0.0)
        sin1 = tl.load(sin_ptr + b * (S * D) + s * D + offs1, mask=mask1, other=0.0)
        # Load second half cos/sin
        cos2 = tl.load(cos_ptr + b * (S * D) + s * D + offs2, mask=mask2, other=0.0)
        sin2 = tl.load(sin_ptr + b * (S * D) + s * D + offs2, mask=mask2, other=0.0)

        # Compute rotated halves in fp32
        k1_f32 = k1.to(tl.float32)
        k2_f32 = k2.to(tl.float32)
        cos1_f32 = cos1.to(tl.float32)
        sin1_f32 = sin1.to(tl.float32)
        cos2_f32 = cos2.to(tl.float32)
        sin2_f32 = sin2.to(tl.float32)

        y1 = k1_f32 * cos1_f32 + (-k2_f32) * sin1_f32
        y2 = k2_f32 * cos2_f32 + (-k1_f32) * sin2_f32
        y = tl.concatenate([y1, y2], axis=0).to(k_ptr.dtype.element_ty)  # cast back to original dtype

        # Write into key_cache and value_cache at position cache_position[s]
        pos = tl.load(cache_pos_ptr + s).to(tl.int32)
        key_cache_row_ptr = key_cache_ptr + b * (N * D) + n * D + pos * D
        value_cache_row_ptr = value_cache_ptr + b * (N * D) + n * D + pos * D

        # Store y into both caches

        # Copy original value row into value_cache (no rotation)
        v = tl.load(v_ptr, mask=offs1 < D_half, other=0.0)  # first half
        v = tl.concatenate([v, tl.load(v_ptr + D_half, mask=offs2 < D, other=0.0)], axis=0).to(v_ptr.dtype.element_ty)

import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel_4d(X_ptr, Y_ptr, M, D, eps, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: RMSNorm across last dimension D for a 4D tensor of shape (M, D),
    where M = B * N * S. Each program handles one row (one token). Writes normalized
    output to Y_ptr.
    """
    row_id = tl.program_id(axis=0)
    if row_id >= M:
        return
    sum_sq = 0.0
    # Reduction across D
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)
    # Scale and store
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        y = (x_f32 / r).to(x.dtype)
        tl.store(Y_ptr + row_id * D + offs, y, mask=mask)

def build_inv_kernel(inv_freq_ptr, inv_ptr, D_HALF, D, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: Build inv of length D from inv_freq of length D_HALF:
    inv[0:D_HALF] = inv_freq, inv[D_HALF:] = inv_freq.
    Stores to inv_ptr of length D.
    """
    # Single program, write entire inv
    for d in range(0, D_HALF):
        val = tl.load(inv_freq_ptr + d)
        tl.store(inv_ptr + d, val)
    for d in range(D_HALF, D):
        idx = d - D_HALF
        val = tl.load(inv_ptr + idx)
        tl.store(inv_ptr + d, val)


@triton.jit
def rotate_rows_kernel(X_ptr, Y_ptr, B, N, S, D, cos_ptr, sin_ptr, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: Rotate rows X_ptr of shape [B, N, S, D] to produce Y_ptr of shape [B, N, S, D].
    For each token s, compute inv = [inv_freq, inv_freq], then cos and sin for pos = position_ids[b, s],
    and apply rotation:
      split x into x1, x2 = x[..., :D//2], x[..., D//2:]
      y = x1 * cos + rotate_half(x) * sin, where rotate_half(x) = [-x2, x1].
    We expect position_ids to be provided implicitly via cos_ptr/sin_ptr which are per-token.
    Grid: (B, N, S).
    """
    b = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    if (b >= B) or (n >= N) or (s >= S):
        return

    base = b * (N * S * D) + n * (S * D) + s * D
    x_ptr = X_ptr + base
    y_ptr = Y_ptr + base

    D_HALF = D // 2  # 64

    # Load first half and second half
    offs1 = tl.arange(0, BLOCK_SIZE)
    offs2 = offs1 + D_HALF
    mask1 = offs1 < D_HALF
    mask2 = offs2 < D

    x1 = tl.load(x_ptr + offs1, mask=mask1, other=0.0)
    x2 = tl.load(x_ptr + offs2, mask=mask2, other=0.0)

    # Load cos and sin vectors for this token s
    # cos_ptr and sin_ptr are of shape [B, S, D], contiguous across D
    cos_ptr_s = cos_ptr + b * (S * D) + s * D
    sin_ptr_s = sin_ptr + b * (S * D) + s * D

    cos1 = tl.load(cos_ptr_s + offs1, mask=mask1, other=0.0)
    sin1 = tl.load(sin_ptr_s + offs1, mask=mask1, other=0.0)
    cos2 = tl.load(cos_ptr_s + offs2, mask=mask2, other=0.0)
    sin2 = tl.load(sin_ptr_s + offs2, mask=mask2, other=0.0)

    x1_f32 = x1.to(tl.float32)
    x2_f32 = x2.to(tl.float32)
    cos1_f32 = cos1.to(tl.float32)
    sin1_f32 = sin1.to(tl.float32)
    cos2_f32 = cos2.to(tl.float32)
    sin2_f32 = sin2.to(tl.float32)

    # Compute rotated parts
    y1 = x1_f32 * cos1_f32 + (-x2_f32) * sin1_f32
    y2 = x2_f32 * cos2_f32 + (-x1_f32) * sin2_f32

    # Combine halves
    # offs1 covers [0,63], offs2 covers [64,127]
    # Store y1 to y_ptr[0:64], y2 to y_ptr[64:128]
    tl.store(y_ptr + offs1, y1.to(x_ptr.dtype.element_ty), mask=mask1)
    tl.store(y_ptr + offs2, y2.to(x_ptr.dtype.element_ty), mask=mask2)
