import triton
import triton.language as tl

# Triton kernel: RMSNorm per row
# Input X_ptr: [NUM_ROWS, D], contiguous along D
# Weight_ptr: [D], 1D vector (bfloat16 or float32), will be cast to fp32 for compute
# Output Y_ptr: [NUM_ROWS, D]
@triton.jit
def rmsnorm_row_kernel(
    X_ptr, W_ptr, Y_ptr,
    NUM_ROWS: tl.constexpr, D: tl.constexpr, eps: tl.constexpr,
    stride_x_row, stride_x_col,
    stride_y_row, stride_y_col,
    row_id: tl.constexpr,
):
    # One program per row
    # Compute base pointers for this row
    x_row_ptr = X_ptr + row_id * stride_x_row
    y_row_ptr = Y_ptr + row_id * stride_y_row

    # Accumulate sum of squares across D in fp32
    sumsq = 0.0
    for col in range(0, D, 128):
        offs = col + tl.arange(0, 128)
        mask = offs < D
        x = tl.load(x_row_ptr + offs * stride_x_col, mask=mask, other=0.0)
        x32 = x.to(tl.float32)
        sumsq += tl.sum(x32 * x32, axis=0)

    mean = sumsq / D
    inv_scale = 1.0 / tl.sqrt(mean + eps)

    # Write normalized output: y = w * x * inv_scale
    for col in range(0, D, 128):
        offs = col + tl.arange(0, 128)
        mask = offs < D
        x = tl.load(x_row_ptr + offs * stride_x_col, mask=mask, other=0.0)
        w = tl.load(W_ptr + offs, mask=mask, other=1.0).to(tl.float32)  # weight is 1D
        y32 = w * x.to(tl.float32) * inv_scale
        # Cast back to original dtype of X (assumed same as Y); Triton will keep Y dtype from allocation
        y = y32.to(x.dtype)
        tl.store(y_row_ptr + offs * stride_y_col, y, mask=mask)

    stride_x_row, stride_x_col,
