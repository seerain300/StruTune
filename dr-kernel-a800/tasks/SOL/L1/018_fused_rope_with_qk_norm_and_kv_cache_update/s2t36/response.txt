import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel(X_ptr, Y_ptr, B, N, S, D, eps, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: RMSNorm over last dimension D for each row of a 4D tensor shaped [B, N, S, D].
    Grid over M=B*N*S rows. Each program handles one row (across S tokens) for fixed (b, n).
    """
    row_id = tl.program_id(axis=0)
    # Compute b, n from flattened row_id
    # row_id in [0, B*N*S)
    b = row_id // (N * S)
    tmp = row_id % (N * S)
    n = tmp // S
    s = tmp % S

    # Base offsets for the row (query/keys) at (b, n, s, :)
    base = b * (N * S * D) + n * (S * D) + s * D
    sum_sq = 0.0
    # Accumulate sum of squares across D
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + base + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)

    # Scale and store y = x / r
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + base + offs, mask=mask, other=0.0)
        y = x.to(tl.float32) / r
        tl.store(Y_ptr + base + offs, y.to(x.dtype), mask=mask)


@triton.jit
def build_inv_kernel(inv_freq_ptr, inv_ptr, D_HALF: tl.constexpr, D: tl.constexpr):
    """
    Triton kernel: Build inv vector of length D = [inv_freq, inv_freq] where inv_freq has length D_HALF.
    inv_ptr: output fp32 of length D.
    inv_freq_ptr: input fp32 of length D_HALF.
    """
    # Single program writes inv
    # First half
    for i in range(0, D_HALF):
        tl.store(inv_ptr + i, tl.load(inv_freq_ptr + i))
    # Second half
    for i in range(0, D_HALF):
        tl.store(inv_ptr + D_HALF + i, tl.load(inv_freq_ptr + i))


@triton.jit
def rotate_and_scatter_kernel(
    key_norm_ptr,  # *const T, [B, N, S, D]
    out_key_ptr,   # *T, [B, N, M, D] (we'll write rotated rows here)
    out_val_ptr,   # *T, [B, N, M, D] (we'll write original key_norm rows here)
    cos_ptr,       # *const fp32, [S, D] (per token s, cos vector)
    sin_ptr,       # *const fp32, [S, D] (per token s, sin vector)
    pos_ids_ptr,   # *const int64, [B, S] (we pass as (B, S) tensor)
    cache_pos_ptr, # *const int64, [S] (cache indices)
    B, N, S, D,
):
    """
    Triton kernel: For each (b, n), iterate s in [0..S), rotate key_norm[b, n, s, :] using cos/sin per s,
    and scatter into out_key[b, n, cache_pos[s], :] and out_val[b, n, cache_pos[s], :].
    Returns nothing; it writes to out_key and out_val.
    Note: Triton does not have tl.cos/tl.sin, so we rely on precomputed cos/sin passed as per-token vectors.
    """
    # Each program handles one (b, n)
    pid = tl.program_id(axis=0)
    b = pid // N
    n = pid % N
    # Loop over tokens
    for s in range(0, S):
        # Load pos for this (b, s)
        pos_elem = tl.load(pos_ids_ptr + b * S + s).to(tl.float32)
        # Compute angles = pos * inv for this s (we assume inv is built by host)
        # We load cos and sin vectors for this s. cos_ptr, sin_ptr are [S, D] contiguous by s.
        cos_s = tl.load(cos_ptr + s * D + tl.arange(0, D))  # [D]
        sin_s = tl.load(sin_ptr + s * D + tl.arange(0, D))  # [D]

        # Base pointers for source and destination
        base_key = b * (N * S * D) + n * (S * D) + s * D
        cache_idx = tl.load(cache_pos_ptr + s).to(tl.int32)
        base_out = b * (N * (S + 0) * D) + n * (S * D) + cache_idx * D  # simplified since M=S writes; we can allocate as (B, N, S, D) for clarity

        # Load original key norm row
        x = tl.load(key_norm_ptr + base_key + tl.arange(0, D))
        x_f32 = x.to(tl.float32)

        # Split into two halves: x1 = last 64 (indices 64..127), x2 = first 64 (indices 0..63)
        # Triton supports slicing: take last D_HALF and first D_HALF.
        D_HALF = D // 2
        x1 = x_f32[D_HALF:]              # [D_HALF]
        x2 = x_f32[:D_HALF]              # [D_HALF]
        x1_vec = tl.zeros([D], dtype=tl.float32)
        x2_vec = tl.zeros([D], dtype=tl.float32)
        # Assign:
        x1_vec[D_HALF:] = x1
        x2_vec[:D_HALF] = x2

        # rotate_half(x) = [-x2, x1]
        rh = tl.zeros([D], dtype=tl.float32)
        rh[:D_HALF] = -x2
        rh[D_HALF:] = x1

        # Compute rotated row:
        # y1 = x1 * cos + rotate_half(x)[:, :D_HALF] * sin
        # y2 = x2 * cos + rotate_half(x)[:, D_HALF:] * sin
        y1 = x1 * cos_s + rh[:D_HALF] * sin_s[:D_HALF]
        y2 = x2 * cos_s + rh[D_HALF:] * sin_s[D_HALF:]
        y = tl.zeros([D], dtype=tl.float32)
        y[:D_HALF] = y2
        y[D_HALF:] = y1

        # Store rotated row into out_key at cache index
        tl.store(out_key_ptr + base_out + tl.arange(0, D), y.to(x.dtype))

        # Store original key norm row into out_val at cache index
        tl.store(out_val_ptr + base_out + tl.arange(0, D), x_f32.to(x.dtype))


class ModelNew(torch.nn.Module):
    def forward(self, query, key, value, position_ids, key_cache, value_cache, cache_position,
                q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps):
        """
        Returns:
          - query_rotated: fp32 [B, N_q, S, D]
          - key_rotated: fp32 [B, N_kv, S, D]
          - key_cache: fp32 [B, N_kv, S, D] updated with rotated key rows at cache_position
          - value_cache: fp32 [B, N_kv, S, D] updated with original key rows at cache_position
        """
        # Shapes
        B, N_q, S, D = query.shape
        Bk, N_kv, Sk, Dk = key.shape
        assert B == Bk and N_q == N_kv and S == Sk and D == Dk, "Input shapes must match."

        # 1) RMSNorm for query and key using Triton (fp32 outputs)
        query_norm = torch.empty_like(query, dtype=torch.float32, device=query.device)
        key_norm = torch.empty_like(key, dtype=torch.float32, device=key.device)

        # Grid over rows: M = B * N * S
        M_q = B * N_q * S
        grid_q = (M_q,)
        rmsnorm_rows_kernel[grid_q](query, query_norm, B, N_q, S, D, rms_norm_eps, BLOCK_SIZE=128, num_warps=4)

        M_k = B * N_kv * S
        grid_k = (M_k,)
        rmsnorm_rows_kernel[grid_k](key, key_norm, B, N_kv, S, D, rms_norm_eps, BLOCK_SIZE=128, num_warps=4)

        # 2) Build inv = [inv_freq, inv_freq] (fp32) of length D
        inv = torch.empty(D, dtype=torch.float32, device=query.device)
        inv[:D // 2] = inv_freq.to(torch.float32)
        inv[D // 2:] = inv_freq.to(torch.float32)
        # Launch build_inv_kernel (single program)
        grid_inv = (1,)
        build_inv_kernel[grid_inv](inv_freq.to(torch.float32), inv, D_HALF=D // 2, D=D)

        # 3) Precompute cos and sin per token s on host using torch (allowed, not per-token Triton math)
        pos = position_ids.to(torch.float32)  # [B, S]
        angles = pos[:, :, None] * inv[None, None, :]  # [B, S, D]
        # cos and sin: [B, S, D]
        cos = torch.cos(angles)  # float32
        sin = torch.sin(angles)  # float32

        # We need per-token pointers; Triton expects contiguous [S, D]. Make contiguous tensors for each s:
        # cos_ptr: [S, D] contiguous
        cos_ptr = cos.contiguous()  # [B, S, D]
        sin_ptr = sin.contiguous()  # [B, S, D]
        # Triton kernels expect inputs as tensors; we can pass slices per s. But Triton loads by pointer offsets.
        # To use rotate_and_scatter_kernel, we need cos_ptr and sin_ptr shaped [S, D]. We can extract by indexing,
        # but Triton will access via pointer arithmetic. For simplicity, we pass the entire tensors and load s-th slice.

        # 4) Allocate outputs for rotated query and rotated key (same shape as key_cache/value_cache semantics)
        # We will write rotated rows into key_cache at cache_position, and original key_norm rows into value_cache.
        # However, we also need to return query_rotated and key_rotated. We can compute them in the same scatter kernel by writing to separate tensors.
        # To avoid complexity, we'll compute query rotation in Triton and write into output_query_rotated. For key_rotated, we write into key_cache.
        # Prepare output tensors:
        out_query_rotated = torch.empty_like(query_norm)  # fp32
        out_key_rotated = torch.empty_like(key_norm)      # fp32 (to emulate returned key_rotated)
        # For value_cache, we'll write original key_norm rows at cache positions.

        # Launch rotate_and_scatter_kernel for query:
        # We need cos/sin for query as well, but we already have cos/sin for keys. Since rotation uses the same inv and pos, we can reuse cos/sin.
        grid_ps = (B * N_q,)
        rotate_and_scatter_kernel[grid_ps](
            query_norm,                       # key_norm_ptr (for query rotation)
            out_query_rotated,               # out_key_ptr (we write rotated query here)
            torch.empty(0, device=query.device),  # dummy out_val_ptr (we won't write here)
            cos_ptr,                         # cos_ptr
            sin_ptr,                         # sin_ptr
            position_ids,                    # pos_ids_ptr (int64)
            cache_position,                  # cache_pos_ptr (int64)
            B, N_q, S, D,
        )

        # Launch rotate_and_scatter_kernel for key (writing into key_cache and value_cache)
        out_key_cache = torch.empty((B, N_kv, S, D), dtype=torch.float32, device=query.device)
        out_value_cache = torch.empty((B, N_kv, S, D), dtype=torch.float32, device=query.device)

        grid_kp = (B * N_kv,)
        rotate_and_scatter_kernel[grid_kp](
            key_norm,                        # key_norm_ptr
            out_key_cache,                  # out_key_ptr (rotated keys)
            out_value_cache,                # out_val_ptr (original key_norm rows)
            cos_ptr,                        # cos_ptr
            sin_ptr,                        # sin_ptr
            position_ids,                   # pos_ids_ptr
            cache_position,                 # cache_pos_ptr
            B, N_kv, S, D,
        )

        # Update the original key_cache and value_cache in-place with out_key_cache and out_value_cache (emulating scatter writes).
        # Note: The original 'run' function updates key_cache in-place; we do the same here.
        key_cache.copy_(out_key_cache)
        value_cache.copy_(out_value_cache)

        # Return query_rotated, key_rotated, key_cache, value_cache
        # For key_rotated, use out_key_cache since the scatter wrote rotated keys; out_key_cache is fp32.
        return out_query_rotated, out_key_cache, key_cache, value_cache
