import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: RMSNorm per row
# Input: x_ptr [n_rows, head_dim], weight_ptr [head_dim], Output: y_ptr [n_rows, head_dim]
# We flatten [B, heads, seq_len, head_dim] into n_rows = B * heads * seq_len rows, and process each row independently.
@triton.jit
def rms_norm_row_kernel(x_ptr, y_ptr, weight_ptr, eps: tl.float32, head_dim: tl.constexpr, BLOCK: tl.constexpr):
    row_id = tl.program_id(0)
    # Each program handles one row
    row_offset = row_id * head_dim

    # Compute variance in fp32 for numerical stability
    # Load x row (masked to head_dim)
    offs = tl.arange(0, BLOCK)
    mask = offs < head_dim
    x = tl.load(x_ptr + row_offset + offs, mask=mask, other=0.0)
    x_fp32 = x.to(tl.float32)
    # sum of squares
    sum_sq = tl.sum(x_fp32 * x_fp32, axis=0)
    mean = sum_sq / head_dim
    scale = tl.rsqrt(mean + eps)  # fp32

    # Load weight and apply scaling
    w = tl.load(weight_ptr + offs, mask=mask, other=1.0)  # masked since offs < head_dim
    w_fp32 = w.to(tl.float32)
    y = (x_fp32 * (w_fp32 * scale)).to(x.dtype)

    # Store output
    tl.store(y_ptr + row_offset + offs, y, mask=mask)


# Triton kernel: build cos/sin rotation vector
# inv_freq_ptr: [half_dim] float32
# cos_ptr: [head_dim] float32
# sin_ptr: [head_dim] float32
@triton.jit
def build_cos_sin_vec(inv_freq_ptr, cos_ptr, sin_ptr, head_dim: tl.constexpr, half_dim: tl.constexpr, BLOCK: tl.constexpr):
    # We build emb = cat([pos * inv_freq, pos * inv_freq]) with pos=0, since the rotation is position-agnostic in this setup.
    offs = tl.arange(0, BLOCK)
    mask = offs < head_dim

    idx_first = offs % 2  # indices for first half
    idx_half = offs // 2  # map to half_dim

    # Load inv_freq
    inv_freq = tl.load(inv_freq_ptr + idx_half, mask=idx_first == 0, other=0.0)  # valid only for even indices
    # pos = 0, so emb_first = 0 * inv_freq = 0
    emb_first = inv_freq * 0.0
    emb_second = emb_first

    # Concatenate: for i in [0, head_dim): i < half_dim -> emb[i] = emb_first[i], else emb[i] = emb_second[i-half_dim]
    is_first_half = offs < half_dim
    emb_val = tl.where(is_first_half, emb_first, emb_second)

    cos_val = tl.cos(emb_val)
    sin_val = tl.sin(emb_val)

    # Store to cos/sin vectors
    tl.store(cos_ptr + offs, cos_val, mask=mask)
    tl.store(sin_ptr + offs, sin_val, mask=mask)


# Triton kernel: apply rotation to a tensor of shape [n_rows, head_dim]
# y = x * cos + (rotate_half(x)) * sin
# rotate_half(x) = cat([x2, -x1], dim=-1), where x1 = x[..., :half_dim], x2 = x[..., half_dim:]
# Here we implement per row: for each i, if i < half_dim => x2[i], else x1[i-half_dim]
@triton.jit
def rotate_rows_kernel(x_ptr, y_ptr, cos_ptr, sin_ptr, head_dim: tl.constexpr, half_dim: tl.constexpr, BLOCK: tl.constexpr):
    row_id = tl.program_id(0)
    row_offset = row_id * head_dim

    offs = tl.arange(0, BLOCK)
    mask = offs < head_dim

    x = tl.load(x_ptr + row_offset + offs, mask=mask, other=0.0)
    x_fp32 = x.to(tl.float32)

    # Load cos/sin
    cos = tl.load(cos_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    sin = tl.load(sin_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    # Split into two halves
    is_first_half = offs < half_dim
    i1 = offs  # positions in first half
    i2 = offs - half_dim  # positions in second half

    x1 = tl.where(is_first_half, x_fp32, 0.0)           # values for first half positions
    x2 = tl.where(~is_first_half, x_fp32, 0.0)          # values for second half positions

    # Build rotated x: for index i, take x2[i] for i < half_dim, -x1[i] for i >= half_dim
    rotated_i = tl.where(is_first_half, x2[i1], -x1[i2])

    y = x_fp32 * cos + rotated_i * sin
    y = y.to(x.dtype)

    tl.store(y_ptr + row_offset + offs, y, mask=mask)


class ModelNew(torch.nn.Module):
