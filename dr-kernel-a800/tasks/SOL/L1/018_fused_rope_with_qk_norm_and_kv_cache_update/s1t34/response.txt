import math
import torch
import triton
import triton.language as tl

# Triton kernel: RMSNorm per row over the last dimension (head_dim)
# x_ptr: *const element, shape [N, D], contiguous
# y_ptr: *element, shape [N, D], contiguous
# weight_ptr: *const element, shape [D], dtype matches x
# eps: float32 scalar
@triton.jit
def rmsnorm_row_kernel(x_ptr, y_ptr, weight_ptr, N, D, eps):
    row_id = tl.program_id(0)  # 0..N-1
    # Offsets for the row
    offs = tl.arange(0, D)
    # Pointer to start of this row
    x_row_ptr = x_ptr + row_id * D
    y_row_ptr = y_ptr + row_id * D

    # Load x and weight, cast to fp32 for accumulation
    x = tl.load(x_row_ptr + offs)  # D elements
    w = tl.load(weight_ptr + offs)  # D elements
    x32 = x.to(tl.float32)
    w32 = w.to(tl.float32)

    # Sum of squares across head_dim
    sumsq = tl.sum(x32 * x32, axis=0)
    mean = sumsq / D  # float32 division
    inv_scale = tl.rsqrt(mean + eps)  # 1/sqrt(var + eps)
    y32 = w32 * x32 * inv_scale  # elementwise multiply in fp32

    # Cast back to original dtype of x and store
    y = y32.to(x.dtype)
    tl.store(y_row_ptr + offs, y)

            query_norm = rmsnorm_torch(query, q_norm_weight, rms_norm_eps)
            key_norm = rmsnorm_torch(key, k_norm_weight, rms_norm_eps)
