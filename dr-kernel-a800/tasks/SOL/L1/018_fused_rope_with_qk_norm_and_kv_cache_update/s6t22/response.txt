import torch
import triton
import triton.language as tl


@triton.jit
def rms_sum_kernel(x_ptr, sum_ptr,
                    B: tl.constexpr, num_heads: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
                    batch_stride_x, head_stride_x, s_stride_x,
                    sum_stride0, sum_stride1, sum_stride2,
                    BLOCK_H: tl.constexpr):
    # One program per (b, head, s)
    pid = tl.program_id(0)
    if pid >= B * num_heads * S:
        return
    b = pid // (num_heads * S)
    h = (pid % (num_heads * S)) // S
    s = pid % S

    total = tl.zeros((), dtype=tl.float32)
    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        offs_x = b * batch_stride_x + h * head_stride_x + s * s_stride_x + idx
        x_vals = tl.load(x_ptr + offs_x, mask=mask, other=0.0).to(tl.float32)
        total += tl.sum(x_vals * x_vals, axis=0)
    out_off = b * sum_stride0 + h * sum_stride1 + s * sum_stride2
    tl.store(sum_ptr + out_off, total)


@triton.jit
def rms_norm_kernel(x_ptr, weight_ptr, out_ptr, sum_ptr,
                    B: tl.constexpr, num_heads: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
                    batch_stride_x, head_stride_x, s_stride_x,
                    batch_stride_out, head_stride_out, s_stride_out,
                    weight_stride,  # typically 1 for [H]
                    sum_stride0, sum_stride1, sum_stride2,
                    eps: tl.constexpr,
                    BLOCK_H: tl.constexpr):
    # One program per (b, head, s)
    pid = tl.program_id(0)
    if pid >= B * num_heads * S:
        return
    b = pid // (num_heads * S)
    h = (pid % (num_heads * S)) // S
    s = pid % S

    # Load sum for this (b, h, s)
    sum_val = tl.load(sum_ptr + (b * sum_stride0 + h * sum_stride1 + s * sum_stride2))
    mean = sum_val / H
    inv_rms = 1.0 / tl.sqrt(mean + eps)

    # Normalize and apply weight, store
    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        offs_x = b * batch_stride_x + h * head_stride_x + s * s_stride_x + idx
        x_vals = tl.load(x_ptr + offs_x, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + idx * weight_stride, mask=mask, other=1.0).to(tl.float32)
        y = x_vals * inv_rms * w
        offs_out = b * batch_stride_out + h * head_stride_out + s * s_stride_out + idx
        tl.store(out_ptr + offs_out, y, mask=mask)


@triton.jit
def rotate_sin_cos_kernel_b_s(position_ids_ptr, inv_freq_ptr, cos_ptr, sin_ptr,
                               B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
                               position_stride,  # usually 1
                               cos_stride0, cos_stride1, cos_stride2,
                               sin_stride0, sin_stride1, sin_stride2,
                               BLOCK_H: tl.constexpr):
    # One program per (b, s)
    pid = tl.program_id(0)
    if pid >= B * S:
        return
    b = pid // S
    s = pid % S
    pos = tl.load(position_ids_ptr + b * position_stride + s * position_stride)

    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        # first half: i < H//2 -> inv_freq[i//2] only used for even i, but inv_freq has length H//2
        half = H // 2
        first_half_mask = idx < half
        inv_freq_idx = idx // 2  # only valid for idx < half
        emb_first = pos * tl.load(inv_freq_ptr + inv_freq_idx, mask=first_half_mask, other=0.0).to(tl.float32)
        # second half duplicates first half
        emb = tl.where(first_half_mask, emb_first, emb_first)

        c = tl.cos(emb)
        s_ = tl.sin(emb)
        base = b * cos_stride0 + s * cos_stride1
        tl.store(cos_ptr + base + idx * cos_stride2, c, mask=mask)
        tl.store(sin_ptr + base + idx * sin_stride2, s_, mask=mask)


@triton.jit
def apply_rotation_kernel(x_ptr, cos_ptr, sin_ptr, out_ptr,
                           B: tl.constexpr, num_heads: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
                           batch_stride_x, head_stride_x, s_stride_x,
                           batch_stride_out, head_stride_out, s_stride_out,
                           cos_stride0, cos_stride1, cos_stride2,
                           sin_stride0, sin_stride1, sin_stride2,
                           BLOCK_H: tl.constexpr):
    # One program per (b, head, s)
    pid = tl.program_id(0)
    if pid >= B * num_heads * S:
        return
    b = pid // (num_heads * S)
    h = (pid % (num_heads * S)) // S
    s = pid % S

    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        offs_x = b * batch_stride_x + h * head_stride_x + s * s_stride_x + idx

        # Load x
        x = tl.load(x_ptr + offs_x, mask=mask, other=0.0).to(tl.float32)
        base = b * cos_stride0 + s * cos_stride1
        cos_vals = tl.load(cos_ptr + base + idx * cos_stride2, mask=mask, other=0.0).to(tl.float32)
        sin_vals = tl.load(sin_ptr + base + idx * sin_stride2, mask=mask, other=0.0).to(tl.float32)

        # Compute rotated = [-x[half:], x[:half]]
        half = H // 2
        rotated_first = x[:half]                   # elements 0..half-1
        rotated_second = -x[half:]                # elements half..H-1 (negative of second half of x)
        rotated = tl.zeros((H,), dtype=tl.float32)
        rotated = tl.where(idx < half, rotated_first, rotated_second)

        y = x * cos_vals + rotated * sin_vals

        offs_out = b * batch_stride_out + h * head_stride_out + s * s_stride_out + idx
        tl.store(out_ptr + offs_out, y, mask=mask)


@triton.jit
def update_cache_kernel(query_ptr, key_cache_ptr, value_ptr, value_cache_ptr,
                         cache_position_ptr,
                         B: tl.constexpr, num_kv_heads: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
                         batch_stride_q, head_stride_q, s_stride_q,
                         batch_stride_key_cache, head_stride_key_cache, pos_stride_key_cache,
                         batch_stride_value_cache, head_stride_value_cache, pos_stride_value_cache,
                         cache_position_stride,
                         BLOCK_H: tl.constexpr):
    # One program per (b, kv_head, s)
    pid = tl.program_id(0)
    if pid >= B * num_kv_heads * S:
        return
    b = pid // (num_kv_heads * S)
    h = (pid % (num_kv_heads * S)) // S
    s = pid % S

    dest_pos = tl.load(cache_position_ptr + s * cache_position_stride)

    # Load query rotated for key_cache, or key rotated for key_cache (assuming rotated_query is passed for key_cache)
    # Here, query_ptr is the rotated tensor we want to write into key_cache at cache_position[s].
    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        offs_q = b * batch_stride_q + h * head_stride_q + s * s_stride_q + idx
        q = tl.load(query_ptr + offs_q, mask=mask, other=0.0).to(tl.float32)

        offs_key_cache = b * batch_stride_key_cache + h * head_stride_key_cache + dest_pos * pos_stride_key_cache + idx
        tl.store(key_cache_ptr + offs_key_cache, q, mask=mask)

    # Update value_cache with original value
    v = tl.load(value_ptr + (b * batch_stride_value_cache + h * head_stride_value_cache + s * s_stride_q), mask=True, other=0.0).to(tl.float32)
    # Note: we need to iterate over H to store each element; we can use a loop to fill entire H (for simplicity, we store v across idx)
    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        offs_value_cache = b * batch_stride_value_cache + h * head_stride_value_cache + dest_pos * pos_stride_value_cache + idx
        tl.store(value_cache_ptr + offs_value_cache, v, mask=mask)
