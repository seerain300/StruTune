# 1) Triton kernel for RMSNorm over last dim: Out = X / sqrt(mean(X^2) + eps) * W
# Inputs:
#   X_ptr: pointer to input [rows, head_dim], float32
#   W_ptr: pointer to weight [head_dim], float32
#   Out_ptr: pointer to output [rows, head_dim], float32
#   rows: number of rows
#   head_dim: length of last dimension (128)
#   eps: epsilon for RMSNorm
@triton.jit
def rmsnorm_rows_kernel(X_ptr, W_ptr, Out_ptr,
                         rows, head_dim,
                         eps: tl.constexpr,
                         BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    sumsq = 0.0
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / head_dim
    r = tl.rsqrt(mean + eps)
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        w = tl.load(W_ptr + offs, mask=mask, other=0.0)
        out = x * r * w
        tl.store(Out_ptr + row_id * head_dim + offs, out, mask=mask)

        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        cos = tl.load(COS_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        sin = tl.load(SIN_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        # We need to implement rotation mapping across the 128 dims. Conceptually:
        # For offs in [0..63]: y = x[offs] * cos[offs] + x[offs+64] * sin[offs]
        # For offs in [64..127]: y = x[offs] * cos[offs] + x[offs-64] * sin[offs]
        # Triton doesn't support indexing into loaded vectors with vectorized offsets, so we handle two segments with masks.
        mask_first = offs < 64
        mask_second = offs >= 64
        # Prepare y_first and y_second vectors
        y_first = x * cos
        # x_at_64 = x[offs+64] for offs < 64; invalid for offs >= 64
        x_at_64 = tl.load(X_ptr + row_id * head_dim + offs + 64, mask=mask_first, other=0.0)
        sin_first = tl.load(SIN_ptr + row_id * head_dim + offs, mask=mask_first, other=0.0)
        y_first = y_first + x_at_64 * sin_first

        # y_second
        x_at_prev = tl.load(X_ptr + row_id * head_dim + offs - 64, mask=mask_second, other=0.0)
        sin_second = tl.load(SIN_ptr + row_id * head_dim + offs, mask=mask_second, other=0.0)
        y_second = x * cos + x_at_prev * sin_second

        # Combine by selecting per-mask
        # Triton doesn't support vectorized where; we store each mask segment with its own store.
        tl.store(Y_ptr + row_id * head_dim + offs, y_first, mask=mask_first)
        tl.store(Y_ptr + row_id * head_dim + offs, y_second, mask=mask_second)
