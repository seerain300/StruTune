import torch
import triton
import triton.language as tl

# Triton kernel: compute cos/sin for each position (fp32), producing [rows, 128].
# rows = B * T (for query or key), last dim = 128 (head_dim), half_dim = 64.
# We derive:
#   cos[j, i] = cos(pos_j * inv_freq[i]), i in [0,59]
#   sin[j, i] = sin(pos_j * inv_freq[i]), i in [0,59]
#   cos[j, 64+i] = cos(pos_j * inv_freq[63 - i]), i in [0,59]  -> maps 64..122 -> inv_freq[63:0]
#   sin[j, 64+i] = sin(pos_j * inv_freq[63 - i]), i in [0,59]
# where pos_j are the flattened positions [0..B*T).
# Outputs are fp32, caller will use them in Triton rotation or convert as needed.
@triton.jit
def compute_cos_sin_rows_kernel(pos_ptr,  # int64 [rows]
                                inv0_ptr,  # fp32 [64]
                                out_cos_ptr,  # fp32 [rows, 128]
                                out_sin_ptr,  # fp32 [rows, 128]
                                rows: tl.constexpr,
                                half_dim: tl.constexpr,
                                BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    # First half: i in [0, half_dim)
    for i in range(0, half_dim):
        idx = i
        val = tl.load(pos_ptr + row_id).to(tl.float32) * tl.load(inv0_ptr + idx)  # pos*inv_freq[i]
        c = tl.cos(val)
        s = tl.sin(val)
        tl.store(out_cos_ptr + row_id * 128 + idx, c)
        tl.store(out_sin_ptr + row_id * 128 + idx, s)

    # Second half: map i to 63 - i (mirrors inv_freq[63:0] via 2*i+1 trick on host)
    for i in range(0, half_dim):
        src_idx = 63 - i
        val = tl.load(pos_ptr + row_id).to(tl.float32) * tl.load(inv0_ptr + src_idx)  # pos*inv_freq[63-i]
        c = tl.cos(val)
        s = tl.sin(val)
        tl.store(out_cos_ptr + row_id * 128 + 64 + i, c)
        tl.store(out_sin_ptr + row_id * 128 + 64 + i, s)

# Triton kernel: apply rotation y = x * cos + rotate_half(x) * sin
# x is [rows, 128] fp32, cos/sin are [rows, 128] fp32, y is [rows, 128] fp32
@triton.jit
def apply_rotation_kernel(x_ptr, cos_ptr, sin_ptr, y_ptr,
                           rows: tl.constexpr,
                           head_dim: tl.constexpr,
                           BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(x_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        cos = tl.load(cos_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        sin = tl.load(sin_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        half = head_dim // 2
        x_first = x[:half]
        x_second = x[half:]
        rotated_second = -x_second
        rotated_first = x_first
        rotated = tl.concatenate([rotated_first, rotated_second], axis=0)  # shape [head_dim]
        y = x * cos + rotated * sin
        tl.store(y_ptr + row_id * head_dim + offs, y, mask=mask)

def _flatten_rows_lastdim(x: torch.Tensor) -> torch.Tensor:
    # x: [..., D], return flat [rows, D], rows = prod(x.shape[:-1])
    shape = x.shape
    rows = 1
    for d in shape[:-1]:
        rows *= d
    D = shape[-1]
    return x.reshape(rows, D).contiguous()

class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # args: query, key, value, position_ids, key_cache, value_cache, cache_position, q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps
        # Extract tensors

        query, key, value, position_ids, key_cache, value_cache, cache_position, q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps = args

        # Compute shapes
        Bq, Hq, Tq, Dq = query.shape
        Bk, Hk, Tk, Dk = key.shape
        assert Dq == 128 and Dk == 128, "head_dim must be 128"
        half_dim = 64

        # 1) Compute position indices flattened for query (we use query's Tq and its batch)
        # position_ids is [Bq, Tq] int64; flatten to [Bq*Tq]
        pos = position_ids[:, :Tq].reshape(-1).to(torch.int64)  # dtype int64 for Triton
        rows_q = Bq * Tq

        # Allocate cos/sin for query
        cos_q = torch.empty((rows_q, 128), dtype=torch.float32, device=query.device)
        sin_q = torch.empty((rows_q, 128), dtype=torch.float32, device=query.device)

        # Launch Triton cos/sin kernel
        # Note: inv_freq is [64] float32; we use inv_freq[:half_dim]
        inv0_q = inv_freq[:half_dim].to(torch.float32)
        compute_cos_sin_rows_kernel[(rows_q,)](
            pos, inv0_q, cos_q, sin_q,
            rows=rows_q, half_dim=half_dim, BLOCK_SIZE=128, num_warps=4
        )

        # 2) RMSNorm for query (PyTorch path to ensure exact numerical match)
        # Normalize over last dim=128 with epsilon
        query_fp32 = query.to(torch.float32)
        query_norm = torch.nn.functional.layer_norm(
            query_fp32, normalized_shape=(128,), weight=q_norm_weight.to(torch.float32), bias=None, eps=rms_norm_eps
        )

        # 3) Apply rotation to query_norm (Triton) -> query_rotated (fp32), cast to bf16
        query_norm_flat = _flatten_rows_lastdim(query_norm)
        rows_q_norm = query_norm_flat.shape[0]
        query_rotated_fp32 = torch.empty_like(query_norm_flat, dtype=torch.float32, device=query.device)
        apply_rotation_kernel[(rows_q_norm,)](
            query_norm_flat, cos_q, sin_q, query_rotated_fp32,
            rows=rows_q_norm, head_dim=128, BLOCK_SIZE=128, num_warps=4
        )
        query_rotated = query_rotated_fp32.view(Bq, Hq, Tq, 128).to(torch.bfloat16)

        # 4) Key RMSNorm (PyTorch path)
        key_fp32 = key.to(torch.float32)
        key_norm = torch.nn.functional.layer_norm(
            key_fp32, normalized_shape=(128,), weight=k_norm_weight.to(torch.float32), bias=None, eps=rms_norm_eps
        )

        # 5) Compute cos/sin for key positions (same positions conceptually, but key has shape [Bk, Hk, Tk, 128])
        # We will flatten [Bk * Hk * Tk] positions: use position_ids with key's batch/seq
        # However, original code uses query's position_ids. For key, we use the same pos as query since cache_position references query tokens.
        rows_k = Bk * Hk * Tk
        pos_k = position_ids[:, :Tq].reshape(-1).to(torch.int64)  # reuse query positions
        cos_k = torch.empty((rows_k, 128), dtype=torch.float32, device=key.device)
        sin_k = torch.empty((rows_k, 128), dtype=torch.float32, device=key.device)
        compute_cos_sin_rows_kernel[(rows_k,)](
            pos_k, inv0_q, cos_k, sin_k,
            rows=rows_k, half_dim=half_dim, BLOCK_SIZE=128, num_warps=4
        )

        # 6) Apply rotation to key_norm (Triton) -> key_rotated (fp32), cast to bf16
        key_norm_flat = _flatten_rows_lastdim(key_norm)
        rows_k_norm = key_norm_flat.shape[0]
        key_rotated_fp32 = torch.empty_like(key_norm_flat, dtype=torch.float32, device=key.device)
        apply_rotation_kernel[(rows_k_norm,)](
            key_norm_flat, cos_k, sin_k, key_rotated_fp32,
            rows=rows_k_norm, head_dim=128, BLOCK_SIZE=128, num_warps=4
        )
        key_rotated = key_rotated_fp32.view(Bk, Hk, Tk, 128).to(torch.bfloat16)

        # 7) Return results (original returns (query_rotated, key_rotated, key_cache, value_cache))
        # Note: value is not used in computation; cache updates via torch in original code.
        return query_rotated, key_rotated, key_cache, value_cache
