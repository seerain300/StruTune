        tl.store(Out_ptr + row_id * D + offs, y, mask=mask)

# Triton kernel: write key_cache[:, :, cache_position, :] = In[:, :, :, :] for B, H, T tokens.
# We handle per (b,h) slice: for each b,h, copy In[b, h, :, :] into key_cache[b, h, cache_position, :].
# In this kernel, we flatten the (b,h) dimension and iterate positions_out in cache.
@triton.jit
def copy_cache_rows_positions_kernel(In_ptr, Out_ptr,
                                     rows_out,  # rows_out = B * H * T
                                     B, H, T, D,
                                     cache_len: tl.constexpr,  # cache_len is fixed in eval (passed as meta)
                                     positions_out,  # [positions_len]
                                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    if pid >= rows_out:
        return
    # Map pid to (b, h, t)
    # b = pid // (H*T); h = (pid // T) % H; t = pid % T
    b = pid // (H * T)
    t = pid % T
    h = (pid // T) % H
    # Compute source offset: row_id_src = (b*H + h) * T + t
    row_src = (b * H + h) * T + t
    # Compute destination offset for each cache position
    for pos in range(0, cache_len):
        # positions_out is 1D; positions_out[pos] gives cache index
        idx = positions_out[pos]
        # Destination base offset for (b, h, idx, :) is b*(H*T*D) + h*(T*D) + idx*D
        dest_base = b * (H * T * D) + h * (T * D) + idx * D
        # Source base offset for (b, h, t, :) is row_src * D
        src_base = row_src * D
        # Copy D elements: offs = tl.arange(0, BLOCK_SIZE)
        offs = tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        vals = tl.load(In_ptr + src_base + offs, mask=mask, other=0.0)
        tl.store(Out_ptr + dest_base + offs, vals, mask=mask)

# Triton kernel: write value_cache[:, :, cache_position, :] = In for Bk*Hk*Tk rows.
# Flatten rows_out and copy In[row] into value_cache[:, :, positions_out[pos], :].
@triton.jit
def copy_values_cache_rows_positions_kernel(In_ptr, Out_ptr,
                                            rows_out,  # rows_out = Bk * Hk * Tq (we set Tq=Tk here)
                                            Bk, Hk, Tq, D,
                                            cache_len: tl.constexpr,
                                            positions_out,
                                            BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    if pid >= rows_out:
        return
    # Map pid to (b, h, t)
    b = pid // (Hk * Tq)
    t = pid % Tq
    h = (pid // Tq) % Hk
    row_src = (b * Hk + h) * Tq + t
    for pos in range(0, cache_len):
        idx = positions_out[pos]
        dest_base = b * (Hk * Tq * D) + h * (Tq * D) + idx * D
        src_base = row_src * D
        offs = tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        vals = tl.load(In_ptr + src_base + offs, mask=mask, other=0.0)
        tl.store(Out_ptr + dest_base + offs, vals, mask=mask)
