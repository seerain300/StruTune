- Cos/Sin in Triton:

- Rotation in Triton:
  - Apply y = x * cos + rotate_half(x) * sin on normalized tensors. Broadcast cos/sin across rows. Launch once for query, once for key.

```python
import torch
import triton
import triton.language as tl

# Triton kernel: RMSNorm over the last dimension (head_dim) for each row.
# Inputs:
#   X_ptr: pointer to input [rows, head_dim], dtype float32
#   W_ptr: pointer to weight [head_dim], dtype float32
#   Out_ptr: pointer to output [rows, head_dim], dtype float32
#   rows: number of rows
#   head_dim: length of last dimension
# eps: epsilon for RMSNorm
@triton.jit
def rmsnorm_rows_kernel(X_ptr, W_ptr, Out_ptr,
                         rows, head_dim,
                         eps: tl.constexpr,
                         BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    sumsq = 0.0
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / head_dim
    r = tl.rsqrt(mean + eps)
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        w = tl.load(W_ptr + offs, mask=mask, other=1.0)
        y = x * r * w
        tl.store(Out_ptr + row_id * head_dim + offs, y, mask=mask)

# Triton kernel: compute cos/sin vectors for positions [0..T) and frequency indices [:half_dim].
# Inputs:
#   pos_ptr: [T] int64 positions
#   inv_ptr: [half_dim] float32 inv_freq for first 64 dims
#   cos_ptr: [T, head_dim] float32 output cos
#   sin_ptr: [T, head_dim] float32 output sin
@triton.jit
def cos_sin_positions_kernel(pos_ptr, inv_ptr, cos_ptr, sin_ptr,
                              T: tl.constexpr, half_dim: tl.constexpr,
                              BLOCK_SIZE: tl.constexpr):
    t = tl.program_id(0)
    if t >= T:
        return
    angle = t.to(tl.float32) * inv_ptr[tl.arange(0, half_dim)]
    cos_vec = tl.cos(angle)
    sin_vec = tl.sin(angle)
    for i in range(half_dim):
        tl.store(cos_ptr + t * (half_dim * 2) + (2 * i), cos_vec[i])
        tl.store(sin_ptr + t * (half_dim * 2) + (2 * i) + 1, sin_vec[i])

# Triton kernel: apply rotation to normalized tensor using cos/sin vectors broadcast over rows.
# Inputs:
#   X_ptr: [rows, head_dim] float32 normalized input
#   cos_ptr: [T, head_dim] float32 cos (but we broadcast across rows: per row, cos/sin are per t)
#   sin_ptr: [T, head_dim] float32 sin
#   Out_ptr: [rows, head_dim] float32 output
@triton.jit
def rotate_y_kernel(X_ptr, cos_ptr, sin_ptr, Out_ptr,
                    rows, head_dim,
                    BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)  # fp32
        cos_vec = tl.load(cos_ptr + offs, mask=mask, other=0.0)
        sin_vec = tl.load(sin_ptr + offs, mask=mask, other=0.0)
        half = head_dim // 2
        first = x[:half]
        second = x[half:]
        rotated_half = -second + first  # rotate_half(x)
        y = x * cos_vec + rotated_half * sin_vec
        tl.store(Out_ptr + row_id * head_dim + offs, y, mask=mask)

def _run(query, key, value, position_ids, key_cache, value_cache, cache_position, q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps):
    device = query.device
    Bq, Hq, Tq, Dq = query.shape
    Bk, Hk, Tk, Dk = key.shape
    assert Dq == 128 and Dk == 128, "head_dim must be 128"
    half_dim = Dq // 2  # 64

    # 1) RMSNorm for query (fp32 compute)
    query_norm = torch.empty((Bq, Hq, Tq, Dq), dtype=torch.float32, device=device)
    query_rows = Bq * Hq * Tq
    query_flat = query.reshape(query_rows, Dq)
    rmsnorm_rows_kernel[(query_rows,)](
        query_flat, q_norm_weight.to(torch.float32), query_norm.reshape(query_rows, Dq),
        query_rows, Dq, rms_norm_eps, BLOCK_SIZE=128, num_warps=4
    )

    # 2) Compute cos/sin for positions (position_ids): shape [Bq, Tq]
    pos_q = position_ids[:, :Tq].reshape(-1).to(torch.int64)  # [Bq*Tq]
    inv_half = inv_freq[:half_dim].to(torch.float32)          # [64]
    cos_q = torch.empty((Tq, Dq), dtype=torch.float32, device=device)
    sin_q = torch.empty((Tq, Dq), dtype=torch.float32, device=device)
    cos_sin_positions_kernel[(Tq,)](
        pos_q, inv_half, cos_q, sin_q, Tq, half_dim, BLOCK_SIZE=128, num_warps=4
    )

    # 3) Apply rotation to query_norm -> query_rotated (fp32 compute, cast to bf16)
    query_rows_k = query_rows
    query_rotated_fp32 = torch.empty((Bq, Hq, Tq, Dq), dtype=torch.float32, device=device)
    rotate_y_kernel[(query_rows_k,)](
        query_norm.reshape(query_rows_k, Dq),
        cos_q.to(torch.float32), sin_q.to(torch.float32),
        query_rotated_fp32.reshape(query_rows_k, Dq),
        query_rows_k, Dq, BLOCK_SIZE=128, num_warps=4
    )
    query_rotated = query_rotated_fp32.to(torch.bfloat16)

    # 4) RMSNorm for key (fp32 compute)
    key_norm = torch.empty((Bk, Hk, Tk, Dk), dtype=torch.float32, device=device)
    key_rows = Bk * Hk * Tk
    key_flat = key.reshape(key_rows, Dk)
    rmsnorm_rows_kernel[(key_rows,)](
        key_flat, k_norm_weight.to(torch.float32), key_norm.reshape(key_rows, Dk),
        key_rows, Dk, rms_norm_eps, BLOCK_SIZE=128, num_warps=4
    )

    # 5) Compute cos/sin for key positions: cache_position is [Tk]
    pos_k = cache_position.to(torch.int64)  # [Tk]
    cos_k = torch.empty((Tk, Dk), dtype=torch.float32, device=device)
    sin_k = torch.empty((Tk, Dk), dtype=torch.float32, device=device)
    cos_sin_positions_kernel[(Tk,)](
        pos_k, inv_half, cos_k, sin_k, Tk, half_dim, BLOCK_SIZE=128, num_warps=4
    )

    # 6) Apply rotation to key_norm -> key_rotated (fp32 compute, cast to bf16)
    key_rotated_fp32 = torch.empty((Bk, Hk, Tk, Dk), dtype=torch.float32, device=device)
    rotate_y_kernel[(key_rows,)](
        key_norm.reshape(key_rows, Dk),
        cos_k.to(torch.float32), sin_k.to(torch.float32),
        key_rotated_fp32.reshape(key_rows, Dk),
        key_rows, Dk, BLOCK_SIZE=128, num_warps=4
    )
    key_rotated = key_rotated_fp32.to(torch.bfloat16)

    # 7) Cache updates (PyTorch indexing for correctness across dynamic shapes)
    # For each (b,h), copy rotated_key[b,h,:,:] into key_cache[b, h, cache_position, :]
    # and copy value[b,h,:,:] into value_cache[b, h, cache_position, :].
    # Since Triton copy with 4D dynamic strides is fragile, we do it via torch ops.
    # Note: cache_position is int64 [Tk], so indexing is valid for key/value shapes.
    for b in range(Bk):
        for h in range(Hk):
            rows_bh = Tq  # We only update the first Tq positions (matching query rotation)
            # Copy key rotated (slice first Tq tokens)
            # Construct src: key_rotated[b, h, :rows_bh, :]
            src_key = key_rotated[b, h, :rows_bh, :]
            # Destination indices: key_cache[b, h, cache_position[:rows_bh], :]
            dest_key = key_cache[b, h, cache_position[:rows_bh].to(torch.long), :]
            dest_key.copy_(src_key.to(dest_key.dtype))  # dtype conversion if needed

            # Copy original value (slice first Tq tokens)
            src_val = value  # value is [B, S, D] -> assuming S == Tq; else use appropriate slice.
            # Destination indices: value_cache[b, h, cache_position[:rows_bh], :]
            dest_val = value_cache[b, h, cache_position[:rows_bh].to(torch.long), :]
            # If value is not matching exactly, we assume it's provided aligned; otherwise slice appropriately.
            # Here, we keep it simple: copy the original value tensor as-is to the cache positions.
            # If value has different shape, alignment must be ensured by caller. For this task, assume value is [B, S, D] with S == Tq.
            if value.shape[1] >= rows_bh:
                dest_val.copy_(value[b, :rows_bh, :].to(dest_val.dtype))

    return query_rotated, key_rotated, key_cache, value_cache
```
