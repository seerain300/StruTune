    Triton kernel: RMSNorm over last dimension D for M rows.
    Each program handles one row. Computes r = sqrt(mean(x^2) + eps) and writes y = x / r.
    X_ptr and Y_ptr are base pointers for input/output; strides must be handled by caller.
    Assumes 4D tensors with shape (B, N, S, D). M = B * N * S.

        return
    sum_sq = 0.0
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D

import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel(X_ptr, Y_ptr, M, D, eps, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: RMSNorm over last dimension D for M rows (view of 4D tensors).
    Each program handles one row. Computes r = sqrt(mean(x^2) + eps) and writes y = x / r.
    """
    row_id = tl.program_id(axis=0)
    if row_id >= M:
        return
    sum_sq = 0.0
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        y = (x_f32 / r).to(x.dtype)
        tl.store(Y_ptr + row_id * D + offs, y, mask=mask)


@triton.jit
def build_inv_kernel(inv_freq_ptr, inv_ptr, D_HALF: tl.constexpr, D: tl.constexpr):
    """
    Triton kernel: Build inv of length D = 2 * D_HALF from inv_freq of length D_HALF.
    inv[0:D_HALF] = inv_freq, inv[D_HALF:] = inv_freq.
    """
    for i in range(0, D_HALF):
        tl.store(inv_ptr + i, tl.load(inv_freq_ptr + i))
    for i in range(0, D_HALF):
        tl.store(inv_ptr + D_HALF + i, tl.load(inv_freq_ptr + i))


@triton.jit
def rotate_and_scatter_kernel(
    key_norm_ptr, value_ptr,
    inv_ptr, cos_ptr, sin_ptr,
    key_cache_ptr, value_cache_ptr,
    B, N, S, D,
    cache_pos_ptr,  # int64[S]
    BLOCK_SIZE: tl.constexpr,
):
    """
    Triton kernel: For each (b, n), iterates over tokens s and:
      - Loads key_norm[b, n, s, :] and value[b, n, s, :].
      - Computes cos and sin from inv and pos=s inside kernel (we'll pass cos/sin precomputed).
      - Applies rotation: split x into halves, rotate_half(x) = [-x2, x1].
        y1 = x1 * cos + rotate_half(x)[:, :D_HALF] * sin
        y2 = x2 * cos + rotate_half(x)[:, D_HALF:] * sin
      - Writes y into key_cache[b, n, cache_position[s], :] and value_cache[b, n, cache_position[s], :].
    Grid: (B, N).
    """
    b = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    if b >= B or n >= N:
        return
    for s in range(0, S):
        base = b * (N * S * D) + n * (S * D) + s * D
        key_row_ptr = key_norm_ptr + base
        val_row_ptr = value_ptr + base

        D_HALF = D // 2
        offs1 = tl.arange(0, BLOCK_SIZE)
        offs2 = offs1 + D_HALF
        mask1 = offs1 < D_HALF
        mask2 = offs2 < D

        k1 = tl.load(key_row_ptr + offs1, mask=mask1, other=0.0)
        k2 = tl.load(key_row_ptr + offs2, mask=mask2, other=0.0)

        # Load cos and sin for this token position s
        cos_ptr_s = cos_ptr + s * D
        sin_ptr_s = sin_ptr + s * D

        cos1 = tl.load(cos_ptr_s + offs1, mask=mask1, other=0.0)
        sin1 = tl.load(sin_ptr_s + offs1, mask=mask1, other=0.0)
        cos2 = tl.load(cos_ptr_s + offs2, mask=mask2, other=0.0)
        sin2 = tl.load(sin_ptr_s + offs2, mask=mask2, other=0.0)

        k1_f32 = k1.to(tl.float32)
        k2_f32 = k2.to(tl.float32)
        cos1_f32 = cos1.to(tl.float32)
        sin1_f32 = sin1.to(tl.float32)
        cos2_f32 = cos2.to(tl.float32)
        sin2_f32 = sin2.to(tl.float32)

        # rotate_half(x) across two halves:
        # y1 = k1*cos + (-k2)*sin (first half)
        y1 = k1_f32 * cos1_f32 + (-k2_f32) * sin1_f32
        # y2 = k2*cos + (-k1)*sin (second half)
        y2 = k2_f32 * cos2_f32 + (-k1_f32) * sin2_f32

        # Concatenate y1 and y2 back to D
        # We'll store with masks
        # Write to key_cache at position cache_position[s]
        pos = tl.load(cache_pos_ptr + s)  # int64 scalar
        out_base = pos * (B * N * D) + b * (N * D) + n * D
        # Store first half
        tl.store(key_cache_ptr + out_base + offs1, y1.to(k1.dtype), mask=mask1)
        # Store second half
        tl.store(key_cache_ptr + out_base + offs2, y2.to(k2.dtype), mask=mask2)

        # Write original value rows into value_cache at the same positions (not rotated)
        # out_base is already computed
        val1 = tl.load(val_row_ptr + offs1, mask=mask1, other=0.0)
        val2 = tl.load(val_row_ptr + offs2, mask=mask2, other=0.0)
        tl.store(value_cache_ptr + out_base + offs1, val1.to(val_row_ptr.dtype.element_ty), mask=mask1)
        tl.store(value_cache_ptr + out_base + offs2, val2.to(val_row_ptr.dtype.element_ty), mask=mask2)

        """
        Compute:
          - query_rotated, key_rotated
          - update key_cache, value_cache (key_rotated at cache_position, value at the same positions)
        Triton kernels are invoked for RMSNorm and rotation/scatter.

        # Ensure tensors are contiguous and devices match
        device = query.device
        B, N_q, S, D = query.shape
        N_kv = key.shape[1]

     - A Triton kernel rotate_query_kernel will rotate query rows: split x into halves, rotate_half(x) = [-x2, x1], then y = x1*cos + rotate_half(x)*sin, and store result.
     - A Triton kernel rotate_key_scatter_kernel will rotate key rows and write into key_cache at cache_position[s], also write original value rows into value_cache at the same positions.
  3) Ensure all kernels are actually launched from ModelNew.forward. No torch.cos/torch.sin in host code other than per-token vectors.

Performance considerations
- Triton kernels for RMSNorm and rotation will operate on contiguous rows; D=128 makes a BLOCK_SIZE of 128 efficient and simple. We use fp32 for reductions and math for stability, casting back to bfloat16 on store.
- We allocate and fill cos and sin per workload using torch; while this is per-token, it’s minimal compared to the overall attention computation in realistic scenarios. The evaluator runs multiple workloads; this approach is robust and avoids Triton limitations for sin/cos broadcasting.

Final code

import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel(X_ptr, Y_ptr, M, D, eps, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: RMSNorm over last dimension D for M rows (view of 4D tensors).
    Each program handles one row. Computes r = sqrt(mean(x^2) + eps) and writes y = x / r.
    """
    row_id = tl.program_id(axis=0)
    if row_id >= M:
        return
    sum_sq = 0.0
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        y = (x_f32 / r).to(x.dtype)
        tl.store(Y_ptr + row_id * D + offs, y, mask=mask)


@triton.jit
def build_inv_kernel(inv_freq_ptr, inv_ptr, D_HALF: tl.constexpr, D: tl.constexpr):
    """
    Triton kernel: Build inv of length D = 2 * D_HALF from inv_freq of length D_HALF.
    inv[0:D_HALF] = inv_freq, inv[D_HALF:] = inv_freq.
    """
    for i in range(0, D_HALF):
        tl.store(inv_ptr + i, tl.load(inv_freq_ptr + i))
    for i in range(0, D_HALF):
        tl.store(inv_ptr + D_HALF + i, tl.load(inv_freq_ptr + i))


@triton.jit
def rotate_query_kernel(query_norm_ptr, inv_ptr, cos_ptr, sin_ptr, query_rot_ptr, S, D, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: For each token s in [0, S), rotate each query row query_norm[b, n, s, :] using inv, cos[s, :], sin[s, :],
    and store into query_rot[b, n, s, :].
    Grid: (B, N_q, S).
    """
    # Triton does not support dynamic grid via B,N_q directly; we construct B,N_q,S via launch calls.
    # This kernel is launched with grid (B, N_q, S) by ModelNew.forward.
    b = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    base = b * (N_q * S * D) + n * (S * D) + s * D
    row_ptr = query_norm_ptr + base
    out_ptr = query_rot_ptr + base

    D_HALF = D // 2
    offs1 = tl.arange(0, BLOCK_SIZE)
    offs2 = offs1 + D_HALF
    mask1 = offs1 < D_HALF
    mask2 = offs2 < D

    q1 = tl.load(row_ptr + offs1, mask=mask1, other=0.0)
    q2 = tl.load(row_ptr + offs2, mask=mask2, other=0.0)

    # Load cos and sin for this token s (global s since cache_position is global indices)
    cos_ptr_s = cos_ptr + s * D
    sin_ptr_s = sin_ptr + s * D

    cos1 = tl.load(cos_ptr_s + offs1, mask=mask1, other=0.0)
    sin1 = tl.load(sin_ptr_s + offs1, mask=mask1, other=0.0)
    cos2 = tl.load(cos_ptr_s + offs2, mask=mask2, other=0.0)
    sin2 = tl.load(sin_ptr_s + offs2, mask=mask2, other=0.0)

    q1_f32 = q1.to(tl.float32)
    q2_f32 = q2.to(tl.float32)
    cos1_f32 = cos1.to(tl.float32)
    sin1_f32 = sin1.to(tl.float32)
    cos2_f32 = cos2.to(tl.float32)
    sin2_f32 = sin2.to(tl.float32)

    # y1 = q1*cos + (-q2)*sin (first half)
    y1 = q1_f32 * cos1_f32 + (-q2_f32) * sin1_f32
    # y2 = q2*cos + (-q1)*sin (second half)
    y2 = q2_f32 * cos2_f32 + (-q1_f32) * sin2_f32

    tl.store(out_ptr + offs1, y1.to(q1.dtype), mask=mask1)
    tl.store(out_ptr + offs2, y2.to(q2.dtype), mask=mask2)


@triton.jit
def rotate_key_scatter_kernel(
    key_norm_ptr, value_ptr,
    inv_ptr, cos_ptr, sin_ptr,
    key_cache_ptr, value_cache_ptr,
    B, N, S, D,
    cache_pos_ptr,  # int64[S], global positions
    BLOCK_SIZE: tl.constexpr,
):
    """
    Triton kernel: For each (b, n), iterates over tokens s and:
      - Loads key_norm[b, n, s, :] and value[b, n, s, :].
      - Loads cos and sin for this s.
      - Applies rotation: split x into halves, rotate_half(x) = [-x2, x1].
        y1 = k1 * cos + (-k2) * sin (first half)
        y2 = k2 * cos + (-k1) * sin (second half)
      - Writes y into key_cache[b, n, cache_position[s], :] and value_cache[b, n, cache_position[s], :].
    Grid: (B, N).
    """
    b = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    if b >= B or n >= N:
        return
    for s in range(0, S):
        base = b * (N * S * D) + n * (S * D) + s * D
        key_row_ptr = key_norm_ptr + base
        val_row_ptr = value_ptr + base

        D_HALF = D // 2
        offs1 = tl.arange(0, BLOCK_SIZE)
        offs2 = offs1 + D_HALF
        mask1 = offs1 < D_HALF
        mask2 = offs2 < D

        k1 = tl.load(key_row_ptr + offs1, mask=mask1, other=0.0)
        k2 = tl.load(key_row_ptr + offs2, mask=mask2, other=0.0)

        # Load cos and sin for this token position s (global indices)
        cos_ptr_s = cos_ptr + s * D
        sin_ptr_s = sin_ptr + s * D

        cos1 = tl.load(cos_ptr_s + offs1, mask=mask1, other=0.0)
        sin1 = tl.load(sin_ptr_s + offs1, mask=mask1, other=0.0)
        cos2 = tl.load(cos_ptr_s + offs2, mask=mask2, other=0.0)
        sin2 = tl.load(sin_ptr_s + offs2, mask=mask2, other=0.0)

        k1_f32 = k1.to(tl.float32)
        k2_f32 = k2.to(tl.float32)
        cos1_f32 = cos1.to(tl.float32)
        sin1_f32 = sin1.to(tl.float32)
        cos2_f32 = cos2.to(tl.float32)
        sin2_f32 = sin2.to(tl.float32)

        # y1 = k1*cos + (-k2)*sin (
