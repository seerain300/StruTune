# We will use grid = (B*H*L,) and compute base pointer for each row based on strides.
@triton.jit
def rmsnorm_reduce(x_ptr, out_ptr, rows, D, stride_x_row, stride_x_d):
    row_id = tl.program_id(0)
    # Compute base pointer for this row: base = x_ptr + row_id * stride_x_row
    # We loop over D elements in the row and accumulate sum of squares.
    sumsq = 0.0
    for i in range(0, D):
        ptr = x_ptr + row_id * stride_x_row + i * stride_x_d
        val = tl.load(ptr)
        sumsq += val * val
    tl.store(out_ptr + row_id, sumsq)


# Triton kernel: RMSNorm scale per row -> y = weight * x / sqrt(sumsq + eps)
@triton.jit
def rmsnorm_scale(x_ptr, y_ptr, weight_ptr, rows, D, eps, stride_x_row, stride_x_d, stride_y_row, stride_y_d):
    row_id = tl.program_id(0)
