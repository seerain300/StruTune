import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_row_kernel(
    x_ptr,               # *T (bfloat16)
    weight_ptr,          # *T (same dtype as x)
    y_ptr,               # *T (same dtype as x)
    B, H, L, D,          # int32 dims
    stride_x0, stride_x1, stride_x2, stride_x3,
    stride_y0, stride_y1, stride_y2, stride_y3,
    stride_w,            # int32: stride for weight (length D)
    eps,                 # float32
):
    # Flatten rows: pid in [0, B*H*L)
    pid = tl.program_id(axis=0)
    # Reconstruct (b, h, l)
    Lb = H * L
    b = pid // Lb
    rem = pid % Lb
    h = rem // L
    l = rem % L

    # Base pointers for the row (b, h, l, :)
    x_row_base = x_ptr + b * stride_x0 + h * stride_x1 + l * stride_x2
    y_row_base = y_ptr + b * stride_y0 + h * stride_y1 + l * stride_y2

    # Compute sum of squares in fp32
    sumsq = 0.0
    d = 0
    while d < D:
        offs = d + tl.arange(0, 1)
        mask = offs < D
        x = tl.load(x_row_base + offs * stride_x3, mask=mask, other=0.0)
        x_fp32 = x.to(tl.float32)
        sumsq += tl.sum(x_fp32 * x_fp32, axis=0)
        d += 1

    mean = sumsq / D
    var = mean
    inv_scale = tl.rsqrt(var + eps)  # fp32 scalar

    # Apply weight and store
    d = 0
    while d < D:
        offs = d + tl.arange(0, 1)
        mask = offs < D
        x = tl.load(x_row_base + offs * stride_x3, mask=mask, other=0.0)
        w = tl.load(weight_ptr + (d + offs) * stride_w, mask=mask, other=0.0)  # weight for this dim
        x_fp32 = x.to(tl.float32)
        y_fp32 = x_fp32 * inv_scale * w.to(tl.float32)
        y = y_fp32.to(x.dtype)
        tl.store(y_row_base + offs * stride_y3, y, mask=mask)
        d += 1

    def __init__(self):

        # Enforce contiguity for predictable strides
