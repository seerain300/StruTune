- Original computation summary:

import torch
import triton
import triton.language as tl

# RMSNorm kernel: per (b, h, s) row, normalize across D and scale by weight
@triton.jit
def rmsnorm_kernel(
    x_ptr,          # *dtype
    y_ptr,          # *dtype
    w_ptr,          # *dtype (weight), length D
    B: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    eps,            # float32
    # strides for x: stride_b, stride_h, stride_s, stride_d
    x_stride_b, x_stride_h, x_stride_s, x_stride_d,
    # strides for y: stride_b, stride_h, stride_s, stride_d
    y_stride_b, y_stride_h, y_stride_s, y_stride_d,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    # base pointers for row [b, h, s]
    x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
    y_row_ptr = y_ptr + b * y_stride_b + h * y_stride_h + s * y_stride_s

    # accumulate sum of squares in fp32
    sum_sq = 0.0
    for d in range(0, D):
        x_val = tl.load(x_row_ptr + d * x_stride_d)
        x_fp32 = x_val.to(tl.float32)
        sum_sq += x_fp32 * x_fp32

    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + eps)

    # scale by weight
    for d in range(0, D):
        x_val = tl.load(x_row_ptr + d * x_stride_d)
        w_val = tl.load(w_ptr + d)
        w_fp32 = w_val.to(tl.float32)
        y_val = (x_val.to(tl.float32) * inv_rms) * w_fp32
        # store back to y in original dtype
        tl.store(y_row_ptr + d * y_stride_d, y_val.to(x_val.dtype))

# Rotation kernel: y = x * cos - rotate_half(x) * sin (for query) or y = x * sin - rotate_half(x) * cos (for key)
# We pass cos_all and sin_all precomputed on device. cos_all and sin_all are [D].
@triton.jit
use_query: tl.constexpr,  # 1 if query rotation (uses cos_all), 0 if key rotation (uses sin_all)
def rotate_kernel(
    x_ptr,          # *dtype (normalized tensor), [B, H, S, D]
    y_ptr,          # *dtype (output rotated tensor), [B, H, S, D]
    cos_ptr,        # *dtype (cos_all), [D]
    sin_ptr,        # *dtype (sin_all), [D]
    B: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    # strides for x: stride_b, stride_h, stride_s, stride_d
    x_stride_b, x_stride_h, x_stride_s, x_stride_d,
    # strides for y: stride_b, stride_h, stride_s, stride_d
    y_stride_b, y_stride_h, y_stride_s, y_stride_d,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
    y_row_ptr = y_ptr + b * y_stride_b + h * y_stride_h + s * y_stride_s

    # Load x row as fp32
    x_fp32 = tl.zeros([D], dtype=tl.float32)
    for d in range(0, D):
        x_val = tl.load(x_row_ptr + d * x_stride_d)
        x_fp32[d] = x_val.to(tl.float32)

    # Load rotation coefficients (cos or sin) as fp32
    cos_all = tl.zeros([D], dtype=tl.float32)
    sin_all = tl.zeros([D], dtype=tl.float32)
    for d in range(0, D):
        cos_val = tl.load(cos_ptr + d)
        sin_val = tl.load(sin_ptr + d)
        cos_all[d] = cos_val.to(tl.float32)
        sin_all[d] = sin_val.to(tl.float32)

    # Compute rotate_half(x)
    half = D // 2
    x2 = x_fp32[half:]
    x1 = x_fp32[:half]
    rotate_half_x = tl.cat([-x2, x1], axis=0)

    # Apply rotation depending on use_query
    if use_query:
        # y = x * cos - rotate_half(x) * sin
        y_fp32 = x_fp32 * cos_all - rotate_half_x * sin_all
    else:
        # y = x * sin - rotate_half(x) * cos
        y_fp32 = x_fp32 * sin_all - rotate_half_x * cos_all

    # Store back to y (implicit cast to y dtype if needed)
    for d in range(0, D):
        tl.store(y_row_ptr + d * y_stride_d, y_fp32[d])

# Cache scatter update: write rotated_key into key_cache[b, h, cache_position[s], :] and value into value_cache[b, h, cache_position[s], :]
@triton.jit
def scatter_cache_kernel(
    src_ptr,        # *dtype (rotated key or original value), [B, H, S, D]
    dst_ptr,        # *dtype (cache), [B, H, L, D]
    B: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    L,              # int (max cache length), runtime scalar
    cache_pos_ptr,  # *int32, [S]
    # strides for src: stride_b, stride_h, stride_s, stride_d
    src_stride_b, src_stride_h, src_stride_s, src_stride_d,
    # strides for dst: stride_b, stride_h, stride_l, stride_d
    dst_stride_b, dst_stride_h, dst_stride_l, dst_stride_d,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    # get cache position for this s
    pos = tl.load(cache_pos_ptr + s).to(tl.int32)

    # compute linear index into dst for (b, h, pos, :)
    # Note: we assume dst is laid out as [B, H, L, D] with provided strides.
    dst_row_ptr = dst_ptr + b * dst_stride_b + h * dst_stride_h + pos * dst_stride_l

    # copy entire D-vector from src[b, h, s, :] to dst[b, h, pos, :]
    src_row_ptr = src_ptr + b * src_stride_b + h * src_stride_h + s * src_stride_s

    for d in range(0, D):
        val = tl.load(src_row_ptr + d * src_stride_d)
        tl.store(dst_row_ptr + d * dst_stride_d, val)

# Helper to run RMSNorm on a tensor (returns normalized & scaled tensor)
def triton_rmsnorm(x: torch.Tensor, weight: torch.Tensor, eps: float):
    B, H, S, D = x.shape
    y = torch.empty_like(x)
    # Launch grid over (B, H, S)
    grid = (B, H, S)
    rmsnorm_kernel[grid](
        x, y, weight,
        B, H, S, D,
        eps,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        y.stride(0), y.stride(1), y.stride(2), y.stride(3),
    )
    return y

# Helper to run rotation on a normalized tensor (returns rotated tensor)
def triton_rotate(x_norm: torch.Tensor, cos_all: torch.Tensor, sin_all: torch.Tensor, is_query: bool):
    B, H, S, D = x_norm.shape
    y = torch.empty_like(x_norm)
    grid = (B, H, S)
    # For query rotation we use cos_all; for key rotation we use sin_all. We pass cos_all and sin_all anyway and set use_query flag.
    # If is_query is False, cos_all is ignored in-kernel; sin_all is used. To keep signature consistent, pass cos_all=sin_all when is_query=True, or pass cos_all=x_norm when is_query=False? The simple approach: we always pass both, but in-kernel we pick based on is_query const flag.
    # Note: Triton requires compile-time flag; we can pass a Python bool as tl.constexpr meta-parameter.
    triton_rotate_kernel = rotate_kernel
    triton_rotate_kernel[grid](
        x_norm, y, cos_all, sin_all,
        B, H, S, D,
        x_norm.stride(0), x_norm.stride(1), x_norm.stride(2), x_norm.stride(3),
        y.stride(0), y.stride(1), y.stride(2), y.stride(3),
        use_query=is_query,
    )
    return y

# Helper to update caches: scatter rotated_key and value into key_cache/value_cache at cache_position
def triton_scatter_update(key_rot: torch.Tensor, value: torch.Tensor,
                           key_cache: torch.Tensor, value_cache: torch.Tensor,
                           cache_position: torch.Tensor):
    B, H_kv, S, D = key_rot.shape
    L = key_cache.shape[2]
    grid = (B, H_kv, S)
    # Ensure cache_position is int32 for Triton
    cache_pos = cache_position.to(torch.int32)
    # For key_cache, src is key_rot; for value_cache, src is value
    scatter_cache_kernel[grid](
        key_rot, key_cache,
        B, H_kv, S, D, L,
        cache_pos,
        key_rot.stride(0), key_rot.stride(1), key_rot.stride(2), key_rot.stride(3),
        key_cache.stride(0), key_cache.stride(1), key_cache.stride(2), key_cache.stride(3),
    )
    scatter_cache_kernel[grid](
        value, value_cache,
        B, H_kv, S, D, L,
        cache_pos,
        value.stride(0), value.stride(1), value.stride(2), value.stride(3),
        value_cache.stride(0), value_cache.stride(1), value_cache.stride(2), value_cache.stride(3),
    )

class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, query, key, value, position_ids, key_cache, value_cache, cache_position,
                q_norm_weight, k_norm_weight, inv_freq, rms_norm_eps):
        # Ensure bf16 tensors
        dtype = torch.bfloat16
        B, num_q_heads, S, D = query.shape
        num_kv_heads = key.shape[1]

        # 1) RMSNorm on query and key
        query_norm = triton_rmsnorm(query, q_norm_weight, rms_norm_eps)  # [B, H_q, S, D]
        key_norm = triton_rmsnorm(key, k_norm_weight, rms_norm_eps)     # [B, H_kv, S, D]

        # 2) Precompute cos_all and sin_all on device using torch (host-side ops allowed here):
        #    For query: cos_all = cos(angle), sin_all = sin(angle), where angle = position_ids * inv_freq and inv_freq has length D//2.
        #    For key: sin_all is used. We'll compute both using torch to ensure correctness; then we pass cos_all and sin_all to Triton kernels.
        #    Note: inv_freq is [D//2] float32; position_ids is [B, S] int64.
        # Build angle [B, S, D//2] then expand to [B, S, D] by concatenation or by constructing full vectors.
        # Since Triton expects flat vectors, we build cos_all and sin_all as [D] tensors per (b, s) context, but here we can use torch to create them.

        # Helper to compute angle and rotation vectors for a given [B, S] positions and inv_freq
        def build_rotation_vectors(position_ids, inv_freq):
            # position_ids: [B, S], inv_freq: [D//2]
            B_s, S_s = position_ids.shape
            half = D // 2
            # angle [B, S, half]
            angle = (position_ids.to(torch.float32) * inv_freq).unsqueeze(-1)  # [B, S, half]
            # cos_all and sin_all [B, S, D]
            cos_all = torch.cos(angle).expand(B_s, S_s, D).contiguous()
            sin_all = torch.sin(angle).expand(B_s, S_s, D).contiguous()
            # But we need per-(b,s) flattened vectors of length D. Since D is small and fixed (128), we can compute per batch and keep as [B, S, D].
            # However, Triton kernel expects a single [D] vector; simplest is to compute per (b, s) and pass. We'll create cos_all and sin_all as [B, S, D] and inside Triton read them as [D] using pointer arithmetic.
            # To pass to Triton, we just pass the [B, S, D] tensors; kernel will use pointer arithmetic to read cos_all[d] and sin_all[d] via linear indexing.
            return cos_all, sin_all

        # Compute query rotation vectors
        cos_all_q, sin_all_q = build_rotation_vectors(position_ids, inv_freq)  # [B, S, D]
        # Compute key rotation vectors (keys use sin-based rotation)
        cos_all_k, sin_all_k = build_rotation_vectors(position_ids, inv_freq)  # [B, S, D] (cos is unused for keys, but we still pass sin for consistency)

        # 3) Rotate query and key using Triton
        query_rotated = triton_rotate(query_norm, cos_all_q, sin_all_q, is_query=True)  # [B, H_q, S, D]
        key_rotated = triton_rotate(key_norm, cos_all_k, sin_all_k, is_query=False)     # [B, H_kv, S, D]

        # 4) Scatter update caches
        triton_scatter_update(key_rotated, value, key_cache, value_cache, cache_position)

        # Return query_rotated, key_rotated, key_cache, value_cache
        return query_rotated, key_rotated, key_cache, value_cache

@triton.jit
def rmsnorm_kernel(

    y_ptr,          # *dtype
    w_ptr,          # *dtype (weight), length D
    B: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    eps,            # float32
    # strides for x: stride_b, stride_h, stride_s, stride_d
    x_stride_b, x_stride_h, x_stride_s, x_stride_d,
    # strides for y: stride_b, stride_h, stride_s, stride_d
    y_stride_b, y_stride_h, y_stride_s, y_stride_d,
):

    h = tl.program_id(1)
    s = tl.program_id(2)

    x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
    y_row_ptr = y_ptr + b * y_stride_b + h * y_stride_h + s * y_stride_s

    # Accumulate sum of squares in fp32
    sum_sq = 0.0
    for d in range(0, D):
        x_val = tl.load(x_row_ptr + d * x_stride_d)
        x_fp32 = x_val.to(tl.float32)
        sum_sq += x_fp32 * x_fp32

    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + eps)

    # Scale by weight and store
    for d in range(0, D):
        x_val = tl.load(x_row_ptr + d * x_stride_d)
        w_val = tl.load(w_ptr + d)
        w_fp32 = w_val.to(tl.float32)
        y_val = (x_val.to(tl.float32) * inv_rms) * w_fp32
        tl.store(y_row_ptr + d * y_stride_d, y_val.to(x_val.dtype))

    rotate_half_x = x1 + [-x for x in x2]

    # y = x * cos - rotate_half(x) * sin
    y_fp32 = [x_fp32[d] * cos_all[d] - rotate_half_x[d] * sin_all[d] for d in range(0, D)]

    # Store back
    for d in range(0, D):
        tl.store(y_row_ptr + d * y_stride_d, y_fp32[d])

# Triton kernel: key rotation y = x * sin - rotate_half(x) * cos
@triton.jit
def rotate_key_kernel(
    x_ptr,          # *dtype (normalized tensor), [B, H_kv, S, D]
    y_ptr,          # *dtype (output rotated tensor), [B, H_kv, S, D]
    cos_ptr,        # *dtype (cos_all), [D]
    sin_ptr,        # *dtype (sin_all), [D]
    B: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    # strides for x: stride_b, stride_h, stride_s, stride_d
    x_stride_b, x_stride_h, x_stride_s, x_stride_d,
    # strides for y: stride_b, stride_h, stride_s, stride_d
    y_stride_b, y_stride_h, y_stride_s, y_stride_d,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    x_row_ptr = x_ptr + b * x_stride_b + h * x_stride_h + s * x_stride_s
    y_row_ptr = y_ptr + b * y_stride_b + h * y_stride_h + s * y_stride_s

    # Load x row as fp32
    x_fp32 = [0.0] * D
    for d in range(0, D):
        x_val = tl.load(x_row_ptr + d * x_stride_d)
        x_fp32[d] = x_val.to(tl.float32)

    # Load rotation coefficients as fp32
    cos_all = [0.0] * D
    sin_all = [0.0] * D
    for d in range(0, D):
        cos_val = tl.load(cos_ptr + d)
        sin_val = tl.load(sin_ptr + d)
        cos_all[d] = cos_val.to(tl.float32)
        sin_all[d] = sin_val.to(tl.float32)

    # Compute rotate_half(x)
    half = D // 2
    x2 = x_fp32[half:]
    x1 = x_fp32[:half]
    rotate_half_x = x1 + [-x for x in x2]

    # y = x * sin - rotate_half(x) * cos
    y_fp32 = [x_fp32[d] * sin_all[d] - rotate_half_x[d] * cos_all[d] for d in range(0, D)]

    # Store back
    for d in range(0, D):
        tl.store(y_row_ptr + d * y_stride_d, y_fp32[d])

# Triton kernel: scatter rotated_key and value into key_cache/value_cache at cache_position[s]
@triton.jit
def scatter_cache_kernel(
    src_ptr,        # *dtype (rotated key or original value), [B, H, S, D]
    dst_ptr,        # *dtype (cache), [B, H, L, D]
    B: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    L,              # int (max cache length), runtime scalar
    cache_pos_ptr,  # *int32, [S]
    # strides for src: stride_b, stride_h, stride_s, stride_d
    src_stride_b, src_stride_h, src_stride_s, src_stride_d,
    # strides for dst: stride_b, stride_h, stride_l, stride_d
    dst_stride_b, dst_stride_h, dst_stride_l, dst_stride_d,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    pos = tl.load(cache_pos_ptr + s).to(tl.int32)
    dst_row_ptr = dst_ptr + b * dst_stride_b + h * dst_stride_h + pos * dst_stride_l
    src_row_ptr = src_ptr + b * src_stride_b + h * src_stride_h + s * src_stride_s

    for d in range(0, D):
        val = tl.load(src_row_ptr + d * src_stride_d)
        tl.store(dst_row_ptr + d * dst_stride_d, val)

# Helper to precompute cos/sin vectors using torch (host-side allowed here)
def build_rotation_vectors(position_ids: torch.Tensor, inv_freq: torch.Tensor):
    # position_ids: [B, S] int64; inv_freq: [D//2] float32
    B, S = position_ids.shape
    half = 128 // 2  # since D=128
    angle = (position_ids.to(torch.float32) * inv_freq).unsqueeze(-1)  # [B, S, half]
    # Construct cos_all and sin_all as [B, S, D] by repeating along last dim
    cos_all = torch.cos(angle).expand(B, S, 128).contiguous()  # [B, S, D], all cos(angle) repeated
    sin_all = torch.sin(angle).expand(B, S, 128).contiguous()  # [B, S, D], all sin(angle) repeated
    return cos_all, sin_all  # both [B, S, D], bf16 in model, but we pass as tensors
