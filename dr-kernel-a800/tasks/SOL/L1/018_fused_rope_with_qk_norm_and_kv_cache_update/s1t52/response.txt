    # We'll traverse the last dim D for each row.

    row_size = NUM_ROWS * D  # not directly needed
    # Derive row base index: since pid indexes rows linearly, we can recover l from pid using modulo if needed.
    # Simpler: we map pid to row index in the flattened buffer. So for each row, we iterate over D.
    # However, Triton programs run independently; we can compute per-row base as pid * D and loop over D.
    # But to support arbitrary NUM_ROWS, we need to map pid to (b,h,l). Triton doesn't provide direct access to B,H,L.
    # Therefore, we make the caller launch exactly NUM_ROWS programs. Inside each program, we compute base = pid * D.
    # We will rely on the caller to ensure exactly NUM_ROWS = B*H*L. Then base = pid * D.
    base = pid * D
    # Accumulate sum of squares in fp32
    sumsq = 0.0
    # Loop over D in chunks
    for off in range(0, D, BLOCK):
        idx = off + tl.arange(0, BLOCK)
        mask = idx < D
        x = tl.load(X_ptr + base + idx, mask=mask, other=0.0)
        x_fp32 = x.to(tl.float32)
        sumsq += tl.sum(x_fp32 * x_fp32, axis=0)
    mean = sumsq / D
    inv_scale = 1.0 / tl.sqrt(mean + eps)
    # Apply weight
    for off in range(0, D, BLOCK):
        idx = off + tl.arange(0, BLOCK)
        mask = idx < D
        x = tl.load(X_ptr + base + idx, mask=mask, other=0.0)
        w = tl.load(W_ptr + idx, mask=mask, other=1.0)
        y = (x.to(tl.float32) * w.to(tl.float32)) * inv_scale
        y = y.to(x.dtype)  # cast back to original dtype (bfloat16)
        tl.store(Y_ptr + base + idx, y, mask=mask)
