```python
import triton
import triton.language as tl


@triton.jit
def rmsnorm_row_kernel(
    x_ptr,         # *const T, input tensor
    w_ptr,         # *const T, weight (length D)
    y_ptr,         # *T, output tensor
    B,             # int32, number of rows
    D,             # int32, head_dim (length of each row)
    eps,           # float32
    BLOCK: tl.constexpr,
):
    # One program per row
    row = tl.program_id(0)
    if row >= B:
        return

    # We assume x is contiguous with row-major over head_dim.
    # Compute base offset for this row: row * D
    base = row * D

    # Accumulate sum of squares in fp32
    sumsq = 0.0
    # Loop over head_dim in chunks of BLOCK; here BLOCK == D, so one iteration.
    for off in range(0, BLOCK, D):
        idx = off + tl.arange(0, D)
        x = tl.load(x_ptr + base + idx)  # load original dtype
        x32 = x.to(tl.float32)
        sumsq += tl.sum(x32 * x32, axis=0)

    mean = sumsq / D
    inv_scale = 1.0 / tl.sqrt(mean + eps)

    # Load weight and compute normalized output
    # weight is 1D of length D; dtype should match x's original dtype (bf16 in provided inputs)
    w = tl.load(w_ptr + tl.arange(0, D)).to(tl.float32)  # weight in fp32 for stability

    for off in range(0, BLOCK, D):
        idx = off + tl.arange(0, D)
        x = tl.load(x_ptr + base + idx)
        y32 = (x.to(tl.float32) * inv_scale) * w
        # Store back in original dtype of x_ptr (assumed same as input)
        tl.store(y_ptr + base + idx, y32.to(tl.typeof(tl.load(x_ptr + base))))  # cast to original dtype on store
