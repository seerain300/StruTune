# Triton kernel: RMSNorm over the last dimension (head_dim) for each row.
# Inputs:
#   X_ptr: pointer to input [rows, head_dim], dtype float32
#   W_ptr: pointer to weight [head_dim], dtype float32
#   Out_ptr: pointer to output [rows, head_dim], dtype float32
#   rows: number of rows
#   head_dim: length of last dimension
# eps: epsilon for RMSNorm
@triton.jit
def rmsnorm_rows_kernel(X_ptr, W_ptr, Out_ptr,
                         rows, head_dim,
                         eps: tl.constexpr,
                         BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)
    if row_id >= rows:
        return
    sumsq = 0.0
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)  # fp32
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / head_dim
    r = tl.rsqrt(mean + eps)
    for col in range(0, head_dim, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < head_dim
        x = tl.load(X_ptr + row_id * head_dim + offs, mask=mask, other=0.0)
        w = tl.load(W_ptr + offs, mask=mask, other=1.0)
        y = x * r * w
        tl.store(Out_ptr + row_id * head_dim + offs, y, mask=mask)

@triton.jit
def compute_cos_sin_kernel(pos_ptr, inv_ptr, cos_ptr, sin_ptr,
                           rows, D: tl.constexpr, half_dim: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(0)

        return
    for col in range(0, D, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        # Load inv for first half and repeat for second half
        inv_half = tl.load(inv_ptr + tl.arange(0, half_dim))  # [64]
        # Build emb for this row: emb[0:D/2] = pos * inv_half; emb[D/2:] = pos * inv_half (repeat)
        pos_val = tl.load(pos_ptr + row_id)
        emb_first = pos_val * inv_half  # [64]
        # Create emb by concatenating first half with itself
        # Triton supports broadcasting in tl.load/tl.store but not cat; we write two parts manually
        for i in range(64):
            tl.store(cos_ptr + row_id * D + i, tl.cos(emb_first[i]))
            tl.store(sin_ptr + row_id * D + i, tl.sin(emb_first[i]))
        for i in range(64):
            tl.store(cos_ptr + row_id * D + (64 + i), tl.cos(emb_first[i]))
            tl.store(sin_ptr + row_id * D + (64 + i), tl.sin(emb_first[i]))

# Triton kernel: apply rotation y = x * cos + rotate_half(x) * sin
# x: [rows, D], cos: [rows, D], sin: [rows, D]
@triton.jit
def apply_rotation_kernel(X_ptr, Cos_ptr, Sin_ptr, Out_ptr,
                          rows, D: tl.constexpr, half_dim: tl.constexpr, BLOCK_SIZE: tl.constexpr):

    if row_id >= rows:
        return
    for col in range(0, D, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        cos = tl.load(Cos_ptr + row_id * D + offs, mask=mask, other=0.0)
        sin = tl.load(Sin_ptr + row_id * D + offs, mask=mask, other=0.0)
        # Rotate half across the last dimension:
        # x is [D], we treat it as [first_half, second_half], rotate_half(x) = [-second_half, first_half]
        first = x[:half_dim]
        second = x[half_dim:]
        rotated_half = -second + first  # combining cat of [-second, first] as vector update
        y = x * cos + rotated_half * sin
        tl.store(Out_ptr + row_id * D + offs, y, mask=mask)
