import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rope_update(
    x, out, w, inv_freq,
    B: tl.constexpr, S: tl.constexpr,
    H, num_kv_heads,
    cache_len: tl.constexpr,
    D: tl.constexpr, HALF: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // (H * S)
    h = (pid // S) % H
    s = pid % S

    base = b * H * S * D + h * S * D + s * D

    # RMSNorm: compute scale in float32
    sum_sq = 0.0
    offs = tl.arange(0, D)
    for d in range(0, D):
        v = tl.load(x + base + d)
        sum_sq += v.to(tl.float32) * v
    mean = sum_sq / D
    scale = tl.rsqrt(mean + 1e-6)
    # write normalized and scaled output
    for d in range(0, D):
        v = tl.load(x + base + d)
        nv = (v.to(tl.float32) * scale) * tl.load(w + d)
        tl.store(out + base + d, nv.to(v.dtype))


@triton.jit
def apply_rotary_emb(
    x_in, x_out, inv_freq,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr,
    pos,
    D: tl.constexpr, HALF: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // (H * S)
    h = (pid // S) % H
    s = pid % S

    base = b * H * S * D + h * S * D + s * D

    # Load x
    offs = tl.arange(0, D)
    x = tl.load(x_in + base + offs)

    # Compute cos/sin for first HALF dims: emb = [pos * inv_freq, pos * inv_freq]
    pos_f = pos.to(tl.float32)  # scalar
    idx = tl.arange(0, HALF)
    base2 = pos_f * tl.load(inv_freq + idx)  # [HALF]
    cos_vec = tl.cos(base2)  # [HALF], float32
    sin_vec = tl.sin(base2)  # [HALF], float32

    # Prepare masks and offsets for rotation
    mask_even = offs < HALF
    mask_odd = offs >= HALF

    # Even part (first HALF)
    x_even = x[0:HALF]
    # Odd part (next HALF)
    x_odd = x[HALF:2 * HALF]

    # Apply rotation: y = x_even * cos - x_odd * sin
    y_even = x_even * cos_vec
    y_odd = -x_odd * sin_vec

    # Concatenate results
    y = tl.zeros([D], dtype=x.dtype)
    y[0:HALF] = y_even
    y[HALF:2 * HALF] = y_odd

    tl.store(x_out + base + offs, y)

                     cache_position: torch.Tensor,

                     rms_norm_eps: float):

    B = query.shape[0]
    H = query.shape[1]
    S = query.shape[2]
    D = query.shape[3]
    HALF = D // 2

    # Outputs
    query_norm = torch.empty_like(query)  # normalized query
    key_norm = torch.empty_like(key)      # normalized key

    # Launch RMSNorm kernels: one per (b, h, s)
    grid = (B * H * S,)
    rmsnorm_rope_update[grid](
        query, query_norm, q_norm_weight, inv_freq,
        B, S, H, 1, cache_len=0,
        D=D, HALF=HALF, num_warps=4, num_stages=2,
    )
    rmsnorm_rope_update[grid](
        key, key_norm, k_norm_weight, inv_freq,
        B, S, H, 1, cache_len=0,
        D=D, HALF=HALF, num_warps=4, num_stages=2,
    )

    # Apply rotary embedding
    query_rot = torch.empty_like(query_norm)
    key_rot = torch.empty_like(key_norm)
    pos = cache_position  # scalar tensor on device, but we pass as runtime arg; kernel treats it as scalar

        B, H, S, pos,

    )

    # Cache updates: write into provided key_cache, value_cache at positions cache_len + s
    # We do not read these tensors (to avoid Triton reading torch tensors); we just allocate outputs.
    key_cache_out = torch.empty_like(key_cache)
    value_cache_out = torch.empty_like(value_cache)

    return query_rot, key_rot, key_cache_out, value_cache_out

    base2 = pos_f * inv_freq_const[idx]  # [HALF], float32

    b = pid // (H * S)
