    sum_sq = tl.zeros((), dtype=tl.float32)

    for i in range(0, F, BLOCK_F):

        mask = offs < F
        x = tl.load(x_ptr + row_start + offs, mask=mask, other=0.0)

        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)
