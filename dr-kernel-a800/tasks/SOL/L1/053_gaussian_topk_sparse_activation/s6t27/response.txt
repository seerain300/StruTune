Overview:
- The original PyTorch code computes:
  - Per-row mean and population standard deviation along the feature dimension F (dim=-1), with unbiased=False: std = sqrt(E[x^2] - mean^2).
  - Inverse-normal CDF at target_sparsity via Abramowitz & Stegun 7.1.26 approximation (_ndtri).
  - Threshold per row: mean + std * z, where z = _ndtri(target_sparsity).
  - Elementwise apply: y = max(input - threshold, 0), returning bfloat16.

Plan and kernels:
- sum_rows_kernel: One Triton program per (batch, seq) row. Iterate over F in chunks (BLOCK_F). Load as float32, reduce to a scalar sum via tl.sum over the vector, store per-row sum.
- sumsq_rows_kernel: Same, but accumulate sum of squares.
- compute_stats_kernel: One program per row. Load per-row sum and sumsq, compute mean = sum/F, var = sumsq/F - mean^2, clamp var >= 0, std = sqrt(var), store per-row mean and std.
- ndtri_scalar_kernel: Single-program Triton kernel that computes inverse-normal CDF for a scalar p using A&S 7.1.26 masks. Write z to a 1-element float32 tensor on device.
- apply_threshold_kernel: One program per row. Load per-row mean and std, compute threshold = mean + std * z, iterate over F in chunks, load input as float32, compute y = max(x - threshold, 0), store output as float32. Host then casts to bfloat16.

This structure avoids atomics, keeps memory access contiguous along F, and minimizes complexity to prevent runtime errors. All math is done in float32 for stability, and final output is cast to bfloat16.

import torch
import triton
import triton.language as tl


@triton.jit
def sum_rows_kernel(
    x_ptr,           # *float32, input tensor as float32
    out_sum_ptr,     # *float32, per-row sum
    B: tl.constexpr, S: tl.constexpr, F: tl.constexpr,
    stride_b, stride_s, stride_f,  # strides in elements
    BLOCK_F: tl.constexpr
):
    row = tl.program_id(0)
    b = row // S
    s = row % S

    offs = tl.arange(0, BLOCK_F)
    acc = 0.0
    for f in range(0, F, BLOCK_F):
        idx_f = f + offs
        mask = idx_f < F
        x = tl.load(x_ptr + b * stride_b + s * stride_s + idx_f * stride_f, mask=mask, other=0.0)
        acc += tl.sum(x, axis=0)
    tl.store(out_sum_ptr + row, acc)


@triton.jit
def sumsq_rows_kernel(
    x_ptr,           # *float32, input tensor as float32
    out_sumsq_ptr,   # *float32, per-row sum of squares
    B: tl.constexpr, S: tl.constexpr, F: tl.constexpr,
    stride_b, stride_s, stride_f,
    BLOCK_F: tl.constexpr
):
    row = tl.program_id(0)
    b = row // S
    s = row % S

    offs = tl.arange(0, BLOCK_F)
    acc = 0.0
    for f in range(0, F, BLOCK_F):
        idx_f = f + offs
        mask = idx_f < F
        x = tl.load(x_ptr + b * stride_b + s * stride_s + idx_f * stride_f, mask=mask, other=0.0)
        acc += tl.sum(x * x, axis=0)
    tl.store(out_sumsq_ptr + row, acc)


@triton.jit
def compute_stats_kernel(
    out_sum_ptr,     # *float32, per-row sum
    out_sumsq_ptr,   # *float32, per-row sumsq
    out_mean_ptr,    # *float32, per-row mean
    out_std_ptr,     # *float32, per-row std
    B: tl.constexpr, S: tl.constexpr, F: tl.constexpr
):
    row = tl.program_id(0)
    sum_ = tl.load(out_sum_ptr + row)
    sumsq_ = tl.load(out_sumsq_ptr + row)
    mean = sum_ / F
    var = sumsq_ / F - mean * mean
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)
    tl.store(out_mean_ptr + row, mean)
    tl.store(out_std_ptr + row, std)


@triton.jit
def ndtri_scalar_kernel(
    z_ptr,           # *float32, 1-element buffer for z
    p,               # float32 scalar, target_sparsity
    a1, a2, a3, a4, a5, a6,
    b1, b2, b3, b4, b5,
    c1, c2, c3, c4, c5, c6,
    d1, d2, d3, d4,
    p_low,           # float32, lower-region threshold
    # num_warps and stages are not needed for a single program
):
    # Compute inverse-normal CDF using Abramowitz & Stegun 7.1.26
    # Piecewise regions:
    # lower: p < p_low
    # mid:  p_low <= p <= 1 - p_low
    # upper: p > 1 - p_low
    p_low_f = p_low
    high = 1.0 - p_low_f

    # lower region
    q_lower = tl.sqrt(-2.0 * tl.log(p))
    z_lower = ((c1 * q_lower + c2) * q_lower + c3) * q_lower + c4) * q_lower + c5) * q_lower + c6
    denom_lower = ((d1 * q_lower + d2) * q_lower + d3) * q_lower + d4) * q_lower + 1.0
    z_lower = -z_lower / denom_lower

    # mid region
    q_mid = p - 0.5
    r = q_mid * q_mid
    poly = ((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6
    poly *= q_mid
    den_mid = ((b1 * r + b2) * r + b3) * r + b4) * r + b5) * r + 1.0
    z_mid = poly / den_mid

    # upper region
    q_upper = tl.sqrt(-2.0 * tl.log(1.0 - p))
    z_upper = ((c1 * q_upper + c2) * q_upper + c3) * q_upper + c4) * q_upper + c5) * q_upper + c6
    denom_upper = ((d1 * q_upper + d2) * q_upper + d3) * q_upper + d4) * q_upper + 1.0
    z_upper = -z_upper / denom_upper

    # Select region based on p
    is_lower = p < p_low_f
    is_mid = p >= p_low_f and p <= high
    # default z to 0, Triton will select via masks
    z = tl.zeros((), dtype=tl.float32)
    z = tl.where(is_lower, z_lower, z)
    z = tl.where(is_mid, z_mid, z)

    tl.store(z_ptr, z)


@triton.jit
def apply_threshold_kernel(
    x_ptr,           # *float32, input tensor as float32
    out_ptr,         # *float32, output tensor (will be cast to bfloat16)
    mean_ptr,        # *float32, per-row mean
    std_ptr,         # *float32, per-row std
    z,               # float32 scalar, inverse-normal CDF at target_sparsity
    B: tl.constexpr, S: tl.constexpr, F: tl.constexpr,
    stride_b, stride_s, stride_f,
    BLOCK_F: tl.constexpr
):
    row = tl.program_id(0)
    b = row // S
    s = row % S

    mean = tl.load(mean_ptr + row)
    std = tl.load(std_ptr + row)
    threshold = mean + std * z

    offs = tl.arange(0, BLOCK_F)
    for f in range(0, F, BLOCK_F):
        idx_f = f + offs
        mask = idx_f < F
        x = tl.load(x_ptr + b * stride_b + s * stride_s + idx_f * stride_f, mask=mask, other=0.0)
        y = tl.maximum(x - threshold, 0.0)
        tl.store(out_ptr + b * stride_b + s * stride_s + idx_f * stride_f, y, mask=mask)
