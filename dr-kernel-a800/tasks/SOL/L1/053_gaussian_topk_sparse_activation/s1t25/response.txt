Optimization plan
- keep two-pass approach: first reduction to compute per-row sum and sum of squares, second pass to apply activation. This ensures correctness and good performance; attempting to fuse into one pass with atomics is risky for general D without careful design.
- Tune reduce_mean_std_kernel:
  - Use BLOCK_SIZE = 1024 and num_warps = 8 to maximize memory throughput for large D (e.g., 12288).
  - Vectorize loads and reduction across the feature dimension with a simple loop; Triton will generate efficient code for reductions across tl.arange lanes.
- Compute mean and std with a tiny Triton kernel (compute_mean_std_kernel): one program per row. This is already minimal; no changes needed.
- Apply activation and cast to bfloat16 in a Triton kernel:
  - Keep BLOCK_SIZE = 1024 and num_warps = 8 for high throughput.
  - Cast to bfloat16 before store to match the original function’s output dtype. Keep math in float32 for precision.
- Compute inverse normal CDF (ndtri) via A&S 5.2.23 in a tiny Triton kernel that writes to a 1-element device tensor. Launch once per forward.

These changes should provide incremental speedups by improving memory-bound performance in the main reduction and application kernels, while maintaining correctness.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def reduce_mean_std_kernel(
    X_ptr,           # *input tensor [B, S, D], contiguous
    SUM_ptr,         # *float32, length B*S
    SUMSQ_ptr,       # *float32, length B*S
    B, S, D,         # int32
    BLOCK_SIZE: tl.constexpr,
):
    # One program per (b, s) row
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    # Compute base linear index for this row in contiguous [B, S, D]
    # Layout: index = ((b * S) + s) * D
    base = (b * S + s) * D

    # Accumulate sum and sumsq in float32
    sum_val = tl.zeros((), tl.float32)
    sumsq_val = tl.zeros((), tl.float32)

    # Loop over feature dimension in chunks of BLOCK_SIZE
    for start in range(0, D, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + base + offs, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sumsq_val += tl.sum(x * x, axis=0)

    tl.store(SUM_ptr + pid, sum_val)
    tl.store(SUMSQ_ptr + pid, sumsq_val)


@triton.jit
def compute_mean_std_kernel(
    SUM_ptr,         # *float32, length B*S
    SUMSQ_ptr,       # *float32, length B*S
    MEAN_ptr,        # *float32, length B*S
    STD_ptr,         # *float32, length B*S
    D,               # int32
):
    pid = tl.program_id(axis=0)
    sum_val = tl.load(SUM_ptr + pid)
    sumsq_val = tl.load(SUMSQ_ptr + pid)
    d = tl.full((), D, tl.int32)
    mean = sum_val / d
    var = sumsq_val / d - mean * mean
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)
    tl.store(MEAN_ptr + pid, mean)
    tl.store(STD_ptr + pid, std)


@triton.jit
def ndtri_approx_kernel(
    P_ptr,           # *float32, length 1 (scalar p in [0,1])
    OUT_ptr,         # *float32, length 1 (output z-score)
):
    # Abramowitz & Stegun 5.2.23 approximation
    p = tl.load(P_ptr)
    # Regions
    p_low = 0.02425
    p_high = 1.0 - p_low

    # Constants for lower region
    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    # Constants for central region
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    # Constants for upper region
    # Same as lower, but with negative sign for z
    # We will compute q = sqrt(-2*log(1-p)) for upper region and use the lower formula.
    # Branch in Triton:
    # lower: p < p_low
    # mid: p_high >= p >= p_low
    # upper: p > p_high

    # For exactness, we branch on p
    mask_low = p < p_low
    mask_high = p > p_high

    # Initialize output as 0
    z = tl.zeros((), tl.float32)

    # Lower region
    # Compute q = sqrt(-2*log(p)) for lower region
    # Note: Triton supports tl.log and tl.sqrt
    q_low = tl.sqrt(-2.0 * tl.log(p))
    poly_low = (((((c1 * q_low + c2) * q_low + c3) * q_low + c4) * q_low + c5) * q_low + c6)
    den_low = (((((d1 * q_low + d2) * q_low + d3) * q_low + d4) * q_low + 1.0))
    z_low = poly_low / den_low

    # Upper region
    q_high = tl.sqrt(-2.0 * tl.log(1.0 - p))
    poly_high = (((((c1 * q_high + c2) * q_high + c3) * q_high + c4) * q_high + c5) * q_high + c6)
    den_high = (((((d1 * q_high + d2) * q_high + d3) * q_high + d4) * q_high + 1.0))
    z_high = -poly_high / den_high

    # Central region
    # q = p - 0.5
    q_mid = p - 0.5
    r_mid = q_mid * q_mid
    poly_mid = (((((a1 * r_mid + a2) * r_mid + a3) * r_mid + a4) * r_mid + a5) * r_mid + a6) * q_mid
    den_mid = (((((b1 * r_mid + b2) * r_mid + b3) * r_mid + b4) * r_mid + b5) * r_mid + 1.0)
    z_mid = poly_mid / den_mid

    # Select region
    # Triton will evaluate all branches; we rely on scalar selection
    # If p < p_low: z = z_low
    # elif p > p_high: z = z_high
    # else: z = z_mid
    z = tl.where(mask_low, z_low, z)
    z = tl.where(mask_high, z_high, z)
    z = tl.where((~mask_low) & (~mask_high), z_mid, z)  # mid region

    tl.store(OUT_ptr, z)


@triton.jit
def apply_activation_kernel(
    X_ptr,           # *input tensor [B, S, D], contiguous (input dtype)
    MEAN_ptr,        # *float32, length B*S
    STD_ptr,         # *float32, length B*S
    Z_ptr,           # *float32, length 1 (scalar z-score)
    OUT_ptr,         # *bfloat16, output [B, S, D]
    B, S, D,         # int32
    BLOCK_SIZE: tl.constexpr,
):
    # 3D grid over (B, S, tiles of D)
    b = tl.program_id(axis=0)
    s = tl.program_id(axis=1)
    tile = tl.program_id(axis=2)

    base = b * S + s
    base_idx = base * D

    offs = tile * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < D

    # Load per-row mean and std (float32)
    mean = tl.load(MEAN_ptr + base)
    std = tl.load(STD_ptr + base)
    z_score = tl.load(Z_ptr)  # scalar float32
    threshold = mean + std * z_score

    # Load input, apply activation (float32 math), cast to bfloat16, store
    x = tl.load(X_ptr + base_idx + offs, mask=mask, other=0.0).to(tl.float32)
    y = x - threshold  # scalar broadcast
    y = tl.maximum(y, 0.0)  # ReLU
    y_bf16 = y.to(tl.bfloat16)
    tl.store(OUT_ptr + base_idx + offs, y_bf16, mask=mask)


def _ceil_div(a, b):
    return (a + b - 1) // b


def run(inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
    """
    Triton-optimized version of the original function.
    - Computes per-row mean and std across the last dimension (features).
    - Uses inverse-normal CDF (A&S 5.2.23) to get z_score for target sparsity.
    - Applies activation: y = max(0, x - (mean + std * z_score)).
    - Returns output in bfloat16, matching the original function.
    """
    assert inputs.dim() == 3, "inputs must be of shape [batch_size, seq_len, intermediate_size]"
    assert target_sparsity >= 0.0 and target_sparsity <= 1.0, "target_sparsity must be in [0, 1]"

    B, S, D = inputs.shape

    # Ensure contiguous input
    x = inputs.contiguous()
    device = x.device

    # Allocate buffers for sums (float32)
    sum_buf = torch.empty(B * S, dtype=torch.float32, device=device)
    sumsq_buf = torch.empty(B * S, dtype=torch.float32, device=device)

    # Launch reduction kernel: one program per (b, s)
    grid_reduce = (B * S,)
    reduce_mean_std_kernel[grid_reduce](
        x, sum_buf, sumsq_buf, B, S, D, BLOCK_SIZE=1024, num_warps=8
    )

    # Allocate mean and std buffers (float32)
    mean_buf = torch.empty(B * S, dtype=torch.float32, device=device)
    std_buf = torch.empty(B * S, dtype=torch.float32, device=device)

    # Launch compute mean/std kernel
    compute_mean_std_kernel[(B * S,)](
        sum_buf, sumsq_buf, mean_buf, std_buf, D
    )

    # Prepare p tensor for ndtri (device scalar tensor)
    p_tensor = torch.tensor([float(target_sparsity)], dtype=torch.float32, device=device)
    z_buf = torch.empty(1, dtype=torch.float32, device=device)

    # Launch ndtri approximation kernel
    ndtri_approx_kernel[(1,)](p_tensor, z_buf)

    # Allocate output in bfloat16
    out = torch.empty((B, S, D), dtype=torch.bfloat16, device=device)

    # Launch apply activation kernel: 3D grid over (B, S, tiles of D)
    grid_apply = (B, S, _ceil_div(D, 1024))
    apply_activation_kernel[grid_apply](
        x, mean_buf, std_buf, z_buf, out, B, S, D, BLOCK_SIZE=1024, num_warps=8
    )

    return out


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # Expect a single 3D tensor input: [batch_size, seq_len, intermediate_size]
        if len(args) != 1:
            raise RuntimeError("ModelNew.forward expects a single 3D tensor input.")
        inputs = args[0]
        target_sparsity = 0.5  # same default as original code
        return run(inputs, target_sparsity)
```
