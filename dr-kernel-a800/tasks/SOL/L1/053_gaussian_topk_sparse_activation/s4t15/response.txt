import torch
import triton
import triton.language as tl


@triton.jit
def _row_reduce_mean_std_kernel(inp_ptr, mean_ptr, std_ptr, N, BLOCK_SIZE: tl.constexpr):
    """
    For each (batch, seq) row, compute mean and std across the last dimension of length N.
    inp_ptr: *f32, shape [B, S, N], contiguous
    mean_ptr: *f32, shape [B*S]
    std_ptr: *f32, shape [B*S]
    N: int, length of last dim
    """
    pid = tl.program_id(0)
    row_start = pid * N

    # Accumulators in f32
    sum_x = 0.0
    sum_x2 = 0.0

    # Number of chunks and loop
    num_chunks = (N + BLOCK_SIZE - 1) // BLOCK_SIZE
    for chunk in range(0, num_chunks):
        off = chunk * BLOCK_SIZE
        idx = off + tl.arange(0, BLOCK_SIZE)
        mask = idx < N
        x = tl.load(inp_ptr + row_start + idx, mask=mask, other=0.0)
        sum_x += tl.sum(x, axis=0)
        sum_x2 += tl.sum(x * x, axis=0)

    n = N
    mean = sum_x / n
    var = sum_x2 / n - mean * mean  # population variance (unbiased=False)
    std = tl.sqrt(var)

    # Write results
    tl.store(mean_ptr + pid, mean)
    tl.store(std_ptr + pid, std)


@triton.jit
def _ndtri_scalar_kernel(p_dev_ptr, out_ptr):
    """
    Compute inverse standard normal CDF for scalar p via A&S approximation (7.1.26).
    Stores the result to out_ptr[0].
    p_dev_ptr: *f32, shape [1] (device scalar tensor; kernel loads it)
    out_ptr: *f32, shape [1] (output buffer for scalar result)
    """

    p = tl.load(p_dev_ptr)

    # Constants for A&S approximation
    # Lower region approximation for p in [0, 0.5)
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    # Polynomial c(z), d(z) for z = sqrt(-2 * log(p))
    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p_high = 1.0 - p_low

    # If p < p_low, use lower region approximation (as in original code)
    use_low = p < p_low

    # Compute z = sqrt(-2 * log(p)) for lower region, else use central or upper
    # We need to guard log domain: for p<=0, log undefined; here p is in (0,1) due to CDF.
    # In Triton, tl.log supports positive inputs; p>0 in typical cases. We proceed.
    if use_low:
        z = tl.sqrt(-2.0 * tl.log(p))
        t = z
        poly_c = (((((c1 * t + c2) * t + c3) * t + c4) * t + c5) * t + c6)
        poly_d = (((((d1 * t + d2) * t + d3) * t + d4) * t + 1.0))
        nd = poly_c / poly_d
        result = -z - nd
    else:
        # Central region: p in [p_low, p_high]
        # If p > 0.5, use symmetry
        if p > 0.5:
            p_cent = 1.0 - p
        else:
            p_cent = p
        z = tl.sqrt(-2.0 * tl.log(p_cent))
        t = z
        # Approximation of erf(z/2): use polynomial with t (z)
        # Note: erf(z/2) ≈ sign(p - 0.5) * (1 - poly_erf(2z)) from Abramowitz & Stegun.
        # Here, we implement the central formula directly:
        poly_a = (((((a1 * t * t + a2) * t * t + a3) * t * t + a4) * t * t + a5) * t * t + a6) * z
        poly_b = (((((b1 * t * t + b2) * t * t + b3) * t * t + b4) * t * t + b5) * t * t + 1.0)
        nd = poly_a / poly_b
        # For p > 0.5, map back: erf(p-0.5) ≈ erf(1-p) as used, but our formula already accounts for p>0.5 via p_cent.
        # Result has correct sign already.
        result = nd if p <= 0.5 else -nd

    # Store the scalar result
    tl.store(out_ptr, result)

def _sparsify_row_kernel(inp_ptr, mean_ptr, std_ptr, threshold_scale_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise sparsification per row:
      cutoff = mean + std * threshold_scale
      out[i] = max(0, inp[i] - cutoff)
    inp_ptr: *f32, shape [B, S, N]
    mean_ptr: *f32, shape [B*S]
    std_ptr: *f32, shape [B*S]
    threshold_scale_ptr: *f32, shape [1] (device scalar with inv-std-normal for target_sparsity)
    out_ptr: *f32, shape [B, S, N]
    N: int, length of last dim
    """
    pid = tl.program_id(0)
    row_start = pid * N
    mean = tl.load(mean_ptr + pid)
    std = tl.load(std_ptr + pid)
    scale = tl.load(threshold_scale_ptr)  # scalar
    cutoff = mean + std * scale

    for off in range(0, N, BLOCK_SIZE):
        idx = off + tl.arange(0, BLOCK_SIZE)
        mask = idx < N
        x = tl.load(inp_ptr + row_start + idx, mask=mask, other=0.0)
        y = x - cutoff
        y = tl.maximum(y, 0.0)  # ReLU
        tl.store(out_ptr + row_start + idx, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        """
        Triton-optimized forward:
          - If target_sparsity == 0.0: return inputs.
          - Else: compute per-row mean/std across last dim, compute inv-std-normal multiplier via Triton,
                  then apply adaptive threshold ReLU in Triton.
        """
        if target_sparsity == 0.0:
            return inputs

        # Ensure contiguous float32 for computation
        inputs = inputs.contiguous()
        inputs_f32 = inputs.to(torch.float32)

        B, S, N = inputs_f32.shape
        # Allocate per-row stats

        std = torch.empty(B * S, dtype=torch.float32, device=inputs.device)

        # Launch sparsification kernel
        _sparsify_row_kernel[grid](inputs_f32, mean, std, std_multiplier, out, N, BLOCK_SIZE=BLOCK_SIZE, num_warps=8, num_stages=4)

        # Cast back to original dtype if needed
        return out.to(inputs.dtype)

      cutoff = mean + std * threshold_scale
      out[i] = max(0, inp[i] - cutoff)
    inp_ptr: *f32, shape [B, S, N]
    sum_ptr: *f32, shape [B*S]
    sumsq_ptr: *f32, shape [B*S]
    threshold_scale_ptr: *f32, shape [1] (device scalar with inv-std-normal for target_sparsity)
    out_ptr: *f32, shape [B, S, N]
    N: int, length of last dim
    """
    pid = tl.program_id(0)
    row_start = pid * N
    sum_x = tl.load(sum_ptr + pid)
    sum_x2 = tl.load(sumsq_ptr + pid)
    n = N
    mean = sum_x / n
    var = sum_x2 / n - mean * mean  # population variance (unbiased=False)
    std = tl.sqrt(var)
    scale = tl.load(threshold_scale_ptr)  # scalar
    cutoff = mean + std * scale

    for off in range(0, N, BLOCK_SIZE):
        idx = off + tl.arange(0, BLOCK_SIZE)
        mask = idx < N
        x = tl.load(inp_ptr + row_start + idx, mask=mask, other=0.0)
        y = x - cutoff
        y = tl.maximum(y, 0.0)  # ReLU
        tl.store(out_ptr + row_start + idx, y, mask=mask)
