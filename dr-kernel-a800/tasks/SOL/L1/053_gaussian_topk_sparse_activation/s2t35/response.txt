import triton
import triton.language as tl


@triton.jit
def sparsity_row_kernel(x_ptr, out_ptr,
                         B, S, N, p,
                         target_sparsity: tl.constexpr,
                         BLOCK: tl.constexpr):
    """
    One program per row (b, s).
    Computes mean and std across the N elements of that row in float32,
    then applies gating: out = max(0, x - (mean + std * invphi(p))).
    Writes float32 output. Forward will cast to bfloat16.
    """
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    # Base linear index for the row
    base = (b * S + s) * N

    # First pass: compute sum and sum of squares
    sum_val = 0.0
    sum_sq = 0.0
    offset = 0
    while offset < N:
        idx = offset + tl.arange(0, BLOCK)
        mask = idx < N
        x = tl.load(x_ptr + base + idx, mask=mask, other=0.0).to(tl.float32)
        # Reduce within the vector to scalars
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        offset += BLOCK

    # Compute mean and std (population std, unbiased=False)
    mean = sum_val / N
    var = sum_sq / N - mean * mean
    # Ensure non-negative variance due to numerical errors
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)

    # Compute inv-Phi(p) via bisection in [-6, 6] for high accuracy
    # invphi ~ 0 when p ~ 1, ~ -6 when p ~ 0. We perform 30 iterations for good precision.
    z_lo = -6.0
    z_hi = 6.0
    invphi = 0.0
    for _ in range(30):
        z = 0.5 * (z_lo + z_hi)
        # erf approximation (Abramowitz & Stegun 7.1.26)
        sign = tl.where(z >= 0.0, 1.0, -1.0)
        z_abs = tl.abs(z)
        t = 1.0 / (1.0 + 0.3275911 * z_abs)
        a1 = 0.254829592
        a2 = -0.284496736
        a3 = 1.421413741
        a4 = -1.453152027
        a5 = 1.061405429
        poly = (((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t)
        erf_z = sign * (1.0 - poly * tl.exp(-z_abs * z_abs))
        phi = 0.5 * (1.0 + erf_z)  # standard normal CDF
        # If phi > p, move z_lo up; else move z_hi down
        move_lo = phi > p
        z_hi = tl.where(move_lo, z, z_hi)
        z_lo = tl.where(move_lo, z_lo, z)
        invphi = 0.5 * (z_lo + z_hi)

    threshold = mean + std * invphi

    # Second pass: apply ReLU gating and write output
    offset = 0
    while offset < N:
        idx = offset + tl.arange(0, BLOCK)
        mask = idx < N
        x = tl.load(x_ptr + base + idx, mask=mask, other=0.0).to(tl.float32)
        out = tl.maximum(x - threshold, 0.0)
        tl.store(out_ptr + base + idx, out, mask=mask)
        offset += BLOCK


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        """
        Triton-based implementation of run(inputs, target_sparsity).
        - No torch ops in forward (except return cast).
        - Computes adaptive sparsity threshold per row and applies ReLU gating.
        - Returns tensor in bfloat16 (to match original intent).
        """
        # If no sparsity requested, return input cast to bfloat16
        if target_sparsity == 0.0:
            return x.to(torch.bfloat16)

        # Ensure contiguous float32 for kernel computation
        x_contig = x.contiguous()
        B, S, N = x_contig.shape
        x_f32 = x_contig.to(torch.float32)

        # Output buffer in float32 (kernel writes float32)
        out_f32 = torch.empty((B, S, N), dtype=torch.float32, device=x.device)

        # Launch one program per row
        grid = (B * S,)
        sparsity_row_kernel[grid](
            x_f32, out_f32,
            B, S, N, float(target_sparsity),
            target_sparsity=0.0,  # unused parameter to keep signature consistent; Triton ignores kwargs not used
            BLOCK=1024,
            num_warps=8
        )

        # Cast to bfloat16 to match original output dtype
        return out_f32.to(torch.bfloat16)
