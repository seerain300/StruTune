import torch
import triton
import triton.language as tl


@triton.jit
def _row_reduce_mean_std_kernel(
    inp_ptr,            # *const float, input tensor base
    mean_ptr,           # *float, output means, length B*S
    std_ptr,            # *float, output stds, length B*S
    N,                  # int: feature dimension size
    BLOCK_SIZE: tl.constexpr,
):
    row_id = tl.program_id(axis=0)
    # Map row_id to (batch, seq) if needed; here we assume linear rows
    # Compute base offset for this row in the flattened [B,S,N] layout
    # If we had explicit strides, we'd compute b and s, but we use linear index.
    # We'll iterate over N and load contiguous elements from inp_ptr
    # Initialize accumulators
    sum_val = 0.0
    sum_sq = 0.0

    # Loop over N in chunks
    for offs in range(0, N, BLOCK_SIZE):
        idx = offs + tl.arange(0, BLOCK_SIZE)
        mask = idx < N
        vals = tl.load(inp_ptr + row_id * N + idx, mask=mask, other=0.0)
        sum_val += tl.sum(vals, axis=0)
        sum_sq += tl.sum(vals * vals, axis=0)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    # Clamp var to non-negative to avoid tiny negative due to FP errors
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)

    # Write results
    tl.store(mean_ptr + row_id, mean)
    tl.store(std_ptr + row_id, std)


@triton.jit
def _ndtri_scalar_kernel(
    p_ptr,              # *const float, 1-element tensor with target_sparsity
    out_ptr,            # *float, 1-element tensor to store inv std norm
):
    # Load p
    p = tl.load(p_ptr)
    # Abramowitz & Stegun 7.1.26 approximation for inverse standard normal CDF
    # Note: we guard regions as in A&S
    p_low = 0.02425
    p_high = 1.0 - p_low

    # Lower region (p < p_low)
    # q = sqrt(-2*log(p)) and polynomial approximation
    # We use a where-style branching: compute both and select
    # Since Triton does not have dynamic if-return, we compute both branches.
    # For masked computation, use tl.where to select per-element, but here p is scalar.
    # Compute lower branch
    # We cannot guard on p (Triton branch on scalars limited), so just compute both.
    # However, to reduce work, we implement branching via tl.where using a scalar mask.

    # Helper to compute the polynomial approximation
    def asheness(p_sub):
        a1 = -3.969683028665376e+01
        a2 = 2.209460984245205e+02
        a3 = -2.759285104469687e+02
        a4 = 1.383577518672690e+02
        a5 = -3.066479806614716e+01
        a6 = 2.506628277459239e+00

        b1 = -5.447609879822406e+01
        b2 = 1.615858368580409e+02
        b3 = -1.556989798598866e+02
        b4 = 6.680131188771972e+01
        b5 = -1.328068155288572e+01

        c1 = -7.784894002430293e-03
        c2 = -3.223964580411365e-01
        c3 = -2.400758277161838e+00
        c4 = -2.549732539343734e+00
        c5 = 4.374664141464968e+00
        c6 = 2.938163982698783e+00

        d1 = 7.784695709041462e-03
        d2 = 3.224671290700398e-01
        d3 = 2.445134137142996e+00
        d4 = 3.754408661907416e+00

        q = tl.sqrt(-2.0 * tl.log(p_sub))
        poly_low = (((((c1 * q + c2) * q + c3) * q + c4) * q + c5) * q + c6)
        den_low = (((((d1 * q + d2) * q + d3) * q + d4) * q + 1.0))
        y_low = poly_low / den_low

        q2 = p_sub - 0.5
        r2 = q2 * q2
        poly_mid = (((((a1 * r2 + a2) * r2 + a3) * r2 + a4) * r2 + a5) * r2 + a6)
        den_mid = (((((b1 * r2 + b2) * r2 + b3) * r2 + b4) * r2 + b5) * r2 + 1.0)
        y_mid = poly_mid / den_mid

        q3 = tl.sqrt(-2.0 * tl.log(1.0 - p_sub))
        poly_high = (((((c1 * q3 + c2) * q3 + c3) * q3 + c4) * q3 + c5) * q3 + c6)
        den_high = (((((d1 * q3 + d2) * q3 + d3) * q3 + d4) * q3 + 1.0))
        y_high = - (poly_high / den_high)

        # Select based on p_sub
        # Triton does not allow branching on scalars here, so we compute via where.
        # However, since p_sub is scalar, we can evaluate masks and return appropriate value.
        # Implement selection: if p_sub < p_low: y_low; elif p_sub > p_high: y_high; else y_mid.
        # We'll use tl.where with scalar comparisons.
        y = tl.where(p_sub < p_low, y_low, tl.where(p_sub > p_high, y_high, y_mid))
        return y

    y = asheness(p)
    tl.store(out_ptr, y)


@triton.jit
def _row_sparsify_kernel(
    inp_ptr,            # *const float, input base
    out_ptr,            # *float, output base
    mean_ptr,           # *const float, means per row
    std_ptr,            # *const float, stds per row
    N,                  # int: feature dimension size
    std_multiplier,     # float: inv std norm from host
    BLOCK_SIZE: tl.constexpr,
):
    row_id = tl.program_id(axis=0)
    mean = tl.load(mean_ptr + row_id)
    std = tl.load(std_ptr + row_id)
    cutoff = mean + std * std_multiplier

    # Iterate over N in chunks, compute relu(inp - cutoff)
    for offs in range(0, N, BLOCK_SIZE):
        idx = offs + tl.arange(0, BLOCK_SIZE)
        mask = idx < N
        vals = tl.load(inp_ptr + row_id * N + idx, mask=mask, other=0.0)
        diff = vals - cutoff  # cutoff is scalar, broadcast
        # ReLU: max(diff, 0)
        diff = tl.maximum(diff, 0.0)
        tl.store(out_ptr + row_id * N + idx, diff, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        # If no sparsity requested, return original
        if target_sparsity == 0.0:
            return inputs

        # Ensure contiguous and compute in float32
        inp = inputs.contiguous().to(torch.float32)
        B, S, N = inp.shape

        # Buffers for mean and std per row
        mean_buf = torch.empty(B * S, dtype=torch.float32, device=inp.device)
        std_buf = torch.empty(B * S, dtype=torch.float32, device=inp.device)

        # Launch reduction kernel: one program per (batch, seq) row
        BLOCK_SIZE = 1024
        grid = (B * S,)
        _row_reduce_mean_std_kernel[grid](inp, mean_buf, std_buf, N, BLOCK_SIZE)

        # Compute inv std norm for target_sparsity via Triton scalar kernel
        p_dev = torch.tensor(float(target_sparsity), dtype=torch.float32, device=inp.device)
        std_multiplier_buf = torch.empty(1, dtype=torch.float32, device=inp.device)
        _ndtri_scalar_kernel[(1,)](p_dev, std_multiplier_buf)
        std_multiplier_scalar = float(std_multiplier_buf.item())

        # Output buffer
        out = torch.empty_like(inp)

        # Launch sparsification kernel
        _row_sparsify_kernel[grid](inp, out, mean_buf, std_buf, N, std_multiplier_scalar, BLOCK_SIZE)

        # Cast back to original dtype
        return out.to(inputs.dtype)
