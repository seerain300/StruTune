    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# -----------------------------
# Triton kernels
# -----------------------------
if TRITON_AVAILABLE:
    @triton.jit
    def reduce_sum_sumsq_rows_kernel(
        x_ptr,            # *const float32
        sums_ptr,         # *float32, length B*S
        sums2_ptr,        # *float32, length B*S
        B: tl.constexpr,  # batch size
        S: tl.constexpr,  # seq_len
        F,                # feature_size (intermediate dimension)
        BLOCK_SIZE: tl.constexpr,
    ):
        # One program per (b, s) row
        row_id = tl.program_id(axis=0)
        b = row_id // S
        s = row_id % S
        base = (b * S + s) * F

        acc_sum = 0.0
        acc_sumsq = 0.0

        # Loop over feature dimension in chunks
        for offs in range(0, F, BLOCK_SIZE):
            idx = offs + tl.arange(0, BLOCK_SIZE)
            mask = idx < F
            ptrs = x_ptr + base + idx
            vals = tl.load(ptrs, mask=mask, other=0.0)
            acc_sum += tl.sum(vals, axis=0)
            acc_sumsq += tl.sum(vals * vals, axis=0)

        tl.store(sums_ptr + row_id, acc_sum)
        tl.store(sums2_ptr + row_id, acc_sumsq)

    @triton.jit
    def compute_mean_std_rows_kernel(
        sums_ptr,         # *const float32, length B*S
        sums2_ptr,        # *const float32, length B*S
        mean_ptr,         # *float32, length B*S
        std_ptr,          # *float32, length B*S
        F: tl.constexpr,  # feature_size
    ):
        row_id = tl.program_id(axis=0)
        sum_row = tl.load(sums_ptr + row_id)
        sumsq_row = tl.load(sums2_ptr + row_id)
        mean = sum_row / F
        var = sumsq_row / F - mean * mean
        var = tl.maximum(var, 0.0)  # clamp to non-negative
        std = tl.sqrt(var)
        tl.store(mean_ptr + row_id, mean)
        tl.store(std_ptr + row_id, std)

    @triton.jit
    def inv_normal_cdf_scalar_kernel(
        out_ptr,          # *float32, single-element output
        p,                # float32 scalar in (0,1)
        # Abramowitz & Stegun 7.1.26 coefficients for normal PPF
        a1 = -3.969683028665376e+01,
        a2 = 2.209460984245205e+02,
        a3 = -2.759285104469687e+02,
        a4 = 1.383577518672690e+02,
        a5 = -3.066479806614716e+01,
        a6 = 2.506628277459239e+00,
        b1 = -5.447609879822406e+01,
        b2 = 1.615858368580409e+02,
        b3 = -1.556989798598866e+02,
        b4 = 6.680131188771972e+01,
        b5 = -1.328068155288572e+01,
        c1 = -7.784894002430293e-03,
        c2 = -3.223964580411365e-01,
        c3 = -2.400758277161838e+00,
        c4 = -2.549732539343734e+00,
        c5 = 4.374664141464968e+00,
        c6 = 2.938163982698783e+00,
        d1 = 7.784695709041462e-03,
        d2 = 3.224671290700398e-01,
        d3 = 2.445134137142996e+00,
        d4 = 3.754408661907416e+00,
    ):
        # Compute x = sqrt(2) * sqrt(-log(p))
        # log and sqrt are available in Triton
        log_p = tl.log(p)
        x = tl.sqrt(-2.0 * log_p)

        # Lower region approximation: p < 0.02425
        # We will branch in Triton; p and bounds are scalars.
        p_low = 0.02425

        # Compute poly_c and poly_d
        # poly_c(x) = c1*x + c2*x^2 + ... + c6*x^6
        # poly_d(x) = d1*x + d2*x^2 + ... + d4*x^4
        x2 = x * x
        poly_c = c1 * x
        poly_c = poly_c + c2 * x
        poly_c = poly_c + c3 * x2
        poly_c = poly_c + c4 * x2
        poly_c = poly_c + c5 * x2
        poly_c = poly_c + c6 * x2

        x3 = x2 * x
        poly_d = d1 * x
        poly_d = poly_d + d2 * x2
        poly_d = poly_d + d3 * x2
        poly_d = poly_d + d4 * x3

        # Central region approximation for p in [low, 1-low]
        p_high = 1.0 - p_low

        # poly_a and poly_b are degree-5 polynomials in q = p - 0.5
        # poly_a(q) = a1*q + a2*q^2 + ... + a6*q^6
        # poly_b(q) = b1*q + b2*q^2 + ... + b5*q^5
        q = 0.0  # placeholder; will be set below
        # We'll compute q and corresponding polynomials when p is in mid region.

        # Precompute q_mid for central region to ensure type consistency
        q = p - 0.5
        q2 = q * q

        poly_a = a1 * q
        poly_a = poly_a + a2 * q
        poly_a = poly_a + a3 * q2
        poly_a = poly_a + a4 * q2
        poly_a = poly_a + a5 * q2
        poly_a = poly_a + a6 * q2

        q3 = q2 * q
        q4 = q3 * q
        poly_b = b1 * q
        poly_b = poly_b + b2 * q2
        poly_b = poly_b + b3 * q2
        poly_b = poly_b + b4 * q3
        poly_b = poly_b + b5 * q4

        # Choose the appropriate branch using scalar conditions
        # For lower region:
        if p < p_low:
            result = poly_c / (poly_d + 1.0)
        # For upper region:
        elif p > p_high:
            # For upper region, use symmetry: ndtri(1-p) = -ndtri(p)
            # Since we already computed x for p, just negate the central result's x?
            # Better: compute the same formulas for 1-p and negate the final result.
            # But since we only need scalar, we can directly evaluate the central formula for 1-p and negate.
            p1 = 1.0 - p
            x1 = tl.sqrt(-2.0 * tl.log(p1))
            q1 = p1 - 0.5
            q2_1 = q1 * q1
            poly_a1 = a1 * q1
            poly_a1 = poly_a1 + a2 * q1
            poly_a1 = poly_a1 + a3 * q2_1
            poly_a1 = poly_a1 + a4 * q2_1
            poly_a1 = poly_a1 + a5 * q2_1
            poly_a1 = poly_a1 + a6 * q2_1
            q3_1 = q2_1 * q1
            q4_1 = q3_1 * q1
            poly_b1 = b1 * q1
            poly_b1 = poly_b1 + b2 * q2_1
            poly_b1 = poly_b1 + b3 * q2_1
            poly_b1 = poly_b1 + b4 * q3_1
            poly_b1 = poly_b1 + b5 * q4_1
            result1 = poly_a1 / (poly_b1 + 1.0)
            result = -result1
        else:
            result = (poly_a / (poly_b + 1.0))

        # Store result to out_ptr[0]
        tl.store(out_ptr, result)

    @triton.jit
    def sparsify_relu_kernel(
        x_ptr,            # *const float32 input
        out_ptr,          # *float32 output
        mean_ptr,         # *const float32, length B*S
        std_ptr,          # *const float32, length B*S
        threshold_scale,  # float32 scalar multiplier
        B: tl.constexpr,
        S: tl.constexpr,
        F: tl.constexpr,
        total_elems,      # int32
        BLOCK_SIZE: tl.constexpr,
    ):
        # 1D tiling over all elements
        pid = tl.program_id(axis=0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < total_elems

        SF = S * F
        b = offs // SF
        rem = offs % SF
        s = rem // F
        f = rem % F

        x_ptrs = x_ptr + b * SF + s * F + f
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)

        row_id = b * S + s
        mean = tl.load(mean_ptr + row_id, mask=mask, other=0.0)
        std = tl.load(std_ptr + row_id, mask=mask, other=0.0)

        threshold = mean + std * threshold_scale
        y = x_vals - threshold
        y = tl.maximum(y, 0.0)

        out_ptrs = out_ptr + b * SF + s * F + f
        tl.store(out_ptrs, y, mask=mask)

class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # No state needed; all work is in Triton kernels.

    @torch.no_grad()
    def forward(self, inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        # If Triton is unavailable or inputs are not CUDA, fall back to PyTorch (not allowed in evaluation).
        assert TRITON_AVAILABLE, "Triton is not available"
        assert inputs.is_cuda, "Input must be on CUDA for Triton execution"

        # Ensure contiguous and float32 for stats
        x = inputs.to(torch.float32).contiguous()
        B, S, F = x.shape
        device = x.device

        # Allocate per-(batch, seq) sums and sums2
        sums = torch.empty(B * S, dtype=torch.float32, device=device)
        sums2 = torch.empty(B * S, dtype=torch.float32, device=device)
        mean = torch.empty(B * S, dtype=torch.float32, device=device)
        std = torch.empty(B * S, dtype=torch.float32, device=device)

        # 1) Reduce to per-row sum and sum of squares
        BLOCK_SIZE_RED = 1024
        grid_red = (B * S,)
        reduce_sum_sumsq_rows_kernel[grid_red](
            x, sums, sums2, B, S, F, BLOCK_SIZE=BLOCK_SIZE_RED
        )

        # 2) Compute mean and std per (b, s) row
        grid_stats = (B * S,)
        compute_mean_std_rows_kernel[grid_stats](sums, sums2, mean, std, F=F)

        # 3) Compute inverse normal CDF for target_sparsity in Triton
        # Create a single-element output buffer on device. Use p = target_sparsity as a scalar.
        multiplier = torch.empty(1, dtype=torch.float32, device=device)
        # Pass p as a scalar in the kernel (compile-time constant is not allowed to be arbitrary; Triton treats
        # function arguments as runtime values. We simply pass the Python float.)
        inv_normal_cdf_scalar_kernel[(1,)](multiplier, target_sparsity)

        # 4) Apply sparsity: out = max(0, x - (mean + std * multiplier[0]))
        total_elems = B * S * F
        out_fp32 = torch.empty((B, S, F), dtype=torch.float32, device=device)
        BLOCK_SIZE_POINT = 1024
        grid_point = (triton.cdiv(total_elems, BLOCK_SIZE_POINT),)
        sparsify_relu_kernel[grid_point](
            x, out_fp32, mean, std, multiplier[0], B, S, F, total_elems, BLOCK_SIZE=BLOCK_SIZE_POINT
        )

        # Return as bfloat16 to match original behavior
        return out_fp32.to(torch.bfloat16)
