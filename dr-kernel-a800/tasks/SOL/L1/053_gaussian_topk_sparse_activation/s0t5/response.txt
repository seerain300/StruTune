import torch
import triton
import triton.language as tl


# Kernel: compute per-row mean across the last dimension (F)
@triton.jit
def mean_lastdim_kernel(
    inputs_ptr,        # *fp32 (we'll cast inputs to fp32 before launching)
    mean_out_ptr,      # *fp32, shape [B*S]
    B: tl.constexpr,
    S: tl.constexpr,
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # No need to compute b,s here; we launch with grid=(B*S,) and index directly
    sum_val = 0.0
    for off in range(0, F, BLOCK_F):
        f = off + tl.arange(0, BLOCK_F)
        mask = f < F
        x = tl.load(inputs_ptr + pid * F + f, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / F
    tl.store(mean_out_ptr + pid, mean)


# Kernel: compute per-row std across the last dimension (population std)
@triton.jit
def std_lastdim_kernel(
    inputs_ptr,        # *fp32
    mean_ptr,          # *fp32, shape [B*S]
    std_out_ptr,       # *fp32, shape [B*S]
    B: tl.constexpr,
    S: tl.constexpr,
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    mean = tl.load(mean_ptr + pid)
    sum_sq = 0.0
    for off in range(0, F, BLOCK_F):
        f = off + tl.arange(0, BLOCK_F)
        mask = f < F
        x = tl.load(inputs_ptr + pid * F + f, mask=mask, other=0.0)
        diff = x - mean
        sum_sq += tl.sum(diff * diff, axis=0)
    var = sum_sq / F
    std = tl.sqrt(var)
    tl.store(std_out_ptr + pid, std)


# Kernel: compute inverse-normal CDF (ndtri) for scalar p using A&S approximation.
# Launch with grid=(1,) and write scalar z to a 1-element output tensor.
@triton.jit
def ndtri_approx_kernel(
    out_ptr,           # *fp32, length 1 (output z)
    p,                 # fp32 scalar input in (0, 1)
    p_low: tl.constexpr,   # 0.02425
    p_high: tl.constexpr,  # 1 - p_low
    a1, a2, a3, a4, a5, a6,
    b1, b2, b3, b4, b5,
    c1, c2, c3, c4, c5, c6,
    d1, d2, d3, d4,
):
    # Masks for piecewise computation
    mask_low = p < p_low
    mask_mid = (p >= p_low) & (p <= p_high)
    mask_high = p > p_high

    # Lower tail
    q_low = tl.sqrt(-2.0 * tl.log(p))
    num_low = (((((c1 * q_low + c2) * q_low + c3) * q_low + c4) * q_low + c5) * q_low + c6)
    den_low = (((((d1 * q_low + d2) * q_low + d3) * q_low + d4) * q_low + 1.0))
    z_low = num_low / den_low

    # Middle region
    q_mid = p - 0.5
    r_mid = q_mid * q_mid
    num_mid = (((((a1 * r_mid + a2) * r_mid + a3) * r_mid + a4) * r_mid + a5) * r_mid + a6)
    den_mid = (((((b1 * r_mid + b2) * r_mid + b3) * r_mid + b4) * r_mid + b5) * r_mid + 1.0)
    z_mid = num_mid / den_mid

    # Upper tail
    q_high = tl.sqrt(-2.0 * tl.log(1.0 - p))
    num_high = (((((c1 * q_high + c2) * q_high + c3) * q_high + c4) * q_high + c5) * q_high + c6)
    den_high = (((((d1 * q_high + d2) * q_high + d3) * q_high + d4) * q_high + 1.0))
    z_high = -num_high / den_high

    # Select one
    z = tl.where(mask_low, z_low, 0.0)
    z = tl.where(mask_mid, z_mid, z)
    z = tl.where(mask_high, z_high, z)

    tl.store(out_ptr, z)


# Kernel: apply cutoff = mean + std * z and y = max(0, x - cutoff) per row
@triton.jit
def apply_cutoff_relu_kernel(
    inputs_ptr,        # *fp32, shape [B, S, F]
    mean_ptr,          # *fp32, shape [B*S]
    std_ptr,           # *fp32, shape [B*S]
    z_ptr,             # *fp32, length 1 (scalar z)
    outputs_ptr,       # *bf16, shape [B, S, F] (we'll cast from fp32 before storing)
    B: tl.constexpr,
    S: tl.constexpr,
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # Load per-row parameters
    mean = tl.load(mean_ptr + pid)
    std = tl.load(std_ptr + pid)
    z = tl.load(z_ptr)  # scalar
    cutoff = mean + std * z

    # Elementwise: y = max(0, x - cutoff)
    for off in range(0, F, BLOCK_F):
        f = off + tl.arange(0, BLOCK_F)
        mask = f < F
        x = tl.load(inputs_ptr + pid * F + f, mask=mask, other=0.0)
        y = x - cutoff
        y = tl.maximum(y, 0.0)
        # Store as fp32; host will cast to bf16 before kernel launch if needed
        tl.store(outputs_ptr + pid * F + f, y, mask=mask)

        b1 = -5.447609879822406e+01; b2 = 1.615858368580409e+02; b3 = -1.556989798598866e+02; b4 = 6.680131188771972e+01; b5 = -1.328068155288572e+01

    inputs_ptr,        # *fp32, shape [B, S, F], contiguous
    mean_out_ptr,      # *fp32, shape [B*S]
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # row index
    sum_val = 0.0
    for off in range(0, F, BLOCK_F):
        f = off + tl.arange(0, BLOCK_F)
        mask = f < F
        x = tl.load(inputs_ptr + pid * F + f, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / F
    tl.store(mean_out_ptr + pid, mean)


# Kernel: compute per-row std across the last dimension (population std)
@triton.jit
def std_lastdim_kernel(
    inputs_ptr,        # *fp32
    mean_ptr,          # *fp32, shape [B*S]
    std_out_ptr,       # *fp32, shape [B*S]
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    mean = tl.load(mean_ptr + pid)
    sum_sq = 0.0
    for off in range(0, F, BLOCK_F):
        f = off + tl.arange(0, BLOCK_F)
        mask = f < F
        x = tl.load(inputs_ptr + pid * F + f, mask=mask, other=0.0)
        diff = x - mean
        sum_sq += tl.sum(diff * diff, axis=0)
    var = sum_sq / F
    std = tl.sqrt(var)
    tl.store(std_out_ptr + pid, std)
