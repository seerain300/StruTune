import torch
import triton
import triton.language as tl


@triton.jit
def mean_lastdim_kernel(x_ptr, means_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    total = tl.zeros((), dtype=tl.float32)
    # Iterate over the feature dimension in chunks
    for i in range(0, F, BLOCK_F):
        offs = i + tl.arange(0, BLOCK_F)
        mask = offs < F
        ptrs = x_ptr + row_offset + offs
        x = tl.load(ptrs, mask=mask, other=0.0)
        total += tl.sum(x, axis=0)
    mean = total / F
    tl.store(means_ptr + pid, mean)


@triton.jit
def std_lastdim_kernel(x_ptr, stds_ptr, means_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    mean = tl.load(means_ptr + pid)
    sum_sq = tl.zeros((), dtype=tl.float32)
    for i in range(0, F, BLOCK_F):
        offs = i + tl.arange(0, BLOCK_F)
        mask = offs < F
        ptrs = x_ptr + row_offset + offs
        x = tl.load(ptrs, mask=mask, other=0.0)
        diff = x - mean
        sum_sq += tl.sum(diff * diff, axis=0)
    # population std (unbiased=False): sqrt(sum_sq / F)
    std = tl.sqrt(sum_sq / F)
    tl.store(stds_ptr + pid, std)

def ndtri_approx_kernel(p, z_ptr, a1, a2, a3, a4, a5, a6,
                         b1, b2, b3, b4, b5,
                         c1, c2, c3, c4, c5, c6,
                         d1, d2, d3, d4,
                         p_low, p_high):

    # Constants for lower, central, and upper regions
    q_low = tl.sqrt(-2.0 * tl.log(p))  # for p < p_low
    t_low = c1 * q_low + c2

    t3_low = t2_low * q_low
    t4_low = t3_low * q_low
    t5_low = t4_low * q_low

    den_low = (d1 * q_low + d2) * q_low + d3
    den_low = den_low * q_low + d4
    den_low = den_low * q_low + 1.0
    z_low = num_low / den_low

    q_mid = p - 0.5
    r_mid = q_mid * q_mid
    num_mid = ((a1 * r_mid + a2) * r_mid + a3) * r_mid + a4
    num_mid = num_mid * r_mid + a5
    num_mid = num_mid * r_mid + a6
    den_mid = ((b1 * r_mid + b2) * r_mid + b3) * r_mid + b4
    den_mid = den_mid * r_mid + b5
    den_mid = den_mid * r_mid + 1.0

    q_high = tl.sqrt(-2.0 * tl.log(1.0 - p))
    t_high = c1 * q_high + c2
    t2_high = t_high * q_high
    t3_high = t2_high * q_high
    t4_high = t3_high * q_high
    t5_high = t4_high * q_high
    num_high = t5_high + c6
    den_high = (d1 * q_high + d2) * q_high + d3
    den_high = den_high * q_high + d4

        y = tl.maximum(y, 0.0)
        tl.store(out_ptr + row_offset + offs, y, mask=mask)

  - Increase num_stages to 3 to better pipeline loads/stores for these memory-bound kernels.

- Correctness:
  - std computed with unbiased=False (population std), matching torch.std(..., unbiased=False).
  - ndtri uses the A&S 5.2.23 piecewise approximation exactly.
  - Compute in float32; cast final output to bfloat16 to match original behavior.

This keeps all computation in Triton, avoids any host-side torch tensor creation, and should provide consistent or slightly better performance across diverse shapes.

import torch
import triton
import triton.language as tl


@triton.jit
def mean_lastdim_kernel(x_ptr, means_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    total = tl.zeros((), dtype=tl.float32)
    # Iterate over the feature dimension in chunks of BLOCK_F
    for i in range(0, F, BLOCK_F):
        offs = i + tl.arange(0, BLOCK_F)
        mask = offs < F
        ptrs = x_ptr + row_offset + offs
        x = tl.load(ptrs, mask=mask, other=0.0)
        total += tl.sum(x, axis=0)
    mean = total / F
    tl.store(means_ptr + pid, mean)


@triton.jit
def std_lastdim_kernel(x_ptr, stds_ptr, means_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    mean = tl.load(means_ptr + pid)
    sum_sq = tl.zeros((), dtype=tl.float32)
    for i in range(0, F, BLOCK_F):
        offs = i + tl.arange(0, BLOCK_F)
        mask = offs < F
        ptrs = x_ptr + row_offset + offs
        x = tl.load(ptrs, mask=mask, other=0.0)
        diff = x - mean
        sum_sq += tl.sum(diff * diff, axis=0)
    # population std (unbiased=False): sqrt(sum_sq / F)
    std = tl.sqrt(sum_sq / F)
    tl.store(stds_ptr + pid, std)


@triton.jit
def ndtri_approx_kernel(p, z_ptr, a1, a2, a3, a4, a5, a6,
                         b1, b2, b3, b4, b5,
                         c1, c2, c3, c4, c5, c6,
                         d1, d2, d3, d4,
                         p_low, p_high):
    # Abramowitz & Stegun 5.2.23 approximation (scalar p -> scalar z)
    # Lower region
    q_low = tl.sqrt(-2.0 * tl.log(p))
    t_low = c1 * q_low + c2
    t2_low = t_low * q_low
    t3_low = t2_low * q_low
    t4_low = t3_low * q_low
    t5_low = t4_low * q_low
    num_low = t5_low + c6
    den_low = (d1 * q_low + d2) * q_low + d3
    den_low = den_low * q_low + d4
    den_low = den_low * q_low + 1.0
    z_low = num_low / den_low

    # Central region
    q_mid = p - 0.5
    r_mid = q_mid * q_mid
    num_mid = ((a1 * r_mid + a2) * r_mid + a3) * r_mid + a4
    num_mid = num_mid * r_mid + a5
    num_mid = num_mid * r_mid + a6
    den_mid = ((b1 * r_mid + b2) * r_mid + b3) * r_mid + b4
    den_mid = den_mid * r_mid + b5
    den_mid = den_mid * r_mid + 1.0
    z_mid = num_mid * q_mid / den_mid

    # Upper region
    q_high = tl.sqrt(-2.0 * tl.log(1.0 - p))
    t_high = c1 * q_high + c2
    t2_high = t_high * q_high
    t3_high = t2_high * q_high
    t4_high = t3_high * q_high
    t5_high = t4_high * q_high
    num_high = t5_high + c6
    den_high = (d1 * q_high + d2) * q_high + d3
    den_high = den_high * q_high + d4
    den_high = den_high * q_high + 1.0
    z_high = -num_high / den_high

    mask_low = p < p_low
    mask_mid = (p >= p_low) & (p <= p_high)
    mask_high = p > p_high
    z = tl.where(mask_low, z_low, 0.0) + tl.where(mask_mid, z_mid, 0.0) + tl.where(mask_high, z_high, 0.0)
    tl.store(z_ptr, z)
