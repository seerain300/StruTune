- The inverse-normal CDF (_ndtri) is computed using the Abramowitz & Stegun 7.1.26 approximation. - Dtype handling: The original code converts inputs to float32, computes statistics, applies threshold, then returns bfloat16. I will perform all computation in float32 inside Triton, and write the final output as bfloat16 to match the original.

Plan:
- Reductions:
  - sum_rows_kernel: One program per row (b, s). Iterate over the feature dimension F in chunks of BLOCK_F. Load elements, cast to float32, accumulate sum in registers, atomic_add into a per-row sum output.
  - sumsq_rows_kernel: Same, accumulate sum of squares.
- compute_stats_kernel: One program per row. Load sum and sumsq, compute mean = sum/F, var = sumsq/F - mean^2, var = max(var, 0), std = sqrt(var). Store mean and std per row.

import torch
import triton
import triton.language as tl


# Kernel: compute per-(b, s) row sums across feature dimension F
@triton.jit
def sum_rows_kernel(input_ptr, out_sum_ptr,
                    B, S, F, stride_b, stride_s, stride_f,
                    BLOCK_F: tl.constexpr):
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    base = b * stride_b + s * stride_s

    sum_val = 0.0
    for off in range(0, F, BLOCK_F):
        idx = off + tl.arange(0, BLOCK_F)
        mask = idx < F
        ptrs = input_ptr + base + idx * stride_f
        vals = tl.load(ptrs, mask=mask, other=0.0)
        vals = vals.to(tl.float32)
        sum_val += tl.sum(vals, axis=0)
    tl.atomic_add(out_sum_ptr + pid, sum_val)


# Kernel: compute per-(b, s) row sum of squares across feature dimension F
@triton.jit
def sumsq_rows_kernel(input_ptr, out_sumsq_ptr,
                      B, S, F, stride_b, stride_s, stride_f,
                      BLOCK_F: tl.constexpr):
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    base = b * stride_b + s * stride_s

    sumsq_val = 0.0
    for off in range(0, F, BLOCK_F):
        idx = off + tl.arange(0, BLOCK_F)
        mask = idx < F
        ptrs = input_ptr + base + idx * stride_f
        vals = tl.load(ptrs, mask=mask, other=0.0)
        vals = vals.to(tl.float32)
        sumsq_val += tl.sum(vals * vals, axis=0)
    tl.atomic_add(out_sumsq_ptr + pid, sumsq_val)


# Kernel: compute per-(b, s) mean and std from sum and sumsq
# std is population std = sqrt(sumsq/F - mean^2)
@triton.jit
def compute_stats_kernel(sum_ptr, sumsq_ptr, mean_out_ptr, std_out_ptr,
                         B, S, F):
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    sum_val = tl.load(sum_ptr + pid)
    sumsq_val = tl.load(sumsq_ptr + pid)
    mean = sum_val / F
    var = sumsq_val / F - mean * mean
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)
    tl.store(mean_out_ptr + pid, mean)
    tl.store(std_out_ptr + pid, std)


# Kernel: compute inverse-normal CDF (quantile) for a single scalar p
# Implements Abramowitz & Stegun 7.1.26 approximation (A&S formula 26.2.23).
@triton.jit
def ndtri_scalar_kernel(z_out_ptr,
                        p,  # scalar float, passed from host
                        a1, a2, a3, a4, a5, a6,
                        b1, b2, b3, b4, b5,
                        c1, c2, c3, c4, c5, c6,
                        d1, d2, d3, d4,
                        p_low):
    # Create a 1-element buffer for p in float32
    p_buf = tl.full((1,), p, tl.float32)
    # Piecewise computation without Python ifs
    # Lower region: p < p_low
    mask_low = p_buf < p_low
    q_low = tl.sqrt(-2.0 * tl.log(p_buf))  # for p < p_low, q = sqrt(-2*log(p))
    z_low = (((((c1 * q_low + c2) * q_low + c3) * q_low + c4) * q_low + c5) * q_low + c6) / \
            ((((d1 * q_low + d2) * q_low + d3) * q_low + d4) * q_low + 1.0)
    # Central region: p_low <= p <= (1 - p_low)
    mask_mid = (p_buf >= p_low) & (p_buf <= (1.0 - p_low))
    p_mid = p_buf
    q_mid = p_mid - 0.5
    r_mid = q_mid * q_mid
    poly_mid = (((((a1 * r_mid + a2) * r_mid + a3) * r_mid + a4) * r_mid + a5) * r_mid + a6)
    den_mid = (((((b1 * r_mid + b2) * r_mid + b3) * r_mid + b4) * r_mid + b5) * r_mid + 1.0)
    z_mid = poly_mid * q_mid / den_mid
    # Upper region: p > (1 - p_low)
    mask_up = p_buf > (1.0 - p_low)
    q_up = tl.sqrt(-2.0 * tl.log(1.0 - p_buf))
    z_up = -(((((c1 * q_up + c2) * q_up + c3) * q_up + c4) * q_up + c5) * q_up + c6) / \
           ((((d1 * q_up + d2) * q_up + d3) * q_up + d4) * q_up + 1.0)
    # Select via masks; since p is a single scalar, one mask will be true
    z = tl.where(mask_low, z_low, 0.0) + tl.where(mask_mid, z_mid, 0.0) + tl.where(mask_up, z_up, 0.0)
    # Store to z_out_ptr[0]
    tl.store(z_out_ptr, z)

# For each (b, s) row, load mean and std, compute threshold = mean + std * z,
# then compute y = max(input - threshold, 0). Store bfloat16 output.
@triton.jit
def apply_threshold_kernel(input_ptr, mean_ptr, std_ptr, z_ptr, output_ptr,
                           B, S, F, stride_b, stride_s, stride_f,
                           BLOCK_F: tl.constexpr):
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    base = b * stride_b + s * stride_s

    mean = tl.load(mean_ptr + pid)
    std = tl.load(std_ptr + pid)
    z = tl.load(z_ptr)  # single scalar

    threshold = mean + std * z

    # Process the row in chunks of BLOCK_F
    for off in range(0, F, BLOCK_F):
        idx = off + tl.arange(0, BLOCK_F)
        mask = idx < F
        in_ptrs = input_ptr + base + idx * stride_f
        in_vals = tl.load(in_ptrs, mask=mask, other=0.0).to(tl.float32)
        y = in_vals - threshold
        y = tl.maximum(y, 0.0)  # ReLU
        # store as bfloat16
        out_ptrs = output_ptr + base + idx * stride_f
        tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask)

- Removed all torch operations in forward. No torch.tensor() creation, no torch.clamp, no torch.sqrt, no elementwise operations on tensors. All math is done in Triton kernels.
- The ndtri_scalar_kernel implements the Abramowitz & Stegun 7.1.26 approximation exactly, with masks to select the appropriate region. No Python control flow based on runtime tensor values; this avoids Triton’s “if mask.any():” pitfalls.
- Reductions (sum and sumsq) use atomic adds per row, which is correct for parallel programs. Computing stats is a simple per-row operation.
- apply_threshold_kernel computes the per-row threshold and applies ReLU in one pass, writing the final bfloat16 output. This mirrors the original behavior and dtype.

Performance considerations:
- The reduction kernels read the input once and produce per-row sums and sumsq, which is efficient. The apply kernel is bandwidth-bound and processes the row in chunks. For the provided workloads, BLOCK_F=256 is a reasonable choice; you can tune it (e.g., 128 or 512) depending on your GPU and feature size.
- Atomic adds are on B*S scalars, which is negligible compared to the input size. If needed, you could implement a two-stage reduction (partial sums then a final reduction), but for these sizes, the simple atomic approach is fine and fast.
