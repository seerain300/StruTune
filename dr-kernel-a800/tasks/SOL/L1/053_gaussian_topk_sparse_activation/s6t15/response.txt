import torch
import triton
import triton.language as tl


@triton.jit
def sum_sumsq_rows_kernel(
    x_ptr,            # *const float (input, will be cast to float32 inside)
    out_sum_ptr,      # *float32, shape [B*S]
    out_sumsq_ptr,    # *float32, shape [B*S]
    B: tl.constexpr,  # int
    S: tl.constexpr,  # int
    F: tl.constexpr,  # int (feature size)
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,   # int
):
    pid = tl.program_id(0)  # program id: 0..(B*S-1)
    b = pid // S
    s = pid % S

    offs = tl.arange(0, BLOCK_F)
    total_sum = 0.0
    total_sumsq = 0.0

    # Loop across the feature dimension in chunks
    for start in range(0, F, BLOCK_F):
        idx = start + offs
        mask = idx < F
        # Compute linear index for x[b, s, idx]
        x_index = b * stride_b + s * stride_s + idx * stride_f
        x_vals = tl.load(x_ptr + x_index, mask=mask, other=0.0)
        # Accumulate in float32
        x_vals = x_vals.to(tl.float32)
        total_sum += tl.sum(x_vals, axis=0)
        total_sumsq += tl.sum(x_vals * x_vals, axis=0)

    # Write per-row results
    tl.atomic_add(out_sum_ptr + pid, total_sum)
    tl.atomic_add(out_sumsq_ptr + pid, total_sumsq)


@triton.jit
def compute_stats_kernel(
    out_sum_ptr,       # *const float32, shape [B*S]
    out_sumsq_ptr,     # *const float32, shape [B*S]
    out_mean_ptr,      # *float32, shape [B*S]
    out_std_ptr,       # *float32, shape [B*S]
    B: tl.constexpr,   # int
    S: tl.constexpr,   # int
    F: tl.constexpr,   # int
):
    pid = tl.program_id(0)  # 0..(B*S-1)
    # Load accumulated sum and sumsq
    sum_val = tl.load(out_sum_ptr + pid)
    sumsq_val = tl.load(out_sumsq_ptr + pid)
    # Compute mean and std (population std)
    mean = sum_val / F
    var = sumsq_val / F - mean * mean
    var = tl.maximum(var, 0.0)  # numerical safety
    std = tl.sqrt(var)
    # Store
    tl.store(out_mean_ptr + pid, mean)
    tl.store(out_std_ptr + pid, std)

def apply_threshold_kernel(

    mean_ptr,           # *const float32, shape [B*S]
    std_ptr,            # *const float32, shape [B*S]

    std = tl.load(std_ptr + pid)

    # Stream through the feature dimension and apply ReLU(x - threshold)

def compute_z_kernel(
    out_z_ptr,         # *float32, shape [1]
    p: tl.constexpr,   # float32 scalar p in (0,1)
    a1, a2, a3, a4, a5, a6,
    b1, b2, b3, b4, b5,
    c1, c2, c3, c4, c5, c6,
    d1, d2, d3, d4,
    p_low,
):
    # A&S 7.1.26 approximation for inverse-normal CDF
    # z = sign(p-0.5) * { (((((a1 t + a2) t + a3) t + a4) t + a5) t + a6) / (1 + (((b1 t + b2) t + b3) t + b4) t + b5) t ) }
    # where t = 1 / (1 + p*(1-p)^0.5)
    p_ = p  # scalar float32
    # lower region
    # t_low = 1 / (1 + p_low * (1 - p_low)^0.5)
    t_low = 1.0 / (1.0 + p_low * tl.sqrt(1.0 - p_low))
    z_low = (((((c1 * t_low + c2) * t_low + c3) * t_low + c4) * t_low + c5) * t_low + c6) / \
            ((((d1 * t_low + d2) * t_low + d3) * t_low + d4) * t_low + 1.0)
    # mid region
    # t_mid = 1 / (1 + (p - 0.5) * (1 - (p - 0.5))^0.5)
    p_minus_half = p_ - 0.5
    t_mid = 1.0 / (1.0 + p_minus_half * tl.sqrt(1.0 - p_minus_half))
    z_mid = (((((a1 * t_mid + a2) * t_mid + a3) * t_mid + a4) * t_mid + a5) * t_mid + a6) / \
            ((((b1 * t_mid + b2) * t_mid + b3) * t_mid + b4) * t_mid + b5)
    # upper region
    # t_up = 1 / (1 + (1 - p) * p^0.5)
    one_minus_p = 1.0 - p_
    t_up = 1.0 / (1.0 + one_minus_p * tl.sqrt(p_))
    z_up = -(((((c1 * t_up + c2) * t_up + c3) * t_up + c4) * t_up + c5) * t_up + c6) / \
           ((((d1 * t_up + d2) * t_up + d3) * t_up + d4) * t_up + 1.0)

    # Choose region and compute sign
    in_low = p_ < p_low
    in_mid = (p_ >= p_low) & (p_ <= (1.0 - p_low))
    in_up = p_ > (1.0 - p_low)

    # sign(p - 0.5)
    sign = 1.0 if p_ >= 0.5 else -1.0

    z = tl.zeros((), dtype=tl.float32)
    # Select appropriate z based on region
    z = tl.where(in_low, z_low, z)
    z = tl.where(in_mid, z_mid, z)
    z = tl.where(in_up, z_up, z)

    z = z * sign

    # Store scalar z into out_z_ptr
    tl.store(out_z_ptr, z)

    out_ptr,            # *float32, shape [B*S,F] (output buffer as float32)
    B: tl.constexpr,    # int
    S: tl.constexpr,    # int
    F: tl.constexpr,    # int
    z: tl.constexpr,    # float32, scalar inverse-normal CDF for target_sparsity
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    out_stride_b: tl.constexpr,  # int
    out_stride_s: tl.constexpr,  # int
    out_stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,       # int
):
    pid = tl.program_id(0)  # 0..(B*S-1)
    b = pid // S
    s = pid % S

    mean = tl.load(mean_ptr + pid)
    std = tl.load(std_ptr + pid)
    threshold = mean + std * z

    offs = tl.arange(0, BLOCK_F)
    for start in range(0, F, BLOCK_F):
        idx = start + offs
        mask = idx < F
        x_index = b * stride_b + s * stride_s + idx * stride_f
        x_vals = tl.load(x_ptr + x_index, mask=mask, other=0.0)
        x_vals = x_vals.to(tl.float32)
        y = tl.maximum(x_vals - threshold, 0.0)
        out_index = b * out_stride_b + s * out_stride_s + idx * out_stride_f
        tl.store(out_ptr + out_index, y, mask=mask)
