We convert inputs to float32 for computation and return bfloat16.

Current Triton approach
- Four Triton kernels:
  1) mean_lastdim_kernel: per-row mean across F.
  2) std_lastdim_kernel: per-row population std across F (unbiased=False).
  3) ndtri_approx_kernel: Triton-only inverse-normal CDF for scalar target_sparsity.
  4) apply_cutoff_relu_kernel: elementwise output = max(0, x - (mean + std * z)) in float32, then cast to bfloat16 on host.

Optimization strategy
- Keep feature-wise reduction kernels vectorized by using a reasonably large BLOCK_F. To minimize loop iterations and improve throughput:
  - If F <= 4096: BLOCK_F=4096, num_warps=4, num_stages=2.
  - If 4096 < F <= 8192: BLOCK_F=8192, num_warps=8, num_stages=2.
  - If 8192 < F <= 16384: BLOCK_F=16384, num_warps=8, num_stages=2.
  - If F > 16384: BLOCK_F=16384, num_warps=8, num_stages=2.

import torch
import triton
import triton.language as tl


@triton.jit
def mean_lastdim_kernel(x_ptr, means_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    total = tl.zeros((), dtype=tl.float32)
    # Loop over features in chunks
    for i in range(0, F, BLOCK_F):
        idx = i + tl.arange(0, BLOCK_F)
        mask = idx < F
        x = tl.load(x_ptr + row_offset + idx, mask=mask, other=0.0)
        total += tl.sum(x, axis=0)
    mean = total / F
    tl.store(means_ptr + pid, mean)


@triton.jit
def std_lastdim_kernel(x_ptr, stds_ptr, means_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    mean = tl.load(means_ptr + pid)
    sum_sq = tl.zeros((), dtype=tl.float32)
    for i in range(0, F, BLOCK_F):
        idx = i + tl.arange(0, BLOCK_F)
        mask = idx < F
        x = tl.load(x_ptr + row_offset + idx, mask=mask, other=0.0)
        diff = x - mean
        sum_sq += tl.sum(diff * diff, axis=0)
    # population std (unbiased=False): sqrt(sum_sq / F)
    std = tl.sqrt(sum_sq / F)
    tl.store(stds_ptr + pid, std)


@triton.jit
def ndtri_approx_kernel(p: tl.float32, out_ptr: tl.pointer_type(tl.float32)):
    # Abramowitz and Stegun 5.2.23 approximation
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p_high = 1.0 - p_low

    # lower region
    z_low = (((((c1 * tl.sqrt(-2.0 * tl.log(p))) + c2) * tl.sqrt(-2.0 * tl.log(p)) +
               c3) * tl.sqrt(-2.0 * tl.log(p)) + c4) * tl.sqrt(-2.0 * tl.log(p)) +
             c5) * tl.sqrt(-2.0 * tl.log(p)) + c6) / (
            ((((d1 * tl.sqrt(-2.0 * tl.log(p))) + d2) * tl.sqrt(-2.0 * tl.log(p)) +
              d3) * tl.sqrt(-2.0 * tl.log(p)) + d4) * tl.sqrt(-2.0 * tl.log(p)) + 1.0)

    # central region
    q = p - 0.5
    r = q * q
    z_mid = (((((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6) * q / (
            (((b1 * r + b2) * r + b3) * r + b4) * r + b5)

    # upper region
    qh = tl.sqrt(-2.0 * tl.log(1.0 - p))
    z_high = -(((((c1 * qh + c2) * qh + c3) * qh + c4) * qh + c5) * qh + c6) / (
            ((((d1 * qh + d2) * qh + d3) * qh + d4) * qh + 1.0)

    # select region based on p
    z = tl.where(p < p_low, z_low, 0.0)
    z = tl.where(p >= p_low, z_mid, z)
    z = tl.where(p > p_high, z_high, z)

    tl.store(out_ptr, z)


@triton.jit
def apply_cutoff_relu_kernel(x_ptr, out_ptr, means_ptr, stds_ptr, z_ptr, B, S, F, BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    mean = tl.load(means_ptr + pid)
    std = tl.load(stds_ptr + pid)
    z = tl.load(z_ptr)  # scalar inverse-N(0,1) CDF
    cutoff = mean + std * z
    # Elementwise: max(0, x - cutoff)
    for i in range(0, F, BLOCK_F):
        idx = i + tl.arange(0, BLOCK_F)
        mask = idx < F
        x = tl.load(x_ptr + row_offset + idx, mask=mask, other=0.0)
        y = x - cutoff
        y = tl.maximum(y, 0.0)
        tl.store(out_ptr + row_offset + idx, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        # Ensure CUDA and contiguous; compute in float32
        assert x.is_cuda, "Input must be on CUDA device for Triton kernels."
        x = x.contiguous().to(torch.float32)
        B, S, F = x.shape
        total_rows = B * S

        # Per-row statistics (float32) on device
        means = torch.empty(total_rows, device=x.device, dtype=torch.float32)
        stds = torch.empty(total_rows, device=x.device, dtype=torch.float32)

        # Heuristic tuning for BLOCK_F, num_warps, num_stages
        if F <= 4096:
            BLOCK_F = 4096
            num_warps = 4
            num_stages = 2
        elif F <= 8192:
            BLOCK_F = 8192
            num_warps = 8
            num_stages = 2
        elif F <= 16384:
            BLOCK_F = 16384
            num_warps = 8
            num_stages = 2
        else:
            BLOCK_F = 16384
            num_warps = 8
            num_stages = 2

        # Launch mean and std kernels
        grid = (total_rows,)
        mean_lastdim_kernel[grid](x, means, B, S, F, BLOCK_F=BLOCK_F, num_warps=num_warps, num_stages=num_stages)
        std_lastdim_kernel[grid](x, stds, means, B, S, F, BLOCK_F=BLOCK_F, num_warps=num_warps, num_stages=num_stages)

        # Scalar z via Triton ndtri kernel
        z_scalar = torch.empty(1, device=x.device, dtype=torch.float32)
        ndtri_approx_kernel[(1,)](float(target_sparsity), z_scalar)  # pass scalar directly

        # Allocate output (float32) for apply kernel
        out_fp32 = torch.empty_like(x, dtype=torch.float32)

        # Launch apply kernel
        apply_cutoff_relu_kernel[grid](x, out_fp32, means, stds, z_scalar, B, S, F, BLOCK_F=BLOCK_F, num_warps=num_warps, num_stages=num_stages)

        # Cast to bfloat16 to match original behavior
        return out_fp32.to(torch.bfloat16)
