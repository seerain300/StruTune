import torch
import triton
import triton.language as tl


@triton.jit
def compute_invphi_kernel(prob: tl.float32, out_ptr):
    """
    Compute inverse standard normal CDF for p = prob using bisection.
    Writes result to out_ptr[0] as float32.
    """
    low = -6.0
    high = 6.0
    # Bisection iterations: 30 provides good accuracy for p in (0, 1)
    for _ in range(30):
        mid = 0.5 * (low + high)
        # erf approximation (Abramowitz & Stegun 7.1.26)
        # erf(x) ≈ sign(x) * [1 - (((a5*t + a4)*t + a3)*t + a2)*t + a1)*t * exp(-x^2)]
        # with t = 1 / (1 + p_const * |x|)
        sqrt2 = 1.4142135623730951
        z = mid / sqrt2
        az = tl.abs(z)
        p_const = 0.3275911
        t = 1.0 / (1.0 + p_const * az)
        # Polynomial in t (ascending powers)
        # poly = t - a1*t^2 + a2*t^3 - a3*t^4 + a4*t^5 - a5*t^6
        poly = t
        poly = poly - 0.254829592 * t * t
        poly = poly + 0.284496736 * t * t * t
        poly = poly - 0.161512557 * t * t * t * t
        poly = poly + 0.027643881 * t * t * t * t * t
        poly = poly - 0.003840873 * t * t * t * t * t * t
        erf_approx = 1.0 - poly * tl.exp(-z * z)
        # Since erf(z/sqrt(2)) is odd, use sign(z) to place it in [-1, 1]
        sign = tl.where(z >= 0.0, 1.0, -1.0)
        erf_approx = sign * erf_approx

        # CDF: 0.5 * (1 + erf(z))
        cdf = 0.5 * (1.0 + erf_approx)
        # Update bounds using bisection
        direction = tl.where(cdf > prob, -1.0, 1.0)
        low = low + direction * (mid - low)  # mid moves to high when cdf > prob, else low moves to mid
        high = mid + (1.0 - direction) * (high - mid)  # mid moves to low when cdf > prob, else high moves to mid
    # Write result
    tl.store(out_ptr, mid)


@triton.jit
def row_sparsity_kernel(x_ptr, out_ptr, B: tl.int32, S: tl.int32, N: tl.int32, invphi_ptr, BLOCK_SIZE: tl.constexpr):
    """
    One program per (b, s) row. Computes:
      - mean and std across N in float32
      - threshold = mean + std * invphi
      - y = max(0, x - threshold) elementwise
    """
    # Decode program id into (b, s)
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S
    # Base linear offset into [B, S, N] contiguous layout: row index = b*S + s
    row_index = b * S + s
    base = row_index * N  # contiguous so [B, S, N] has linear indexing as above

    # Accumulators in float32
    sum_val = 0.0
    sum_sq = 0.0

    # First pass: reduce sum and sum of squares
    for start in range(0, N, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < N
        # Load as float32
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    n_f = tl.float32(N)
    mean = sum_val / n_f
    var = sum_sq / n_f - mean * mean
    # Guard against tiny negative due to roundoff
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)

    # Load inv-Phi scalar
    invphi = tl.load(invphi_ptr)  # float32 scalar
    threshold = mean + std * invphi

    # Second pass: elementwise ReLU gate
    for start in range(0, N, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < N
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        y = x - threshold
        y = tl.maximum(y, 0.0)  # ReLU
        tl.store(out_ptr + base + offs, y, mask=mask)

import torch
import torch.nn.functional as F
import math


def _ndtri(p: torch.Tensor) -> torch.Tensor:
    """Inverse of the standard normal CDF (quantile function).
    Uses Abramowitz and Stegun approximation (formula 26.2.23).
    This is a rational approximation that works well for p in (0, 1).
    """
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p_high = 1.0 - p_low

    result = torch.zeros_like(p)

    # Lower region
    mask_low = p < p_low
    if mask_low.any():
        q = torch.sqrt(-2.0 * torch.log(p[mask_low]))
        result[mask_low] = (((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6) / \
                           ((((d1*q + d2)*q + d3)*q + d4)*q + 1.0)

    # Central region
    mask_mid = (p >= p_low) & (p <= p_high)
    if mask_mid.any():
        q = p[mask_mid] - 0.5
        r = q * q
        result[mask_mid] = (((((a1*r + a2)*r + a3)*r + a4)*r + a5)*r + a6)*q / \
                           (((((b1*r + b2)*r + b3)*r + b4)*r + b5)*r + 1.0)

    # Upper region
    mask_high = p > p_high
    if mask_high.any():
        q = torch.sqrt(-2.0 * torch.log(1.0 - p[mask_high]))
        result[mask_high] = -(((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6) / \
                            ((((d1*q + d2)*q + d3)*q + d4)*q + 1.0)

    return result


@torch.no_grad()
def run(inputs: torch.Tensor, target_sparsity: float) -> torch.Tensor:
    """
    Gaussian-based top-k sparse activation.

    Computes adaptive sparsity threshold based on input statistics:
    1. Compute mean and std of input across feature dimension
    2. Calculate threshold = mean + std * norm.icdf(target_sparsity)
    3. Apply ReLU(input - threshold) to create sparse activations

    Args:
        inputs: Input tensor of shape [batch_size, seq_len, intermediate_size]
        target_sparsity: Float in [0, 1] indicating target sparsity level.
                        0.0 means no sparsity (all activations pass through).

    Returns:
        Sparsified tensor of same shape as input.
    """
    # Early return if no sparsity requested
    if target_sparsity == 0.0:
        return inputs

    # Convert to float32 for numerical stability in statistics computation
    inputs_f32 = inputs.to(torch.float32)

    # Compute statistics along the feature dimension (last dim)
    inputs_mean = torch.mean(inputs_f32, dim=-1, keepdim=True)
    inputs_std = torch.std(inputs_f32, dim=-1, keepdim=True, unbiased=False)

    # Compute the standard deviation multiplier using inverse CDF
    target_sparsity_tensor = torch.tensor(target_sparsity, dtype=torch.float32, device=inputs.device)
    std_multiplier = _ndtri(target_sparsity_tensor)

    # Compute the adaptive cutoff threshold
    cutoff_threshold = inputs_mean + inputs_std * std_multiplier

    # Apply ReLU with the adaptive threshold
    sparse_output = F.relu(inputs_f32 - cutoff_threshold)

    return sparse_output.to(torch.bfloat16)

class Model(torch.nn.Module):
    def forward(self, *args):
        return run(*args)

  - Takes a scalar prob (Python float) and writes the inverse standard normal CDF to a 1-element device tensor out_ptr[0].
  - Implement inverse CDF via bisection over z in [-6, 6]:
    - erf approximation (Abramowitz & Stegun 7.1.26):
      erf(x) ≈ sign(x) * [1 - P(t) * exp(-x^2)] where t = 1 / (1 + p_const * |x|), P(t) is a 5th-degree polynomial in t. Use only +, -, *, /, and tl.exp.
    - CDF Φ(z) = 0.5 * (1 + erf(z / sqrt(2))).
    - Update bounds using bisection: if Φ(z) > p then move high to z, else move low to z. With 30 iterations, accuracy is high.
  - This avoids any torch operations and produces a scalar on device.
- row_sparsity_kernel:
  - One Triton program per (b, s) row. Use linear indexing: row_base = (b*S + s)*N, since inputs are made contiguous by ModelNew.forward.
  - First loop over N in chunks of BLOCK_SIZE (e.g., 1024) to accumulate sum and sum of squares in float32. Compute mean and var, guard var >= 0, then std = sqrt(var).
  - Load the scalar invphi via tl.load(invphi_ptr). Compute threshold = mean + std * invphi.
  - Second loop: compute y = x - threshold, y = max(y, 0), store. Use masks offs < N.
- Performance:
  - The bisection loop is small (30 iterations), invphi is computed once per forward. The per-row work is two linear passes over N (sum, then elementwise). BLOCK_SIZE=1024, num_warps=8 is reasonable for N up to 16384.
  - All computation is in Triton; no host-side torch ops.

get_inputs
- The harness expects get_inputs to return a list of 1 element: a tensor. We will create a random tensor using pure Python’s random.randint to generate shape parameters, then build the tensor using nested loops (this avoids torch entirely).
- Example: create [B, S, N] with B=1, S=512, N=12288 (matching the first workload). Fill with random values in [-1, 1) using randint.

Edge cases and correctness
- If target_sparsity == 0.0, return inputs unchanged (no gating).
- Ensure inputs are contiguous before launching the kernel; use inputs.contiguous().
- Output is float32 in the kernel, then cast to bfloat16 on return (to match the original behavior). Triton kernels don’t need to know dtype, we cast in PyTorch after kernel execution.
