import torch
import triton
import triton.language as tl


@triton.jit
def compute_mean_std_fp32(in_ptr, mean_ptr, std_ptr, B, S, H, BLOCK: tl.constexpr):
    # One Triton program per (b, s) row: pid in [0, B*S)
    row = tl.program_id(0)
    b = row // S
    s = row % S

    # Base offset for this row in flattened [B, S, H] layout (row-major: B*S contiguous rows, each of length H)
    base = b * S * H + s * H

    # First pass: accumulate sum and sum of squares (fp32)
    sum_val = 0.0
    sum_sq = 0.0
    offsets = tl.arange(0, BLOCK)
    i = 0
    while i < H:
        idx = i + offsets
        mask = idx < H
        x = tl.load(in_ptr + base + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        i += BLOCK

    # Compute mean and std (population, unbiased=False)
    n = H
    mean = sum_val / n
    # var = E[x^2] - (E[x])^2
    var = sum_sq / n - mean * mean
    std = tl.sqrt(var)

    # Store per-row mean and std
    tl.store(mean_ptr + row, mean)
    tl.store(std_ptr + row, std)


@triton.jit
def compute_icdf_scalar(out_ptr, p_val: tl.float32, BLOCK: tl.constexpr):
    # Abramowitz & Stegun 5th-order rational approximation for standard normal inverse CDF
    # p_val is scalar float32
    # Constants
    p_low = 0.02425
    p_high = 1.0 - p_low

    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    # Compute inverse normal CDF for p_val (piecewise)
    if p_val <= p_low:
        # Lower tail: x = sqrt(-2*log(p))
        q = tl.sqrt(-2.0 * tl.log(p_val))
        poly = -(((((c1 * q + c2) * q + c3) * q + c4) * q + c5) * q + c6)
        den = (((((d1 * q + d2) * q + d3) * q + d4) * q + 1.0))
        nd = poly / den
    elif p_val >= p_high:
        # Upper tail: x = sqrt(-2*log(1-p))
        q = tl.sqrt(-2.0 * tl.log(1.0 - p_val))
        poly = -(((((c1 * q + c2) * q + c3) * q + c4) * q + c5) * q + c6)
        den = (((((d1 * q + d2) * q + d3) * q + d4) * q + 1.0))
        nd = poly / den
    else:
        # Center region: q = p - 0.5
        q = p_val - 0.5
        r = q * q
        poly = (((((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6)
        den = (((((b1 * r + b2) * r + b3) * r + b4) * r + b5))
        nd = poly * q / den

    tl.store(out_ptr, nd)


@triton.jit
def apply_threshold_relu_to_bf16(in_ptr, out_ptr, mean_ptr, std_ptr, B, S, H, z_ptr, BLOCK: tl.constexpr):
    # One Triton program per (b, s) row
    row = tl.program_id(0)
    b = row // S
    s = row % S

    # Load per-row mean and std (fp32)
    mean = tl.load(mean_ptr + row)
    std = tl.load(std_ptr + row)
    # Load scalar z (icdf) from z_ptr[0]
    z = tl.load(z_ptr)

    # Compute threshold
    thr = mean + std * z  # fp32

    # Base offset for this row
    base = b * S * H + s * H

    offsets = tl.arange(0, BLOCK)
    i = 0
    while i < H:
        idx = i + offsets
        mask = idx < H
        x = tl.load(in_ptr + base + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        y = x - thr
        y = tl.maximum(y, 0.0)
        tl.store(out_ptr + base + idx, y.to(tl.bfloat16), mask=mask)
        i += BLOCK


class ModelNew(torch.nn.Module):
    def forward(self, input: torch.Tensor, target_sparsity: float) -> torch.Tensor:
        # Ensure 3D input: [B, S, H]
        assert input.ndim == 3, "Input must be of shape [batch_size, seq_len, intermediate_size]"
        B, S, H = input.shape

        # If no sparsity, return input unchanged (preserve dtype and shape)
        if target_sparsity == 0.0:
            return input

        # Compute per-row mean and std in fp32 using Triton
        # Allocate mean/std buffers: shape [B*S]
        mean = torch.empty(B * S, dtype=torch.float32, device=input.device)
        std = torch.empty(B * S, dtype=torch.float32, device=input.device)

        # Launch compute_mean_std_fp32 kernel
        grid = (B * S,)
        compute_mean_std_fp32[grid](input, mean, std, B, S, H, BLOCK=1024)

        # Compute z = icdf(target_sparsity) via Triton kernel, output is 1-element tensor
        z = torch.empty(1, dtype=torch.float32, device=input.device)
        compute_icdf_scalar[(1,)](z, target_sparsity, BLOCK=1024)

        # Prepare output tensor in bfloat16
        output = torch.empty((B, S, H), dtype=torch.bfloat16, device=input.device)

        # Launch apply_threshold_relu_to_bf16 kernel
        grid_apply = (B * S,)
        apply_threshold_relu_to_bf16[grid_apply](input, output, mean, std, B, S, H, z, BLOCK=1024)

        return output

3) Triton-only requirement means forward() must not do any tensor math using PyTorch (no torch.mean, torch.std, F.relu, torch.tensor(...)). All computations must be inside @triton.jit kernels. Host code may only:
   - allocate tensors
   - set grid and launch Triton kernels
   - pass scalars to kernels (e.g., target_sparsity)
   - return the final tensor

既往失败原因推测
- 不完整的 Triton kernel主体（例如 compute_mean_std_fp32 应该是两遍 pass: first accumulate sum and sum of squares in fp32, second compute mean and std, then apply threshold in a per-row kernel）。截断导致统计不准确或者应用不完全，从而数值错误。
- apply kernel的截断导致某些行/特征可能未被处理，引起系统性错误。

行动计划
- 提供完整的 Triton kernels，并确保它们被实际调用：
  1) compute_mean_std_fp32: robust two-pass per (b, s) row using BLOCK and while loop to traverse H:
     - pass 1: sum and sum_sq in fp32
     - pass 2: mean = sum/H, var = sum_sq/H - mean^2, std = sqrt(var)
     - store mean and std to fp32 buffers of size [B*S]
  2) compute_icdf_scalar: A&S 5th-order rational approximation for inverse normal CDF (Abramowitz & Stegun). Inputs are scalars, outputs 1-element fp32 tensor z for device use.
  3) apply_threshold_relu_to_bf16: per-row kernel reads mean and std, loads z, computes thr, then iterates over H in chunks to compute y = max(x - thr, 0) and writes bfloat16 output.
- Host code:
  - Ensure input is 3D [B, S, H], convert to fp32 for computation (statistics and application), and allocate outputs.
  - Launch compute_mean_std_fp32 with grid (B*S,)
  - Launch compute_icdf_scalar with a 1-item grid, pass target_sparsity (float) and store z to device fp32 tensor
  - Launch apply_threshold_relu_to_bf16 with grid (B*S,) and write bfloat16 output
- Do not do any PyTorch mean/std/relu or tensor creation (e.g., torch.tensor(target_sparsity, ...)) on the host. Only scalar passing and tensor allocations are allowed.

进一步性能考虑
- H values are large (e.g., 12288). Using BLOCK=2048 reduces loop iterations compared to 1024 (6 vs 12), improving performance. We will set BLOCK=2048 and pass num_warps=4 or 8 to Triton launch for good throughput. This should maintain correctness (two-pass accumulation remains exact for fp32).
- apply kernel’s while loop uses BLOCK=2048 with mask, which is efficient and avoids unnecessary iterations.

import torch
import triton
import triton.language as tl


@triton.jit
def compute_mean_std_fp32(in_ptr, mean_ptr, std_ptr, B, S, H, BLOCK: tl.constexpr):
    # One Triton program per (b, s) row: pid in [0, B*S)
    row = tl.program_id(0)
    b = row // S
    s = row % S

    # Base offset for this row in flattened [B, S, H] layout
    base = b * S * H + s * H

    # Pass 1: accumulate sum and sum of squares (fp32)
    sum_val = 0.0
    sum_sq = 0.0
    offsets = tl.arange(0, BLOCK)
    i = 0
    while i < H:
        idx = i + offsets
        mask = idx < H
        x = tl.load(in_ptr + base + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        i += BLOCK

    # Pass 2: compute mean and std (population, unbiased=False)
    n = H
    mean = sum_val / n
    var = sum_sq / n - mean * mean
    std = tl.sqrt(var)

    # Store per-row mean and std
    tl.store(mean_ptr + row, mean)
    tl.store(std_ptr + row, std)
