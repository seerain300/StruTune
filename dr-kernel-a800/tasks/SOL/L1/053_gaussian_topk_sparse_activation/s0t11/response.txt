Detailed analysis
- Original PyTorch behavior:
  - Inputs: x of shape [B, S, F].
  - Compute mean and population std across last dim (features): x.to(torch.float32), then:
    - mean_x = mean(x, dim=-1, keepdim=True) -> shape [B, S, 1]
    - std_x = std(x, dim=-1, keepdim=True, unbiased=False) -> shape [B, S, 1]
  - z = ndtri(target_sparsity), where ndtri is the inverse of the standard normal CDF. For p in (0,1), the reference uses the Abramowitz & Stegun 5.2.23 approximation.
  - cutoff = mean_x + std_x * z -> shape [B, S, 1]
  - sparse_output = relu(x - cutoff) elementwise over F, returning in x.dtype (in tests typically bfloat16).

- Potential sources of mismatch:
  - Dtype handling: Must compute in float32, but return in original dtype (bfloat16 in the provided tests). The PyTorch code casts the final result to bfloat16; we will do that as well.
  - Indexing and layout: The kernels must operate on a contiguous layout with correct row/feature mapping. Using a 2D view [NROWS, F] where NROWS=B*S is correct if we maintain contiguous inputs and reshape outputs accordingly.
  - Scalar z computation: We must compute ndtri(target_sparsity) inside Triton (no torch math on device). Reading the resulting scalar once on host is acceptable.

Plan
1) Ensure inputs are contiguous and on CUDA; convert to float32 for computations.
2) Compute per-row mean and std along the last dimension:
   - mean_lastdim_kernel: one program per row (pid in [0, B*S)). Loop over F in chunks (BLOCK_F=2048), accumulate sum, divide by F, store mean.
   - std_lastdim_kernel: one program per row. Loop over F again, accumulate sum of squared diffs from mean, divide by F, sqrt, store std.
3) Compute z = ndtri(target_sparsity) in a Triton kernel using the Abramowitz & Stegun 5.2.23 approximation, writing to a 1-element device tensor.
4) Apply elementwise: out = max(0, x - (mean + std*z)), compute in float32, cast to original dtype before returning.
5) Keep host code minimal: allocate, launch kernels, reshape and cast, return.

Key points to prevent recurrence
- Use explicit contiguous [B, S, F] tensor for inputs and create a 2D view [NROWS, F] for kernels. This avoids any ambiguous strides.
- Use a 1D grid over NROWS and iterate over F with masked loads/stores to handle any F.
- Avoid any torch math in host code except for dtype casting of the final output.

import torch
import triton
import triton.language as tl


# Kernel: compute per-row mean across the last dimension (features F)
@triton.jit
def mean_lastdim_kernel(
    inputs_ptr,        # *fp32, shape [NROWS, F]
    mean_out_ptr,      # *fp32, shape [NROWS]
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(0)
    # Accumulate sum across F in chunks
    total = 0.0
    for start in range(0, F, BLOCK_F):
        offs = start + tl.arange(0, BLOCK_F)
        mask = offs < F
        row_ptr = inputs_ptr + pid * F + offs
        vals = tl.load(row_ptr, mask=mask, other=0.0)
        total += tl.sum(vals, axis=0)
    mean = total / F
    tl.store(mean_out_ptr + pid, mean)


# Kernel: compute per-row std (population, unbiased=False) across the last dimension
@triton.jit
def std_lastdim_kernel(
    inputs_ptr,        # *fp32, shape [NROWS, F]
    mean_ptr,          # *fp32, shape [NROWS]
    std_out_ptr,       # *fp32, shape [NROWS]
    F,                 # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(0)
    mean = tl.load(mean_ptr + pid)
    total = 0.0
    for start in range(0, F, BLOCK_F):
        offs = start + tl.arange(0, BLOCK_F)
        mask = offs < F
        row_ptr = inputs_ptr + pid * F + offs
        vals = tl.load(row_ptr, mask=mask, other=0.0)
        diff = vals - mean
        total += tl.sum(diff * diff, axis=0)
    var = total / F
    std = tl.sqrt(var)
    tl.store(std_out_ptr + pid, std)


# Kernel: compute inverse-normal CDF (ndtri) for a scalar p in (0,1) using A&S 5.2.23
@triton.jit
def ndtri_kernel(
    out_ptr,           # *fp32, shape [1]
    p,                 # fp32 scalar (0,1)
    p_low,             # fp32
    p_high,            # fp32
    a1, a2, a3, a4, a5, a6,    # fp32
    b1, b2, b3, b4, b5,        # fp32
    c1, c2, c3, c4, c5, c6,    # fp32
    d1, d2, d3, d4,            # fp32
):
    # Compute z based on p
    # Masks for regions
    # Lower region
    if p < p_low:
        q = tl.sqrt(-2.0 * tl.log(p))
        poly = (((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6)
        denom = ((((d1*q + d2)*q + d3)*q + d4)*q + 1.0)
        z = poly / denom
    # Central region
    elif p <= p_high:
        q = p - 0.5
        r = q * q
        poly = (((((a1*r + a2)*r + a3)*r + a4)*r + a5)*r + a6)
        denom = (((((b1*r + b2)*r + b3)*r + b4)*r + b5)*r + 1.0)
        z = poly * q / denom
    else:
        q = tl.sqrt(-2.0 * tl.log(1.0 - p))
        poly = (((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6)
        denom = ((((d1*q + d2)*q + d3)*q + d4)*q + 1.0)
        z = -poly / denom
    # Store scalar
    tl.store(out_ptr, z)


# Kernel: apply cutoff and ReLU elementwise over rows: out = max(0, x - (mean + std*z))
@triton.jit
def apply_cutoff_relu_kernel(
    inputs_ptr,         # *fp32, shape [NROWS, F]
    mean_ptr,           # *fp32, shape [NROWS]
    std_ptr,            # *fp32, shape [NROWS]
    z_ptr,              # *fp32, shape [1]
    out_ptr,            # *fp32, shape [NROWS, F]
    F,                  # int32
    BLOCK_F: tl.constexpr,
):
    pid = tl.program_id(0)
    mean = tl.load(mean_ptr + pid)
    std = tl.load(std_ptr + pid)
    z = tl.load(z_ptr)  # scalar z
    cutoff = mean + std * z
    for start in range(0, F, BLOCK_F):
        offs = start + tl.arange(0, BLOCK_F)
        mask = offs < F
        row_in_ptr = inputs_ptr + pid * F + offs
        row_out_ptr = out_ptr + pid * F + offs
        x = tl.load(row_in_ptr, mask=mask, other=0.0)
        y = x - cutoff
        # ReLU
        y = tl.maximum(y, 0.0)
        tl.store(row_out_ptr, y, mask=mask)
