# Kernel: for each (b, s) row, accumulate sum of inputs across F in chunks of BLOCK_F.
# One program handles a chunk: it computes local sum and atomic_add to out_sum[pid_row].
@triton.jit
def sum_rows_kernel(inp_ptr, out_sum_ptr, B, S, F,
                     stride_b, stride_s, stride_f,
                     BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row = pid
    # guard: pid in [0, B*S)
    # compute base offset for this row
    # Note: Triton expects linear indexing, so we map pid -> (b, s) implicitly via row index
    # Here, we assume grid=(B*S,)
    # Each program handles a single row, and we loop over F in chunks of BLOCK_F
    start = 0
    local_sum = 0.0
    while start < F:
        offs = start + tl.arange(0, BLOCK_F)
        mask = offs < F
        # linear address for a row is: b*stride_b + s*stride_s + f*stride_f
        # For pid in [0, B*S), we can recover b and s as: b = pid // S, s = pid % S
        # However, since we use grid=(B*S,), we can directly index using row and strides
        # The pointer arithmetic below assumes row corresponds to (b, s) pair via row = b*S + s
        b = row // S
        s = row % S
        base = b * stride_b + s * stride_s
        x = tl.load(inp_ptr + base + offs * stride_f, mask=mask, other=0.0)
        x = x.to(tl.float32)
        local_sum += tl.sum(x, axis=0)
        start += BLOCK_F
    tl.atomic_add(out_sum_ptr + row, local_sum)


# Kernel: same as sum_rows_kernel but accumulates sum of squares.

def sumsq_rows_kernel(inp_ptr, out_sumsq_ptr, B, S, F,

    sumsq_val = tl.load(out_sumsq_ptr + row)

    var = sumsq_val / F - mean * mean

    std = tl.sqrt(var)

    tl.store(out_std_ptr + row, std)

    q_low = tl.sqrt(-2.0 * tl.log(p))

@triton.jit
def add_kernel(
    x_ptr,  # Pointer to first input
    y_ptr,  # Pointer to second input
    out_ptr,  # Pointer to output
    n_elements,  # Total number of elements in input/output
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles a contiguous block of data of size BLOCK_SIZE
    block_start = tl.program_id(0) * BLOCK_SIZE
    # Create a range of offsets [0..BLOCK_SIZE-1]
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    # Mask to ensure we don't go out of bounds
    mask = offsets < n_elements
    # Load input values
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    # Perform the elementwise addition
    out = x + y
    # Store the result
    tl.store(out_ptr + offsets, out, mask=mask)

def triton_add(x: torch.Tensor, y: torch.Tensor):
    """
    This function wraps the Triton kernel call. It:
      1. Ensures the inputs are contiguous on GPU.
      2. Calculates the grid (blocks) needed.
      3. Launches the Triton kernel.
    """
    assert x.is_cuda and y.is_cuda, "Tensors must be on CUDA."
    x = x.contiguous()
    y = y.contiguous()

    # Prepare output tensor
    out = torch.empty_like(x)

    # Number of elements in the tensor
    n_elements = x.numel()
    BLOCK_SIZE = 128  # Tunable parameter for block size

    # Determine the number of blocks needed
    grid = lambda meta: ((n_elements + meta["BLOCK_SIZE"] - 1) // meta["BLOCK_SIZE"],)

    # Launch the Triton kernel
    add_kernel[grid](x, y, out, n_elements, BLOCK_SIZE=BLOCK_SIZE)

  - One-pass reductions reduce memory traffic and kernel launches. For large F (e.g., 12288), this can offer noticeable speedups without changing numerical behavior.
  - Choose BLOCK_F heuristically (1024, 512, 256) based on F to balance occupancy and register usage. We’ll use 1024 when F ≥ 1024, otherwise 512 or 256.

import torch
import triton
import triton.language as tl


# Kernel: for each (b, s) row, accumulate sum of inputs across F in chunks of BLOCK_F.
# Each program handles a single row and loops over F; local sum is atomic added to out_sum[row].
@triton.jit
def sum_rows_kernel(inp_ptr, out_sum_ptr, B, S, F,
                     stride_b, stride_s, stride_f,
                     BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row = pid
    start = 0
    local_sum = 0.0
    while start < F:
        offs = start + tl.arange(0, BLOCK_F)
        mask = offs < F
        b = row // S
        s = row % S
        base = b * stride_b + s * stride_s
        x = tl.load(inp_ptr + base + offs * stride_f, mask=mask, other=0.0).to(tl.float32)
        local_sum += tl.sum(x, axis=0)
        start += BLOCK_F
    tl.atomic_add(out_sum_ptr + row, local_sum)


# Kernel: same as sum_rows_kernel but accumulates sum of squares.
@triton.jit
def sumsq_rows_kernel(inp_ptr, out_sumsq_ptr, B, S, F,
                       stride_b, stride_s, stride_f,
                       BLOCK_F: tl.constexpr):
    pid = tl.program_id(axis=0)
    row = pid
    start = 0
    local_sumsq = 0.0
    while start < F:
        offs = start + tl.arange(0, BLOCK_F)
        mask = offs < F
        b = row // S
        s = row % S
        base = b * stride_b + s * stride_s
        x = tl.load(inp_ptr + base + offs * stride_f, mask=mask, other=0.0).to(tl.float32)
        local_sumsq += tl.sum(x * x, axis=0)
        start += BLOCK_F
    tl.atomic_add(out_sumsq_ptr + row, local_sumsq)


# Kernel: compute mean and std for each row using out_sum and out_sumsq.
# population std: std = sqrt(E[x^2] - mean^2), var = max(E[x^2] - mean^2, 0)
@triton.jit
def compute_stats_kernel(out_sum_ptr, out_sumsq_ptr, out_mean_ptr, out_std_ptr, B, S, F):
    row = tl.program_id(axis=0)
    sum_val = tl.load(out_sum_ptr + row)
    sumsq_val = tl.load(out_sumsq_ptr + row)
    mean = sum_val / F
    var = sumsq_val / F - mean * mean
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)
    tl.store(out_mean_ptr + row, mean)
    tl.store(out_std_ptr + row, std)
