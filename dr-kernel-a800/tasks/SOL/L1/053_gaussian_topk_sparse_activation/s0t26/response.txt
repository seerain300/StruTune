@triton.jit
def mean_lastdim_kernel(x_ptr, means_ptr, B, S, F, NUM_ITERS: tl.constexpr, BLOCK_F: tl.constexpr):
    # One program per row; row index = program_id(0)
    pid = tl.program_id(axis=0)
    row_offset = pid * F
    total = tl.zeros((), dtype=tl.float32)

    # Loop over features in chunks of BLOCK_F
    i = 0
    while i < NUM_ITERS:
        offsets = i * BLOCK_F + tl.arange(0, BLOCK_F)
        mask = offsets < F
        x_ptrs = x_ptr + row_offset + offsets
        vals = tl.load(x_ptrs, mask=mask, other=0.0)
        # accumulate in float32
        total += tl.sum(vals, axis=0)
        i += 1

    mean = total / F
    tl.store(means_ptr + pid, mean)

        mask = offsets < F
        vals = tl.load(x_ptr + row_offset + offsets, mask=mask, other=0.0)
        diff = vals - mean
        sumsq += tl.sum(diff * diff, axis=0)

    # Population std: unbiased=False, divide by F
    std = tl.sqrt(sumsq / F)
    tl.store(stds_ptr + pid, std)
