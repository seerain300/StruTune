import triton
import triton.language as tl


@triton.jit
def reduce_sum_sumsq_kernel(
    X_ptr,           # *input tensor [B, S, D], contiguous
    SUM_ptr,         # *float32, length B*S
    SUMSQ_ptr,       # *float32, length B*S
    B: tl.constexpr, # int32 dimensions (will pass runtime int, but constexpr helps compilation)
    S: tl.constexpr,
    D: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    # One program per (b, s) row
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S
    # Base offset for this row (row-major [B, S, D] with contiguous => stride along D is 1)
    base = (b * S + s) * D

    acc_sum = tl.zeros((), dtype=tl.float32)
    acc_sumsq = tl.zeros((), dtype=tl.float32)

    # Iterate over D in tiles
    for off in range(0, D, BLOCK_SIZE):
        idx = off + tl.arange(0, BLOCK_SIZE)
        mask = idx < D
        x = tl.load(X_ptr + base + idx, mask=mask, other=0.0).to(tl.float32)
        acc_sum += tl.sum(x, axis=0)
        acc_sumsq += tl.sum(x * x, axis=0)

    mean = acc_sum / D
    var_num = acc_sumsq / D - mean * mean
    var_num = tl.maximum(var_num, 0.0)  # guard against tiny negative due to rounding
    std = tl.sqrt(var_num)

    tl.store(SUM_ptr + pid, acc_sum)     # store sum for optional debug, not used in next kernel
    tl.store(SUMSQ_ptr + pid, acc_sumsq)  # store sumsq for optional debug
    tl.store(SUM_ptr + pid + B * S, mean)  # store mean at position pid (pid already covers B*S)
    tl.store(SUMSQ_ptr + pid + B * S, std)  # store std at position pid (pid already covers B*S)

    var_num = acc_sumsq / D - mean * mean
    var_num = tl.maximum(var_num, 0.0)  # guard against tiny negative due to rounding
    std = tl.sqrt(var_num)

    tl.store(MEAN_ptr + pid, mean)
    tl.store(STD_ptr + pid, std)
