Design choices for correctness and robustness

  - grid = (L, ceil_div(B*S, BLOCK_R))
  - Each program computes a partial sum and sumsq for its feature f across its row-chunk and writes to out_sum[pid_f] and out_sumsq[pid_f]. So we’ll instead do a 2D grid where each program writes its partials, then host-side torch.sum (not in forward). This would violate Triton-only constraint. Therefore, we’ll instead use a single Triton program per feature and loop over all row chunks, which guarantees correctness but needs dynamic loops. Triton supports while loops with runtime conditions.

To satisfy Triton-only constraint fully and avoid torch reductions, we will:
- Implement a Triton kernel sum_sumsq_per_feature that loops over all row chunks for a given feature f. It iterates groups with a while loop until it covers all rows. This keeps the whole reduction in Triton and avoids decoy kernels. It uses runtime rows and BLOCK_R, masking out-of-range indices. This is the safest way to handle dynamic sizes. The cost is the kernel doing more iterations, but for typical sizes, it’s fine, and correctness is paramount.
- Compute mean and std per feature in a Triton kernel that reads sum and sumsq and writes mean_f and std_f to 1D device arrays. Pass rows and L as runtime scalars.
- Compute threshold per feature in a Triton kernel using thr = mean + std * std_multiplier (std_multiplier is provided once, not computed inside forward). Launch compute_thr_per_feature_kernel.
- Apply sparse ReLU per feature using a Triton kernel sparse_relu_per_feature with a 2D grid (rows, features) to broadcast thr per feature. Launch this kernel.
- Cast output to bfloat16 via cast_bf16_kernel. Launch this kernel.

Performance considerations
- The per-feature reduction kernel loops over row chunks; it is correct and simple. - We use reasonably large BLOCK_R (e.g., 1024) to limit loop iterations. Even if rows = 4096, the loop runs only 4 iterations per feature, which is acceptable.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def sum_sumsq_per_feature(x_ptr,  # *fp32, input flattened to [rows, L]
                           out_sum_ptr,  # *fp32, length L
                           out_sumsq_ptr,  # *fp32, length L
                           rows,  # int32, number of rows = B*S
                           L,  # int32, number of features
                           BLOCK_R: tl.constexpr):
    """
    For a given feature f, accumulate sum and sumsq over all rows.
    We use one Triton program per feature f, and loop over row chunks.
    """
    f = tl.program_id(0)
    # Guard: if f >= L, return (normally grid=(L,) ensures f < L)
    if f >= L:
        return

    # Initialize accumulators
    sum_acc = 0.0
    sumsq_acc = 0.0

    # Iterate over row chunks
    r = 0
    while r < rows:
        # For each row in the chunk, accumulate
        # Note: rows is runtime; Triton supports while loops with runtime conditions.
        for i in range(BLOCK_R):
            row_idx = r + i
            row_mask = row_idx < rows
            # Load x[row_idx, f] as fp32. Since x is fp32, this is straightforward.
            # Compute pointer offset: x_ptr is laid out as [rows, L] contiguous,
            # so offset = row_idx * L + f
            val = tl.load(x_ptr + row_idx * L + f, mask=row_mask, other=0.0)
            sum_acc += val
            sumsq_acc += val * val
        r += BLOCK_R

    # Write results
    tl.store(out_sum_ptr + f, sum_acc)
    tl.store(out_sumsq_ptr + f, sumsq_acc)


@triton.jit
def compute_mean_std_per_feature(sum_ptr,  # *fp32, length L
                                 sumsq_ptr,  # *fp32, length L
                                 mean_ptr,   # *fp32, length L (output)
                                 std_ptr,    # *fp32, length L (output)
                                 rows,       # int32
                                 L):         # int32
    """
    Compute mean and std per feature:
      mean = sum / rows
      var = sumsq / rows - mean^2
      std = sqrt(max(var, 0))
    """
    f = tl.program_id(0)
    if f >= L:
        return
    sum_f = tl.load(sum_ptr + f)
    sumsq_f = tl.load(sumsq_ptr + f)
    mean_f = sum_f / rows
    var_f = sumsq_f / rows - mean_f * mean_f
    # Ensure non-negative variance for numerical stability
    var_f = tl.maximum(var_f, 0.0)
    std_f = tl.sqrt(var_f)
    tl.store(mean_ptr + f, mean_f)
    tl.store(std_ptr + f, std_f)


@triton.jit
def compute_thr_per_feature(mean_ptr,  # *fp32, length L
                            std_ptr,   # *fp32, length L
                            thr_ptr,   # *fp32, length L (output)
                            multiplier,  # fp32 scalar
                            L):         # int32
    """
    Compute per-feature threshold: thr = mean + std * multiplier
    """
    f = tl.program_id(0)
    if f >= L:
        return
    mean_f = tl.load(mean_ptr + f)
    std_f = tl.load(std_ptr + f)
    thr_f = mean_f + std_f * multiplier
    tl.store(thr_ptr + f, thr_f)


@triton.jit
def sparse_relu_per_feature(x_ptr,      # *fp32, input flattened [rows, L]
                            thr_ptr,    # *fp32, length L
                            out_ptr,    # *fp32, output flattened [rows, L]
                            rows,       # int32
                            L):         # int32
    """
    Apply elementwise ReLU with per-feature threshold: out[b,s,f] = max(x[b,s,f] - thr[f], 0)
    We launch a 2D grid: pid_r across rows, pid_f across features.
    Each program handles one row index and one feature.
    """
    pid_r = tl.program_id(0)  # row index
    pid_f = tl.program_id(1)  # feature index
    if pid_r >= rows or pid_f >= L:
        return
    # Load x[pid_r, pid_f]
    val = tl.load(x_ptr + pid_r * L + pid_f)
    thr = tl.load(thr_ptr + pid_f)
    out_val = tl.maximum(val - thr, 0.0)
    tl.store(out_ptr + pid_r * L + pid_f, out_val)


@triton.jit
def cast_bf16_kernel(in_ptr,  # *fp32
                     out_ptr,  # *bf16
                     numel,    # int32
                     BLOCK_SIZE: tl.constexpr):
    """
    Cast fp32 to bf16 elementwise.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    vals = tl.load(in_ptr + offs, mask=mask, other=0.0)
    vals_bf16 = vals.to(tl.bfloat16)
    tl.store(out_ptr + offs, vals_bf16, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self, target_sparsity: float, multiplier: float):
        super().__init__()
        # Store multiplier = ndtri(target_sparsity), expected to be precomputed.
        # Keep as Python float; pass to Triton kernel as fp32.
        self.target_sparsity = float(target_sparsity)
        self.multiplier = float(multiplier)  # ndtri(target_sparsity)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        x: input tensor of shape [B, S, L], dtype float32 (for numerical stability).
        Returns: sparse output tensor of same shape as x, dtype bfloat16.
        """
        assert x.is_cuda, "ModelNew.forward requires CUDA tensors"
        assert x.dtype == torch.float32, "Expected input dtype to be float32"

        B, S, L = x.shape
        rows = B * S

        # Prepare flattened input [rows, L] contiguous
        x_flat = x.contiguous().view(rows, L)

        # 1) Compute per-feature sum and sumsq using Triton reduction over all rows.
        #    One program per feature f; loop over row chunks.
        sum_per_feature = torch.empty(L, dtype=torch.float32, device=x.device)
        sumsq_per_feature = torch.empty(L, dtype=torch.float32, device=x.device)

        # Launch reduction kernel: grid=(L,)
        sum_sumsq_per_feature[(L,)](
            x_flat, sum_per_feature, sumsq_per_feature,
            rows, L,
            BLOCK_R=1024,  # chunk size over rows; mask handles rows < BLOCK_R * groups
        )

        # 2) Compute mean and std per feature in Triton
        mean_per_feature = torch.empty(L, dtype=torch.float32, device=x.device)
        std_per_feature = torch.empty(L, dtype=torch.float32, device=x.device)

        compute_mean_std_per_feature[(L,)](
            sum_per_feature, sumsq_per_feature,
            mean_per_feature, std_per_feature,
            rows, L
        )

        # 3) Compute per-feature threshold thr = mean + std * multiplier (std_multiplier is self.multiplier)
        thr_per_feature = torch.empty(L, dtype=torch.float32, device=x.device)

        compute_thr_per_feature[(L,)](
            mean_per_feature, std_per_feature,
            thr_per_feature,
            self.multiplier, L
        )

        # 4) Apply ReLU with per-feature thresholds using Triton, output in fp32
        out_fp32 = torch.empty_like(x_flat, dtype=torch.float32, device=x.device)

        sparse_relu_per_feature[(rows, L)](
            x_flat, thr_per_feature, out_fp32,
            rows, L
        )

        # 5) Cast to bfloat16 via Triton kernel and reshape to [B, S, L]
        out_bf16 = torch.empty(B * S * L, dtype=torch.bfloat16, device=x.device)

        # Flatten out_fp32 for casting
        numel = B * S * L
        cast_bf16_kernel[(triton.cdiv(numel, 1024),)](
            out_fp32.view(-1), out_bf16,
            numel, BLOCK_SIZE=1024
        )

        out = out_bf16.view(B, S, L)
        return out
```

- Cast: Triton elementwise cast from fp32 to bf16 over the flattened output. Launch 1D grid.

Performance improvement ideas (not all applied here due to evaluator constraints)
- Two-stage reduction to reduce loop iterations: compute partial sums/sumsq with a grid over (features, row-chunks) and then reduce those partials. Triton doesn’t provide fp32 atomics; doing torch.sum on partials would trigger torch ops in forward (forbidden). Without torch, we’re limited; the current approach is correct and simple.
- Precompute multiplier = ndtri(target_sparsity) once, which we do in the module init (outside forward), avoiding any torch computation in forward.

Final code with all kernels invoked in forward

import torch
import triton
import triton.language as tl


@triton.jit
def sum_sumsq_per_feature(x_ptr,      # *fp32, input flattened [rows, L]
                           out_sum_ptr,  # *fp32, length L
                           out_sumsq_ptr,  # *fp32, length L
                           rows,        # int32, number of rows (B*S)
                           L,           # int32, number of features
                           BLOCK_R: tl.constexpr):
    """
    One Triton program per feature f. Loop over all rows in chunks of BLOCK_R,
    accumulate sum and sumsq for that feature, and write results.
    """
    f = tl.program_id(0)
    if f >= L:
        return

    sum_acc = 0.0
    sumsq_acc = 0.0

    r = 0
    while r < rows:
        for i in range(BLOCK_R):
            row_idx = r + i
            row_mask = row_idx < rows
            # Load x[row_idx, f] as fp32
            val = tl.load(x_ptr + row_idx * L + f, mask=row_mask, other=0.0)
            sum_acc += val
            sumsq_acc += val * val
        r += BLOCK_R

    tl.store(out_sum_ptr + f, sum_acc)
    tl.store(out_sumsq_ptr + f, sumsq_acc)


@triton.jit
def compute_mean_std_per_feature(sum_ptr,     # *fp32, length L
                                 sumsq_ptr,   # *fp32, length L
                                 mean_ptr,    # *fp32, length L
                                 std_ptr,     # *fp32, length L
                                 rows,        # int32
                                 L):          # int32
    """
    Compute mean and std per feature:
      mean = sum / rows
      var = sumsq / rows - mean^2
      std = sqrt(max(var, 0))
    """
    f = tl.program_id(0)
    if f >= L:
        return
    sum_f = tl.load(sum_ptr + f)
    sumsq_f = tl.load(sumsq_ptr + f)
    mean_f = sum_f / rows
    var_f = sumsq_f / rows - mean_f * mean_f
    var_f = tl.maximum(var_f, 0.0)
    std_f = tl.sqrt(var_f)
    tl.store(mean_ptr + f, mean_f)
    tl.store(std_ptr + f, std_f)


@triton.jit
def compute_thr_per_feature(mean_ptr,     # *fp32, length L
                            std_ptr,      # *fp32, length L
                            thr_ptr,      # *fp32, length L
                            multiplier,   # fp32 scalar
                            L):           # int32
    """
    Compute per-feature threshold: thr = mean + std * multiplier
    """
    f = tl.program_id(0)
    if f >= L:
        return
    mean_f = tl.load(mean_ptr + f)
    std_f = tl.load(std_ptr + f)
    thr_f = mean_f + std_f * multiplier
    tl.store(thr_ptr + f, thr_f)


@triton.jit
def sparse_relu_per_feature(x_ptr,       # *fp32, input flattened [rows, L]
                            thr_ptr,     # *fp32, length L
                            out_ptr,     # *fp32, output flattened [rows, L]
                            rows,        # int32
                            L):          # int32
    """
    Apply elementwise ReLU with per-feature threshold:
      out[row, f] = max(x[row, f] - thr[f], 0)
    2D grid: program_id(0) over rows, program_id(1) over features.
    """
    pid_r = tl.program_id(0)
    pid_f = tl.program_id(1)
    if pid_r >= rows or pid_f >= L:
        return
    val = tl.load(x_ptr + pid_r * L + pid_f)
    thr = tl.load(thr_ptr + pid_f)
    out_val = tl.maximum(val - thr, 0.0)
    tl.store(out_ptr + pid_r * L + pid_f, out_val)


@triton.jit
def cast_bf16_kernel(in_ptr,  # *fp32
                     out_ptr,  # *bf16
                     numel,    # int32
                     BLOCK_SIZE: tl.constexpr):
    """
    Cast fp32 to bf16 elementwise.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    vals = tl.load(in_ptr + offs, mask=mask, other=0.0)
    vals_bf16 = vals.to(tl.bfloat16)
    tl.store(out_ptr + offs, vals_bf16, mask=mask)
