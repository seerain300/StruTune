import triton
import triton.language as tl


@triton.jit
def scatter_add_full_row_kernel(
    out_ptr,            # *float32 [B, H], out_stride0 = H
    indices_ptr,        # *int32 [N]
    vals_ptr,           # *float32 [N, H], vals_stride0 = H, vals_stride1 = 1

    H,                  # int32
    out_stride0,        # int32
    vals_stride0,       # int32
    vals_stride1,       # int32
    T: tl.constexpr,    # number of updates per program (compile-time constant)
    BLOCK_H: tl.constexpr,  # full row width, we assume H <= BLOCK_H
):
    # Each program handles up to T updates.
    pid = tl.program_id(axis=0)
    # Use vectorized k for multiple updates per program
    k = tl.arange(0, T)  # shape [T]
    # Compute positions in [0, N)
    pos = pid * T + k  # shape [T]
    # Masks for valid positions (tail handling)
    mask_pos = pos < N

    # Load indices for these positions
    idx = tl.load(indices_ptr + pos, mask=mask_pos, other=0)  # int32
    # Compute base pointers for rows in output and vals
    out_row_ptr = out_ptr + idx * out_stride0  # shape [T]
    vals_row_ptr = vals_ptr + pos * vals_stride0  # shape [T]

    # Since H <= BLOCK_H, we can atomic_add the entire row in one vector operation.
    # We loop over columns in chunks of BLOCK_H (here it's a single chunk when H <= BLOCK_H).
    # For robustness, we keep a loop that Triton can unroll at compile time.
    for h in range(0, tl.max_contiguous(H, BLOCK_H), BLOCK_H):
        h_offsets = h + tl.arange(0, BLOCK_H)  # shape [BLOCK_H]
        mask_cols = h_offsets < H
        # For full-row, mask_pos ensures pos is valid; for columns, mask_cols ensures H coverage
        mask = mask_pos[:, None] & mask_cols[None, :]

        # Atomic add into output
        tl.atomic_add(out_row_ptr[:, None], v, mask=mask)
