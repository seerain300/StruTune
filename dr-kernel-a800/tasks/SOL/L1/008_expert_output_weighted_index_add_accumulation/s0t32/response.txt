```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_2d_kernel(
    output_ptr,            # *bf16 or *fp16, shape [M, H]
    expert_ptr,            # *bf16 or *fp16, shape [N, H]
    index_ptr,             # *int32, shape [N]
    M: tl.constexpr,       # number of rows in output
    H: tl.constexpr,       # hidden size (columns)
    N: tl.constexpr,       # number of expert outputs
    BLOCK_N: tl.constexpr, # tile size along N
    BLOCK_H: tl.constexpr, # tile size along H
):
    # Program ids for 2D grid: rows tile and hidden tile
    pid_n = tl.program_id(0)
    pid_h = tl.program_id(1)

    # Compute offsets for this tile
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]

    # Masks for valid n and h
    n_mask = n_offsets < N
    h_mask = h_offsets < H

    # Load token indices for this n tile (int32)
    # Masked loads for n that are out of bounds
    idx = tl.load(index_ptr + n_offsets, mask=n_mask, other=0)  # [BLOCK_N]

    # Loop over H in chunks of BLOCK_H, perform 2D atomic adds
    start = 0
    while start < H:
        h_block = start + h_offsets  # [BLOCK_H]
        h_mask_block = h_block < H

        # Broadcast to 2D: [BLOCK_N, BLOCK_H]
        # Build pointers for expert outputs and destinations in output
        # expert outputs: base + n_offsets[:, None] * H + h_block[None, :]
        # output destinations: base + idx[:, None] * H + h_block[None, :]

        # Note: Triton uses row-major, so we can compute linear offsets as above.

        # We need to load expert chunk for each n in the tile and add to output.
        # Use nested loops over BLOCK_N and BLOCK_H to form 2D broadcasted pointers.
        for i in range(BLOCK_N):
            n = n_offsets[i]
            n_valid = n < N
            # For rows beyond N, skip by masking (vals will be zeros)
            # Load expert values for this row and h-block
            vals = tl.load(
                expert_ptr + n * H + h_block,
                mask=h_mask_block & n_valid,
                other=0.0,
            )  # [BLOCK_H]
            # Compute destination offsets
            dest_offsets = idx[i] * H + h_block  # [BLOCK_H]
            # Atomic add into output
            tl.atomic_add(
                output_ptr + dest_offsets,
                vals,
                mask=h_mask_block & n_valid,
            )

        start += BLOCK_H

            token_indices_i32 = token_indices.to(torch.int32)
        else:
            token_indices_i32 = token_indices

        # Shapes
        M = final_hidden_states.shape[0]
        H = final_hidden_states.shape[1]
        N = expert_outputs.shape[0]

        # Make tensors contiguous
        output = final_hidden_states.contiguous()
        expert = expert_outputs.contiguous()
        indices = token_indices_i32.contiguous()

        # Choose tile sizes adaptively
        if H >= 2048:
            BLOCK_H = 256
            num_warps = 8
            num_stages = 2
        elif H >= 512:
            BLOCK_H = 128
            num_warps = 4
            num_stages = 2
        else:
            BLOCK_H = 64
            num_warps = 2
            num_stages = 2

        BLOCK_N = 64
        grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(H, BLOCK_H))

        scatter_add_2d_kernel[grid](
            output, expert, indices,
            M, H, N,
            BLOCK_N=BLOCK_N, BLOCK_H=BLOCK_H,
            num_warps=num_warps, num_stages=num_stages,
        )
        return output
```
