import triton
import triton.language as tl

    H: tl.constexpr,     # int: hidden size (number of columns)
    T: tl.constexpr,     # int: number of expert outputs (source rows)
):

    i = tl.program_id(0)
    # Guard: if i >= T, do nothing (in case grid > T, but we set grid == T)
    # Load token index (int64), then cast to int32 for pointer arithmetic
    idx64 = tl.load(indices_ptr + i)  # scalar
    # Triton allows mixing int64/int32; casting is safe for row addressing

    # Loop over hidden columns and add elementwise
    # This scalar approach minimizes vectorization-related precision differences.
    for h in range(0, H):

        v = tl.load(expert_ptr + i * H + h)  # scalar load of row i, column h
        # Store into output row at idx, column h
        tl.store(output_ptr + idx * H + h, v)

import triton
import triton.language as tl

    expert_ptr,          # *const bfloat16, shape (T, H)
