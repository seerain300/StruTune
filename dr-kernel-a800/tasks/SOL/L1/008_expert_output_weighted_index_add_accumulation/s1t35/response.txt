import torch
import triton
import triton.language as tl


def _select_block_h_and_warps(H: int):
    # Choose BLOCK_H as next power-of-two of H, clamped to [128, 1024].
    # This minimizes loop iterations over the hidden dimension.
    if H <= 128:
        block_h = 128
    elif H <= 256:
        block_h = 256
    elif H <= 512:
        block_h = 512
    else:
        block_h = 1024  # H is typically 1024 in these workloads
    # Heuristic for num_warps based on tile size
    if block_h <= 256:
        num_warps = 4
    elif block_h <= 512:
        num_warps = 8
    else:
        num_warps = 8
    return block_h, num_warps


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,          # *const bfloat16, shape [M, H]
    src_ptr,          # *const bfloat16, shape [N, H]
    indices_ptr,      # *const int32,    shape [N]
    M: tl.constexpr,  # total rows in out_ptr/src_ptr (used only for bounds checking)
    N: tl.constexpr,  # total source rows (launch grid size)
    H: tl.constexpr,  # hidden size
    BLOCK_H: tl.constexpr,  # tile size for H
):
    # One Triton program per source row i
    i = tl.program_id(0)  # i in [0, N)
    # Guard against out-of-range program ids (safety)
    if i >= N:
        return

    # Load the token index for this row
    idx = tl.load(indices_ptr + i)  # int32
    # If idx is out of [0, M), skip (defensive; indices are expected valid)
    if (idx < 0) or (idx >= M):
        return

    # Vectorized traversal over H in tiles of size BLOCK_H
    offs_h = tl.arange(0, BLOCK_H)  # column offsets
    # Loop over tiles: range(0, H, BLOCK_H) -> number of tiles = ceil_div(H, BLOCK_H)
    for start in range(0, H, BLOCK_H):
        cols = start + offs_h
        mask = cols < H

        # Load src[i, cols]
        src_row_ptr = src_ptr + i * H + cols
        src_vals = tl.load(src_row_ptr, mask=mask, other=0.0)  # bfloat16

        # Compute output row base pointer
        out_row_ptr = out_ptr + idx * H + cols

        # Atomic add into output
        # Note: tl.atomic_add supports bfloat16 accumulation in Triton
        tl.atomic_add(out_row_ptr, src_vals, mask=mask)
