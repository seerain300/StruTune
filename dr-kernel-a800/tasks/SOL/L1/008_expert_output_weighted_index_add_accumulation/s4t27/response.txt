```python
import torch
import triton
import triton.language as tl


@triton.jit
def _copy_and_scatter_add_kernel(
    output_ptr,            # *bf16, shape (B, H)
    final_ptr,             # *bf16, shape (B, H)
    expert_ptr,            # *bf16, shape (T, H)
    indices_ptr,           # *int64, shape (T,)
    B: tl.constexpr,       # int
    H: tl.constexpr,       # int
    T: tl.constexpr,       # int
):
    # Phase 1: copy final_hidden_states into output (clone). Row-major: row_offset = b * H
    b = 0
    while b < B:
        # copy entire row b
        row_offset = b * H
        h = 0
        while h < H:
            val = tl.load(final_ptr + row_offset + h)
            tl.store(output_ptr + row_offset + h, val)
            h += 1
        b += 1

    # Phase 2: scatter-add: output[token_indices[i]] += expert_outputs[i]
    i = 0
    while i < T:
        # load idx as int64, then cast to int32 for pointer arithmetic
        idx64 = tl.load(indices_ptr + i)
        idx = idx64.to(tl.int32)
        # loop over hidden dimension
        h = 0
        while h < H:
            val = tl.load(expert_ptr + i * H + h)
            tl.store(output_ptr + idx * H + h, tl.load(output_ptr + idx * H + h) + val, mask=1)
            h += 1
        i += 1


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-optimized implementation of:
            output = final_hidden_states.clone()
            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
        We perform both the clone and the scatter-add in a single Triton kernel to ensure correctness.
        """
        # Ensure tensors are on CUDA for Triton
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, \
            "All tensors must be on CUDA for Triton."
        assert final_hidden_states.dtype == torch.bfloat16 and expert_outputs.dtype == torch.bfloat16, \
            "final_hidden_states and expert_outputs must be bfloat16."

        # Ensure contiguous
        final = final_hidden_states.contiguous()
        expert = expert_outputs.contiguous()
        indices = token_indices.contiguous()

        B, H = final.shape
        T = indices.shape[0]

        # Allocate output (we'll fill it via the Triton kernel)
        output = torch.empty((B, H), dtype=final.dtype, device=final.device)

        # Launch Triton kernel: one program per phase's iteration is fine; here we use a single grid with enough programs.
        # To keep it simple and robust, we launch a 1D grid with size B*H for copy and T*H for scatter-add via while loops inside.
        # Triton requires an integer grid size; we choose grid=(1,) and iterate in the kernel. Alternatively, we can use nested loops.
        # Using grid=(1,) is fine; the kernel uses while loops to cover all rows/columns.
        grid = (1,)

        _copy_and_scatter_add_kernel[grid](
            output, final, expert, indices,
            B=B, H=H, T=T,
            num_warps=1,
            num_stages=1,
        )

        return output
```
