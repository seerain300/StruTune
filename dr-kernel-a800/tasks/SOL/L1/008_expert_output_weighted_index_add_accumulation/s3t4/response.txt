```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_no_atomic_kernel(
    out_ptr,  # fp32 output buffer [M, H]
    A_ptr,    # fp32 expert_outputs buffer [N, H]
    idx_ptr,  # int32 token_indices buffer [N]
    N,        # number of tokens (rows to add)
    H,        # hidden size (number of columns)
    BLOCK_SIZE: tl.constexpr,
):
    # One program per token i
    i = tl.program_id(0)

    # Load the target row index for this token (int32)
    idx_i = tl.load(idx_ptr + i)

    # Iterate across the hidden dimension in chunks of BLOCK_SIZE
    # We do sequential masked loads/stores to avoid atomics and ensure correctness.
    for h in range(0, H):
        offs = h + tl.arange(0, BLOCK_SIZE)
        mask = offs < H

        # Compute pointers
        # out_ptr is [M, H], row idx_i, cols offs
        out_row_ptr = out_ptr + idx_i * H
        A_row_ptr = A_ptr + i * H

        # Load current out row chunk and A row chunk (fp32)
        out_vals = tl.load(out_row_ptr + offs, mask=mask, other=0.0)
        v = tl.load(A_row_ptr + offs, mask=mask, other=0.0)

        # Add and store back
        out_vals += v
        tl.store(out_row_ptr + offs, out_vals, mask=mask)

    def forward(self, final_hidden_states: torch.Tensor,
                expert_outputs: torch.Tensor,

            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
        Returns output cast to bfloat16 to match original interface.
```

```python
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_chunked_kernel(
    out_ptr,   # *fp32, shape [M, H], M = batch_seq_len
    A_ptr,     # *fp32, shape [N, H]

    N,         # int32, number of tokens
    H,         # int32, hidden size
    BLOCK_SIZE: tl.constexpr,  # chunk size for columns
):

    idx_i = tl.load(idx_ptr + i)

    # out_ptr is [M, H], A_ptr is [N, H]
```
