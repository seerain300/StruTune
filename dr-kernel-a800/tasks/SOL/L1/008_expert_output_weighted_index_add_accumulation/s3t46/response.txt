import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_full_row_multi_update(
    out_ptr,            # *float32, shape [B, H]
    idx_ptr,            # *int32,   shape [N]
    src_ptr,            # *float32, shape [N, H] (row i to idx[i])
    N,                  # int32: number of updates
    B,                  # int32: number of tokens (unused but kept for symmetry)
    H,                  # int32: hidden size
    T: tl.constexpr,    # number of updates per program
    BLOCK_H: tl.constexpr,  # vector width
):
    pid = tl.program_id(0)
    base = pid * T
    # lane offsets [0..T)
    k = tl.arange(0, T)
    mask_lane = k < (N % T if N % T != 0 else T)  # lane is valid for tail

    # Compute per-lane absolute positions
    pos = base + k
    mask_pos = pos < N

    # Load indices for each lane
    idx = tl.load(idx_ptr + pos, mask=mask_pos & mask_lane, other=0)
    # Compute row start pointers for each lane
    row_start = idx * H

    # Load source rows (vector across H) for each lane
    v = tl.load(
        src_ptr + pos[:, None] * H + tl.arange(0, BLOCK_H)[None, :],
        mask=(mask_pos[:, None] & mask_lane[:, None]) & (tl.arange(0, BLOCK_H)[None, :] < H),
        other=0.0
    )  # shape [T, BLOCK_H], float32

    # Atomic add each lane's row into output
    # Note: for lanes where pos >= N or lane >= tail, v is zero and atomic_add is masked.
    tl.atomic_add(out_ptr + row_start[:, None] + tl.arange(0, BLOCK_H)[None, :],
                  v,
                  mask=(mask_pos[:, None] & mask_lane[:, None]) &
                  (tl.arange(0, BLOCK_H)[None, :] < H))

    N,                  # int32
    B,                  # int32
    H,                  # int32
    T: tl.constexpr,    # updates per program
    BLOCK_H: tl.constexpr,  # chunk size
):

    base = pid * T
    k = tl.arange(0, T)
    mask_lane = k < (N % T if N % T != 0 else T)
    pos = base + k
    mask_pos = pos < N
    idx = tl.load(idx_ptr + pos, mask=mask_pos & mask_lane, other=0)
    row_start = idx * H

    # Iterate over hidden dimension in BLOCK_H chunks
    h = 0
    while h < H:
        offs = tl.arange(0, BLOCK_H)
        h_offsets = h + offs
        mask_chunk = h_offsets < H

        # Load v for each lane: shape [T, BLOCK_H]
        v = tl.load(
            src_ptr + pos[:, None] * H + h_offsets[None, :],
            mask=(mask_pos[:, None] & mask_lane[:, None]) & mask_chunk[None, :],
            other=0.0

        tl.atomic_add(
            out_ptr + row_start[:, None] + h_offsets[None, :],
            v,
            mask=(mask_pos[:, None] & mask_lane[:, None]) & mask_chunk[None, :]
        )
        h += BLOCK_H

                            expert_outputs: torch.Tensor,

    - Accumulates in float32 (required for Triton atomic_add).
    - Returns tensor cast to bfloat16 (to match original interface).

        num_warps = 8
    elif H >= 512:
        BLOCK_H = 512
        num_warps = 4
    else:
        BLOCK_H = 256
        num_warps = 2

    grid = lambda META: (triton.cdiv(N, T),)

    if H <= BLOCK_H:
        # Fast path: write entire rows in one atomic add per lane
        _scatter_add_full_row_multi_update[grid](

            num_warps=num_warps, num_stages=4
        )
