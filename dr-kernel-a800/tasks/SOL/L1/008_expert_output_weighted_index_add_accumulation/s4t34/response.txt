import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: per-source-row accumulation along hidden dimension.
# Each program handles one row i and loops over H, updating output[token_indices[i], h] += expert_outputs[i, h]
if TRITON_AVAILABLE:
    @triton.jit
    def scatter_add_rows_loop_kernel(
        output_ptr,          # *const TOutput (we'll assume bfloat16)
        expert_ptr,          # *const TExpert (bfloat16)
        indices_ptr,         # *const int64
        B: tl.constexpr,     # batch_seq_len (rows in output)
        H: tl.constexpr,     # hidden_size (columns)
        T: tl.constexpr,     # number of expert outputs
        BLOCK_H: tl.constexpr,  # loop tile over H (set to 1 for robustness)
    ):
        # One program per source row i
        i = tl.program_id(0)
        # Guard: if i >= T, return (safety, though grid should match)
        if i >= T:
            return

        # Load the destination row index (int64)
        idx64 = tl.load(indices_ptr + i)
        # Cast to int32 for address arithmetic (B is number of rows; idx must be < B)
        idx = idx64.to(tl.int32)

        # Loop over hidden dimension in chunks of BLOCK_H
        # For robustness and correctness, set BLOCK_H=1 to avoid vectorization issues.
        # This ensures deterministic behavior and minimizes numerical discrepancies.
        for h in range(0, H):
            # Compute pointers for the current element
            # output is row-major: offset = idx * H + h
            out_off = idx * H + h
            # expert_outputs row-major: offset = i * H + h
            exp_off = i * H + h

            # Load v from expert_outputs in bf16
            v = tl.load(expert_ptr + exp_off)
            # Load the current output value (may be random from clone), add v
            out_val = tl.load(output_ptr + out_off)
            out_val = out_val + v
            # Store back
            tl.store(output_ptr + out_off, out_val)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-optimized version of:
            output = final_hidden_states.clone()
            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
        We implement the index_add via a Triton kernel using a deterministic per-element update.
        """
        # Fallback to PyTorch if Triton not available or tensors not on CUDA
        if (not TRITON_AVAILABLE) or (not final_hidden_states.is_cuda) or (not expert_outputs.is_cuda) or (not token_indices.is_cuda):
            # Fallback: exact PyTorch behavior
            output = final_hidden_states.clone()
            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
            return output

        # Ensure contiguity
        output = final_hidden_states.clone()
        output = output.contiguous()
        expert_outputs = expert_outputs.contiguous()
        token_indices = token_indices.contiguous()

        # Shapes
        B = output.shape[0]  # batch_seq_len
        H = output.shape[1]  # hidden_size
        T = token_indices.shape[0]  # number of expert outputs

        # Launch Triton kernel: one program per source row
        # To maximize correctness and avoid vectorization, we use a simple loop over H.
        # Note: grid=(T,) ensures one program per row i.
        # num_warps=1 keeps it simple and deterministic for small H; increase if needed.
        scatter_add_rows_loop_kernel[(T,)](
            output, expert_outputs, token_indices,
            B=B, H=H, T=T, BLOCK_H=1,
            num_warps=1, num_stages=1,
        )

        return output
