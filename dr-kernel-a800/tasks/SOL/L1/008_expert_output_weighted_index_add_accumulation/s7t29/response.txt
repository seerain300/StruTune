```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_per_row_chunked_kernel(
    out_ptr,        # *bf16, pointer to out tensor [N, H]
    src_ptr,        # *bf16, pointer to src tensor [M, H]
    indices_ptr,    # *int32, pointer to token indices [M]
    M,              # int32, number of rows in src (num_selected_tokens)
    N,              # int32, number of rows in out (batch_seq_len)
    H,              # int32, hidden size
    BLOCK_SIZE: tl.constexpr,  # chunk size for unrolling along H
):
    pid = tl.program_id(axis=0)
    if pid >= M:
        return

    # Destination row index for this source row
    dst = tl.load(indices_ptr + pid)
    # dst must be in [0, N). Reference indices are valid, but keep a defensive check.
    if (dst < 0) or (dst >= N):
        return

    # Iterate over hidden dimension in chunks
    # This is a runtime loop; no tl.static_range to avoid Triton version issues.
    for start in range(0, H, BLOCK_SIZE):
        # Compute a vector of column offsets within the chunk
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < H

        # Load source row slice
        src_offsets = pid * H + offs
        src_vals = tl.load(src_ptr + src_offsets, mask=mask, other=0.0)

        # Compute destination offsets and atomic add
        out_offsets = dst * H + offs
        # For masked-out lanes, ensure we don't perform atomic adds.
        tl.atomic_add(out_ptr + out_offsets, src_vals, mask=mask)

        H = final_hidden_states.shape[1]

        # Choose a reasonable block size for unrolling; 128 works well for typical H in your workloads.
        BLOCK_SIZE = 128

        grid = (M,)
        scatter_add_per_row_chunked_kernel[grid](
            out, expert_outputs, token_indices,
            M, N, H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=1,
            num_stages=1,
        )
        return out
```
