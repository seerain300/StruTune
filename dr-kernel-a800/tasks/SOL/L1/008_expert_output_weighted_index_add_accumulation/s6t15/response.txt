```python
import torch
import triton
import triton.language as tl


@triton.jit
def row_copy_kernel(
    src_ptr,  # *bf16
    dst_ptr,  # *bf16
    batch_seq_len,  # int
    hidden_size,    # int
    BLOCK_ROWS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Copy rows from src (final_hidden_states) to dst (output) in tiles.
    Each program handles a tile of size [BLOCK_ROWS, BLOCK_H].
    """
    pid_rows = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_rows * BLOCK_ROWS + tl.arange(0, BLOCK_ROWS)
    cols = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # Create 2D mask for in-bounds rows and cols
    mask_rows = rows < batch_seq_len
    mask_cols = cols < hidden_size
    mask = mask_rows[:, None] & mask_cols[None, :]

    # Compute flat offsets for src and dst
    # Layout assumed row-major: offset = row * hidden_size + col
    src_offsets = rows[:, None] * hidden_size + cols[None, :]
    dst_offsets = rows[:, None] * hidden_size + cols[None, :]

    # Load from src and store to dst
    # Note: Use 0.0 for out-of-bounds elements; they won't be stored due to mask.
    vals = tl.load(src_ptr + src_offsets, mask=mask, other=0.0)
    tl.store(dst_ptr + dst_offsets, vals, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-optimized implementation:
          - Copies final_hidden_states into output using a Triton kernel (robust, fast).
          - Performs index_add on the output using torch (exact PyTorch semantics).
        """
        # Ensure tensors are on the same device
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, \
            "All tensors must be on CUDA for Triton kernel."
        # Allocate output as a copy of final_hidden_states (clone semantics)
        batch_seq_len, hidden_size = final_hidden_states.shape
        output = torch.empty_like(final_hidden_states)

        # Launch Triton copy kernel in 2D grid over rows and hidden dimension
        BLOCK_ROWS = 32
        BLOCK_H = 256

        grid = (
            triton.cdiv(batch_seq_len, BLOCK_ROWS),
            triton.cdiv(hidden_size, BLOCK_H),
        )

        row_copy_kernel[grid](
            final_hidden_states,  # src
            output,                # dst
            batch_seq_len,
            hidden_size,
            BLOCK_ROWS=BLOCK_ROWS,
            BLOCK_H=BLOCK_H,
            num_warps=4,           # good balance for these tile sizes
            num_stages=2,
        )

        # Now perform scatter-add exactly as in the original: output.index_add_(dim=0, token_indices, expert_outputs)
        # token_indices are int64 by default; ensure they are on the same device and dtype-compatible.
        output.index_add_(0, token_indices, expert_outputs)

        return output
```
