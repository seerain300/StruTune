import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_row_chunk_kernel(
    output_ptr,    # *bf16 or *fp16, shape [M, H]
    expert_ptr,    # *bf16 or *fp16, shape [N, H]
    index_ptr,     # *int32, shape [N]
    M, H, N,       # int32 runtime sizes
    BLOCK_H: tl.constexpr,
):
    # One program per source row (n)
    n = tl.program_id(0)

    # Iterate over H in chunks
    start = 0
    while start < H:
        h_offsets = start + tl.arange(0, BLOCK_H)
        h_mask = h_offsets < H

        # Load token index for this source row
        dest = tl.load(index_ptr + n)

        # Compute pointers for this row chunk
        expert_row_ptr = expert_ptr + n * H + h_offsets
        out_row_ptr = output_ptr + dest * H + h_offsets

        # Load expert outputs and atomic add to output
        vals = tl.load(expert_row_ptr, mask=h_mask, other=0.0)
        tl.atomic_add(out_row_ptr, vals, mask=h_mask)

        start += BLOCK_H


@triton.jit
def scatter_add_tile_2d_kernel(
    output_ptr,    # *bf16 or *fp16, shape [M, H]
    expert_ptr,    # *bf16 or *fp16, shape [N, H]
    index_ptr,     # *int32, shape [N]
    M, H, N,       # int32 runtime sizes
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # 2D tiling: each program handles a tile of N rows and H columns
    pid_n = tl.program_id(0)
    pid_h = tl.program_id(1)

    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]

    n_mask = n_offsets < N
    h_mask = h_offsets < H

    # Broadcast masks
    mask = n_mask[:, None] & h_mask[None, :]

    # Load destination indices for each n in the tile
    dests = tl.load(index_ptr + n_offsets, mask=n_mask, other=0).to(tl.int32)  # [BLOCK_N]

    # Compute pointers for the tile
    # expert_ptr has row-major layout: element offset = n * H + h
    expert_ptrs = expert_ptr + n_offsets[:, None] * H + h_offsets[None, :]  # [BLOCK_N, BLOCK_H]
    out_ptrs = output_ptr + dests[:, None] * H + h_offsets[None, :]         # [BLOCK_N, BLOCK_H]

    vals = tl.load(expert_ptrs, mask=mask, other=0.0)  # [BLOCK_N, BLOCK_H]
    tl.atomic_add(out_ptrs, vals, mask=mask)


def _choose_row_chunk_params(H: int):
    # Heuristic for BLOCK_H and kernel launch params
    if H >= 2048:
        BLOCK_H = 256
        num_warps = 8
        num_stages = 3
    elif H >= 512:
        BLOCK_H = 128
        num_warps = 4
        num_stages = 2
    else:
        BLOCK_H = 64
        num_warps = 2
        num_stages = 2
    return BLOCK_H, num_warps, num_stages


def _choose_tile_params(H: int, N: int):
    # Heuristic for 2D tiling
    # Use larger tiles when H and N are large to reduce programs and increase per-program work.
    if H >= 1024 and N >= 1024:
        BLOCK_N = 64
        BLOCK_H = 128
        num_warps = 4
        num_stages = 2
    elif H >= 512 and N >= 512:
        BLOCK_N = 32
        BLOCK_H = 128
        num_warps = 4
        num_stages = 2
    else:
        BLOCK_N = 16
        BLOCK_H = 64
        num_warps = 2
        num_stages = 2
    return BLOCK_N, BLOCK_H, num_warps, num_stages


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor,
                expert_outputs: torch.Tensor,
                token_indices: torch.Tensor):
        # Ensure device and contiguity
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Tensors must be on CUDA"
        M, H = final_hidden_states.shape
        N = expert_outputs.shape[0]
        # Output buffer: keep the same tensor (accumulate in-place)
        output = final_hidden_states

        # Ensure types and contiguity
        # Keep expert_outputs contiguous in memory
        if not expert_outputs.is_contiguous():
            expert_outputs = expert_outputs.contiguous()
        # Use int32 for indices
        if token_indices.dtype != torch.int32:
            token_indices = token_indices.to(torch.int32)
        if not token_indices.is_contiguous():
            token_indices = token_indices.contiguous()

        # Heuristic: if H is large and N is large, use 2D tiling; otherwise use row-chunk
        use_tile = (H >= 512 and N >= 512)

        if use_tile:
            BLOCK_N, BLOCK_H, num_warps, num_stages = _choose_tile_params(H, N)
            grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(H, BLOCK_H))
            scatter_add_tile_2d_kernel[grid](
                output, expert_outputs, token_indices,
                M, H, N,
                BLOCK_N=BLOCK_N, BLOCK_H=BLOCK_H,
                num_warps=num_warps, num_stages=num_stages
            )
        else:
            # Row-chunk kernel
            BLOCK_H, num_warps, num_stages = _choose_row_chunk_params(H)
            grid = (N,)
            scatter_add_row_chunk_kernel[grid](
                output, expert_outputs, token_indices,
                M, H, N,
                BLOCK_H=BLOCK_H,
                num_warps=num_warps, num_stages=num_stages
            )

        return output
