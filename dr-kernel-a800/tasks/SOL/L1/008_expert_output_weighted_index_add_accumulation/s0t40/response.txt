import triton
import triton.language as tl

def scatter_add_row_kernel(
    output_ptr,            # *bf16 or *fp16, pointer to [M, H]
    expert_ptr,            # *bf16 or *fp16, pointer to [N, H]
    index_ptr,             # *int32, pointer to [N]
    M: tl.constexpr,       # number of rows in output (batch_size * seq_len)
    H,                     # hidden size (runtime)
    BLOCK_H: tl.constexpr  # chunk size along H
):
    # Each program handles one source row 'n' (token), and adds all H features
    n = tl.program_id(axis=0)
    if n >= H:  # guard, though typically grid=N (H<=N) and H is runtime; keep for safety
        return

    # Row base offsets
    # Address: output[index[n], :] and expert[n, :]
    dst_row = tl.load(index_ptr + n).to(tl.int32)
    # We will add chunks of H: start from 0 to H, step BLOCK_H
    start = 0
    while start < H:
        offs = start + tl.arange(0, BLOCK_H)
        mask = offs < H

        # Load expert outputs for this chunk
        expert_row_ptr = expert_ptr + n * H + offs
        vals = tl.load(expert_row_ptr, mask=mask, other=0.0)

        # Compute destination addresses: output[dst_row, offs]
        dst_ptrs = output_ptr + dst_row * H + offs
        # Atomic add into output
        tl.atomic_add(dst_ptrs, vals, mask=mask)

        start += BLOCK_H
