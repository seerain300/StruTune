No torch ops are used in the Triton kernels; all computation is performed by the Triton kernel(s).

```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_full_row_multi_update(
    out_ptr,            # *float32, shape [B, H], B = batch_seq_len
    idx_ptr,            # *int32,   shape [N]
    src_ptr,            # *float32, shape [N, H] (row i maps to idx[i])
    N,                  # int32: number of updates
    B,                  # int32: number of tokens (batch_seq_len)
    H,                  # int32: hidden_size
    T: tl.constexpr,    # number of updates per program (compile-time constant)
    BLOCK_H: tl.constexpr,  # vector width across hidden dim
):
    pid = tl.program_id(0)
    # Each program handles up to T updates starting at position pid*T
    # We use a vectorized lane k in [0, T) with masks for tail.
    for k in range(T):
        pos = pid * T + k
        # Mask to ensure we don't process out-of-range updates
        mask_update = pos < N
        # Load the target token index (int32)
        idx = tl.load(idx_ptr + pos, mask=mask_update, other=0)  # scalar int32
        # Compute base offset for the row in out_ptr
        out_base = idx * H
        # Load the source row vector (float32)
        src_row = tl.load(src_ptr + pos * H + tl.arange(0, BLOCK_H), mask=mask_update, other=0.0)
        # Atomic add the vector into out at the target row
        tl.atomic_add(out_ptr + out_base + tl.arange(0, BLOCK_H), src_row, mask=mask_update)


@triton.jit
def scatter_add_chunked_multi_update(
    out_ptr,            # *float32, shape [B, H]
    idx_ptr,            # *int32,   shape [N]
    src_ptr,            # *float32, shape [N, H]
    N,                  # int32: number of updates
    B,                  # int32: number of tokens (batch_seq_len)
    H,                  # int32: hidden_size
    T: tl.constexpr,    # number of updates per program (compile-time constant)
    BLOCK_H: tl.constexpr,  # vector width across hidden dim
):
    pid = tl.program_id(0)
    # Each program handles up to T updates starting at position pid*T
    for k in range(T):
        pos = pid * T + k
        mask_update = pos < N
        idx = tl.load(idx_ptr + pos, mask=mask_update, other=0)  # int32
        out_base = idx * H
        # Iterate over hidden dimension in chunks
        for h in range(0, H, BLOCK_H):
            offs = tl.arange(0, BLOCK_H) + h
            # Load chunk from src
            src_chunk = tl.load(src_ptr + pos * H + offs, mask=mask_update & (offs < H), other=0.0)
            # Atomic add chunk into out
            tl.atomic_add(out_ptr + out_base + offs, src_chunk, mask=mask_update)


class ModelNew(torch.nn.Module):
    def forward(
        final_hidden_states: torch.Tensor,
        expert_outputs: torch.Tensor,
        token_indices: torch.Tensor,
    ) -> torch.Tensor:
        # Ensure inputs are on the same CUDA device and contiguous
        device = final_hidden_states.device
        assert device.type == "cuda", "ModelNew requires CUDA device."
        B = final_hidden_states.shape[0]
        H = final_hidden_states.shape[1]
        N = expert_outputs.shape[0]
        # Prepare fp32 accumulation buffer (clone to avoid in-place on input)
        out_fp32 = final_hidden_states.clone().to(torch.float32).contiguous()
        # Prepare indices and source as required by kernel
        idx_i32 = token_indices.to(torch.int32).contiguous()
        src_fp32 = expert_outputs.contiguous().to(torch.float32)

        # Heuristics for BLOCK_H and num_warps
        if H >= 1024:
            BLOCK_H = 1024
            num_warps = 8
        elif H >= 512:
            BLOCK_H = 512
            num_warps = 4
        else:
            BLOCK_H = 256
            num_warps = 2

        # Number of updates per program
        T = 4  # good balance; masks handle tails

        # Grid size: each program handles T updates, so grid = ceil_div(N, T)
        grid = (triton.cdiv(N, T),)

        if H <= BLOCK_H:
            # Fast path: one atomic add per update lane (no inner loop)
            scatter_add_full_row_multi_update[grid](
                out_fp32,
                idx_i32,
                src_fp32,
                N,
                B,
                H,
                T=T,
                BLOCK_H=BLOCK_H,
                num_warps=num_warps,
                num_stages=4,
            )
        else:
            # General path: chunk over hidden dim
            scatter_add_chunked_multi_update[grid](
                out_fp32,
                idx_i32,
                src_fp32,
                N,
                B,
                H,
                T=T,
                BLOCK_H=BLOCK_H,
                num_warps=num_warps,
                num_stages=4,
            )

        # Cast back to bfloat16 to match original interface
        return out_fp32.to(torch.bfloat16)
```

```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_2d_chunks(
    out_ptr,        # *float32, shape [B, H], B = batch_seq_len
    idx_ptr,        # *int32,   shape [N]
    src_ptr,        # *float32, shape [N, H]
    N,              # int32: number of updates (len(token_indices))
    B,              # int32: number of tokens (batch_seq_len), not used directly
    H,              # int32: hidden_size
    BLOCK_H: tl.constexpr,  # vector width across hidden dim
):
    # 2D grid: pid0 over tokens, pid1 over hidden chunks
    pid_i = tl.program_id(0)  # token index
    pid_c = tl.program_id(1)  # chunk index over hidden dimension

    # Mask for valid token
    mask_i = pid_i < N
    # Compute hidden offsets for this chunk
    h_start = pid_c * BLOCK_H
    offs = h_start + tl.arange(0, BLOCK_H)
    mask_h = offs < H

    # Load target token index
    idx = tl.load(idx_ptr + pid_i, mask=mask_i, other=0)  # int32 scalar
    # Compute row base offset in output
    out_row_base = idx * H

    # Compute source row base offset
    src_row_base = pid_i * H

    # Load source chunk (vector), masked for H tail
    v = tl.load(src_ptr + src_row_base + offs, mask=mask_i & mask_h, other=0.0)

    # Atomic add into output row chunk
    tl.atomic_add(out_ptr + out_row_base + offs, v, mask=mask_i & mask_h)

    def forward(
        final_hidden_states: torch.Tensor,
        expert_outputs: torch.Tensor,
        token_indices: torch.Tensor,
    ) -> torch.Tensor:
        # Device checks
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "CUDA tensors required."
        device = final_hidden_states.device

        # Shapes
        B = final_hidden_states.shape[0]
        H = final_hidden_states.shape[1]
        N = expert_outputs.shape[0]

        src_fp32 = expert_outputs.contiguous().to(torch.float32)
```
