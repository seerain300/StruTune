import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_per_col_kernel(
    output_ptr,       # *const bfloat16, shape (B, H)
    expert_ptr,       # *const bfloat16, shape (T, H)
    indices_ptr,      # *const int64,    shape (T,)
    B: tl.constexpr,  # int
    H: tl.constexpr,  # int
    T: tl.constexpr,  # int
):
    """
    One program per source row i. For each hidden column h, write:
        output[token_indices[i], h] = expert_outputs[i, h]
    This matches PyTorch's index_add(dim=0, index=token_indices, source=expert_outputs)
    without using atomics, ensuring deterministic write order.
    """
    i = tl.program_id(0)  # program id along the "row" source dimension (i in [0, T))
    # Guard: if grid > T, early return (though we set grid=(T,) so not necessary)
    # We still do bounds checks just in case.
    if i >= T:
        return

    # Load token index (int64), then cast to int32 for pointer arithmetic
    idx64 = tl.load(indices_ptr + i)
    idx = idx64.to(tl.int32)

    # Iterate over hidden dimension columns
    # Note: we treat H as a runtime integer (Triton will JIT-specialize per H),
    # but we keep a loop to minimize any vectorization-related precision discrepancies.
    for h in range(0, H):
        # Load expert value as bfloat16
        v = tl.load(expert_ptr + i * H + h, eviction_policy="evict_last")
        # Compute output pointer for row idx, column h
        out_ptr = output_ptr + idx * H + h
        # Store v; if idx is invalid, Triton will not store (but idx is in [0, B)),
        # and we already ensured token_indices are valid. Still guard explicitly.
        if idx >= 0 and idx < B:
            tl.store(out_ptr, v)

        )
