import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    output_ptr,        # *bf16, shape [B, H]
    expert_ptr,        # *bf16, shape [T, H]
    indices_ptr,       # *i64,  shape [T]

    H: tl.constexpr,   # hidden_size (number of columns)
    T: tl.constexpr,   # number of source rows (num_selected_tokens)
    BLOCK_COLS: tl.constexpr,  # number of columns processed per iteration
):

    i = tl.program_id(0)

    # Load token index (row in output to add to)
    idx64 = tl.load(indices_ptr + i)  # int64
    idx = idx64.to(tl.int32)

    # Bound check: ensure idx is within [0, B)
    # (inputs should be valid, but this makes the kernel robust)
    valid = (idx >= 0) & (idx < B)

    # Loop over hidden columns in blocks of BLOCK_COLS
    for col_start in range(0, H, BLOCK_COLS):
        cols = col_start + tl.arange(0, BLOCK_COLS)
        mask = (cols < H) & valid

        # Load the vector of size BLOCK_COLS from expert_outputs[i, :]
        # Note: expert_outputs assumed row-major contiguous: offset = i * H + cols
        v = tl.load(expert_ptr + i * H + cols, mask=mask, other=0.0)  # bf16

        # Compute output row base: idx * H + cols
        out_ptrs = output_ptr + idx * H + cols
        tl.store(out_ptrs, v, mask=mask)
