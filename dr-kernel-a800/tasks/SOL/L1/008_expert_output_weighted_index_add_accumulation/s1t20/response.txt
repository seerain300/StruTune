import torch
import triton
import triton.language as tl


def _next_power_of_two(x: int) -> int:
    # Compute next power of two >= x
    if x <= 1:
        return 1
    return 1 << ((x - 1).bit_length())


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,           # *const bfloat16, shape [M, H]
    src_ptr,           # *const bfloat16, shape [N, H]
    indices_ptr,       # *const int32,    shape [N]
    M,                 # int32
    H,                 # int32
    BLOCK_H: tl.constexpr,
):
    # Each program handles one source row i
    row = tl.program_id(axis=0)  # i in [0, N)
    # Bounds check for grid
    if row >= tl.num_programs(axis=0):
        return

    # Compute base pointers for this row
    # out_row_ptr points to out[row % M, :]
    # We assume grid launches exactly N programs, but keep a safe index
    row_in_out = row % M
    out_row_ptr = out_ptr + row_in_out * H
    src_row_ptr = src_ptr + row * H
    idx = tl.load(indices_ptr + row)  # int32

    # Vectorize across hidden dimension in tiles of size BLOCK_H
    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask = h_offsets < H

        # Load source values for this tile
        src_vals = tl.load(src_row_ptr + h_offsets, mask=mask, other=0.0)

        # Compute destination pointer for the token index
        dest_ptr = out_ptr + idx * H + h_offsets

        # Atomic add into output
        tl.atomic_add(dest_ptr, src_vals, mask=mask)


def _select_block_h_and_warps(H: int):
    # Choose BLOCK_H as next power-of-two of H, clamped to [128, 1024]
    bh = _next_power_of_two(H)
    if bh < 128:
        bh = 128
    if bh > 1024:
        bh = 1024
    # Heuristic for num_warps
    if bh <= 256:
        num_warps = 4
    elif bh <= 512:
        num_warps = 8
    else:
        num_warps = 8
    return bh, num_warps

    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):

        output[token_indices[i]] += expert_outputs[i] for i in [0, N),

        N = src.shape[0]
        H = out.shape[1]

            M, H,
            BLOCK_H=BLOCK_H,
            num_warps=num_warps,
            num_stages=2,
        )
        return out
