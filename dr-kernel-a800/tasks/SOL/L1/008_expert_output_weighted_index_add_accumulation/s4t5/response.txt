import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_per_element_kernel(
    output_ptr,      # *bf16, shape (B, H)
    expert_ptr,      # *bf16, shape (T, H)
    indices_ptr,     # *int64, shape (T,)
    B: tl.int32,     # batch_seq_len (rows in output)
    H: tl.int32,     # hidden_size (cols)
    T: tl.int32,     # number of source rows
    BLOCK_H: tl.constexpr,  # number of columns processed per program
):
    # Each program handles one source row i
    i = tl.program_id(0)
    if i >= T:
        return

    # Load the destination row index for this source row
    idx64 = tl.load(indices_ptr + i)  # int64
    idx = idx64.to(tl.int32)          # Triton expects int32 for pointer arithmetic

    # Iterate over hidden dimension in tiles
    # Note: We use a simple loop here to avoid any potential vectorization issues
    # and to ensure deterministic per-element addition.
    for h_start in range(0, H, BLOCK_H):
        offs = h_start + tl.arange(0, BLOCK_H)  # [BLOCK_H] vector of column indices
        mask = offs < H

        # Load the source vector (bf16) for this row i
        v = tl.load(expert_ptr + i * H + offs, mask=mask, other=0.0)

        # Compute destination addresses: output[idx, offs]
        dest_addr = idx * H + offs
        # Store/add: since output was cloned and index_add would perform +=,
        # we need to add v to existing output. We do this by loading, adding, and storing back.
        # But to avoid extra loads, we can directly atomic add if we switch to that approach.
        # For correctness now, we implement a two-phase: write zeros initially and then add.
        # However, since the original uses index_add on a pre-existing tensor, we can do:
        # We will do in-place add by loading existing, adding, and storing back.
        # To minimize complexity, we'll do the add via an atomic path (below), which is more robust.
        # Here, we simply perform atomic add once, which is fine as long as H is modest.
        # However, to avoid potential atomic precision differences, we implement deterministic add:
        # We will load current output row slice, add v, and store back. Since we don't have output ptr
        # in this function, we cannot load; so we use atomic add in the next kernel.

        # Instead of trying to read output (which we cannot from this kernel), we directly
        # perform atomic add. This is the correct approach for scatter-add semantics.
        # Triton supports atomic_add for bf16 in recent versions; we guard by using bf16 pointers.
        tl.atomic_add(output_ptr + dest_addr, v, mask=mask)
