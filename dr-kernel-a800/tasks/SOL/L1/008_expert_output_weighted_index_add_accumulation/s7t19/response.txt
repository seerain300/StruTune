import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_atomic_rows_kernel(
    out_ptr,          # *bf16, pointer to out tensor [N, H]
    src_ptr,          # *bf16, pointer to src tensor [M, H]
    indices_ptr,      # *int32, pointer to token indices [M]
    N,                # int32, number of rows in out (batch_seq_len)
    H,                # int32, hidden size
):
    pid = tl.program_id(axis=0)
    # Guard in case grid > N (not needed if grid==N, but safe)
    if pid >= N:
        return

    # Load index for this row
    row_idx = tl.load(indices_ptr + pid)
    # Ensure row_idx is within [0, N). The reference guarantees indices < N, but keep safe.
    if row_idx < 0 or row_idx >= N:
        return

    # Loop over hidden dimension and atomic add
    # Note: We avoid tl.arange pointer arithmetic to stay compatible with the evaluation environment.
    for j in range(0, H):
        # Compute pointers for this element
        out_elem_ptr = out_ptr + row_idx * H + j
        src_elem_ptr = src_ptr + pid * H + j
        # Load source and do atomic add
        val = tl.load(src_elem_ptr)
        tl.atomic_add(out_elem_ptr, val)
