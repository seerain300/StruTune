# Triton imports
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: scatter-add expert_outputs[i, :] into output[token_indices[i], :]
# One program per selected token index (i). Vectorize across hidden dimension in chunks of BLOCK_SIZE=256.
# Robust handling of any hidden_size via masking.
@triton.jit
def scatter_add_rows_kernel(
    output_ptr,         # *ptr to output [rows, hidden]
    expert_ptr,         # *ptr to expert_outputs [num_selected_tokens, hidden]
    indices_ptr,        # *ptr to token_indices [num_selected_tokens] (int32)
    rows,               # number of rows in output (i.e., batch_size * seq_len)
    hidden,             # hidden_size (number of columns)
    num_selected,       # number of selected tokens
    BLOCK_SIZE: tl.constexpr,  # chunk size across hidden dimension
):

    i = tl.program_id(axis=0)
    # guard: if grid > num_selected, do nothing (typically grid == num_selected)
    if i >= num_selected:
        return

    # load the destination row index for this selected token
    row_idx = tl.load(indices_ptr + i)

    # bounds check on row_idx (in case of any misconfiguration)
    if row_idx < 0 or row_idx >= rows:
        return

    # Iterate over hidden dimension in chunks
    # We use a loop with masking to handle tails safely.
    for k in range(0, hidden, BLOCK_SIZE):
        offs = k + tl.arange(0, BLOCK_SIZE)
        # mask for valid hidden positions in this chunk
        mask = offs < hidden

        # Compute pointers for output row slice and expert slice
        output_row_ptr = output_ptr + row_idx * hidden + offs
        expert_row_ptr = expert_ptr + i * hidden + offs

        # Load current output slice and expert slice under mask
        out_slice = tl.load(output_row_ptr, mask=mask, other=0.0)
        exp_slice = tl.load(expert_row_ptr, mask=mask, other=0.0)

        # Atomic add only for masked positions
        if tl.any(mask):
            tl.atomic_add(output_row_ptr, exp_slice, mask=mask)

    hidden = output.shape[1]
