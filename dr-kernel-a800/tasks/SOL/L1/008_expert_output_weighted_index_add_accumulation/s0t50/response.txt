- The operation is a scatter-add along dim=0: output[token_indices[i]] += expert_outputs[i]. Output is shape (M, H) where M = batch_size * seq_len; expert_outputs is (N, H); token_indices length N.
- Our Triton kernels have been correct and delivered speedups between ~1.27 and ~1.83. The variability suggests that the dominant cost is atomic contention and the distribution of token_indices (how many indices map to the same row). Scatter-add inherently has contention when indices are not unique.
- We cannot avoid atomics unless we implement a two-phase per-row reduction (block-local reduction + second pass), which is more complex and may not improve in general. The simpler, robust path is to maximize per-program work, improve memory access, and tune launch parameters.

Optimization strategy
- Keep one program per source row (n). Each program handles that row’s H in chunks to maintain simplicity and minimize index-dependent branching.
- Increase per-program work by choosing larger BLOCK_H for larger H to reduce loop iterations and improve throughput. Use:
  - BLOCK_H = 256 when H >= 1024
  - BLOCK_H = 128 when H >= 256
  - BLOCK_H = 64 otherwise
- Tune num_warps and num_stages:
  - num_warps = 8 for BLOCK_H >= 256
  - num_warps = 4 for BLOCK_H >= 128
  - num_warps = 2 for BLOCK_H = 64
  - num_stages = 3 for BLOCK_H >= 128, else 2
- Ensure inputs are contiguous and use int32 for token_indices (convert from long on host if needed). This keeps pointer arithmetic simple and efficient.
- Keep the while loop over H with runtime termination, which avoids Triton compile-time constraints and works for arbitrary H.
- While we can’t remove atomic contention entirely, the above tiling and warp tuning should help maintain and improve performance across a range of shapes.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_row_kernel(
    output_ptr,            # *bf16 or *fp16 pointer, shape [M, H]
    expert_ptr,            # *bf16 or *fp16 pointer, shape [N, H]
    index_ptr,             # *int32 pointer, shape [N]
    M: tl.constexpr,       # number of rows in output
    H: tl.constexpr,       # number of columns (hidden size)
    BLOCK_H: tl.constexpr  # chunk size over H
):
    # One program per source row (n in [0, N))
    n = tl.program_id(0)

    # Load the destination row index for this source row
    dest_row = tl.load(index_ptr + n)
    # Guard: if index is out of range, skip (defensive; not expected in benchmark)
    if dest_row < 0 or dest_row >= M:
        return

    start = 0
    while start < H:
        h_offsets = start + tl.arange(0, BLOCK_H)
        mask = h_offsets < H

        # Load expert values for this row chunk
        expert_vals = tl.load(expert_ptr + n * H + h_offsets, mask=mask, other=0.0)

        # Compute output pointers for this chunk and atomically add
        out_ptrs = output_ptr + dest_row * H + h_offsets
        # Atomic add accumulates contributions from all source rows with the same dest_row
        tl.atomic_add(out_ptrs, expert_vals, mask=mask)

        start += BLOCK_H


def _choose_block_params(H: int):
    # Heuristic tuning for block size and warps
    if H >= 1024:
        return 256, 8, 3
    elif H >= 256:
        return 128, 4, 3
    else:
        return 64, 2, 2


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-optimized scatter-add:
        output[token_indices[i]] += expert_outputs[i]
        - output: (M, H), dtype bfloat16/fp16
        - expert_outputs: (N, H)
        - token_indices: (N,) int64; we will convert to int32 for kernel
        """
        # Ensure device is CUDA and tensors are contiguous
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Triton requires CUDA tensors."
        output = final_hidden_states
        # Ensure output is contiguous row-major
        if not output.is_contiguous():
            output = output.contiguous()
        # Ensure expert_outputs is contiguous (usually true in this setup)
        if not expert_outputs.is_contiguous():
            expert_outputs = expert_outputs.contiguous()

        # Convert indices to int32 for efficient address arithmetic
        index32 = token_indices.to(torch.int32)

        M = output.shape[0]
        H = output.shape[1]
        N = index32.shape[0]

        BLOCK_H, num_warps, num_stages = _choose_block_params(H)

        # Launch one program per source row
        grid = (N,)
        scatter_add_row_kernel[grid](
            output, expert_outputs, index32,
            M, H, BLOCK_H,
            num_warps=num_warps,
            num_stages=num_stages,
        )
        return output
```

```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_tile_kernel(
    output_ptr,            # *bf16 or *fp16, shape [M, H]
    expert_ptr,            # *bf16 or *fp16, shape [N, H]
    index_ptr,             # *int32, shape [N]
    M: tl.constexpr,       # number of rows in output
    N: tl.constexpr,       # number of source rows (token indices length)
    H: tl.constexpr,       # hidden size
    BLOCK_M: tl.constexpr, # number of source rows per program
    BLOCK_H: tl.constexpr  # chunk size over H
):
    pid = tl.program_id(0)
    # Compute the slice of source rows this program will handle
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)
    mask_rows = rows < N

    # Load destination row indices for these source rows
    dest_rows = tl.load(index_ptr + rows, mask=mask_rows, other=0)
    # Compute base output pointers for each dest row (vector of BLOCK_M)
    out_bases = dest_rows * H  # int32 vector

    # Iterate over H in chunks
    start_h = 0
    while start_h < H:
        h_offsets = start_h + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H

        # Loop over each source row in the tile and atomically add its H-chunk
        # This keeps atomic adds per feature per source row but increases per-program work.
        for i in range(BLOCK_M):
            # Only process valid source rows
            valid_i = row_start + i < N
            # If not valid, skip this iteration; Triton supports Python-level control flow
            if not valid_i:
                continue
            # Compute base pointer for expert row i and current H-chunk
            expert_base = (row_start + i) * H
            vals = tl.load(expert_ptr + expert_base + h_offsets, mask=mask_h, other=0.0)
            # Atomic add to output for each dest row in this tile
            out_ptrs = output_ptr + out_bases[i] + h_offsets
            tl.atomic_add(out_ptrs, vals, mask=mask_h & mask_rows[i])

        start_h += BLOCK_H


def _choose_block_params(N: int, H: int):
    # Heuristic tuning for tile sizes and launch config
    # Choose BLOCK_M based on N to reduce program count
    BLOCK_M = 8 if N >= 4096 else 4
    # Choose BLOCK_H and warps/stages based on H
    if H >= 1024:
        BLOCK_H, num_warps, num_stages = 256, 8, 3
    elif H >= 256:
        BLOCK_H, num_warps, num_stages = 128, 4, 3
    else:
        BLOCK_H, num_warps, num_stages = 64, 2, 2
    return BLOCK_M, BLOCK_H, num_warps, num_stages


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-optimized scatter-add:
        output[token_indices[i]] += expert_outputs[i]
        - output: (M, H), dtype bfloat16/fp16
        - expert_outputs: (N, H)
        - token_indices: (N,) int64; converted to int32 for kernel
        """
        # Triton requires CUDA tensors
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Triton requires CUDA tensors."
        output = final_hidden_states
        # Ensure contiguous row-major layout
        if not output.is_contiguous():
            output = output.contiguous()
        if not expert_outputs.is_contiguous():
            expert_outputs = expert_outputs.contiguous()

        # Convert indices to int32 for efficient address arithmetic
        index32 = token_indices.to(torch.int32)

        M = output.shape[0]
        N = index32.shape[0]
        H = output.shape[1]

        BLOCK_M, BLOCK_H, num_warps, num_stages = _choose_block_params(N, H)

        # Launch one program per tile of source rows
        grid = (triton.cdiv(N, BLOCK_M),)

        scatter_add_tile_kernel[grid](
            output, expert_outputs, index32,
            M, N, H,
            BLOCK_M, BLOCK_H,
            num_warps=num_warps,
            num_stages=num_stages,
        )
        return output
```
