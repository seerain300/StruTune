```python
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: vectorized across hidden dimension (columns), 1D grid over rows.
# out: (N, H), expert_outputs: (N, H), token_indices: (N,)
@triton.jit
def scatter_add_rows_atomic_cols_kernel(
    out_ptr,           # *bfloat16, pointer to out tensor
    expert_ptr,        # *bfloat16, pointer to expert_outputs tensor
    indices_ptr,       # *int64 (will cast to int32), pointer to token_indices
    N,                 # int32, number of rows (tokens)
    H: tl.constexpr,   # hidden size, compile-time constant for vectorization
):
    # One program per row
    pid = tl.program_id(0)
    if pid >= N:
        return

    # Load target row index for this expert output
    # indices_ptr is int64; Triton allows int64 arithmetic. We'll cast to int32 for pointer math.
    row_idx = tl.load(indices_ptr + pid)  # int64
    row_idx32 = row_idx.to(tl.int32)

    # Column offsets for vectorized operations
    offs = tl.arange(0, H)  # vector of column indices [0..H-1]

    # Compute source and destination pointers for this row
    # out_ptr and expert_ptr are bfloat16; indices use int32 arithmetic
    src_ptrs = expert_ptr + pid * H + offs
    dst_ptrs = out_ptr + row_idx32 * H + offs

    # Load values from expert_outputs and atomically add to out
    vals = tl.load(src_ptrs)  # bfloat16 vector
    tl.atomic_add(dst_ptrs, vals)  # vectorized atomic add across columns


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor) -> torch.Tensor:
        # Ensure device and contiguity
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "All tensors must be on CUDA for Triton kernel"
        # Clone to match original behavior (index_add_ on a cloned tensor)
        out = final_hidden_states.clone()

        # Shapes
        N = out.shape[0]  # number of rows (tokens) = batch_size * seq_len
        H = out.shape[1]  # hidden size

        # Ensure tensors are contiguous
        out = out.contiguous()
        expert_outputs = expert_outputs.contiguous()
        # Triton can handle int64 indices; casting to int32 is fine as long as N < 2^31, which is true in our use.
        token_indices = token_indices.contiguous()

        # Launch Triton kernel: 1D grid over rows
        # Modest number of warps; H is small to moderate, so 4 warps is fine.
        grid = (N,)
        scatter_add_rows_atomic_cols_kernel[grid](
            out, expert_outputs, token_indices,
            N, H,
            num_warps=4,
            num_stages=2,
        )
        return out
```
