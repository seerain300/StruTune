```python
import torch

# Triton imports
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: scatter-add expert_outputs[i, :] into output[token_indices[i], :]
# One program per index (i). Vectorize across hidden dimension in chunks of BLOCK_SIZE=256.
@triton.jit
def scatter_add_rows_kernel(
    output_ptr,         # *ptr to output [rows, hidden]
    expert_ptr,         # *ptr to expert_outputs [num_selected_tokens, hidden]
    indices_ptr,        # *ptr to token_indices [num_selected_tokens] (int32)
    rows,               # number of rows in output (i.e., batch_size * seq_len)
    hidden,             # hidden_size (number of columns)
    num_selected,       # number of expert outputs to add
    BLOCK_SIZE: tl.constexpr,  # hidden chunk size (e.g., 256)
):
    pid = tl.program_id(axis=0)
    # One program per selected index
    # If grid is larger than num_selected, guard with mask (not strictly needed here)
    in_bounds = pid < num_selected

    # Load token index for this program
    index = tl.load(indices_ptr + pid, mask=in_bounds, other=0)
    index = tl.max(index, 0)  # defensive clamp

    # Base offsets for the row in output and the row in expert
    # We assume row-major contiguous layout: offset = row * hidden + col
    # For atomic add, we use the base row pointer and then add col offsets.
    # Loop over hidden dimension in chunks
    start = 0
    while start < hidden:
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = in_bounds & (offs < hidden)
        # Load a chunk of expert outputs
        expert_vals = tl.load(expert_ptr + pid * hidden + offs, mask=mask, other=0.0)
        # Atomic add to the corresponding row in output
        tl.atomic_add(output_ptr + index * hidden + offs, expert_vals, mask=mask)
        start += BLOCK_SIZE


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        # Clone to avoid modifying input in-place for correctness
        output = final_hidden_states.clone()

        # Ensure dtypes are consistent
        if output.dtype != expert_outputs.dtype:
            # If expert_outputs is bfloat16 as in the provided get_inputs, we keep it.
            # No host-side arithmetic; we only scatter-add.
            pass

        # Ensure indices are int32 for Triton
        if token_indices.dtype != torch.int32:
            token_indices = token_indices.to(torch.int32)

        # If Triton not available, fallback to torch implementation
        if not TRITON_AVAILABLE:
            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
            return output

        # Launch kernel: one program per selected token
        num_selected = expert_outputs.shape[0]
        grid = (triton.cdiv(num_selected, 1),)  # axis=0 over selected tokens
        BLOCK_SIZE = 256  # Vectorize across hidden dim; 256 works well for bfloat16 and common sizes
        scatter_add_rows_kernel[grid](
            output, expert_outputs, token_indices,
            output.shape[0], output.shape[1], num_selected,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,  # reasonable default for simple vectorized loops
        )
        return output
```

- Keep the single-program-per-index design (simple, robust, and fast for these shapes).
- Improve vectorization over the hidden dimension using a 2D grid: split the hidden dimension into column tiles and parallelize both across selected tokens and across column tiles. This increases parallelism and can help for larger hidden_size and high num_selected_tokens.
- Continue to use atomic_add for correctness when indices repeat, but ensure the kernel minimizes per-chunk overhead and uses masks for tails.

Detailed plan
- Kernel design:
  - Use a 2D launch grid:
    - axis=0: each program handles one selected token i (grid = (num_selected_tokens,)).
    - axis=1: each program handles a tile of the hidden dimension of size BLOCK_SIZE (e.g., 256). The number of tiles is triton.cdiv(hidden_size, BLOCK_SIZE).
  - Inside the kernel, each program:
    - Loads the token index i and the corresponding row in output.
    - Loads a tile of expert_outputs[i, tile] and performs an atomic_add into output[index, tile].
    - Uses masks to handle the tail when hidden_size is not divisible by BLOCK_SIZE.
- Host code:
  - Minimal operations: clone final_hidden_states to output, cast token_indices to int32 if needed, and launch the Triton kernel with grid = (num_selected_tokens, num_hidden_tiles).
  - Keep a fallback to torch.index_add if Triton is not available.

Why this should help
- The 2D grid increases the number of concurrent programs, improving utilization across the GPU when hidden_size is large or num_selected_tokens is high. Even though atomics serialize writes per row, increasing the number of programs often improves throughput due to better occupancy and parallelism.
- The per-token loop across hidden tiles remains simple and correct; the atomic_add ensures accumulation correctness even with duplicate indices.

```python
try:

# 2D grid: axis=0 over selected tokens (i), axis=1 over hidden tiles.
@triton.jit
def scatter_add_rows_kernel_2d(
    output_ptr,         # *ptr to output [rows, hidden]
    expert_ptr,         # *ptr to expert_outputs [num_selected_tokens, hidden]
    indices_ptr,        # *ptr to token_indices [num_selected_tokens] (int32)
    rows,               # number of rows in output (i.e., batch_size * seq_len)
    hidden,             # hidden_size (number of columns)
    num_selected,       # number of expert outputs to add
    BLOCK_SIZE: tl.constexpr,  # tile size along hidden dimension
):
    pid_i = tl.program_id(axis=0)  # selected token index
    pid_tile = tl.program_id(axis=1)  # tile along hidden dimension

    # Guard for case where grid > num_selected (shouldn't happen if we set grid correctly)
    in_bounds_i = pid_i < num_selected

    # Load target row index for this selected token
    index = tl.load(indices_ptr + pid_i, mask=in_bounds_i, other=0)
    index = tl.max(index, 0)  # defensive clamp

    # Compute column offsets for this tile
    start = pid_tile * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = in_bounds_i & (offs < hidden)

    # Load a tile of expert outputs for this selected token
    vals = tl.load(expert_ptr + pid_i * hidden + offs, mask=mask, other=0.0)

    # Atomic add into the corresponding row in output
    tl.atomic_add(output_ptr + index * hidden + offs, vals, mask=mask)

        num_selected = expert_outputs.shape[0]
        hidden_size = output.shape[1]
        BLOCK_SIZE = 256  # tile size along hidden dimension; 256 works well for bfloat16
        num_tiles = triton.cdiv(hidden_size, BLOCK_SIZE)
        grid = (num_selected, num_tiles)

        scatter_add_rows_kernel_2d[grid](
            output, expert_outputs, token_indices,
            output.shape[0], hidden_size, num_selected,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,  # reasonable default; tuneable
        )
        return output
```
