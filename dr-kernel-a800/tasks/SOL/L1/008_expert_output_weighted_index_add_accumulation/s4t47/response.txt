import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_loop_kernel(
    output_ptr,        # *bf16, shape [B, H]
    expert_ptr,        # *bf16, shape [T, H]
    indices_ptr,       # *i64,  shape [T]
    B: tl.constexpr,   # batch_seq_len
    H: tl.constexpr,   # hidden_size
    T: tl.constexpr,   # number of expert outputs
):
    # One program per source row i
    i = tl.program_id(0)
    # Guard in case grid > T (not needed if grid=(T,), but safe)
    if i >= T:
        return

    # Load token index as int64, then cast to int32 for pointer arithmetic
    idx64 = tl.load(indices_ptr + i)
    idx = idx64.to(tl.int32)

    # Loop over hidden dimension and add one element at a time
    # This avoids atomics and minimizes any vectorization discrepancies.
    for h in range(0, H):
        # Load expert_outputs[i, h] as bf16
        v = tl.load(expert_ptr + i * H + h)  # v is bfloat16 (Triton infers from expert_ptr dtype)
        # Store v into output[idx, h]
        tl.store(output_ptr + idx * H + h, v)

    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):

            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
        We implement the same scatter-add along rows (dim=0).
        """

import triton
import triton.language as tl

def scatter_add_rows_cols_kernel(
    output_ptr,        # *bf16, shape [B, H]
    expert_ptr,        # *bf16, shape [T, H]
    indices_ptr,       # *i64 or *i32, shape [T]
    B: tl.int32,       # batch_seq_len (runtime int)
    H: tl.int32,       # hidden_size (runtime int)
    T: tl.int32,       # number of expert outputs (runtime int)
):

    i = tl.program_id(0)  # row index
    h = tl.program_id(1)  # column index

    # Store v into output[idx, h] (bf16)
    tl.store(output_ptr + idx * H + h, v)
