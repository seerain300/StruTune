import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_atomic_kernel(
    out_ptr,      # *bf16, shape (N, H)
    src_ptr,      # *bf16, shape (M, H)
    indices_ptr,  # *int32, shape (M,)
    N,            # int32
    M,            # int32
    H,            # int32
):
    # 1D grid: each program handles one row 'i' from (0..M-1)
    i = tl.program_id(0)
    if i >= M:
        return
    # Compute destination row index
    row_idx = tl.load(indices_ptr + i)  # int32
    # Loop over hidden dimension, atomic add each element
    for j in range(0, H):
        val = tl.load(src_ptr + i * H + j)
        tl.atomic_add(out_ptr + row_idx * H + j, val)


@triton.jit
def scatter_copy_noatomic_kernel(
    out_ptr,      # *bf16, shape (N, H)
    src_ptr,      # *bf16, shape (M, H)
    indices_ptr,  # *int32, shape (M,)
    M,            # int32
    H,            # int32
):
    # 1D grid: each program handles one row 'i' from (0..M-1)
    i = tl.program_id(0)
    if i >= M:
        return
    row_idx = tl.load(indices_ptr + i)  # int32
    # Copy src[i, :] into out[row_idx, :]
    for j in range(0, H):
        val = tl.load(src_ptr + i * H + j)
        tl.store(out_ptr + row_idx * H + j, val)


def _launch_atomic_scatter(out: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor, H: int):
    M = expert_outputs.shape[0]
    N = out.shape[0]
    # Ensure contiguity
    out = out.contiguous()
    expert_outputs = expert_outputs.contiguous()
    token_indices = token_indices.to(torch.int32).contiguous()
    # Launch 1D grid over M rows
    grid = (triton.cdiv(M, 1),)
    scatter_add_atomic_kernel[grid](out, expert_outputs, token_indices, N, M, H, num_warps=4, num_stages=1)

    grid = (triton.cdiv(M, 1),)
    scatter_copy_noatomic_kernel[grid](out, expert_outputs, token_indices, M, H, num_warps=4, num_stages=1)

        # Note: torch.unique_count may not be available in all environments; implement via equality.
