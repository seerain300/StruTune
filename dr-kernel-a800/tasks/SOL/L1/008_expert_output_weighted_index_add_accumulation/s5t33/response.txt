Current design and why it works
- One Triton program per selected index (grid = (num_selected_tokens,)).
- Inside each program, iterate over the hidden dimension in chunks of BLOCK_SIZE=256 (vectorized), with masks for the tail when hidden_size isn’t divisible by 256.
- Use atomic_add per chunk to accumulate contributions safely.
- Host code performs minimal operations: clone, cast token_indices to int32, and launch the kernel. No PyTorch tensor ops in host.

- Preserve host-side minimal logic.

Final code
```python
import torch

# Triton imports
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: scatter-add expert_outputs[i, :] into output[token_indices[i], :]
# One program per index (i). Vectorize across hidden dimension in chunks of BLOCK_SIZE=256.
@triton.jit
def scatter_add_rows_kernel(
    output_ptr,         # *ptr to output [rows, hidden]
    expert_ptr,         # *ptr to expert_outputs [num_selected_tokens, hidden]
    indices_ptr,        # *ptr to token_indices [num_selected_tokens] (int32)
    rows,               # number of rows in output (i.e., batch_size * seq_len)
    hidden,             # hidden_size (number of columns)
    num_selected,       # num_selected_tokens
    BLOCK_SIZE: tl.constexpr,  # chunk size across hidden dimension (256)
):
    pid = tl.program_id(axis=0)
    # Guard: only run if pid < num_selected
    if pid >= num_selected:
        return

    # Load the destination row index for this selected token
    idx = tl.load(indices_ptr + pid)
    # Guard against out-of-range indices (though token_indices are in [0, rows), keep it for safety)
    if (idx < 0) or (idx >= rows):
        return

    # Base offsets for this row in output and for this selected token in expert
    base_out = idx * hidden
    base_exp = pid * hidden

    # Vectorized iteration across hidden dimension in chunks of BLOCK_SIZE
    for col_start in range(0, hidden, BLOCK_SIZE):
        offs = col_start + tl.arange(0, BLOCK_SIZE)
        mask = offs < hidden

        # Load expert outputs for this chunk
        vals = tl.load(expert_ptr + base_exp + offs, mask=mask, other=0.0)

        # Atomic add into output
        tl.atomic_add(output_ptr + base_out + offs, vals, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(
        self,
        final_hidden_states: torch.Tensor,
        expert_outputs: torch.Tensor,
        token_indices: torch.Tensor,
    ) -> torch.Tensor:
        # Minimal host-side logic: clone and launch Triton kernel
        # Ensure dtype is bfloat16 and output is contiguous
        output = final_hidden_states.clone()

        # Triton requires int32 indices
        if token_indices.dtype != torch.int32:
            token_indices = token_indices.to(torch.int32)

        # Ensure tensors are on CUDA and contiguous
        if TRITON_AVAILABLE and output.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda:
            rows = output.shape[0]
            hidden = output.shape[1]
            num_selected = expert_outputs.shape[0]

            # Configure grid: one program per selected token
            grid = (num_selected,)

            # Launch kernel; use a stable configuration that has shown good performance
            scatter_add_rows_kernel[grid](
                output,
                expert_outputs,
                token_indices,
                rows,
                hidden,
                num_selected,
                BLOCK_SIZE=256,
                num_warps=4,
                num_stages=2,
            )

        return output
```
