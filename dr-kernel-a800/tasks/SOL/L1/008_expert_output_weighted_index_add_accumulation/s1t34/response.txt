```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,          # *const bfloat16, shape [M, H]
    src_ptr,          # *const bfloat16, shape [N, H]
    indices_ptr,      # *const int32,    shape [N]
    N: tl.constexpr,  # number of source rows
    H: tl.constexpr,  # hidden size
    BLOCK_H: tl.constexpr,  # tile size across H
):
    # One program per source row
    row = tl.program_id(0)
    if row >= N:
        return

    # Load token index (int32)
    idx = tl.load(indices_ptr + row)

    # Vector of hidden indices
    offs = tl.arange(0, BLOCK_H)
    mask = offs < H

    # Compute pointers for this row
    out_row_ptr = out_ptr + idx * H + offs
    src_row_ptr = src_ptr + row * H + offs

    # Load source values and accumulate into output
    val = tl.load(src_row_ptr, mask=mask, other=0.0)
    out_vals = tl.load(out_row_ptr, mask=mask, other=0.0)
    out_vals = out_vals + val
    tl.store(out_row_ptr, out_vals, mask=mask)


def _select_block_h(H: int) -> int:
    # Choose BLOCK_H to minimize loop iterations while keeping good occupancy.
    # For typical H <= 1024, set BLOCK_H = H to avoid looping over tiles.
    # Clamp to [128, 1024] to avoid pathological cases.
    if H >= 1024:
        return 1024
    if H <= 128:
        return 128
    # Round up to next multiple of 32 for better alignment (H is often power-of-two)
    # but prefer exact H when H <= 1024 to avoid looping.
    return H


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        # Ensure tensors are on the same device and dtypes
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Inputs must be on CUDA for Triton"
        assert final_hidden_states.dtype == torch.bfloat16 and expert_outputs.dtype == torch.bfloat16, "Expected bfloat16 tensors"
        assert token_indices.dtype == torch.int32, "token_indices must be int32"

        # Clone final_hidden_states to output
        output = final_hidden_states.clone()

        M = output.shape[0]
        N = expert_outputs.shape[0]
        H = output.shape[1]

        # Select BLOCK_H to minimize loop iterations (for H <= 1024, BLOCK_H = H)
        BLOCK_H = _select_block_h(H)
        grid = (N,)

        scatter_add_rows_kernel[grid](
            output,               # out_ptr
            expert_outputs,       # src_ptr
            token_indices,        # indices_ptr
            N=N,                  # constexpr
            H=H,                  # constexpr
            BLOCK_H=BLOCK_H,      # constexpr
            num_warps=4,
            num_stages=2,
        )
        return output
```

Notes:
- The kernel processes each source row independently, vectorizing across the hidden dimension with a BLOCK_H that is chosen to match H when H ≤ 1024, removing the inner loop over tiles. This reduces control overhead and should improve performance.
- We keep one program per source row (grid = N). This approach previously performed well and maintains parallelism across N. Using a 2D grid over (N, tiles) would require per-tile atomics or a reduction pattern, which can hurt performance and increase complexity without clear gains here.
- Inputs are expected to be on CUDA, with bfloat16 dtype for tensors and int32 for token_indices. The forward clones the final_hidden_states to create the output buffer, then performs the scatter-add in-place on the output.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,          # *const bfloat16, shape [M, H]
    src_ptr,          # *const bfloat16, shape [N, H]
    indices_ptr,      # *const int32,    shape [N]
    N: tl.constexpr,  # number of source rows
    H: tl.constexpr,  # hidden size
    BLOCK_H: tl.constexpr,  # tile size across H
):
    # One program per source row
    row = tl.program_id(0)
    if row >= N:
        return

    # Load token index (int32)
    idx = tl.load(indices_ptr + row)

    # Vector of hidden offsets for this tile
    offs = tl.arange(0, BLOCK_H)
    mask = offs < H

    # Compute pointers for this row
    out_row_ptr = out_ptr + idx * H + offs
    src_row_ptr = src_ptr + row * H + offs

    # Load source values and atomically accumulate into output
    val = tl.load(src_row_ptr, mask=mask, other=0.0)
    # out_vals = tl.load(out_row_ptr, mask=mask, other=0.0)  # not needed for atomic_add with addend
    tl.atomic_add(out_row_ptr, val, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        # Triton requires CUDA tensors
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Inputs must be on CUDA for Triton"
        # Ensure dtypes and contiguity
        assert final_hidden_states.dtype == torch.bfloat16 and expert_outputs.dtype == torch.bfloat16, "Expected bfloat16 tensors"
        assert token_indices.dtype == torch.int32, "token_indices must be int32"
        final_hidden_states = final_hidden_states.contiguous()
        expert_outputs = expert_outputs.contiguous()
        token_indices = token_indices.contiguous()

        # Clone final_hidden_states to output
        output = final_hidden_states.clone()

        M = output.shape[0]
        N = expert_outputs.shape[0]
        H = output.shape[1]

        # Choose a robust tile size: 256 balances occupancy and vector width
        BLOCK_H = 256
        grid = (N,)

        scatter_add_rows_kernel[grid](
            output,               # out_ptr
            expert_outputs,       # src_ptr
            token_indices,        # indices_ptr
            N=N,                  # constexpr
            H=H,                  # constexpr
            BLOCK_H=BLOCK_H,      # constexpr
            num_warps=4,
            num_stages=2,
        )
        return output
```
