import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_fp32_kernel(
    out_ptr,            # *fp32, shape [M, H], M = batch_seq_len
    expert_ptr,         # *fp32, shape [N, H], N = num_selected_tokens
    indices_ptr,        # *int32, shape [N]
    M: tl.constexpr,    # number of rows in out (not used for scatter, kept for clarity)
    N: tl.constexpr,    # number of updates
    H: tl.constexpr,    # hidden size (number of columns)
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # Guard against overlaunch (in case grid > N)
    if pid >= N:
        return

    # Destination row index for this update
    idx = tl.load(indices_ptr + pid)  # scalar int32

    # Process the row in BLOCK_H chunks
    # We use a simple loop over offsets, masked by H
    for offs in range(0, H, BLOCK_H):
        col = offs + tl.arange(0, BLOCK_H)
        mask = col < H
        # Load the vector of values from expert_ptr[pid, :]
        v = tl.load(expert_ptr + pid * H + col, mask=mask, other=0.0)
        # Atomic add into out_ptr[idx, :]
        tl.atomic_add(out_ptr + idx * H + col, v, mask=mask)

import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_2d_fp32_kernel(
    out_ptr,         # *fp32, shape [M, H]
    expert_ptr,      # *fp32, shape [N, H]
    indices_ptr,     # *int32, shape [N]
    M: tl.constexpr, # number of rows in output
    N: tl.constexpr, # number of updates
    H: tl.constexpr, # hidden size
    BLOCK_H: tl.constexpr,
):

    pid_m = tl.program_id(0)  # row index
    pid_h = tl.program_id(1)  # block id along hidden dimension

    if pid_m >= M:

    mask = offs < H

    # Loop over all updates and accumulate into acc
    # Duplicate token_indices will be accumulated multiple times, which is correct
    for i in range(0, N):

        # Load expert_outputs[i, offs] as a vector

    idx = tl.load(indices_ptr + pid)  # int32

- If batch_seq_len M is extremely large, consider tiling across rows as well; however, the current approach keeps one program per update and writes directly, which is straightforward and efficient for most configurations.
