import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_atomic_kernel(
    out_ptr,            # *bf16, shape (N, H)
    src_ptr,            # *bf16, shape (M, H)
    indices_ptr,        # *int32, shape (M,)
    N,                  # int32, number of rows in out (batch_seq_len)
    H,                  # int32, hidden size
    M,                  # int32, number of expert outputs
    BLOCK_SIZE: tl.constexpr,  # rows per program
):
    # Each program handles BLOCK_SIZE rows
    pid = tl.program_id(0)
    row_start = pid * BLOCK_SIZE
    # Iterate rows this program is responsible for
    # This loop keeps correctness across all N; no reliance on grid * BLOCK_SIZE
    for i in range(BLOCK_SIZE):
        row = row_start + i
        # If row >= N, skip (safest approach; although grid is chosen so rows fit)
        # In practice, choose grid such that row < N, but this guard is cheap and safe.
        if row >= N:
            break
        # Load the destination index for this row
        idx = tl.load(indices_ptr + row)
        # Ensure idx is in range [0, N). The provided inputs guarantee this.
        # Loop over hidden dimension and atomically add
        for j in range(0, H):
            src_offset = row * H + j
            dst_offset = idx * H + j
            val = tl.load(src_ptr + src_offset)
            # Atomic add in-place
            tl.atomic_add(out_ptr + dst_offset, val)
