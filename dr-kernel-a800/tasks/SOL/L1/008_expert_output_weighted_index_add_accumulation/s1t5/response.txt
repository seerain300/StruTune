import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_expert_outputs_kernel(
    output_ptr,         # *bf16, shape (M, H)
    source_ptr,         # *bf16, shape (N, H)
    index_ptr,          # *int32, shape (N,)
    M: tl.constexpr,    # int: number of rows in output (batch_seq_len)
    N: tl.constexpr,    # int: number of source rows (num_selected_tokens)
    H: tl.constexpr,    # int: hidden_size
    BLOCK_H: tl.constexpr  # tile size along hidden dimension
):
    pid = tl.program_id(axis=0)  # program id: which source row to process
    if pid >= N:
        return

    # Load the destination row index for this source row
    dest = tl.load(index_ptr + pid)  # int32
    if dest < 0 or dest >= M:
        # Defensive: ignore out-of-range indices (though inputs should be valid)
        return

    # Vectorized loop over hidden dimension in tiles
    for offs in range(0, H, BLOCK_H):
        h = offs + tl.arange(0, BLOCK_H)          # [BLOCK_H] vector of column offsets
        mask = h < H                              # mask for valid columns

        # Compute source and output offsets
        src_offs = pid * H + h                    # vector of source offsets
        out_offs = dest * H + h                  # vector of output offsets

        # Load the expert output slice (bf16)
        src_vals = tl.load(source_ptr + src_offs, mask=mask, other=0.0)

        # Atomic add into the output
        tl.atomic_add(output_ptr + out_offs, src_vals, mask=mask)

            "All tensors must be on CUDA device for Triton kernels."

        # Ensure contiguity and dtypes
        final_hidden_states = final_hidden_states.contiguous()

        token_indices = token_indices.contiguous()

        # We start from a fresh tensor (equivalent to cloning original provided final_hidden_states),
        # but since the original function clones, we should mirror that to be safe.

        H = final_hidden_states.shape[1]  # hidden_size
