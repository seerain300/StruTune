- Advantages:

import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_fp32_kernel(
    out_ptr,          # *fp32, shape [M, H] output buffer (already initialized)
    expert_ptr,       # *fp32, shape [N, H] expert outputs
    idx_ptr,          # *int32, shape [N] token indices
    M,                # int32, number of rows (batch_seq_len)
    N,                # int32, number of updates (num_selected_tokens)
    H,                # int32, hidden size
    BLOCK_SIZE: tl.constexpr,  # tile size along hidden dimension
):
    # program ids
    row_id = tl.program_id(0)    # which output row we're handling
    tile_id = tl.program_id(1)   # which hidden-dimension tile we're handling

    # compute offsets for this tile along hidden dimension
    start = tile_id * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask_vec = offsets < H  # vectorized mask for bounds

    # base pointer for this row and tile
    row_base_out = row_id * H + offsets
    # initialize local vector to zero for the chunk we're handling
    acc = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    # loop over all updates and accumulate into acc
    # This ensures duplicates are summed correctly.
    i = 0
    while i < N:
        idx = tl.load(idx_ptr + i)  # int32 scalar
        # guard for safety (though idx is guaranteed in [0, M))
        valid_i = (idx >= 0) & (idx < M)
        if valid_i:
            # load the i-th expert output vector chunk
            exp_base = i * H + offsets
            v = tl.load(expert_ptr + exp_base, mask=mask_vec, other=0.0)
            acc += v
        i += 1

    # atomically add the accumulated chunk into the output row
    tl.atomic_add(out_ptr + row_base_out, acc, mask=mask_vec)

    if x <= 64:
        return 64
    elif x <= 128:
        return 128
    elif x <= 256:
        return 256
    elif x <= 512:
        return 512
    else:
        return 1024  # cap at 1024 for reasonable tile size

                expert_outputs: torch.Tensor, token_indices: torch.Tensor):

            "All inputs must be on CUDA for Triton execution."
        final_hidden_states = final_hidden_states.contiguous()
        expert_outputs = expert_outputs.contiguous()
        token_indices = token_indices.contiguous()

        M = final_hidden_states.shape[0]  # batch_seq_len
        H = final_hidden_states.shape[1]
        N = expert_outputs.shape[0]

            out_fp32, expert_fp32, idx_i32,
            M, N, H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,  # reasonable default; can tune per H
        )

        # Cast back to bfloat16 to match original behavior
        result = out_fp32.to(torch.bfloat16)
        return result
