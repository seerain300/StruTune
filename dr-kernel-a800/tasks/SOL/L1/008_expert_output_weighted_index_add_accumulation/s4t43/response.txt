```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_deterministic(
    output_ptr,       # *bf16, shape [B, H]
    expert_ptr,       # *bf16, shape [T, H]
    indices_ptr,      # *int64, shape [T]
    B: tl.constexpr,  # batch_seq_len (number of rows in output)
    H: tl.constexpr,  # hidden_size (number of columns)
    T: tl.constexpr,  # number of expert outputs to scatter
    BLOCK_H: tl.constexpr,  # number of columns processed per iteration
):
    # One program per source row i
    i = tl.program_id(0)

    # If i >= T, do nothing (safety in case grid > T)
    if i >= T:
        return

    # Load token index (int64), cast to int32 for pointer arithmetic
    idx64 = tl.load(indices_ptr + i)
    idx = idx64.to(tl.int32)

    # Guard against any out-of-range index (shouldn't happen, but safe)
    if idx < 0 or idx >= B:
        return

    # Loop over hidden columns in chunks of BLOCK_H
    offs = tl.arange(0, BLOCK_H)
    # Start column
    h = 0
    while h < H:
        col = h + offs
        mask = col < H
        # Compute linear offsets
        src_offsets = i * H + col
        dst_offsets = idx * H + col

        # Load source row vector (bf16) and store into destination row
        # Masked loads/stores ensure we don't read/write beyond H
        v = tl.load(expert_ptr + src_offsets, mask=mask, other=0.0)
        # Store into output; masked to handle tail
        tl.store(output_ptr + dst_offsets, v, mask=mask)

        h += BLOCK_H


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-optimized version of:
            output = final_hidden_states.clone()
            output.index_add_(dim=0, index=token_indices, source=expert_outputs)
        We implement the index_add via a Triton kernel that deterministically adds each source row
        to the output at the corresponding token index, handling duplicates via sequential writes.
        """
        # Ensure tensors are on CUDA
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "All tensors must be on CUDA for Triton."

        # Clone to match original behavior
        output = final_hidden_states.clone()

        # Make tensors contiguous for performance and predictable strides
        output = output.contiguous()
        expert_outputs = expert_outputs.contiguous()
        token_indices = token_indices.contiguous()

        # Shapes
        B = output.shape[0]  # batch_seq_len (number of rows)
        H = output.shape[1]  # hidden_size (number of columns)
        T = token_indices.shape[0]  # number of expert outputs to scatter

        # Choose a reasonable BLOCK_H; 128 works well for typical hidden sizes and keeps register use manageable
        BLOCK_H = 128

        # Launch one program per source row
        grid = (T,)

        # Run Triton kernel
        scatter_add_rows_deterministic[grid](
            output, expert_outputs, token_indices,
            B=B, H=H, T=T, BLOCK_H=BLOCK_H,
            num_warps=1,  # keep lightweight; correctness is priority
            num_stages=1,
        )

        return output
```
