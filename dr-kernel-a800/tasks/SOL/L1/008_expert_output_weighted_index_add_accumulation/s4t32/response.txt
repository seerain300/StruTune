import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_bf16_kernel(
    output_ptr,      # *bfloat16
    expert_ptr,      # *bfloat16
    indices_ptr,     # *int64
    B, H, T,         # int32 scalars
):
    # One program per source row i
    i = tl.program_id(0)
    if i >= T:
        return

    # Load token index idx (int64)
    idx64 = tl.load(indices_ptr + i)
    # Use as int32 for pointer arithmetic (safe because B fits in int32)
    idx = idx64.to(tl.int32)

    # Iterate over hidden dimension H, one column at a time
    # Keep operations in bfloat16 to avoid any fp32 promotion
    for h in range(0, H):
        # Load value from expert_outputs[i, h] (row-major contiguous)
        v = tl.load(expert_ptr + i * H + h)  # bfloat16
        # Load current output value
        old = tl.load(output_ptr + idx * H + h)  # bfloat16
        # Accumulate: output[idx, h] += v
        new = old + v
        # Store the result back
        tl.store(output_ptr + idx * H + h, new)
