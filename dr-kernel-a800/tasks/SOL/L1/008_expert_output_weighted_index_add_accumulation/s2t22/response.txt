import triton.language as tl

def _scatter_add_rows_kernel(
    out_ptr,            # *bf16, (M, H)
    in_ptr,             # *bf16, (N, H)
    idx_ptr,            # *int32, (N,)
    M,                  # int: number of rows in output (i.e., batch_size * seq_len)
    N,                  # int: number of rows in in_ptr/idx_ptr (num_selected_tokens)
    H,                  # int: hidden size

    stride_out_col: tl.constexpr,  # int: col stride of out (usually 1)
    stride_in_row: tl.constexpr,   # int: row stride of in (usually 1)
    stride_in_col: tl.constexpr,   # int: col stride of in (usually 1)
    BLOCK: tl.constexpr,           # int: chunk size across hidden dimension
):
    # One program per row
    pid = tl.program_id(0)
    # Guard: if grid is larger than N (rare), early return
    if pid >= N:
        return

    # Load destination row index for this row
    idx = tl.load(idx_ptr + pid)

        return

    # Compute base pointers for this row

    in_row_base = in_ptr + pid * stride_in_row

    # Vectorized chunked accumulation across hidden dimension
    col_offsets = tl.arange(0, BLOCK)
    # Loop over chunks of size BLOCK
    # We use a python range with dynamic H to avoid large compile-time loops
    # Triton will unroll or handle this loop at runtime.
    for start in range(0, H, BLOCK):

        mask = cols < H

        # Atomic add to handle duplicates in idx
        tl.atomic_add(out_ptrs, vals, mask=mask)
