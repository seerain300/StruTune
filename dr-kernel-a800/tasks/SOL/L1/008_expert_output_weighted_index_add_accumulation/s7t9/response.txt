    def scatter_add_rows_atomic_64_kernel(
        out_ptr,            # *bf16, shape [N, H], contiguous
        src_ptr,            # *bf16, shape [M, H], contiguous
        idx_ptr,            # *int64, shape [M], contiguous

        H,                  # int32: hidden_size
        M                   # int32: number of expert outputs (num_selected_tokens)

        # Load destination row index as int64 for 64-bit addressing
        row_idx = tl.load(idx_ptr + i)  # int64

        # Compute base offsets in int64
        row_base = row_idx * H  # int64
        i_base = i * H          # int64

        # Loop over hidden_size elements, accumulate with atomic add
        # We use a scalar loop to avoid Triton vectorization pitfalls in this environment.
        for j in range(0, 32768):  # upper bound; will break when j >= H
            if j >= H:
                break
            dst_offset = row_base + j
            src_offset = i_base + j
            # Load element from src row i, column j
            val = tl.load(src_ptr + src_offset)  # bf16
            # Atomic add into out[row_idx, j]

    Requirements:

    """
