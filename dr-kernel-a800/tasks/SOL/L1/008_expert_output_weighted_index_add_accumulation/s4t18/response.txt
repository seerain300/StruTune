import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_dim0_per_element_kernel(
    output_ptr,         # *bf16, shape [B, H]
    expert_ptr,         # *bf16, shape [T, H]
    indices_ptr,        # *int64, shape [T]
    B: tl.constexpr,    # number of rows in output (batch_seq_len)
    H: tl.constexpr,    # number of columns (hidden_size)
    T: tl.constexpr,    # number of expert outputs
):
    # 2D grid: one program per (row i, column h)
    row = tl.program_id(0)
    col = tl.program_id(1)

    # Bounds checks (defensive)
    if row >= T:
        return

    # Load the token index for this row (int64), then cast to int32 for pointer arithmetic
    idx64 = tl.load(indices_ptr + row)
    idx = idx64.to(tl.int32)

    # Load the value from expert_outputs[row, col] (bf16)
    v = tl.load(expert_ptr + row * H + col)

    # Compute destination pointer for output[idx, col]
    out_ptr = output_ptr + idx * H + col

    # Atomic add the bfloat16 value; since we launch one program per (row, col),
    # only one program will write to each output element, so atomic_add behaves like a plain store.
    tl.atomic_add(out_ptr, v)
