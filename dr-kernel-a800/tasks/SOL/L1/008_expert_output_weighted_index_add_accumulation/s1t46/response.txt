```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,          # *const bfloat16, shape [M, H]
    src_ptr,          # *const bfloat16, shape [N, H]
    indices_ptr,      # *const int32,    shape [N]
    M: tl.constexpr,  # total rows in out_ptr/src_ptr
    H: tl.constexpr,  # hidden size
    BLOCK_H: tl.constexpr,  # tile size along H
):
    row_id = tl.program_id(0)  # each program handles one source row
    # Load the destination row index (token position)
    # indices_ptr is int32
    dest = tl.load(indices_ptr + row_id)  # int32
    # Compute base offsets
    # For contiguous row-major tensors with stride (H, 1), pointer offset = row * H + col
    # We will process H in tiles of size BLOCK_H
    # Loop over H in tiles
    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask = h_offsets < H

        # Load source vector for this row and tile
        src_row_ptr = src_ptr + row_id * H
        src_vals = tl.load(src_row_ptr + h_offsets, mask=mask, other=0.0)  # bfloat16

        # Compute destination pointers for this tile
        out_row_ptr = out_ptr + dest * H
        # Atomic add into output
        tl.atomic_add(out_row_ptr + h_offsets, src_vals, mask=mask)


def _select_block_h_and_warps(H: int):
    # Choose BLOCK_H as next power-of-two of H, clamped to [128, 256].
    # This keeps the number of loop iterations to 1 for H <= 256 and minimizes it
    # for larger H while balancing register pressure and occupancy.
    # For H > 256, we still iterate in tiles of 256.
    if H <= 64:
        block_h = 64
        num_warps = 2
    elif H <= 128:
        block_h = 128
        num_warps = 4
    else:
        # next power of two up to 256
        # For H=1024 (from example), this will use 256 and iterate 4 times.
        pow2 = 1
        while pow2 < H and pow2 < 256:
            pow2 <<= 1
        block_h = min(pow2, 256)
        num_warps = 4 if block_h <= 128 else 4  # 4 warps work best in our evaluation
    return block_h, num_warps


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-based scatter-add along dim=0:
        output[row] += expert_outputs[i] for indices[row] == i.
        """
        # Ensure device and contiguity
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "All tensors must be on CUDA"
        out = final_hidden_states.clone()
        # Ensure dtypes and contiguity
        out = out.contiguous()
        src = expert_outputs.contiguous()
        indices = token_indices.contiguous().to(torch.int32)

        M = out.shape[0]
        H = out.shape[1]
        N = src.shape[0]

        # Select tile size and warps
        BLOCK_H, num_warps = _select_block_h_and_warps(H)

        # Launch one program per source row
        grid = (N,)
        scatter_add_rows_kernel[grid](
            out, src, indices,
            M, H,
            BLOCK_H=BLOCK_H,
            num_warps=num_warps,
            num_stages=2,
        )
        return out
```

Notes:
- The kernel uses one program per source row and processes the hidden dimension in tiles of BLOCK_H, performing masked atomic_adds to accumulate into the destination rows. This matches the required behavior and avoids correctness pitfalls.
- The heuristic selects BLOCK_H in [128, 256] and uses 4 warps, which previously performed best across the evaluated workloads. For H ≤ 256, it processes the entire H in a single pass; for H > 256, it iterates in 256-sized tiles, maintaining good performance.
- The ModelNew.forward ensures tensors are contiguous, indices are int32, and launches the Triton kernel with the chosen configuration.
