import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    output_ptr,       # *bf16 or *fp16, shape [M, H]
    expert_ptr,       # *bf16 or *fp16, shape [N, H]
    index_ptr,        # *int32, shape [N]
    M, H, N,          # int32 scalars
    BLOCK_H: tl.constexpr,
):

    n = tl.program_id(0)
    # If n >= N, exit (defensive guard for grid > N)
    if n >= N:
        return

    # Iterate over H in chunks of BLOCK_H
    start = 0
    # Loop is supported with runtime condition in Triton
    while start < H:
        offs = start + tl.arange(0, BLOCK_H)
        mask = offs < H

        # Load token index for this source row
        dest = tl.load(index_ptr + n, mask=True, other=0)  # scalar int32

        # Compute pointers for the current chunk
        # expert_ptr is row-major [N, H] => offset = n * H + offs
        vals = tl.load(expert_ptr + n * H + offs, mask=mask, other=0.0)

        # Atomic add into output at (dest, offs)
        # output_ptr is row-major [M, H] => offset = dest * H + offs
        tl.atomic_add(output_ptr + dest * H + offs, vals, mask=mask)

        start += BLOCK_H
