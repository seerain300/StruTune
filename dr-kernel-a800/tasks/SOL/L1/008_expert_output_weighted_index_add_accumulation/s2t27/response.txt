```python
import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_rows_kernel(
    out_ptr,         # *bf16, pointer to (M, H) output tensor
    in_ptr,          # *bf16, pointer to (N, H) expert_outputs tensor
    idx_ptr,         # *int32, pointer to (N,) token_indices
    M,               # int, number of rows in output (i.e., batch_size * seq_len)
    N,               # int, number of selected tokens (i.e., N = M * num_experts_per_tok)
    H,               # int, hidden size (columns)
    BLOCK: tl.constexpr,  # chunk size along hidden dimension
):
    # Each program handles one selected token (row i)
    pid = tl.program_id(0)
    # Guard: if grid > N (defensive, though we launch exactly N programs)
    if pid >= N:
        return

    # Load source row i
    in_row_ptr = in_ptr + pid * H
    # Create column offsets for a chunk
    col_offsets = tl.arange(0, BLOCK)
    # Iterate over hidden dimension in chunks of BLOCK
    num_chunks = (H + BLOCK - 1) // BLOCK
    # We need to compute start indices for each chunk; do this with a simple loop
    # Triton allows Python-side loops with constexpr; this is fine for typical H.
    for chunk in range(0, num_chunks):
        start = chunk * BLOCK
        cols = start + col_offsets
        mask = cols < H
        # Load source values (bf16)
        vals = tl.load(in_row_ptr + cols, mask=mask, other=0.0)  # shape (BLOCK,), dtype bf16

        # Read destination row index
        dest_row = tl.load(idx_ptr + pid)  # int32
        dest_row = dest_row.to(tl.int32)

        # Compute output row base pointer
        out_row_ptr = out_ptr + dest_row * H + start + col_offsets

        # Atomic add the vector chunk into the output
        tl.atomic_add(out_row_ptr, vals, mask=mask)
```
