import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    output_ptr,            # *const float (final_hidden_states clone), shape (M, H), bf16
    expert_ptr,            # *const float, shape (N, H), bf16
    indices_ptr,           # *const int64, shape (N,), int64
    N: tl.constexpr,       # number of updates
    H: tl.constexpr,       # hidden size (columns)
    BLOCK_H: tl.constexpr, # number of hidden features processed per iteration
):
    # Each program handles one source row (n in [0, N))
    n = tl.program_id(axis=0)

    # If grid > N, mask out-of-range
    if n >= N:
        return

    # Load target index for this update
    # indices are int64; Triton will handle cast to address offsets when used in pointer arithmetic
    idx = tl.load(indices_ptr + n)  # int64

    # Iterate over hidden dimension in BLOCK_H chunks
    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask = h_offsets < H

        # Load expert row block
        vals = tl.load(
            expert_ptr + n * H + h_offsets,
            mask=mask,
            other=0.0,  # safe default for masked elements
        )

        # Compute output destination addresses
        dest = idx * H + h_offsets

        # Atomically add the vals into output
        tl.atomic_add(output_ptr + dest, vals, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-accelerated scatter-add along dim=0:
          output[token_indices[i]] += expert_outputs[i]
        where output is a clone of final_hidden_states.
        """
        # Ensure tensors are on CUDA and contiguous
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Inputs must be on CUDA for Triton."
        device = final_hidden_states.device
        # Clone to match original semantics
        output = final_hidden_states.clone()

        # Shapes
        M = output.shape[0]
        H = output.shape[1]
        N = expert_outputs.shape[0]
        assert expert_outputs.shape[1] == H, "expert_outputs hidden size must match output's hidden size"
        assert token_indices.numel() == N, "token_indices length must equal N"

        # Tuning parameters
        # Choose BLOCK_H based on H to balance vectorization and register pressure
        if H >= 256:
            BLOCK_H = 128
            num_warps = 4
        elif H >= 128:
            BLOCK_H = 128
            num_warps = 4
        elif H >= 64:
            BLOCK_H = 64
            num_warps = 4
        else:
            BLOCK_H = 32
            num_warps = 2

        # 1D grid over N rows
        grid = (N,)

        # Launch Triton kernel
        scatter_add_rows_kernel[grid](
            output, expert_outputs, token_indices,
            N, H,
            BLOCK_H=BLOCK_H,
            num_warps=num_warps,
            num_stages=2,
        )

        return output

- torch.index_add 内部使用原子操作来处理重复索引，并且对整行添加专家输出。

现有 Triton 实现的性能瓶颈
- 之前的正确版本使用 1D grid（每个 program 处理一行），并在 H 维度上循环进行原子添加：

- 提高并行度并减少控制开销：

import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_2d_kernel(
    output_ptr,            # *bf16, shape (M, H)
    expert_ptr,            # *bf16, shape (N, H)
    indices_ptr,           # *int64, shape (N,)
    N: tl.constexpr,       # number of updates
    H: tl.constexpr,       # hidden size
    BLOCK_N: tl.constexpr, # number of rows processed per program (tile along N)
    BLOCK_H: tl.constexpr, # number of hidden features processed per iteration (tile along H)
):
    # 2D grid: programs tile over N (rows) and H (features)
    pid_n = tl.program_id(axis=0)
    pid_h = tl.program_id(axis=1)

    # Compute row offsets this program handles
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    n_mask = n_offsets < N

    # Load token indices for these rows
    idxs = tl.load(indices_ptr + n_offsets, mask=n_mask, other=0).to(tl.int64)

    # Compute hidden offsets this program handles
    h_start = pid_h * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    h_mask = h_offsets < H

    # Build 2D mask for valid loads/stores
    mask_2d = n_mask[:, None] & h_mask[None, :]

    # Load expert outputs for this (BLOCK_N x BLOCK_H) tile
    # Addresses: expert_ptr + n_offsets[:, None] * H + h_offsets[None, :]
    expert_addrs = n_offsets[:, None] * H + h_offsets[None, :]
    vals = tl.load(expert_ptr + expert_addrs, mask=mask_2d, other=0.0)

    # Compute destination addresses in output: idxs[:, None] * H + h_offsets[None, :]
    dest_addrs = idxs[:, None] * H + h_offsets[None, :]

    # Atomically add vals into output at dest_addrs
    tl.atomic_add(output_ptr + dest_addrs, vals, mask=mask_2d)

        # Shapes
        M = output.shape[0]
        H = output.shape[1]
        N = expert_outputs.shape[0]
        assert expert_outputs.shape[1] == H, "expert_outputs hidden size must match output's hidden size"

import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_2d_kernel(
    output_ptr,            # *bf16, shape (M, H)
    expert_ptr,            # *bf16, shape (N, H)
    indices_ptr,           # *int64, shape (N,)
    N: tl.constexpr,       # number of updates
    H: tl.constexpr,       # hidden size
    BLOCK_N: tl.constexpr, # rows per program
    BLOCK_H: tl.constexpr, # hidden features processed per iteration (can loop to cover all H)
):
    # 2D grid: axis 0 over rows (N tiles), axis 1 over H tiles
    pid_n = tl.program_id(axis=0)
    pid_h = tl.program_id(axis=1)

    # Rows handled by this program
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    n_mask = n_offsets < N

    # Load target indices (int64)
    idxs = tl.load(indices_ptr + n_offsets, mask=n_mask, other=0).to(tl.int64)

    # Loop over H in BLOCK_H chunks to perform atomic adds
    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        h_mask = h_offsets < H

        # 2D mask for valid loads/stores

        # Load expert outputs for this (BLOCK_N x BLOCK_H) tile
        expert_addrs = n_offsets[:, None] * H + h_offsets[None, :]
        vals = tl.load(expert_ptr + expert_addrs, mask=mask_2d, other=0.0)

        # Destination addresses in output: idxs[:, None] * H + h_offsets[None, :]
        dest_addrs = idxs[:, None] * H + h_offsets[None, :]
