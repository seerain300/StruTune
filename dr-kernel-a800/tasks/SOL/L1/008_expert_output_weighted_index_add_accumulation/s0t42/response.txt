```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_row_kernel(
    output_ptr,          # *bf16 or *fp16, pointer to output [M, H]
    expert_ptr,          # *bf16 or *fp16, pointer to expert outputs [N, H]
    index_ptr,           # *int32, pointer to token indices [N]
    M,                   # int32, number of rows in output (M = batch_size * seq_len)
    H,                   # int32, number of columns in output
    BLOCK_H: tl.constexpr,  # tile size along H
):
    # Each program handles one source row i
    pid = tl.program_id(axis=0)
    # Load token index for this source row
    token = tl.load(index_ptr + pid)  # int32
    # Vector of column offsets for this tile
    h_offsets = tl.arange(0, BLOCK_H)

    start = 0
    while start < H:
        offs = start + h_offsets
        mask = offs < H
        # Load expert_outputs[i, offs]
        expert_row_ptr = expert_ptr + pid * H + offs
        val = tl.load(expert_row_ptr, mask=mask, other=0.0)
        # Compute destination address: output[token, offs]
        # Token may be >= M (in general), but M here is batch_size * seq_len and token indices are in [0, M).
        # For safety, cast to 64-bit for pointer arithmetic.
        dest_row = token.to(tl.int64)
        dest_base = output_ptr + dest_row * H
        dest_ptr = dest_base + offs.to(tl.int64)
        # Atomic add into output
        tl.atomic_add(dest_ptr, val, mask=mask)
        start += BLOCK_H

    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        final_hidden_states: (M, H), accumulation buffer to be updated
        expert_outputs: (N, H), weighted expert outputs to scatter-add
        token_indices: (N,), indices into final_hidden_states' rows
        """

        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Inputs must be on CUDA"
        output = final_hidden_states  # we accumulate in-place
        # Ensure contiguity
        output = output.contiguous()
        expert_outputs = expert_outputs.contiguous()
        token_indices = token_indices.contiguous()
        # Triton expects int32 indices for simplicity; convert if needed

        H = output.shape[1]
        N = expert_outputs.shape[0]

        # Grid: one program per source row
        grid = (N,)

        # Choose BLOCK_H adaptively
        if H >= 1024:
            BLOCK_H = 256
            num_warps = 8
            num_stages = 2
        elif H >= 256:
            BLOCK_H = 128
            num_warps = 4
            num_stages = 2
        else:
            BLOCK_H = 64
            num_warps = 2
            num_stages = 2

            output,
            expert_outputs,
            token_indices,
            M,
            H,
            BLOCK_H=BLOCK_H,
            num_warps=num_warps,
            num_stages=num_stages,
        )
        return output
```

```python
import triton
import triton.language as tl

def scatter_add_grid_kernel(
    output_ptr,          # *bf16 or *fp16, pointer to output [M, H]
    expert_ptr,          # *bf16 or *fp16, pointer to expert outputs [N, H]
    index_ptr,           # *int32, pointer to token indices [N]
    M,                   # int32, number of rows in output (M = batch_size * seq_len)
    H,                   # int32, number of columns in output
    N,                   # int32, number of source rows
    BLOCK_H: tl.constexpr,  # tile size along H
):
    # 2D launch: axis 0 = source row i, axis 1 = tile id along H
    row_id = tl.program_id(axis=0)
    tile_id = tl.program_id(axis=1)

    # Load token index for this source row (int32)
    token = tl.load(index_ptr + row_id)

    # Compute column offsets for this tile
    h_start = tile_id * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)  # int32 vector

    # Mask for valid columns
    mask = h_offsets < H

    # Load expert outputs for this row and tile
    expert_row_ptr = expert_ptr + row_id * H + h_offsets  # int64 arithmetic will be handled below
    val = tl.load(expert_row_ptr, mask=mask, other=0.0)

    # Compute destination pointer: output[token, h_offsets]
    # Cast to int64 for safe pointer arithmetic
    dest_row_i64 = token.to(tl.int64)
    dest_base = output_ptr + dest_row_i64 * H
    dest_ptr = dest_base + h_offsets.to(tl.int64)

    # Atomic add into output
    tl.atomic_add(dest_ptr, val, mask=mask)

        BLOCK_H = 128 if H >= 128 else 64
        num_warps = 4
        num_stages = 2

        grid = (N, triton.cdiv(H, BLOCK_H))

        scatter_add_grid_kernel[grid](
            output,
            expert_outputs,
            token_indices,
            M,
            H,
            N,
            BLOCK_H=BLOCK_H,
            num_warps=num_warps,
            num_stages=num_stages,
        )
```
