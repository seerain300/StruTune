    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:

        out_ptr,          # *bf16, shape (n_rows, n_cols)
        idx_ptr,          # *i32, shape (n_indices,)

        n_indices: tl.int32,
        n_cols: tl.int32,
        BLOCK_SIZE: tl.constexpr,
    ):
        pid = tl.program_id(axis=0)
        if pid >= n_indices:
            return

        idx = tl.load(idx_ptr + pid)  # int32
