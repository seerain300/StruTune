import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    output_ptr,            # *const bf16, shape (M, H)
    expert_ptr,            # *const bf16, shape (N, H)
    idx_ptr,               # *const int32, shape (N,)
    N,                     # int32: number of rows to process (len(expert_ptr))
    H,                     # int32: hidden size (columns)
    BLOCK_H: tl.constexpr, # tile size over hidden dimension
):
    # Each program handles one source row n
    n = tl.program_id(0)
    if n >= N:
        return

    # Load target row index for this source row
    # token_indices are provided as int32 in device memory
    idx = tl.load(idx_ptr + n)
    # If idx is out of range, we still guard with H in mask below

    # Loop over hidden dimension in chunks of BLOCK_H
    # We cast indices to int64 for address computation
    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask = h_offsets < H

        # Load the block of hidden features for this source row
        vals = tl.load(expert_ptr + n * H + h_offsets, mask=mask, other=0.0)

        # Compute destination addresses: output[idx, h_offsets]
        # Cast idx to int64 for pointer arithmetic
        dest = idx.to(tl.int64) * H + h_offsets.to(tl.int64)

        # Atomically add to output
        tl.atomic_add(output_ptr + dest, vals, mask=mask)
