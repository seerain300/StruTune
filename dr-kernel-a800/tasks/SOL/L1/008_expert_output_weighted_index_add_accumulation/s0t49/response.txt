import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_row_kernel(
    output_ptr,            # *bf16 or *fp16, pointer to output (M, H)
    expert_ptr,            # *bf16 or *fp16, pointer to expert outputs (N, H)
    index_ptr,             # *int32, pointer to token indices (N,)
    M, N, H,               # int32 runtime sizes
    BLOCK_H: tl.constexpr  # tile size along H
):
    # One program per source row (n)
    n = tl.program_id(0)
    # If n >= N, exit (guard against any oversized grid)
    if n >= N:
        return

    # Load token index for this source row
    idx = tl.load(index_ptr + n)  # int32
    # Compute base offset for this token row in output: idx * H
    # Note: idx is in [0, M), H is number of columns; output layout is row-major.
    # We will perform atomic adds for each H chunk.

    # Iterate over H in chunks of BLOCK_H
    start = 0
    while start < H:
        offs = start + tl.arange(0, BLOCK_H)
        mask = offs < H

        # Load expert_outputs[n, offs] as a vector
        expert_off = n * H + offs
        vals = tl.load(expert_ptr + expert_off, mask=mask, other=0)

        # Compute output addresses: output[idx, offs]
        out_off = idx * H + offs
        # Atomic add into output
        tl.atomic_add(output_ptr + out_off, vals, mask=mask)

        start += BLOCK_H
