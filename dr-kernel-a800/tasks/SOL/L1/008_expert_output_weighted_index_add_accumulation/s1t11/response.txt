```python
import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_rows_bf16_kernel(
    out_ptr,            # *bf16, shape (M, H)
    expert_ptr,         # *bf16, shape (N, H)
    idx_ptr,            # *int32, shape (N,)
    M: tl.constexpr,    # int
    H: tl.constexpr,    # int
    N: tl.constexpr,    # int
    BLOCK_H: tl.constexpr,
):
    # One program per source row
    row = tl.program_id(0)
    # Each program handles a single row; loop over hidden dimension in tiles.
    # This preserves correctness via atomic adds.
    for j in range(0, H, BLOCK_H):
        offs = j + tl.arange(0, BLOCK_H)
        mask = offs < H

        # Load token index for this row
        idx = tl.load(idx_ptr + row)
        # Compute output row pointer
        out_row_ptr = out_ptr + idx * H

        # Compute expert row pointer (contiguous along H)
        expert_row_ptr = expert_ptr + row * H

        # Load a vector of expert outputs for this row and offsets
        val = tl.load(expert_row_ptr + offs, mask=mask, other=0.0)

        # Atomic add into the output at the token index's row
        tl.atomic_add(out_row_ptr + offs, val, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-based implementation of:
            output = final_hidden_states.clone()
            output.index_add_(0, token_indices, expert_outputs)
        """
        # Ensure device and contiguity
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Triton requires CUDA tensors"
        out = final_hidden_states.clone()

        # Triton prefers int32 for indexing; ensure indices are int32
        idx32 = token_indices.to(torch.int32)

        # Shapes
        M, H = out.shape
        N, H_exp = expert_outputs.shape
        assert H_exp == H, "expert_outputs hidden size must match output hidden size"

        # Launch kernel: one program per row in expert_outputs
        grid = (N,)
        # Choose a reasonable tile size; 128 works well for typical H <= 1024.
        BLOCK_H = 128

        _scatter_add_rows_bf16_kernel[grid](
            out, expert_outputs, idx32,
            M, H, N,
            BLOCK_H=BLOCK_H,
            num_warps=4,  # modest parallelism per program; can tune
            num_stages=2,
        )

        return out
```

```python
import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_multi_n_kernel(
    out_ptr,            # *bf16, shape (M, H)
    expert_ptr,         # *bf16, shape (N, H)
    idx_ptr,            # *int32, shape (N,)
    M: tl.constexpr,    # int
    H: tl.constexpr,    # int
    N: tl.constexpr,    # int
    TILE_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # 2D grid: pid0 over row tiles, pid1 over hidden tiles
    pid0 = tl.program_id(0)  # tile index across N
    pid1 = tl.program_id(1)  # tile index across H

    # Compute the rows this program will handle
    n_start = pid0 * TILE_N
    n_offsets = n_start + tl.arange(0, TILE_N)  # shape (TILE_N,)
    n_mask = n_offsets < N

    # Loop over hidden dimension in tiles
    for j in range(0, H, BLOCK_H):
        h_offs = j + tl.arange(0, BLOCK_H)  # shape (BLOCK_H,)
        h_mask = h_offs < H

        # Load token indices for these rows
        idxs = tl.load(idx_ptr + n_offsets, mask=n_mask, other=0).to(tl.int32)  # (TILE_N,)
        # Compute output pointers for each row
        out_row_ptrs = out_ptr + idxs[:, None] * H  # (TILE_N, 1) broadcastable
        # Compute expert pointers for each row
        expert_row_ptrs = expert_ptr + n_offsets[:, None] * H  # (TILE_N, 1)

        # Full mask for the 2D tile: valid rows and valid hidden positions
        full_mask = n_mask[:, None] & h_mask[None, :]

        # Load expert outputs for this tile: (TILE_N, BLOCK_H)
        vals = tl.load(expert_row_ptrs + h_offs[None, :], mask=full_mask, other=0.0)

        # Atomic add to output at corresponding token indices
        tl.atomic_add(out_row_ptrs + h_offs[None, :], vals, mask=full_mask)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-based implementation of:
            output = final_hidden_states.clone()
            output.index_add_(0, token_indices, expert_outputs)
        """
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Triton requires CUDA tensors"
        out = final_hidden_states.clone()

        # Triton prefers int32 for indexing
        idx32 = token_indices.to(torch.int32).contiguous()
        # Ensure inputs are contiguous
        expert_outputs = expert_outputs.contiguous()

        # Shapes
        M, H = out.shape
        N, H_exp = expert_outputs.shape
        assert H_exp == H, "expert_outputs hidden size must match output hidden size"

        # Tile sizes
        TILE_N = 64        # number of rows handled per program
        BLOCK_H = 128      # hidden dimension tile size

        # Grid: tile across N and H
        grid0 = (N + TILE_N - 1) // TILE_N
        grid1 = (H + BLOCK_H - 1) // BLOCK_H
        grid = (grid0, grid1)

        _scatter_add_multi_n_kernel[grid](
            out, expert_outputs, idx32,
            M, H, N,
            TILE_N=TILE_N,
            BLOCK_H=BLOCK_H,
            num_warps=4,
            num_stages=2,
        )

        return out
```
