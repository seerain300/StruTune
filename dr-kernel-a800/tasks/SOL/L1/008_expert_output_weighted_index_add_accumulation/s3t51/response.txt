Final Triton-optimized code (ModelNew.forward)
```python
import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_experts_kernel(
    out_ptr,                # *float32, [batch_seq_len, hidden_size]
    idx_ptr,                # *int32,   [N]
    src_ptr,                # *float32, [N, hidden_size]
    N: tl.constexpr,        # number of updates (num_selected_tokens)
    H: tl.constexpr,        # hidden_size (columns)
    T: tl.constexpr,        # updates per program
    BLOCK_H: tl.constexpr,  # chunk size across hidden dimension
):
    pid = tl.program_id(axis=0)
    start = pid * T
    # Vector of lanes across the T updates handled by this program
    k = tl.arange(0, T)
    pos = start + k  # [T]
    mask_pos = pos < N  # mask for valid updates

    # Load indices for these T updates (int32)
    idx = tl.load(idx_ptr + pos, mask=mask_pos, other=0)  # [T], int32

    # Iterate over hidden dimension in chunks
    for h in range(0, H, BLOCK_H):
        h_vec = h + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = h_vec < H

        # For each lane (k), load v_chunk = src[pos, h:h+BLOCK_H]
        # We need a 2D view: rows=pos (T), columns=h_vec (BLOCK_H)
        # Build pointer for each lane:
        # src_ptr at [pos, h_vec]
        # We can compute as: src_ptr + pos * H + h_vec
        # Note: src_ptr is row-major [N, H], contiguous per row.
        # Compute per-lane pointers
        # To do vectorized loads, we can loop over k (T lanes) and load per lane:
        # But Triton supports broadcasting with 2D pointer: pos[:, None]*H + h_vec[None, :].
        # However, Triton expects pointer arithmetic with 1D offsets; we emulate by
        # loading each lane individually using a loop over k (T is small, acceptable).
        for kk in range(T):
            p = pos[kk]
            valid_k = p < N  # scalar
            # Compute per-lane column offsets for this chunk
            offs = p * H + h_vec  # [BLOCK_H]
            # Load v_chunk from src for this lane
            v = tl.load(src_ptr + offs, mask=mask_pos[kk] & mask_h, other=0.0)  # [BLOCK_H], float32
            # Compute output row pointer: out_ptr + idx[kk] * H + h_vec
            out_offs = idx[kk] * H + h_vec  # [BLOCK_H]
            # Atomic add for this lane
            # Note: atomic_add supports float32 in Triton
            tl.atomic_add(out_ptr + out_offs, v, mask=mask_pos[kk] & mask_h)


def _choose_kernel_params(H: int):
    # Choose BLOCK_H and num_warps based on hidden size
    if H >= 1024:
        BLOCK_H = 1024
        num_warps = 8
    elif H >= 512:
        BLOCK_H = 512
        num_warps = 4
    else:
        BLOCK_H = 256
        num_warps = 2
    return BLOCK_H, num_warps


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        # Ensure CUDA and contiguity
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "All tensors must be on CUDA device."
        final_hidden_states = final_hidden_states.contiguous()
        expert_outputs = expert_outputs.contiguous()
        token_indices = token_indices.contiguous()

        # Shapes
        N = expert_outputs.shape[0]
        H = final_hidden_states.shape[1]

        # Accumulate in float32; original uses bfloat16, but atomics are on fp32
        out_fp32 = final_hidden_states.clone().to(torch.float32)

        # Cast expert_outputs to float32 for accumulation
        src_fp32 = expert_outputs.to(torch.float32)

        # Triton kernel launch
        BLOCK_H, num_warps = _choose_kernel_params(H)
        # Each program handles T updates
        T = 4
        grid = (triton.cdiv(N, T),)

        _scatter_add_experts_kernel[grid](
            out_fp32,
            token_indices.to(torch.int32),
            src_fp32,
            N=N,
            H=H,
            T=T,
            BLOCK_H=BLOCK_H,
            num_warps=num_warps,
            num_stages=2,
        )

        # Cast back to bfloat16 to match original API
        return out_fp32.to(torch.bfloat16)
```
