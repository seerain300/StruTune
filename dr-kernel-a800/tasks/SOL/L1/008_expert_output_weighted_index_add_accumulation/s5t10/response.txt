try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: scatter-add expert_outputs[i, :] into output[token_indices[i], :]
# One program per index (i). Vectorize across hidden_size in chunks of BLOCK_SIZE.
@triton.jit
def scatter_add_rows_kernel(
    output_ptr,           # *pointer* to output tensor (float32/float16/bfloat16), shape [M, H]
    expert_ptr,           # *pointer* to expert_outputs tensor, shape [N, H]
    indices_ptr,          # *pointer* to token_indices tensor, int32, shape [N]
    M,                    # int: number of rows in output (batch_size * seq_len)
    N,                    # int: number of expert outputs
    H,                    # int: hidden_size (number of columns)
    BLOCK_SIZE: tl.constexpr,  # chunk size for columns, power-of-two up to 512
):
    pid = tl.program_id(axis=0)  # program id over indices
    # Guard in case grid > N (not strictly necessary if grid == N)
    if pid >= N:
        return

    # Load destination row index and cast to int32
    idx = tl.load(indices_ptr + pid).to(tl.int32)

    # Base offsets for rows in 2D tensors: row-major, contiguous, strides (H, 1) for last-dim contiguous
    # For a contiguous tensor of shape [M, H], linear index = row * H + col.
    # We'll compute per-chunk column offsets as j + arange(0, BLOCK_SIZE)
    # and atomic_add the corresponding slice from expert_ptr[pid, :] into output[idx, :].
    j = 0
    # Loop over hidden_size in chunks
    while j < H:
        cols = j + tl.arange(0, BLOCK_SIZE)
        mask = cols < H

        # Compute pointers for the current chunk:
        # Output row: idx * H + cols
        out_ptrs = output_ptr + idx * H + cols
        # Expert row: pid * H + cols
        exp_ptrs = expert_ptr + pid * H + cols

        # Load the chunk from expert_outputs; 'other' is for masked lanes, but masked lanes won't be stored
        vals = tl.load(exp_ptrs, mask=mask, other=0)

        # Atomic add into output
        # Note: Triton's atomic_add supports fp16/bf16/fp32. Ensure output_ptr points to correct dtype.
        tl.atomic_add(out_ptrs, vals, mask=mask)

        j += BLOCK_SIZE

        else:

            block_size = min(block_size, 512)        # cap to 512

        grid = (N,)

        # Run Triton kernel
        scatter_add_rows_kernel[grid](
            output,                    # output_ptr
            expert_outputs,            # expert_ptr
            indices_i32,               # indices_ptr
            M, N, H,                   # dimensions
            BLOCK_SIZE=block_size,     # constexpr meta-parameter
            num_warps=4,               # balanced default
            num_stages=2,              # modest pipelining
        )

        return output
