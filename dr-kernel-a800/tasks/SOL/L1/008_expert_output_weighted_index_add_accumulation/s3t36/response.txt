import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_full_row_kernel(
    out_ptr,        # *float32, shape [B, H]
    indices_ptr,    # *int32,   shape [N]
    vals_ptr,       # *float32, shape [N, H]
    N,              # int32, number of updates
    B,              # int32, number of rows in output (batch_seq_len)
    H,              # int32, hidden size
    T: tl.constexpr,     # number of updates per program
    BLOCK_H: tl.constexpr,  # should be >= H in full-row case (we choose 1024)
):
    # Each program handles up to T updates: pos = pid * T + k
    pid = tl.program_id(axis=0)
    # Construct N as Triton scalar for arithmetic
    N_s = tl.full((), N, tl.int32)
    # Loop over T updates handled by this program
    for k in range(T):
        pos = pid * T + k
        # mask for valid pos < N
        valid = pos < N_s
        # Load index and guard with valid
        idx = tl.load(indices_ptr + pos, mask=valid, other=0)
        # Compute base row pointer
        # out_ptr is row-major: row start = idx * H
        row_start = idx * H
        # Load the full vals row and atomic add into output
        # vals_ptr indexing: row of vals for update pos starts at vals_ptr + pos * H
        # We read H elements; masked load handles valid
        # We'll iterate h from 0 to H-1 with a vectorized assignment
        # Triton allows simple elementwise pointer arithmetic:
        # Build a vector of offsets for h in [0, H)
        h = tl.arange(0, H)
        vals_row_ptrs = vals_ptr + pos * H + h
        out_row_ptrs = out_ptr + row_start + h
        # Mask: only perform atomic add if valid
        # For invalid pos, load zeros and do not add
        vals_vec = tl.load(vals_row_ptrs, mask=valid, other=0.0)
        # atomic add only where valid
        tl.atomic_add(out_row_ptrs, vals_vec, mask=valid)


@triton.jit
def scatter_add_chunked_kernel(
    out_ptr,        # *float32, shape [B, H]
    indices_ptr,    # *int32,   shape [N]
    vals_ptr,       # *float32, shape [N, H]
    N,              # int32, number of updates
    B,              # int32, number of rows in output (batch_seq_len)
    H,              # int32, hidden size
    T: tl.constexpr,     # number of updates per program
    BLOCK_H: tl.constexpr,  # chunk size for hidden dimension
):
    pid = tl.program_id(axis=0)
    N_s = tl.full((), N, tl.int32)
    # Process up to T updates per program
    for k in range(T):
        pos = pid * T + k
        valid = pos < N_s
        idx = tl.load(indices_ptr + pos, mask=valid, other=0)
        row_start = idx * H
        # Iterate over hidden dimension in chunks of BLOCK_H
        # h is a runtime scalar; we build offsets per chunk
        h = 0
        while h < H:
            offs = h + tl.arange(0, BLOCK_H)
            mask = (offs < H) & valid
            vals_ptrs = vals_ptr + pos * H + offs
            out_ptrs = out_ptr + row_start + offs
            vals_chunk = tl.load(vals_ptrs, mask=mask, other=0.0)
            tl.atomic_add(out_ptrs, vals_chunk, mask=mask)
            h += BLOCK_H


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        # Ensure CUDA tensors
        assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "Inputs must be on CUDA device for Triton kernel."
        # Ensure contiguous
        out_fp32 = final_hidden_states.to(torch.float32).contiguous()
        vals_fp32 = expert_outputs.to(torch.float32).contiguous()
        indices_i32 = token_indices.to(torch.int32).contiguous()
        B = out_fp32.shape[0]
        H = out_fp32.shape[1]
        N = indices_i32.shape[0]

        # Heuristics for kernels
        # Choose BLOCK_H and num_warps based on H
        if H <= 1024:
            BLOCK_H = 1024
            num_warps = 8
            T = 8  # number of updates per program
            grid = (triton.cdiv(N, T),)
            # Full-row kernel: minimal atomics
            scatter_add_full_row_kernel[grid](
                out_fp32, indices_i32, vals_fp32,
                N, B, H,
                T=T, BLOCK_H=BLOCK_H,
                num_warps=num_warps,
                num_stages=1,
            )
        else:
            # Chunked kernel for larger H
            if H >= 2048:
                BLOCK_H = 1024
                num_warps = 8
                T = 8
            elif H >= 1024:
                BLOCK_H = 512
                num_warps = 4
                T = 8
            else:
                BLOCK_H = 256
                num_warps = 2
                T = 8
            grid = (triton.cdiv(N, T),)

                out_fp32, indices_i32, vals_fp32,
                N, B, H,
                T=T, BLOCK_H=BLOCK_H,
                num_warps=num_warps,
                num_stages=1,
