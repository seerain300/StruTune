import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_chunks_kernel(
    out_ptr: tl.pointer_type(dtype=tl.float32),   # we accumulate in fp32
    indices_ptr: tl.pointer_type(dtype=tl.int32),  # token_indices
    src_ptr: tl.pointer_type(dtype=tl.bfloat16),    # expert_outputs (bfloat16)
    N: tl.int32,
    H: tl.int32,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)  # one program per update
    # Load token index
    idx = tl.load(indices_ptr + pid)  # int32

    # Iterate over hidden dimension in BLOCK_H chunks
    # Note: Triton loops require compile-time constants; step is BLOCK_H
    for h in range(0, H, BLOCK_H):
        offs = h + tl.arange(0, BLOCK_H)
        mask = offs < H

        # Load expert output chunk (bfloat16), cast to float32 for atomic_add
        v = tl.load(src_ptr + pid * H + offs, mask=mask, other=0.0)  # bfloat16
        v = v.to(tl.float32)

        # Atomic add into output at row idx
        # Pointer arithmetic: out_ptr is 2D (rows=N, cols=H), but we only add at one row (idx)
        # We need to compute the linear offset for each column: idx * H + offs
        out_row_ptr = out_ptr + idx * H
        tl.atomic_add(out_row_ptr + offs, v, mask=mask)


def _triton_scatter_add(final_hidden_states: torch.Tensor,
                        expert_outputs: torch.Tensor,
                        token_indices: torch.Tensor,
                        hidden_size: int):
    """
    Triton scatter-add in fp32 for performance; returns bfloat16 to match original API.
    final_hidden_states: (batch_seq_len, hidden_size), bfloat16
    expert_outputs: (num_selected_tokens, hidden_size), bfloat16
    token_indices: (num_selected_tokens,), long
    Returns: updated final_hidden_states (bfloat16)
    """
    assert final_hidden_states.is_cuda and expert_outputs.is_cuda and token_indices.is_cuda, "All tensors must be on CUDA."
    # Ensure contiguity
    final_hidden_states = final_hidden_states.contiguous()
    expert_outputs = expert_outputs.contiguous()
    token_indices = token_indices.contiguous()

    # Prepare dtypes for Triton
    token_indices_i32 = token_indices.to(torch.int32)  # indices for Triton

    # We accumulate in float32 for atomic_add support
    # Clone to avoid modifying input
    out_fp32 = final_hidden_states.to(torch.float32).clone()

    N = expert_outputs.shape[0]
    H = hidden_size

    # Heuristic selection for BLOCK_H and num_warps
    if H >= 1024:
        BLOCK_H = 1024
        num_warps = 8
    elif H >= 512:
        BLOCK_H = 512
        num_warps = 4
    else:
        BLOCK_H = 256
        num_warps = 2

    # Launch kernel: one program per update
    grid = (N,)
    _scatter_add_chunks_kernel[grid](
        out_fp32,
        token_indices_i32,
        expert_outputs,  # keep bfloat16; cast inside kernel to float32
        N,
        H,
        BLOCK_H=BLOCK_H,
        num_warps=num_warps,
        num_stages=2,
    )

    # Cast back to bfloat16 to match original API
    return out_fp32.to(torch.bfloat16)

          final_hidden_states[token_indices[i]] += expert_outputs[i, :]
        Returns updated final_hidden_states.
        """

        assert final_hidden_states.dim() == 2, "final_hidden_states must be (batch_seq_len, hidden_size)"
        hidden_size = final_hidden_states.shape[1]
        return _triton_scatter_add(final_hidden_states, expert_outputs, token_indices, hidden_size)
