```python
import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,        # *const bfloat16, shape [M, H]
    src_ptr,        # *const bfloat16, shape [N, H]
    indices_ptr,    # *const int32,    shape [N]
    M: tl.constexpr,    # number of rows in output
    H: tl.constexpr,    # hidden size (columns)
    N: tl.constexpr,    # number of source rows (N = M * num_experts_per_tok)
    BLOCK_H: tl.constexpr,  # tile size along hidden dim
):
    # One program handles one source row i
    i = tl.program_id(0)
    # If i >= N, nothing to do (safety, though grid should be N)
    if i >= N:
        return

    # Load the token index for this row
    idx = tl.load(indices_ptr + i)  # int32
    # Compute base offsets for this row
    # out_ptr is row-major: row offset = i * H
    # src_ptr is row-major: row offset = i * H
    # Note: out is [M, H], src is [N, H]; i < N guaranteed by grid. idx is in [0, M).
    # We will atomic add each H-tile of src[i, :] into out[idx, :].

    # Loop over hidden dimension in tiles of size BLOCK_H
    for h_off in range(0, H, BLOCK_H):
        offs = h_off + tl.arange(0, BLOCK_H)
        mask = offs < H

        # Load src[i, offs]
        src_row_ptr = src_ptr + i * H
        vals = tl.load(src_row_ptr + offs, mask=mask, other=0.0)  # bfloat16

        # Compute destination pointers: out[idx, offs]
        out_row_ptr = out_ptr + idx * H
        # Atomic add into output
        tl.atomic_add(out_row_ptr + offs, vals, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, final_hidden_states: torch.Tensor, expert_outputs: torch.Tensor, token_indices: torch.Tensor):
        """
        Triton-based implementation of:
          output = final_hidden_states.clone()
          output.index_add_(dim=0, index=token_indices, source=expert_outputs)
        Requirements:
        - Use Triton kernels for the atomic accumulation.
        - No torch.index_add or torch ops in host code other than tensor creation and launch.
        """
        # Ensure inputs are contiguous and dtypes are as expected
        out = final_hidden_states.clone()  # clone to match reference semantics
        # Triton prefers int32 for indices
        indices_i32 = token_indices.to(torch.int32)
        # Source tensor (may be non-contiguous if sliced, but we can enforce contiguous)
        src = expert_outputs

        # Shapes
        M, H = out.shape
        N = expert_outputs.shape[0]

        # Launch Triton kernel: one program per source row
        grid = (N,)

        # Choose a robust tile size and launch configuration
        BLOCK_H = 256
        scatter_add_rows_kernel[grid](
            out, src, indices_i32,
            M=M, H=H, N=N,
            BLOCK_H=BLOCK_H,
            num_warps=4,  # good balance for this workload
            num_stages=2,  # modest pipelining
        )

        return out
```

- We keep everything Triton-only on the hot path: no torch.index_add in the host code other than the initial clone, which is necessary to match the reference semantics. All accumulation is done by the Triton kernel with proper atomics.
