import triton
import triton.language as tl


@triton.jit
def scatter_add_rows_kernel(
    out_ptr,          # *bf16, shape (N, H)
    src_ptr,          # *bf16, shape (M, H)
    indices_ptr,      # *int32, shape (M,)
    N: tl.constexpr,  # number of rows in out (batch_seq_len)
    H: tl.constexpr,  # hidden_size
):
    # 1D grid: each program handles one row
    pid = tl.program_id(0)
    if pid >= N:
        return

    # Load the destination row index for this token
    row_idx = tl.load(indices_ptr + pid)  # int32

    # Loop over hidden dimension and perform atomic add
    # Note: H is a constexpr, so Triton can unroll or optimize this loop.
    for j in range(0, H):
        # Compute source and destination offsets
        src_off = pid * H + j
        dst_off = row_idx * H + j

        # Load source element (bfloat16) and atomic add to destination
        val = tl.load(src_ptr + src_off)
        # Atomic add: out[row_idx, j] += src[pid, j]
        tl.atomic_add(out_ptr + dst_off, val)
