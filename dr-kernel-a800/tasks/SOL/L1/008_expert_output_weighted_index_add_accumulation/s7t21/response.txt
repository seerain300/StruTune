import torch
import triton
import triton.language as tl


@triton.jit
def scatter_add_scalar_kernel(
    out_ptr,        # *bf16, shape (N, H)
    src_ptr,        # *bf16, shape (M, H)
    indices_ptr,    # *int32, shape (M,)
    N,              # int32, number of rows in out (batch_seq_len)
    H: tl.constexpr # hidden size (compile-time constant for loop unrolling)
):
    # One program per row i
    i = tl.program_id(0)
    # If grid > N, mask out extra programs
    if i >= N:
        return

    # Compute destination row index
    row_idx = tl.load(indices_ptr + i)  # int32

    # Loop over hidden dimension and accumulate with atomic add
    for j in range(H):
        # Compute source and destination addresses
        src_offset = i * H + j
        dst_offset = row_idx * H + j

        # Load source value (bf16)
        val = tl.load(src_ptr + src_offset)

        # Atomic add to destination
        tl.atomic_add(out_ptr + dst_offset, val)

            num_stages=1

import triton
import triton.language as tl


@triton.jit
def scatter_add_rowvec_kernel(
    out_ptr,        # *bf16, shape (N, H)
    src_ptr,        # *bf16, shape (M, H)
    indices_ptr,    # *int32, shape (M,)
    N,              # int32, number of rows in out (batch_seq_len)
    H: tl.constexpr # hidden size (compile-time constant)
):

    if i >= N:
        return

    # Destination row index for this selected token
    row_idx = tl.load(indices_ptr + i)  # int32

    # Vector of column offsets
    offs = tl.arange(0, H)  # [0..H-1]

    # Compute source row base and load entire row vector
    src_base = i * H

    # Load destination row (out[row_idx, :]) as vector
