import torch
import triton
import triton.language as tl


@triton.jit
def _scatter_add_rows_kernel(
    out_ptr,        # *bf16, shape (M, H), destination
    in_ptr,         # *bf16, shape (N, H), source rows to add
    idx_ptr,        # *int32, shape (N,), token indices (row positions in output)
    M: tl.constexpr,     # number of rows in output (batch_size * seq_len)
    H: tl.constexpr,     # hidden size
    N: tl.constexpr,     # number of selected tokens (rows in expert_outputs)
    BLOCK: tl.constexpr  # columns processed per chunk
):
    # One program per selected token (row in expert_outputs)
    i = tl.program_id(0)

    # Guard against overlaunch
    if i >= N:
        return

    # Load the entire expert row i into a vector
    cols = tl.arange(0, BLOCK)
    # Compute the starting offset for this row
    in_row_offset = i * H
    # Pointer to this row in in_ptr
    in_row_ptrs = in_ptr + in_row_offset + cols
    # Mask for valid columns (handles tail when H not divisible by BLOCK)
    mask_in = cols < H
    # Load with other=0 to zero out masked elements
    vals = tl.load(in_row_ptrs, mask=mask_in, other=0.0)

    # Destination row index
    dest = tl.load(idx_ptr + i)  # int32
    # Destination row base pointer
    out_row_ptrs = out_ptr + dest * H + cols

    # Perform vectorized atomic add for this chunk
    # Note: only one chunk here to keep correctness and simplicity.
    tl.atomic_add(out_row_ptrs, vals, mask=mask_in)
