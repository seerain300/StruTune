核心分析
- 原 PyTorch 逻辑步骤：
  1) A_cumsum: shape [B, H, N, S]
  2) Lower-triangular mask on the chunk dimension (S x S): diagonal=-1, so for each chunk n, i >= j.
  3) Segment sum along the S dimension: for each i, segment_sum = sum_{k=0..i} A[b,h,n,k] with j <= i; j > i contributes 0. Then L = exp(segment_sum). This produces L[b,h,n,i,j] where j <= i gets exp(sum_k^i A), else 0.
  4) Contract B and C (already expanded to H) to form G[i,j,h] = sum_d C[b,n,i,h,d] * B[b,n,j,h,d].
  5) Apply mask: M = G * L.
  6) Reduce: Y[b,n,i,h] = sum_j M[b,n,i,j,h] * hidden_states[b,n,j,h,:].

- 我之前的尝试中出现的问题：
  - 或许没有正确地使用 Triton 内核完成所有数值计算，导致被视为 decoy。
  - 也可能存在 Triton 内核中的 stride 计算错误、grid 不覆盖所有情况，或者数据类型不匹配导致运行时错误。
  - L 的计算必须严格遵循 tril(diagonal=-1) 语义：对于每个 chunk n 的矩阵 S x S，下三角部分 j <= i 有效，上三角部分 j > i 应该为 0。

解决方案
- 为了确保 correctness 优先，我将重新构建每个步骤的 Triton 内核，并且 ModelNew.forward 将实际调用它们：
  1) Triton kernel tril_mask_kernel: 生成 1D int8 mask of length S*S, where lower[i*S + j] = 1 if j <= i else 0.
  2) Triton kernel a_segment_sum_exp_kernel: 为每个 (b,h,n), 计算 L[b,h,n,i,j] = exp(cumulative sum from 0..i) only if j <= i, else 0.0. Grid dims: (B,H,N). Inside kernel, we loop i and j.
  3) Triton kernel g_contract_kernel: compute G[i,j,h] via reduction over D. Grid dims: (B,N,S,S,H), loop over D in tiles.
  4) Triton kernel m_mul_kernel: elementwise multiply M = G * L, after permute L to [B,N,S,S,H] to match G’s last dim order.
  5) Triton kernel y_diag_reduce_kernel: compute Y[b,n,i,h] = sum_j M[b,n,i,j,h] * hidden_states[b,n,j,h,:]. Grid dims: (B,N,S,H), loop j and D in tiles.

- 为了保证安全性，我将所有 tensor 在 kernel 之前设置为 contiguous 并使用 .to(torch.float32) for computation; output cast to bfloat16 at the end to match the original code.

- Performance considerations:
  - The heavy part is the construction of L (B*H*N*S*S). Triton loops over i and j; we use grid (B,H,N) and loop inside. This is necessary for correctness with tril(diagonal=-1). For many (B,N,S), this may be slower than optimized PyTorch, but the requirement is correctness first. If all workloads pass, we can later optimize by:
    - Vectorizing i/j tiles.
    - Using more warps/stages.
    - Specializing when S is small (e.g., S=128).

```python
import torch
import triton
import triton.language as tl


# Kernel 1: Generate 1D lower-triangular mask with diagonal=-1.
# Outputs mask_lower as int8[ S*S ]: 1 where j <= i, else 0.
@triton.jit
def tril_mask_kernel(mask_ptr, S: tl.constexpr):
    idx = tl.program_id(0)
    S_int = S
    i = idx // S
    j = idx % S
    if j <= i:
        tl.store(mask_ptr + idx, 1)
    else:
        tl.store(mask_ptr + idx, 0)


# Kernel 2: Compute L = exp(segment_sum(A)) with tril(diagonal=-1).
# Input A: [B, H, N, S], float32.
# Output L: [B, H, N, S, S], float32.
@triton.jit
def a_segment_sum_exp_kernel(
    A_ptr, L_ptr,
    B_size, H_size, N_size, S_size,
    A_stride_b, A_stride_h, A_stride_n, A_stride_s,
    L_stride_b, L_stride_h, L_stride_n, L_stride_s1, L_stride_s2,
):
    pid_b = tl.program_id(0)  # b
    pid_h = tl.program_id(1)  # h
    pid_n = tl.program_id(2)  # n

    # We fill L[b,h,n,i,j] for all i,j in [0..S-1]
    # Segment sum from 0..i only if j <= i; else 0.0
    for i in range(0, S_size):
        seg_sum = 0.0
        # Compute segment sum up to i
        # A[b,h,n,i] address: b*stride_b + h*stride_h + n*stride_n + i*stride_s
        # Loop k from 0 to i (inclusive)
        # Note: S_size is constexpr in Triton, so this is fine.
        for k in range(0, i + 1):
            a_val = tl.load(A_ptr + pid_b * A_stride_b + pid_h * A_stride_h + pid_n * A_stride_n + k * A_stride_s)
            seg_sum += a_val
        # Now fill j from 0..i with exp(seg_sum), and j>i with 0.0
        for j in range(0, S_size):
            if j <= i:
                l_val = tl.exp(seg_sum)
            else:
                l_val = 0.0
            # L address: b*stride_b + h*stride_h + n*stride_n + i*stride_s1 + j*stride_s2
            tl.store(L_ptr + pid_b * L_stride_b + pid_h * L_stride_h + pid_n * L_stride_n + i * L_stride_s1 + j * L_stride_s2, l_val)


# Kernel 3: Compute G[i,j,h] = sum_d C[b,n,i,h,d] * B[b,n,j,h,d]
# Inputs: B [B,N,S,H,D], C [B,N,S,H,D], Output: G [B,N,S,S,H], float32.
@triton.jit
def g_contract_kernel(
    B_ptr, C_ptr, G_ptr,
    B_size, N_size, S_size, H_size, D_size,
    B_stride_b, B_stride_n, B_stride_s, B_stride_h, B_stride_d,
    C_stride_b, C_stride_n, C_stride_s, C_stride_h, C_stride_d,
    G_stride_b, G_stride_n, G_stride_s1, G_stride_s2, G_stride_h,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)  # b
    pid_n = tl.program_id(1)  # n
    pid_si = tl.program_id(2)  # i in [0..S-1]
    pid_sj = tl.program_id(3)  # j in [0..S-1]
    pid_h = tl.program_id(4)   # h in [0..H-1]

    # Accumulate over D in tiles
    acc = 0.0
    # Loop d from 0 to D in chunks of BLOCK_D
    for d_start in range(0, D_size, BLOCK_D):
        d_idx = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_idx < D_size
        # Load B[b,n,j,h,d] and C[b,n,i,h,d]
        b_vec = tl.load(
            B_ptr + pid_b * B_stride_b + pid_n * B_stride_n + pid_sj * B_stride_s + pid_h * B_stride_h + d_idx * B_stride_d,
            mask=mask_d, other=0.0
        )
        c_vec = tl.load(
            C_ptr + pid_b * C_stride_b + pid_n * C_stride_n + pid_si * C_stride_s + pid_h * C_stride_h + d_idx * C_stride_d,
            mask=mask_d, other=0.0
        )
        # Fused multiply and reduce
        acc += tl.sum(b_vec * c_vec, axis=0)
    # Store G[b,n,i,j,h] = acc
    tl.store(
        G_ptr + pid_b * G_stride_b + pid_n * G_stride_n + pid_si * G_stride_s1 + pid_sj * G_stride_s2 + pid_h * G_stride_h,
        acc
    )


# Kernel 4: Elementwise M = G * L. G: [B,N,S,S,H], L: [B,N,S,S,H]. Output M same shape.
@triton.jit
def m_mul_kernel(
    G_ptr, L_ptr, M_ptr,
    B_size, N_size, S_size, H_size,
    G_stride_b, G_stride_n, G_stride_s1, G_stride_s2, G_stride_h,
    L_stride_b, L_stride_n, L_stride_s1, L_stride_s2, L_stride_h,
    M_stride_b, M_stride_n, M_stride_s1, M_stride_s2, M_stride_h,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)
    pid_sj = tl.program_id(3)
    pid_h = tl.program_id(4)

    g_val = tl.load(G_ptr + pid_b * G_stride_b + pid_n * G_stride_n + pid_si * G_stride_s1 + pid_sj * G_stride_s2 + pid_h * G_stride_h)
    l_val = tl.load(L_ptr + pid_b * L_stride_b + pid_n * L_stride_n + pid_si * L_stride_s1 + pid_sj * L_stride_s2 + pid_h * L_stride_h)
    m_val = g_val * l_val
    tl.store(M_ptr + pid_b * M_stride_b + pid_n * M_stride_n + pid_si * M_stride_s1 + pid_sj * M_stride_s2 + pid_h * M_stride_h, m_val)


# Kernel 5: Reduce Y[b,n,i,h] = sum_j M[b,n,i,j,h] * hidden_states[b,n,j,h,:].
# Inputs: M [B,N,S,S,H], hidden_states [B,N,S,H,D], Output Y [B,N,S,H], float32.
@triton.jit
def y_diag_reduce_kernel(
    M_ptr, hidden_ptr, Y_ptr,
    B_size, N_size, S_size, H_size, D_size,
    M_stride_b, M_stride_n, M_stride_s1, M_stride_s2, M_stride_h,
    hidden_stride_b, hidden_stride_n, hidden_stride_s, hidden_stride_h, hidden_stride_d,
    Y_stride_b, Y_stride_n, Y_stride_s1, Y_stride_h,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)  # i
    pid_h = tl.program_id(3)   # h

    acc = 0.0
    # Loop j over S dimension in tiles
    for j_start in range(0, S_size, 1):  # S is small (128), one tile per j
        j = j_start
        # Load M[b,n,i,j,h]
        m_val = tl.load(
            M_ptr + pid_b * M_stride_b + pid_n * M_stride_n + pid_si * M_stride_s1 + j * M_stride_s2 + pid_h * M_stride_h
        )
        # Load hidden[b,n,j,h,:] vector over D
        d_idx = tl.arange(0, BLOCK_D)
        mask_d = d_idx < D_size
        hidden_vec = tl.load(
            hidden_ptr + pid_b * hidden_stride_b + pid_n * hidden_stride_n + j * hidden_stride_s + pid_h * hidden_stride_h + d_idx * hidden_stride_d,
            mask=mask_d, other=0.0
        )
        # Multiply M scalar with hidden vector
        # We need to broadcast m_val to vector length D
        hidden_vec = hidden_vec * m_val
        # Reduce over D
        acc += tl.sum(hidden_vec, axis=0)
    # Store Y[b,n,i,h]
    tl.store(Y_ptr + pid_b * Y_stride_b + pid_n * Y_stride_n + pid_si * Y_stride_s1 + pid_h * Y_stride_h, acc)


class ModelNew(torch.nn.Module):
    def __init__(self, CHUNK_SIZE=128, NUM_HEADS=32, N_GROUPS=8, D_STATE=32):
        super().__init__()
        self.CHUNK_SIZE = CHUNK_SIZE
        self.NUM_HEADS = NUM_HEADS
        self.N_GROUPS = N_GROUPS
        self.D_STATE = D_STATE  # state_size, used for G contraction and Y reduction

    def forward(self, hidden_states: torch.Tensor, A_cumsum: torch.Tensor, B: torch.Tensor, C: torch.Tensor) -> torch.Tensor:
        """
        hidden_states: [B, N, S, H, D]
        A_cumsum:      [B, H, N, S]
        B:             [B, N, S, G, D]
        C:             [B, N, S, G, D]
        Output:        [B, N, S, H] in bfloat16
        """
        device = hidden_states.device
        B_size, N_size, S_size, H_size, D_size = hidden_states.shape

        # Ensure inputs are contiguous and float32 for computation
        A = A_cumsum.to(torch.float32).contiguous()        # [B,H,N,S]
        Bc = B.to(torch.float32).contiguous()              # [B,N,S,G,D]
        Cc = C.to(torch.float32).contiguous()              # [B,N,S,G,D]

        # 1) Generate lower-triangular mask (int8) of length S*S on device
        mask_lower = torch.empty(S_size * S_size, dtype=torch.int8, device=device)
        tril_mask_kernel[(S_size * S_size,)](mask_lower, S_size)

        # 2) Compute L = exp(segment_sum(A)) with tril(diagonal=-1) via Triton
        #    L: [B,H,N,S,S] float32
        L = torch.empty((B_size, H_size, N_size, S_size, S_size), dtype=torch.float32, device=device)
        a_segment_sum_exp_kernel[(B_size, H_size, N_size)](
            A, L,
            B_size, H_size, N_size, S_size,
            A.stride(0), A.stride(1), A.stride(2), A.stride(3),
            L.stride(0), L.stride(1), L.stride(2), L.stride(3), L.stride(4),
            num_warps=1, num_stages=1
        )

        # 3) Expand B/C from G to H (original code uses repeat_interleave)
        #    Since the provided B/C are not expanded, we assume NUM_HEADS/H and try to match original assumption:
        #    Original: B has [G = N_GROUPS], and forward expands to H=NUM_HEADS via repeat_interleave in PyTorch.
        #    Here, to keep Triton, we launch kernels assuming H is already expanded. If not, we expand B/C before kernel.
        #    Note: The original B/C shapes should be [B,N,S,H,D]. If they are [B,N,S,G,D], expand G->H.
        #    We expand to H using repeat_interleave: H = NUM_HEADS, G = N_GROUPS, NUM_HEADS // N_GROUPS == 4.
        #    Perform expansion via .repeat_interleave(4) on dim=3.
        if Bc.shape[3] != H_size:
            Bc = Bc.repeat_interleave(H_size // Bc.shape[3], dim=3)
        if Cc.shape[3] != H_size:
            Cc = Cc.repeat_interleave(H_size // Cc.shape[3], dim=3)

        # Ensure Bc, Cc have shape [B,N,S,H,D]
        assert Bc.shape[3] == H_size and Cc.shape[3] == H_size, "B/C must be expanded to NUM_HEADS."

        # 4) Compute G[i,j,h] via Triton contraction over D
        G = torch.empty((B_size, N_size, S_size, S_size, H_size), dtype=torch.float32, device=device)
        g_contract_kernel[(B_size, N_size, S_size, S_size, H_size)](
            Bc, Cc, G,
            B_size, N_size, S_size, H_size, self.D_STATE,
            Bc.stride(0), Bc.stride(1), Bc.stride(2), Bc.stride(3), Bc.stride(4),
            Cc.stride(0), Cc.stride(1), Cc.stride(2), Cc.stride(3), Cc.stride(4),
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
            BLOCK_D=64,  # D_STATE=32, using 64 for robustness; masked loads handle tail
            num_warps=4, num_stages=2
        )

        # 5) M = G * L (elementwise). Permute L to match G dims: [B,N,S,S,H]
        L_perm = L.permute(0, 2, 3, 4, 1).contiguous()  # [B,N,S,S,H]
        M = torch.empty_like(G)
        m_mul_kernel[(B_size, N_size, S_size, S_size, H_size)](
            G, L_perm, M,
            B_size, N_size, S_size, H_size,
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
            L_perm.stride(0), L_perm.stride(1), L_perm.stride(2), L_perm.stride(3), L_perm.stride(4),
            M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
            num_warps=4, num_stages=2
        )

        # 6) Y[b,n,i,h] = sum_j M[b,n,i,j,h] * hidden_states[b,n,j,h,:]
        #    hidden_states: [B,N,S,H,D], Y: [B,N,S,H]
        Y = torch.empty((B_size, N_size, S_size, H_size), dtype=torch.float32, device=device)
        y_diag_reduce_kernel[(B_size, N_size, S_size, H_size)](
            M, hidden_states.to(torch.float32), Y,
            B_size, N_size, S_size, H_size, self.D_STATE,
            M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2), hidden_states.stride(3), hidden_states.stride(4),
            Y.stride(0), Y.stride(1), Y.stride(2), Y.stride(3),
            BLOCK_D=64,  # D=32, masked
            num_warps=4, num_stages=2
        )

        # Cast to bfloat16 to match original output dtype
        return Y.to(torch.bfloat16)
```

    A_stride_b, A_stride_h, A_stride_n, A_stride_s,
    L_stride_b, L_stride_h, L_stride_n, L_stride_s1, L_stride_s2,

    pid_h = tl.program_id(1)  # h

                l_val = 0.0
            tl.store(
                L_ptr + pid_b * L_stride_b + pid_h * L_stride_h + pid_n * L_stride_n + i * L_stride_s1 + j * L_stride_s2,
                l_val
            )


# Kernel 2: Compute G[i,j,h] = sum_d C[b,n,i,h,d] * B[b,n,j,h,d]
# Inputs: B [B,N,S,H,D], C [B,N,S,H,D], Output G [B,N,S,S,H], float32
@triton.jit
def g_contract_kernel(
    B_ptr, C_ptr, G_ptr,
    B_size, N_size, S_size, H_size, D_size,
    B_stride_b, B_stride_n, B_stride_s, B_stride_h, B_stride_d,
    C_stride_b, C_stride_n, C_stride_s, C_stride_h, C_stride_d,
    G_stride_b, G_stride_n, G_stride_s1, G_stride_s2, G_stride_h,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)  # b
    pid_n = tl.program_id(1)  # n
    pid_si = tl.program_id(2)  # i
    pid_sj = tl.program_id(3)  # j
    pid_h = tl.program_id(4)   # h

    acc = 0.0
    for d_start in range(0, D_size, BLOCK_D):
        d_idx = d_start + tl.arange(0, BLOCK_D)
        mask = d_idx < D_size
        b_vec = tl.load(B_ptr + pid_b * B_stride_b + pid_n * B_stride_n + pid_sj * B_stride_s + pid_h * B_stride_h + d_idx * B_stride_d, mask=mask, other=0.0)
        c_vec = tl.load(C_ptr + pid_b * C_stride_b + pid_n * C_stride_n + pid_si * C_stride_s + pid_h * C_stride_h + d_idx * C_stride_d, mask=mask, other=0.0)
        acc += tl.sum(b_vec * c_vec, axis=0)
    tl.store(G_ptr + pid_b * G_stride_b + pid_n * G_stride_n + pid_si * G_stride_s1 + pid_sj * G_stride_s2 + pid_h * G_stride_h, acc)


# Kernel 3: Elementwise M = G * L
# G: [B,N,S,S,H], L: [B,N,S,S,H], Output M: [B,N,S,S,H]
@triton.jit
def m_mul_kernel(
    G_ptr, L_ptr, M_ptr,
    B_size, N_size, S_size, H_size,
    G_stride_b, G_stride_n, G_stride_s1, G_stride_s2, G_stride_h,
    L_stride_b, L_stride_n, L_stride_s1, L_stride_s2, L_stride_h,
    M_stride_b, M_stride_n, M_stride_s1, M_stride_s2, M_stride_h,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)
    pid_sj = tl.program_id(3)
    pid_h = tl.program_id(4)

    g_val = tl.load(G_ptr + pid_b * G_stride_b + pid_n * G_stride_n + pid_si * G_stride_s1 + pid_sj * G_stride_s2 + pid_h * G_stride_h)
    l_val = tl.load(L_ptr + pid_b * L_stride_b + pid_n * L_stride_n + pid_si * L_stride_s1 + pid_sj * L_stride_s2 + pid_h * L_stride_h)
    m_val = g_val * l_val
    tl.store(M_ptr + pid_b * M_stride_b + pid_n * M_stride_n + pid_si * M_stride_s1 + pid_sj * M_stride_s2 + pid_h * M_stride_h, m_val)


# Kernel 4: Reduce Y[b,n,i,h] = sum_j M[b,n,i,j,h] * hidden_states[b,n,j,h,:]
# M: [B,N,S,S,H], hidden: [B,N,S,H,D], Output Y: [B,N,S,H] float32
@triton.jit
def y_diag_reduce_kernel(
    M_ptr, hidden_ptr, Y_ptr,
    B_size, N_size, S_size, H_size, D_size,
    M_stride_b, M_stride_n, M_stride_s1, M_stride_s2, M_stride_h,
    hidden_stride_b, hidden_stride_n, hidden_stride_s, hidden_stride_h, hidden_stride_d,
    Y_stride_b, Y_stride_n, Y_stride_s1, Y_stride_h,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)  # i
    pid_h = tl.program_id(3)   # h

    acc = 0.0
    for j in range(0, S_size):
        m_val = tl.load(M_ptr + pid_b * M_stride_b + pid_n * M_stride_n + pid_si * M_stride_s1 + j * M_stride_s2 + pid_h * M_stride_h)
        d_idx = tl.arange(0, BLOCK_D)
        mask = d_idx < D_size
        hidden_vec = tl.load(
            hidden_ptr + pid_b * hidden_stride_b + pid_n * hidden_stride_n + j * hidden_stride_s + pid_h * hidden_stride_h + d_idx * hidden_stride_d,
            mask=mask, other=0.0
        )
        acc += tl.sum(hidden_vec * m_val, axis=0)
    tl.store(Y_ptr + pid_b * Y_stride_b + pid_n * Y_stride_n + pid_si * Y_stride_s1 + pid_h * Y_stride_h, acc)

        """
        hidden_states: [B, N, S, H, D]
        A_cumsum:      [B, H, N, S]

        C:             [B, N, S, G, D]
        Output:        [B, N, S, H] in bfloat16

        B_size, N_size, S_size, H_size, D_size = hidden_states.shape

        A = A_cumsum.to(torch.float32).contiguous()       # [B,H,N,S]
        Bc = B.to(torch.float32).contiguous()             # [B,N,S,G,D]
        Cc = C.to(torch.float32).contiguous()             # [B,N,S,G,D]

        # Expand B/C from G to H (NUM_HEADS=32, G=N_GROUPS=8 => repeat_interleave by 4)
        if Bc.shape[3] != self.NUM_HEADS:
            Bc = Bc.repeat_interleave(self.NUM_HEADS // Bc.shape[3], dim=3)
        if Cc.shape[3] != self.NUM_HEADS:
            Cc = Cc.repeat_interleave(self.NUM_HEADS // Cc.shape[3], dim=3)

        )

            Cc.stride(0), Cc.stride(1), Cc.stride(2), Cc.stride(3), Cc.stride(4),
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),

            num_warps=4, num_stages=2
        )
