    A_ptr, A_masked_ptr,
    B_batch, B_heads, B_chunks,
    A_stride_b, A_stride_h, A_stride_n, A_stride_i, A_stride_j,
    Ams_stride_b, Ams_stride_n, Ams_stride_i, Ams_stride_j, Ams_stride_h,
    CHUNK: tl.constexpr
):

    h = tl.program_id(1)
    n = tl.program_id(2)
    i = tl.program_id(3)

    # We'll compute cumsum as a loop over j, but Triton does not support dynamic loops well here.
    # Instead, we use a host-side tensor and just write the masked value for each j, since cumsum here
    # is along the j-dimension with strict lower-triangular condition (j <= i).

    # will implement full L generation in Triton in the next step.
    # For now, just zero out upper triangle in masked buffer.

    out_stride_b, out_stride_n, out_stride_k, out_stride_h, out_stride_d,

    n = tl.program_id(1)
    i = tl.program_id(2)

            G_off = b * G_stride_b + n * G_stride_n + i * G_stride_k + j * G_stride_j + h * G_stride_h

            acc += G_val * L_val * hidden_vals
        out_off = b * out_stride_b + n * out_stride_n + i * out_stride_k + h * out_stride_h + d * out_stride_d
        tl.store(out_ptr + out_off, acc, mask=mask_d)

        )

import torch
import triton
import triton.language as tl

# Constants
CHUNK_SIZE = 128
NUM_HEADS = 32
N_GROUPS = 8
REPEAT = NUM_HEADS // N_GROUPS  # 4
STATE_SIZE = 64  # assumed in this benchmark setup
HEAD_DIM = 64  # output's last dimension

# Triton kernel: generate lower-triangular mask (diagonal=-1) as a 2D tensor of 0/1, then compute L via cumsum(exp).
# We need mask as an input to the kernel. Triton allows passing 1D/2D tensors via pointers; here we pass a 2D mask [CHUNK, CHUNK].
@triton.jit
def create_tril_minus1_mask_2d(mask_ptr, CHUNK: tl.constexpr):
    rows = tl.program_id(0)
    cols = tl.program_id(1)
    # Write 1 for j <= i (lower triangle including diagonal), 0 otherwise
    if cols <= rows:
        tl.store(mask_ptr + rows * CHUNK + cols, 1)
    else:
        tl.store(mask_ptr + rows * CHUNK + cols, 0)

# Triton kernel: compute masked cumsum along j, apply exp, and write L[b, h, n, i, j]
# We pass mask_ptr to avoid any torch tril in host code.
@triton.jit
def compute_L_tril_exp_kernel(
    A_ptr, mask_ptr, L_ptr,
    B_batch, B_heads, B_chunks,
    A_stride_b, A_stride_h, A_stride_n, A_stride_i, A_stride_j,
    L_stride_b, L_stride_n, L_stride_i, L_stride_j, L_stride_h,
    mask_stride_0, mask_stride_1,  # strides for mask [CHUNK, CHUNK]
    CHUNK: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    n = tl.program_id(2)
    # For each i, compute cumsum across j only for lower triangle (j <= i), others 0, then exp
    for i in range(CHUNK):
        acc = 0.0
        for j in range(CHUNK):
            # Load mask value
            mj = tl.load(mask_ptr + j * CHUNK + i)  # mask[i, j]
            # If mask is 1, load A; else contribute 0
            A_off = b * A_stride_b + h * A_stride_h + n * A_stride_n + i * A_stride_i + j * A_stride_j
            A_val = tl.load(A_ptr + A_off) if mj != 0 else 0.0
            acc += A_val
            L_off = b * L_stride_b + h * L_stride_h + n * L_stride_n + i * L_stride_i + j * L_stride_j
            tl.store(L_ptr + L_off, tl.exp(acc))
        # Upper triangle positions already stored as exp(0) == 1 from the loop, but we can explicitly set them to 1:
        for j in range(CHUNK):
            if j > i:
                L_off = b * L_stride_b + h * L_stride_h + n * L_stride_n + i * L_stride_i + j * L_stride_j
                tl.store(L_ptr + L_off, 1.0)

# Triton kernel: contract B and C into G: G[i, j, h] = sum over s of C[b, n, i, g, s] * B[b, n, j, g, s]
@triton.jit
def contract_BC_to_G_kernel(
    B_ptr, C_ptr, G_ptr,
    B_batch, B_n, B_K, B_groups, B_state,
    G_stride_b, G_stride_n, G_stride_k, G_stride_j, G_stride_h,
    B_stride_b, B_stride_n, B_stride_k, B_stride_g, B_stride_s,
    C_stride_b, C_stride_n, C_stride_k, C_stride_g, C_stride_s,
    CHUNK: tl.constexpr, REPEAT: tl.constexpr, BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    n = tl.program_id(1)
    h = tl.program_id(2)
    g = h // REPEAT
    for i in range(CHUNK):
        for j in range(CHUNK):
            acc = 0.0
            for s_start in range(0, B_state, BLOCK_S):
                s = s_start + tl.arange(0, BLOCK_S)
                mask_s = s < B_state
                # B[b, n, j, g, s]
                B_off = b * B_stride_b + n * B_stride_n + j * B_stride_k + g * B_stride_g + s * B_stride_s
                B_vals = tl.load(B_ptr + B_off, mask=mask_s, other=0.0)
                # C[b, n, i, g, s]
                C_off = b * C_stride_b + n * C_stride_n + i * C_stride_k + g * C_stride_g + s * C_stride_s
                C_vals = tl.load(C_ptr + C_off, mask=mask_s, other=0.0)
                acc += tl.sum(B_vals * C_vals, axis=0)
            G_off = b * G_stride_b + n * G_stride_n + i * G_stride_k + j * G_stride_j + h * G_stride_h
            tl.store(G_ptr + G_off, acc)

# Triton kernel: final reduction to compute Y_diag: out[b, n, i, h, d] = sum_j G[b, n, i, j, h] * L[b, n, i, j, h] * hidden[b, n, j, h, d]
# hidden is [B, N, CHUNK, H, D] with D=HEAD_DIM, here 64.
@triton.jit
def final_reduce_to_Ydiag_kernel(
    G_ptr, L_ptr, hidden_ptr, out_ptr,
    B_batch, B_n, B_K, B_heads, B_D,
    G_stride_b, G_stride_n, G_stride_k, G_stride_j, G_stride_h,
    L_stride_b, L_stride_h, L_stride_n, L_stride_k, L_stride_j,
    hidden_stride_b, hidden_stride_n, hidden_stride_k, hidden_stride_h, hidden_stride_d,
    out_stride_b, out_stride_n, out_stride_k, out_stride_h, out_stride_d,
    CHUNK: tl.constexpr, BLOCK_D: tl.constexpr
):
    b = tl.program_id(0)
    n = tl.program_id(1)
    i = tl.program_id(2)
    h = tl.program_id(3)
    for d_start in range(0, B_D, BLOCK_D):
        d = d_start + tl.arange(0, BLOCK_D)
        mask_d = d < B_D
        acc = tl.zeros([BLOCK_D], dtype=tl.float32)
        for j in range(CHUNK):
            G_off = b * G_stride_b + n * G_stride_n + i * G_stride_k + j * G_stride_j + h * G_stride_h
            G_val = tl.load(G_ptr + G_off)  # scalar
            L_off = b * L_stride_b + h * L_stride_h + n * L_stride_n + i * L_stride_k + j * L_stride_j
            L_val = tl.load(L_ptr + L_off)  # scalar
            hidden_off = b * hidden_stride_b + n * hidden_stride_n + j * hidden_stride_k + h * hidden_stride_h + d * hidden_stride_d
            hidden_vals = tl.load(hidden_ptr + hidden_off, mask=mask_d, other=0.0)
            acc += G_val * L_val * hidden_vals
        out_off = b * out_stride_b + n * out_stride_n + i * out_stride_k + h * out_stride_h + d * out_stride_d
        tl.store(out_ptr + out_off, acc, mask=mask_d)
