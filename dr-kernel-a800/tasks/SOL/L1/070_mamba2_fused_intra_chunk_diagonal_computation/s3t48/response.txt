    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    S: tl.constexpr,  # chunk_size (128)
    H: tl.constexpr,  # num_heads (32)
    num_warps=8, num_stages=2
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    # Loop over i (rows of output L), vectorize over j (columns)
    for i in range(S):
        j_vec = tl.arange(0, S)  # vector of j indices
        # Compute cumsum across j for each head h, then write L[i, j, h] with triangular mask
