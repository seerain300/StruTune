import triton
import triton.language as tl


@triton.jit
def compute_L_exp_cumsum_kernel(
    A_ptr, L_ptr,
    Bsz, Csz,
    stride_A_b, stride_A_h, stride_A_c, stride_A_s,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    S: tl.constexpr,  # chunk_size = 128
    H: tl.constexpr,  # num_heads = 32
    num_warps=4, num_stages=2
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    # For each i, build cumsum across j and write L[i, j, h] with lower-triangular mask (i >= j)
    for i in range(S):
        # cumsum vector s[j]

            s_j = 0.0
            for hh in range(H):
                a_off = b * stride_A_b + hh * stride_A_h + c * stride_A_c + j * stride_A_s
                a_val = tl.load(A_ptr + a_off)
                s_j += a_val

            for hh in range(H):
                l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + hh * stride_L_h
                l_val = tl.exp(s_j)

    - Set L[i, j, h] = exp(s[j]) if j <= i, else 0. This matches the original triangular mask (lower-triangular including diagonal) after cumsum and exp.
  - This is implemented as a single Triton program per (b, c), looping over i and j, and writing L values.

- Kernel 2: contract_BC_to_G_kernel
  - Input: B [B, C, S, G_CONST, K], C [B, C, S, G_CONST, K], Output: G [B, C, S, S, H]
  - Semantics: For each (b, c), loop over i and j and accumulate G[i, j, h] = sum over groups g and state k of C[b, c, i, g, k] * B[b, c, j, g, k].
  - This is straightforward Triton code with nested loops over groups and states; grid is (Bsz, Csz).

- Kernel 3: contract_M_with_hidden_kernel
  - Input: G [B, C, S, S, H], L [B, C, S, S, H], hidden_states [B, C, S, H, D], Output: Y_diag [B, C, S, H, D]
  - Semantics: For each output element (b, c, i, h, d), compute Y[b, c, i, h, d] = sum_j (G[b, c, i, j, h] * L[b, c, i, j, h]) * hidden[b, c, j, h, d].
  - This avoids torch einsum by performing the reduction in Triton.

2) Shape specialization
- The original model uses constants: S=128, H=32, D=32, G_CONST=8, K=32. The evaluation workloads provided use these axes. We will specialize kernels for these constants to ensure correctness and avoid dynamic loops that can cause runtime errors.
- ModelNew.forward asserts these shapes; if inputs don’t match, the code can be adjusted, but for the given evaluation they do.

3) Ensuring Triton-only computation
- No torch elementwise operations or einsum are used in ModelNew.forward. All computation is done in Triton kernels with explicit grid sizes. Outputs are allocated and filled by Triton kernels.

import torch
import triton
import triton.language as tl


@triton.jit
def compute_L_exp_cumsum_kernel(
    A_ptr, L_ptr,
    Bsz, Csz,
    stride_A_b, stride_A_h, stride_A_c, stride_A_s,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    S: tl.constexpr,  # chunk_size = 128
    H: tl.constexpr,  # num_heads = 32
    num_warps=4, num_stages=2
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    # For each i, build cumsum across j and write L[i, j, h] with lower-triangular mask (i >= j)
    for i in range(S):
        # cumsum vector s[j]
        for j in range(S):
            s_j = 0.0
            for hh in range(H):
                a_off = b * stride_A_b + hh * stride_A_h + c * stride_A_c + j * stride_A_s
                a_val = tl.load(A_ptr + a_off)
                s_j += a_val
            # Write exp(s_j) for all h if j <= i, else 0
            for hh in range(H):
                l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + hh * stride_L_h
                l_val = tl.exp(s_j)
                if j <= i:
                    tl.store(L_ptr + l_off, l_val)
                else:
                    tl.store(L_ptr + l_off, 0.0)


@triton.jit
def contract_BC_to_G_kernel(
    B_ptr, C_ptr, G_ptr,
    Bsz, Csz,
    stride_B_b, stride_B_c, stride_B_s, stride_B_g, stride_B_k,
    stride_C_b, stride_C_c, stride_C_s, stride_C_g, stride_C_k,
    stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h,
    S: tl.constexpr,  # 128
    G_CONST: tl.constexpr,  # 8
    K: tl.constexpr,  # 32
    num_warps=4, num_stages=2
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    for i in range(S):
        for j in range(S):
            acc = 0.0
            # Accumulate over groups and states
            for g in range(G_CONST):
                for k in range(K):
                    b_off = b * stride_B_b + c * stride_B_c + j * stride_B_s + g * stride_B_g + k * stride_B_k
                    c_off = b * stride_C_b + c * stride_C_c + i * stride_C_s + g * stride_C_g + k * stride_C_k
                    b_val = tl.load(B_ptr + b_off)
                    c_val = tl.load(C_ptr + c_off)
                    acc += b_val * c_val
            # Write acc to G[i, j, h] for all h
            for hh in range(H):
                g_off = b * stride_G_b + c * stride_G_c + i * stride_G_i + j * stride_G_j + hh * stride_G_h
                tl.store(G_ptr + g_off, acc)


@triton.jit
def contract_M_with_hidden_kernel(

    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,

    h = tl.program_id(3)
    d = tl.program_id(4)

    for j in range(S):
        g_off = b * stride_G_b + c * stride_G_c + i * stride_G_i + j * stride_G_j + h * stride_G_h
        l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + h * stride_L_h
