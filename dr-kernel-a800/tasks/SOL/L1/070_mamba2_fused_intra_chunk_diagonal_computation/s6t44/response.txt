```python
import torch
import triton
import triton.language as tl


# Kernel 1: Generate 1D lower-triangular mask (diagonal=-1) of length S*S into mask[0:S*S].
# mask[i*S + j] = 1 if j <= i, else 0. Output as int8 (1/0).
@triton.jit
def tril_mask_kernel(mask_ptr, S: tl.constexpr):
    idx = tl.program_id(0)  # linear index over S*S
    i = idx // S
    j = idx % S
    lower = j <= i
    # store 1 for True, 0 for False
    if lower:
        tl.store(mask_ptr + idx, 1)
    else:
        tl.store(mask_ptr + idx, 0)


# Kernel 2: Compute L[b,h,n,i,j] = exp(sum_{k=0..i} A[b,h,n,k]) if j <= i else 0.0
# A: [B,H,N,S] float32, L: [B,H,N,S,S] float32
@triton.jit
def a_segment_sum_exp_kernel(
    A_ptr, L_ptr,
    B_size, H_size, N_size, S_size,
    A_stride_b, A_stride_h, A_stride_n, A_stride_s,
    L_stride_b, L_stride_h, L_stride_n, L_stride_s1, L_stride_s2,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Loop over i and j; within i we maintain a running sum for segment sum at that i
    for i in range(S_size):
        sum_val = 0.0  # float32 accumulator
        # compute segment sum from k=0..i
        for k in range(i + 1):  # inclusive up to i
            a_off = pid_b * A_stride_b + pid_h * A_stride_h + pid_n * A_stride_n + k * A_stride_s
            a_val = tl.load(A_ptr + a_off)
            sum_val += a_val
        # Now fill L for all j <= i: L[b,h,n,i,j] = exp(sum_val), else 0.0
        # We can fill for all j in [0, S_size) but we must gate j <= i to match tril(diagonal=-1)
        for j in range(S_size):
            if j <= i:
                val = tl.exp(sum_val)
            else:
                val = 0.0
            l_off = pid_b * L_stride_b + pid_h * L_stride_h + pid_n * L_stride_n + i * L_stride_s1 + j * L_stride_s2
            tl.store(L_ptr + l_off, val)


# Kernel 3: Compute G[i,j,h] = sum_d B[b,n,j,h,d] * C[b,n,i,h,d]
# Inputs: B [B,N,S,H,D], C [B,N,S,H,D], Output: G [B,N,S,S,H] float32
@triton.jit
def g_contract_kernel(
    B_ptr, C_ptr, G_ptr,
    B_size, N_size, S_size, H_size, D_size,
    B_stride_b, B_stride_n, B_stride_s, B_stride_h, B_stride_d,
    C_stride_b, C_stride_n, C_stride_s, C_stride_h, C_stride_d,
    G_stride_b, G_stride_n, G_stride_s1, G_stride_s2, G_stride_h,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)  # i
    pid_sj = tl.program_id(3)  # j
    pid_h = tl.program_id(4)

    # Accumulator for sum over D
    acc = 0.0
    # Loop over D in tiles
    for d_start in range(0, D_size, BLOCK_D):
        d_offsets = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_offsets < D_size
        # Load B[b,n,j,h,d]
        b_off = (
            pid_b * B_stride_b
            + pid_n * B_stride_n
            + pid_sj * B_stride_s
            + pid_h * B_stride_h
            + d_offsets * B_stride_d
        )
        b_vals = tl.load(B_ptr + b_off, mask=mask_d, other=0.0)
        # Load C[b,n,i,h,d]
        c_off = (
            pid_b * C_stride_b
            + pid_n * C_stride_n
            + pid_si * C_stride_s
            + pid_h * C_stride_h
            + d_offsets * C_stride_d
        )
        c_vals = tl.load(C_ptr + c_off, mask=mask_d, other=0.0)
        # Multiply and reduce over this tile
        acc += tl.sum(b_vals * c_vals, axis=0)

    # Store G[b,n,i,j,h] = acc
    g_off = (
        pid_b * G_stride_b
        + pid_n * G_stride_n
        + pid_si * G_stride_s1
        + pid_sj * G_stride_s2
        + pid_h * G_stride_h
    )
    tl.store(G_ptr + g_off, acc)


# Kernel 4: Elementwise M = G * L, with lower-triangular gate (diagonal=-1).
# G: [B,N,S,S,H], L: [B,N,S,S,H], M: [B,N,S,S,H]
@triton.jit
def m_mul_kernel(
    G_ptr, L_ptr, M_ptr,
    B_size, N_size, S_size, H_size,
    G_stride_b, G_stride_n, G_stride_s1, G_stride_s2, G_stride_h,
    L_stride_b, L_stride_n, L_stride_s1, L_stride_s2, L_stride_h,
    M_stride_b, M_stride_n, M_stride_s1, M_stride_s2, M_stride_h,
    MASK_ptr,  # 1D mask of size S*S (int8), lower[i*S + j] = 1 if j <= i
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)  # i
    pid_sj = tl.program_id(3)  # j
    pid_h = tl.program_id(4)

    # Read mask for (i,j): linear index idx = si*S + sj
    idx = pid_si * S_size + pid_sj
    mask_val = tl.load(MASK_ptr + idx)  # int8
    lower = mask_val != 0

    # Load G[b,n,i,j,h] and L[b,n,i,j,h]
    g_off = (
        pid_b * G_stride_b
        + pid_n * G_stride_n
        + pid_si * G_stride_s1
        + pid_sj * G_stride_s2
        + pid_h * G_stride_h
    )
    l_off = (
        pid_b * L_stride_b
        + pid_n * L_stride_n
        + pid_si * L_stride_s1
        + pid_sj * L_stride_s2
        + pid_h * L_stride_h
    )
    g_val = tl.load(G_ptr + g_off)
    l_val = tl.load(L_ptr + l_off)

    # Apply gate: if j > i (upper triangle), set M to 0; else multiply
    if lower:
        m_val = g_val * l_val
    else:
        m_val = 0.0
    m_off = (
        pid_b * M_stride_b
        + pid_n * M_stride_n
        + pid_si * M_stride_s1
        + pid_sj * M_stride_s2
        + pid_h * M_stride_h
    )
    tl.store(M_ptr + m_off, m_val)


# Kernel 5: Reduce Y[b,n,i,h] = sum_j M[b,n,i,j,h] * hidden[b,n,j,h,:]
# Inputs: M [B,N,S,S,H], hidden [B,N,S,H,D], Output: Y [B,N,S,H] float32
@triton.jit
def y_diag_reduce_kernel(
    M_ptr, hidden_ptr, Y_ptr,
    B_size, N_size, S_size, H_size, D_size,
    M_stride_b, M_stride_n, M_stride_s1, M_stride_s2, M_stride_h,
    hidden_stride_b, hidden_stride_n, hidden_stride_s, hidden_stride_h, hidden_stride_d,
    Y_stride_b, Y_stride_n, Y_stride_s1, Y_stride_h,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_si = tl.program_id(2)  # i
    pid_h = tl.program_id(3)

    acc = 0.0
    # Loop over j in tiles
    for j_start in range(0, S_size, 1):  # j loop is fine; we can tile over D
        # For each j, load M[b,n,i,j,h] and sum over hidden[b,n,j,h,:] to form dot
        # We can do a scalar loop over j since S is small (128), but we use BLOCK_D for D accumulation.
        # However, Y reduction only needs per j contribution; we'll compute dot per j and add to acc.
        # Better: keep acc vector per (b,n,i,h) across j? Here, we accumulate into scalar acc across j.
        # Implementing per j accumulation:
        for j in range(S_size):
            m_off = (
                pid_b * M_stride_b
                + pid_n * M_stride_n
                + pid_si * M_stride_s1
                + j * M_stride_s2
                + pid_h * M_stride_h
            )
            m_val = tl.load(M_ptr + m_off)

            h_off = (
                pid_b * hidden_stride_b
                + pid_n * hidden_stride_n
                + j * hidden_stride_s
                + pid_h * hidden_stride_h
            )
            # h_off points to [B,N,S,H,1] but hidden is [B,N,S,H,D]; we need to load hidden[b,n,j,h,:] vector
            # We'll read a vector along D dimension by iterating BLOCK_D
            acc_j = 0.0
            for d_start in range(0, D_size, BLOCK_D):
                d_offsets = d_start + tl.arange(0, BLOCK_D)
                mask_d = d_offsets < D_size
                # hidden[b,n,j,h,d_offsets]
                h_vec_ptr = hidden_ptr + h_off + d_offsets * hidden_stride_d
                h_vec = tl.load(h_vec_ptr, mask=mask_d, other=0.0)
                # M_val is scalar; multiply broadcast
                acc_j += tl.sum(h_vec * m_val, axis=0)

            acc += acc_j

    # Store Y[b,n,i,h] = acc
    y_off = pid_b * Y_stride_b + pid_n * Y_stride_n + pid_si * Y_stride_s1 + pid_h * Y_stride_h
    tl.store(Y_ptr + y_off, acc)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # Constants from the original code
        self.NUM_HEADS = 32
        self.D_STATE = 32  # state size in B/C tensors; assumed 32 as in original

    def forward(self, hidden_states: torch.Tensor, A_cumsum: torch.Tensor, B: torch.Tensor, C: torch.Tensor):
        """
        hidden_states: [B, N, S, H, D]
        A_cumsum: [B, H, N, S]
        B, C: [B, N, S, G, D], with G=N_GROUPS=8
        Returns Y_diag: [B, N, S, H] (cast to bfloat16)
        """
        device = hidden_states.device
        B_size, N_size, S_size, H_size, D_size = hidden_states.shape

        # Ensure inputs are contiguous and float32 for computation
        A = A_cumsum.to(torch.float32).contiguous()            # [B,H,N,S]
        hidden = hidden_states.to(torch.float32).contiguous()  # [B,N,S,H,D]

        # Expand B/C from G to H (NUM_HEADS=32, G=N_GROUPS=8 => repeat_interleave by 4)
        # Note: in the original, B/C are expanded on host to H before forward; here we perform it:
        if B.shape[3] != self.NUM_HEADS:
            B = B.repeat_interleave(self.NUM_HEADS // B.shape[3], dim=3)
        if C.shape[3] != self.NUM_HEADS:
            C = C.repeat_interleave(self.NUM_HEADS // C.shape[3], dim=3)
        B = B.contiguous()
        C = C.contiguous()

        # 1) Compute tril mask (diagonal=-1) for chunk dimension SxS
        mask_1d = torch.empty(S_size * S_size, dtype=torch.int8, device=device)
        tril_mask_kernel[(S_size * S_size,)](mask_1d, S=S_size)

        # 2) Compute L[b,h,n,i,j] = exp(sum_{k=0..i} A[b,h,n,k]) if j <= i else 0.0 via Triton
        L = torch.empty((B_size, N_size, S_size, S_size), dtype=torch.float32, device=device)
        a_segment_sum_exp_kernel[(B_size, H_size, N_size)](
            A, L,
            B_size, H_size, N_size, S_size,
            A.stride(0), A.stride(1), A.stride(2), A.stride(3),
            L.stride(0), L.stride(1), L.stride(2), L.stride(3), L.stride(4),
            num_warps=1, num_stages=1
        )
        # Broadcast L to [B,N,S,S,H] by adding H dimension
        L4 = L.unsqueeze(-1).expand(B_size, N_size, S_size, S_size, H_size).contiguous()

        # 3) Compute G[i,j,h] via Triton contraction over D
        G = torch.empty((B_size, N_size, S_size, S_size, H_size), dtype=torch.float32, device=device)
        g_contract_kernel[(B_size, N_size, S_size, S_size, H_size)](
            B, C, G,
            B_size, N_size, S_size, H_size, self.D_STATE,
            B.stride(0), B.stride(1), B.stride(2), B.stride(3), B.stride(4),
            C.stride(0), C.stride(1), C.stride(2), C.stride(3), C.stride(4),
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
            BLOCK_D=64,
            num_warps=4, num_stages=2
        )

        # 4) Elementwise multiply M = G * L (with lower-triangular gate)
        M = torch.empty_like(G, dtype=torch.float32, device=device)
        m_mul_kernel[(B_size, N_size, S_size, S_size, H_size)](
            G, L4, M,
            B_size, N_size, S_size, H_size,
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
            L4.stride(0), L4.stride(1), L4.stride(2), L4.stride(3), L4.stride(4),
            M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
            mask_1d,  # pass 1D mask
            num_warps=1, num_stages=1
        )

        # 5) Reduce to Y[b,n,i,h] via Triton kernel (launch)
        Y = torch.empty((B_size, N_size, S_size, H_size), dtype=torch.float32, device=device)
        y_diag_reduce_kernel[(B_size, N_size, S_size, H_size)](

            B_size, N_size, S_size, H_size, D_size,
            M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
            hidden.stride(0), hidden.stride(1), hidden.stride(2), hidden.stride(3), hidden.stride(4),
            Y.stride(0), Y.stride(1), Y.stride(2), Y.stride(3),
            BLOCK_D=64,
            num_warps=4, num_stages=2
        )

        # Cast to bfloat16 to match original output dtype
        return Y.to(torch.bfloat16)
```
