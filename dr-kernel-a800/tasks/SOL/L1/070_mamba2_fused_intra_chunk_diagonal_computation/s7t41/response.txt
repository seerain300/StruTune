import triton
import triton.language as tl


@triton.jit
def masked_cumsum_lower_exp(
    A_ptr,  # *float32, shape [B, C, S, N] (we'll pass contiguous)
    L_ptr,  # *float32, shape [B, C, S, S, N] (we'll allocate and fill)
    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, N: tl.constexpr,
    A_stride_b, A_stride_c, A_stride_j, A_stride_n,
    L_stride_b, L_stride_c, L_stride_i, L_stride_j, L_stride_n,
):
    # Each program handles one (b, c, i)
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)

    # Running sum vector of length N (num_heads)
    running = tl.zeros((N,), dtype=tl.float32)

    # Loop over j from 0 to S-1; masked by (j < i)
    for j in range(0, S):
        include = j < i
        # Load A[b, c, j, n] for all n; create index vector for n
        n_idx = tl.arange(0, N)
        a_offsets = (
            b * A_stride_b
            + c * A_stride_c
            + j * A_stride_j
            + n_idx * A_stride_n
        )
        a_vals = tl.load(A_ptr + a_offsets)
        # If j >= i, set to 0; else keep
        a_vals = tl.where(include, a_vals, tl.zeros_like(a_vals))
        running += a_vals

        # Store L[b, c, i, j, n] = exp(running)
        l_offsets = (
            b * L_stride_b
            + c * L_stride_c
            + i * L_stride_i
            + j * L_stride_j
            + n_idx * L_stride_n
        )
        tl.store(L_ptr + l_offsets, tl.exp(running))

    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, N: tl.constexpr, K: tl.constexpr,

    C_stride_b, C_stride_c, C_stride_i, C_stride_n, C_stride_k,

):

    c = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    n = tl.program_id(4)

    # Accumulate sum over k
    acc = tl.zeros((), dtype=tl.float32)

        bval = tl.load(
            B_ptr
            + b * B_stride_b
            + c * B_stride_c
            + j * B_stride_j
            + n * B_stride_n
            + k * B_stride_k
        )
        cval = tl.load(
            C_ptr
            + b * C_stride_b
            + c * C_stride_c
            + i * C_stride_i
            + n * C_stride_n
            + k * C_stride_k

        acc += bval * cval

    # Store G[b, c, i, j, n] = acc
    g_offsets = (
        b * G_stride_b
        + c * G_stride_c
        + i * G_stride_i
        + j * G_stride_j
        + n * G_stride_n
    )

    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, N: tl.constexpr,
    A_stride_b, A_stride_c, A_stride_j, A_stride_n,

):
