import triton

@triton.jit
def compute_L_exp_cumsum_kernel(
    A_ptr, L_ptr,
    Bsz, Csz,
    stride_A_b, stride_A_h, stride_A_c, stride_A_s,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    S: tl.constexpr,  # chunk_size (128)
    H: tl.constexpr,  # num_heads (32)
    num_warps=8, num_stages=2
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    # For each i, compute cumsum across j and write L[i, j, h] = exp(cumsum[j]) for all h
    for i in range(S):
        # Compute cumsum vector s[j] = sum_{t <= j} A[b, :, c, t]
        s = [0.0] * S  # vector of size S
        for j in range(S):
            s_j = 0.0
            # sum over cumsum axis (j index t <= j)
            for t in range(j + 1):
                # A[b, :, c, t] sums across heads h
                for h in range(H):

                    a_val = tl.load(A_ptr + a_off)
                    s_j += a_val

        for j in range(S):
