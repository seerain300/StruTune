import triton
import triton.language as tl


@triton.jit
def l_masked_cumsum_exp_kernel(
    A_ptr,  # pointer to A_cumsum: [B, Csz, S, N]
    L_ptr,  # pointer to L: [B, Csz, S, S, N], float32
    B_size, C_size, S, N,
    A_stride_b, A_stride_c, A_stride_j, A_stride_n,
    L_stride_b, L_stride_c, L_stride_i, L_stride_j, L_stride_n,
    BLOCK_S: tl.constexpr
):
    # program ids for b, c, n
    b = tl.program_id(0)
    c = tl.program_id(1)
    n = tl.program_id(2)

    # we will iterate over i (the "i" index in L)
    # Triton launches one program per (b, c, n); within the program, we compute for all i
    # since BLOCK_S can be >= S, we can compute all i in a single pass
    i_idx = 0
    while i_idx < S:
        # running sum for cumsum along j
        running = tl.zeros((), dtype=tl.float32)

        j = 0
        while j < S:
            # mask: include j < i, exclude j == i (diagonal=-1). Note: i is scalar here; j is scalar.
            include = j < i_idx  # scalar boolean
            a_val = tl.load(A_ptr + b * A_stride_b + c * A_stride_c + j * A_stride_j + n * A_stride_n)
            a_val = a_val.to(tl.float32)
            if include:
                running += a_val
            # else: exclude j==i, so not adding

            # After loop: running is the cumsum up to j. But we need cumsum up to i (exclusive). So we need running_i_minus1.
            # We'll implement: running_i = running at j=i (but we excluded j==i). We need to produce L[i, j] = exp(cumsum up to i-1).
            # However, we don't have L[i, j] here inside this while; we must store L[i, j] = exp(running) only for j < i.
            # For j >= i, L[i, j] = 1.0 (since the original code applies tril(diagonal=-1) and exp to the masked cumsum; here we are building L elementwise).
            # Compute whether to store: include implies j < i. For j >= i, store 1.0.
            l_val = tl.exp(running) if include else 1.0
            # store to L[b, c, i, j, n]
            tl.store(L_ptr + b * L_stride_b + c * L_stride_c + i_idx * L_stride_i + j * L_stride_j + n * L_stride_n, l_val)
            j += 1
        i_idx += 1

):

    c = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)

        hidden_states_f32 = hidden_states.to(torch.float32)

        L = torch.empty((Bsz, Csz, S, S, N), device=hidden_states.device, dtype=torch.float32)

        A_stride_b, A_stride_c, A_stride_j, A_stride_n = A_cumsum_f32.stride()

            BLOCK_S=S,
            num_warps=1, num_stages=1
        )

        G = torch.empty((Bsz, Csz, S, S, N), device=hidden_states.device, dtype=torch.float32)

        B_stride_b, B_stride_c, B_stride_j, B_stride_n, B_stride_k = B_expanded_f32.stride()

            BLOCK_K=K,
            num_warps=1, num_stages=1
        )

        # Elementwise M = G * L (PyTorch for simplicity; heavy numeric work is already Triton above)
        M = G * L  # float32

        # Allocate Y: [B, Csz, S, N, D], float32
        Y = torch.empty((Bsz, Csz, S, N, D), device=hidden_states.device, dtype=torch.float32)

        grid_Y = (Bsz, Csz, S, N, D)
        diag_contract_Y_kernel[grid_Y](

            Bsz, Csz, S, N, D,

            Y_stride_b, Y_stride_c, Y_stride_i, Y_stride_n, Y_stride_d,
            BLOCK_J=S,
            num_warps=1, num_stages=1
        )

        # Cast to bfloat16 to match original signature
        return Y.to(torch.bfloat16)
