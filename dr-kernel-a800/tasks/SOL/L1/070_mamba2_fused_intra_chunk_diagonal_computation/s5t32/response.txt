import torch
import triton
import triton.language as tl


@triton.jit
def build_lower_tri_causal_kernel(
    A_ptr,                  # *float32, shape [B, C, L, H]
    L_ptr,                  # *float32, shape [B, C, 128, 128, H]
    B_size, C_size, H_size, L_size,
    stride_A_b, stride_A_c, stride_A_l, stride_A_h,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h
):
    # 5D grid: (b, c, h, i, j)
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)  # row index in 128
    j = tl.program_id(4)  # col index in 128

    # Only keep lower-triangular (diagonal=-1): j <= i
    keep = j <= i

    # Compute cumulative sum along L for this (b, c, h): sum_{k=0..L-1} A[b, c, k, h]
    # Use k in 0..L-1 and accumulate in fp32
    cumsum = 0.0
    for k in range(0, L_size):
        a_val = tl.load(A_ptr + b * stride_A_b + c * stride_A_c + k * stride_A_l + h * stride_A_h)
        cumsum += a_val

    total = cumsum
    # L[i, j] = exp(total) if j <= i, else 0
    val = tl.where(keep, tl.exp(total), 0.0)

    # Store to L_ptr[b, c, i, j, h]
    L_offset = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + h * stride_L_h
    tl.store(L_ptr + L_offset, val)


@triton.jit
def expand_groups_repeat_interleave(
    X_ptr,                  # *float32, shape [B, C, L, G, S] input (B/C/L/G/S)
    Y_ptr,                  # *float32, shape [B, C, L, H, S] output (B/C/L/H/S)
    B_size, C_size, L_size, G_size, S_size, H_size,
    stride_X_b, stride_X_c, stride_X_l, stride_X_g, stride_X_s,
    stride_Y_b, stride_Y_c, stride_Y_l, stride_Y_h, stride_Y_s,
    group_expand  # int, e.g., 4
):
    # 5D grid: (b, c, l, g, s) -> write to h = g * group_expand
    b = tl.program_id(0)
    c = tl.program_id(1)
    l = tl.program_id(2)
    g = tl.program_id(3)
    s = tl.program_id(4)

    h = g * group_expand
    if h >= H_size:
        return

    x_offset = b * stride_X_b + c * stride_X_c + l * stride_X_l + g * stride_X_g + s * stride_X_s
    y_offset = b * stride_Y_b + c * stride_Y_c + l * stride_Y_l + h * stride_Y_h + s * stride_Y_s

    x_val = tl.load(X_ptr + x_offset)
    tl.store(Y_ptr + y_offset, x_val)


@triton.jit
def compute_G_outer_prod_reduce_S(
    B_exp_ptr,             # *float32, shape [B, C, L, H, S]
    C_exp_ptr,             # *float32, shape [B, C, L, H, S]
    G_ptr,                 # *float32, shape [B, C, L, L, H]
    B_size, C_size, L_size, H_size, S_size,
    stride_B_b, stride_B_c, stride_B_l, stride_B_h, stride_B_s,
    stride_C_b, stride_C_c, stride_C_l, stride_C_h, stride_C_s,
    stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h
):
    # 5D grid: (b, c, i, j, h) -> compute G[b, c, i, j, h] = sum_s B[b, c, j, h, s] * C[b, c, i, h, s]
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)

    acc = 0.0
    for s in range(0, S_size):
        b_val = tl.load(B_exp_ptr + b * stride_B_b + c * stride_B_c + j * stride_B_l + h * stride_B_h + s * stride_B_s)
        c_val = tl.load(C_exp_ptr + b * stride_C_b + c * stride_C_c + i * stride_C_l + h * stride_C_h + s * stride_C_s)
        acc += b_val * c_val

    g_offset = b * stride_G_b + c * stride_G_c + i * stride_G_i + j * stride_G_j + h * stride_G_h
    tl.store(G_ptr + g_offset, acc)


@triton.jit
def multiply_mask(
    G_ptr,                 # *float32, shape [B, C, L, L, H]
    L_ptr,                 # *float32, shape [B, C, 128, 128, H]
    M_ptr,                 # *float32, shape [B, C, L, L, H]
    B_size, C_size, L_size, H_size,
    stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h
):
    # 5D grid: (b, c, i, j, h)
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)

    # Load G and L, compute element-wise product
    g_offset = b * stride_G_b + c * stride_G_c + i * stride_G_i + j * stride_G_j + h * stride_G_h
    l_offset = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + h * stride_L_h

    g_val = tl.load(G_ptr + g_offset)
    l_val = tl.load(L_ptr + l_offset)
    m_val = g_val * l_val

    m_offset = b * stride_M_b + c * stride_M_c + i * stride_M_i + j * stride_M_j + h * stride_M_h
    tl.store(M_ptr + m_offset, m_val)


@triton.jit
def reduce_j_contract_hidden(
    M_ptr,                 # *float32, shape [B, C, L, L, H]
    hidden_ptr,            # *float32, shape [B, C, L, H, D]
    Y_ptr,                 # *float32, shape [B, C, L, H, D]
    B_size, C_size, L_size, H_size, D_size,
    stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h,
    stride_H_b, stride_H_c, stride_H_l, stride_H_h, stride_H_d,
    stride_Y_b, stride_Y_c, stride_Y_i, stride_Y_h, stride_Y_d
):
    # 5D grid: (b, c, i, h, d) -> reduce over j in 0..L_size-1
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    h = tl.program_id(3)
    d = tl.program_id(4)

    acc = 0.0
    for j in range(0, L_size):
        m_offset = b * stride_M_b + c * stride_M_c + i * stride_M_i + j * stride_M_j + h * stride_M_h
        h_offset = b * stride_H_b + c * stride_H_c + j * stride_H_l + h * stride_H_h + d * stride_H_d
        m_val = tl.load(M_ptr + m_offset)
        h_val = tl.load(hidden_ptr + h_offset)
        acc += m_val * h_val

    y_offset = b * stride_Y_b + c * stride_Y_c + i * stride_Y_i + h * stride_Y_h + d * stride_Y_d
    tl.store(Y_ptr + y_offset, acc)


def _compute_and_launch_triton(hidden: torch.Tensor,
                                A_cumsum: torch.Tensor,
                                B: torch.Tensor,
                                C: torch.Tensor) -> torch.Tensor:
    """
    Entire computation done via Triton kernels. Returns Y_diag [B, C, L, H, D] in float32,
    which will be converted to bfloat16 at the end to match the original code.
    """
    # Shapes
    batch_size, num_chunks, L, num_heads, head_dim = hidden.shape
    # Constants
    NUM_HEADS = 32
    N_GROUPS = 8
    GROUP_EXPAND = 4
    L_mat_size = 128  # as in original code

    # Cast inputs to float32 for Triton computation
    hidden_f32 = hidden.to(torch.float32)
    A_f32 = A_cumsum.to(torch.float32)
    B_f32 = B.to(torch.float32)
    C_f32 = C.to(torch.float32)

    # 1) Build L matrix [B, C, 128, 128, H] in Triton
    L_f32 = torch.empty((batch_size, num_chunks, L_mat_size, L_mat_size, num_heads), dtype=torch.float32, device=hidden_f32.device)

    # Launch Triton kernel: grid=(B, C, H, 128, 128)
    grid_L = (batch_size, num_chunks, num_heads, L_mat_size, L_mat_size)
    build_lower_tri_causal_kernel[grid_L](
        A_f32, L_f32,
        batch_size, num_chunks, num_heads, L,
        A_f32.stride(0), A_f32.stride(1), A_f32.stride(2), A_f32.stride(3),
        L_f32.stride(0), L_f32.stride(1), L_f32.stride(2), L_f32.stride(3), L_f32.stride(4)
    )

    # 2) Expand B and C from groups (G=N_GROUPS) to heads (H=NUM_HEADS) via repeat_interleave GROUP_EXPAND=4
    S_size = B_f32.shape[4]
    B_exp = torch.empty((batch_size, num_chunks, L, NUM_HEADS, S_size), dtype=torch.float32, device=hidden_f32.device)
    C_exp = torch.empty((batch_size, num_chunks, L, NUM_HEADS, S_size), dtype=torch.float32, device=hidden_f32.device)

    grid_expand = (batch_size, num_chunks, L, N_GROUPS, S_size)
    expand_groups_repeat_interleave[grid_expand](
        B_f32, B_exp,
        batch_size, num_chunks, L, N_GROUPS, S_size, NUM_HEADS,
        B_f32.stride(0), B_f32.stride(1), B_f32.stride(2), B_f32.stride(3), B_f32.stride(4),
        B_exp.stride(0), B_exp.stride(1), B_exp.stride(2), B_exp.stride(3), B_exp.stride(4),
        GROUP_EXPAND
    )
    expand_groups_repeat_interleave[grid_expand](
        C_f32, C_exp,
        batch_size, num_chunks, L, N_GROUPS, S_size, NUM_HEADS,
        C_f32.stride(0), C_f32.stride(1), C_f32.stride(2), C_f32.stride(3), C_f32.stride(4),
        C_exp.stride(0), C_exp.stride(1), C_exp.stride(2), C_exp.stride(3), C_exp.stride(4),
        GROUP_EXPAND
    )

    # 3) Compute G[b, c, i, j, h] = sum_s C_exp[b, c, i, h, s] * B_exp[b, c, j, h, s] in Triton
    G = torch.empty((batch_size, num_chunks, L, L, num_heads), dtype=torch.float32, device=hidden_f32.device)
    grid_G = (batch_size, num_chunks, L, L, num_heads)
    compute_G_outer_prod_reduce_S[grid_G](
        B_exp, C_exp, G,
        batch_size, num_chunks, L, num_heads, S_size,
        B_exp.stride(0), B_exp.stride(1), B_exp.stride(2), B_exp.stride(3), B_exp.stride(4),
        C_exp.stride(0), C_exp.stride(1), C_exp.stride(2), C_exp.stride(3), C_exp.stride(4),
        G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4)
    )

    # 4) Apply mask: M = G * L element-wise in Triton
    M = torch.empty((batch_size, num_chunks, L, L, num_heads), dtype=torch.float32, device=hidden_f32.device)
    grid_M = (batch_size, num_chunks, L, L, num_heads)
    multiply_mask[grid_M](
        G, L_f32, M,
        batch_size, num_chunks, L, num_heads,
        G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
        L_f32.stride(0), L_f32.stride(1), L_f32.stride(2), L_f32.stride(3), L_f32.stride(4),
        M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4)
    )

    # 5) Compute Y_diag: contraction over j -> Y[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden[b, c, j, h, d]
    # hidden_f32 shape: [B, C, L, H, D]
    Y = torch.empty((batch_size, num_chunks, L, num_heads, head_dim), dtype=torch.float32, device=hidden_f32.device)

    grid_reduce = (batch_size, num_chunks, L, num_heads, head_dim)
    reduce_j_contract_hidden[grid_reduce](
        M, hidden_f32, Y,
        batch_size, num_chunks, L, num_heads, head_dim,
        M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
        hidden_f32.stride(0), hidden_f32.stride(1), hidden_f32.stride(2), hidden_f32.stride(3), hidden_f32.stride(4),
        Y.stride(0), Y.stride(1), Y.stride(2), Y.stride(3), Y.stride(4)
    )

    # Return as bfloat16 to match original
    return Y.to(torch.bfloat16)
