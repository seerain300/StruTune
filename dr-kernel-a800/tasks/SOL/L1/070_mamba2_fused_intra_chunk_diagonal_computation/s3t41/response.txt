        # Output: [B, C, S, H, D]
        Y_diag = torch.empty((Bsz, Csz, S, H, D), dtype=torch.float32, device=hidden_states.device)
        for b in range(Bsz):
            for c in range(Csz):
                for i in range(S):
                    for h in range(H):
                        acc = 0.0
                        for j in range(S):
                            m_val = M[b, c, i, j, h]
                            hs_val = hidden_states[b, c, j, h, :]
                            acc += m_val * torch.sum(hs_val)
                        Y_diag[b, c, i, h, :] = acc

        return Y_diag.to(torch.bfloat16)
