import torch
import triton
import triton.language as tl

# Compile-time constants for the given model
CHUNK_SIZE: tl.constexpr = 128
NUM_HEADS: tl.constexpr = 32
HEAD_DIM: tl.constexpr = 128
N_GROUPS: tl.constexpr = 8

# Triton kernels

# 1) Build L: L[b, h, i, k, j] = exp(cumsum(A[b, h, i, 0:j])) if i >= j else 0
@triton.jit
def build_L_kernel(
    A_ptr,                # *float32, [B, NUM_HEADS, NUM_CHUNKS, CHUNK_SIZE]
    L_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,      # int32: batch_size and num_chunks (not used here)
):
    b = tl.program_id(0)
    c = tl.program_id(1)  # i (source chunk)
    h = tl.program_id(2)  # head
    # Loop over k (destination position within chunk)
    for k in range(CHUNK_SIZE):
        # Compute cumsum along j for A[b, h, c, :]
        cumsum = 0.0
        for j in range(CHUNK_SIZE):
            # Load A[b, h, c, j]
            a_off = b * (NUM_HEADS * NC_size * CHUNK_SIZE) + h * (NC_size * CHUNK_SIZE) + c * CHUNK_SIZE + j
            a_val = tl.load(A_ptr + a_off)
            cumsum += a_val
            # Store L[b, c, k, j, h] = exp(cumsum) if k >= j else 0
            if k >= j:
                L_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
                tl.store(L_ptr + L_off, tl.exp(cumsum))
            else:
                L_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
                tl.store(L_ptr + L_off, 0.0)

# 2) Compute G[i, j, h] = sum_s C_expanded[i, k, h, s] * B_expanded[j, k, h, s]
@triton.jit
def compute_G_kernel(
    B_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM] (expanded)
    C_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM] (expanded)
    G_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,      # int32: batch_size and num_chunks (not used here)
):
    b = tl.program_id(0)
    c = tl.program_id(1)  # i (source chunk)
    k = tl.program_id(2)  # destination position
    h = tl.program_id(3)  # head
    # Accumulate G[i=k, j, h]
    for j in range(CHUNK_SIZE):
        G_val = 0.0
        # Sum over state dimension (HEAD_DIM == 128)
        for s in range(HEAD_DIM):
            B_off = b * (NC_size * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + c * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + s
            C_off = b * (NC_size * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + c * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + j * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + s
            B_val = tl.load(B_ptr + B_off)
            C_val = tl.load(C_ptr + C_off)
            G_val += C_val * B_val
        G_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
        tl.store(G_ptr + G_off, G_val)

# 3) Multiply M = G * L with mask k >= j
@triton.jit
def multiply_LG_kernel(
    G_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    L_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    M_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,      # int32: batch_size and num_chunks (not used here)
):
    b = tl.program_id(0)
    c = tl.program_id(1)  # i
    k = tl.program_id(2)  # destination position
    h = tl.program_id(3)  # head
    for j in range(CHUNK_SIZE):
        G_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
        L_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
        G_val = tl.load(G_ptr + G_off)
        L_val = tl.load(L_ptr + L_off)
        if k >= j:
            M_val = G_val * L_val
        else:
            M_val = 0.0
        M_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
        tl.store(M_ptr + M_off, M_val)

# 4) Contract to Y_diag: Y[b, i, k, h, d] = sum_j M[b, i, k, j, h] * hidden[b, i, k, j, h, d]
@triton.jit
def contract_to_Ydiag_kernel(
    M_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    hidden_ptr,           # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    Y_ptr,                # *bfloat16, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    B_size, NC_size,      # int32: batch_size and num_chunks
):
    b = tl.program_id(0)
    c = tl.program_id(1)  # i
    k = tl.program_id(2)  # destination position
    h = tl.program_id(3)  # head
    d = tl.program_id(4)  # head_dim index
    Y_val = 0.0
    for j in range(CHUNK_SIZE):
        M_off = b * (NC_size * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + c * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h
        M_val = tl.load(M_ptr + M_off)
        h_off = b * (NC_size * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + c * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + j
        hidden_val = tl.load(hidden_ptr + h_off)
        Y_val += M_val * hidden_val
    # Store as bfloat16
    Y_off = b * (NC_size * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + c * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + d
    tl.store(Y_ptr + Y_off, tl.cast(Y_val, tl.bfloat16))

# ModelNew: entry point, must launch Triton kernels
class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
                A_cumsum: torch.Tensor,
                B: torch.Tensor,
                C: torch.Tensor) -> torch.Tensor:
        # Ensure inputs are CUDA and contiguous
        device = hidden_states.device
        assert device.type == 'cuda', "ModelNew requires CUDA tensors"
        B_size = hidden_states.shape[0]  # batch_size
        NC_size = hidden_states.shape[1]  # num_chunks
        # Prepare expanded B and C for heads (repeat_interleave 4 since NUM_HEADS//N_GROUPS=4)
        # Note: We operate on float32; outputs are cast to bfloat16 at the end
        B_exp = B.repeat_interleave(REPEAT, dim=3).contiguous().to(torch.float32)
        C_exp = C.repeat_interleave(REPEAT, dim=3).contiguous().to(torch.float32)
        hidden_exp = hidden_states.to(torch.float32)

        # Allocate outputs
        L = torch.empty((B_size, NC_size, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS), device=device, dtype=torch.float32)
        G = torch.empty((B_size, NC_size, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS), device=device, dtype=torch.float32)
        M = torch.empty((B_size, NC_size, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS), device=device, dtype=torch.float32)
        Y = torch.empty((B_size, NC_size, CHUNK_SIZE, NUM_HEADS, HEAD_DIM), device=device, dtype=torch.bfloat16)

        # Launch kernels
        # Grid for build_L: (B, NC, NUM_HEADS)
        grid_L = (B_size, NC_size, NUM_HEADS)
        build_L_kernel[grid_L](A_cumsum, L, B_size, NC_size)

        # Grid for compute_G: (B, NC, CHUNK_SIZE, NUM_HEADS)
        grid_G = (B_size, NC_size, CHUNK_SIZE, NUM_HEADS)
        compute_G_kernel[grid_G](B_exp, C_exp, G, B_size, NC_size)

        # Grid for multiply_LG: (B, NC, CHUNK_SIZE, NUM_HEADS)
        grid_M = (B_size, NC_size, CHUNK_SIZE, NUM_HEADS)
        multiply_LG_kernel[grid_M](G, L, M, B_size, NC_size)

        # Grid for contract_to_Ydiag: (B, NC, CHUNK_SIZE, NUM_HEADS, HEAD_DIM)
        grid_Y = (B_size, NC_size, CHUNK_SIZE, NUM_HEADS, HEAD_DIM)
        contract_to_Ydiag_kernel[grid_Y](M, hidden_exp, Y, B_size, NC_size)

        return Y
