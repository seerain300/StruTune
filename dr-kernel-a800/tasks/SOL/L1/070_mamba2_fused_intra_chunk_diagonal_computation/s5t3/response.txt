import triton
import triton.language as tl


@triton.jit
def mask_lower_tri_kernel(
    mask_ptr,      # *int32, [L*L] linear buffer for mask
    L_len: tl.constexpr,
):
    idx = tl.program_id(0)
    # Linear index i, j from idx
    i = idx // L_len
    j = idx % L_len
    # Lower-triangular mask with diagonal = -1 -> keep if j <= i
    if j <= i:
        tl.store(mask_ptr + idx, 1)
    else:
        tl.store(mask_ptr + idx, 0)

    L_len,          # int

    BLOCK_J: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    c = tl.program_id(2)

    # We will iterate i and j
    i = 0
    while i < L_len:
        # Compute sum s_i = sum_{k=0..i} A[b, c, k, h]
        s = 0.0
        k = 0
        while k <= i:
            # A[b, c, k, h] addressing: A strides (C*L*H, L*H, H, 1)
            a_val = tl.load(A_ptr + b * (C_size * L_len * H_size) + c * (L_len * H_size) + k * H_size + h)
            s += a_val
            k += 1
        # Now set L[i, j] = exp(s) for j <= i
        j = 0
        while j < L_len:
            idx = i * L_len + j
            m_val = tl.load(mask_ptr + idx)
            if m_val != 0:
                L_base = b * (H_size * C_size * L_len * L_len) + h * (C_size * L_len * L_len) + c * (L_len * L_len)
                addr = L_base + i * L_len + j
                tl.store(L_ptr + addr, tl.exp(s))
            j += 1
        i += 1


@triton.jit
def compute_G_kernel(
    B_exp_ptr,      # *float32, [B, C, L, H, S]
    C_exp_ptr,      # *float32, [B, C, L, H, S]
    G_ptr,          # *float32, [B, C, L, L, H]
    B_size,         # int
    C_size,         # int
    L_len,          # int
    H_size,         # int
    S_size,         # int
    groups,          # int
    group_expand,    # int
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)

    i = 0
    while i < L_len:
        j = 0
        while j < L_len:
            g_val = 0.0
            s = 0
            while s < S_size:
                # Addresses for B_exp[b, c, j, h, s] and C_exp[b, c, i, h, s]
                stride_s = 1
                stride_h = S_size
                stride_j = H_size * S_size
                stride_c = L_len * H_size * S_size
                stride_b = C_size * L_len * H_size * S_size

                B_addr = b * stride_b + c * stride_c + j * stride_j + h * stride_h + s * stride_s
                C_addr = b * stride_b + c * stride_c + i * stride_j + h * stride_h + s * stride_s

                B_val = tl.load(B_exp_ptr + B_addr)
                C_val = tl.load(C_exp_ptr + C_addr)
                g_val += C_val * B_val
                s += 1
            # Store G[b, c, i, j, h]
            G_stride_H = 1
            G_stride_L = H_size
            G_stride_C = L_len * H_size
            G_stride_B = C_size * L_len * H_size

            G_addr = b * G_stride_B + c * G_stride_C + i * L_len + j * G_stride_L + h * G_stride_H
            tl.store(G_ptr + G_addr, g_val)
            j += 1
        i += 1
