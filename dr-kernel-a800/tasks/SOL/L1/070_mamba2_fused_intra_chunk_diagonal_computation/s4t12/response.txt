import triton.language as tl

# For each (n, h, t), compute segment_sum[i, j] = sum_{m=0..j} A_cumsum[n, h, t, i] if i <= j else 0.
# Then L[n, h, t, i, j] = exp(segment_sum[i, j]) if i <= j, else 0.
# Output L_out: [N, H, T, L, L] (float32)
@triton.jit
def _build_lower_tri_exp_mask_from_cumsum(
    A_cumsum_ptr, L_out_ptr,
    N, H, T, L,
    stride_A_n, stride_A_h, stride_A_t, stride_A_l,
    stride_L_n, stride_L_h, stride_L_t, stride_L_i, stride_L_j,
):
    n = tl.program_id(0)
    h = tl.program_id(1)
    t = tl.program_id(2)

    # We will fill L_out[n, h, t, i, j] for all i, j in [0..L-1]
    for i in range(L):
        # Initialize segment_sum for this i
        seg_sum = 0.0
        # For each j, compute segment_sum[i, j] with lower-triangular mask (i <= j)
        for j in range(L):
            if j < i:
                # upper triangular: segment_sum is 0
                val = 0.0
            else:
                # lower triangular: include A_cumsum[n, h, t, i]
                # Address: A_cumsum[n, h, t, i]
                a_ptr = A_cumsum_ptr + n * stride_A_n + h * stride_A_h + t * stride_A_t + i * stride_A_l
                a_val = tl.load(a_ptr)
                seg_sum += a_val
                val = seg_sum
            # Store exp(val) if lower-triangular, else 0
            l_ptr = L_out_ptr + n * stride_L_n + h * stride_L_h + t * stride_L_t + i * stride_L_i + j * stride_L_j
            if j >= i:
                tl.store(l_ptr, tl.exp(val))
            else:
                tl.store(l_ptr, 0.0)


# Kernel 2: Contract B @ C^T to form G: G[n, t, i, j, h] = sum over g and k of C[n, t, i, g, k] * B[n, t, j, g, k]
# Inputs:
#   B_in: [N, T, L, G, K]
#   C_in: [N, T, L, G, K]
# Output:
#   Gout: [N, T, L, L, H] (float32)

def _contract_bc_to_g(
    B_in_ptr, C_in_ptr, Gout_ptr,
    N, T, L, H, G, K,
    stride_B_n, stride_B_t, stride_B_l, stride_B_g, stride_B_k,
    stride_C_n, stride_C_t, stride_C_l, stride_C_g, stride_C_k,
    stride_G_n, stride_G_t, stride_G_i, stride_G_j, stride_G_h,
):
    # Grid over (n, t, i, j, h)
    n = tl.program_id(0)
    t = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)

    acc = 0.0
    # Sum over groups g and K (runtime loops; Triton supports Python range for small sizes)
    for g in range(G):
        for k in range(K):
            b_ptr = B_in_ptr + n * stride_B_n + t * stride_B_t + j * stride_B_l + g * stride_B_g + k * stride_B_k
            c_ptr = C_in_ptr + n * stride_C_n + t * stride_C_t + i * stride_C_l + g * stride_C_g + k * stride_C_k
            b = tl.load(b_ptr)
            c = tl.load(c_ptr)
            acc += c * b

    gout_ptr = Gout_ptr + n * stride_G_n + t * stride_G_t + i * stride_G_i + j * stride_G_j + h * stride_G_h
    tl.store(gout_ptr, acc)


# Kernel 3: Apply mask L (already permuted to [N, T, L, L, H]) to G to produce M: M = G * L
# Inputs:
#   G: [N, T, L, L, H]
#   L_perm: [N, T, L, L, H]
# Output:

@triton.jit

    h = tl.program_id(2)

    # Initialize per d accumulator arrays (small D). We'll do one d per program by looping over d.
    # Alternative: launch grid over (N, T, H, D) and compute single d; here we loop over D inside.
    for d in range(D):
        acc = 0.0
        for i in range(L):

                hs = tl.load(hs_ptr)
