def build_L_kernel(

    L_ptr,            # *ptr to L: [B, C, S, S, H]

    C,                # int

    H: tl.constexpr,  # compile-time constant 32
    stride_ab, stride_ah, stride_ac, stride_as,  # strides for A
    stride_lb, stride_lc, stride_li, stride_lj, stride_lh  # strides for L
):
    # Grid: (B, C, H)
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)

    # Compute cumsum along j for A[b, h, c, :]
    # We'll accumulate in float32 for numerical stability.
    # We'll write to L at each (i, j) where i >= j.
    for j in range(0, S):
        # Load A[b, h, c, j]
        a_val = tl.load(A_ptr + b * stride_ab + h * stride_ah + c * stride_ac + j * stride_as)
