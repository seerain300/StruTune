# Kernel 1: Build L via segment_sum(A_cumsum) with lower-triangular mask (diagonal = -1), then exp.
# L_out: [N, H, T, L, L] (float32)
# We vectorize over i in [0..L-1], build segment_sum for all j in [0..L-1] using a lower-triangular mask.
@triton.jit
def _build_lower_tri_exp_mask_from_cumsum_1d(
    A_cumsum_ptr, L_out_ptr,
    N, H, T, L,
    stride_A_n, stride_A_h, stride_A_t, stride_A_l,
    stride_L_n, stride_L_h, stride_L_t, stride_L_i, stride_L_j,
):
    n = tl.program_id(0)
    h = tl.program_id(1)
    t = tl.program_id(2)

    for i in range(L):
        seg_sum = 0.0
        j_vec = tl.arange(0, L)
        mask_vec = j_vec >= i
        a_ptr = A_cumsum_ptr + n * stride_A_n + h * stride_A_h + t * stride_A_t + i * stride_A_l
        # load A[i] with mask; if j<i, set to 0
        a_vals = tl.load(a_ptr, mask=mask_vec, other=0.0)
        # inclusive cumsum across j
        seg_sum_vec = tl.cumsum(a_vals, axis=0)
        # exp of segment_sum
        l_vals = tl.exp(seg_sum_vec)
        # store to L[n, h, t, i, j]
        l_ptrs = L_out_ptr + n * stride_L_n + h * stride_L_h + t * stride_L_t + i * stride_L_i + j_vec * stride_L_j
        tl.store(l_ptrs, l_vals, mask=mask_vec)


# Kernel 2: Contract B @ C^T to form G: G[n, t, i, j, h] = sum over groups g and state K of C[n,t,i,g,k] * B[n,t,j,g,k]
# Inputs:
#   B_in: [N, T, L, G, K]
#   C_in: [N, T, L, G, K]
# Output:
#   Gout: [N, T, L, L, H] (float32)
# Grid: (N, T, L, L, H)
@triton.jit
def _contract_bc_to_g(
    B_in_ptr, C_in_ptr, Gout_ptr,
    N, T, L, H, G, K,
    stride_B_n, stride_B_t, stride_B_l, stride_B_g, stride_B_k,
    stride_C_n, stride_C_t, stride_C_l, stride_C_g, stride_C_k,
    stride_G_n, stride_G_t, stride_G_i, stride_G_j, stride_G_h,
):
    n = tl.program_id(0)
    t = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)

            c_ptr = C_in_ptr + n * stride_C_n + t * stride_C_t + i * stride_C_l + g * stride_C_g + k * stride_C_k
            b = tl.load(b_ptr)
            c = tl.load(c_ptr)

@triton.jit

@triton.jit

    A_cumsum_ptr, L_out_ptr,
    N, H, T, L,
    stride_A_n, stride_A_h, stride_A_t, stride_A_l,
    stride_L_n, stride_L_h, stride_L_t, stride_L_i, stride_L_j,
