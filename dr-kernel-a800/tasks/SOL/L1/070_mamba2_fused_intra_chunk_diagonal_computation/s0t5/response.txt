    A_ptr,            # [B, H, C, S], float32
    L_ptr,            # [B, C, S, S, H], float32
    Bsz: tl.constexpr,
    Csz: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
):
    bc = tl.program_id(0)  # over B*C
    i = tl.program_id(1)   # row index i in [0..S-1]
    h = tl.program_id(2)   # head index h in [0..H-1]
    b = bc // Csz
    c = bc % Csz

    # Base offset for A[b, h, c, :]
    base_A = b * (H * Csz * S) + h * (Csz * S) + c * S

    # Cumulative sum vector cum[j]
    cum = tl.zeros([S], dtype=tl.float32)

    # For each j, compute cum up to j and store exp(cum[j]) at L[b, c, i, j, h]
    for j in range(0, S):
        # Accumulate A[b, h, c, j] only if j <= i (lower-triangular). For j > i, cum stays the same.
        a_val = tl.load(A_ptr + base_A + j)
        if j <= i:
            # cum[j] = sum_{k=0..j} A[b,h,c,k] but since we only include up to i, we maintain cum[j] = cum[i] at j>i
            # For j <= i, include a_val; for j > i, skip (cum unchanged).
            cum[j] = cum[j - 1] + a_val  # cum[j] = previous cum plus a_val
        else:
            cum[j] = cum[j - 1]  # no contribution beyond i
        # L[b, c, i, j, h] = exp(cum[j]) for j <= i, else 0
        L_val = tl.exp(cum[j])
        if j > i:
            L_val = 0.0
        L_offset = b * (Csz * S * S * H) + c * (S * S * H) + i * (S * H) + j * H + h
        tl.store(L_ptr + L_offset, L_val)


# Kernel: Compute G[b, c, i, j, h] = sum_n C_exp[b, c, i, n, j, h] * B_exp[b, c, j, n, i, h]
# Grid: (B*C, S, H) => each program handles a fixed (b,c,i,h) and loops over j
@triton.jit
def compute_G_rows_kernel(
    B_exp_ptr,        # [B, C, S, H, N], float32
    C_exp_ptr,        # [B, C, S, H, N], float32
    G_ptr,            # [B, C, S, S, H], float32
    Bsz: tl.constexpr,
    Csz: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    N: tl.constexpr,
):
    bc = tl.program_id(0)  # over B*C
    i = tl.program_id(1)   # row i in [0..S-1]
    h = tl.program_id(2)   # head h in [0..H-1]
    b = bc // Csz
    c = bc % Csz

    # Compute G[i, j, h] for all j in [0..S-1]
    for j in range(0, S):
        acc = 0.0
        # Sum over n from 0 to N-1 (N=128 typically)
        for n in range(0, N):
            # B_exp[b, c, j, h, n] offset
            B_off = b * (Csz * S * H * N) + c * (S * H * N) + j * (H * N) + h * N + n
            # C_exp[b, c, i, n, j, h] offset
            C_off = b * (Csz * S * H * N) + c * (S * H * N) + i * (H * N) + j * (H * N) + n
            B_val = tl.load(B_exp_ptr + B_off)
            C_val = tl.load(C_exp_ptr + C_off)
            acc += C_val * B_val
        G_off = b * (Csz * S * S * H) + c * (S * S * H) + i * (S * H) + j * H + h
        tl.store(G_ptr + G_off, acc)


# Kernel: Compute Y_diag[b, c, i, h, d] = sum_j G[b, c, i, j, h] * hidden_states[b, c, j, h, d]
# Grid: (B*C*S*H, head_dim) => each program handles a fixed (b, c, i, h, d) and loops over j

):
    pid = tl.program_id(0)  # over B*C*S*H planes
    d = tl.program_id(1)    # over head_dim
    bc = pid // (S * H)
    rem = pid % (S * H)
    i = rem // H
