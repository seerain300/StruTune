```python
import torch
import triton
import triton.language as tl


@triton.jit
def build_lower_tri_causal_kernel(
    A_ptr,  # [B, C, L, H], float32
    L_ptr,  # [B, C, 128, 128, H], float32
    Bsz: tl.constexpr, Csz: tl.constexpr, Lsz: tl.constexpr, Hsz: tl.constexpr,
    stride_A_b, stride_A_c, stride_A_i, stride_A_h, stride_A_s,  # A strides
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,  # L strides
):
    # program ids
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)  # row in 0..127
    j = tl.program_id(4)  # col in 0..127

    # Bounds check: if out of (i,j) range, return
    if i >= 128 or j >= 128:
        return

    # Compute cumsum for row i: total = sum_{k=0..L-1} A[b, c, k, h]
    total = tl.zeros((), dtype=tl.float32)
    for k in range(Lsz):
        a_off = b * stride_A_b + c * stride_A_c + k * stride_A_i + h * stride_A_h
        a_val = tl.load(A_ptr + a_off)
        total += a_val

    # If j <= i, set L[i, j] = exp(total), else 0
    if j <= i:
        l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + h * stride_L_h
        tl.store(L_ptr + l_off, tl.exp(total))
    else:
        l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + h * stride_L_h
        tl.store(L_ptr + l_off, 0.0)


@triton.jit
def expand_groups_repeat_interleave(
    X_ptr,  # [B, C, L, G, S], float32
    Xexp_ptr,  # [B, C, L, H, S], float32
    Bsz: tl.constexpr, Csz: tl.constexpr, Lsz: tl.constexpr, Hsz: tl.constexpr, Gsz: tl.constexpr, Ssz: tl.constexpr,
    GROUP_EXPAND: tl.constexpr,
    stride_X_b, stride_X_c, stride_X_i, stride_X_g, stride_X_s,  # X strides
    stride_Xexp_b, stride_Xexp_c, stride_Xexp_i, stride_Xexp_h, stride_Xexp_s,  # Xexp strides
):
    # program ids
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)  # i in 0..L-1
    h = tl.program_id(3)  # h in 0..H-1
    s = tl.program_id(4)  # s in 0..S-1

    # Map head h to group g: h = g * GROUP_EXPAND
    g = h // GROUP_EXPAND

    # Compute offsets
    x_off = b * stride_X_b + c * stride_X_c + i * stride_X_i + g * stride_X_g + s * stride_X_s
    xexp_off = b * stride_Xexp_b + c * stride_Xexp_c + i * stride_Xexp_i + h * stride_Xexp_h + s * stride_Xexp_s

    val = tl.load(X_ptr + x_off)
    tl.store(Xexp_ptr + xexp_off, val)


@triton.jit
def compute_G_outer_product(
    Bexp_ptr,  # [B, C, L, H, S], float32
    Cexp_ptr,  # [B, C, L, H, S], float32
    G_ptr,     # [B, C, L, L, H], float32
    Bsz: tl.constexpr, Csz: tl.constexpr, Lsz: tl.constexpr, Hsz: tl.constexpr, Ssz: tl.constexpr,
    stride_Bexp_b, stride_Bexp_c, stride_Bexp_i, stride_Bexp_h, stride_Bexp_s,
    stride_Cexp_b, stride_Cexp_c, stride_Cexp_i, stride_Cexp_h, stride_Cexp_s,
    stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h,
):
    # program ids
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)  # head
    i = tl.program_id(3)  # row
    j = tl.program_id(4)  # col

    if i >= Lsz or j >= Lsz or h >= Hsz:
        return

    # Accumulate G[i, j, h]
    acc = tl.zeros((), dtype=tl.float32)
    for s in range(Ssz):
        b_off = b * stride_Bexp_b + c * stride_Bexp_c + i * stride_Bexp_i + h * stride_Bexp_h + s * stride_Bexp_s
        c_off = b * stride_Cexp_b + c * stride_Cexp_c + j * stride_Cexp_i + h * stride_Cexp_h + s * stride_Cexp_s
        bval = tl.load(Bexp_ptr + b_off)
        cval = tl.load(Cexp_ptr + c_off)
        acc += bval * cval

    g_off = b * stride_G_b + c * stride_G_c + i * stride_G_i + j * stride_G_j + h * stride_G_h
    tl.store(G_ptr + g_off, acc)


@triton.jit
def apply_mask_Multiply_L_and_G(
    L_ptr,  # [B, C, 128, 128, H], float32
    G_ptr,  # [B, C, L, L, H], float32
    M_ptr,  # [B, C, L, L, H], float32
    Bsz: tl.constexpr, Csz: tl.constexpr, Lsz: tl.constexpr, Hsz: tl.constexpr,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h,
    stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h,
):
    # program ids
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)  # head
    i = tl.program_id(3)  # row
    j = tl.program_id(4)  # col

    if i >= Lsz or j >= Lsz or h >= Hsz:
        return

    # Load L[i, j] for (b, c, h). Even though L is 128x128, we only need the lower-triangular and j < 128 always matches since j in [0, L-1] and L <= 128 in practice.
    l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + h * stride_L_h
    l_val = tl.load(L_ptr + l_off)

    # Load G[i, j, h]
    g_off = b * stride_G_b + c * stride_G_c + i * stride_G_i + j * stride_G_j + h * stride_G_h
    g_val = tl.load(G_ptr + g_off)

    m_val = g_val * l_val
    m_off = b * stride_M_b + c * stride_M_c + i * stride_M_i + j * stride_M_j + h * stride_M_h
    tl.store(M_ptr + m_off, m_val)


@triton.jit
def compute_Y_diag_Contract_j(
    M_ptr,     # [B, C, L, L, H], float32
    Hs_ptr,    # [B, C, L, H, D], float32 (hidden_states)
    Y_ptr,     # [B, C, L, H, D], float32
    Bsz: tl.constexpr, Csz: tl.constexpr, Lsz: tl.constexpr, Hsz: tl.constexpr, Dsz: tl.constexpr,
    stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h,
    stride_H_b, stride_H_c, stride_H_i, stride_H_h, stride_H_d,
    stride_Y_b, stride_Y_c, stride_Y_i, stride_Y_h, stride_Y_d,
):
    # Grid: (b, c, i)
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)

    # Vectorize over head h and dim d inside the kernel
    h = 0
    while h < Hsz:
        d = 0
        while d < Dsz:
            acc = tl.zeros((), dtype=tl.float32)
            j = 0
            while j < Lsz:
                m_off = b * stride_M_b + c * stride_M_c + i * stride_M_i + j * stride_M_j + h * stride_M_h
                # hidden_states[b, c, j, h, d]
                hs_off = b * stride_H_b + c * stride_H_c + j * stride_H_i + h * stride_H_h + d * stride_H_d
                m_val = tl.load(M_ptr + m_off)
                hs_val = tl.load(Hs_ptr + hs_off)
                acc += m_val * hs_val
                j += 1
            # Store acc into Y[b, c, i, h, d]
            y_off = b * stride_Y_b + c * stride_Y_c + i * stride_Y_i + h * stride_Y_h + d * stride_Y_d
            tl.store(Y_ptr + y_off, acc)
            d += 1
        h += 1


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
                A_cumsum: torch.Tensor,
                B: torch.Tensor,
                C: torch.Tensor) -> torch.Tensor:
        # Extract shapes
        Bsz, Csz, Lsz, Hsz, Dsz = hidden_states.shape

        # Constants
        NUM_HEADS = 32
        N_GROUPS = 8
        GROUP_EXPAND = 4  # 32 // 8
        L_mat_size = 128  # as in original code

        # Cast inputs to float32 for compute
        hidden_f32 = hidden_states.to(torch.float32)
        A_f32 = A_cumsum.to(torch.float32)  # [B, C, L, H]
        B_f32 = B.to(torch.float32)         # [B, C, L, G, S]
        C_f32 = C.to(torch.float32)         # [B, C, L, G, S]

        # 1) Build L in Triton: L[b, c, 128, 128, h]
        L = torch.empty((Bsz, Csz, L_mat_size, L_mat_size, Hsz), dtype=torch.float32, device=hidden_f32.device)

        stride_A_b, stride_A_c, stride_A_i, stride_A_h, stride_A_s = A_f32.stride()
        stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h = L.stride()

        grid_L = (Bsz, Csz, Hsz, L_mat_size, L_mat_size)
        build_lower_tri_causal_kernel[grid_L](
            A_f32, L,
            Bsz, Csz, Lsz, Hsz,
            stride_A_b, stride_A_c, stride_A_i, stride_A_h, stride_A_s,
            stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
            num_warps=4, num_stages=2
        )

        # 2) Expand B and C from groups to heads
        Ssz = B_f32.shape[-1]
        B_exp = torch.empty((Bsz, Csz, Lsz, Hsz, Ssz), dtype=torch.float32, device=hidden_f32.device)
        C_exp = torch.empty((Bsz, Csz, Lsz, Hsz, Ssz), dtype=torch.float32, device=hidden_f32.device)

        stride_B_b, stride_B_c, stride_B_i, stride_B_g, stride_B_s = B_f32.stride()
        stride_C_b, stride_C_c, stride_C_i, stride_C_g, stride_C_s = C_f32.stride()
        stride_Bexp_b, stride_Bexp_c, stride_Bexp_i, stride_Bexp_h, stride_Bexp_s = B_exp.stride()
        stride_Cexp_b, stride_Cexp_c, stride_Cexp_i, stride_Cexp_h, stride_Cexp_s = C_exp.stride()

        grid_expand = (Bsz, Csz, Lsz, Hsz, Ssz)
        expand_groups_repeat_interleave[grid_expand](
            B_f32, B_exp,
            Bsz, Csz, Lsz, Hsz, N_GROUPS, Ssz,
            GROUP_EXPAND,
            stride_B_b, stride_B_c, stride_B_i, stride_B_g, stride_B_s,
            stride_Bexp_b, stride_Bexp_c, stride_Bexp_i, stride_Bexp_h, stride_Bexp_s,
            num_warps=4, num_stages=2
        )
        expand_groups_repeat_interleave[grid_expand](
            C_f32, C_exp,
            Bsz, Csz, Lsz, Hsz, N_GROUPS, Ssz,
            GROUP_EXPAND,
            stride_C_b, stride_C_c, stride_C_i, stride_C_g, stride_C_s,
            stride_Cexp_b, stride_Cexp_c, stride_Cexp_i, stride_Cexp_h, stride_Cexp_s,
            num_warps=4, num_stages=2
        )

        # 3) Compute G via outer product contraction over S
        G = torch.empty((Bsz, Csz, Lsz, Lsz, Hsz), dtype=torch.float32, device=hidden_f32.device)

        stride_Bexp_b, stride_Bexp_c, stride_Bexp_i, stride_Bexp_h, stride_Bexp_s = B_exp.stride()
        stride_Cexp_b, stride_Cexp_c, stride_Cexp_i, stride_Cexp_h, stride_Cexp_s = C_exp.stride()
        stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h = G.stride()

        grid_G = (Bsz, Csz, Hsz, Lsz, Lsz)
        compute_G_outer_product[grid_G](
            B_exp, C_exp, G,
            Bsz, Csz, Lsz, Hsz, Ssz,
            stride_Bexp_b, stride_Bexp_c, stride_Bexp_i, stride_Bexp_h, stride_Bexp_s,
            stride_Cexp_b, stride_Cexp_c, stride_Cexp_i, stride_Cexp_h, stride_Cexp_s,
            stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h,
            num_warps=4, num_stages=2
        )

        # 4) Apply mask: M = G * L
        M = torch.empty((Bsz, Csz, Lsz, Lsz, Hsz), dtype=torch.float32, device=hidden_f32.device)

        stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h = L.stride()
        stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h = G.stride()
        stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h = M.stride()

        grid_M = (Bsz, Csz, Hsz, Lsz, Lsz)
        apply_mask_Multiply_L_and_G[grid_M](
            L, G, M,
            Bsz, Csz, Lsz, Hsz,
            stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
            stride_G_b, stride_G_c, stride_G_i, stride_G_j, stride_G_h,
            stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h,
            num_warps=4, num_stages=2
        )

        # 5) Compute Y_diag by contracting over j: Y[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden[b, c, j, h, d]
        # hidden_states is already provided as [B, C, L, H, D] in the original signature; we use it directly.
        Y = torch.empty((Bsz, Csz, Lsz, Hsz, Dsz), dtype=torch.float32, device=hidden_f32.device)

        stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h = M.stride()
        stride_H_b, stride_H_c, stride_H_i, stride_H_h, stride_H_d = hidden_f32.stride()
        stride_Y_b, stride_Y_c, stride_Y_i, stride_Y_h, stride_Y_d = Y.stride()

        grid_Y = (Bsz, Csz, Lsz)
        compute_Y_diag_Contract_j[grid_Y](
            M, hidden_f32, Y,
            Bsz, Csz, Lsz, Hsz, Dsz,
            stride_M_b, stride_M_c, stride_M_i, stride_M_j, stride_M_h,
            stride_H_b, stride_H_c, stride_H_i, stride_H_h, stride_H_d,
            stride_Y_b, stride_Y_c, stride_Y_i, stride_Y_h, stride_Y_d,
            num_warps=4, num_stages=2
        )

        # Return in bfloat16 as original model does
        return Y.to(torch.bfloat16)
```
