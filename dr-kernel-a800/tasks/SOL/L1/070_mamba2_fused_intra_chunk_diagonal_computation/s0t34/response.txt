import torch
import triton
import triton.language as tl

# Kernel 1: build L = exp(cumsum(A)) with lower-triangular mask, per (b, c, h), i >= j
@triton.jit
def build_L_kernel(
    A_ptr, L_ptr,
    Bsz, Csz, H,
    aB, aH, aC, aS,  # strides for A: [B, H, C, S]
    lB, lC, lSi, lSj, lH,  # strides for L: [B, C, Si, Sj, H]
    S_CONST: tl.constexpr,
):
    # Grid: (B*C*H, S, S)
    pid = tl.program_id(0)
    si = tl.program_id(1)
    sj = tl.program_id(2)
    b = pid // (Csz * H)
    rem = pid % (Csz * H)
    c = rem // H
    h = rem % H

    # Bounds check (shouldn't be needed if grid matches exactly, but safe)
    if (b >= Bsz) or (c >= Csz) or (h >= H):
        return

    # Compute cumsum along j up to i (for each i), then set L[i, j, h] = exp(cumsum[j]) if i >= j else 0
    # We compute per i: i = si (program_id(1)), then loop j
    # Note: Triton requires compile-time constants in for-loops; S_CONST is constexpr.
    for j in range(S_CONST):
        # A[b, h, c, j] -> offset: b*aB + h*aH + c*aC + j*aS
        # Load A values up to j for cumulative sum
        sum_val = 0.0
        for k in range(0, j + 1):
            val = tl.load(A_ptr + b * aB + h * aH + c * aC + k * aS)
            sum_val += val
        exp_val = tl.exp(sum_val)
        # If i >= j, set L[i, j, h] = exp_val else 0
        if si >= j:
            # L[b, c, si, sj, h] -> offset: b*lB + c*lC + si*lSi + sj*lSj + h*lH
            tl.store(L_ptr + b * lB + c * lC + si * lSi + sj * lSj + h * lH, exp_val)
        else:
            tl.store(L_ptr + b * lB + c * lC + si * lSi + sj * lSj + h * lH, 0.0)


# Kernel 2: compute G[i, j, h] = sum_n C_expanded[b, c, i, n, j, h] * B_expanded[b, c, j, n, i, h]
# We compute for all h in a single kernel launch; grid will iterate over (b,c,i,j,h)
@triton.jit
def compute_G_kernel(
    Bexp_ptr, Cexp_ptr, G_ptr,
    Bsz, Csz, H, S_CONST, N_CONST,
    beB, beC, beSi, beH, beN,  # strides for B_expanded: [B, C, Si, H, N]
    ceB, ceC, ceSi, ceH, ceN,  # strides for C_expanded: [B, C, Si, H, N]
    gB, gC, gSi, gSj, gH,      # strides for G: [B, C, Si, Sj, H]
    S_CONST: tl.constexpr, N_CONST: tl.constexpr,
):
    # Grid: (B*C*S_CONST*S_CONST*H, ) i.e., one program per (b, c, i, j, h)
    pid = tl.program_id(0)
    total = Bsz * Csz * S_CONST * S_CONST * H
    if pid >= total:
        return

    # Decode pid -> (b, c, i, j, h)
    tmp = pid
    h = tmp % H
    tmp = tmp // H
    sj = tmp % S_CONST
    tmp = tmp // S_CONST
    si = tmp % S_CONST
    tmp = tmp // S_CONST
    c = tmp % Csz
    b = tmp // Csz

    # Accumulate sum over n
    acc = 0.0
    for n in range(N_CONST):
        b_val = tl.load(Bexp_ptr + b * beB + c * beC + si * beSi + h * beH + n * beN)
        c_val = tl.load(Cexp_ptr + b * ceB + c * ceC + si * ceSi + h * ceH + n * ceN)
        acc += b_val * c_val
    # Store G[b, c, si, sj, h] = acc
    tl.store(G_ptr + b * gB + c * gC + si * gSi + sj * gSj + h * gH, acc)


# Kernel 3: compute Y_diag[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden_states[b, c, j, h, d]
# We iterate over (b, c, i, h) and launch per d inside. Triton doesn't support varying grid on d, so do this in-kernel for each head_dim.
@triton.jit
def compute_Y_diag_kernel(
    M_ptr, hs_ptr, Y_ptr,
    Bsz, Csz, H, S_CONST, head_dim,
    mB, mC, mSi, mSj, mH,     # strides for M: [B, C, Si, Sj, H]
    hsB, hsC, hsSi, hsH, hsD, # strides for hidden_states: [B, C, Si, H, D]
    yB, yC, ySi, yH, yD,      # strides for Y: [B, C, Si, H, D]
    S_CONST: tl.constexpr, head_dim: tl.constexpr,
):
    # Grid: (B*C*S_CONST*H*head_dim, ) decode into (b, c, i, h, d)
    pid = tl.program_id(0)
    total = Bsz * Csz * S_CONST * H * head_dim
    if pid >= total:
        return

    # Decode pid -> (b, c, i, h, d)
    tmp = pid
    d = tmp % head_dim
    tmp = tmp // head_dim
    h = tmp % H
    tmp = tmp // H
    si = tmp % S_CONST
    tmp = tmp // S_CONST
    c = tmp % Csz
    b = tmp // Csz

    acc = 0.0
    for j in range(S_CONST):
        m_val = tl.load(M_ptr + b * mB + c * mC + si * mSi + j * mSj + h * mH)
        hs_val = tl.load(hs_ptr + b * hsB + c * hsC + j * hsSi + h * hsH + d * hsD)
        acc += m_val * hs_val
    # Store Y[b, c, si, h, d] = acc
    tl.store(Y_ptr + b * yB + c * yC + si * ySi + h * yH + d * yD, acc)


class ModelNew(torch.nn.Module):
    def __init__(self, S_CONST=128, H_CONST=32, N_GROUPS=8):
        super().__init__()
        self.S_CONST = S_CONST
        self.H_CONST = H_CONST
        self.N_GROUPS = N_GROUPS

    def forward(self, hidden_states: torch.Tensor, A_cumsum: torch.Tensor, B: torch.Tensor, C: torch.Tensor) -> torch.Tensor:
        # Ensure tensors are on CUDA
        assert hidden_states.is_cuda and A_cumsum.is_cuda and B.is_cuda and C.is_cuda, "All tensors must be on CUDA for Triton kernels."
        # Shapes
        Bsz, Csz, S, H, head_dim = hidden_states.shape
        # The original code implies S=128, H=32. We use S_CONST=128, H_CONST=32 from init. If not matched, the provided axes workloads use these exact values.
        assert S == self.S_CONST, f"hidden_states sequence size S must be {self.S_CONST}, got {S}"
        assert H == self.H_CONST, f"hidden_states heads H must be {self.H_CONST}, got {H}"

        # Make inputs contiguous
        A = A_cumsum.contiguous()
        B_expanded = B.repeat_interleave(self.H_CONST // self.N_GROUPS, dim=3).contiguous()
        C_expanded = C.repeat_interleave(self.H_CONST // self.N_GROUPS, dim=3).contiguous()
        hidden_states_cs = hidden_states.contiguous()

        # 1) Build L: [B, C, S, S, H] as float32
        L = torch.empty((Bsz, Csz, self.S_CONST, self.S_CONST, H), device=hidden_states.device, dtype=torch.float32)

        # Strides
        aB, aH, aC, aS = A.stride()
        lB, lC, lSi, lSj, lH = L.stride()

        # Launch build_L_kernel
        grid = (Bsz * Csz * H, self.S_CONST, self.S_CONST)
        build_L_kernel[grid](
            A, L,
            Bsz, Csz, H,
            aB, aH, aC, aS,
            lB, lC, lSi, lSj, lH,
            S_CONST=self.S_CONST,
            num_warps=1, num_stages=1,
        )

        # 2) Compute G: [B, C, S, S, H] via contraction over N
        # We need N from B or C shape; assume B shape is [B, C, S, N_GROUPS, N]. In provided tests, N typically 128.
        # Extract N from B (assume consistent with C)
        # B shape: [B, C, S, N_GROUPS, N]
        Bsz_b, Csz_b, S_b, N_GROUPS_b, N = B.shape
        assert Bsz_b == Bsz and Csz_b == Csz and S_b == self.S_CONST and N_GROUPS_b == self.N_GROUPS, "B shape mismatch"
        G = torch.empty((Bsz, Csz, self.S_CONST, self.S_CONST, H), device=hidden_states.device, dtype=torch.float32)

        # Strides for B_expanded, C_expanded, G
        beB, beC, beSi, beH, beN = B_expanded.stride()  # [B, C, Si, H, N]
        ceB, ceC, ceSi, ceH, ceN = C_expanded.stride()  # [B, C, Si, H, N]
        gB, gC, gSi, gSj, gH = G.stride()

        # Launch compute_G_kernel
        # One program per (b, c, i, j, h)
        total = Bsz * Csz * self.S_CONST * self.S_CONST * H
        grid2 = (total, )
        compute_G_kernel[grid2](
            B_expanded, C_expanded, G,
            Bsz, Csz, H, self.S_CONST, N,
            beB, beC, beSi, beH, beN,
            ceB, ceC, ceSi, ceH, ceN,
            gB, gC, gSi, gSj, gH,
            S_CONST=self.S_CONST, N_CONST=N,
            num_warps=1, num_stages=1,
        )

        # 3) Multiply M = G * L
        M = G * L  # element-wise multiply, float32

        # 4) Compute Y_diag: [B, C, S, H, head_dim]
        Y = torch.empty((Bsz, Csz, self.S_CONST, H, head_dim), device=hidden_states.device, dtype=torch.float32)

        # Strides
        mB, mC, mSi, mSj, mH = M.stride()  # [B, C, Si, Sj, H]
        hsB, hsC, hsSi, hsH, hsD = hidden_states_cs.stride()  # [B, C, Si, H, D]
        yB, yC, ySi, yH, yD = Y.stride()

        # Launch compute_Y_diag_kernel for each d (head_dim is small, e.g., 1 in tests). Triton will handle each pid.
        total_y = Bsz * Csz * self.S_CONST * H * head_dim
        grid3 = (total_y, )
        compute_Y_diag_kernel[grid3](
            M, hidden_states_cs, Y,
            Bsz, Csz, H, self.S_CONST, head_dim,
            mB, mC, mSi, mSj, mH,
            hsB, hsC, hsSi, hsH, hsD,
            yB, yC, ySi, yH, yD,
            S_CONST=self.S_CONST, head_dim=head_dim,
            num_warps=1, num_stages=1,
        )

        # Return in bfloat16 as original
        return Y.to(torch.bfloat16)
