    pid = tl.program_id(0)
    if pid >= total_elems:
        return

    # Map pid -> (b, c, i, j, h)
    tmp = pid
    h = tmp % H
    tmp = tmp // H
    j = tmp % S
    tmp = tmp // S
    i = tmp % S
    tmp = tmp // S
    c = tmp % C
    b = tmp // C

    # Compute cumsum along j positions up to j, per (b, h, c)
    # cumsum = sum_{t=0..j} A[b, h, c, t]
    cumsum = 0.0
    for t in range(0, j + 1):
        a_idx = b * A_s0 + h * A_s1 + c * A_s2 + t * A_s3
        a_val = tl.load(A_ptr + a_idx)
        cumsum += a_val

    l_idx = b * L_s0 + c * L_s1 + i * L_s2 + j * L_s3 + h * L_s4
    tl.store(L_ptr + l_idx, l_val)
