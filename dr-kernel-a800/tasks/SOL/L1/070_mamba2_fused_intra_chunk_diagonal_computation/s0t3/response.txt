import torch
import triton
import triton.language as tl

# Kernel 1: Build L_flat = exp(cumsum(A_lower_tri)) linearized as [B*C*S*S*H]
@triton.jit
def build_L_flat_kernel(
    A_ptr,  # [B, H, C, S]
    L_flat_ptr,  # [B*C*S*S*H]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, S: tl.constexpr
):
    pid = tl.program_id(axis=0)
    total = B * C * S * S * H
    if pid >= total:
        return
    # Compute indices (b, c, i, j, h) from linear pid
    tmp = pid
    h = tmp % H
    tmp = tmp // H
    j = tmp % S
    tmp = tmp // S
    i = tmp % S
    tmp = tmp // S
    c = tmp % C
    b = tmp // C

    # Only lower-triangular positions (i >= j) are valid
    if i < j:
        # Store zero
        # Linear index in L_flat: ((b*C + c)*S + i)*S*H + j*H + h
        idx = ((b * C + c) * S + i) * (S * H) + (j * H + h)
        # Store float32 0.0
        tl.store(L_flat_ptr + idx, 0.0)
        return

    # Compute cumsum of A[b, h, c, :] up to j
    cum = 0.0
    # A[b, h, c, j] linear index: ((b*H + h)*C*S + c*S + j)
    # We loop j from 0 to i
    for jj in range(0, S):
        if jj > i:
            break
        a_idx = ((b * H + h) * C * S + c * S + jj)
        a_val = tl.load(A_ptr + a_idx)
        cum += a_val
    # L[i, j, h] = exp(cum)
    l_val = tl.exp(cum)
    # Store to L_flat at linear index: ((b*C + c)*S + i)*S*H + (j*H + h)
    idx = ((b * C + c) * S + i) * (S * H) + (j * H + h)
    tl.store(L_flat_ptr + idx, l_val)


# Kernel 2: Compute G_flat[i, j, h] = sum_n C_expanded[b, c, i, n, j, h] * B_expanded[b, c, j, n, i, h]
# C_expanded: [B, C, S, H, N], B_expanded: [B, C, S, H, N]
@triton.jit
def compute_G_flat_kernel(
    B_exp_ptr, C_exp_ptr, G_flat_ptr,
    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, H: tl.constexpr, N: tl.constexpr
):
    pid = tl.program_id(axis=0)
    total = Bsz * Csz * S * S * H
    if pid >= total:
        return
    # Compute indices (b, c, i, j, h)
    tmp = pid
    h = tmp % H
    tmp = tmp // H
    j = tmp % S
    tmp = tmp // S
    i = tmp % S
    tmp = tmp // S
    c = tmp % Csz
    b = tmp // Csz

    sum_val = 0.0
    # Loop over n in 0..N-1 (N=128), sum C_exp[b,c,i,n,j,h] * B_exp[b,c,j,n,i,h]
    for n in range(0, N):
        # C_exp[b, c, i, n, j, h]: offset = ((b*Csz + c)*S*H*N + i*H*N + n*H + j)*H + h
        C_off = ((b * Csz + c) * S * H * N + i * (H * N) + n * H + j) * H + h
        # B_exp[b, c, j, n, i, h]: offset = ((b*Csz + c)*S*H*N + j*H*N + n*H + i)*H + h
        B_off = ((b * Csz + c) * S * H * N + j * (H * N) + n * H + i) * H + h
        c_val = tl.load(C_exp_ptr + C_off)
        b_val = tl.load(B_exp_ptr + B_off)
        sum_val += c_val * b_val

    # Store G_flat at linear index: ((b*Csz + c)*S + i)*S*H + (j*H + h)
    G_idx = ((b * Csz + c) * S + i) * (S * H) + (j * H + h)
    tl.store(G_flat_ptr + G_idx, sum_val)


# Kernel 3: Compute Y_diag_flat[b, c, i, h, d] = sum_j G[b, c, i, j, h] * hidden_states[b, c, j, h, d]
# Y_flat: [B*C*S*H*head_dim], linearized. hidden_states: [B, C, S, H, head_dim]
@triton.jit
def compute_Y_diag_kernel(
    G_flat_ptr, hidden_ptr, Y_flat_ptr,
    B: tl.constexpr, C: tl.constexpr, S: tl.constexpr, H: tl.constexpr, head_dim: tl.constexpr
):
    pid = tl.program_id(axis=0)
    total = B * C * S * H * head_dim
    if pid >= total:
        return

    # Decode indices (b, c, i, h, d)
    d = pid % head_dim
    tmp = pid // head_dim
    h = tmp % H
    tmp = tmp // H
    i = tmp % S
    tmp = tmp // S
    c = tmp % C
    b = tmp // C

    # Sum over j
    sum_y = 0.0
    for j in range(0, S):
        # G_flat index: ((b*C + c)*S + i)*S*H + (j*H + h)
        G_idx = ((b * C + c) * S + i) * (S * H) + (j * H + h)
        g = tl.load(G_flat_ptr + G_idx)
        # hidden_ptr[b, c, j, h, d]: offset = ((b*C + c)*S*H*head_dim + j*H*head_dim + h*head_dim + d)
        h_off = ((b * C + c) * S * H * head_dim + j * (H * head_dim) + h * head_dim + d)
        hs_val = tl.load(hidden_ptr + h_off)
        sum_y += g * hs_val

    # Y_flat index: ((b*C + c)*S*H*head_dim + i*(H*head_dim) + h*head_dim + d)
    Y_idx = ((b * C + c) * S * H * head_dim + i * (H * head_dim) + h * head_dim + d)
    tl.store(Y_flat_ptr + Y_idx, sum_y)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # Constants as in the original reference
        self.CHUNK_SIZE = 128  # S
        self.NUM_HEADS = 32    # H
        self.N_GROUPS = 8      # groups in B/C

    def forward(self, hidden_states: torch.Tensor, A_cumsum: torch.Tensor, B: torch.Tensor, C: torch.Tensor) -> torch.Tensor:
        """
        Compute Y_diag = sum_j M * hidden_states, where:
        - hidden_states: [B, C, S, H, head_dim]
        - A_cumsum: [B, H, C, S]
        - B: [B, C, S, N_GROUPS, N]
        - C: [B, C, S, N_GROUPS, N]
        """
        assert hidden_states.is_cuda and A_cumsum.is_cuda and B.is_cuda and C.is_cuda, "All tensors must be on CUDA for Triton."
        Bsz, Csz, S, H, head_dim = hidden_states.shape
        assert S == self.CHUNK_SIZE, f"hidden_states chunk size must be {self.CHUNK_SIZE}, got {S}"
        assert H == self.NUM_HEADS, f"num_heads must be {self.NUM_HEADS}, got {H}"
        # A_cumsum is [B, H, C, S]; we'll use it to build L
        A = A_cumsum  # already [B, H, C, S]
        # Prepare expanded B and C along H (repeat_interleave by 4)
        B_exp = B.repeat_interleave(self.NUM_HEADS // self.N_GROUPS, dim=3)  # [B, C, S, H, N]
        C_exp = C.repeat_interleave(self.NUM_HEADS // self.N_GROUPS, dim=3)  # [B, C, S, H, N]
        # Ensure contiguity
        A = A.contiguous()
        B_exp = B_exp.contiguous()
        C_exp = C_exp.contiguous()
        hidden_states = hidden_states.contiguous()

        # Allocate flat buffers (float32 for computation)
        L_flat = torch.empty(Bsz * Csz * S * S * H, dtype=torch.float32, device=hidden_states.device)
        G_flat = torch.empty(Bsz * Csz * S * S * H, dtype=torch.float32, device=hidden_states.device)
        Y_flat = torch.empty(Bsz * Csz * S * H * head_dim, dtype=torch.float32, device=hidden_states.device)

        # Launch build_L_flat_kernel
        total_L = Bsz * Csz * S * S * H
        grid_L = (total_L,)
        build_L_flat_kernel[grid_L](
            A, L_flat,
            B=Bsz, C=Csz, H=H, S=S
        )

        # Launch compute_G_flat_kernel
        total_G = Bsz * Csz * S * S * H
        grid_G = (total_G,)
        compute_G_flat_kernel[grid_G](
            B_exp, C_exp, G_flat,
            Bsz=Bsz, Csz=Csz, S=S, H=H, N=128  # N is state_size in original (128)
        )

        # Launch compute_Y_diag_kernel
        total_Y = Bsz * Csz * S * H * head_dim
        grid_Y = (total_Y,)
        compute_Y_diag_kernel[grid_Y](
            G_flat, hidden_states, Y_flat,
            B=Bsz, C=Csz, S=S, H=H, head_dim=head_dim
        )

        # Reshape Y_flat to [B, C, S, H, head_dim]
        Y = Y_flat.view(Bsz, Csz, S, H, head_dim)
        # Return in bfloat16 to match original behavior
        return Y.to(torch.bfloat16)

    total = Bsz * Csz * S * S * H
    if pid >= total:

    tmp = pid

    tmp = tmp // S

    tmp = tmp // S

        C_off = ((b * Csz + c) * S * H * N + i * (H * N) + n * H + j) * H + h
        # B_exp[b, c, j, n, i, h] linear index:
        # ((b*Csz + c)*S*H*N + j*H*N + n*H + i)*H + h
        B_off = ((b * Csz + c) * S * H * N + j * (H * N) + n * H + i) * H + h
        c_val = tl.load(C_exp_ptr + C_off)
        b_val = tl.load(B_exp_ptr + B_off)
        sum_val += c_val * b_val

    # Store to G_flat at linear index: ((b*Csz + c)*S + i)*S*H + (j*H + h)
    G_idx = ((b * Csz + c) * S + i) * (S * H) + (j * H + h)
    tl.store(G_flat_ptr + G_idx, sum_val)


# Kernel 3: Compute Y_diag_flat[b, c, i, h, d] = sum_j G[b, c, i, j, h] * hidden_states[b, c, j, h, d]
# hidden_states: [B, C, S, H, head_dim]
@triton.jit
def compute_Y_diag_kernel(
    G_flat_ptr, hidden_ptr, Y_flat_ptr,
    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, H: tl.constexpr, head_dim: tl.constexpr
):
    pid = tl.program_id(axis=0)
    total = Bsz * Csz * S * H * head_dim
    if pid >= total:
        return

    # Decode (b, c, i, h, d)
    d = pid % head_dim
    tmp = pid // head_dim
    h = tmp % H
    tmp = tmp // H
    i = tmp % S
    tmp = tmp // S
    c = tmp % Csz
    b = tmp // Csz

    sum_y = 0.0
    for j in range(0, S):
        # G_flat index: ((b*Csz + c)*S + i)*S*H + (j*H + h)
        G_idx = ((b * Csz + c) * S + i) * (S * H) + (j * H + h)
        g = tl.load(G_flat_ptr + G_idx)
        # hidden_ptr[b, c, j, h, d] linear index:
        # ((b*Csz + c)*S*H*head_dim + j*H*head_dim + h*head_dim + d)
        h_off = ((b * Csz + c) * S * H * head_dim + j * (H * head_dim) + h * head_dim + d)
        hs_val = tl.load(hidden_ptr + h_off)
        sum_y += g * hs_val

    # Store to Y_flat at linear index: ((b*Csz + c)*S*H*head_dim + i*(H*head_dim) + h*head_dim + d)
    Y_idx = ((b * Csz + c) * S * H * head_dim + i * (H * head_dim) + h * head_dim + d)
    tl.store(Y_flat_ptr + Y_idx, sum_y)
