import torch
import triton
import triton.language as tl

# Triton kernels: all math in Triton, no torch ops in forward

# 1) Build L: L[b, h, i, k, j] = exp(cumsum(A[b, h, i, 0:j])) for i >= j, else 0
@triton.jit
def build_L_kernel(
    A_ptr,               # *float32, [B, NUM_HEADS, NUM_CHUNKS, CHUNK_SIZE]
    L_ptr,               # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,     # int32: batch_size, num_chunks
    CHUNK_SIZE: tl.constexpr
):
    b = tl.program_id(0)
    i = tl.program_id(1)  # chunk position i (same as 'c' in original code)
    h = tl.program_id(2)  # head index
    # Only launch when b < B_size and i < NC_size, but Triton grid ensures bounds
    for k in range(CHUNK_SIZE):
        # compute cumsum along j from 0..k
        # A[b, h, i, j] is contiguous along j
        acc = 0.0
        for jj in range(k + 1):
            a_val = tl.load(A_ptr + b * NUM_HEADS * NUM_CHUNKS * CHUNK_SIZE + h * NUM_CHUNKS * CHUNK_SIZE + i * CHUNK_SIZE + jj)
            acc += a_val
        L_val = tl.exp(acc)
        for j in range(CHUNK_SIZE):
            # lower-triangular mask: i >= j
            if (i >= j):
                tl.store(L_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h, L_val)
            else:
                tl.store(L_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h, 0.0)

# 2) Compute G: G[i, j, h] = sum over s of C_expanded[i, k, h, s] * B_expanded[j, k, h, s]
@triton.jit
def compute_G_kernel(
    B_exp_ptr,           # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    C_exp_ptr,           # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    G_ptr,               # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,     # int32: batch_size, num_chunks (runtime)
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr,
    HEAD_DIM: tl.constexpr
):
    # Grid: (B, NC, NUM_HEADS)
    b = tl.program_id(0)
    i = tl.program_id(1)  # source chunk position
    h = tl.program_id(2)  # head index

    # For each destination j, compute G[i, j, h]
    for j in range(CHUNK_SIZE):
        G_val = 0.0  # accumulate scalar
        for k in range(CHUNK_SIZE):
            # C_exp[b, i, k, h, :] and B_exp[b, j, k, h, :]
            # We load vectors of length HEAD_DIM and compute dot-product
            dot = 0.0
            for s in range(HEAD_DIM):  # state_dim loop
                c_vec = tl.load(C_exp_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + i * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + s)
                b_vec = tl.load(B_exp_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + j * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + s)
                dot += c_vec * b_vec
            G_val += dot
        # store G[i, j, h]
        tl.store(G_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h, G_val)

# 3) Multiply M = G * L (elementwise with mask i >= j)
@triton.jit
def multiply_LG_kernel(
    G_ptr,               # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    L_ptr,               # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    M_ptr,               # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,     # int32
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    h = tl.program_id(2)
    for k in range(CHUNK_SIZE):
        for j in range(CHUNK_SIZE):
            # lower-triangular mask: i >= j
            if (i >= j):
                G_val = tl.load(G_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h)
                L_val = tl.load(L_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h)
                M_val = G_val * L_val
            else:
                M_val = 0.0
            tl.store(M_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h, M_val)

# 4) Contract to Y_diag: Y[b, i, k, h, d] = sum_j M[b, i, k, j, h] * hidden[b, i, k, j, h, d]
@triton.jit
def contract_to_Ydiag_kernel(
    M_ptr,               # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    hidden_ptr,          # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    Y_ptr,               # *bfloat16, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    B_size, NC_size,     # int32
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr,
    HEAD_DIM: tl.constexpr
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    k = tl.program_id(2)
    h = tl.program_id(3)
    for d in range(HEAD_DIM):
        # Compute scalar Y[b, i, k, h, d]
        Y_val = 0.0
        for j in range(CHUNK_SIZE):
            M_val = tl.load(M_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + i * (CHUNK_SIZE * CHUNK_SIZE * NUM_HEADS) + k * (CHUNK_SIZE * NUM_HEADS) + j * NUM_HEADS + h)
            hidden_vec = tl.load(hidden_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + i * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + d)
            Y_val += M_val * hidden_vec
        # Store as bfloat16
        tl.store(Y_ptr + b * (NUM_CHUNKS * CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + i * (CHUNK_SIZE * NUM_HEADS * HEAD_DIM) + k * (NUM_HEADS * HEAD_DIM) + h * HEAD_DIM + d, tl.cast(Y_val, tl.bfloat16))

class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
