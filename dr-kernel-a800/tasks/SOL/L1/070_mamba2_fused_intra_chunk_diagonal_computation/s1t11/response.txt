        )

        # 3) Final reduction to Y_diag: [B_batch, B_n, B_K, B_heads, B_D]
        out = torch.empty((B_batch, B_n, B_K, B_heads, B_D), dtype=torch.float32, device=device)

            CHUNK=B_K, BLOCK_D=64, num_warps=4, num_stages=2
        )

        # Return in bfloat16 to match original
        return out.to(torch.bfloat16)

        tl.store(L_ptr + L_off, tl.exp(cumsum))
