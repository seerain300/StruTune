import torch
import triton
import triton.language as tl


# 1) Triton kernel: build L = lower-triangular exp(cumsum(A)) per (b, c, h)
# Input: A_cumsum [B, C, L, H] float32
# Output: L_out [B, C, L, L, H] float32
@triton.jit
def build_L_kernel(
    A_ptr,  # *float32
    L_ptr,  # *float32
    B: tl.constexpr,  # batch size
    C: tl.constexpr,  # num_chunks
    H: tl.constexpr,  # num_heads
    L_dim: tl.constexpr,  # chunk_size
    A_stride_b, A_stride_c, A_stride_l, A_stride_h,
    L_stride_b, L_stride_c, L_stride_i, L_stride_j, L_stride_h,
):
    # program ids
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)  # row
    j = tl.program_id(4)  # col

    # bounds check (in case grid > L)
    if (i >= L_dim) or (j >= L_dim) or (b >= B) or (c >= C) or (h >= H):
        return

    # Compute cumulative sum along L_dim for A[b, c, :, h]
    # cumsum_i = sum_{k=0..i} A[b, c, k, h]
    cumsum_i = 0.0
    # We assume A is zero outside i, but we can compute up to L_dim by masking: A[i, h] exists for i < L_dim
    # Loop over k from 0 to L_dim
    for k in range(0, L_dim):
        # addr = b*A_stride_b + c*A_stride_c + k*A_stride_l + h*A_stride_h
        addr = b * A_stride_b + c * A_stride_c + k * A_stride_l + h * A_stride_h
        # load A[b, c, k, h]
        a_val = tl.load(A_ptr + addr)
        cumsum_i += a_val

    # L[i, j] = exp(cumsum_i) if j <= i else 0
    if j <= i:
        l_val = tl.exp(cumsum_i)
    else:
        l_val = 0.0

    # Store to L_out[b, c, i, j, h]
    L_addr = b * L_stride_b + c * L_stride_c + i * L_stride_i + j * L_stride_j + h * L_stride_h
    tl.store(L_ptr + L_addr, l_val)


# 2) Triton kernel: expand groups -> heads for B (repeat_interleave with GROUP_EXPAND=4)
# Input: B_in [B, C, L, G, S] float32, G = N_GROUPS
# Output: B_exp [B, C, L, H, S] float32
@triton.jit
def expand_groups_B_kernel(
    B_in_ptr, B_out_ptr,
    B: tl.constexpr, C: tl.constexpr, L: tl.constexpr, G: tl.constexpr, H: tl.constexpr, S: tl.constexpr,
    B_in_stride_b, B_in_stride_c, B_in_stride_l, B_in_stride_g, B_in_stride_s,
    B_out_stride_b, B_out_stride_c, B_out_stride_l, B_out_stride_h, B_out_stride_s,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)  # i in [0, L)
    h = tl.program_id(3)  # h in [0, H)
    s = tl.program_id(4)  # s in [0, S)

    # h maps to group g
    g = (h // 4)  # GROUP_EXPAND=4, NUM_HEADS=32, N_GROUPS=8, so h in [0..31], g in [0..7]
    if g >= G:
        return

    # input address: [b, c, i, g, s]
    in_addr = b * B_in_stride_b + c * B_in_stride_c + i * B_in_stride_l + g * B_in_stride_g + s * B_in_stride_s
    val = tl.load(B_in_ptr + in_addr)

    # output address: [b, c, i, h, s]
    out_addr = b * B_out_stride_b + c * B_out_stride_c + i * B_out_stride_l + h * B_out_stride_h + s * B_out_stride_s
    tl.store(B_out_ptr + out_addr, val)


# 3) Triton kernel: expand groups -> heads for C (repeat_interleave with GROUP_EXPAND=4)
# Same as expand_groups_B_kernel, just for C.
@triton.jit
def expand_groups_C_kernel(
    C_in_ptr, C_out_ptr,
    B: tl.constexpr, C: tl.constexpr, L: tl.constexpr, G: tl.constexpr, H: tl.constexpr, S: tl.constexpr,
    C_in_stride_b, C_in_stride_c, C_in_stride_l, C_in_stride_g, C_in_stride_s,
    C_out_stride_b, C_out_stride_c, C_out_stride_l, C_out_stride_h, C_out_stride_s,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    h = tl.program_id(3)
    s = tl.program_id(4)
    g = (h // 4)
    if g >= G:
        return

    in_addr = b * C_in_stride_b + c * C_in_stride_c + i * C_in_stride_l + g * C_in_stride_g + s * C_in_stride_s
    val = tl.load(C_in_ptr + in_addr)
    out_addr = b * C_out_stride_b + c * C_out_stride_c + i * C_out_stride_l + h * C_out_stride_h + s * C_out_stride_s
    tl.store(C_out_ptr + out_addr, val)


# 4) Triton kernel: compute G[b, c, i, j, h] = sum_s C_exp[b,c,i,h,s] * B_exp[b,c,j,h,s]
# Input: C_exp [B, C, L, H, S], B_exp [B, C, L, H, S]
# Output: G_out [B, C, L, L, H]
@triton.jit
def compute_G_kernel(
    C_exp_ptr, B_exp_ptr, G_out_ptr,
    B: tl.constexpr, C: tl.constexpr, L: tl.constexpr, H: tl.constexpr, S: tl.constexpr,
    C_stride_b, C_stride_c, C_stride_i, C_stride_h, C_stride_s,
    B_stride_b, B_stride_c, B_stride_j, B_stride_h, B_stride_s,
    G_stride_b, G_stride_c, G_stride_i, G_stride_j, G_stride_h,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)  # row
    j = tl.program_id(3)  # col
    h = tl.program_id(4)

    if (b >= B) or (c >= C) or (i >= L) or (j >= L) or (h >= H):
        return

    g_val = 0.0
    # loop over state_size S
    for s in range(0, S):
        c_in_addr = b * C_stride_b + c * C_stride_c + i * C_stride_i + h * C_stride_h + s * C_stride_s
        b_in_addr = b * B_stride_b + c * B_stride_c + j * B_stride_j + h * B_stride_h + s * B_stride_s
        c_val = tl.load(C_exp_ptr + c_in_addr)
        b_val = tl.load(B_exp_ptr + b_in_addr)
        g_val += c_val * b_val

    # store G[b, c, i, j, h] = g_val
    G_addr = b * G_stride_b + c * G_stride_c + i * G_stride_i + j * G_stride_j + h * G_stride_h
    tl.store(G_out_ptr + G_addr, g_val)


# 5) Triton kernel: elementwise multiply G * L -> M
# Input: G [B, C, L, L, H], L [B, C, L, L, H]
# Output: M [B, C, L, L, H]
@triton.jit
def elemwise_mul_GL_kernel(
    G_ptr, L_ptr, M_ptr,
    B: tl.constexpr, C: tl.constexpr, L: tl.constexpr, H: tl.constexpr,
    G_stride_b, G_stride_c, G_stride_i, G_stride_j, G_stride_h,
    L_stride_b, L_stride_c, L_stride_i, L_stride_j, L_stride_h,
    M_stride_b, M_stride_c, M_stride_i, M_stride_j, M_stride_h,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)
    if (b >= B) or (c >= C) or (i >= L) or (j >= L) or (h >= H):
        return

    g_addr = b * G_stride_b + c * G_stride_c + i * G_stride_i + j * G_stride_j + h * G_stride_h
    l_addr = b * L_stride_b + c * L_stride_c + i * L_stride_i + j * L_stride_j + h * L_stride_h
    g_val = tl.load(G_ptr + g_addr)
    l_val = tl.load(L_ptr + l_addr)
    m_val = g_val * l_val
    m_addr = b * M_stride_b + c * M_stride_c + i * M_stride_i + j * M_stride_j + h * M_stride_h
    tl.store(M_ptr + m_addr, m_val)


# 6) Triton kernel: compute Y_diag[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden[b, c, j, h, d]
# Input: M [B, C, L, L, H], hidden [B, C, L, H, D]
# Output: Y [B, C, L, H, D]
@triton.jit
def compute_Y_diag_kernel(
    M_ptr, hidden_ptr, Y_ptr,
    B: tl.constexpr, C: tl.constexpr, L: tl.constexpr, H: tl.constexpr, D: tl.constexpr,
    M_stride_b, M_stride_c, M_stride_i, M_stride_j, M_stride_h,
    hidden_stride_b, hidden_stride_c, hidden_stride_l, hidden_stride_h, hidden_stride_d,
    Y_stride_b, Y_stride_c, Y_stride_i, Y_stride_h, Y_stride_d,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)  # row
    h = tl.program_id(3)  # head
    d = tl.program_id(4)  # feature dim

    if (b >= B) or (c >= C) or (i >= L) or (h >= H) or (d >= D):
        return

    y_val = 0.0
    for j in range(0, L):
        M_addr = b * M_stride_b + c * M_stride_c + i * M_stride_i + j * M_stride_j + h * M_stride_h
        hidden_addr = b * hidden_stride_b + c * hidden_stride_c + j * hidden_stride_l + h * hidden_stride_h + d * hidden_stride_d
        m_val = tl.load(M_ptr + M_addr)
        hidden_val = tl.load(hidden_ptr + hidden_addr)
        y_val += m_val * hidden_val

    Y_addr = b * Y_stride_b + c * Y_stride_c + i * Y_stride_i + h * Y_stride_h + d * Y_stride_d
    tl.store(Y_ptr + Y_addr, y_val)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, A_cumsum: torch.Tensor, B: torch.Tensor, C: torch.Tensor) -> torch.Tensor:
        # Shapes as per original: hidden_states: [B, C, L, H, D]
        batch_size, num_chunks, chunk_size, num_heads, head_dim = hidden_states.shape

        # Constants from original
        NUM_HEADS = 32
        N_GROUPS = 8
        GROUP_EXPAND = 4  # NUM_HEADS // N_GROUPS

        # Ensure dtype float32 for kernels
        device = hidden_states.device
        A = A_cumsum.to(torch.float32).contiguous()  # [B, C, L, H]
        B_in = B.to(torch.float32).contiguous()      # [B, C, L, G, S]
        C_in = C.to(torch.float32).contiguous()      # [B, C, L, G, S]

        # 1) Build L in Triton: L[b, c, i, j, h] = exp(sum_{k=0..i} A[b, c, k, h]) if j <= i else 0
        # Allocate L
        L = torch.empty((batch_size, num_chunks, chunk_size, chunk_size, num_heads), dtype=torch.float32, device=device)
        grid_L = (batch_size, num_chunks, num_heads, chunk_size, chunk_size)
        build_L_kernel[grid_L](
            A, L,
            batch_size, num_chunks, num_heads, chunk_size,
            A.stride(0), A.stride(1), A.stride(2), A.stride(3),
            L.stride(0), L.stride(1), L.stride(2), L.stride(3), L.stride(4),
        )

        # 2) Expand B and C from groups to heads via Triton repeat_interleave
        S_size = B_in.shape[4]  # state_size (same for B_in and C_in)
        B_exp = torch.empty((batch_size, num_chunks, chunk_size, num_heads, S_size), dtype=torch.float32, device=device)
        C_exp = torch.empty((batch_size, num_chunks, chunk_size, num_heads, S_size), dtype=torch.float32, device=device)

        grid_expand = (batch_size, num_chunks, chunk_size, num_heads, S_size)
        expand_groups_B_kernel[grid_expand](
            B_in, B_exp,
            batch_size, num_chunks, chunk_size, N_GROUPS, num_heads, S_size,
            B_in.stride(0), B_in.stride(1), B_in.stride(2), B_in.stride(3), B_in.stride(4),
            B_exp.stride(0), B_exp.stride(1), B_exp.stride(2), B_exp.stride(3), B_exp.stride(4),
        )
        expand_groups_C_kernel[grid_expand](
            C_in, C_exp,
            batch_size, num_chunks, chunk_size, N_GROUPS, num_heads, S_size,
            C_in.stride(0), C_in.stride(1), C_in.stride(2), C_in.stride(3), C_in.stride(4),
            C_exp.stride(0), C_exp.stride(1), C_exp.stride(2), C_exp.stride(3), C_exp.stride(4),
        )

        # 3) Compute G in Triton: G[b, c, i, j, h] = sum_s C_exp[b,c,i,h,s] * B_exp[b,c,j,h,s]
        G = torch.empty((batch_size, num_chunks, chunk_size, chunk_size, num_heads), dtype=torch.float32, device=device)
        grid_G = (batch_size, num_chunks, chunk_size, chunk_size, num_heads)
        compute_G_kernel[grid_G](
            C_exp, B_exp, G,
            batch_size, num_chunks, chunk_size, num_heads, S_size,
            C_exp.stride(0), C_exp.stride(1), C_exp.stride(2), C_exp.stride(3), C_exp.stride(4),
            B_exp.stride(0), B_exp.stride(1), B_exp.stride(2), B_exp.stride(3), B_exp.stride(4),
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
        )

        # 4) Elementwise multiply G * L to get M in Triton
        M = torch.empty_like(G)
        grid_M = (batch_size, num_chunks, chunk_size, chunk_size, num_heads)
        elemwise_mul_GL_kernel[grid_M](
            G, L, M,
            batch_size, num_chunks, chunk_size, num_heads,
            G.stride(0), G.stride(1), G.stride(2), G.stride(3), G.stride(4),
            L.stride(0), L.stride(1), L.stride(2), L.stride(3), L.stride(4),
            M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
        )

        # 5) Compute Y_diag in Triton: Y[b, c, i, h, d] = sum_j M[b,c,i,j,h] * hidden[b,c,j,h,d]
        hidden_f32 = hidden_states.to(torch.float32).contiguous()  # [B, C, L, H, D]
        Y = torch.empty((batch_size, num_chunks, chunk_size, num_heads, head_dim), dtype=torch.float32, device=device)

        grid_Y = (batch_size, num_chunks, chunk_size, num_heads, head_dim)
        compute_Y_diag_kernel[grid_Y](
            M, hidden_f32, Y,
            batch_size, num_chunks, chunk_size, num_heads, head_dim,
            M.stride(0), M.stride(1), M.stride(2), M.stride(3), M.stride(4),
            hidden_f32.stride(0), hidden_f32.stride(1), hidden_f32.stride(2), hidden_f32.stride(3), hidden_f32.stride(4),
            Y.stride(0), Y.stride(1), Y.stride(2), Y.stride(3), Y.stride(4),
        )

        # Return in bfloat16 (original returns bfloat16)
        return Y.to(torch.bfloat16)
