import torch
import triton
import triton.language as tl


@triton.jit
def compute_G_per_i_h_kernel(
    B_ptr,  # *float32, [B, num_chunks, chunk_size, num_heads, state_size]
    C_ptr,  # *float32, [B, num_chunks, chunk_size, num_heads, state_size]
    G_ptr,  # *float32, [B, num_chunks, chunk_size, num_heads]  (we'll only store per (i, h) across j)
    B_stride_b, B_stride_ci, B_stride_ck, B_stride_ch, B_stride_cs,
    C_stride_b, C_stride_ci, C_stride_ck, C_stride_ch, C_stride_cs,
    G_stride_b, G_stride_ci, G_stride_ch,
    num_chunks, chunk_size, num_heads, state_size,
    BLOCK_J: tl.constexpr,
    BLOCK_S: tl.constexpr,
):
    # Each program handles one (b, i, h) triplet and computes G for all j
    b = tl.program_id(0)
    i = tl.program_id(1)
    h = tl.program_id(2)

    # Accumulator for G[j, h] across all j
    G_j = tl.zeros([BLOCK_J], dtype=tl.float32)

    # Iterate over j (compile-time constant)
    for j in range(BLOCK_J):
        # Initialize G_j[j] to 0
        G_j[j] = 0.0
        # Compute G_j[j] = sum over k and s of C[b, i, k, h, s] * B[b, j, k, h, s]
        # Iterate k
        for k in range(chunk_size):
            # Iterate over state dimension with compile-time BLOCK_S
            for s in range(BLOCK_S):
                # Compute pointers
                B_ptr_val = B_ptr \
                    + b * B_stride_b \
                    + j * B_stride_ci \
                    + k * B_stride_ck \
                    + h * B_stride_ch \
                    + s * B_stride_cs

                C_ptr_val = C_ptr \
                    + b * C_stride_b \
                    + i * C_stride_ci \
                    + k * C_stride_ck \
                    + h * C_stride_ch \
                    + s * C_stride_cs

                # Load as float32
                B_val = tl.load(B_ptr_val, eviction_policy='evict_last').to(tl.float32)
                C_val = tl.load(C_ptr_val, eviction_policy='evict_last').to(tl.float32)

                G_j[j] += C_val * B_val

        # Store G_j[j] into G[b, i, j, h]
        G_ptr_val = G_ptr + b * G_stride_b + i * G_stride_ci + j * G_stride_ch + h
        tl.store(G_ptr_val, G_j[j])

    return


@triton.jit
def contract_to_Y_diag_kernel(
    G_ptr,            # *float32, [B, num_chunks, chunk_size, num_heads]
    hidden_ptr,       # *bfloat16, [B, num_chunks, chunk_size, num_heads, head_dim]
    Y_ptr,            # *bfloat16, [B, num_chunks, chunk_size, num_heads, head_dim]
    B_stride_b, B_stride_ci, B_stride_ck, B_stride_ch, B_stride_cd,
    G_stride_b, G_stride_ci, G_stride_ch,
    Y_stride_b, Y_stride_ci, Y_stride_ck, Y_stride_ch, Y_stride_cd,
    batch_size, num_chunks, chunk_size, num_heads, head_dim,
    BLOCK_D: tl.constexpr,
):
    # Grid over (b, i, k, h). Each program writes the whole head_dim for one (b, i, k, h)
    b = tl.program_id(0)
    i = tl.program_id(1)
    k = tl.program_id(2)
    h = tl.program_id(3)

    # Compute output vector
    for d in range(BLOCK_D):
        # Initialize sum
        sum_val = 0.0
        # Sum over j
        for j in range(chunk_size):
            G_val = tl.load(G_ptr + b * G_stride_b + i * G_stride_ci + j * G_stride_ch + h).to(tl.float32)
            hidden_ptr_val = hidden_ptr \
                + b * B_stride_b \
                + i * B_stride_ci \
                + k * B_stride_ck \
                + h * B_stride_ch \
                + d * B_stride_cd
            hidden_val = tl.load(hidden_ptr_val, eviction_policy='evict_last').to(tl.float32)
            sum_val += G_val * hidden_val

        # Store to Y
        Y_ptr_val = Y_ptr + b * Y_stride_b + i * Y_stride_ci + k * Y_stride_ck + h * Y_stride_ch + d * Y_stride_cd
        tl.store(Y_ptr_val, sum_val.to(tl.bfloat16))

    return

import torch

# Original computation (unchanged), required for correctness
@torch.no_grad()
def run(
    hidden_states: torch.Tensor,
    A_cumsum: torch.Tensor,
    B: torch.Tensor,
    C: torch.Tensor
) -> torch.Tensor:
    """
    Compute intra-chunk diagonal output Y_diag for Mamba2 SSD.
    Steps:
    1) Build segment_sum(A) to get cumulative decay factors, apply lower-triangular mask for causality, exp to get L.
    2) Expand B and C from n_groups to num_heads, compute G = sum_s C[..., s] * B[..., s].
    3) Apply mask L to G to get M.
    4) Contract M with hidden_states to produce Y_diag of shape [B, num_chunks, chunk_size, num_heads, head_dim].
    """
    # Constants
    CHUNK_SIZE = 128
    NUM_HEADS = 32
    N_GROUPS = 8

    batch_size, num_chunks, chunk_size, num_heads, head_dim = hidden_states.shape

    # Step 1: Build L
    # A_cumsum: [batch, num_heads, num_chunks, chunk_size]
    A_cumsum_f32 = A_cumsum.to(torch.float32)
    # Expand to [batch, num_heads, num_chunks, chunk_size, chunk_size]
    A_expanded = A_cumsum_f32[..., None].expand(*A_cumsum_f32.size(), CHUNK_SIZE)

    # Lower-triangular mask (exclude diagonal)
    mask = torch.tril(
        torch.ones(CHUNK_SIZE, CHUNK_SIZE, device=A_cumsum.device, dtype=torch.bool),
        diagonal=-1
    )
    A_masked = A_expanded.masked_fill(~mask, 0)

    # Cumulative sum along source dimension (axis=-2)
    A_cumsum_seg = torch.cumsum(A_masked, dim=-2)

    # Include diagonal
    mask_with_diag = torch.tril(
        torch.ones(CHUNK_SIZE, CHUNK_SIZE, device=A_cumsum.device, dtype=torch.bool),
        diagonal=0
    )
    segment_sum_A = A_cumsum_seg.masked_fill(~mask_with_diag, 0)

    # Apply exponential to get causal decay mask L
    L = torch.exp(segment_sum_A)  # [batch, num_heads, num_chunks, chunk_size, chunk_size]

    # Step 2: Expand B and C from n_groups to num_heads
    B_f32 = B.to(torch.float32)
    C_f32 = C.to(torch.float32)
    # NUM_HEADS_PER_GROUP = 4
    B_expanded = B_f32.repeat_interleave(NUM_HEADS // N_GROUPS, dim=3)
    C_expanded = C_f32.repeat_interleave(NUM_HEADS // N_GROUPS, dim=3)

    # Compute G: G[i, j, h] = sum_s C[i, s] * B[j, s] -> [batch, num_chunks, chunk_size, chunk_size, num_heads]
    C_for_G = C_expanded[:, :, :, None, :, :]  # [B, nc, ch, 1, num_heads, state_size]
    B_for_G = B_expanded[:, :, None, :, :, :]  # [B, nc, 1, ch, num_heads, state_size]
    G = (C_for_G * B_for_G).sum(dim=-1)        # [B, nc, ch, ch, num_heads]

    # Step 3: Apply L to G: M = G * L
    L_permuted = L.permute(0, 2, 3, 4, 1)     # [B, nc, ch, ch, num_heads]
    M = G * L_permuted

    # Step 4: Contract M with hidden_states to get Y_diag
    hidden_states_f32 = hidden_states.to(torch.float32)
    M_expanded = M[..., None]                 # [B, nc, ch, ch, num_heads, 1]
    hidden_expanded = hidden_states_f32[:, :, None, :, :, :]  # [B, nc, 1, ch, num_heads, head_dim]
    Y_diag = (M_expanded * hidden_expanded).sum(dim=3)        # [B, nc, ch, num_heads, head_dim]

    return Y_diag.to(torch.bfloat16)
