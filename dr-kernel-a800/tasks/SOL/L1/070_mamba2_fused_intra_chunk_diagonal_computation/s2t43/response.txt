import torch
import triton
import triton.language as tl

# Compile-time constants for the given model
CHUNK_SIZE: tl.constexpr = 128
NUM_HEADS: tl.constexpr = 32
HEAD_DIM: tl.constexpr = 128
N_GROUPS: tl.constexpr = 8

# Triton kernels: all math in Triton, no torch ops in forward

# 1) Build L: L[b, h, i, k, j] = exp(cumsum(A[b, h, i, 0:j])) for i >= j, else 0
@triton.jit
def build_L_kernel(
    A_ptr,                # *float32, [B, NUM_HEADS, NUM_CHUNKS, CHUNK_SIZE]
    L_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,      # int32: batch_size and num_chunks (not used here, but kept for generality)
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr,
):
    b = tl.program_id(0)   # batch
    i = tl.program_id(1)   # chunk position i
    h = tl.program_id(2)   # head index
    # We will fill L[b, i, k, j, h] for all k,j
    for k in range(CHUNK_SIZE):
        # lower-triangular mask: i >= j
        for j in range(CHUNK_SIZE):
            if i >= j:
                # cumsum along j (source positions)
                ssum = 0.0
                for t in range(j + 1):
                    # A layout: [B, NUM_HEADS, NUM_CHUNKS, CHUNK_SIZE]
                    # index = (((b * NUM_HEADS + h) * NUM_CHUNKS + i) * CHUNK_SIZE + t)
                    idx = (((b * NUM_HEADS + h) * NUM_CHUNKS + i) * CHUNK_SIZE + t)
                    a_val = tl.load(A_ptr + idx)
                    ssum += a_val
                L_val = tl.exp(ssum)
            else:
                L_val = 0.0
            # store L[b, i, k, j, h]
            L_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * CHUNK_SIZE + j) * NUM_HEADS + h
            tl.store(L_ptr + L_index, L_val)

# 2) Compute G[i, j, h] = sum_s C_expanded[i, k, h, s] * B_expanded[j, k, h, s]
@triton.jit
def compute_G_kernel(
    B_exp_ptr,            # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    C_exp_ptr,            # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    G_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr,
    HEAD_DIM: tl.constexpr,
):
    b = tl.program_id(0)   # batch
    i = tl.program_id(1)   # chunk position i
    h = tl.program_id(2)   # head index
    # Compute G for all j (we will fill entire G[b, i, :, :, h])
    for j in range(CHUNK_SIZE):
        G_vals = tl.zeros([CHUNK_SIZE], dtype=tl.float32)
        for k in range(CHUNK_SIZE):
            for s in range(HEAD_DIM):
                # B_exp and C_exp layout: [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
                B_idx = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * NUM_HEADS + h) * HEAD_DIM + s
                C_idx = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * NUM_HEADS + h) * HEAD_DIM + s
                B_val = tl.load(B_exp_ptr + B_idx)
                C_val = tl.load(C_exp_ptr + C_idx)
                G_vals[k] += C_val * B_val
        # Store G[b, i, k, j, h] for all k
        for k in range(CHUNK_SIZE):
            G_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * CHUNK_SIZE + j) * NUM_HEADS + h
            tl.store(G_ptr + G_index, G_vals[k])

# 3) Multiply M = G * L (elementwise), with same lower-triangular mask (i >= j)
@triton.jit
def multiply_LG_kernel(
    G_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    L_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    M_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    B_size, NC_size,
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr,
):
    b = tl.program_id(0)   # batch
    i = tl.program_id(1)   # chunk position i
    h = tl.program_id(2)   # head index
    for k in range(CHUNK_SIZE):
        for j in range(CHUNK_SIZE):
            if i >= j:
                G_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * CHUNK_SIZE + j) * NUM_HEADS + h
                L_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * CHUNK_SIZE + j) * NUM_HEADS + h
                M_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * CHUNK_SIZE + j) * NUM_HEADS + h
                G_val = tl.load(G_ptr + G_index)
                L_val = tl.load(L_ptr + L_index)
                M_val = G_val * L_val
            else:
                M_val = 0.0
            tl.store(M_ptr + M_index, M_val)

# 4) Contract to Y_diag: Y[b, i, k, h, d] = sum_j M[b, i, k, j, h] * hidden[b, i, k, j, h, d]
@triton.jit
def contract_to_Ydiag_kernel(
    M_ptr,                # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
    hidden_ptr,           # *float32, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    Y_ptr,                # *bfloat16, [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM]
    B_size, NC_size,
    CHUNK_SIZE: tl.constexpr,
    NUM_HEADS: tl.constexpr,
    HEAD_DIM: tl.constexpr,
):
    b = tl.program_id(0)   # batch
    i = tl.program_id(1)   # chunk position i
    k = tl.program_id(2)   # inner chunk position k
    h = tl.program_id(3)   # head index
    for d in range(HEAD_DIM):
        Y_row = tl.zeros([HEAD_DIM], dtype=tl.float32)
        for j in range(CHUNK_SIZE):
            M_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * CHUNK_SIZE + j) * NUM_HEADS + h
            M_val = tl.load(M_ptr + M_index)
            H_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * NUM_HEADS + h) * HEAD_DIM + d
            H_val = tl.load(hidden_ptr + H_index)
            Y_row[d] += M_val * H_val
        # Store Y[b, i, k, h, d] as bfloat16
        Y_index = (((b * NUM_CHUNKS + i) * CHUNK_SIZE + k) * NUM_HEADS + h) * HEAD_DIM + d
        tl.store(Y_ptr + Y_index, Y_row[d].to(tl.bfloat16))

class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,

        hidden_states = hidden_states.contiguous()
        A_cumsum = A_cumsum.contiguous()
        # Expand B and C across heads: repeat_interleave NUM_HEADS // N_GROUPS = 4
        # Original code does: B_exp = B.repeat_interleave(NUM_HEADS // N_GROUPS, dim=3)
        # Here we assume NUM_HEADS = 32, N_GROUPS = 8, so repeat 4
        # But we don't have NUM_HEADS here; set explicit constants.
        B_exp = B.repeat_interleave(4, dim=3).contiguous()
        C_exp = C.repeat_interleave(4, dim=3).contiguous()

        B_size = hidden_states.shape[0]  # batch_size
        NC_size = hidden_states.shape[1] # num_chunks

        # Allocate outputs
        # L: [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
        L = torch.empty((B_size, NC_size, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS), device=hidden_states.device, dtype=torch.float32)
        # G: [B, NUM_CHUNKS, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS]
        G = torch.empty((B_size, NC_size, CHUNK_SIZE, CHUNK_SIZE, NUM_HEADS), device=hidden_states.device, dtype=torch.float32)
        # M: same shape as G
        M = torch.empty_like(G, dtype=torch.float32)
        # Y_diag: [B, NUM_CHUNKS, CHUNK_SIZE, NUM_HEADS, HEAD_DIM], bfloat16
        Y_diag = torch.empty((B_size, NC_size, CHUNK_SIZE, NUM_HEADS, HEAD_DIM), device=hidden_states.device, dtype=torch.bfloat16)

        grid_L = (B_size, NC_size, NUM_HEADS)
        build_L_kernel[grid_L](

        )

        compute_G_kernel[grid_G](

        return Y_diag
