# Computes L[i, j] = exp(sum_{k=0..i} A[b, c, k, h]) if i >= j else 0
@triton.jit
def build_lower_tri_causal_kernel(
    A_ptr,         # *float32, [B, C, L, H]
    L_ptr,         # *float32, [B, C, L, L, H]
    B_size,        # int
    C_size,        # int
    L_len,         # int
    H_size,        # int
    BLOCK_J: tl.constexpr,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)

    # Loop over rows i
    i = 0
    while i < L_len:
        # Compute sum over k from 0..i
        total = 0.0
        k = 0
        while k <= i:
            # load A[b, c, k, h]
            addr = b * (C_size * L_len * H_size) + c * (L_len * H_size) + k * H_size + h
            a_val = tl.load(A_ptr + addr)
            total += a_val
            k += 1
        # Now set L[i, j] = exp(total) for j <= i
        j = 0
        while j < L_len:
            if j <= i:
                l_val = tl.exp(total)
            else:
                l_val = 0.0
            # store L[b, c, i, j, h] = l_val
            L_addr = b * (C_size * L_len * L_len * H_size) + c * (L_len * L_len * H_size) + i * (L_len * H_size) + j * H_size + h
            tl.store(L_ptr + L_addr, l_val)
            j += 1
        i += 1


# Triton kernel: expand along groups dimension by repeat_interleave (GROUP_EXPAND=4)

@triton.jit
def expand_groups_repeat_interleave(
    X_ptr,          # *float32 input tensor
    X_exp_ptr,      # *float32 output tensor
    B_size,         # int
    C_size,         # int
    L_len,          # int
    groups,         # int
    H_size,         # int (groups * GROUP_EXPAND)
    S_size,         # int
    GROUP_EXPAND: tl.constexpr,
):

    c = tl.program_id(1)
    i = tl.program_id(2)  # along L
    # h index in output
    h = tl.program_id(3)

        s += 1

   - The original code expands B and C from groups to H by repeating, then computes G[i, j, h] = sum_s C[b, c, i, h, s] * B[b, c, j, h, s]. Implement this in Triton with a 3D grid over (b, c, h), loop over i, j, and s. We avoid torch einsum.

3) Final reduction Y_diag:
   - Y[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden_states[b, c, j, h, d], where M = G * L. Implement Triton kernel that launches over (b, c, i, h) and vectorizes along d.

import triton

# Triton kernel: build lower-triangular causal matrix L for each (b, c, h)
# Computes L[i, j] = exp(sum_{k=0..i} A[b, c, k, h]) if i >= j else 0
@triton.jit
def build_lower_tri_causal_kernel(
    A_ptr,         # *float32, [B, C, L, H]
    L_ptr,         # *float32, [B, C, L, L, H]
    B_size,        # int
    C_size,        # int
    L_len,         # int
    H_size,        # int
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)

    # Loop over rows i in [0, L_len)
    i = 0
    while i < L_len:
        # Compute sum over k from 0..i
        total = 0.0
        k = 0
        while k <= i:
            addr = b * (C_size * L_len * H_size) + c * (L_len * H_size) + k * H_size + h
            a_val = tl.load(A_ptr + addr)
            total += a_val
            k += 1
        # Set L[i, j] = exp(total) for j <= i, else 0
        j = 0
        while j < L_len:
            if j <= i:
                l_val = tl.exp(total)
            else:
                l_val = 0.0
            L_addr = b * (C_size * L_len * L_len * H_size) + c * (L_len * L_len * H_size) + i * (L_len * H_size) + j * H_size + h
            tl.store(L_ptr + L_addr, l_val)
            j += 1
        i += 1
