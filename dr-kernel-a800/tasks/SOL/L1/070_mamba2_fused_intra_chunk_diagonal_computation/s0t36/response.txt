@triton.jit
def compute_Y_diag_1d(
    M_ptr,           # [B, C, S, S, H]
    hidden_ptr,      # [B, C, S, H, head_dim]
    Y_ptr,           # [B, C, S, H, head_dim]
    total_elems,     # B * C * S * H * head_dim
    S, H, head_dim,
    M_s0, M_s1, M_s2, M_s3, M_s4,
    hidden_s0, hidden_s1, hidden_s2, hidden_s3, hidden_s4,
    Y_s0, Y_s1, Y_s2, Y_s3, Y_s4
):
    pid = tl.program_id(0)
    if pid >= total_elems:
        return

    # Map pid to (b, c, i, h, d)
    tmp = pid
    d = tmp % head_dim
    tmp = tmp // head_dim
    h = tmp % H
    tmp = tmp // H
    i = tmp % S
    tmp = tmp // S
    c = tmp % C
    b = tmp // C

    acc = 0.0
    for j in range(0, S):
        M_idx = b * M_s0 + c * M_s1 + i * M_s2 + j * M_s3 + h * M_s4
        hidden_idx = b * hidden_s0 + c * hidden_s1 + j * hidden_s2 + h * hidden_s3 + d * hidden_s4
        M_val = tl.load(M_ptr + M_idx)
        hidden_val = tl.load(hidden_ptr + hidden_idx)
        acc += M_val * hidden_val

    Y_idx = b * Y_s0 + c * Y_s1 + i * Y_s2 + h * Y_s3 + d * Y_s4
    tl.store(Y_ptr + Y_idx, acc)

):

    if pid >= total_elems:

    j = tmp % S
    tmp = tmp // S

        B_val = tl.load(B_exp_ptr + B_idx)

    G_idx = b * G_s0 + c * G_s1 + i * G_s2 + j * G_s3 + h * G_s4
    tl.store(G_ptr + G_idx, acc)
