@triton.jit
def compute_L_exp_cumsum_kernel(
    A_ptr, L_ptr,
    Bsz, Csz,
    stride_A_b, stride_A_h, stride_A_c, stride_A_s,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
    S: tl.constexpr,  # chunk_size (128)
    H: tl.constexpr,  # num_heads (32)
    num_warps=8, num_stages=2
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    # Loop over i (source position), build cumsum over j (target position)
    for i in range(S):
        # maintain cumsum vector s[j] = sum_{t <= j} A[b, h, c, t]
        # We'll build s for all j in parallel using a vectorized approach
        # s is a vector of length S
        # We'll do scalar accumulation since Triton doesn't support per-element vector indexing in loops cleanly here.
        # Instead, we loop j in a separate manner, but to keep it simple and correct, we use a single j loop here.
        # However, Triton requires explicit ranges; better: build s[j] by looping j and writing L[i, j, h] directly.
        # So we'll loop j explicitly to avoid illegal memory access.
        for j in range(S):
            # Compute s[j] = sum_{t=0..j} A[b, h, c, t] across h
            # We'll recompute s[j] by summing A for this j across h
            s_j = 0.0
            for hh in range(H):
                a_off = b * stride_A_b + hh * stride_A_h + c * stride_A_c + j * stride_A_s
                a_val = tl.load(A_ptr + a_off)  # A[b, h, c, j] scalar
                s_j += a_val
            # Now set L[i, j, h] = exp(s_j) for all h and then write per h
            # We need to write L for each h. We'll loop over h and store.
            for hh in range(H):
                l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j + hh * stride_L_h
                l_val = tl.exp(s_j)  # since s_j is the cumsum up to j, this is correct for lower-triangular mask (i >= j)
                tl.store(L_ptr + l_off, l_val)
            # Note: For j > i, we want L[i, j, h] = 0. We don't need to store zeros because we already store exp(s_j) which is >0 only when i >= j.
            # To ensure zeros when i < j, we can skip store above for those j. But since s_j is 0 when i<j, exp(0)=1 would be wrong.

            # We can zero out L[i, j, h] for j > i by writing zeros in the next loop. However, Triton doesn't allow if on runtime j easily.
            # Simpler: rely on L initialized to zeros and not writing for j > i. Since our loop runs for all j, we must explicitly zero for j > i.
            # We will run a second loop to zero-out when j > i.

                    tl.store(L_ptr + l_off, 0.0)

def contract_BC_to_G_kernel(
