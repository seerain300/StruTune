Plan:
1) Triton Kernel for Masked Cumsum and Exp (L):
   - Inputs: A_cumsum [B, Csz, S, N] (float32), output L [B, Csz, S, S, N] (float32).
   - For each (b, c, n, i), compute a running sum across j in [0..S-1], include j < i (diagonal=-1), i.e., exclude j == i. Apply mask mask[j, i] = (j < i). Then compute cumsum_masked[j] = sum_{r<=j, r<i} A[b, c, r], store L[b, c, i, j, n] = exp(cumsum_masked[j]). This matches torch.tril(diagonal=-1) + cumsum + exp.

2) Triton Kernel for B/C Contraction to G:
   - Inputs: B_expanded [B, Csz, S, N, D], C_expanded [B, Csz, S, N, D], output G [B, Csz, S, S, N] (float32).
   - For each (b, c, i, j, n), G[i,j,n] = sum over k in [0..D-1] of C[b, c, i, n, k] * B[b, c, j, n, k]. D is passed as a constexpr.

3) Triton Kernel for Diagonal Contraction to Y_diag:
   - Inputs: M = G * L (float32), hidden_states [B, Csz, S, N, D] (float32), output Y [B, Csz, S, N, D] (float32).
   - For each (b, c, i, n, d), Y[b, c, i, n, d] = sum over j in [0..S-1] of M[b, c, i, j, n] * hidden_states[b, c, j, n, d].

All heavy numeric work is done by Triton; ModelNew.forward allocates tensors, casts to float32, and launches the Triton kernels. The final output is cast to bfloat16 to match the original signature.

import torch
import triton
import triton.language as tl

@triton.jit
def masked_cumsum_exp_kernel(
    A_ptr, L_ptr,
    B: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, N: tl.constexpr,
    A_stride_b, A_stride_c, A_stride_s, A_stride_n,
    L_stride_b, L_stride_c, L_stride_i, L_stride_j, L_stride_n,
):
    # Grid: (B, Csz, N, S)
    b = tl.program_id(0)
    c = tl.program_id(1)
    n = tl.program_id(2)
    i = tl.program_id(3)  # fixed i for this program, loop over j inside

    # Running sum across j for this (b, c, n, i), include j < i
    running = 0.0  # accumulate in float32
    # We'll compute L[j] = exp(running) for j < i, else 0. We only write L[i, j, n].
    for j in range(0, S):
        # Load A[b, c, j, n], guard if needed (though S is constexpr and j < S)
        a = tl.load(A_ptr + b * A_stride_b + c * A_stride_c + j * A_stride_s + n * A_stride_n)
        include = j < i
        running += a * include
        # We only need to store at (i, j), but since this kernel is per (i), we store all L[i, j, n]
        # Compute exp of the masked cumsum at position j: exp(running) if include else 0
        l_val = tl.exp(running) if include else 0.0
        tl.store(L_ptr + b * L_stride_b + c * L_stride_c + i * L_stride_i + j * L_stride_j + n * L_stride_n, l_val)

@triton.jit
def contract_BC_to_G_kernel(
    B_ptr, C_ptr, G_ptr,
    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, N: tl.constexpr, D: tl.constexpr,
    B_stride_b, B_stride_c, B_stride_j, B_stride_n, B_stride_d,
    C_stride_b, C_stride_c, C_stride_i, C_stride_n, C_stride_d,
    G_stride_b, G_stride_c, G_stride_i, G_stride_j, G_stride_n,
):
    # Grid: (Bsz, Csz, S, S, N)
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    n = tl.program_id(4)

    acc = 0.0  # float32 accumulator
    # Loop over state dimension K (D is constexpr)
    for k in range(0, D):
        b_val = tl.load(B_ptr + b * B_stride_b + c * B_stride_c + j * B_stride_j + n * B_stride_n + k * B_stride_d)
        c_val = tl.load(C_ptr + b * C_stride_b + c * C_stride_c + i * C_stride_i + n * C_stride_n + k * C_stride_d)
        acc += b_val * c_val
    tl.store(G_ptr + b * G_stride_b + c * G_stride_c + i * G_stride_i + j * G_stride_j + n * G_stride_n, acc)

@triton.jit
def diag_contract_Y_kernel(
    M_ptr, HS_ptr, Y_ptr,
    Bsz: tl.constexpr, Csz: tl.constexpr, S: tl.constexpr, N: tl.constexpr, D: tl.constexpr,
    M_stride_b, M_stride_c, M_stride_i, M_stride_j, M_stride_n,
    HS_stride_b, HS_stride_c, HS_stride_j, HS_stride_n, HS_stride_d,
    Y_stride_b, Y_stride_c, Y_stride_i, Y_stride_n, Y_stride_d,
):
    # Grid: (Bsz, Csz, S, N, D)
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    n = tl.program_id(3)
    d = tl.program_id(4)

    acc = 0.0  # float32 accumulator over j
    for j in range(0, S):
        m = tl.load(M_ptr + b * M_stride_b + c * M_stride_c + i * M_stride_i + j * M_stride_j + n * M_stride_n)
        hs = tl.load(HS_ptr + b * HS_stride_b + c * HS_stride_c + j * HS_stride_j + n * HS_stride_n + d * HS_stride_d)
        acc += m * hs
    tl.store(Y_ptr + b * Y_stride_b + c * Y_stride_c + i * Y_stride_i + n * Y_stride_n + d * Y_stride_d, acc)

        # 3) Compute G = contraction of B and C: G[b, c, i, j, n] = sum_k B[b,c,j,n,k] * C[b,c,i,n,k]
        G = torch.empty((Bsz, Csz, S, S, N), device=device, dtype=torch.float32)

            C_exp_stride_b, C_exp_stride_c, C_exp_stride_i, C_exp_stride_n, C_exp_stride_d,
            G_stride_b, G_stride_c, G_stride_i, G_stride_j, G_stride_n,
            num_warps=1, num_stages=1
        )

        # 4) Compute M = G * L elementwise in PyTorch (avoid torch.tril/cumsum here)
        M = G * L  # float32

        # 5) Diagonal contraction to compute Y_diag: [B, C, S, N, D]
        hidden_states_f32 = hidden_states.to(torch.float32).contiguous()
        Y = torch.empty((Bsz, Csz, S, N, D), device=device, dtype=torch.float32)

        M_stride_b, M_stride_c, M_stride_i, M_stride_j, M_stride_n = M.stride()
        HS_stride_b, HS_stride_c, HS_stride_j, HS_stride_n, HS_stride_d = hidden_states_f32.stride()

        grid_Y = (Bsz, Csz, S, N, D)
        diag_contract_Y_kernel[grid_Y](
            M, hidden_states_f32, Y,
            Bsz, Csz, S, N, D,
            M_stride_b, M_stride_c, M_stride_i, M_stride_j, M_stride_n,
            HS_stride_b, HS_stride_c, HS_stride_j, HS_stride_n, HS_stride_d,
            Y_stride_b, Y_stride_c, Y_stride_i, Y_stride_n, Y_stride_d,
            num_warps=1, num_stages=1
        )

        # Cast to bfloat16 to match original signature
        return Y.to(torch.bfloat16)
