```python
    A_ptr, L_ptr,
    B, H, C, S,
```

```python
    # 5) Compute Y_diag: [B, C, S, H, head_dim]
    hidden_f32 = hidden_states.to(torch.float32)
    Y = torch.empty((batch_size, num_chunks, chunk_size, num_heads, head_dim), dtype=torch.float32, device=hidden_states.device)
    for b in range(batch_size):
        for c in range(num_chunks):
            for i in range(chunk_size):
                for h in range(num_heads):
                    # sum over j of M[b, c, i, j, h] * hidden_states[b, c, j, h, d]
                    # We loop over head_dim d as well to build full output
                    for d in range(head_dim):
                        acc = 0.0
                        for j in range(chunk_size):
                            # M[b, c, i, j, h] * hidden[b, c, j, h, d]
                            m_val = M[b, c, i, j, h]
                            hs_val = hidden_f32[b, c, j, h, d]
                            acc += m_val * hs_val
                        # Store into Y[b, c, i, h, d]
                        Y[b, c, i, h, d] = acc

    return Y.to(torch.bfloat16)

class Model(torch.nn.Module):
    def forward(self, *args):
        return run(*args)
```

Summary of original computation and key points to reproduce in Triton

- Shapes and constants:
  - hidden_states: [B, C, S, H, head_dim], where S = CHUNK_SIZE = 128, H = NUM_HEADS = 32, N_GROUPS = 8, repeat_factor = 4.
  - A_cumsum: [B, H, C, S] where S is the sequence length per chunk. We must use this layout exactly (H as dim=1, C as dim=2).
  - B, C: [B, C, S, N_GROUPS, N], N_GROUPS=8, N=128 typically.
- Step 1: Build L with lower-triangular causal mask:
  - Create A_expanded [B, H, C, S, S] by repeating along the last dimension S (using expand).
  - Compute cumsum along j for each row i: cumsum_j[j] = sum_{t=0..j} A[b, h, c, t].
  - L[i, j, h] = exp(cumsum_j[j]) for i >= j, else 0. Note the “lower triangular” exclusion of diagonal (diagonal=-1).
  - In the provided code, they build L with expanded A_cumsum and cumsum along the last expanded dim. We should match that logic.
- Step 2: Expand B and C along H dimension to get [B, C, S, H, N] by repeat_interleave 4.
- Step 3: Compute G[i, j, h] = sum over n of C_exp[b, c, i, n, j, h] * B_exp[b, c, j, n, i, h]. This is a contraction over N and forms a [S, S] matrix per (b, c, h).
- Step 4: M = G * L.
- Step 5: Y_diag[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden_states[b, c, j, h, d]. Loop over j and head_dim d to accumulate.

Potential pitfalls and how to avoid them

- Triton kernel indexing: To avoid shape and stride mismatches, we’ll:
  - Compute L in Triton per (b, c, h) by looping over i and j and using A’s strides to address elements. This mirrors the original logic closely.
  - Compute G in Triton per (b, c, i, h) by looping over j and N, accumulating dot products. This ensures correctness before attempting more complex vectorized contractions.
  - Compute Y_diag in Triton per (b, c, i, h) by looping over j and head_dim, accumulating into a vector. This mirrors the PyTorch output generation.
- Avoid torch operations inside Triton kernels where possible. Use simple 1D grid per program and explicit loops over small, fixed sizes (S=128, N=128, H=32). This reduces the risk of runtime errors.
- Ensure inputs are contiguous and on CUDA. Use .to(torch.float32) for compute and cast outputs as needed.
- Return dtype bfloat16 to match the original.

Below is the Triton implementation that is actually invoked from ModelNew.forward and performs the heavy computations in Triton. It uses three kernels: build_L, compute_G, and compute_Y_diag. The forward ensures all kernels are launched and shapes are correctly handled. Note that some operations (like expand) are done in PyTorch to create the required layout, but the heavy numeric work is in Triton.

import torch
import triton
import triton.language as tl


# Kernel: Build L[b, c, i, j, h] = exp(sum_{t=0..j} A[b,h,c,t]) for i >= j, else 0
# A: [B, H, C, S], L: [B, C, S, S, H]
@triton.jit
def build_L_kernel(
    A_ptr, L_ptr,
    Bsz, H, Csz, S,
    stride_A_b, stride_A_h, stride_A_c, stride_A_s,
    stride_L_b, stride_L_c, stride_L_i, stride_L_j, stride_L_h,
):
    # Grid: (Bsz * Csz * S, H)
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    b = pid0 // (Csz * S)
    rem = pid0 % (Csz * S)
    c = rem // S
    i = rem % S
    h = pid1

    # Compute cumsum for j positions
    # L[i, j, h] requires cumsum_j[j] only; we compute it per (b, c, h) and store for all i>=j
    # Initialize cumsum scalar
    cumsum = 0.0
    for j in range(0, S):
        # A[b, h, c, j]
        a_off = b * stride_A_b + h * stride_A_h + c * stride_A_c + j * stride_A_s
        a_val = tl.load(A_ptr + a_off)
        cumsum += a_val
        # If i >= j, write exp(cumsum) to L[i, j, h], else 0
        if i >= j:
            l_off = b * stride_L_b + c * stride_L_c + i * stride_L_i + j * stride_L_j
