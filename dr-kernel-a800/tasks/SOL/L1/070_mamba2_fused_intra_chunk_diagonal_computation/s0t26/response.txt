        # Compute cumsum for A[b, h, c, :]
        cumsum = tl.zeros([CHUNK_SIZE], dtype=tl.float32)
        for j in range(CHUNK_SIZE):
            a_val = tl.load(
                A_ptr + b * A_stride_b + h_idx * A_stride_h + c * A_stride_c + j * A_stride_s,
                mask=True,
                other=0.0
            )
            cumsum[j] = a_val  # maintain cumsum[j] = sum_{t=0..j} A[b, h, c, t]
        # Write L[i, j, h] = exp(cumsum[j]) for i >= j
        for i in range(CHUNK_SIZE):
            for j in range(CHUNK_SIZE):
                if i >= j:
                    val = tl.exp(cumsum[j])  # float32
                    tl.store(
                        L_ptr + b * L_stride_b + c * L_stride_c + i * L_stride_i + j * L_stride_j + h_idx * L_stride_h,
                        val
                    )

# Kernel: compute G[i, j, h] = sum over n of C_exp[b, c, i, h, n] * B_exp[b, c, j, h, n]
# Simple 1D grid decoding to avoid complicated 2D decoding; robust and correct for small sizes.
@triton.jit
def compute_G_kernel_simple(B_exp_ptr, C_exp_ptr, G_ptr,
                            B_stride_b, B_stride_c, B_stride_s, B_stride_h, B_stride_n,
                            C_stride_b, C_stride_c, C_stride_s, C_stride_h, C_stride_n,
                            G_stride_b, G_stride_c, G_stride_i, G_stride_j, G_stride_h,
                            N: tl.constexpr, S: tl.constexpr, H: tl.constexpr):
    total = S * S * H * N
    pid = tl.program_id(0)
    b = pid // (total)
    tmp = pid % (total)
    c = tmp // (S * S * H)
    tmp2 = tmp % (S * S * H)
    i = tmp2 // (S * H)
    tmp3 = tmp2 % (S * H)
    j = tmp3 // H
    h = tmp3 % H
    acc = 0.0
    for n in range(N):
        C_val = tl.load(
            C_exp_ptr + b * C_stride_b + c * C_stride_c + i * C_stride_s + h * C_stride_h + n * C_stride_n
        )
        B_val = tl.load(
            B_exp_ptr + b * B_stride_b + c * B_stride_c + j * B_stride_s + h * B_stride_h + n * B_stride_n
        )
        acc += C_val * B_val
    tl.store(
        G_ptr + b * G_stride_b + c * G_stride_c + i * G_stride_i + j * G_stride_j + h * G_stride_h,
        acc
    )

# Kernel: compute Y_diag[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden_states[b, c, j, h, d]
@triton.jit
def compute_Y_diag_kernel(M_ptr, hidden_ptr, Y_ptr,
                          M_stride_b, M_stride_c, M_stride_i, M_stride_j, M_stride_h,
                          hidden_stride_b, hidden_stride_c, hidden_stride_s, hidden_stride_h, hidden_stride_d,
                          Y_stride_b, Y_stride_c, Y_stride_i, Y_stride_h, Y_stride_d,
                          S: tl.constexpr, H: tl.constexpr, head_dim: tl.constexpr):
    pid0 = tl.program_id(0)  # over B*C*S*H
    pid1 = tl.program_id(1)  # over head_dim
    bc_ih = S * H
    b = pid0 // (bc_ih)
    tmp = pid0 % (bc_ih)
    c = tmp // (S * H)
    tmp2 = tmp % (S * H)
    i = tmp2 // H
    h = tmp2 % H
    d = pid1
    acc = 0.0
    for j in range(S):
        M_val = tl.load(
            M_ptr + b * M_stride_b + c * M_stride_c + i * M_stride_i + j * M_stride_j + h * M_stride_h
        )
        hidden_val = tl.load(
            hidden_ptr + b * hidden_stride_b + c * hidden_stride_c + j * hidden_stride_s + h * hidden_stride_h + d * hidden_stride_d
        )
        acc += M_val * hidden_val
    tl.store(
        Y_ptr + b * Y_stride_b + c * Y_stride_c + i * Y_stride_i + h * Y_stride_h + d * Y_stride_d,
        acc
    )
