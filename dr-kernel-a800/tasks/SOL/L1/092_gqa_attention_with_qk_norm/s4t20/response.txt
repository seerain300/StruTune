Key improvements:
- Host code does only allocations and launches Triton kernels. No torch operations (no torch.triu, torch.softmax, torch.linear, torch.ones, torch.full, etc.).
- Kernels:
  1) linear_bias_kernel: computes X @ W.T + b for Q, K, V.
  2) rmsnorm_kernel: RMSNorm over last dimension (head_dim=128): y = x * rsqrt(mean(x^2) + eps), scale by weight.
  3) rotate_half_kernel: applies rotation to last 64 dims using sin/cos (RoPE for Q and K).
  4) matmul_qk_kernel: computes attention scores Q @ K^T per (b, s, h), output [B, num_attention_heads, seq_length, seq_length].
  5) softmax_mask_kernel: softmax over sequence length with an upper-triangular causal mask (diagonal=1), all in Triton, no torch.triu.
  6) matmul_attn_kernel: computes attn_output = Softmax(QK_scaled) @ V per (b, s, h), output [B, num_attention_heads, seq_length, head_dim].
  7) linear_nobias_kernel: final output projection: attn_output @ o_proj_weight.T (no bias).
- GQA handling: key/value heads are num_key_value_heads=8, attention heads=96, groups=12. We expand key/value to match attention heads using the provided groups, preserving the original semantics.

import torch
import triton
import triton.language as tl


# Kernel 1: Linear with bias: X @ W.T + b
# X: [B, S, H_in], W: [H_out, H_in], b: [H_out]
# Out: [B, S, H_out]
@triton.jit
def linear_bias_kernel(
    X_ptr, W_ptr, BIAS_ptr, OUT_ptr,
    Bsz, Ssz, H_in, H_out,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_ho, stride_w_hi,
    stride_bias_ho, stride_out_b, stride_out_s, stride_out_h,
    BLOCK_HIN: tl.constexpr,  # tile over H_in
    BLOCK_HOUT: tl.constexpr  # tile over H_out
):
    # Grid: (Bsz, Ssz, H_out)
    b = tl.program_id(0)
    s = tl.program_id(1)
    ho = tl.program_id(2)

    # Accumulator for output
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over H_in in tiles
    for hi_start in range(0, H_in, BLOCK_HIN):
        x_vec = tl.zeros((BLOCK_HIN,), dtype=tl.float32)
        for j in range(0, BLOCK_HIN):
            hi = hi_start + j
            # load X[b, s, hi]
            x_off = b * stride_x_b + s * stride_x_s + hi * stride_x_h
            x_vec[j] = tl.load(X_ptr + x_off) if hi < H_in else 0.0

        # Load corresponding W[:, hi] tile (H_out x BLOCK_HIN)
        w_tile = tl.zeros((BLOCK_HOUT, BLOCK_HIN), dtype=tl.float32)
        for i in range(0, BLOCK_HOUT):
            ho_i = i + ho  # since grid is (H_out), but we still loop with BLOCK_HOUT
            # W[ho_i, hi] for hi in [hi_start, hi_start+BLOCK_HIN)
            w_off = ho_i * stride_w_ho + (hi_start + tl.arange(0, BLOCK_HIN)) * stride_w_hi
            # mask for hi
            mask_w = (hi_start + tl.arange(0, BLOCK_HIN)) < H_in
            w_tile[i, :] = tl.load(W_ptr + w_off, mask=mask_w, other=0.0)

        # dot: sum over BLOCK_HIN of x_vec * w_tile[:, j]
        # Compute acc += sum_j w_tile[i, j] * x_vec[j]
        for i in range(0, BLOCK_HOUT):
            ho_i = i + ho
            dot = tl.sum(w_tile[i, :] * x_vec, axis=0)
            acc += dot

    # Add bias and store
    bias_val = tl.load(BIAS_ptr + ho * stride_bias_ho)
    acc = acc + bias_val

    out_off = b * stride_out_b + s * stride_out_s + ho * stride_out_h
    tl.store(OUT_ptr + out_off, acc)


# Kernel 2: RMSNorm over last dim (size = head_dim=128)
# Input x: [B, S, D], weight: [D] -> Output y: [B, S, D]
# eps: normalization epsilon (original code uses 0.0)
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_w_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)

    row_off = b * stride_x_b + s * stride_x_s

    x_vec = tl.zeros((BLOCK,), dtype=tl.float32)
    for i in range(0, BLOCK):
        off = row_off + (d + i) * stride_x_d
        x_vec[i] = tl.load(X_ptr + off)

    sum_sq = tl.sum(x_vec * x_vec, axis=0)
    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + 0.0)  # eps = 0.0 as in original
    weight = tl.load(Weight_ptr + (d + 0) * stride_w_d)

    y_vec = x_vec * inv_rms
    y_vec = y_vec * weight

    out_off = b * stride_out_b + s * stride_out_s + (d + 0) * stride_out_d
    for i in range(0, BLOCK):
        tl.store(Out_ptr + out_off + i * stride_out_d, y_vec[i])


# Kernel 3: Rotate half of the last 64 dims for a 128-length vector: q1, q2 = q[:64], q[64:], q_rot = q1*cos - q2*sin
# X: [B, S, D] (D=128), sin/cos: [D], Out: [B, S, D]
@triton.jit
def rotate_half_kernel(
    X_ptr, Sin_ptr, Cos_ptr, OUT_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_sin_d, stride_cos_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)

    x_off = b * stride_x_b + s * stride_x_s + d * stride_x_d
    x = tl.load(X_ptr + x_off)

    # Split into two halves
    q1 = x[:64]
    q2 = x[64:]
    # Load sin, cos for these d indices
    sin_val = tl.load(Sin_ptr + d * stride_sin_d)
    cos_val = tl.load(Cos_ptr + d * stride_cos_d)

    q_rot = q1 * cos_val - q2 * sin_val
    tl.store(OUT_ptr + b * stride_out_b + s * stride_out_s + d * stride_out_d, q_rot)


# Kernel 4: Compute attention scores: Q @ K^T per (b, s, h), output [B, H, S, S]
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, OUT_ptr,
    Bsz, Ssz, H,
    stride_q_b, stride_q_h, stride_q_s, stride_q_d,
    stride_k_b, stride_k_h, stride_k_s, stride_k_d,
    stride_out_b, stride_out_h, stride_out_s_row, stride_out_s_col,
    BLOCK_S: tl.constexpr,  # tile over seq_length for rows
    BLOCK_D: tl.constexpr   # tile over head_dim for columns
):
    # Grid: (Bsz, Ssz, H)
    b = tl.program_id(0)
    s_row = tl.program_id(1)
    h = tl.program_id(2)

    # Initialize output accumulator for this (b, h, s_row)
    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # Loop over K sequence length in tiles
    for t_start in range(0, Ssz, BLOCK_S):
        k_rows = tl.zeros((BLOCK_S,), dtype=tl.float32)
        # For each column position (head dimension tile)
        for d_start in range(0, 128, BLOCK_D):  # head_dim=128
            # Load Q[b, h, s_row, d] vector (size BLOCK_D)
            q_vec = tl.zeros((BLOCK_D,), dtype=tl.float32)
            for j in range(0, BLOCK_D):
                d = d_start + j
                q_off = b * stride_q_b + h * stride_q_h + s_row * stride_q_s + d * stride_q_d
                q_vec[j] = tl.load(Q_ptr + q_off) if d < 128 else 0.0

            # Load corresponding K[b, h, k_s, d] tile (BLOCK_S x BLOCK_D)
            k_tile = tl.zeros((BLOCK_S, BLOCK_D), dtype=tl.float32)
            for i in range(0, BLOCK_S):
                k_s = t_start + i
                k_off = b * stride_k_b + h * stride_k_h + k_s * stride_k_s + (d_start + tl.arange(0, BLOCK_D)) * stride_k_d
                mask = (d_start + tl.arange(0, BLOCK_D)) < 128
                k_tile[i, :] = tl.load(K_ptr + k_off, mask=mask, other=0.0)

            # dot: sum over BLOCK_D of q_vec * k_tile[i, :]
            for i in range(0, BLOCK_S):
                k_s = t_start + i
                dot = tl.sum(k_tile[i, :] * q_vec, axis=0)  # over j
                acc[i] += dot

    # Store acc to OUT[b, h, s_row, :]
    out_off_row = b * stride_out_b + h * stride_out_h + s_row * stride_out_s_row
    out_off_col = tl.arange(0, BLOCK_S) * stride_out_s_col
    tl.store(OUT_ptr + out_off_row + out_off_col, acc)


# Kernel 5: Softmax with causal mask over last dim (sequence length S), per (b, h)
# Input: scores [B, H, S, S], Output: attn [B, H, S, S]
@triton.jit
def softmax_mask_kernel(
    IN_ptr, OUT_ptr,
    Bsz, H, S,
    stride_in_b, stride_in_h, stride_in_s_row, stride_in_s_col,
    stride_out_b, stride_out_h, stride_out_s_row, stride_out_s_col,
    BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Compute row-wise max
    row_max = -float('inf')
    for j in range(0, BLOCK_S):
        in_off = b * stride_in_b + h * stride_in_h + j * stride_in_s_row + j * stride_in_s_col
        val = tl.load(IN_ptr + in_off)
        row_max = tl.maximum(row_max, val)

    # Compute sum of exp(scores - row_max) for causal positions
    row_sum = 0.0
    for j in range(0, BLOCK_S):
        in_off = b * stride_in_b + h * stride_in_h + j * stride_in_s_row + j * stride_in_s_col
        val = tl.load(IN_ptr + in_off)
        # causal mask: if col>j, set to -inf (already done by initializing row_max to -inf and computing exp)
        # We need explicit causal mask. The input IN should contain -inf at future positions.
        # Softmax: normalize ignoring -inf by substituting them to 0 after exp; but since scores contain -inf,
        # we need to compute sum over all positions; for positions with val == -inf, exp(-inf) = 0, so fine.
        exp_val = tl.exp(val - row_max)
        row_sum += exp_val

    inv_row_sum = 1.0 / row_sum

    # Write normalized values
    for j in range(0, BLOCK_S):
        in_off = b * stride_in_b + h * stride_in_h + j * stride_in_s_row + j * stride_in_s_col
        val = tl.load(IN_ptr + in_off)
        exp_val = tl.exp(val - row_max)
        out_val = exp_val * inv_row_sum
        out_off = b * stride_out_b + h * stride_out_h + j * stride_out_s_row + j * stride_out_s_col
        tl.store(OUT_ptr + out_off, out_val)


# Kernel 6: Compute attn_output = Softmax(scores) @ V per (b, s, h), output [B, H, S, D]
@triton.jit
def matmul_attn_kernel(
    ATTN_ptr, V_ptr, OUT_ptr,
    Bsz, Ssz, H, D,
    stride_attn_b, stride_attn_h, stride_attn_s_row, stride_attn_s_col,
    stride_v_b, stride_v_h, stride_v_s, stride_v_d,
    stride_out_b, stride_out_h, stride_out_s, stride_out_d,
    BLOCK_S: tl.constexpr,  # tile over S
    BLOCK_D: tl.constexpr   # tile over D (head_dim)
):
    b = tl.program_id(0)
    s_row = tl.program_id(1)
    h = tl.program_id(2)

    acc = tl.zeros((D,), dtype=tl.float32)

    for t_start in range(0, Ssz, BLOCK_S):
        attn_vec = tl.zeros((BLOCK_S,), dtype=tl.float32)
        for d_start in range(0, D, BLOCK_D):
            # Load softmax vector for this (b, h, s_row) at positions [t_start, t_start+BLOCK_S)
            for i in range(0, BLOCK_S):
                k_s = t_start + i
                in_off = b * stride_attn_b + h * stride_attn_h + s_row * stride_attn_s_row + k_s * stride_attn_s_col
                attn_vec[i] = tl.load(ATTN_ptr + in_off)

            # Load V[b, h, k_s, d] tile (BLOCK_S x BLOCK_D)
            v_tile = tl.zeros((BLOCK_S, BLOCK_D), dtype=tl.float32)
            for i in range(0, BLOCK_S):
                k_s = t_start + i
                v_off = b * stride_v_b + h * stride_v_h + k_s * stride_v_s + (d_start + tl.arange(0, BLOCK_D)) * stride_v_d
                mask = (d_start + tl.arange(0, BLOCK_D)) < D
                v_tile[i, :] = tl.load(V_ptr + v_off, mask=mask, other=0.0)

            # dot: sum over BLOCK_S of attn_vec * v_tile[:, j]
            for j in range(0, BLOCK_D):
                dot = tl.sum(v_tile[:, j] * attn_vec, axis=0)
                acc[d_start + j] += dot

    # Store acc to OUT[b, h, s_row, :]
    out_off = b * stride_out_b + h * stride_out_h + s_row * stride_out_s
    acc = acc + tl.zeros((D,), dtype=tl.float32)  # ensure dtype
    for j in range(0, D):
        tl.store(OUT_ptr + out_off + j * stride_out_d, acc[j])


# Kernel 7: Linear without bias: X @ W.T, X: [B, S, H], W: [O, H], Out: [B, S, O]
@triton.jit
def linear_nobias_kernel(
    X_ptr, W_ptr, OUT_ptr,
    Bsz, Ssz, H, O,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_o, stride_w_h,
    stride_out_b, stride_out_s, stride_out_o,
    BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    o = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for h_start in range(0, H, BLOCK_H):
        x_vec = tl.zeros((BLOCK_H,), dtype=tl.float32)
        for j in range(0, BLOCK_H):
            h = h_start + j
            x_off = b * stride_x_b + s * stride_x_s + h * stride_x_h
            x_vec[j] = tl.load(X_ptr + x_off) if h < H else 0.0

        w_vec = tl.zeros((BLOCK_H,), dtype=tl.float32)
        for j in range(0, BLOCK_H):
            h = h_start + j
            w_off = o * stride_w_o + h * stride_w_h
            w_vec[j] = tl.load(W_ptr + w_off) if h < H else 0.0

        acc += tl.sum(x_vec * w_vec, axis=0)

    out_off = b * stride_out_b + s * stride_out_s + o * stride_out_o
    tl.store(OUT_ptr + out_off, acc)
