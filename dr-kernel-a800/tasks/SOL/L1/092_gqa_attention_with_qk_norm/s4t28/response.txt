import torch
import triton
import triton.language as tl


# Kernel 1: Linear with bias: X @ W.T + b
# X: [B, S, H_in], W: [H_out, H_in], b: [H_out]
# Output: Y: [B, S, H_out]
@triton.jit
def linear_bias_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    Bsz, Ssz, H_in, H_out,
    stride_x_b, stride_x_s, stride_x_hin,
    stride_w_hout, stride_w_hin,
    stride_b_hout,
    stride_y_b, stride_y_s, stride_y_hout,
    BLOCK_HIN: tl.constexpr, BLOCK_HOUT: tl.constexpr,
):
    # Grid: (Bsz, Ssz, H_out)
    b = tl.program_id(0)
    s = tl.program_id(1)
    hout = tl.program_id(2)

    acc = 0.0

    # Accumulate over H_in in tiles of BLOCK_HIN
    for hin_start in range(0, H_in, BLOCK_HIN):
        x_vec = tl.zeros((BLOCK_HIN,), dtype=tl.float32)
        for i in range(0, BLOCK_HIN):
            hin = hin_start + i
            x_off = b * stride_x_b + s * stride_x_s + hin * stride_x_hin
            x_vec[i] = tl.load(X_ptr + x_off)
        # Load corresponding W[hout, :] across H_in (vector over H_in for fixed hout)
        w_vec = tl.zeros((BLOCK_HOUT,), dtype=tl.float32)
        for i in range(0, BLOCK_HOUT):
            w_off = hout * stride_w_hout + (hin_start + i) * stride_w_hin  # index over H_in
            w_vec[i] = tl.load(W_ptr + w_off)
        # Dot product accumulate for this hout
        acc += tl.sum(x_vec * w_vec, axis=0)

    # Add bias
    bias = tl.load(B_ptr + hout * stride_b_hout)
    acc = acc + bias

    # Store result to Y[b, s, hout]
    y_off = b * stride_y_b + s * stride_y_s + hout * stride_y_hout
    tl.store(Y_ptr + y_off, acc)


# Kernel 2: RMSNorm over last dim (size = head_dim=128): y = x * rsqrt(mean(x^2) + eps), then scale by weight
# Input x: [B, S, D], weight: [D] -> Output y: [B, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_w_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    # d is computed via index in loop
    # We'll process row (b, s) across D in tiles of BLOCK
    row_off_x = b * stride_x_b + s * stride_x_s
    row_off_out = b * stride_out_b + s * stride_out_s

    sum_sq = 0.0
    # First pass: compute sum of squares
    for i in range(0, D):
        xi = tl.load(X_ptr + row_off_x + i * stride_x_d)
        sum_sq += xi * xi
    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + 0.0)  # rms_norm_eps set to 0.0 as in original code

    # Second pass: write normalized and scaled output
    for i in range(0, D):
        xi = tl.load(X_ptr + row_off_x + i * stride_x_d)
        wi = tl.load(Weight_ptr + i * stride_w_d)
        yi = xi * inv_rms
        yi = yi * wi
        tl.store(Out_ptr + row_off_out + i * stride_out_d, yi)


# Kernel 3: Rotate half for a 128-length vector: q1, q2 = q[:64], q[64:], q_rot = q1*cos - q2*sin
# This kernel applies rotation to Q and K vectors. We assume D=128 and cos, sin are 128-element arrays.
@triton.jit
def rotate_half_kernel(
    Q_ptr, Sin_ptr, Cos_ptr, Out_ptr,
    Bsz, Ssz, D,  # D should be 128
    stride_q_b, stride_q_s, stride_q_d,
    stride_s_d, stride_c_d, stride_out_b, stride_out_s, stride_out_d,
):
    # Grid: (Bsz, Ssz)
    b = tl.program_id(0)
    s = tl.program_id(1)

    # We process D in chunks of 64 for simplicity
    q1 = tl.zeros((64,), dtype=tl.float32)
    q2 = tl.zeros((64,), dtype=tl.float32)

    # Load q across D
    for i in range(0, 64):
        qi = tl.load(Q_ptr + b * stride_q_b + s * stride_q_s + i * stride_q_d)
        q1[i] = qi
    for i in range(0, 64):
        qi = tl.load(Q_ptr + b * stride_q_b + s * stride_q_s + (64 + i) * stride_q_d)
        q2[i] = qi

    # Load sin and cos for last half
    sin_half = tl.zeros((64,), dtype=tl.float32)
    cos_half = tl.zeros((64,), dtype=tl.float32)
    for i in range(0, 64):
        sin_half[i] = tl.load(Sin_ptr + i * stride_s_d)
        cos_half[i] = tl.load(Cos_ptr + i * stride_c_d)

    q1_rot = q1 * cos_half - q2 * sin_half

    # Store rotated q for first half
    for i in range(0, 64):
        tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + i * stride_out_d, q1[i])
    # Store rotated q for second half
    for i in range(0, 64):
        tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + (64 + i) * stride_out_d, q1_rot[i])


# Kernel 4: Compute attention scores Q @ K^T per (b, s, h): Out[b,h,i,j] = dot(Q[b,h,i,:], K[b,h,j,:])
# We use Triton to compute scores over sequence pairs (i,j). Output shape: [B, num_attention_heads, S, S]
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_q_b, stride_q_h, stride_q_s, stride_q_d,
    stride_k_b, stride_k_h, stride_k_s, stride_k_d,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    h: tl.constexpr,  # attention head index
):
    # Grid over (b, i)
    b = tl.program_id(0)
    i = tl.program_id(1)

    # Load Q[b, h, i, :]
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_off = b * stride_q_b + h * stride_q_h + i * stride_q_s + d * stride_q_d
        q_vec[d] = tl.load(Q_ptr + q_off)

    # Accumulate scores for all j
    scores_row = tl.zeros((Ssz,), dtype=tl.float32)
    for j in range(0, Ssz):
        k_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            k_off = b * stride_k_b + h * stride_k_h + j * stride_k_s + d * stride_k_d
            k_vec[d] = tl.load(K_ptr + k_off)
        dot = 0.0
        for d in range(0, D):
            dot += q_vec[d] * k_vec[d]
        scores_row[j] = dot

    # Store scores[b, h, i, :]
    for j in range(0, Ssz):
        out_off = b * stride_out_b + h * stride_out_h + i * stride_out_i + j * stride_out_j
        tl.store(Out_ptr + out_off, scores_row[j])


# Kernel 5: Softmax over last dimension (sequence length) with causal mask (upper-triangular, diagonal=1)
# Inputs: Attn[b,h,i,j], Output: Soft[b,h,i,j]
@triton.jit
def softmax_mask_kernel(
    Attn_ptr, Soft_ptr,
    Bsz, Ssz,
    stride_attn_b, stride_attn_h, stride_attn_i, stride_attn_j,
    stride_soft_b, stride_soft_h, stride_soft_i, stride_soft_j,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    # For each row i, compute softmax over j
    for i in range(0, Ssz):
        # Compute row_max
        row_max = -1e30
        for j in range(0, Ssz):
            attn_off = b * stride_attn_b + h * stride_attn_h + i * stride_attn_i + j * stride_attn_j
            val = tl.load(Attn_ptr + attn_off)
            # Apply causal mask: if j > i, val = -inf
            if j > i:
                val = -1e30
            if val > row_max:
                row_max = val
        # Compute sum of exp
        row_sum = 0.0
        for j in range(0, Ssz):
            attn_off = b * stride_attn_b + h * stride_attn_h + i * stride_attn_i + j * stride_attn_j
            val = tl.load(Attn_ptr + attn_off)
            if j > i:
                val = -1e30
            expv = tl.exp(val - row_max)
            row_sum += expv
        # Write softmax
        for j in range(0, Ssz):
            attn_off = b * stride_attn_b + h * stride_attn_h + i * stride_attn_i + j * stride_attn_j
            val = tl.load(Attn_ptr + attn_off)
            if j > i:
                val = -1e30
            expv = tl.exp(val - row_max)
            soft = expv / row_sum
            soft_off = b * stride_soft_b + h * stride_soft_h + i * stride_soft_i + j * stride_soft_j
            tl.store(Soft_ptr + soft_off, soft)


# Kernel 6: Compute attention output: Soft[b,h,i,:] @ V[b,h,:,d] -> Out[b,h,i,d]
@triton.jit
def matmul_attn_v_kernel(
    Soft_ptr, V_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_soft_b, stride_soft_h, stride_soft_i, stride_soft_j,
    stride_v_b, stride_v_h, stride_v_s, stride_v_d,
    stride_out_b, stride_out_h, stride_out_i, stride_out_d,
    h: tl.constexpr,  # attention head index
):
    b = tl.program_id(0)
    i = tl.program_id(1)

    # Accumulator for output vector at position i across D
    out_vec = tl.zeros((D,), dtype=tl.float32)

    # For each j, load Soft[b,h,i,j], and V[b,h,j,:] and accumulate
    for j in range(0, Ssz):
        soft_off = b * stride_soft_b + h * stride_soft_h + i * stride_soft_i + j * stride_soft_j
        soft = tl.load(Soft_ptr + soft_off)
        v_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            v_off = b * stride_v_b + h * stride_v_h + j * stride_v_s + d * stride_v_d
            v_vec[d] = tl.load(V_ptr + v_off)
        out_vec += soft * v_vec

    # Store out[b,h,i,:]
    for d in range(0, D):
        out_off = b * stride_out_b + h * stride_out_h + i * stride_out_i + d * stride_out_d
        tl.store(Out_ptr + out_off, out_vec[d])
