import math
import torch
import triton
import triton.language as tl


# -----------------------------
# Triton kernel: dense linear for output[b, s, h] = sum_k input[b, :, k] * weight[h, k] + bias[h]
# Input shape: [B, S, K]
# Weight shape: [H, K]
# Output shape: [B, S, H]
# Grid: (B, H, S)
# -----------------------------
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, b_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    acc = tl.zeros([BLOCK_K], dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        # Load input row for (b, s) across K
        x_row = tl.load(
            x_ptr + b * x_stride0 + s * x_stride1 + k * x_stride2,
            mask=mask_k, other=0.0
        )
        # Load weights for this h across K
        w_row = tl.load(
            w_ptr + h * w_stride0 + k * w_stride1,
            mask=mask_k, other=0.0
        )
        acc += x_row * w_row

    # Add bias
    bias = tl.load(b_ptr + h)
    acc += bias

    # Store
    tl.store(
        out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2,
        acc,
        mask=True
    )


# -----------------------------
# Triton RMSNorm: normalize each row (b, h) across S
# Input out_in has shape [B, S, H]
# weight[h] is scale factor
# Output out_out has shape [B, S, H]
# Grid: (B, H), iterate S in BLOCK_S tiles
# -----------------------------
@triton.jit
def triton_rmsnorm(x_ptr, weight_ptr, out_ptr,
                    B, S, H,
                    x_stride0, x_stride1, x_stride2,
                    out_stride0, out_stride1, out_stride2,
                    eps, BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    sum_sq = 0.0
    for s0 in range(0, S, BLOCK_S):
        s = s0 + tl.arange(0, BLOCK_S)
        mask_s = s < S
        x = tl.load(
            x_ptr + b * x_stride0 + s * x_stride1 + h * x_stride2,
            mask=mask_s, other=0.0
        ).to(tl.float32)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = tl.load(weight_ptr + h) * inv_rms  # weight[h] is the gamma scale

    for s0 in range(0, S, BLOCK_S):
        s = s0 + tl.arange(0, BLOCK_S)
        mask_s = s < S
        x = tl.load(
            x_ptr + b * x_stride0 + s * x_stride1 + h * x_stride2,
            mask=mask_s, other=0.0
        ).to(tl.float32)
        y = x * scale
        tl.store(
            out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2,
            y.to(tl.float16),  # keep output dtype consistent (as in original model likely fp16)
            mask=mask_s
        )


# -----------------------------
# Triton RoPE: rotate query/key: split 128 into 64+64 and apply q_out = q*cos + [-q2, q1]*sin
# Input has shape [B, H, S], head_dim=128, grid (B, H, S)
# cos, sin: shape [128]
# -----------------------------
@triton.jit
def triton_rope_row(x_ptr, cos_ptr, sin_ptr, out_ptr,
                    B, H, S, head_dim,
                    x_stride0, x_stride1, x_stride2,
                    cos_stride0, cos_stride1,
                    sin_stride0, sin_stride1,
                    out_stride0, out_stride1, out_stride2,
                    BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=0.0)
        sin_vec = tl.load(sin_ptr + d, mask=mask, other=0.0)
        # split
        q1 = q[:64]
        q2 = q[64:]
        rotated = -q2 + q1  # corresponds to [-q2, q1] concatenated
        q_out = q * cos_vec + rotated * sin_vec
        tl.store(out_ptr + base_out + d, q_out, mask=mask)


# -----------------------------
# Triton GQA expand: expand KV heads from KVH to H with GROUPS = H // KVH (for given config, 96 // 8 = 12)
# Input: K/V [B, KVH, S, KD], Output: K/V expanded [B, H, S, KD]
# Grid: (B, KVH, GROUPS, S)
# -----------------------------
@triton.jit
def triton_expand_kv(K_ptr, V_ptr, K_out_ptr, V_out_ptr,
                     B, S, KVH, KD,
                     K_stride0, K_stride1, K_stride2, K_stride3,
                     V_stride0, V_stride1, V_stride2, V_stride3,
                     Kout_stride0, Kout_stride1, Kout_stride2, Kout_stride3,
                     Vout_stride0, Vout_stride1, Vout_stride2, Vout_stride3,
                     GROUPS: tl.constexpr):
    b = tl.program_id(0)
    kh = tl.program_id(1)
    g = tl.program_id(2)
    j = tl.program_id(3)

    h_target = kh * GROUPS + g

    # K
    k_val = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + kh * K_stride2 + 0 * K_stride3)  # KD is last dim index 0
    tl.store(K_out_ptr + b * Kout_stride0 + j * Kout_stride1 + h_target * Kout_stride2 + 0 * Kout_stride3, k_val)

    # V
    v_val = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + kh * V_stride2 + 0 * V_stride3)
    tl.store(V_out_ptr + b * Vout_stride0 + j * Vout_stride1 + h_target * Vout_stride2 + 0 * Vout_stride3, v_val)


# -----------------------------
# Triton attention compute: for each (b, h), compute scores over tiled i,j, apply causal mask, softmax, and attn_output += scores @ V
# Grid: (B, H)
# We implement:
#   - compute Q[b, h] @ K[b].T -> scores shape [S, S]
#   - apply causal mask (lower-triangular with diagonal=1)
#   - softmax along j dimension
#   - multiply by V and accumulate into out[b, h, :]
# Note: we assume head_dim=128 (as in the original code) and perform math in fp32.
# -----------------------------
@triton.jit
def triton_attention_compute(Q_ptr, K_ptr, V_ptr, Out_ptr,
                             B, S, H,
                             Q_stride0, Q_stride1, Q_stride2, Q_stride3,
                             K_stride0, K_stride1, K_stride2, K_stride3,
                             V_stride0, V_stride1, V_stride2, V_stride3,
                             Out_stride0, Out_stride1, Out_stride2,
                             scaling,  # float32 scalar
                             BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Prepare output accumulator
    # We will accumulate attn_output per token i, then write final out[b, :, h]
    # But Out tensor is [B, S, H], we write each row.
    # Implement as: for each i, compute scores_i, softmax, and accumulate to Out[b, i, h].

    # Loop over i tiles
    for i0 in range(0, S, BLOCK_I):
        i = i0 + tl.arange(0, BLOCK_I)
        mask_i = i < S
        # Initialize scores_i as zeros [BLOCK_I, BLOCK_J] for this tile
        scores = tl.zeros([BLOCK_I, BLOCK_J], dtype=tl.float32)

        # Loop over j tiles
        for j0 in range(0, S, BLOCK_J):
            j = j0 + tl.arange(0, BLOCK_J)
            mask_j = j < S

            # Load Q[i] and K[j]
            # Q shape [B, S, H, KD], we access row i for h
            Q_row = tl.load(
                Q_ptr + b * Q_stride0 + i * Q_stride1 + h * Q_stride2 + 0 * Q_stride3,  # KD last dim is 0 here
                mask=mask_i, other=0.0
            )  # [BLOCK_I, KD]
            K_row = tl.load(
                K_ptr + b * K_stride0 + j * K_stride1 + h * K_stride2 + 0 * K_stride3,
                mask=mask_j, other=0.0
            )  # [BLOCK_J, KD]

            # Compute dot: [BLOCK_I, KD] @ [KD, BLOCK_J] -> [BLOCK_I, BLOCK_J]
            dot = tl.dot(Q_row, tl.trans(K_row))  # [BLOCK_I, BLOCK_J]
            # Add scaling
            scores += dot * scaling

        # Apply causal mask: lower-triangular (i >= j), diagonal=1 => allow i==j
        # Build masks
        i_mat = i[:, None]  # [BLOCK_I, 1]
        j_mat = j[None, :]  # [1, BLOCK_J]
        allow = i_mat >= j_mat
        # Negative infinity for masked elements
        scores = tl.where(allow, scores, -1e30)

        # Softmax along j for each i
        # For numerical stability, subtract row-wise max
        row_max = tl.max(scores, axis=1)  # [BLOCK_I]
        scores = scores - row_max[:, None]  # broadcast over j
        exp_scores = tl.exp(scores)
        row_sum = tl.sum(exp_scores, axis=1)  # [BLOCK_I]
        softmax = exp_scores / row_sum[:, None]  # [BLOCK_I, BLOCK_J]

        # Now accumulate output: out[b, i, h] += softmax @ V[j]
        for j0 in range(0, S, BLOCK_J):
            j = j0 + tl.arange(0, BLOCK_J)
            mask_j = j < S
            V_row = tl.load(
                V_ptr + b * V_stride0 + j * V_stride1 + h * V_stride2 + 0 * V_stride3,
                mask=mask_j, other=0.0
            )  # [BLOCK_J, KD]
            # output_row_i: [BLOCK_I]
            output_row_i = tl.sum(softmax * V_row, axis=1)  # [BLOCK_I]

            # Store into Out[b, i, h]
            # Note: Out is [B, S, H], contiguous in last dim
            base_out = b * Out_stride0 + h * Out_stride2
            for k in range(0, BLOCK_I):
                ii = i[k]
                store_mask = (ii < S) & mask_i[k]
                tl.store(Out_ptr + base_out + ii * Out_stride1, output_row_i[k], mask=store_mask)


# -----------------------------
# Triton linear for output[b, s, h] = sum_k x[b, s, k] * w[h, k]
# x shape: [B, S, H] (here we use attn_output), w shape: [H, K], out shape: [B, S, H]
# Grid: (B, H, S), tile over K
# -----------------------------
@triton.jit
def triton_linear_out(x_ptr, w_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    acc = tl.zeros([1], dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        x_row = tl.load(
            x_ptr + b * x_stride0 + s * x_stride1 + k * x_stride2,
            mask=mask_k, other=0.0
        ).to(tl.float32)
        w_row = tl.load(
            w_ptr + h * w_stride0 + k * w_stride1,
            mask=mask_k, other=0.0
        ).to(tl.float32)
        acc += tl.sum(x_row * w_row, axis=0)

    # Store
    tl.store(
        out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2,
        acc.to(tl.float16), mask=True
    )

        Out = torch.empty((B, S, H), device=device, dtype=torch.float16)

        scaling = 1.0 / math.sqrt(self.head_dim)

            0, 1, 2, 3,
            0, 1, 2, 3,

            BLOCK_I=self.BLOCK_I, BLOCK_J=self.BLOCK_J
        )
        # Note: In Triton, Out strides are accessed via tensor strides. The call above is a placeholder; Triton will
        # use Out's actual strides. We pass Out_ptr and use Out's strides.

        # 6) Output projection: Out = Q @ O_proj^T, no bias
        Out_final = torch.empty((B, S, H), device=device, dtype=torch.float16)

        grid_out = (B, H, S)
        # For output projection, we need x: [B, S, H] (i.e., Out), w: [H, K], here K=H (o_proj_weight dim 1 equals H?):
        # The original code uses o_proj_weight of shape [H, H]. We linear over K=H.
        triton_linear_out[grid_out](

            B, S, H, self.o_proj_weight.shape[1],
            0, 1, 2,
            0, 1,
            0, 1, 2,
            BLOCK_K=self.BLOCK_K
        )

        return Out_final
