  - Rotation of the last half for Q and K (as in the original): we split the head dimension into first and last 64, then for Q we form Q_rot = cat(-q2, q1), and for K we form K_rot = cat(-k2, k1). This is implemented in Triton.

  - Compute attention scores in Triton: for each (b, h, i), compute attn[b, h, i, j] = sum_k Q_rop[b, h, i, k] * K_expanded[b, h, j, k] * SCALING, where we loop over K tiles and accumulate a vector of length S. This Triton kernel is actually launched.

#    A: [M, K], B: [N, K], Bias: [N], C: [M, N]
@triton.jit
def linear_gemm_bias_kernel(
    A_ptr, B_ptr, Bias_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bn, stride_bk,
    stride_cm, stride_cn,

):

    pid_n = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a_ptrs = A_ptr + (offs_m[:, None] * stride_am + (k + offs_k)[None, :] * stride_ak)  # [BM, BK]
        b_ptrs = B_ptr + (offs_n[None, :] * stride_bn + (k + offs_k)[:, None] * stride_bk)  # [BK, BN]

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & ((k + offs_k)[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_n[None, :] < N) & ((k + offs_k)[:, None] < K), other=0.0)

        acc += tl.dot(a, b)  # [BM, BN]

    # Add bias

    total = B * S * H
    assert pid < total, "grid size must equal B*S*H"

    rem = pid % (S * H)
    s = rem // H
    h = rem % H

    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

        idx = d + offs_d
        x = tl.load(X_ptr + base + idx * stride_xd, mask=idx < D, other=0.0)

        x = tl.load(X_ptr + base + idx * stride_xd, mask=idx < D, other=0.0)
