我明白了评分器的严格要求：所有计算必须由 @triton.jit kernels 在 forward 中完成，且每个定义的 kernel 必须实际被调用。之前的多次提交出现 runtime 错误的主要原因有两个：
- kernel pointer arithmetic 和 grid 维度设置不正确，导致非法访问或未正确覆盖所有元素；
- host 代码仍然使用了 torch.triu 或其他 torch.compute 方法来创建因果掩码或进行计算，触发了评分器的违规检测。

为了解决这些问题，我将提供一个完整的 Triton 实现，所有 heavy computation 均由 Triton kernel 完成，并且 forward 中仅做必要的 reshape/view/contiguous 操作以及 kernel 调用。特别地：
- 我将 softmax with causal mask 完全实现在 Triton kernel 中，不使用 torch.triu；
- 所有 projection, RMSNorm, rotation, attention matmul, output matmul, final linear projection 都由 Triton kernel 实现；
- 我会仔细检查每个 kernel 的 grid, strides, BLOCK sizes，并确保所有内存访问都通过 mask 进行保护；
- 我会提供详细的 grid 计算方法，使得所有元素都被覆盖，且没有 out-of-bound 访问；
- 我会确保每个 kernel 都被 forward 实际调用，没有 decoy kernel。

Attention 机制细节
- hidden_states shape: [B, L, hidden_dim] = [B, L, 768], 从原始描述中推断 head_dim=128, num_attention_heads=96, num_key_value_heads=8, num_key_value_groups=12, scaling = head_dim ** -0.5.
- Q, K, V 通过线性投影得到，然后重塑为 [B, L, 96, 128]；
- RMSNorm 用于 Q 和 K；
- Q 和 K 应用 Rotary Position Embedding (RoPE): 对前 64 维使用 cos，后 64 维使用 -sin；
- K/V 通过 groups 从 8 heads 扩展到 96 heads: K/V 为 [B, 8, L, 128] expand(2, 12) -> [B, 8, 12, L, 128], 然后 reshape to [B, 96, L, 128] 用于 attention；
- Attention score: attn_scores[b, qh, l, t] = Q[b, qh, l] * K[b, qh, t]；
- Softmax over t with causal mask: for each (b, qh, l), attn_scores[b, qh, l, t] = -inf if t < l else keep, then softmax along t (subtract max, exp, sum, divide)；
- Output: attn_output[b, qh, l] = sum_t attn_scores[b, qh, l, t] * V[b, qh, t]；
- Final linear: final_out[b, l, :] = attn_output_flat[b, l, :] @ o_proj_weight^T, output [B, L, 768] float32.

现在提供完整的 Triton 实现 ModelNew，forward 中启动所有 kernel，并且不包含任何 torch.exp, torch.sum, torch.matmul, softmax, triu 等计算。

```python
import torch
import triton
import triton.language as tl


# 1) Linear projection: y[b, l, n] = sum_k x[b, l, k] * w[n, k], accumulate in f32, output f32
@triton.jit
def linear_proj_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H_in]
    w_ptr,           # *f16/f32/bf16, [N_out, H_in]
    bias_ptr,        # *f32, [N_out] (or dummy)
    y_ptr,           # *f32, [B, L, N_out]
    B, L, H_in, N_out,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    # Each program computes one output element y[b, l, n]
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)
    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_in, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_in
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)
        # promote to f32
        x_vals = x_vals.to(tl.float32)
        w_vals = w_vals.to(tl.float32)
        acc += tl.sum(x_vals * w_vals, axis=0)
    # add bias if provided (bias_ptr can be dummy)
    # store
    y_ptr_out = y_ptr + b * y_bs0 + l * y_bs1 + n * y_bs2
    tl.store(y_ptr_out, acc)


# 2) RMSNorm for x with weight, no bias: y = x * rsqrt(mean(x^2)+eps) * weight
@triton.jit
def rmsnorm_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H]
    weight_ptr,      # *f16/f32/bf16, [H]
    y_ptr,           # *f32, [B, L, H]
    B, L, H,
    x_bs0, x_bs1, x_bs2,
    weight_bs0,
    y_bs0, y_bs1, y_bs2,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    h = tl.program_id(2)
    # compute mean over H
    sumsq = tl.zeros((), dtype=tl.float32)
    for i in range(0, H):
        xi = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + i * x_bs2)
        xi = xi.to(tl.float32)
        sumsq += xi * xi
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + 0.0)  # eps assumed in host
    w = tl.load(weight_ptr + h * weight_bs0).to(tl.float32)
    # write normalized
    for i in range(0, H):
        xi = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + i * x_bs2)
        xi = xi.to(tl.float32)
        yi = xi * inv_rms * w
        tl.store(y_ptr + b * y_bs0 + l * y_bs1 + i * y_bs2, yi)


# 3) Rotate Q and K: split head into two halves, apply sin/cos
# rotate_q: Q_in is [B, L, H], H=128, out Q_out
@triton.jit
def rotate_qk_kernel(
    x_ptr,           # *f32, [B, L, H]
    sin_ptr,         # *f32, [L, H//2]
    cos_ptr,         # *f32, [L, H//2]
    y_ptr,           # *f32, [B, L, H]
    B, L, H,
    x_bs0, x_bs1, x_bs2,
    y_bs0, y_bs1, y_bs2,
    sin_bs0, sin_bs1,
    cos_bs0, cos_bs1,
    HALF: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    h = tl.program_id(2)
    # split
    if h < HALF:
        # first half: use cos
        s = tl.load(sin_ptr + l * sin_bs0 + h * sin_bs1)
        c = tl.load(cos_ptr + l * cos_bs0 + h * cos_bs1)
        # read original
        xi = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + h * x_bs2)
        # rotate: h_out = cos * h_in - sin * h_in_next
        # here h_in_next is h+HALF; if out of bound, we read 0
        h_next = h + HALF
        if h_next < H:
            xi_next = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + h_next * x_bs2)
        else:
            xi_next = 0.0
        yval = xi * c - xi_next * s
    else:
        # second half: use -sin
        s = tl.load(sin_ptr + l * sin_bs0 + (h - HALF) * sin_bs1)
        c = tl.load(cos_ptr + l * cos_bs0 + (h - HALF) * cos_bs1)
        hi = h - HALF  # index into first half
        xi = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + hi * x_bs2)
        h_next = hi + HALF
        if h_next < H:
            xi_next = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + h_next * x_bs2)
        else:
            xi_next = 0.0
        yval = xi * c + xi_next * s  # use -sin here
    tl.store(y_ptr + b * y_bs0 + l * y_bs1 + h * y_bs2, yval)


# 4) Compute attention scores: attn_scores[b, qh, l, t] = Q[b, qh, l] * K[b, qh, t]
# x_ptr: Q [B, NUM_HEADS, L, H], y_ptr: attn_scores [B, NUM_HEADS, L, L], grid over (B, NUM_HEADS, L)
@triton.jit
def attn_matmul_kernel(
    Q_ptr,           # *f32, [B, NUM_HEADS, L, H]
    K_ptr,           # *f32, [B, NUM_HEADS, L, H]
    attn_ptr,        # *f32, [B, NUM_HEADS, L, L]
    B, NUM_HEADS, L, H,
    Q_bs0, Q_bs1, Q_bs2, Q_bs3,
    K_bs0, K_bs1, K_bs2, K_bs3,
    attn_bs0, attn_bs1, attn_bs2, attn_bs3,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)  # output position
    # compute Q[b, qh, l] * K[b, qh, t] for all t in [0..L-1]
    # We store results into attn[b, qh, l, t]
    for t in range(0, L):
        q = tl.load(Q_ptr + b * Q_bs0 + qh * Q_bs1 + l * Q_bs2 + 0 * Q_bs3)  # H dimension is 128
        q = q.to(tl.float32)  # Q already float32
        k = tl.load(K_ptr + b * K_bs0 + qh * K_bs1 + t * K_bs2 + 0 * K_bs3)
        k = k.to(tl.float32)
        score = q * k  # scalar
        tl.store(attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2 + t * attn_bs3, score)


# 5) Softmax with causal mask in-place on attn: per (b, qh, l), softmax over t with upper-triangular mask (diagonal=1)
@triton.jit
def softmax_mask_kernel(
    attn_ptr,        # *f32, [B, NUM_HEADS, L, L]
    B, NUM_HEADS, L,
    attn_bs0, attn_bs1, attn_bs2, attn_bs3,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)
    # row pointer
    row_ptr = attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2
    # 1) subtract max
    max_val = tl.full((), -float('inf'), dtype=tl.float32)
    for t in range(0, L):
        val = tl.load(row_ptr + t * attn_bs3)
        max_val = tl.maximum(max_val, val)
    for t in range(0, L):
        val = tl.load(row_ptr + t * attn_bs3)
        val = val - max_val
        # causal mask: if t < l, set to -inf
        if t < l:
            val = -float('inf')
        exp_val = tl.exp(val)
        # accumulate sum
        # we will store masked exp after sum
        sum_exp = 0.0
        for j in range(0, L):
            vj = tl.load(row_ptr + j * attn_bs3)
            vj = vj - max_val
            if j < l:
                vj = -float('inf')
            sum_exp += tl.exp(vj)
        # write normalized
        for t in range(0, L):
            vt = tl.load(row_ptr + t * attn_bs3)
            vt = vt - max_val
            if t < l:
                vt = -float('inf')
            out = tl.exp(vt) / sum_exp
            tl.store(row_ptr + t * attn_bs3, out)


# 6) Output matmul: attn_output[b, qh, l] = sum_t attn_scores[b, qh, l, t] * V[b, qh, t]
@triton.jit
def output_matmul_kernel(
    attn_ptr,        # *f32, [B, NUM_HEADS, L, L]
    V_ptr,           # *f32, [B, NUM_HEADS, L, H]
    out_ptr,         # *f32, [B, NUM_HEADS, L, H] (placeholder, but we only need 1D out)
    B, NUM_HEADS, L, H,
    attn_bs0, attn_bs1, attn_bs2, attn_bs3,
    V_bs0, V_bs1, V_bs2, V_bs3,
    out_bs0, out_bs1, out_bs2, out_bs3,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)
    # reduce over t: out[l] = sum_t attn[b,qh,l,t] * V[b,qh,t]
    acc = tl.zeros((), dtype=tl.float32)
    for t in range(0, L):
        at = tl.load(attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2 + t * attn_bs3)
        vt = tl.load(V_ptr + b * V_bs0 + qh * V_bs1 + t * V_bs2 + 0 * V_bs3)  # H dim is 128, but we multiply scalar with row
        acc += at * vt
    # store result into out as a scalar per (b, qh, l). We return full [B, L, H] in host by looping in forward.
    # Here we simply store to out_ptr; host will allocate [B, NUM_HEADS, L, 1] to receive scalar per (b,qh,l), then expand.
    tl.store(out_ptr + b * out_bs0 + qh * out_bs1 + l * out_bs2 + 0 * out_bs3, acc)


# 7) Final linear projection: y[b, l, :] = x[b, l, :] @ o_proj_weight^T, x here is attn_output_flat [B, L, H_flat], o_proj_weight [hidden_dim, H_flat], output [B, L, hidden_dim]
@triton.jit
def final_linear_kernel(
    x_ptr,           # *f32, [B, L, H_flat]
    w_ptr,           # *f32, [hidden_dim, H_flat]
    y_ptr,           # *f32, [B, L, hidden_dim]
    B, L, hidden_dim, H_flat,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)  # hidden_dim
    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_flat, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_flat
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)
        acc += tl.sum(x_vals * w_vals, axis=0)
    tl.store(y_ptr + b * y_bs0 + l * y_bs2 + n * y_bs1, acc)  # careful strides: y is [B, L, hidden_dim]


class ModelNew(torch.nn.Module):
    def __init__(self, hidden_dim: int = 768):
        super().__init__()
        self.hidden_dim = hidden_dim

    def forward(
        self,
        hidden_states: torch.Tensor,
        q_proj_weight: torch.Tensor,
        q_proj_bias: torch.Tensor,
        k_proj_weight: torch.Tensor,
        k_proj_bias: torch.Tensor,
        v_proj_weight: torch.Tensor,
        v_proj_bias: torch.Tensor,
        o_proj_weight: torch.Tensor,
        q_norm_weight: torch.Tensor,
        k_norm_weight: torch.Tensor,
        cos: torch.Tensor,    # [L, H/2]
        sin: torch.Tensor,    # [L, H/2]
        rms_norm_eps: float,
    ):
        # Ensure device/dtype
        assert hidden_states.is_cuda, "All inputs must be on CUDA for Triton"
        B, L, H_in = hidden_states.shape
        H = 128
        NUM_HEADS = 96
        NUM_KV_HEADS = 8
        NUM_KV_GROUPS = 12

        # 1) Linear projections
        Q = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)
        K = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)
        V = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)

        # Launch linear_proj_kernel for Q, K, V
        # We use BLOCK_K=128 which covers H_in=128. If H_in != 128, you can set BLOCK_K accordingly.
        linear_proj_kernel[(B, L, H)](
            hidden_states, q_proj_weight, q_proj_bias, Q,
            B, L, H_in, H,
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            q_proj_weight.stride(0), q_proj_weight.stride(1),
            Q.stride(0), Q.stride(1), Q.stride(2),
            BLOCK_K=128,
            num_warps=4, num_stages=2
        )

        linear_proj_kernel[(B, L, H)](
            hidden_states, k_proj_weight, k_proj_bias, K,
            B, L, H_in, H,
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            k_proj_weight.stride(0), k_proj_weight.stride(1),
            K.stride(0), K.stride(1), K.stride(2),
            BLOCK_K=128,
            num_warps=4, num_stages=2
        )

        linear_proj_kernel[(B, L, H)](
            hidden_states, v_proj_weight, v_proj_bias, V,
            B, L, H_in, H,
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            v_proj_weight.stride(0), v_proj_weight.stride(1),
            V.stride(0), V.stride(1), V.stride(2),
            BLOCK_K=128,
            num_warps=4, num_stages=2
        )

        # 2) RMSNorm for Q and K
        Q_norm = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)
        K_norm = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)

        # Q RMSNorm
        rmsnorm_kernel[(B, L, H)](
            Q, q_norm_weight, Q_norm,
            B, L, H,
            Q.stride(0), Q.stride(1), Q.stride(2),
            q_norm_weight.stride(0),
            Q_norm.stride(0), Q_norm.stride(1), Q_norm.stride(2),
            num_warps=4, num_stages=2
        )

        # K RMSNorm
        rmsnorm_kernel[(B, L, H)](
            K, k_norm_weight, K_norm,
            B, L, H,
            K.stride(0), K.stride(1), K.stride(2),
            k_norm_weight.stride(0),
            K_norm.stride(0), K_norm.stride(1), K_norm.stride(2),
            num_warps=4, num_stages=2
        )

        # 3) Rotate Q and K
        Q_rot = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)
        K_rot = torch.empty((B, L, H), dtype=torch.float32, device=hidden_states.device)

        rotate_qk_kernel[(B, L, H)](
            Q_norm, cos, sin, Q_rot,
            B, L, H,
            Q_norm.stride(0), Q_norm.stride(1), Q_norm.stride(2),
            Q_rot.stride(0), Q_rot.stride(1), Q_rot.stride(2),
            cos.stride(0), cos.stride(1),
            sin.stride(0), sin.stride(1),
            HALF=64,
            num_warps=4, num_stages=2
        )

        rotate_qk_kernel[(B, L, H)](
            K_norm, cos, sin, K_rot,
            B, L, H,
            K_norm.stride(0), K_norm.stride(1), K_norm.stride(2),
            K_rot.stride(0), K_rot.stride(1), K_rot.stride(2),
            cos.stride(0), cos.stride(1),
            sin.stride(0), sin.stride(1),
            HALF=64,
            num_warps=4, num_stages=2
        )

        # 4) GQA: expand K/V to 96 heads
        # We need K_rot and V for 96 heads. original code expands K/V from 8 to 96 via groups. Here we use expand view to match attention head count. Note: This assumes no per-head K/V weight; original code doesn't provide them either.
        K_expanded = K_rot.unsqueeze(2).expand(B, NUM_HEADS, L, H).reshape(B, NUM_HEADS, L, H).contiguous()
        V_expanded = V.unsqueeze(2).expand(B, NUM_HEADS, L, H).reshape(B, NUM_HEADS, L, H).contiguous()

        # 5) Allocate attn scores [B, NUM_HEADS, L, L]
        attn_scores = torch.empty((B, NUM_HEADS, L, L), dtype=torch.float32, device=hidden_states.device)

        # Launch attn_matmul_kernel: grid over (B, NUM_HEADS, L)
        attn_matmul_kernel[(B, NUM_HEADS, L)](
            Q_rot, K_expanded, attn_scores,
            B, NUM_HEADS, L, H,
            Q_rot.stride(0), Q_rot.stride(1), Q_rot.stride(2), Q_rot.stride(3),
            K_expanded.stride(0), K_expanded.stride(1), K_expanded.stride(2), K_expanded.stride(3),
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            num_warps=4, num_stages=2
        )

        # 6) Softmax with causal mask: in-place
        softmax_mask_kernel[(B, NUM_HEADS, L)](
            attn_scores,
            B, NUM_HEADS, L,
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            num_warps=4, num_stages=2
        )

        # 7) Output matmul: compute attn_output[b, qh, l] = sum_t attn_scores[b, qh, l, t] * V_expanded[b, qh, t]
        # We'll compute per (b, qh, l) as scalar, then write to [B, NUM_HEADS, L, 1]
        attn_output_scalar = torch.empty((B, NUM_HEADS, L, 1), dtype=torch.float32, device=hidden_states.device)

        # Launch output_matmul_kernel: grid over (B, NUM_HEADS, L)
        # Note: this kernel writes a scalar per (b,qh,l) to attn_output_scalar. We then expand along H dimension in host.
        output_matmul_kernel[(B, NUM_HEADS, L)](
            attn_scores, V_expanded, attn_output_scalar,
            B, NUM_HEADS, L, H,
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            V_expanded.stride(0), V_expanded.stride(1), V_expanded.stride(2), V_expanded.stride(3),
            attn_output_scalar.stride(0), attn_output_scalar.stride(1), attn_output_scalar.stride(2), attn_output_scalar.stride(3),
            num_warps=4, num_stages=2
        )

        # Expand attn_output_scalar to [B, NUM_HEADS, L, H] for final linear
        attn_output = attn_output_scalar.expand(B, NUM_HEADS, L, H).contiguous()

        # 8) Final linear: [B, L, H] @ o_proj_weight^T -> [B, L, hidden_dim]
        final_out = torch.empty((B, L, self.hidden_dim), dtype=torch.float32, device=hidden_states.device)

        # o_proj_weight is [hidden_dim, H_flat], H_flat = NUM_HEADS * H = 96 * 128 = 12288
        final_linear_kernel[(B, L, self.hidden_dim)](
            attn_output, o_proj_weight, final_out,
            B, L, self.hidden_dim, NUM_HEADS * H,
            attn_output.stride(0), attn_output.stride(1), attn_output.stride(2),
            o_proj_weight.stride(0), o_proj_weight.stride(1),
            final_out.stride(0), final_out.stride(1), final_out.stride(2),
            BLOCK_K=64,
            num_warps=4, num_stages=2
        )
```
