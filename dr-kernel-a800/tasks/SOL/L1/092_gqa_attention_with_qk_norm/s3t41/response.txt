Performance notes:
- The Triton kernels replace torch.linear, RMSNorm, and attention (matmul + softmax). This should produce a clear speedup on GPU compared to the original PyTorch implementation for the given sizes, especially since the seq_len up to 2048 and batch sizes up to 32.

Below is the complete Triton-based implementation. It defines the following kernels:
- linear_kernel: computes Y = X @ W^T + B with X of shape [M, K], W of shape [N, K], Y of shape [M, N].
- rmsnorm_kernel: computes y = x * rsqrt(mean(x^2) + eps) * weight over last dim D, for each (b, h, s).
- rotate_half_kernel: applies the rotation for the last dim=128 by splitting into q1 and q2 halves.
- attn_scores_kernel: computes attention scores S[b, h, i, j] = sum_k Q[b,h,i,k] * K[b,h,j,k] * scaling, using Q and K reshaped to [B, H, S, 128].
- softmax_cols_kernel: computes softmax along the sequence dimension for each (b, h), producing Soft[b, h, :, :].
- attn_output_kernel: computes Y[b, h, i] = sum_j Soft[b,h,i,j] * V[b,h,j] for each (b, h, i).

import torch
import torch.nn as nn
import torch.nn.functional as F

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True

# 1) Linear: Y = X @ W^T + B
# X: [M, K], W: [N, K], B: [N], Y: [M, N]
@triton.jit
def linear_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    M, K, N,
    stride_xm, stride_xk,
    stride_w0, stride_w1,
    stride_ym, stride_yn,
    num_warps: tl.constexpr,
):
    m = tl.program_id(axis=0)  # row in M
    n = tl.program_id(axis=1)  # out dim
    acc = tl.zeros((), dtype=tl.float32)
    # loop over K in chunks
    for k0 in range(0, K, 64):
        offs_k = k0 + tl.arange(0, 64)
        mask_k = offs_k < K
        x = tl.load(X_ptr + m * stride_xm + offs_k * stride_xk, mask=mask_k, other=0.0)  # [64]
        w = tl.load(W_ptr + n * stride_w0 + offs_k * stride_w1, mask=mask_k, other=0.0)  # [64]
        acc += tl.sum(x * w, axis=0)
    # add bias
    b = tl.load(B_ptr + n)
    acc += b
    # store
    tl.store(Y_ptr + m * stride_ym + n * stride_yn, acc)


# 2) RMSNorm per (b, h, s, d): y = x * rsqrt(mean(x^2) + eps) * weight
@triton.jit
def rmsnorm_kernel(
    X_ptr, W_ptr, Y_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    eps: tl.float32,
    num_warps: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    d = tl.program_id(axis=3)
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d * stride_xd).to(tl.float32)
    sum_sq = x * x
    mean_sq = tl.sum(sum_sq, axis=0) / D
    inv_rms = tl.rsqrt(mean_sq + eps)
    w = tl.load(W_ptr + d).to(tl.float32)
    y = (x * inv_rms) * w
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d * stride_yd, y)


# 3) Rotate half for last dim=128: q1 = x[:64], q2 = x[64:], rotated = x * cos + cat((-q2, q1), -1) * sin
@triton.jit
def rotate_half_kernel(
    X_ptr, C_ptr, S_ptr, Y_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    stride_c0, stride_c1,     # cos: [S, D/2]
    stride_s0, stride_s1,     # sin: [S, D/2]
    num_warps: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    d = tl.program_id(axis=3)
    # load x, cos, sin
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d * stride_xd).to(tl.float32)
    d2 = D // 2
    cos = tl.load(C_ptr + s * stride_c0 + d * stride_c1).to(tl.float32)
    sin = tl.load(S_ptr + s * stride_s0 + d * stride_s1).to(tl.float32)
    q1 = x[:64]
    q2 = x[64:]
    rotated_half = tl.cat((-q2, q1), axis=0)
    y = x * cos + rotated_half * sin
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d * stride_yd, y)


# 4) Compute attention scores: S[b,h,i,j] = sum_k Q[b,h,i,k] * K[b,h,j,k] * scaling, for each (b,h)
@triton.jit
def attn_scores_kernel(
    Q_ptr, K_ptr, Scores_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_qb, stride_qh, stride_qs, stride_qd,
    stride_kb, stride_kh, stride_ks, stride_kd,
    stride_sb, stride_sh, stride_si, stride_sj,
    scaling: tl.float32,
    num_warps: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    i = tl.program_id(axis=2)  # query token
    # compute scores[i, :]
    acc = tl.zeros((D,), dtype=tl.float32)  # scores vector over D
    for k0 in range(0, D, 64):
        offs = k0 + tl.arange(0, 64)
        q = tl.load(Q_ptr + b * stride_qb + h * stride_qh + i * stride_qs + offs * stride_qd, mask=offs < D, other=0.0)
        for j in range(0, S):
            k = tl.load(K_ptr + b * stride_kb + h * stride_kh + j * stride_ks + offs * stride_kd, mask=offs < D, other=0.0)
            acc += (q * k)  # accumulate over D chunks
    acc = acc * scaling
    # store scores[b, h, i, :]
    tl.store(Scores_ptr + b * stride_sb + h * stride_sh + i * stride_si + tl.arange(0, D) * stride_sj, acc, mask=tl.arange(0, D) < D)


# 5) Softmax over sequence dimension for each (b,h): Soft[b,h,:,j] = softmax(scores[i,j]) over i
@triton.jit
def softmax_cols_kernel(
    Scores_ptr, Soft_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr,
    stride_sb, stride_sh, stride_si, stride_sj,
    stride_pb, stride_ph, stride_pi, stride_pj,
    num_warps: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    # per column j, compute softmax over i
    for j in range(0, S):
        scores_col = tl.load(Scores_ptr + b * stride_sb + h * stride_sh + tl.arange(0, S) * stride_si + j * stride_sj, mask=tl.arange(0, S) < S, other=-float('inf'))
        m = tl.max(scores_col, axis=0)
        scores_col = scores_col - m
        exp_col = tl.exp(scores_col)
        sum_exp = tl.sum(exp_col, axis=0)
        soft_col = exp_col / sum_exp
        tl.store(Soft_ptr + b * stride_pb + h * stride_ph + tl.arange(0, S) * stride_pi + j * stride_pj, soft_col, mask=tl.arange(0, S) < S)


# 6) Compute attention output per (b,h,i): Y[b,h,i] = sum_j Soft[b,h,i,j] * V[b,h,j]
@triton.jit
def attn_output_kernel(
    Soft_ptr, V_ptr, Y_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_sb, stride_sh, stride_si, stride_sj,
    stride_vb, stride_vh, stride_vj,
    stride_yb, stride_yh, stride_yi,
    num_warps: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    acc = tl.zeros((), dtype=tl.float32)
    for j in range(0, S):
        soft_i_j = tl.load(Soft_ptr + b * stride_sb + h * stride_sh + i * stride_si + j * stride_sj)
        v = tl.load(V_ptr + b * stride_vb + h * stride_vh + j * stride_vj)  # vector over D
        acc += soft_i_j * tl.sum(v, axis=0)
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + i * stride_yi, acc)

# Triton kernel: Linear Y = X @ W^T + B
# X: [M, K], W: [N, K], B: [N], Y: [M, N]
@triton.jit
def linear_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    M, K, N,
    stride_xm, stride_xk,
    stride_w0, stride_w1,
    stride_ym, stride_yn,
    num_warps: tl.constexpr,
):
    m = tl.program_id(axis=0)  # row in M
    n = tl.program_id(axis=1)  # output dim
    acc = tl.zeros((), dtype=tl.float32)
    # Loop over K in chunks
    for k0 in range(0, K, 64):
        offs_k = k0 + tl.arange(0, 64)
        mask_k = offs_k < K
        x = tl.load(X_ptr + m * stride_xm + offs_k * stride_xk, mask=mask_k, other=0.0)  # [64]
        w = tl.load(W_ptr + n * stride_w0 + offs_k * stride_w1, mask=mask_k, other=0.0)  # [64]
        acc += tl.sum(x * w, axis=0)
    # Add bias
    b = tl.load(B_ptr + n)
    acc += b
    # Store Y[m, n]
    tl.store(Y_ptr + m * stride_ym + n * stride_yn, acc)


# Triton kernel: RMSNorm per (b, h, s, d) -> y = x * rsqrt(mean(x^2) + eps) * weight
# X: [B, H, S, D], W: [D], Y: [B, H, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, W_ptr, Y_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    eps: tl.float32,
    num_warps: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    d = tl.program_id(axis=3)
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d * stride_xd).to(tl.float32)
    sum_sq = x * x
    mean_sq = tl.sum(sum_sq, axis=0) / D
    inv_rms = tl.rsqrt(mean_sq + eps)
    w = tl.load(W_ptr + d).to(tl.float32)
    y = (x * inv_rms) * w
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d * stride_yd, y)
