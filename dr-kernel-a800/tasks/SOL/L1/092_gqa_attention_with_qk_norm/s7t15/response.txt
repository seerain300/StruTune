# 1) Triton dense linear: out[b, s, h] = sum_k input[b, :, k] * weight[h, k] + bias[h]
# Input is [B, S, K], weight [H, K], bias [H], output [B, S, H]
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, b_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    # Initialize output row
    out_row_ptr = out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over K in tiles
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        # Load x[b, s, k]
        x = tl.load(x_ptr + b * x_stride0 + s * x_stride1 + k * x_stride2, mask=mask_k, other=0.0)
        # Load weight[h, k]
        w = tl.load(w_ptr + h * w_stride0 + k * w_stride1, mask=mask_k, other=0.0)
        # Accumulate dot product for this tile
        acc += tl.sum(x * w, axis=0)

    # Add bias
    bias = tl.load(b_ptr + h)
    acc = acc + bias

    # Store result (cast handled by caller if needed)
    tl.store(out_row_ptr, acc)


# 2) Triton RMSNorm on Q/K: normalize each row (b, h) across S
# out[b, h, s] = x[b, h, s] * (weight[h] / sqrt(mean(x[b,h,:^2]) + eps))
@triton.jit
def triton_rmsnorm_bhs(x_ptr, weight_ptr, out_ptr,
                       B, H, S,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       eps: tl.constexpr,
                       BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    row_sum = 0.0
    for d0 in range(0, S, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < S
        x = tl.load(x_ptr + b * x_stride0 + h * x_stride1 + d * x_stride2, mask=mask, other=0.0)
        row_sum += tl.sum(x * x, axis=0)

    mean = row_sum / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = tl.load(weight_ptr + h) * inv_rms

    for d0 in range(0, S, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < S
        x = tl.load(x_ptr + b * x_stride0 + h * x_stride1 + d * x_stride2, mask=mask, other=0.0)
        y = x * scale
        tl.store(out_ptr + b * out_stride0 + h * out_stride1 + d * out_stride2, y, mask=mask)


# 3) Triton RoPE: rotate query/key rows. Split 128 into 64+64.
@triton.jit
def triton_rope_bhs(x_ptr, cos_ptr, sin_ptr, out_ptr,
                    B, H, S, head_dim,
                    x_stride0, x_stride1, x_stride2,
                    cos_stride0, cos_stride1,
                    sin_stride0, sin_stride1,
                    out_stride0, out_stride1, out_stride2,
                    BLOCK_D: tl.constexpr):

    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=1.0)  # cos initialized

#    scores[i, j] = (Q[i] * K[j]) * scaling; apply causal mask; softmax over j; attn_output[i] += scores[i, j] * V[j]
@triton.jit
def attention_compute_bh(Q_ptr, K_ptr, V_ptr, Out_ptr,

    b = tl.program_id(0)
    h = tl.program_id(1)

    out_row_ptr = Out_ptr + b * Out_stride0 + h * Out_stride1
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over i in tiles
    for i0 in range(0, S, BLOCK_I):
        i_idx = i0 + tl.arange(0, BLOCK_I)
        mask_i = i_idx < S

        # Load Q[i] vector for this tile
        q_vec = tl.load(Q_ptr + b * Q_stride0 + i_idx * Q_stride1 + h * Q_stride2, mask=mask_i, other=0.0)

        # Compute scores[i, j] for j in tiles
        for j0 in range(0, S, BLOCK_J):
            j_idx = j0 + tl.arange(0, BLOCK_J)
            mask_j = j_idx < S

            # Load K[j] and V[j]
            k_mat = tl.load(K_ptr + b * K_stride0 + j_idx * K_stride1 + h * K_stride2, mask=mask_j, other=0.0)
            v_mat = tl.load(V_ptr + b * V_stride0 + j_idx * V_stride1 + h * V_stride2, mask=mask_j, other=0.0)

            # scores = Q[i] dot K[j]
            scores = tl.sum(q_vec[:, None] * k_mat[None, :], axis=1)  # shape [BLOCK_I]
            scores = scores * scaling  # scaling = 1/sqrt(head_dim)

            # Causal mask: for i>t, set to -inf; otherwise 0
            for ii in range(0, BLOCK_I):
                row_i = i0 + ii
                valid_row = row_i < S
                if valid_row:
                    for jj in range(0, BLOCK_J):
                        col_j = j0 + jj
                        valid_col = col_j < S
                        if valid_col:
                            if row_i <= col_j:
                                scores[ii] = 0.0
                            else:
                                scores[ii] = -float('inf')

            # Softmax along j for each i
            scores = scores - tl.max(scores, axis=0)  # subtract max
            scores = tl.exp(scores) / tl.sum(tl.exp(scores), axis=0)  # softmax

            # Accumulate output: attn_output[i] += scores * V[j]
            acc += tl.sum(scores[:, None] * v_mat[None, :], axis=1)

    # Store final acc for this (b, h)
    tl.store(out_row_ptr, acc)
