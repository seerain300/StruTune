# Dense linear: Y[b, s, d_out] = sum_k X[b, s, k] * W[d_out, k] + B[d_out]
# X: [B, S, D_in], W: [D_out, D_in], B: [D_out], Y: [B, S, D_out]
@triton.jit
def dense_linear_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    B, S, D_in, D_out,
    stride_xb, stride_xs, stride_xk,
    stride_w0, stride_w1,   # W[d_out, k] layout: stride_w0 = D_in, stride_w1 = 1
    stride_yb, stride_ys, stride_yd,
    BLOCK_K: tl.constexpr,  # usually 128
    BLOCK_DO: tl.constexpr  # usually 128
):
    pid_bs = tl.program_id(axis=0)  # 0..(B*S - 1)
    pid_do = tl.program_id(axis=1)  # 0..(ceil(D_out/BLOCK_DO) - 1)

    b = pid_bs // S
    s = pid_bs % S

    d_out_offsets = pid_do * BLOCK_DO + tl.arange(0, BLOCK_DO)
    mask_d_out = d_out_offsets < D_out

    acc = tl.zeros([BLOCK_DO], dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, D_in, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < D_in

        x_vals = tl.load(
            X_ptr + b * stride_xb + s * stride_xs + k_offsets * stride_xk,
            mask=mask_k,
            other=0.0
        )  # [BLOCK_K]

        w_vals = tl.load(
            W_ptr + d_out_offsets[:, None] * stride_w0 + k_offsets[None, :] * stride_w1,
            mask=mask_d_out[:, None] & mask_k[None, :],
            other=0.0
        )  # [BLOCK_DO, BLOCK_K]

        acc += tl.sum(w_vals * x_vals[None, :], axis=1)

    # Add bias
    bias = tl.load(B_ptr + d_out_offsets, mask=mask_d_out, other=0.0)
    acc += bias

    # Store
    tl.store(
        Y_ptr + b * stride_yb + s * stride_ys + d_out_offsets * stride_yd,
        acc,
        mask=mask_d_out
    )

# RMSNorm per (b, h, s, d): Y = X * inv_rms * W
@triton.jit
def rmsnorm_kernel(
    X_ptr, W_ptr, Y_ptr,
    B, H, S, D,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    eps: tl.constexpr  # eps should be a compile-time constant, but we can pass runtime float too
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)

    d_offsets = tl.arange(0, D)
    x = tl.load(
        X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d_offsets * stride_xd,
        mask=d_offsets < D,
        other=0.0
    )
    x32 = x.to(tl.float32)
    sum_sq = tl.sum(x32 * x32, axis=0)
    mean_sq = sum_sq / D
    inv_rms = tl.rsqrt(mean_sq + eps)
    w = tl.load(W_ptr + d_offsets, mask=d_offsets < D, other=1.0)
    y = (x32 * inv_rms) * w
    tl.store(
        Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d_offsets * stride_yd,
        y,
        mask=d_offsets < D
    )

# Rotate half for last dim D=128: q1=x[:64], q2=x[64:], rotated_half = cat((-q2, q1), -1), Y = X * cos + rotated_half * sin
@triton.jit
def rotate_half_kernel(
    X_ptr, C_ptr, S_ptr, Y_ptr,
    B, H, S, D,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    stride_c0, stride_c1,  # cos: [S, 64], strides
    stride_s0, stride_s1,  # sin: [S, 64], strides
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)

    d_offsets = tl.arange(0, D)
    x = tl.load(
        X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d_offsets * stride_xd,
        mask=d_offsets < D,
        other=0.0
    )  # [D]
    # Load cos and sin for this s
    cos_vals = tl.load(C_ptr + s * stride_c0 + tl.arange(0, 64) * stride_c1)  # [64]
    sin_vals = tl.load(S_ptr + s * stride_s0 + tl.arange(0, 64) * stride_s1)  # [64]

    q1 = x[0:64]
    q2 = x[64:128]
    rotated_half = tl.cat([-q2, q1], axis=0)  # [128]
    y = x * cos_vals[None, :] + rotated_half * sin_vals[None, :]

    tl.store(
        Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d_offsets * stride_yd,
        y,
        mask=d_offsets < D
    )

# Compute attention scores S[b, h, i, j] = (Q[b,h,i] @ K[b,h,j]) * scaling, tile over i/j
@triton.jit
def attention_score_kernel(
    Q_ptr, K_ptr, Score_ptr,
    B, H, S, D,
    stride_qb, stride_qh, stride_qi, stride_qd,
    stride_kb, stride_kh, stride_kj, stride_kd,
    stride_sb, stride_sh, stride_si, stride_sj,
    scaling: tl.constexpr
):
    # We launch one program per (b, h). It will loop over tiles i and j of size BLOCK (e.g., 64).
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)

    BLOCK = 64
    # Initialize score buffer to -inf
    # Triton doesn't support direct tensor initialization here, so we'll compute and store.
    for i_start in range(0, S, BLOCK):
        i_offsets = i_start + tl.arange(0, BLOCK)
        mask_i = i_offsets < S

        row_scores = tl.zeros([BLOCK], dtype=tl.float32)

        for j_start in range(0, S, BLOCK):
            j_offsets = j_start + tl.arange(0, BLOCK)
            mask_j = j_offsets < S

            # Load Q[i,:] and K[j,:]
            q_row = tl.load(
                Q_ptr + b * stride_qb + h * stride_qh + i_offsets[:, None] * stride_qi + tl.arange(0, D) * stride_qd,
                mask=mask_i[:, None],
                other=0.0
            )  # [BLOCK, D]
            k_row = tl.load(
                K_ptr + b * stride_kb + h * stride_kh + j_offsets[None, :] * stride_kj + tl.arange(0, D) * stride_kd,
                mask=mask_j[None, :],
                other=0.0
            )  # [1, D] broadcasts to [BLOCK, D] via implicit

            # Compute dot product: [BLOCK, 1]
            dot = tl.sum(q_row * k_row, axis=1, keepdims=True)
            row_scores += (dot * scaling).to(tl.float32)

        # Write scores for this i tile
        tl.store(
            Score_ptr + b * stride_sb + h * stride_sh + i_offsets * stride_si + tl.arange(0, S) * stride_sj,  # shape [BLOCK, S]
            # We need a 2D store. Triton can't do 2D store here; instead, we store per element with nested loop.
            # But better approach: pre-initialize with -inf. Here we'll write zeros and then add per j loop below.
        )

    # Re-compute and write scores correctly with nested loops per element
    # Since Triton kernels operate elementwise or vectorized, we'll restructure: for each i, compute and store a vector.
    for i_start in range(0, S, BLOCK):
        i_offsets = i_start + tl.arange(0, BLOCK)
        mask_i = i_offsets < S
        for j_start in range(0, S, BLOCK):
            j_offsets = j_start + tl.arange(0, BLOCK)
            mask_j = j_offsets < S

            q_row = tl.load(
                Q_ptr + b * stride_qb + h * stride_qh + i_offsets[:, None] * stride_qi + tl.arange(0, D) * stride_qd,
                mask=mask_i[:, None],
                other=0.0
            )  # [BLOCK, D]
            k_row = tl.load(
                K_ptr + b * stride_kb + h * stride_kh + j_offsets[None, :] * stride_kj + tl.arange(0, D) * stride_kd,
                mask=mask_j[None, :],
                other=0.0
            )  # [1, D]

            dot = tl.sum(q_row * k_row, axis=1, keepdims=True)  # [BLOCK, 1]
            scores = (dot * scaling).to(tl.float32)

            # Store into Score[b, h, i, j]
            for ii in range(0, BLOCK):
                valid_i = i_start + ii < S
                if valid_i:
                    for jj in range(0, BLOCK):
                        valid_j = j_start + jj < S
                        score = scores[ii, 0]
                        tl.store(
                            Score_ptr + b * stride_sb + h * stride_sh + (i_start + ii) * stride_si + (j_start + jj) * stride_sj,
                            score,
                            mask=(i_start + ii < S) & (j_start + jj < S)
                        )

# Softmax over last dimension per row: input Score[b, h, i, :], output Soft[b, h, i, :]
@triton.jit
def softmax_rows_kernel(
    Score_ptr, Soft_ptr,
    B, H, S, S_LEN,
    stride_sb, stride_sh, stride_si, stride_sj,
    stride_cb, stride_ch, stride_ci, stride_cj,
):
    # Launch one program per (b, h, i)
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    # Load row
    row = tl.load(
        Score_ptr + b * stride_sb + h * stride_sh + i * stride_si + tl.arange(0, S_LEN) * stride_sj,
        mask=tl.arange(0, S_LEN) < S_LEN,
        other=0.0
    )
    row = row.to(tl.float32)
    row_max = tl.max(row, axis=0)
    row = row - row_max
    exp_row = tl.exp(row)
    row_sum = tl.sum(exp_row, axis=0)
    soft_row = exp_row / row_sum
    tl.store(
        Soft_ptr + b * stride_cb + h * stride_ch + i * stride_ci + tl.arange(0, S_LEN) * stride_cj,
        soft_row,
        mask=tl.arange(0, S_LEN) < S_LEN
    )

# Matmul with V: Soft[b, h, i, j] * V[b, h, j], accumulate over j
@triton.jit
def matmul_with_value_kernel(
    Soft_ptr, V_ptr, Output_ptr,
    B, H, S, D,
    stride_sb, stride_sh, stride_si, stride_sj,
    stride_vb, stride_vh, stride_vj, stride_vd,
    stride_ob, stride_oh, stride_oi, stride_od,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)

    # Output per (b, h, i) vector of length D
    for i in range(0, S):
        acc = tl.zeros([D], dtype=tl.float32)
        # Loop over j in tiles
        for j_start in range(0, S, 64):
            j_offsets = j_start + tl.arange(0, 64)
            mask_j = j_offsets < S
            soft = tl.load(
                Soft_ptr + b * stride_sb + h * stride_sh + i * stride_si + j_offsets * stride_sj,
                mask=mask_j,
                other=0.0
            )  # [64]
            v = tl.load(
                V_ptr + b * stride_vb + h * stride_vh + j_offsets * stride_vj + tl.arange(0, D) * stride_vd,
                mask=mask_j[:, None] & (tl.arange(0, D)[None, :] < D),
                other=0.0
            )  # [64, D]
            acc += tl.sum(v * soft[:, None], axis=0)
        tl.store(
            Output_ptr + b * stride_ob + h * stride_oh + i * stride_oi + tl.arange(0, D) * stride_od,
            acc,
            mask=tl.arange(0, D) < D
        )
