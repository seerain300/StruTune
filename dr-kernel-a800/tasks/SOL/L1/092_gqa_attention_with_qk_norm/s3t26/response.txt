- Linear operations in Triton: F.linear(hidden, W, b) for Q, K, V and for the final output projection O. I’ll use a 2D grid over (B*S, D_out) and loop over D_in in chunks (e.g., 64) to compute Y = X @ W^T + B. All accumulation will be in float32.
- RMSNorm in Triton: per (b, h, s, d), compute inv_rms = 1/sqrt(mean(x^2) + eps), then y = x * inv_rms * weight. This matches the original q_norm and k_norm in the provided code.
- Rotation (RoPE) in Triton: per (b, h, s), split head dim into halves q1 (first 64) and q2 (last 64), and rotate using cos/sin tensors of shape [S, 64]: rotated_half = cat((-q2, q1), -1), then Y = X * cos + rotated_half * sin. This matches the original behavior.
- Attention computation in Triton: compute attention scores S[b, h, i, j] = Q[b,h,i,:] @ K[b,h,j,:] * scaling (with scaling = 1/sqrt(head_dim)). I’ll use a 3D grid over (b, h, i) and loop over i and j in tiles, accumulating scores in float32 and applying the causal mask in-kernel. I will not use PyTorch softmax/matmul here.
- Softmax in Triton: column-wise softmax over the sequence dimension (Soft[b, h, i, :]) per (b, h, i), using an online max-reduction for numerical stability, and store Soft in float32. This replaces torch.softmax used earlier.
- Output projection in Triton: compute attn_output @ o_proj_weight^T + bias (bias can be None). This is a linear on [B*S, num_attention_heads*head_dim] to [B*S, model_dim]. We’ll loop over input_dim in chunks and accumulate in float32.

import torch
import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

# Triton kernel: dense linear Y = X @ W^T + B
# X: [M, K] where M = B*S, K = D_in
# W: [N, K] where N = D_out
# B: [N]
# Y: [M, N]
@triton.jit
def linear_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_xm, stride_xk,
    stride_w0, stride_w1,  # W has shape [N, K]
    stride_ym, stride_yn,
    BLOCK_K: tl.constexpr,
):
    m = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        offs = k0 + tl.arange(0, BLOCK_K)
        mask = offs < K
        x = tl.load(X_ptr + m * stride_xm + offs * stride_xk, mask=mask, other=0.0)  # [BLOCK_K]
        w = tl.load(W_ptr + n * stride_w0 + offs * stride_w1, mask=mask, other=0.0)  # [BLOCK_K]
        acc += tl.sum(x * w, axis=0)
    b = tl.load(B_ptr + n)
    acc += b
    tl.store(Y_ptr + m * stride_ym + n * stride_yn, acc)


# Triton kernel: RMSNorm per (b, h, s, d)
# X: [B, H, S, D], W: [D], Y: [B, H, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, W_ptr, Y_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    eps: tl.float32,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    d = tl.program_id(axis=3)
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d * stride_xd).to(tl.float32)
    sum_sq = x * x
    mean_sq = tl.sum(sum_sq, axis=0) / D
    inv_rms = tl.rsqrt(mean_sq + eps)
    w = tl.load(W_ptr + d).to(tl.float32)
    y = (x * inv_rms) * w
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d * stride_yd, y)


# Triton kernel: apply rotation (RoPE) for last dim D=128, split into two halves (64,64)
# X: [B, H, S, D], C: [S, D/2], S: [S, D/2], Y: [B, H, S, D]
@triton.jit
def rotate_half_kernel(
    X_ptr, C_ptr, S_ptr, Y_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    stride_c0, stride_c1,     # cos strides: [S, D/2]
    stride_s0, stride_s1,     # sin strides: [S, D/2]
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    # Load cos and sin for this s
    cos_vals = tl.load(C_ptr + s * stride_c0 + tl.arange(0, D//2) * stride_c1).to(tl.float32)  # [D/2]
    sin_vals = tl.load(S_ptr + s * stride_s0 + tl.arange(0, D//2) * stride_s1).to(tl.float32)  # [D/2]
    # Load original X[b, h, s, :]
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + tl.arange(0, D) * stride_xd).to(tl.float32)
    # Split into halves
    q1 = x[:64]
    q2 = x[64:]
    rotated_half = tl.cat([-q2, q1], axis=0)
    y = x * cos_vals + rotated_half * sin_vals
    # Store result
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + tl.arange(0, D) * stride_yd, y)


# Triton kernel: compute attention scores S[b, h, i, j] = Q[b,h,i,:] @ K[b,h,j,:] * scaling
# Inputs: Q: [B, H, S, D], K: [B, H, S, D], Outputs: S: [B, H, S, S]
@triton.jit
def attn_scores_kernel(
    Q_ptr, K_ptr, S_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, D: tl.constexpr,
    stride_qb, stride_qh, stride_qs, stride_qd,
    stride_kb, stride_kh, stride_ks, stride_kd,
    stride_sb, stride_sh, stride_si, stride_sj,
    scaling: tl.float32,
    BLOCK_J: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    # Accumulate scores for j in tiles
    for j0 in range(0, S, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)  # [BLOCK_J]
        mask_j = j_idx < S
        q_row = tl.load(Q_ptr + b * stride_qb + h * stride_qh + i * stride_qs + tl.arange(0, D) * stride_qd).to(tl.float32)  # [D]
        k_block = tl.load(K_ptr + b * stride_kb + h * stride_kh + j_idx * stride_ks + tl.arange(0, D) * stride_kd)  # [BLOCK_J, D]
        scores = tl.zeros((BLOCK_J,), dtype=tl.float32)
        for d0 in range(0, D, 64):
            offs_d = d0 + tl.arange(0, 64)
            mask_d = offs_d < D
            q_chunk = q_row[offs_d]  # [64]
            k_chunk = k_block[:, offs_d]  # [BLOCK_J, 64]
            scores += tl.sum(q_chunk[None, :] * k_chunk, axis=1)
        scores *= scaling
        # Apply causal mask: if i >= j, keep; else -inf
        causal = (i >= (j0 + tl.arange(0, BLOCK_J)))  # [BLOCK_J]
        scores = tl.where(mask_j & causal, scores, -float('inf'))
        tl.store(S_ptr + b * stride_sb + h * stride_sh + i * stride_si + j_idx * stride_sj, scores, mask=mask_j)


# Triton kernel: softmax over columns (sequence dim) for each (b, h, i): Soft[b,h,i,:]
# Inputs: S: [B, H, S, S], Outputs: Soft: [B, H, S, S]
@triton.jit
def softmax_cols_kernel(
    S_ptr, Soft_ptr,
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr, S2: tl.constexpr,
    stride_sb, stride_sh, stride_si, stride_sj,
    stride_sb_soft, stride_sh_soft, stride_si_soft, stride_sj_soft,
    BLOCK_J: tl.constexpr,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    # Compute max over j for numerical stability
    max_val = -float('inf')
    for j0 in range(0, S2, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)
        mask_j = j_idx < S2
        scores = tl.load(S_ptr + b * stride_sb + h * stride_sh + i * stride_si + j_idx * stride_sj, mask=mask_j, other=-float('inf'))
        # Reduce max over this tile
        tile_max = tl.max(scores, axis=0)
        max_val = tl.maximum(max_val, tile_max)
    # Compute exp and sum
    sum_val = tl.zeros((), dtype=tl.float32)
    for j0 in range(0, S2, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)
        mask_j = j_idx < S2
        scores = tl.load(S_ptr + b * stride_sb + h * stride_sh + i * stride_si + j_idx * stride_sj, mask=mask_j, other=-float('inf'))
        exp_scores = tl.exp(scores - max_val)
        sum_val += tl.sum(exp_scores, axis=0)
    # Write normalized softmax
    for j0 in range(0, S2, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)
        mask_j = j_idx < S2
        scores = tl.load(S_ptr + b * stride_sb + h * stride_sh + i * stride_si + j_idx * stride_sj, mask=mask_j, other=-float('inf'))
        exp_scores = tl.exp(scores - max_val)
        soft = exp_scores / sum_val
        tl.store(Soft_ptr + b * stride_sb_soft + h * stride_sh_soft + i * stride_si_soft + j_idx * stride_sj_soft, soft, mask=mask_j)


# Triton kernel: output projection Y = attn_output_flat @ W^T + b
# X: [M, N_in] where M = B*S, N_in = num_attention_heads*head_dim
# W: [N_out, N_in], B: [N_out] (None possible -> we'll pass a zero if needed)
# Y: [M, N_out]
@triton.jit
def final_linear_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    M: tl.constexpr, N_in: tl.constexpr, N_out: tl.constexpr,
    stride_xm, stride_xn,
    stride_w0, stride_w1,  # W has shape [N_out, N_in]
    stride_ym, stride_yn,
    BLOCK_N: tl.constexpr,
):
    m = tl.program_id(axis=0)
    n_out = tl.program_id(axis=1)
    acc = tl.zeros((), dtype=tl.float32)
    for n0 in range(0, N_in, BLOCK_N):
        offs = n0 + tl.arange(0, BLOCK_N)
        mask = offs < N_in
        x = tl.load(X_ptr + m * stride_xm + offs * stride_xn, mask=mask, other=0.0)  # [BLOCK_N]
        w = tl.load(W_ptr + n_out * stride_w0 + offs * stride_w1, mask=mask, other=0.0)  # [BLOCK_N]
        acc += tl.sum(x * w, axis=0)
    b = tl.load(B_ptr + n_out) if B_ptr != 0 else 0.0
    acc += b
    tl.store(Y_ptr + m * stride_ym + n_out * stride_yn, acc)
