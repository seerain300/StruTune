Here is the complete Triton-based implementation of ModelNew that avoids any torch operations in the host code. All mathematical computations are performed inside Triton kernels. The class defines and launches the necessary Triton kernels and performs only allocations and reshapes in the host code.

```python
import torch
import triton
import triton.language as tl


# Kernel 1: Linear with bias: X @ W.T + b
# X: [B, S, H_in], W: [H_out, H_in], b: [H_out]
# Out: [B, S, H_out]
@triton.jit
def linear_bias_kernel(
    X_ptr, W_ptr, BIAS_ptr, OUT_ptr,
    Bsz, Ssz, H_in, H_out,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_o, stride_w_i,
    stride_out_b, stride_out_s, stride_out_h,
    BLOCK_K: tl.constexpr,
    BLOCK_OUT: tl.constexpr,
):
    # Grid: (Bsz, Ssz, H_out)
    b = tl.program_id(0)
    s = tl.program_id(1)
    o = tl.program_id(2)

    # Accumulator for the output vector
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over H_in in chunks of BLOCK_K
    for k in range(0, H_in, BLOCK_K):
        k_idx = k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < H_in

        # Load X[b, s, k_idx]
        x_off = b * stride_x_b + s * stride_x_s + k_idx * stride_x_h
        x_vec = tl.load(X_ptr + x_off, mask=mask_k, other=0.0)  # shape [BLOCK_K]

        # Load W[o, k_idx]
        w_off = o * stride_w_o + k_idx * stride_w_i
        w_vec = tl.load(W_ptr + w_off, mask=mask_k, other=0.0)  # shape [BLOCK_K]

        # Dot product accumulate
        acc += tl.sum(x_vec * w_vec, axis=0)

    # Add bias
    bias_val = tl.load(BIAS_ptr + o)
    acc = acc + bias_val

    # Store output
    out_off = b * stride_out_b + s * stride_out_s + o * stride_out_h
    tl.store(OUT_ptr + out_off, acc)


# Kernel 2: RMSNorm over last dim (size = head_dim, here 128): y = x * rsqrt(mean(x^2) + eps), then scale by weight
# Input x: [B, S, D], weight: [D] -> Output y: [B, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_w_d, stride_out_b, stride_out_s, stride_out_d,
    EPS: tl.constexpr,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)  # d ranges over 0..D-1

    row_off = b * stride_x_b + s * stride_x_s
    sum_sq = 0.0
    for i in range(0, BLOCK):
        off = row_off + (d + i) * stride_x_d
        x = tl.load(X_ptr + off)
        sum_sq += x * x
    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + EPS)
    weight = tl.load(Weight_ptr + d * stride_w_d)
    y = x * inv_rms
    y = y * weight

    out_off = b * stride_out_b + s * stride_out_s + d * stride_out_d
    tl.store(Out_ptr + out_off, y)


# Kernel 3: Rotate half of the last 64 dims for a 128-length vector: q1, q2 = q[:64], q[64:], q_rot = q1*cos - q2*sin
# We apply rotation to the last dimension via sin/cos tensors. Inputs: Q [B, S, D], Sin [D], Cos [D]. Output: Qrot [B, S, D].
@triton.jit
def rotate_half_kernel(
    Q_ptr, Sin_ptr, Cos_ptr, OUT_ptr,
    Bsz, Ssz, D,
    stride_q_b, stride_q_s, stride_q_d,
    stride_s_d, stride_c_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)  # d ranges over 0..D-1

    q_off = b * stride_q_b + s * stride_q_s + d * stride_q_d
    sin_d = tl.load(Sin_ptr + d * stride_s_d)
    cos_d = tl.load(Cos_ptr + d * stride_c_d)

    # Load q vector
    q = tl.load(Q_ptr + q_off)

    # Split into q1 and q2
    q1 = q[:64]
    q2 = q[64:]

    # Rotate: q1' = q1*cos - q2*sin; q2' = q1*sin + q2*cos
    q1r = q1 * cos_d - q2 * sin_d
    q2r = q1 * sin_d + q2 * cos_d

    # Concatenate
    q_rot = tl.concatenate([q1r, q2r])

    out_off = b * stride_out_b + s * stride_out_s + d * stride_out_d
    tl.store(OUT_ptr + out_off, q_rot)


# Kernel 4: Compute attention scores Q @ K^T for each (b, s, h): output [B, num_attention_heads, seq_length, seq_length]
# Q: [B, num_attention_heads, S, D], K: [B, num_attention_heads, S, D]
# Output: [B, num_attention_heads, S, S]
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, OUT_ptr,
    Bsz, Hsz, Ssz, D,
    stride_q_b, stride_q_h, stride_q_s, stride_q_d,
    stride_k_b, stride_k_h, stride_k_s, stride_k_d,
    stride_out_b, stride_out_h, stride_out_s, stride_out_t,
    BLOCK: tl.constexpr,
):
    # Grid: (Bsz, Hsz, Ssz, Ssz). Each program computes one scalar output OUT[b, h, s, t] = dot(Q[b,h,s,:], K[b,h,t,:])
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)
    t = tl.program_id(3)

    acc = 0.0
    for d in range(0, D, BLOCK):
        idx = d + tl.arange(0, BLOCK)
        mask = idx < D

        q_off = b * stride_q_b + h * stride_q_h + s * stride_q_s + idx * stride_q_d
        q_vec = tl.load(Q_ptr + q_off, mask=mask, other=0.0)

        k_off = b * stride_k_b + h * stride_k_h + t * stride_k_s + idx * stride_k_d
        k_vec = tl.load(K_ptr + k_off, mask=mask, other=0.0)

        acc += tl.sum(q_vec * k_vec, axis=0)

    out_off = b * stride_out_b + h * stride_out_h + s * stride_out_s + t * stride_out_t
    tl.store(OUT_ptr + out_off, acc)


# Kernel 5: Softmax over last dimension (sequence length) with causal mask (upper-triangular, diagonal=1)
# Input: scores [B, H, S, S], Output: softmax [B, H, S, S]
@triton.jit
def softmax_mask_kernel(
    SCORES_ptr, OUT_ptr,
    Bsz, Hsz, Ssz,
    stride_s_b, stride_s_h, stride_s_s, stride_s_t,
    stride_out_b, stride_out_h, stride_out_s, stride_out_t,
    BLOCK: tl.constexpr,
):
    # Grid: (Bsz, Hsz, Ssz)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    # Compute row max with mask: if t > s, set to -inf
    row_max = -float('inf')
    for t in range(0, Ssz, BLOCK):
        idx = t + tl.arange(0, BLOCK)
        mask = idx < Ssz

        s_off = b * stride_s_b + h * stride_s_h + s * stride_s_s + idx * stride_s_t
        vals = tl.load(SCORES_ptr + s_off, mask=mask, other=-float('inf'))
        mask_inf = idx > s
        vals = tl.where(mask_inf, -float('inf'), vals)
        # Reduce max
        block_max = tl.max(vals, axis=0)
        row_max = tl.maximum(row_max, block_max)

    # Compute exp and sum with mask
    exp_sum = 0.0
    for t in range(0, Ssz, BLOCK):
        idx = t + tl.arange(0, BLOCK)
        mask = idx < Ssz

        s_off = b * stride_s_b + h * stride_s_h + s * stride_s_s + idx * stride_s_t
        vals = tl.load(SCORES_ptr + s_off, mask=mask, other=-float('inf'))
        mask_inf = idx > s
        vals = tl.where(mask_inf, -float('inf'), vals)
        vals = vals - row_max
        exp_vals = tl.exp(vals)
        exp_sum += tl.sum(exp_vals, axis=0)

    # Write normalized outputs
    for t in range(0, Ssz, BLOCK):
        idx = t + tl.arange(0, BLOCK)
        mask = idx < Ssz

        s_off = b * stride_s_b + h * stride_s_h + s * stride_s_s + idx * stride_s_t
        vals = tl.load(SCORES_ptr + s_off, mask=mask, other=-float('inf'))
        mask_inf = idx > s
        vals = tl.where(mask_inf, -float('inf'), vals)
        vals = vals - row_max
        exp_vals = tl.exp(vals)
        out_vals = exp_vals / exp_sum

        out_off = b * stride_out_b + h * stride_out_h + s * stride_out_s + idx * stride_out_t
        tl.store(OUT_ptr + out_off, out_vals, mask=mask)


# Kernel 6: Compute attention output Softmax(QK_scaled) @ V for each (b, s, h): output [B, num_attention_heads, S, D]
# A: [B, H, S, S] (softmaxed scores), V: [B, H, S, D]
# Output: [B, H, S, D]
@triton.jit
def matmul_attn_kernel(
    A_ptr, V_ptr, OUT_ptr,
    Bsz, Hsz, Ssz, D,
    stride_a_b, stride_a_h, stride_a_s, stride_a_t,
    stride_v_b, stride_v_h, stride_v_s, stride_v_d,
    stride_out_b, stride_out_h, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr,
):
    # Grid: (Bsz, Hsz, Ssz, D)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)
    d = tl.program_id(3)

    out_vec = tl.zeros((), dtype=tl.float32)

    for t in range(0, Ssz, BLOCK):
        idx = t + tl.arange(0, BLOCK)
        mask_t = idx < Ssz

        # Load A[b, h, s, idx]
        a_off = b * stride_a_b + h * stride_a_h + s * stride_a_s + idx * stride_a_t
        a_vec = tl.load(A_ptr + a_off, mask=mask_t, other=0.0)

        # Load V[b, h, idx, d]
        v_off = b * stride_v_b + h * stride_v_h + idx * stride_v_s + d * stride_v_d
        v_vec = tl.load(V_ptr + v_off, mask=mask_t, other=0.0)

        # Accumulate: out[b, h, s, d] += sum_t A[b,h,s,t] * V[b,h,t,d]
        out_vec += tl.sum(a_vec * v_vec, axis=0)

    out_off = b * stride_out_b + h * stride_out_h + s * stride_out_s + d * stride_out_d
    tl.store(OUT_ptr + out_off, out_vec)


# Kernel 7: Linear without bias: X @ W.T (used for final output projection)
# X: [B, S, H_in], W: [H_out, H_in]
# Out: [B, S, H_out]
@triton.jit
def linear_nobias_kernel(
    X_ptr, W_ptr, OUT_ptr,
    Bsz, Ssz, H_in, H_out,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_o, stride_w_i,
    stride_out_b, stride_out_s, stride_out_h,
    BLOCK_K: tl.constexpr,
    BLOCK_OUT: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    o = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k in range(0, H_in, BLOCK_K):
        k_idx = k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < H_in

        x_off = b * stride_x_b + s * stride_x_s + k_idx * stride_x_h
        x_vec = tl.load(X_ptr + x_off, mask=mask_k, other=0.0)

        w_off = o * stride_w_o + k_idx * stride_w_i
        w_vec = tl.load(W_ptr + w_off, mask=mask_k, other=0.0)

        acc += tl.sum(x_vec * w_vec, axis=0)

    out_off = b * stride_out_b + s * stride_out_s + o * stride_out_h
    tl.store(OUT_ptr + out_off, acc)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # Prepare constants for head dim and scalars
        self.head_dim = 128
        self.num_attention_heads = 96
        self.num_key_value_heads = 8
        self.num_key_value_groups = 12
        self.scaling = 1.0 / (self.head_dim ** 0.5)  # matches original code

    def forward(
        self,
        hidden_states: torch.Tensor,
        q_proj_weight: torch.Tensor,
        q_proj_bias: torch.Tensor,
        k_proj_weight: torch.Tensor,
        k_proj_bias: torch.Tensor,
        v_proj_weight: torch.Tensor,
        v_proj_bias: torch.Tensor,
        o_proj_weight: torch.Tensor,
        q_norm_weight: torch.Tensor,
        k_norm_weight: torch.Tensor,
        cos: torch.Tensor,
        sin: torch.Tensor,
        rms_norm_eps: float,
    ):
        # hidden_states: [B, S, 11008]
        Bsz, Ssz, H_in = hidden_states.shape

        # 1) Linear projections: Q, K, V
        # Allocate outputs
        query_states = torch.empty((Bsz, Ssz, q_proj_weight.shape[0]), dtype=hidden_states.dtype, device=hidden_states.device)
        key_states = torch.empty((Bsz, Ssz, k_proj_weight.shape[0]), dtype=hidden_states.dtype, device=hidden_states.device)
        value_states = torch.empty((Bsz, Ssz, v_proj_weight.shape[0]), dtype=hidden_states.dtype, device=hidden_states.device)

        # Strides
        stride_x_q_b, stride_x_q_s, stride_x_q_h = hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2)
        stride_w_q_o, stride_w_q_i = q_proj_weight.stride(0), q_proj_weight.stride(1)
        stride_out_q_b, stride_out_q_s, stride_out_q_o = query_states.stride(0), query_states.stride(1), query_states.stride(2)

        stride_x_k_b, stride_x_k_s, stride_x_k_h = hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2)
        stride_w_k_o, stride_w_k_i = k_proj_weight.stride(0), k_proj_weight.stride(1)
        stride_out_k_b, stride_out_k_s, stride_out_k_o = key_states.stride(0), key_states.stride(1), key_states.stride(2)

        stride_x_v_b, stride_x_v_s, stride_x_v_h = hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2)
        stride_w_v_o, stride_w_v_i = v_proj_weight.stride(0), v_proj_weight.stride(1)
        stride_out_v_b, stride_out_v_s, stride_out_v_o = value_states.stride(0), value_states.stride(1), value_states.stride(2)

        # Launch linear_bias_kernel for Q, K, V
        # Grid: (Bsz, Ssz, H_out)
        grid_q = (Bsz, Ssz, q_proj_weight.shape[0])
        grid_k = (Bsz, Ssz, k_proj_weight.shape[0])
        grid_v = (Bsz, Ssz, v_proj_weight.shape[0])

        # Choose BLOCK_K as 64 (divisible by 128 chunks), BLOCK_OUT as 1 (scalar accumulation). Triton will broadcast properly.
        # Note: Triton requires static BLOCK size; we use 64 for reduction chunk.
        linear_bias_kernel[grid_q](
            hidden_states, q_proj_weight, q_proj_bias, query_states,
            Bsz, Ssz, H_in, q_proj_weight.shape[0],
            stride_x_q_b, stride_x_q_s, stride_x_q_h,
            stride_w_q_o, stride_w_q_i,
            stride_out_q_b, stride_out_q_s, stride_out_q_o,
            64, 1,
            num_warps=1
        )

        linear_bias_kernel[grid_k](
            hidden_states, k_proj_weight, k_proj_bias, key_states,
            Bsz, Ssz, H_in, k_proj_weight.shape[0],
            stride_x_k_b, stride_x_k_s, stride_x_k_h,
            stride_w_k_o, stride_w_k_i,
            stride_out_k_b, stride_out_k_s, stride_out_k_o,
            64, 1,
            num_warps=1
        )

        linear_bias_kernel[grid_v](
            hidden_states, v_proj_weight, v_proj_bias, value_states,
            Bsz, Ssz, H_in, v_proj_weight.shape[0],
            stride_x_v_b, stride_x_v_s, stride_x_v_h,
            stride_w_v_o, stride_w_v_i,
            stride_out_v_b, stride_out_v_s, stride_out_v_o,
            64, 1,
            num_warps=1
        )

        # 2) RMSNorm for Q and K
        # Allocate normalized tensors
        query_norm = torch.empty_like(query_states)
        key_norm = torch.empty_like(key_states)

        # Launch rmsnorm_kernel
        # For Q
        grid_qn = (Bsz, Ssz, query_states.shape[2])
        rmsnorm_kernel[grid_qn](
            query_states, q_norm_weight, query_norm,
            Bsz, Ssz, query_states.shape[2],
            query_states.stride(0), query_states.stride(1), query_states.stride(2),
            q_norm_weight.stride(0), query_norm.stride(0), query_norm.stride(1), query_norm.stride(2),
            rms_norm_eps,
            128
        )

        # For K
        grid_kn = (Bsz, Ssz, key_states.shape[2])
        rmsnorm_kernel[grid_kn](
            key_states, k_norm_weight, key_norm,
            Bsz, Ssz, key_states.shape[2],
            key_states.stride(0), key_states.stride(1), key_states.stride(2),
            k_norm_weight.stride(0), key_norm.stride(0), key_norm.stride(1), key_norm.stride(2),
            rms_norm_eps,
            128
        )

        # 3) Apply Rotation (RoPE) to Q and K
        query_rot = torch.empty_like(query_norm)
        key_rot = torch.empty_like(key_norm)

        grid_r = (Bsz, Ssz, query_norm.shape[2])
        rotate_half_kernel[grid_r](
            query_norm, sin, cos, query_rot,
            Bsz, Ssz, query_norm.shape[2],
            query_norm.stride(0), query_norm.stride(1), query_norm.stride(2),
            sin.stride(0), cos.stride(0), query_rot.stride(0), query_rot.stride(1), query_rot.stride(2),
            128
        )

        rotate_half_kernel[grid_r](
            key_norm, sin, cos, key_rot,
            Bsz, Ssz, key_norm.shape[2],
            key_norm.stride(0), key_norm.stride(1), key_norm.stride(2),
            sin.stride(0), cos.stride(0), key_rot.stride(0), key_rot.stride(1), key_rot.stride(2),
            128
        )

        # 4) Reshape to heads
        # Query: [B, num_attention_heads, S, head_dim]
        query_heads = query_rot.view(Bsz, self.num_attention_heads, Ssz, self.head_dim)
        key_heads = key_rot.view(Bsz, self.num_key_value_heads, Ssz, self.head_dim)
        value_heads = value_states.view(Bsz, self.num_key_value_heads, Ssz, self.head_dim)

        # 5) Repeat K/V to match attention heads for GQA
        # We expand (num_key_value_heads, num_key_value_groups) to num_attention_heads
        # key_heads: [B, 8, S, 128] -> [B, 96, S, 128]
        # Using groups=12: each attention head maps to a unique (kv_head, group) pair
        key_heads_exp = key_heads[:, :, None, :, :].expand(Bsz, self.num_key_value_heads, self.num_key_value_groups, Ssz, self.head_dim).reshape(Bsz, self.num_attention_heads, Ssz, self.head_dim)
        value_heads_exp = value_heads[:, :, None, :, :].expand(Bsz, self.num_key_value_heads, self.num_key_value_groups, Ssz, self.head_dim).reshape(Bsz, self.num_attention_heads, Ssz, self.head_dim)

        # 6) Compute attention scores, softmax with causal mask, and attention output per head
        # We'll loop over attention heads in host code (since Triton cannot have dynamic loop bounds in kernel signature).
        # Allocate intermediates
        attn_scores = torch.empty((Bsz, self.num_attention_heads, Ssz, Ssz), dtype=query_heads.dtype, device=query_heads.device)
        attn_output = torch.empty((Bsz, self.num_attention_heads, Ssz, self.head_dim), dtype=query_heads.dtype, device=query_heads.device)

        # Launch matmul_qk_kernel: for each (b, h, s, t) compute Q[b,h,s,:] dot K[b,h,t,:]
        # Grid: (Bsz, num_attention_heads, Ssz, Ssz)
        grid_qk = (Bsz, self.num_attention_heads, Ssz, Ssz)
        matmul_qk_kernel[grid_qk](
            query_heads, key_heads_exp, attn_scores,
            Bsz, self.num_attention_heads, Ssz, self.head_dim,
            query_heads.stride(0), query_heads.stride(1), query_heads.stride(2), query_heads.stride(3),
            key_heads_exp.stride(0), key_heads_exp.stride(1), key_heads_exp.stride(2), key_heads_exp.stride(3),
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            128
        )

        # 7) Softmax with causal mask (upper-triangular, diagonal=1)
        # grid: (Bsz, num_attention_heads, Ssz)
        grid_sm = (Bsz, self.num_attention_heads, Ssz)
        softmax_mask_kernel[grid_sm](
            attn_scores, attn_scores,  # in-place softmax result
            Bsz, self.num_attention_heads, Ssz,
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            128
        )

        # 8) Compute attention output per head: Softmax(QK_scaled) @ V
        grid_attn = (Bsz, self.num_attention_heads, Ssz, self.head_dim)
        matmul_attn_kernel[grid_attn](
            attn_scores, value_heads_exp, attn_output,
            Bsz, self.num_attention_heads, Ssz, self.head_dim,
            attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
            value_heads_exp.stride(0), value_heads_exp.stride(1), value_heads_exp.stride(2), value_heads_exp.stride(3),
            attn_output.stride(0), attn_output.stride(1), attn_output.stride(2), attn_output.stride(3),
            128
        )

        # 9) Transpose and reshape to [B, S, num_attention_heads*head_dim]
        attn_out = attn_output.transpose(1, 2).contiguous()  # [B, S, 96*128]
        attn_out = attn_out.reshape(Bsz, Ssz, self.num_attention_heads * self.head_dim)

        # 10) Final output projection (no bias): attn_out @ o_proj_weight.T
        output = torch.empty((Bsz, Ssz, o_proj_weight.shape[0]), dtype=attn_out.dtype, device=attn_out.device)

        stride_x_ob_b, stride_x_ob_s, stride_x_ob_h = attn_out.stride(0), attn_out.stride(1), attn_out.stride(2)
        stride_w_ob_o, stride_w_ob_i = o_proj_weight.stride(0), o_proj_weight.stride(1)
        stride_out_ob_b, stride_out_ob_s, stride_out_ob_o = output.stride(0), output.stride(1), output.stride(2)
```
