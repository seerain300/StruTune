    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # over s dimension tiles
    pid_n = tl.program_id(1)  # over output n dimension tiles

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # s indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # output n indices
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)  # k indices

        # Load A tile: In[s, k] for each s in offs_m and k in offs_k
        a_ptrs = In_ptr + offs_m[:, None] * stride_is + offs_k[None, :] * stride_in
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < S) & (offs_k[None, :] < K), other=0.0).to(tl.float32)

        # Load W tile: W[k]
        w = tl.load(W_ptr + offs_k, mask=(offs_k < K), other=0.0).to(tl.float32)  # [BK]
        # Multiply and reduce over k
        prod = a * w[None, :]  # broadcast over columns
        acc += tl.sum(prod, axis=1)  # reduce over BK -> [BLOCK_M, BLOCK_N]

    # Store acc to Out[s, n]
    out_ptrs = Out_ptr + offs_m[:, None] * stride_os + offs_n[None, :] * stride_on
    tl.store(out_ptrs, acc, mask=(offs_m[:, None] < S) & (offs_n[None, :] < N_out))

# Helper to launch output projection for a given batch b:
def triton_output_projection(In_b: torch.Tensor, W: torch.Tensor, Out_b: torch.Tensor):
    # In_b: [S, N_out], Out_b: [S, N_out]
    S, N_out = In_b.shape
    K = N_out
    grid = (triton.cdiv(S, 1), triton.cdiv(N_out, 128))
    output_projection_kernel_final[grid](
        In_b, W, Out_b,
        S, N_out, K,
        In_b.stride(0), In_b.stride(1),
        Out_b.stride(0), Out_b.stride(1),
        W.stride(0),
        BLOCK_M=1, BLOCK_N=128, BLOCK_K=128,
    )
