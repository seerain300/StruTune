import triton

# 1) Batched GEMM (no bias): C[M, N] = A[M, K] @ B[K, N], where B is weight^T (K, N).
@triton.jit
def triton_linear_no_bias(
    A_ptr,         # *fp32, [M, K] = hidden_states reshaped
    B_ptr,         # *fp32, [K, N] = weight.T
    C_ptr,         # *fp32, [M, N] output
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_am, stride_ak,   # A strides (row, col)
    stride_bk, stride_bn,   # B strides (row=K, col=N)
    stride_cm, stride_cn,   # C strides (row, col)
    BLOCK_K: tl.constexpr,
):
    # one program instance handles one row of C
    row = tl.program_id(0)  # 0..M-1
    if row >= M:
        return
    acc = tl.zeros((N,), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        a_row_vec = tl.load(
            A_ptr + row * stride_am + offs_k * stride_ak,
            mask=offs_k < K,
            other=0.0
        )  # [BLOCK_K]
        b_block = tl.load(
            B_ptr + offs_k[:, None] * stride_bk + tl.arange(0, N)[None, :] * stride_bn,
            mask=(offs_k[:, None] < K) & (tl.arange(0, N)[None, :] < N),
            other=0.0
        )  # [BLOCK_K, N]
        acc += tl.sum(b_block * a_row_vec[:, None], axis=0)
    tl.store(C_ptr + row * stride_cm + tl.arange(0, N) * stride_cn, acc, mask=tl.arange(0, N) < N)


# 2) RMSNorm per head over last dimension (head_dim=128). In this function,
#    we normalize along head_dim and scale by q/k_norm_weight. We assume input is [B, S, H].
#    We will call this kernel per (batch, head) on the reshaped [S, H] slice.
@triton.jit
def triton_rmsnorm_row(
    X_ptr,          # *fp32, [S, H] (row slice of [B, S, H])
    Weight_ptr,     # *fp32, [H] (q/k_norm_weight)
    Y_ptr,          # *fp32, [S, H] output
    S: tl.constexpr, H: tl.constexpr,
    stride_xs, stride_xh,   # X strides
    stride_ws, stride_wh,   # Weight strides (usually 1D, here use 0 and 1)
    stride_ys, stride_yh,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    s = tl.program_id(0)  # 0..S-1
    if s >= S:
        return
    # compute mean of squares across H
    sumsq = 0.0
    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        x = tl.load(X_ptr + s * stride_xs + offs_h * stride_xh, mask=offs_h < H, other=0.0)  # [BLOCK_H]
        sumsq += tl.sum(x * x)
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + eps)
    # normalize and scale
    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        x = tl.load(X_ptr + s * stride_xs + offs_h * stride_xh, mask=offs_h < H, other=0.0)
        w = tl.load(Weight_ptr + offs_h * stride_wh, mask=offs_h < H, other=0.0)  # scale factor
        y = x * inv_rms * w
        tl.store(Y_ptr + s * stride_ys + offs_h * stride_yh, y, mask=offs_h < H)


# 3) Rotate half: for tensors shaped [S, H] (H=128), rotate so that first half maps to second half and vice versa:
#    Given Q/K [S, H], produce rotated [S, H] with cat((-q2, q1), dim=-1). This matches the original PyTorch code.
@triton.jit
def triton_rotate_half(
    In_ptr,    # *fp32, [S, H]
    Cos_ptr,   # *fp32, [H]
    Sin_ptr,   # *fp32, [H]
    Out_ptr,   # *fp32, [S, H]
    S: tl.constexpr, H: tl.constexpr,
    stride_is, stride_ih,
    stride_cos, stride_sin,
    stride_os, stride_oh,
):
    s = tl.program_id(0)  # row index
    if s >= S:
        return
    for h0 in range(0, H, 16):
        offs_h = h0 + tl.arange(0, 16)
        q = tl.load(In_ptr + s * stride_is + offs_h * stride_ih, mask=offs_h < H, other=0.0)
        cosv = tl.load(Cos_ptr + offs_h * stride_cos, mask=offs_h < H, other=1.0)
        sinv = tl.load(Sin_ptr + offs_h * stride_sin, mask=offs_h < H, other=1.0)
        q1 = q[:8]
        q2 = q[8:]
        q_rot = -q2 * cosv[:8] * sinv[:8] + q1 * cosv[:8] * sinv[:8]  # exact rotation per dim; broadcasting cos/sin
        # Store rotated values into Out: for half, positions map as described; we use q_rot for the half-block
        tl.store(Out_ptr + s * stride_os + offs_h * stride_oh, q_rot, mask=offs_h < H)

#    We expand [B, 8, S, H] to [B, 96, S, H] by repeating each kv head g=8 times per group. This is done by writing
#    each slice into output at positions determined by group index.
@triton.jit
def triton_repeat_kv_heads(

    Out_ptr,         # *fp32, [B, 96, S, H]
    B: tl.constexpr, KV_HEADS: tl.constexpr, S: tl.constexpr, H: tl.constexpr,

    G: tl.constexpr,      # num_key_value_groups
):

        val = tl.load(In_ptr + in_offset, mask=tl.arange(0, H) < H, other=0.0)

#    We implement row-wise kernel: each program handles one (batch, head) row and iterates over N (sequence positions).
@triton.jit
def triton_score_row(
    Q_ptr,       # *fp32, [M, D] where M=B*96, D=128
    K_ptr,       # *fp32, [N, D] where N=S
    S_ptr,       # *fp32, [M, N] output
    M: tl.constexpr, N: tl.constexpr, D: tl.constexpr,
    stride_qm, stride_qd,   # Q strides
    stride_kn, stride_kd,   # K strides
    stride_sm, stride_sn,   # S strides
    BLOCK_D: tl.constexpr,
):
    row = tl.program_id(0)  # 0..M-1
    if row >= M:
        return
    acc = tl.zeros((N,), dtype=tl.float32)
    for d0 in range(0, D, BLOCK_D):
        offs_d = d0 + tl.arange(0, BLOCK_D)
        q_vec = tl.load(Q_ptr + row * stride_qm + offs_d * stride_qd, mask=offs_d < D, other=0.0)  # [BLOCK_D]
        k_block = tl.load(
            K_ptr + tl.arange(0, N) * stride_kn + offs_d[None, :] * stride_kd,
            mask=(tl.arange(0, N)[:, None] < N) & (offs_d[None, :] < D),
            other=0.0
        )  # [N, BLOCK_D]
        # dot per position in N
        acc += tl.sum(k_block * q_vec[None, :], axis=1)
    tl.store(S_ptr + row * stride_sm + tl.arange(0, N) * stride_sn, acc, mask=tl.arange(0, N) < N)


# 6) Row-wise softmax over N: apply softmax to S row. We use masked loads to ignore positions beyond N when padding.
@triton.jit
def triton_softmax_row(

    OutS_ptr,         # *fp32, [M, N] output
    M: tl.constexpr, N: tl.constexpr,
    stride_sm, stride_sn,
    stride_om, stride_on,
):
    row = tl.program_id(0)  # 0..M-1

        return

        val = tl.load(S_ptr + row * stride_sm + i * stride_sn)

    sum_exp = 0.0
    for i in range(0, N):
        val = tl.load(S_ptr + row * stride_sm + i * stride_sn)
        sum_exp += tl.exp(val - max_val)
    # write normalized
    for i in range(0, N):
        val = tl.load(S_ptr + row * stride_sm + i * stride_sn)
        out = tl.exp(val - max_val) / sum_exp
        tl.store(OutS_ptr + row * stride_om + i * stride_on, out)


# 7) Final attention output per row: OutRow[M, H] = Softmax[SRow] @ V[M, H] where M=B*num_attention_heads.
#    V has shape [B, S, H], we pass a single row [S, H] and reduce across N (sequence).
@triton.jit

    Softmax_ptr,   # *fp32, [M, N] row-wise softmax scores
    V_ptr,         # *fp32, [S, H] V slice
    OutRow_ptr,    # *fp32, [M, H] output
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_sm, stride_sn,   # Softmax strides
    stride_vs, stride_vh,   # V strides
    stride_om, stride_oh,   # OutRow strides
):
    m = tl.program_id(0)  # 0..M-1
    if m >= M:

    acc = tl.zeros((H,), dtype=tl.float32)

        acc += sm * v
    tl.store(OutRow_ptr + m * stride_om + tl.arange(0, H) * stride_oh, acc, mask=tl.arange(0, H) < H)
