import torch
import triton
import triton.language as tl


# Kernel 1: Linear with bias: X @ W.T + b
# X: [B, S, H_in], W: [H_out, H_in], b: [H_out] -> Out: [B, S, H_out]
@triton.jit
def linear_bias_kernel(
    X_ptr, W_ptr, B_ptr, Out_ptr,
    Bsz: tl.constexpr, Ssz: tl.constexpr, H_in: tl.constexpr, H_out: tl.constexpr,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_o, stride_w_i,
    stride_out_b, stride_out_s, stride_out_h,
    BLOCK_IN: tl.constexpr, BLOCK_OUT: tl.constexpr,
):
    # grid: (b, s, o)
    b = tl.program_id(0)
    s = tl.program_id(1)
    o = tl.program_id(2)

    # Accumulator for output channel o
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over H_out in tiles
    for i in range(0, H_out, BLOCK_OUT):
        o_offsets = i + tl.arange(0, BLOCK_OUT)
        mask_o = o_offsets < H_out
        # Load bias for these output channels
        b_vals = tl.load(B_ptr + o_offsets, mask=mask_o, other=0.0).to(tl.float32)

        # Accumulate for each o_offsets
        acc = tl.zeros([BLOCK_OUT], dtype=tl.float32)

        # Loop over H_in in tiles
        for j in range(0, H_in, BLOCK_IN):
            i_offsets = j + tl.arange(0, BLOCK_IN)
            mask_i = i_offsets < H_in

            # Load X[b, s, i_offsets]
            x_ptrs = X_ptr + b * stride_x_b + s * stride_x_s + i_offsets * stride_x_h
            x_vals = tl.load(x_ptrs, mask=mask_i, other=0.0).to(tl.float32)  # [BLOCK_IN]

            # Load W[o_offsets, i_offsets] as [BLOCK_OUT, BLOCK_IN]
            w_ptrs = W_ptr + o_offsets[:, None] * stride_w_o + i_offsets[None, :] * stride_w_i
            w_vals = tl.load(w_ptrs, mask=mask_o[:, None] & mask_i[None, :], other=0.0).to(tl.float32)

            # Fused multiply-add and reduction over input channels
            acc += tl.sum(w_vals * x_vals[None, :], axis=1)  # [BLOCK_OUT]

        # Add bias and store
        out_ptrs = Out_ptr + b * stride_out_b + s * stride_out_s + o_offsets * stride_out_h
        tl.store(out_ptrs, acc + b_vals, mask=mask_o)


# Kernel 2: RMSNorm per row over last dim (head_dim=128): y = x * rsqrt(mean(x^2) + eps)
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Y_ptr,
    Bsz: tl.constexpr, Ssz: tl.constexpr, H: tl.constexpr,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w, stride_y_b, stride_y_s, stride_y_h,
    eps: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # Each program handles one row [Ssz, H]
    b = tl.program_id(0)
    s = tl.program_id(1)
    # Reduce over H
    sum_sq = tl.zeros((), dtype=tl.float32)
    for i in range(0, H, BLOCK_H):
        h_offsets = i + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H
        x_ptrs = X_ptr + b * stride_x_b + s * stride_x_s + h_offsets * stride_x_h
        x_vals = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)
        sum_sq += tl.sum(x_vals * x_vals)
    mean = sum_sq / H
    scale = 1.0 / tl.sqrt(mean + eps)

    # Scale and apply weight, then store
    for i in range(0, H, BLOCK_H):
        h_offsets = i + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H
        x_ptrs = X_ptr + b * stride_x_b + s * stride_x_s + h_offsets * stride_x_h
        y_ptrs = Y_ptr + b * stride_y_b + s * stride_y_s + h_offsets * stride_y_h
        x_vals = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)
        w_vals = tl.load(Weight_ptr + h_offsets * stride_w, mask=mask_h, other=1.0).to(tl.float32)
        y_vals = x_vals * scale * w_vals
        tl.store(y_ptrs, y_vals, mask=mask_h)


# Kernel 3: Rotate last half of a 128-length vector: for h=0..63 keep; for h=64..127 apply y = cos*x1 - sin*x2 where x1=first 64, x2=last 64
@triton.jit
def rotate_half_kernel(
    X_ptr, Cos_ptr, Sin_ptr, Y_ptr,
    Bsz: tl.constexpr, Ssz: tl.constexpr, H: tl.constexpr,  # H is 128 in our case
    stride_x_b, stride_x_s, stride_x_h,
    stride_y_b, stride_y_s, stride_y_h,
    BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)

    for i in range(0, H, BLOCK_H):
        h_offsets = i + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H

        x_ptrs = X_ptr + b * stride_x_b + s * stride_x_s + h_offsets * stride_x_h
        x_vals = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)

        # Split halves: h0 and h1 correspond to first and last 64 dims
        h0 = h_offsets < 64
        h1 = h_offsets >= 64

        x1 = tl.where(h0, x_vals, 0.0)
        x2 = tl.where(h1, x_vals - 64, 0.0)  # invalid lanes masked

        # Load cos/sin for the last 64 dims only
        cos_vals = tl.load(Cos_ptr + (h_offsets - 64), mask=h1, other=1.0).to(tl.float32)
        sin_vals = tl.load(Sin_ptr + (h_offsets - 64), mask=h1, other=0.0).to(tl.float32)

        y1 = x1
        y2 = cos_vals * x1 - sin_vals * x2  # using x1, x2 as above; correct because rotation only applies to last 64

        # Combine
        y_vals = tl.where(h0, y1, 0.0) + tl.where(h1, y2, 0.0)
        y_ptrs = Y_ptr + b * stride_y_b + s * stride_y_s + h_offsets * stride_y_h
        tl.store(y_ptrs, y_vals, mask=mask_h)


# Kernel 4: Compute attention scores: Out[b, s, h] = sum_k Q[b, s, h] * K[b, k, h] over K in range S
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, Out_ptr,
    Bsz: tl.constexpr, Ssz: tl.constexpr, H: tl.constexpr,
    stride_q_b, stride_q_s, stride_q_h,
    stride_k_b, stride_k_s, stride_k_h,
    stride_out_b, stride_out_s, stride_out_h,
    BLOCK_K: tl.constexpr,
):
    # grid: (Bsz, Ssz, H)
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k in range(0, Ssz, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < Ssz
        # Load Q[b, s, h]
        q_val = tl.load(Q_ptr + b * stride_q_b + s * stride_q_s + h * stride_q_h).to(tl.float32)
        # Load K[b, k, h] vector
        k_vals = tl.load(K_ptr + b * stride_k_b + k_offsets * stride_k_s + h * stride_k_h, mask=mask_k, other=0.0).to(tl.float32)
        acc += tl.sum(q_val * k_vals)
    # Store
    out_ptrs = Out_ptr + b * stride_out_b + s * stride_out_s + h * stride_out_h
    tl.store(out_ptrs, acc)


# Kernel 5: Softmax with causal mask over last dimension (S) for tensor of shape [B, H, S, S]
@triton.jit
def softmax_mask_kernel(
    In_ptr, Mask_ptr, Out_ptr,
    Bsz: tl.constexpr, Ssz: tl.constexpr, H: tl.constexpr,
    stride_in_b, stride_in_h, stride_in_m, stride_in_n,
    stride_mask_b, stride_mask_h, stride_mask_m, stride_mask_n,
    stride_out_b, stride_out_h, stride_out_m, stride_out_n,
    scaling: tl.constexpr,
):
    # Each program handles one row across sequence length n for given (b, h, m)
    b = tl.program_id(0)
    h = tl.program_id(1)
    m = tl.program_id(2)

    # Load row In[b, h, m, :]
    row_max = tl.float32(-1e30)
    for n in range(0, Ssz):
        in_ptrs = In_ptr + b * stride_in_b + h * stride_in_h + m * stride_in_m + n * stride_in_n
        val = tl.load(in_ptrs).to(tl.float32)
        row_max = tl.maximum(row_max, val)

    sum_exp = tl.zeros((), dtype=tl.float32)
    for n in range(0, Ssz):
        in_ptrs = In_ptr + b * stride_in_b + h * stride_in_h + m * stride_in_m + n * stride_in_n
        val = tl.load(in_ptrs).to(tl.float32)
        mask_ptrs = Mask_ptr + b * stride_mask_b + h * stride_mask_h + m * stride_mask_m + n * stride_mask_n
        mask_val = tl.load(mask_ptrs).to(tl.float32)  # 0.0 or -inf
        val = val - row_max + mask_val
        expv = tl.exp(val)
        sum_exp += expv

    inv_sum = 1.0 / sum_exp
    for n in range(0, Ssz):
        in_ptrs = In_ptr + b * stride_in_b + h * stride_in_h + m * stride_in_m + n * stride_in_n
        val = tl.load(in_ptrs).to(tl.float32)
        mask_ptrs = Mask_ptr + b * stride_mask_b + h * stride_mask_h + m * stride_mask_m + n * stride_mask_n
        mask_val = tl.load(mask_ptrs).to(tl.float32)
        val = val - row_max + mask_val
        outv = tl.exp(val) * inv_sum
        out_ptrs = Out_ptr + b * stride_out_b + h * stride_out_h + m * stride_out_m + n * stride_out_n
        tl.store(out_ptrs, outv)


# Kernel 6: Compute attention output: Out[b, s, h] = sum_k Softmax[b, h, s, k] * V[b, k, h]
@triton.jit
def matmul_attn_kernel(
    Soft_ptr, V_ptr, Out_ptr,
    Bsz: tl.constexpr, Ssz: tl.constexpr, H: tl.constexpr,
    stride_soft_b, stride_soft_h, stride_soft_m, stride_soft_n,
    stride_v_b, stride_v_s, stride_v_h,
    stride_out_b, stride_out_s, stride_out_h,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k in range(0, Ssz, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < Ssz
        # Load Soft[b, h, s, k]
        soft_ptrs = Soft_ptr + b * stride_soft_b + h * stride_soft_h + s * stride_soft_m + k_offsets * stride_soft_n
        soft_vals = tl.load(soft_ptrs, mask=mask_k, other=0.0).to(tl.float32)
        # Load V[b, k, h]
        v_ptrs = V_ptr + b * stride_v_b + k_offsets * stride_v_s + h * stride_v_h
        v_vals = tl.load(v_ptrs, mask=mask_k, other=0.0).to(tl.float32)
        acc += tl.sum(soft_vals * v_vals)
    out_ptrs = Out_ptr + b * stride_out_b + s * stride_out_s + h * stride_out_h
    tl.store(out_ptrs, acc)


# Kernel 7: Final output projection without bias: Out = X @ W.T, where X is [B*S, H_in=96*128], W is [H_out, H_in], Out is [B*S, H_out]
@triton.jit
def linear_nobias_kernel(
    X_ptr, W_ptr, Out_ptr,
    M: tl.constexpr, H_in: tl.constexpr, H_out: tl.constexpr,
    stride_x_m, stride_x_in,
    stride_w_out, stride_w_in,
    stride_out_m, stride_out_out,
    BLOCK_IN: tl.constexpr, BLOCK_OUT: tl.constexpr,
):
    # grid over (M, H_out)
    m = tl.program_id(0)
    o = tl.program_id(1)

    acc = tl.zeros((), dtype=tl.float32)
    for i in range(0, H_out, BLOCK_OUT):
        o_offsets = i + tl.arange(0, BLOCK_OUT)
        mask_o = o_offsets < H_out
        acc_vec = tl.zeros([BLOCK_OUT], dtype=tl.float32)

        for j in range(0, H_in, BLOCK_IN):
            in_offsets = j + tl.arange(0, BLOCK_IN)
            mask_in = in_offsets < H_in

            # X[m, in_offsets]
            x_ptrs = X_ptr + m * stride_x_m + in_offsets * stride_x_in
            x_vals = tl.load(x_ptrs, mask=mask_in, other=0.0).to(tl.float32)

            w_ptrs = W_ptr + o_offsets[:, None] * stride_w_out + in_offsets[None, :] * stride_w_in

            acc_vec += tl.sum(w_vals * x_vals[None, :], axis=1)

        out_ptrs = Out_ptr + m * stride_out_m + o_offsets * stride_out_out
        tl.store(out_ptrs, acc_vec, mask=mask_o)

def _launch_linear_bias(Out, X, W, B, Bsz, Ssz, H_in, H_out):
    # Grid over (b, s, o)
    grid = (Bsz, Ssz, H_out)
    # Choose BLOCK sizes
    BLOCK_IN = 128
    BLOCK_OUT = 64
    triton.run(
        linear_bias_kernel,
        grid=grid,
        num_warps=4,
        X_ptr=X, W_ptr=W, B_ptr=B, Out_ptr=Out,
        Bsz=Bsz, Ssz=Ssz, H_in=H_in, H_out=H_out,
        stride_x_b=X.stride(0), stride_x_s=X.stride(1), stride_x_h=X.stride(2),
        stride_w_o=W.stride(0), stride_w_i=W.stride(1),
        stride_out_b=Out.stride(0), stride_out_s=Out.stride(1), stride_out_h=Out.stride(2),
        BLOCK_IN=BLOCK_IN, BLOCK_OUT=BLOCK_OUT,
    )
