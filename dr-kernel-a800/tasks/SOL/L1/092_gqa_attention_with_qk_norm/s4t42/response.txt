import torch
import triton
import triton.language as tl


# Kernel 1: Linear with bias: X @ W.T + b
# X: [B, S, H_in], W: [H_out, H_in], b: [H_out] -> Y: [B, S, H_out]
@triton.jit
def linear_bias_kernel(
    X_ptr, W_ptr, BIAS_ptr, Y_ptr,
    Bsz, Ssz, H_in, H_out,
    stride_x_b, stride_x_s, stride_x_h_in,
    stride_w_hout, stride_w_hin,
    stride_bias_hout,
    stride_y_b, stride_y_s, stride_y_h_out,
    BLOCK_IN: tl.constexpr, BLOCK_OUT: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_out = tl.program_id(2)

    # Accumulate over input dims in tiles
    acc = tl.zeros((), dtype=tl.float32)
    for i in range(0, H_in, BLOCK_IN):
        # load X tile: [BLOCK_IN]
        x_vec = tl.zeros((BLOCK_IN,), dtype=tl.float32)
        for ii in range(0, BLOCK_IN):
            x_off = b * stride_x_b + s * stride_x_s + (i + ii) * stride_x_h_in
            x_vec[ii] = tl.load(X_ptr + x_off)
        # load W tile: [BLOCK_OUT, BLOCK_IN]
        w_mat = tl.zeros((BLOCK_OUT, BLOCK_IN), dtype=tl.float32)
        for j in range(0, BLOCK_OUT):
            w_off_row = h_out * stride_w_hout + (i + 0) * stride_w_hin  # + j * stride_w_hout; W is [H_out, H_in]
            for jj in range(0, BLOCK_IN):
                w_off = w_off_row + (i + jj) * stride_w_hin
                w_mat[j, jj] = tl.load(W_ptr + w_off)
        # acc += sum_j (X[i + ii] * W[j, i + jj])
        # note: W is [H_out, H_in]; we index by (h_out, h_in)
        for j in range(0, BLOCK_OUT):
            # sum over BLOCK_IN
            dot = 0.0
            for jj in range(0, BLOCK_IN):
                dot += w_mat[j, jj] * x_vec[jj]
            acc += dot
    # add bias
    bias_val = tl.load(BIAS_ptr + h_out * stride_bias_hout)
    acc += bias_val
    # store
    y_off = b * stride_y_b + s * stride_y_s + h_out * stride_y_h_out
    tl.store(Y_ptr + y_off, acc)


# Kernel 2: RMSNorm over last dim (size = D=128): y = x * rsqrt(mean(x^2) + eps), then scale by weight
# Input x: [B, S, D], weight: [D] -> Output y: [B, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_w_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)  # process one d per program; grid spans D as well
    row_off = b * stride_x_b + s * stride_x_s

    # compute sum of squares over last dimension
    sum_sq = 0.0
    for i in range(0, BLOCK):
        off = row_off + (d + i) * stride_x_d
        x = tl.load(X_ptr + off)
        sum_sq += x * x

    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + 0.0)  # eps assumed 0.0 as in original code
    weight = tl.load(Weight_ptr + (d + 0) * stride_w_d)

    # normalize and scale
    for i in range(0, BLOCK):
        in_off = row_off + (d + i) * stride_x_d
        x = tl.load(X_ptr + in_off)
        y = x * inv_rms
        y = y * weight
        out_off = b * stride_out_b + s * stride_out_s + (d + i) * stride_out_d
        tl.store(Out_ptr + out_off, y)


# Kernel 3: Rotate half of the last 64 dims for a 128-length vector: q1, q2 = q[:64], q[64:], rotate:
# q1' = q1*cos - q2*sin, q2' = q2*cos + q1*sin.
# This kernel takes Q, Sin, Cos and produces Out. Assumes D=128, and operates per (b, s, d).
@triton.jit
def rotate_half_kernel(
    Q_ptr, Sin_ptr, Cos_ptr, Out_ptr,
    Bsz, Ssz, D,  # D should be 128
    stride_q_b, stride_q_s, stride_q_d,
    stride_out_b, stride_out_s, stride_out_d,
    BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)  # d ranges over 0..D-1

    q_row_off = b * stride_q_b + s * stride_q_s
    for i in range(0, BLOCK_D):
        q_off = q_row_off + (d + i) * stride_q_d
        q_val = tl.load(Q_ptr + q_off)
        sin_val = tl.load(Sin_ptr + (d + i))
        cos_val = tl.load(Cos_ptr + (d + i))

        if i < 64:
            # first half
            q1 = q_val
            q2 = tl.load(Q_ptr + q_row_off + (d + 64 + i) * stride_q_d)  # q2 = q[64 + i]
            q1_rot = q1 * cos_val - q2 * sin_val
            q2_rot = q2 * cos_val + q1 * sin_val
            tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + i * stride_out_d, q1_rot)
            tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + (64 + i) * stride_out_d, q2_rot)


# Kernel 4: Compute attention scores: Q[b,h,:] @ K[b,h,:]^T -> [S, S], tiled.
# Inputs: Q: [B, H, S, D], K: [B, H, S, D], outputs: Scores: [B, H, S, S]
# We compute a full [S, S] matrix per (b, h). Using BLOCK_S and BLOCK_D for tiling.
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, Scores_ptr,
    Bsz, H, Ssz, D,
    stride_q_b, stride_q_h, stride_q_s, stride_q_d,
    stride_k_b, stride_k_h, stride_k_s, stride_k_d,
    stride_s_b, stride_s_h, stride_s_s, stride_s_s2,
    BLOCK_S: tl.constexpr, BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    # We tile over S and S for scores
    s_block = tl.program_id(2)
    i = tl.program_id(3)

    s0 = s_block * BLOCK_S
    for s_i in range(0, BLOCK_S):
        i_idx = s0 + s_i
        if i_idx >= Ssz:
            break
        row_acc = tl.zeros((BLOCK_S,), dtype=tl.float32)
        for d0 in range(0, D, BLOCK_D):
            acc_d = tl.zeros((), dtype=tl.float32)
            for d_i in range(0, BLOCK_D):
                k_col = d0 + d_i
                if k_col >= D:
                    break
                # q[b, h, i, k_col] * k[b, h, i, k_col]
                q_off = b * stride_q_b + h * stride_q_h + i_idx * stride_q_s + k_col * stride_q_d
                k_off = b * stride_k_b + h * stride_k_h + i_idx * stride_k_s + k_col * stride_k_d
                q_val = tl.load(Q_ptr + q_off)
                k_val = tl.load(K_ptr + k_off)
                acc_d += q_val * k_val
            # add to row accumulator
            row_acc[s_i] = acc_d
        # store row into Scores
        scores_off = b * stride_s_b + h * stride_s_h + i_idx * stride_s_s + (s0 + tl.arange(0, BLOCK_S)) * stride_s_s2
        # mask for s_i valid
        for s_j in range(0, BLOCK_S):
            j_idx = s0 + s_j
            if j_idx >= Ssz:
                break
            # row_acc[s_j] is computed; we can store it
            tl.store(Scores_ptr + scores_off + s_j * stride_s_s2, row_acc[s_j])


# Kernel 5: Softmax with causal mask over last dimension (sequence length) for Scores: [B, H, S, S]
# Apply mask: if col > row, set to -inf; else keep value. Then softmax.
@triton.jit
def softmax_mask_kernel(
    Scores_ptr, Out_ptr,
    Bsz, H, Ssz,
    stride_s_b, stride_s_h, stride_s_s, stride_s_s2,
    stride_o_b, stride_o_h, stride_o_s, stride_o_s2,
    BLOCK_S: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    for i in range(0, Ssz, BLOCK_S):
        row = tl.zeros((), dtype=tl.int32)
        for s_i in range(0, BLOCK_S):
            j = i + s_i
            if j >= Ssz:
                break
            # load score for row j
            scores_off = b * stride_s_b + h * stride_s_h + j * stride_s_s + (i + tl.arange(0, BLOCK_S)) * stride_s_s2
            score = tl.load(Scores_ptr + scores_off + s_i * stride_s_s2)
            # causal mask: if col > row, set to -inf
            if j > (i + s_i):
                score = -float('inf')
            row = tl.maximum(row, score)
        # compute sum of exp(scores - row)
        sum_exp = 0.0
        for s_i in range(0, BLOCK_S):
            j = i + s_i
            if j >= Ssz:
                break
            scores_off = b * stride_s_b + h * stride_s_h + j * stride_s_s + (i + tl.arange(0, BLOCK_S)) * stride_s_s2
            score = tl.load(Scores_ptr + scores_off + s_i * stride_s_s2)
            if j > (i + s_i):
                score = -float('inf')
            expv = tl.exp(score - row)
            sum_exp += expv
        # write normalized
        for s_i in range(0, BLOCK_S):
            j = i + s_i
            if j >= Ssz:
                break
            scores_off = b * stride_s_b + h * stride_s_h + j * stride_s_s + (i + tl.arange(0, BLOCK_S)) * stride_s_s2
            score = tl.load(Scores_ptr + scores_off + s_i * stride_s_s2)
            if j > (i + s_i):
                score = -float('inf')
            prob = tl.exp(score - row) / sum_exp
            out_off = b * stride_o_b + h * stride_o_h + j * stride_o_s + (i + s_i) * stride_o_s2
            tl.store(Out_ptr + out_off, prob)


# Kernel 6: Attention output: Softmax(QK_scaled) @ V -> Out[b,h,i,:] per (b,s,i), accumulate over j.
# We compute Out[b,h,i,:] = sum_j Softmax[j,i] * V[b,h,j,:]
@triton.jit
def matmul_attn_kernel(
    Softmax_ptr, V_ptr, Out_ptr,
    Bsz, H, Ssz, D,
    stride_sm_b, stride_sm_h, stride_sm_i, stride_sm_j,
    stride_v_b, stride_v_h, stride_v_s, stride_v_d,
    stride_o_b, stride_o_h, stride_o_s, stride_o_d,
    BLOCK_S: tl.constexpr, BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    for i in range(0, Ssz, BLOCK_S):
        row = tl.zeros((), dtype=tl.float32)
        for d0 in range(0, D, BLOCK_D):
            sum_j = tl.zeros((), dtype=tl.float32)
            for d_i in range(0, BLOCK_D):
                d = d0 + d_i
                if d >= D:
                    break
                # sum over j of Softmax[b,h,i,j] * V[b,h,j,d]
                for j in range(0, Ssz, BLOCK_S):
                    acc = tl.zeros((), dtype=tl.float32)
                    for s_i in range(0, BLOCK_S):
                        j_idx = j + s_i
                        if j_idx >= Ssz:
                            break
                        sm_off = b * stride_sm_b + h * stride_sm_h + i * stride_sm_i + j_idx * stride_sm_j
                        sm_val = tl.load(Softmax_ptr + sm_off)
                        v_off = b * stride_v_b + h * stride_v_h + j_idx * stride_v_s + d * stride_v_d
                        v_val = tl.load(V_ptr + v_off)
                        acc += sm_val * v_val
                    sum_j += acc
            # store Out[b,h,i,d] = sum_j
            out_off = b * stride_o_b + h * stride_o_h + (i + tl.arange(0, BLOCK_S)) * stride_o_s + d * stride_o_d
            tl.store(Out_ptr + out_off, sum_j)
