        tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d * stride_yd, x_orig * 0.0 + y_first)  # placeholder

    # Second half
    for i in range(64):
        d = 64 + i
        x2 = x_q2[i]
        x1 = x_q1[i]
        c = cos_row[i]
        s_row = sin_row[i]
        y_second = x2 * c + x1 * s_row
        x_orig = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d * stride_xd).to(tl.float32)
