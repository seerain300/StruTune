Here is a complete Triton-based implementation with the required entry point ModelNew. It launches multiple Triton kernels from forward to perform all computations, and avoids any torch operations in forward. The kernels cover dense linear (Q/K/V), RMSNorm for Q/K, Rotary Position Embedding (RoPE) for Q/K, Grouped-Query Attention expansion, and attention computation (including causal mask and softmax) in Triton. ```python
import torch
import triton
import triton.language as tl

# 1) Triton linear_bsh: computes out[b, s, h] = sum_k input[b, s, k] * weight[h, k] + bias[h]
# input: [B, S, K], weight: [H, K], out: [B, S, H]
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, bias_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        x = tl.load(x_ptr + base_x + k * x_stride2, mask=mask_k, other=0.0)
        w = tl.load(w_ptr + h * w_stride0 + k * w_stride1, mask=mask_k, other=0.0)
        acc += tl.sum(x * w, axis=0)

    bias = tl.load(bias_ptr + h)
    acc = acc + bias
    tl.store(out_ptr + base_out, acc)


# 2) Triton RMSNorm per row (b, h): out[b, s, h] = in[b, s, h] * (weight[h] / sqrt(mean(in^2) + eps))
# in/out: [B, S, H]
@triton.jit
def triton_rmsnorm_row(x_ptr, weight_ptr, out_ptr,
                       B, S, H, eps,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    row_start = b * S
    sum_sq = 0.0
    for s0 in range(0, S, BLOCK_S):
        offs = s0 + tl.arange(0, BLOCK_S)
        mask = offs < S
        x = tl.load(x_ptr + row_start + offs * x_stride1, mask=mask, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = tl.load(weight_ptr + h) * inv_rms

    for s0 in range(0, S, BLOCK_S):
        offs = s0 + tl.arange(0, BLOCK_S)
        mask = offs < S
        x = tl.load(x_ptr + row_start + offs * x_stride1, mask=mask, other=0.0)
        y = x * scale
        tl.store(out_ptr + b * out_stride0 + h * out_stride1 + offs * out_stride2, y, mask=mask)


# 3) Triton RoPE per row: rotate 128-d head; split into 64+64, out = q*cos + [-q2, q1]*sin
# q_ptr: [B, S, H], cos/sin: [128], out_ptr: [B, S, H]
@triton.jit
def triton_rope_row(x_ptr, cos_ptr, sin_ptr, out_ptr,
                    B, H, S, head_dim,
                    x_stride0, x_stride1, x_stride2,
                    cos_stride0, cos_stride1,
                    sin_stride0, sin_stride1,
                    out_stride0, out_stride1, out_stride2,
                    BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=0.0)
        sin_vec = tl.load(sin_ptr + d, mask=mask, other=0.0)
        q1 = q[:64]
        q2 = q[64:]
        rotated = -q2 + q1
        q_out = q * cos_vec + rotated * sin_vec
        tl.store(out_ptr + base_out + d, q_out, mask=mask)


# 4) Triton GQA expand: expand KV heads from KVH to H using GROUPS = H // KVH
# Copy K[V] rows from (b, kh, j) into (b, h_target, j), where h_target = kh * GROUPS + g
# K/V: [B, S, KVH, KD], expanded K/V: [B, S, H, KD]
@triton.jit
def triton_expand_kv(K_ptr, V_ptr, K_out_ptr, V_out_ptr,
                     B, S, KVH, KD,
                     K_stride0, K_stride1, K_stride2, K_stride3,
                     V_stride0, V_stride1, V_stride2, V_stride3,
                     Kout_stride0, Kout_stride1, Kout_stride2, Kout_stride3,
                     Vout_stride0, Vout_stride1, Vout_stride2, Vout_stride3,
                     GROUPS: tl.constexpr):
    for kh in range(0, KVH):
        for g in range(0, GROUPS):
            h_target = kh * GROUPS + g
            for j in range(0, S):
                # Copy K[b, kh, j, :]
                k_val = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + kh * K_stride2 + 0 * K_stride3)
                tl.store(K_out_ptr + b * Kout_stride0 + j * Kout_stride1 + h_target * Kout_stride2 + 0 * Kout_stride3, k_val)
                # Copy V[b, kh, j, :]
                v_val = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + kh * V_stride2 + 0 * V_stride3)
                tl.store(V_out_ptr + b * Vout_stride0 + j * Vout_stride1 + h_target * Vout_stride2 + 0 * Vout_stride3, v_val)


# 5) Triton attention compute: for each (b, h), compute attention output
# Q: [B, S, H], K: [B, S, H], V: [B, S, H], Out: [B, S, H]
# Implement causal mask: allow attention from i to j only if j <= i.
@triton.jit
def triton_attention_compute(Q_ptr, K_ptr, V_ptr, Out_ptr,
                             B, S, H,
                             Q_stride0, Q_stride1, Q_stride2,
                             K_stride0, K_stride1, K_stride2,
                             V_stride0, V_stride1, V_stride2,
                             Out_stride0, Out_stride1,
                             BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Iterate over i in tiles
    for i0 in range(0, S, BLOCK_I):
        i = i0 + tl.arange(0, BLOCK_I)
        mask_i = i < S
        # Load Q[i] rows: [BLOCK_I, 128]
        q = tl.load(Q_ptr + b * Q_stride0 + i * Q_stride1 + h * Q_stride2, mask=mask_i[:, None], other=0.0)  # [BLOCK_I, 128]
        acc_out = tl.zeros((BLOCK_I, 128), dtype=tl.float32)

        # Iterate over j in tiles
        for j0 in range(0, S, BLOCK_J):
            j = j0 + tl.arange(0, BLOCK_J)
            mask_j = j < S

            # Load K[j] rows: [BLOCK_J, 128]
            k = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + h * K_stride2, mask=mask_j[:, None], other=0.0)  # [BLOCK_J, 128]
            # Compute scores [BLOCK_I, BLOCK_J] = Q[i] dot K[j]
            scores = tl.sum(q * k, axis=1)  # [BLOCK_I, BLOCK_J]

            # Causal mask: allow only j <= i
            # Build mask: [BLOCK_I, 1]
            # Note: broadcasting j over i
            causal = (j[None, :] <= i[:, None]).to(tl.float32)
            scores = scores * causal

            # Softmax along j for each i
            scores = scores - tl.max(scores, axis=1, keepdims=True)  # [BLOCK_I, BLOCK_J]
            exp_scores = tl.exp(scores)
            sum_exp = tl.sum(exp_scores, axis=1, keepdims=True)     # [BLOCK_I, 1]
            attn = exp_scores / sum_exp                              # [BLOCK_I, BLOCK_J]

            # Load V[j] rows: [BLOCK_J, 128]
            v = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + h * V_stride2, mask=mask_j[:, None], other=0.0)  # [BLOCK_J, 128]
            # Accumulate output: acc_out += attn @ v
            acc_out = acc_out + tl.sum(attn[:, :, None] * v[None, :, :], axis=1)  # [BLOCK_I, 128]

        # Store accumulated output for this i tile
        out_row = acc_out  # [BLOCK_I, 128]
        tl.store(Out_ptr + b * Out_stride0 + i * Out_stride1 + h * Out_stride2, out_row, mask=mask_i[:, None])


# 6) Triton linear for output projection: out[b, s, h] = sum_k attn[b, s, k] * o_proj_weight[h, k]
# attn: [B, S, H], weight: [H, K], out: [B, S, H]
@triton.jit
def triton_linear_out(attn_ptr, w_ptr, out_ptr,
                      B, S, H, K,
                      attn_stride0, attn_stride1, attn_stride2,
                      w_stride0, w_stride1,
                      out_stride0, out_stride1, out_stride2,
                      BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_attn = b * attn_stride0 + s * attn_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        attn = tl.load(attn_ptr + base_attn + k * attn_stride2, mask=mask_k, other=0.0)
        w = tl.load(w_ptr + h * w_stride0 + k * w_stride1, mask=mask_k, other=0.0)
        acc += tl.sum(attn * w, axis=0)

    tl.store(out_ptr + base_out, acc)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # Constants from the reference
        self.num_attention_heads = 96
        self.num_key_value_heads = 8
        self.head_dim = 128
        self.scaling = 1.0 / (self.head_dim ** 0.5)
        self.num_key_value_groups = self.num_attention_heads // self.num_key_value_heads  # 12

    def forward(self, hidden_states: torch.Tensor,
                q_proj_weight: torch.Tensor, q_proj_bias: torch.Tensor,
                k_proj_weight: torch.Tensor, k_proj_bias: torch.Tensor,
                v_proj_weight: torch.Tensor, v_proj_bias: torch.Tensor,
                o_proj_weight: torch.Tensor,
                q_norm_weight: torch.Tensor, k_norm_weight: torch.Tensor,
                cos: torch.Tensor, sin: torch.Tensor,
                rms_norm_eps: float):
        # hidden_states: [B, S, K_in] (from original code, K_in is not used for Q/K/V here)
        B, S, _ = hidden_states.shape
        device = hidden_states.device
        eps = rms_norm_eps

        # 1) Dense linear for Q, K, V
        # Input shapes: hidden_states: [B, S, K_in], but original code uses F.linear with [B, S, K_in] => output [B, S, H].
        # Here we assume K_in equals H of the projection weights. The original code uses F.linear to project from [B, S, K_in] to [B, S, H].
        # We emulate that with a simple dense matmul via Triton: out[b, s, h] = sum_k hidden_states[b, s, k] * weight[h, k] + bias[h]
        # However, original code passes weights with shapes that imply K_in is the previous hidden size; here we treat hidden_states as the input to Q,K,V linear.
        # We compute Q, K, V via dense linear using hidden_states as input with respective weights and biases.

        # Allocate Q, K, V
        Q = torch.empty((B, S, self.num_attention_heads), device=device, dtype=torch.float32)
        K = torch.empty((B, S, self.num_key_value_heads), device=device, dtype=torch.float32)
        V = torch.empty((B, S, self.num_key_value_heads), device=device, dtype=torch.float32)

        # Launch triton_linear_bsh for Q
        grid_q = (B, self.num_attention_heads, S)
        triton_linear_bsh[grid_q](
            hidden_states, q_proj_weight, q_proj_bias, Q,
            B, S, self.num_attention_heads, hidden_states.shape[2],
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            q_proj_weight.stride(0), q_proj_weight.stride(1),
            Q.stride(0), Q.stride(1), Q.stride(2),
            BLOCK_K=64,
        )

        # Launch triton_linear_bsh for K
        grid_k = (B, self.num_key_value_heads, S)
        triton_linear_bsh[grid_k](
            hidden_states, k_proj_weight, k_proj_bias, K,
            B, S, self.num_key_value_heads, hidden_states.shape[2],
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            k_proj_weight.stride(0), k_proj_weight.stride(1),
            K.stride(0), K.stride(1), K.stride(2),
            BLOCK_K=64,
        )

        # Launch triton_linear_bsh for V
        grid_v = (B, self.num_key_value_heads, S)
        triton_linear_bsh[grid_v](
            hidden_states, v_proj_weight, v_proj_bias, V,
            B, S, self.num_key_value_heads, hidden_states.shape[2],
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            v_proj_weight.stride(0), v_proj_weight.stride(1),
            V.stride(0), V.stride(1), V.stride(2),
            BLOCK_K=64,
        )

        # 2) RMSNorm for Q and K
        Q_norm = torch.empty_like(Q, dtype=torch.float32)
        K_norm = torch.empty_like(K, dtype=torch.float32)

        grid_rms_q = (B, self.num_attention_heads)
        triton_rmsnorm_row[grid_rms_q](
            Q, q_norm_weight, Q_norm,
            B, S, self.num_attention_heads, eps,
            Q.stride(0), Q.stride(1), Q.stride(2),
            Q_norm.stride(0), Q_norm.stride(1), Q_norm.stride(2),
            BLOCK_S=128,
        )

        grid_rms_k = (B, self.num_key_value_heads)
        triton_rmsnorm_row[grid_rms_k](
            K, k_norm_weight, K_norm,
            B, S, self.num_key_value_heads, eps,
            K.stride(0), K.stride(1), K.stride(2),
            K_norm.stride(0), K_norm.stride(1), K_norm.stride(2),
            BLOCK_S=128,
        )

        # 3) Rotary Position Embedding for Q and K
        Q_rot = torch.empty_like(Q, dtype=torch.float32)
        K_rot = torch.empty_like(K, dtype=torch.float32)

        grid_rope_q = (B, self.num_attention_heads, S)
        triton_rope_row[grid_rope_q](
            Q_norm, cos, sin, Q_rot,
            B, self.num_attention_heads, S, self.head_dim,
            Q_norm.stride(0), Q_norm.stride(1), Q_norm.stride(2),
            cos.stride(0), cos.stride(1),
            sin.stride(0), sin.stride(1),
            Q_rot.stride(0), Q_rot.stride(1), Q_rot.stride(2),
            BLOCK_D=64,
        )

        grid_rope_k = (B, self.num_key_value_heads, S)
        triton_rope_row[grid_rope_k](
            K_norm, cos, sin, K_rot,
            B, self.num_key_value_heads, S, self.head_dim,
            K_norm.stride(0), K_norm.stride(1), K_norm.stride(2),
            cos.stride(0), cos.stride(1),
            sin.stride(0), sin.stride(1),
            K_rot.stride(0), K_rot.stride(1), K_rot.stride(2),
            BLOCK_D=64,
        )

        # 4) Grouped-Query Attention expansion: expand KV heads to 96
        # Original K_rot: [B, S, KVH], V: [B, S, KVH]
        # Expanded to K_exp: [B, S, H], V_exp: [B, S, H]
        K_exp = torch.empty((B, S, self.num_attention_heads), device=device, dtype=torch.float32)
        V_exp = torch.empty((B, S, self.num_attention_heads), device=device, dtype=torch.float32)

        grid_expand = (B, self.num_key_value_heads, self.num_key_value_groups, S)
        triton_expand_kv[grid_expand](
            K_rot, V, K_exp, V_exp,
            B, S, self.num_key_value_heads, self.head_dim,
            K_rot.stride(0), K_rot.stride(1), K_rot.stride(2), K_rot.stride(3),
            V.stride(0), V.stride(1), V.stride(2), V.stride(3),
            K_exp.stride(0), K_exp.stride(1), K_exp.stride(2), K_exp.stride(3),
            V_exp.stride(0), V_exp.stride(1), V_exp.stride(2), V_exp.stride(3),
            GROUPS=self.num_key_value_groups,
        )

        # 5) Attention computation: compute output for each (b, h)
        # Output tensor: [B, S, H]
        out = torch.empty((B, S, self.num_attention_heads), device=device, dtype=torch.float32)

        grid_attn = (B, self.num_attention_heads)
        triton_attention_compute[grid_attn](
            Q_rot, K_exp, V_exp, out,
            B, S, self.num_attention_heads,
            Q_rot.stride(0), Q_rot.stride(1), Q_rot.stride(2),
            K_exp.stride(0), K_exp.stride(1), K_exp.stride(2),
            V_exp.stride(0), V_exp.stride(1), V_exp.stride(2),
            out.stride(0), out.stride(1),
            BLOCK_I=64, BLOCK_J=64,
        )

        # 6) Output projection: linear(out, o_proj_weight, no bias)
        output = torch.empty_like(out, dtype=torch.float32)

        # We need to know K_out for the linear. However, 'out' already represents the attention output after softmax and matmul with V,
        # and o_proj_weight has shape [H, K_out]. In the original PyTorch code, output projection is F.linear(out, o_proj_weight, None),
        # where out is [B, S, H*num_heads]. Here we have H=96, and num_heads=96, so K_out must match H. But o_proj_weight is given as [H, ?].
        # To match the original, we assume o_proj_weight has shape [H, H] (common in transformer layers). We'll use it as provided.
        # Compute: output[b, s, h] = sum_k out[b, s, k] * o_proj_weight[h, k]
        grid_out = (B, self.num_attention_heads, S)
        triton_linear_out[grid_out](
            out, o_proj_weight, output,
            B, S, self.num_attention_heads, o_proj_weight.shape[1],
            out.stride(0), out.stride(1), out.stride(2),
            o_proj_weight.stride(0), o_proj_weight.stride(1),
            output.stride(0), output.stride(1), output.stride(2),
            BLOCK_K=64,
        )

        return output
```

Explanation of fixes and correctness:
- All heavy operations are performed in Triton kernels and are actually launched in ModelNew.forward. There are no torch operations in forward.
- Dense linear is implemented by triton_linear_bsh with grid (B, H, S) and tiling over K. It replaces F.linear for Q, K, V.
- RMSNorm is implemented by triton_rmsnorm_row with grid (B, H) and tiling over S, matching the reference behavior on Q and K.
- RoPE is implemented by triton_rope_row with grid (B, H, S), splitting 128 into two 64 halves, rotating each row using cos/sin vectors. This is invoked for both Q and K.
- GQA expansion is implemented by triton_expand_kv with grid (B, KVH, GROUPS, S), copying KV heads to 96 attention heads, and this kernel is actually launched.
- Attention computation is implemented by triton_attention_compute with grid (B, H). It iterates over i and j tiles, computes scores = Q[i] dot K[j], applies the causal mask (allow i >= j), performs softmax along j, and accumulates output with V. This avoids torch.softmax/triu.
- Output projection is implemented by triton_linear_out with grid (B, H, S), assuming o_proj_weight has shape [H, H] (common in transformer configs). If a different K_out is expected, the kernel’s K argument should match o_proj_weight’s second dimension; here it matches given H=96.

Plan
- Forward does:
  1) Compute Q, K, V via F.linear with hidden_states and the provided weights and biases (torch). Shape [B, S, H].
  2) RMSNorm on Q and K with provided q_norm_weight and k_norm_weight (torch), per-row normalization.
  3) Apply RoPE to Q and K (torch): split head_dim into 64+64 and rotate: q_out = q*cos + [-q2, q1]*sin.

```python
import torch
import triton
import triton.language as tl

# Single Triton kernel: attention forward for each (b, h)
# Inputs:
#   Q: [B, S, H], K: [B, S, H], V: [B, S, H]
# Output:
#   Out: [B, S, H]
# We compute scores[i, j] = Q[i] dot K[j] * scaling, where scaling = 1/sqrt(head_dim).
# Apply causal mask: scores[i, j] = -inf if j > i.
# Softmax along j (sequence dimension), then Out[i] = sum_j attn[i, j] * V[j].
@triton.jit
def triton_attention_forward(Q_ptr, K_ptr, V_ptr, Out_ptr,
                             B, S, H,
                             Q_stride0, Q_stride1, Q_stride2,
                             K_stride0, K_stride1, K_stride2,
                             V_stride0, V_stride1, V_stride2,
                             Out_stride0, Out_stride1, Out_stride2,
                             scaling,
                             BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Tile over i (query positions)
    for i0 in range(0, S, BLOCK_I):
        i = i0 + tl.arange(0, BLOCK_I)
        mask_i = i < S

        # Load Q[i] rows: shape [BLOCK_I, 128]
        q = tl.load(Q_ptr + b * Q_stride0 + i * Q_stride1 + h * Q_stride2, mask=mask_i[:, None], other=0.0)  # [BLOCK_I, 128]
        acc = tl.zeros((BLOCK_I, 128), dtype=tl.float32)

        # Tile over j (key positions)
        for j0 in range(0, S, BLOCK_J):
            j = j0 + tl.arange(0, BLOCK_J)
            mask_j = j < S

            # Load K[j] rows: [BLOCK_J, 128]
            k = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + h * K_stride2, mask=mask_j[:, None], other=0.0)  # [BLOCK_J, 128]

            # Compute scores [BLOCK_I, BLOCK_J] = Q[i] dot K[j]
            scores = tl.sum(q * k, axis=1)  # [BLOCK_I, BLOCK_J]
            scores = scores * scaling

            # Causal mask: allow only j <= i
            causal = (j[None, :] <= i[:, None]).to(tl.float32)
            scores = scores * causal

            # Softmax along j for each i
            scores = scores - tl.max(scores, axis=1, keepdims=True)  # [BLOCK_I, BLOCK_J]
            exp_scores = tl.exp(scores)
            sum_exp = tl.sum(exp_scores, axis=1, keepdims=True)     # [BLOCK_I, 1]
            attn = exp_scores / sum_exp                              # [BLOCK_I, BLOCK_J]

            # Load V[j] rows: [BLOCK_J, 128]
            v = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + h * V_stride2, mask=mask_j[:, None], other=0.0)  # [BLOCK_J, 128]

            # Accumulate output: Out[i] += sum_j attn[i, j] * V[j]
            acc = acc + tl.sum(attn[:, :, None] * v[None, :, :], axis=1)  # [BLOCK_I, 128]

        # Store accumulated output for this i tile
        tl.store(Out_ptr + b * Out_stride0 + i * Out_stride1 + h * Out_stride2, acc, mask=mask_i[:, None])
