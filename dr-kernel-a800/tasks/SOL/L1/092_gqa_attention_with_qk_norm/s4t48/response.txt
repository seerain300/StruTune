import torch
import triton
import triton.language as tl


# Kernel 1: Linear with bias: X @ W.T + b
# X: [B, S, Hin], W: [Hout, Hin], b: [Hout] -> Out: [B, S, Hout]
# Assumes Hout=1152 (compile-time constant).
@triton.jit
def linear_bias_kernel(
    X_ptr, W_ptr, BIAS_ptr, Out_ptr,
    Bsz, Ssz, Hin,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_h, stride_w_in,
    stride_bias_h,
    stride_out_b, stride_out_s, stride_out_h,
    HOUT: tl.constexpr,  # must be 1152
):
    # Grid is (Bsz, Ssz, HOUT)
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_out = tl.program_id(2)

    # Accumulator for output
    acc = 0.0
    # Loop over input dimension Hin
    for h_in in range(0, Hin):
        # Load X[b, s, h_in]
        x_off = b * stride_x_b + s * stride_x_s + h_in * stride_x_h
        x_val = tl.load(X_ptr + x_off)
        # Load W[h_out, h_in]
        w_off = h_out * stride_w_h + h_in * stride_w_in
        w_val = tl.load(W_ptr + w_off)
        acc += x_val * w_val
    # Add bias
    bias_val = tl.load(BIAS_ptr + h_out * stride_bias_h)
    acc += bias_val

    # Store to Out[b, s, h_out]
    out_off = b * stride_out_b + s * stride_out_s + h_out * stride_out_h
    tl.store(Out_ptr + out_off, acc)


# Kernel 2: RMSNorm over last dim (D=128): y = x * rsqrt(mean(x^2) + eps), then scale by weight
# Input x: [B, S, D], weight: [D] -> Output y: [B, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_w_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)

    row_off = b * stride_x_b + s * stride_x_s

    # Load x over head_dim
    x_vec = tl.zeros((BLOCK,), dtype=tl.float32)
    for i in range(0, BLOCK):
        off = row_off + (d + i) * stride_x_d
        x_vec[i] = tl.load(X_ptr + off)

    # RMS computation
    sum_sq = tl.sum(x_vec * x_vec, axis=0)
    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + 0.0)  # eps=0.0 as in original code
    weight = tl.load(Weight_ptr + (d + 0) * stride_w_d)

    y_vec = x_vec * inv_rms
    y_vec = y_vec * weight

    out_off = b * stride_out_b + s * stride_out_s + (d + 0) * stride_out_d
    for i in range(0, BLOCK):
        tl.store(Out_ptr + out_off + i * stride_out_d, y_vec[i])


# Kernel 3: Rotate half (last 64 dims) of 128-dim vector: q1, q2 = q[:64], q[64:], q_rot = q1*cos - q2*sin
@triton.jit
def rotate_half_kernel(
    Q_ptr, Sin_ptr, Cos_ptr, Out_ptr,
    Bsz, Ssz, D,  # D should be 128
    stride_q_b, stride_q_s, stride_q_d,
    stride_s_d, stride_c_d, stride_out_b, stride_out_s, stride_out_d,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)  # 0..D-1

    q_off = b * stride_q_b + s * stride_q_s + d * stride_q_d
    sin_off = d * stride_s_d
    cos_off = d * stride_c_d

    q_val = tl.load(Q_ptr + q_off)
    sin_val = tl.load(Sin_ptr + sin_off)
    cos_val = tl.load(Cos_ptr + cos_off)

    # Compute rotated for the last 64 dims
    q1 = q_val[:64]
    q2 = q_val[64:]
    rotated = q1 * cos_val - q2 * sin_val

    out_off = b * stride_out_b + s * stride_out_s + d * stride_out_d
    tl.store(Out_ptr + out_off, rotated)


# Kernel 4: Compute attention scores QH @ KH^T -> [S, S] per (b, s, h)
# QH: [S, 128], KH: [S, 128]
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, Out_ptr,
    Bsz, Ssz, D,  # D=128
    stride_q_b, stride_q_s, stride_q_d,
    stride_k_b, stride_k_s, stride_k_d,
    stride_out_b, stride_out_s, stride_out_seq,
    BLOCK: tl.constexpr,
):
    # Grid is (Bsz, Ssz, Ssz) i.e., (b, s, seq_idx)
    b = tl.program_id(0)
    s = tl.program_id(1)
    seq_idx = tl.program_id(2)

    # Load Q row s and K row seq_idx
    q_vec = tl.zeros((BLOCK,), dtype=tl.float32)
    k_vec = tl.zeros((BLOCK,), dtype=tl.float32)
    for i in range(0, BLOCK):
        q_off = b * stride_q_b + s * stride_q_s + i * stride_q_d
        k_off = b * stride_k_b + seq_idx * stride_k_s + i * stride_k_d
        q_vec[i] = tl.load(Q_ptr + q_off)
        k_vec[i] = tl.load(K_ptr + k_off)

    score = tl.sum(q_vec * k_vec, axis=0)
    out_off = b * stride_out_b + s * stride_out_s + seq_idx * stride_out_seq
    tl.store(Out_ptr + out_off, score)


# Kernel 5: Softmax with causal mask over last dim (sequence length) per (b, h)
# Input scores: [S, S], Output softmax_scores: [S, S]
@triton.jit
def softmax_mask_kernel(
    Scores_ptr, Mask_ptr, Out_ptr,
    Bsz, Ssz,
    stride_scores_b, stride_scores_s1, stride_scores_s2,
    stride_mask_s1, stride_mask_s2,
    stride_out_b, stride_out_s1, stride_out_s2,
    BLOCK: tl.constexpr,
):
    # Grid is (Bsz, Ssz, Ssz) i.e., (b, row, col)
    b = tl.program_id(0)
    row = tl.program_id(1)
    col = tl.program_id(2)

    # Load score and mask
    score_off = b * stride_scores_b + row * stride_scores_s1 + col * stride_scores_s2
    mask_off = row * stride_mask_s1 + col * stride_mask_s2

    score = tl.load(Scores_ptr + score_off)
    # mask: if col > row, set to -inf, else keep score
    # Create a scalar mask: mask = -float('inf') if col > row else score
    # However, since we can't branch per element here, we compute per row by first taking max.
    # We will implement a row-wise softmax with mask by loading the row and applying mask in kernel.
    # But simpler: load entire row into registers, apply mask, compute max and sum.

    # Load entire row for this (b, row)
    row_scores = tl.zeros((BLOCK,), dtype=tl.float32)
    for j in range(0, BLOCK):
        off = b * stride_scores_b + row * stride_scores_s1 + j * stride_scores_s2
        row_score = tl.load(Scores_ptr + off)
        # Apply mask for each col j
        mask_val = tl.load(Mask_ptr + row * stride_mask_s1 + j * stride_mask_s2)
        row_scores[j] = row_score if mask_val != -float('inf') else -float('inf')

    row_max = tl.max(row_scores, axis=0)
    row_exp = tl.exp(row_scores - row_max)
    row_sum = tl.sum(row_exp, axis=0)
    softmax_row = row_exp / row_sum

    out_off = b * stride_out_b + row * stride_out_s1 + col * stride_out_s2
    tl.store(Out_ptr + out_off, softmax_row[col])

def matmul_attn_kernel(
