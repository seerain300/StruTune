```python
import torch
import triton
import triton.language as tl


# 1) Linear projection: y[b, l, n] = sum_k x[b, l, k] * w[n, k], accumulate in f32, output f32
@triton.jit
def linear_proj_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H_in]
    w_ptr,           # *f16/f32/bf16, [N_out, H_in]
    bias_ptr,        # *f32, [N_out] (dummy if None)
    y_ptr,           # *f32, [B, L, N_out]
    B, L, H_in, N_out,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)
    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_in, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_in
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)
        # multiply-accumulate in f32
        x_vals = x_vals.to(tl.float32)
        w_vals = w_vals.to(tl.float32)
        acc += tl.sum(x_vals * w_vals, axis=0)
    tl.store(y_ptr + b * y_bs0 + l * y_bs1 + n * y_bs2, acc)


# 2) RMSNorm for tensors of shape [B, L, H]: y[b, l, h] = x[b, l, h] * rsqrt(mean(x^2)+eps) * weight[h]
@triton.jit
def rmsnorm_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H]
    weight_ptr,      # *f32, [H]
    y_ptr,           # *f32, [B, L, H]
    B, L, H,
    x_bs0, x_bs1, x_bs2,
    y_bs0, y_bs1, y_bs2,
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    h = tl.program_id(2)
    # compute variance over H dimension
    sum_sq = tl.zeros((), dtype=tl.float32)
    for h0 in range(0, H, BLOCK):
        offs = h0 + tl.arange(0, BLOCK)
        mask = offs < H
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs * x_bs2
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)
        sum_sq += tl.sum(x_vals * x_vals, axis=0)
    mean = sum_sq / H
    inv_rms = tl.rsqrt(mean + eps)
    scale = inv_rms
    # apply weight and store
    x_ptr_el = x_ptr + b * x_bs0 + l * x_bs1 + h * x_bs2
    x_val = tl.load(x_ptr_el).to(tl.float32)
    w_ptr_el = weight_ptr + h
    w_val = tl.load(w_ptr_el)
    y_val = x_val * scale * w_val
    tl.store(y_ptr + b * y_bs0 + l * y_bs1 + h * y_bs2, y_val)


# 3) Rotate Q/K (RoPE) per (b, l), split head into h1[:64], h2[64:], apply cos/sin and concatenate
@triton.jit
def rotate_qk_kernel(
    q_ptr, k_ptr, cos_ptr, sin_ptr,
    q_out_ptr, k_out_ptr,
    B, L, H,
    q_bs0, q_bs1, q_bs2,
    k_bs0, k_bs1, k_bs2,
    q_out_bs0, q_out_bs1, q_out_bs2,
    k_out_bs0, k_out_bs1, k_out_bs2,
    BLOCK: tl.constexpr,
):
    # Each program handles one (b, l)
    b = tl.program_id(0)
    l = tl.program_id(1)
    # H is 128; half is 64
    for h0 in range(0, H, BLOCK):
        offs = h0 + tl.arange(0, BLOCK)
        mask = offs < H
        q_ptrs = q_ptr + b * q_bs0 + l * q_bs1 + offs * q_bs2
        k_ptrs = k_ptr + b * k_bs0 + l * k_bs1 + offs * k_bs2
        q_vals = tl.load(q_ptrs, mask=mask, other=0.0).to(tl.float32)
        k_vals = tl.load(k_ptrs, mask=mask, other=0.0).to(tl.float32)
        h1 = offs[:64]
        h2 = offs[64:]
        # load cos/sin for corresponding positions
        cos_h1 = tl.load(cos_ptr + l * 128 + h1).to(tl.float32)
        sin_h1 = tl.load(cos_ptr + l * 128 + h1 + 64).to(tl.float32)  # wrong, should read sin
        # correction: sin for h1 from sin_ptr
        sin_h1 = tl.load(sin_ptr + l * 128 + h1).to(tl.float32)
        cos_h2 = tl.load(cos_ptr + l * 128 + h2).to(tl.float32)
        sin_h2 = tl.load(sin_ptr + l * 128 + h2).to(tl.float32)
        # q1 = cos * q1 + sin * q2; q2 = -sin * q1 + cos * q2
        q1 = q_vals[:64] * cos_h1 - q_vals[64:] * sin_h1
        q2 = q_vals[:64] * sin_h1 + q_vals[64:] * cos_h2
        q_rot = tl.concatenate([q1, q2], axis=0)
        k1 = k_vals[:64] * cos_h1 - k_vals[64:] * sin_h1
        k2 = k_vals[:64] * sin_h1 + k_vals[64:] * cos_h2
        k_rot = tl.concatenate([k1, k2], axis=0)
        # store
        q_out_ptrs = q_out_ptr + b * q_out_bs0 + l * q_out_bs1 + offs * q_out_bs2
        k_out_ptrs = k_out_ptr + b * k_out_bs0 + l * k_out_bs1 + offs * k_out_bs2
        tl.store(q_out_ptrs, q_rot, mask=mask)
        tl.store(k_out_ptrs, k_rot, mask=mask)


# 4) Attention scores matmul: scores[b, qh, l, t] = Q[b, qh, l] * K[b, qh, t]
@triton.jit
def attn_matmul_kernel(
    q_ptr, k_ptr, scores_ptr,
    B, num_heads, L,
    q_bs0, q_bs1, q_bs2,
    k_bs0, k_bs1, k_bs2,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)
    # iterate t in [0..L-1]
    for t in range(0, L, BLOCK):
        offs_t = t + tl.arange(0, BLOCK)
        mask_t = offs_t < L
        q_el_ptr = q_ptr + b * q_bs0 + qh * q_bs1 + l * q_bs2
        q_val = tl.load(q_el_ptr).to(tl.float32)
        k_vals_ptr = k_ptr + b * k_bs0 + qh * k_bs1 + offs_t * k_bs2
        k_vals = tl.load(k_vals_ptr, mask=mask_t, other=0.0).to(tl.float32)
        score = q_val * k_vals  # [BLOCK]
        # store into scores[b, qh, l, offs_t]
        scores_ptrs = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + offs_t * scores_bs3
        tl.store(scores_ptrs, score, mask=mask_t)


# 5) Softmax + causal mask (upper triangle, diagonal=1): per (b, qh, l) row over t
@triton.jit
def softmax_mask_kernel(
    scores_ptr, mask_ptr, probs_ptr,
    B, num_heads, L,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
    mask_bs0, mask_bs1, mask_bs2, mask_bs3,
    probs_bs0, probs_bs1, probs_bs2, probs_bs3,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)
    max_score = tl.full((), -float('inf'), dtype=tl.float32)
    # compute max over t in [0..L-1]
    for t in range(0, L, BLOCK):
        offs_t = t + tl.arange(0, BLOCK)
        mask_t = offs_t < L
        sptr = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + offs_t * scores_bs3
        scores = tl.load(sptr, mask=mask_t, other=-float('inf'))
        scores = scores.to(tl.float32)
        max_score = tl.maximum(max_score, tl.max(scores, axis=0))
    # exp and sum
    sum_exp = tl.zeros((), dtype=tl.float32)
    for t in range(0, L, BLOCK):
        offs_t = t + tl.arange(0, BLOCK)
        mask_t = offs_t < L
        sptr = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + offs_t * scores_bs3
        scores = tl.load(sptr, mask=mask_t, other=-float('inf'))
        scores = scores.to(tl.float32)
        exp_scores = tl.exp(scores - max_score)
        sum_exp += tl.sum(exp_scores, axis=0)
    # apply causal mask: if t < l, set score to -inf
    for t in range(0, L, BLOCK):
        offs_t = t + tl.arange(0, BLOCK)
        mask_t = offs_t < L
        sptr = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + offs_t * scores_bs3
        mptr = mask_ptr + b * mask_bs0 + qh * mask_bs1 + l * mask_bs2 + offs_t * mask_bs3
        scores = tl.load(sptr, mask=mask_t, other=-float('inf')).to(tl.float32)
        mask_vals = tl.load(mptr, mask=mask_t, other=-float('inf')).to(tl.float32)
        scores = tl.where(mask_vals > 0, -float('inf'), scores)
        exp_scores = tl.exp(scores - max_score)
        probs = exp_scores / sum_exp
        pptr = probs_ptr + b * probs_bs0 + qh * probs_bs1 + l * probs_bs2 + offs_t * probs_bs3
        tl.store(pptr, probs, mask=mask_t)


# 6) Output matmul: attn_output[b, qh, l] = sum_t probs[b, qh, l, t] * V[b, qh, t]
@triton.jit
def output_matmul_kernel(
    probs_ptr, v_ptr, attn_out_ptr,
    B, num_heads, L,
    probs_bs0, probs_bs1, probs_bs2, probs_bs3,
    v_bs0, v_bs1, v_bs2,
    attn_bs0, attn_bs1, attn_bs2,
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)
    acc = tl.zeros((), dtype=tl.float32)
    for t in range(0, L, BLOCK):
        offs_t = t + tl.arange(0, BLOCK)
        mask_t = offs_t < L
        pptr = probs_ptr + b * probs_bs0 + qh * probs_bs1 + l * probs_bs2 + offs_t * probs_bs3
        vptr = v_ptr + b * v_bs0 + qh * v_bs1 + offs_t * v_bs2
        probs = tl.load(pptr, mask=mask_t, other=0.0).to(tl.float32)
        v_vals = tl.load(vptr, mask=mask_t, other=0.0).to(tl.float32)
        acc += tl.sum(probs * v_vals, axis=0)
    tl.store(attn_out_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2, acc)


# 7) Final linear projection: final_out[b, l, n] = sum_k attn_output_flat[b, l, k] * o_proj_weight[n, k] (no bias)
@triton.jit
def final_linear_kernel(
    attn_ptr, o_ptr, out_ptr,
    B, L, H, K_total,
    attn_bs0, attn_bs1, attn_bs2,
    o_bs0, o_bs1,
    out_bs0, out_bs1, out_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)
    acc = tl.zeros((), dtype=tl.float32)
    # attn_ptr is flattened [B, L, H_total]; here H_total is K_total
    for k0 in range(0, K_total, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K_total
        attn_ptrs = attn_ptr + b * attn_bs0 + l * attn_bs1 + offs_k * attn_bs2
        attn_vals = tl.load(attn_ptrs, mask=mask_k, other=0.0).to(tl.float32)
        o_ptrs = o_ptr + n * o_bs0 + offs_k * o_bs1
        o_vals = tl.load(o_ptrs, mask=mask_k, other=0.0).to(tl.float32)
        acc += tl.sum(attn_vals * o_vals, axis=0)
    tl.store(out_ptr + b * out_bs0 + l * out_bs1 + n * out_bs2, acc)

        # 5) Compute attn_output[b, qh, l] = sum_t probs[b, qh, l, t] * V[b, qh, t]
        attn_out = torch.empty((B, num_attention_heads, L), device=device, dtype=torch.float32)

        for qh in range(num_attention_heads):
            # output_matmul_kernel
            output_matmul_kernel[(B, num_attention_heads, L)](
                attn_scores, V, attn_out,
                B, num_attention_heads, L,
                attn_scores.stride(0), attn_scores.stride(1), attn_scores.stride(2), attn_scores.stride(3),
                V.stride(0), V.stride(1), V.stride(2),
                attn_out.stride(0), attn_out.stride(1), attn_out.stride(2),
                BLOCK=128, num_warps=4, num_stages=2
            )

        # 6) Concatenate all heads to produce [B, L, num_attention_heads * head_dim]
        attn_output_concat = attn_out.view(B, L, num_attention_heads * head_dim)

        # 7) Final linear projection to hidden_dim=768 without bias
        final_out = torch.empty((B, L, hidden_dim), device=device, dtype=torch.float32)
        final_linear_kernel[(B, L, hidden_dim)](
            attn_output_concat, o_proj_weight, final_out,
            B, L, num_attention_heads * head_dim, o_proj_weight.shape[1],
            attn_output_concat.stride(0), attn_output_concat.stride(1), attn_output_concat.stride(2),
            o_proj_weight.stride(0), o_proj_weight.stride(1),
            final_out.stride(0), final_out.stride(1), final_out.stride(2),
            BLOCK_K=64, num_warps=4, num_stages=2
        )

        return final_out
```
