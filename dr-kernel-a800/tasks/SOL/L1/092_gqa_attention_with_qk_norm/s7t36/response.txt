import torch
import triton
import triton.language as tl

# 1) Triton linear_bsh: computes out[b, s, h] = sum_k input[b, :, k] * weight[h, k] + bias[h]
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, b_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1  # x is [B, S, K] with strides (x_stride0, x_stride1, x_stride2)
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2  # out is [B, S, H]

    acc = tl.zeros([BLOCK_K], dtype=tl.float32)  # accumulate across K
    for k0 in range(0, K, BLOCK_K):
        kk = k0 + tl.arange(0, BLOCK_K)
        mask_k = kk < K
        x_vec = tl.load(x_ptr + base_x + kk * x_stride2, mask=mask_k, other=0.0)
        w_vec = tl.load(w_ptr + h * w_stride0 + kk * w_stride1, mask=mask_k, other=0.0)
        acc += x_vec * w_vec
    bias = tl.load(b_ptr + h)
    acc = acc + bias
    tl.store(out_ptr + base_out, acc)


# 2) Triton RMSNorm per (b, h) over last dim S: y = x * (weight[h] / sqrt(mean(x^2) + eps))
@triton.jit
def triton_rmsnorm_row(x_ptr, weight_ptr, out_ptr,
                       B, S, H,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       eps: tl.constexpr,
                       BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    row_start = b * S

    sum_sq = 0.0
    for d in range(0, S, BLOCK_S):
        offs = d + tl.arange(0, BLOCK_S)
        mask = offs < S
        x = tl.load(x_ptr + row_start + offs * x_stride1, mask=mask, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = tl.load(weight_ptr + h) * inv_rms

    for d in range(0, S, BLOCK_S):
        offs = d + tl.arange(0, BLOCK_S)
        mask = offs < S
        x = tl.load(x_ptr + row_start + offs * x_stride1, mask=mask, other=0.0)
        y = x * scale
        tl.store(out_ptr + b * out_stride0 + h * out_stride1 + offs * out_stride2, y, mask=mask)


# 3) Triton RoPE per (b, h, s): rotate 128-dim head split into 64+64
# For Q: q_out = q * cos + [-q2, q1] * sin ; For K: same
@triton.jit
def triton_rope_row(x_ptr, cos_ptr, sin_ptr, out_ptr,
                    B, H, S, head_dim,
                    x_stride0, x_stride1, x_stride2,
                    cos_stride0, cos_stride1,
                    sin_stride0, sin_stride1,
                    out_stride0, out_stride1, out_stride2,
                    BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=1.0)
        sin_vec = tl.load(sin_ptr + d, mask=mask, other=0.0)
        q1 = q[:64]
        q2 = q[64:]
        rotated = -q2 + q1
        q_out = q * cos_vec + rotated * sin_vec
        tl.store(out_ptr + base_out + d, q_out, mask=mask)


# 4) Triton GQA expand: expand KV heads from KVH to H using groups
# groups = H // KVH (for config: 96 // 8 = 12). target_h = kh * GROUPS + g
# Copy K[V] row (for each position j) from kh into expanded target_h
@triton.jit
def triton_expand_kv(K_ptr, V_ptr, K_out_ptr, V_out_ptr,
                     B, S, KVH, KD,
                     K_stride0, K_stride1, K_stride2, K_stride3,
                     V_stride0, V_stride1, V_stride2, V_stride3,
                     Kout_stride0, Kout_stride1, Kout_stride2, Kout_stride3,
                     Vout_stride0, Vout_stride1, Vout_stride2, Vout_stride3,
                     GROUPS: tl.constexpr):
    for kh in range(0, KVH):
        for g in range(0, GROUPS):
            h_target = kh * GROUPS + g
            for j in range(0, S):
                k_val = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + kh * K_stride2 + 0 * K_stride3)
                tl.store(K_out_ptr + b * Kout_stride0 + j * Kout_stride1 + h_target * Kout_stride2 + 0 * Kout_stride3, k_val)
                v_val = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + kh * V_stride2 + 0 * V_stride3)
                tl.store(V_out_ptr + b * Vout_stride0 + j * Vout_stride1 + h_target * Vout_stride2 + 0 * Vout_stride3, v_val)


# 5) Triton attention compute per (b, h): tile over i and j positions
# Compute scores[i, j] = Q_i @ K_j^T * scaling, apply causal mask, softmax along j, then accumulate Out_i = scores @ V_i
@triton.jit
def triton_attention_compute(Q_ptr, K_ptr, V_ptr, Out_ptr,
                             B, S, H, head_dim,
                             Q_stride0, Q_stride1, Q_stride2, Q_stride3,
                             K_stride0, K_stride1, K_stride2, K_stride3,
                             V_stride0, V_stride1, V_stride2, V_stride3,
                             Out_stride0, Out_stride1, Out_stride2,
                             scaling: tl.constexpr,
                             BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Accumulator for output row i
    acc = tl.zeros([BLOCK_I], dtype=tl.float32)
    # Loop over j tiles
    for j0 in range(0, S, BLOCK_J):
        j = j0 + tl.arange(0, BLOCK_J)
        mask_j = j < S

        # Build scores[i, j] for i in this tile
        for i0 in range(0, S, BLOCK_I):
            i = i0 + tl.arange(0, BLOCK_I)
            mask_i = i < S

            q = tl.load(Q_ptr + b * Q_stride0 + i * Q_stride1 + h * Q_stride2 + 0 * Q_stride3,
                        mask=mask_i, other=0.0)  # [BLOCK_I]
            k = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + h * K_stride2 + 0 * K_stride3,
                        mask=mask_j, other=0.0)  # [BLOCK_J]
            # scores = q[:, None] * k[None, :]
            scores = tl.sum(q[:, None] * k[None, :], axis=0) * scaling  # scalar per (i0,j0)
            # Causal mask: mask where j > i
            # Build simple mask: keep when j <= i
            causal_mask = (j[None, :] <= i[:, None])
            # For positions beyond S, ensure -inf
            scores = tl.where(causal_mask, scores, -float('inf'))

            # Softmax along j
            exp_scores = tl.exp(scores - tl.max(scores, axis=1, keepdims=True))  # [BLOCK_I, 1]
            sum_exp = tl.sum(exp_scores, axis=1, keepdims=True)  # [BLOCK_I, 1]
            exp_scores = exp_scores / sum_exp  # [BLOCK_I, 1]

            # Accumulate output: acc[i] += sum_j exp_scores[i, j] * V[b, j, h]
            v = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + h * V_stride2 + 0 * V_stride3,
                        mask=mask_j, other=0.0)  # [BLOCK_J]
            acc += tl.sum(exp_scores * v[None, :], axis=1)  # [BLOCK_I]

        # Store accumulated output for i in this tile
        for t in range(0, BLOCK_I):
            i_idx = i0 + t
            if i_idx < S:
                tl.store(Out_ptr + b * Out_stride0 + i_idx * Out_stride1 + h * Out_stride2,
                         acc[t])



# 6) Triton output projection: out[b, s, h] = sum_k attn_output[b, s, k] * o_proj_weight[h, k]
@triton.jit
def triton_linear_o_proj(x_ptr, w_ptr, out_ptr,
                         B, S, H, K,
                         x_stride0, x_stride1,
                         w_stride0, w_stride1,
                         out_stride0, out_stride1, out_stride2,
                         BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros([BLOCK_K], dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        kk = k0 + tl.arange(0, BLOCK_K)
        mask_k = kk < K
        x_vec = tl.load(x_ptr + base_x + kk, mask=mask_k, other=0.0)  # x_ptr strides (x_stride0, x_stride1), last dim is contiguous
        w_vec = tl.load(w_ptr + h * w_stride0 + kk * w_stride1, mask=mask_k, other=0.0)
        acc += x_vec * w_vec
    tl.store(out_ptr + base_out, acc)

            sin.stride(0), sin.stride(1),
            key_rot.stride(0), key_rot.stride(1), key_rot.stride(2),
            BLOCK_D=64,
        )

        # 4) GQA expansion: expand KVH to H with groups=12 (grid over B, KVH, groups, S)
        key_expanded = torch.empty((B, S, H), device=hidden_c.device, dtype=torch.float32)

        output = torch.empty((B_out, S_out, H_out * KD), device=hidden_c.device, dtype=torch.float32)

            o_proj_weight.stride(0), o_proj_weight.stride(1),
            output.stride(0), output.stride(1), output.stride(2),
            BLOCK_K=64,
        )
        return output

                       w_stride0, w_stride1,

def triton_rmsnorm_row(x_ptr, weight_ptr, out_ptr,
                       B, S, H,
                       x_stride0, x_stride1, x_stride2,

    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=1.0)
        sin_vec = tl.load(sin_ptr + d, mask=mask, other=0.0)
        q1 = q[:64]
        q2 = q[64:]
        rotated = -q2 + q1
        q_out = q * cos_vec + rotated * sin_vec
        tl.store(out_ptr + base_out + d, q_out, mask=mask)


# 4) Triton GQA expand: expand KV heads from KVH to H using groups
# groups = H // KVH (for config: 96 // 8 = 12). target_h = kh * GROUPS + g
# Copy K[V] row (for each position j) from kh into expanded target_h
@triton.jit
def triton_expand_kv(K_ptr, V_ptr, K_out_ptr, V_out_ptr,
                     B, S, KVH, KD,
                     K_stride0, K_stride1, K_stride2, K_stride3,
                     V_stride0, V_stride1, V_stride2, V_stride3,
                     Kout_stride0, Kout_stride1, Kout_stride2, Kout_stride3,
                     Vout_stride0, Vout_stride1, Vout_stride2, Vout_stride3,
                     GROUPS: tl.constexpr):
    for kh in range(0, KVH):
        for g in range(0, GROUPS):
            h_target = kh * GROUPS + g
            for j in range(0, S):
                k_val = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + kh * K_stride2 + 0 * K_stride3)
                tl.store(K_out_ptr + b * Kout_stride0 + j * Kout_stride1 + h_target * Kout_stride2 + 0 * Kout_stride3, k_val)
                v_val = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + kh * V_stride2 + 0 * V_stride3)
                tl.store(V_out_ptr + b * Vout_stride0 + j * Vout_stride1 + h_target * Vout_stride2 + 0 * Vout_stride3, v_val)


# 5) Triton attention compute per (b, h): tile over i positions, loop over j tiles
# Compute Out[b, s, h] by iterating j in BLOCK_J chunks. For each j-chunk:
#   - scores[b, s, h] = sum over j_chunk of (exp(soft) * V[b, j, h]) with causal mask j>=s.
@triton.jit
def triton_attention_compute(out_ptr, Q_ptr, K_ptr, V_ptr,
                             B, S, H, head_dim,
                             Out_stride0, Out_stride1, Out_stride2,
                             Q_stride0, Q_stride1, Q_stride2, Q_stride3,
                             K_stride0, K_stride1, K_stride2, K_stride3,
                             V_stride0, V_stride1, V_stride2, V_stride3,
                             scaling: tl.constexpr,
                             BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Accumulator for output row s
    acc = tl.zeros([BLOCK_I], dtype=tl.float32)

    for j0 in range(0, S, BLOCK_J):
        j = j0 + tl.arange(0, BLOCK_J)
        mask_j = j < S

        # We compute attention for each i in tiles
        for i0 in range(0, S, BLOCK_I):
            i = i0 + tl.arange(0, BLOCK_I)
            mask_i = i < S

            q = tl.load(Q_ptr + b * Q_stride0 + i * Q_stride1 + h * Q_stride2 + 0 * Q_stride3,
                        mask=mask_i, other=0.0)  # [BLOCK_I]
            k = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + h * K_stride2 + 0 * K_stride3,
                        mask=mask_j, other=0.0)  # [BLOCK_J]
            # scores = q[:, None] * k[None, :] * scaling
            scores = tl.sum(q[:, None] * k[None, :], axis=0) * scaling  # [BLOCK_I, BLOCK_J]

            # Causal mask: j >= i
            causal_mask = (j[None, :] >= i[:, None])
            scores = tl.where(causal_mask, scores, -float('inf'))

            # Softmax along j dimension
            max_scores = tl.max(scores, axis=1, keepdims=True)  # [BLOCK_I, 1]
            exp_scores = tl.exp(scores - max_scores)
            sum_exp = tl.sum(exp_scores, axis=1, keepdims=True)  # [BLOCK_I, 1]
            soft = exp_scores / sum_exp  # [BLOCK_I, BLOCK_J]

            # Accumulate output: Out[b, i, h] += sum_j soft[b, i, j] * V[b, j, h]
            v = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + h * V_stride2 + 0 * V_stride3,
                        mask=mask_j, other=0.0)  # [BLOCK_J]
            acc += tl.sum(soft * v[None, :], axis=1)  # [BLOCK_I]

        # Store accumulated output for i in this tile
        for t in range(0, BLOCK_I):
            i_idx = i0 + t
            if i_idx < S:
                tl.store(out_ptr + b * Out_stride0 + i_idx * Out_stride1 + h * Out_stride2,
                         acc[t])


# 6) Triton output projection: out[b, s, h] = sum_k attn_output[b, s, k] * o_proj_weight[h, k]
@triton.jit
def triton_linear_o_proj(x_ptr, w_ptr, out_ptr,
                         B, S, H, K,
                         x_stride0, x_stride1,
                         w_stride0, w_stride1,
                         out_stride0, out_stride1, out_stride2,
                         BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros([BLOCK_K], dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        kk = k0 + tl.arange(0, BLOCK_K)
        mask_k = kk < K
        x_vec = tl.load(x_ptr + base_x + kk, mask=mask_k, other=0.0)
        w_vec = tl.load(w_ptr + h * w_stride0 + kk * w_stride1, mask=mask_k, other=0.0)
        acc += x_vec * w_vec
    tl.store(out_ptr + base_out, acc)
