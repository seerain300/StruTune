# 1) Linear projection: y[b, l, n] = sum_k x[b, l, k] * w[n, k], accumulate in f32, output f32
@triton.jit
def linear_proj_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H_in]
    w_ptr,           # *f16/f32/bf16, [N_out, H_in]
    y_ptr,           # *f32, [B, L, N_out]
    B, L, H_in, N_out,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    # Each program computes y[b, l, n]
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_in, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_in
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)
        acc += tl.sum(x_vals.to(tl.float32) * w_vals.to(tl.float32), axis=0)
    tl.store(y_ptr + b * y_bs0 + l * y_bs1 + n * y_bs2, acc)


# 2) RMSNorm for a vector of length H: y = x * rsqrt(mean(x^2) + eps) * weight
@triton.jit
def rmsnorm_kernel(
    x_ptr,        # *f32, [B, L, H]
    w_ptr,        # *f32, [H]
    out_ptr,      # *f32, [B, L, H]
    B, L, H, eps,
    x_bs0, x_bs1, x_bs2,
    w_bs0,
    out_bs0, out_bs1, out_bs2,
    BLOCK: tl.constexpr,
):
    # Each program handles one (b, l, h)
    b = tl.program_id(0)
    l = tl.program_id(1)
    h = tl.program_id(2)

    # Compute variance over h
    sum_sq = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H, BLOCK):
        offs = k0 + tl.arange(0, BLOCK)
        mask = offs < H
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs * x_bs2
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
        sum_sq += tl.sum(x_vals * x_vals, axis=0)

    mean = sum_sq / H
    inv_rms = tl.rsqrt(mean + eps)
    # Scale and apply weight
    for k0 in range(0, H, BLOCK):
        offs = k0 + tl.arange(0, BLOCK)
        mask = offs < H
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs * x_bs2
        w_ptrs = w_ptr + offs * w_bs0
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask, other=1.0)  # weight may be 1 for no scale if not provided; here it's provided
        y_vals = x_vals * inv_rms * w_vals
        out_ptrs = out_ptr + b * out_bs0 + l * out_bs1 + offs * out_bs2
        tl.store(out_ptrs, y_vals)

    out_bs0, out_bs1, out_bs2,
):

    l = tl.program_id(1)
    h = tl.program_id(2)  # h in [0..127]
    # Load x[b, l, h]
    x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + h * x_bs2
    x_val = tl.load(x_ptrs)

    # Determine h1 or h2
    if h < 64:
        cos_val = tl.load(cos_ptr + l * 64 + h)  # cos[l, h]
        y_val = x_val * cos_val
    else:
        sin_index = h - 64
        sin_val = tl.load(sin_ptr + l * 64 + sin_index)  # sin[l, h-64]
        y_val = -x_val * sin_val

    out_ptrs = out_ptr + b * out_bs0 + l * out_bs1 + h * out_bs2
    tl.store(out_ptrs, y_val)

# 5) Softmax with causal mask (per row): stable softmax over t in [0..L-1], mask t < l -> -inf
# Triton does not support torch.triu in kernel, so we implement mask and softmax manually.
# We need to apply mask per (b, qh, l). We will implement a kernel that loads attn_scores row, adds mask, computes softmax, and writes back. However, the attn_scores tensor is [B, HEADS, L, L], and Triton kernels have fixed grid. We'll implement per-(b, qh, l) row softmax over L.

@triton.jit
def softmax_mask_kernel(
    scores_ptr, out_ptr,
    B, HEADS, L,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
    out_bs0, out_bs1, out_bs2, out_bs3,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)

    # Compute row pointer: scores[b, qh, l, :]
    row_base = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2
    row = tl.arange(0, L)
    scores_row_ptrs = row_base + row * scores_bs3
    scores_row = tl.load(scores_row_ptrs)

    # Build causal mask: m[t] = -inf if t < l else 0
    mask = tl.where(row < l, -float('inf'), 0.0)
    scores_row = scores_row + mask

    # Stable softmax: subtract max, exp, sum, normalize
    max_score = tl.max(scores_row, axis=0)
    scores_row = scores_row - max_score
    exp_scores = tl.exp(scores_row)
    sum_exp = tl.sum(exp_scores, axis=0)
    softmax_row = exp_scores / sum_exp

    out_row_ptrs = out_ptr + b * out_bs0 + qh * out_bs1 + l * out_bs2 + row * out_bs3
    tl.store(out_row_ptrs, softmax_row)


# 6) Output matmul: out[b, qh, l] = sum_t scores[b, qh, l, t] * V[b, qh, t]
# Implement per (b, qh, l) as a Triton kernel looping over t. Triton does not support dynamic loop over L here; we will define BLOCK_T and iterate.

@triton.jit
def output_matmul_kernel(
    scores_ptr, V_ptr, out_ptr,
    B, HEADS, L,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
    V_bs0, V_bs1, V_bs2,
    out_bs0, out_bs1, out_bs2,
    BLOCK_T: tl.constexpr,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)

    # Initialize accumulator
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over t in chunks
    for t0 in range(0, L, BLOCK_T):
        offs_t = t0 + tl.arange(0, BLOCK_T)
        mask_t = offs_t < L

        # Load scores row segment: scores[b, qh, l, offs_t]
        row_base = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2
        scores_ptrs = row_base + offs_t * scores_bs3
        scores_vals = tl.load(scores_ptrs, mask=mask_t, other=0.0)

        # Load V segment: V[b, qh, offs_t]
        V_ptrs = V_ptr + b * V_bs0 + qh * V_bs1 + offs_t * V_bs2
        V_vals = tl.load(V_ptrs, mask=mask_t, other=0.0)

        acc += tl.sum(scores_vals * V_vals, axis=0)

    # Store result to out[b, qh, l]
    out_ptr_row = out_ptr + b * out_bs0 + qh * out_bs1 + l * out_bs2
    tl.store(out_ptr_row, acc)


# 7) Final linear projection: y[b, l, n] = sum_k x[b, l, k] * w[n, k] (no bias), x shape [B, L, 12288], w shape [hidden_dim, 12288]
@triton.jit
def final_linear_kernel(
    x_ptr, w_ptr, y_ptr,
    B, L, H_in, N_out,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_in, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_in
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)
        acc += tl.sum(x_vals.to(tl.float32) * w_vals.to(tl.float32), axis=0)
    tl.store(y_ptr + b * y_bs0 + l * y_bs1 + n * y_bs2, acc)
