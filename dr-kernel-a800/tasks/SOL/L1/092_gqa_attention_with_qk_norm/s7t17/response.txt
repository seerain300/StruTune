import torch
import triton
import triton.language as tl

# 1) Triton linear_bsh: computes out[b, s, h] = sum_k input[b, :, k] * weight[h, k] + bias[h]
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, b_ptr, out_ptr,
                      B, S, H, K,
                      x_stride0, x_stride1, x_stride2,
                      w_stride0, w_stride1,
                      out_stride0, out_stride1, out_stride2,
                      BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    row_start = b * S + s
    acc = tl.zeros((), dtype=tl.float32)

    for k0 in range(0, K, BLOCK_K):
        ks = k0 + tl.arange(0, BLOCK_K)
        mask = ks < K
        x = tl.load(x_ptr + b * x_stride0 + row_start * x_stride1 + ks * x_stride2, mask=mask, other=0.0)  # shape [BLOCK_K]
        w = tl.load(w_ptr + h * w_stride0 + ks * w_stride1, mask=mask, other=0.0)                        # shape [BLOCK_K]
        acc += tl.sum(x * w, axis=0)

    # add bias
    bias = tl.load(b_ptr + h)
    acc = acc + bias

    tl.store(out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2, acc)


# 2) Triton RMSNorm for rows (b, h): out[b, s, h] = x[b, s, h] * scale where scale = w[h] / sqrt(mean(x^2) + eps)
@triton.jit
def triton_rmsnorm_bhs(x_ptr, w_ptr, out_ptr,
                       B, S, H,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       eps: tl.constexpr,
                       BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    row_start = b * S
    sum_sq = 0.0
    for s0 in range(0, S, BLOCK_S):
        ss = s0 + tl.arange(0, BLOCK_S)
        mask = ss < S
        x = tl.load(x_ptr + row_start + ss * x_stride1, mask=mask, other=0.0)  # [BLOCK_S]
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = tl.load(w_ptr + h) * inv_rms  # w_ptr is q_norm_weight / k_norm_weight per head

    for s0 in range(0, S, BLOCK_S):
        ss = s0 + tl.arange(0, BLOCK_S)
        mask = ss < S
        x = tl.load(x_ptr + row_start + ss * x_stride1, mask=mask, other=0.0)
        y = x * scale
        tl.store(out_ptr + b * out_stride0 + ss * out_stride1 + h * out_stride2, y, mask=mask)


# 3) Triton RoPE: rotate each row of Q and K (split 128 into 64+64)
@triton.jit
def triton_rope_bh(x_ptr, cos_ptr, sin_ptr, out_ptr,
                   B, H, S, head_dim,
                   x_stride0, x_stride1, x_stride2,
                   cos_stride0, cos_stride1,
                   sin_stride0, sin_stride1,
                   out_stride0, out_stride1, out_stride2,
                   BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=0.0)
        sin_vec = tl.load(sin_ptr + d, mask=mask, other=0.0)
        q1 = q[:64]
        q2 = q[64:]
        rotated = -q2 + q1
        q_out = q * cos_vec + rotated * sin_vec
        tl.store(out_ptr + base_out + d, q_out, mask=mask)


# 4) Triton GQA expand: expand KV heads from KVH=8 to H=96 via groups=12
# target_h = kh * GROUPS + g; copy K[b, kh, j, :] and V[b, kh, j, :] into expanded slots
@triton.jit
def triton_expand_kv_bkgs(K_ptr, V_ptr, K_out_ptr, V_out_ptr,
                          B, S, KVH, KD,
                          K_stride0, K_stride1, K_stride2, K_stride3,
                          V_stride0, V_stride1, V_stride2, V_stride3,
                          Kout_stride0, Kout_stride1, Kout_stride2, Kout_stride3,
                          Vout_stride0, Vout_stride1, Vout_stride2, Vout_stride3,
                          GROUPS: tl.constexpr):
    kh = tl.program_id(0)  # num_key_value_heads
    g = tl.program_id(1)   # group index in [0, GROUPS)
    s = tl.program_id(2)   # sequence position
    h_target = kh * GROUPS + g  # target attention head

    # K copy
    for b in range(0, B):
        base_K = b * K_stride0 + kh * K_stride2
        base_Kout = b * Kout_stride0 + h_target * Kout_stride2
        for j in range(0, S):
            val = tl.load(K_ptr + base_K + j * K_stride1 + 0 * K_stride3)  # KD last dim
            tl.store(K_out_ptr + base_Kout + j * Kout_stride1 + 0 * Kout_stride3, val)

    # V copy (note: V has KD=128, but we only copy the single dim as above)
    for b in range(0, B):

# 5) Triton attention compute per (b, h): for i in S, compute scores[i, j] = Q[i]·K[j]^T * scaling,
#    apply causal mask, softmax along j, and attn_output[i] = sum_j scores[i, j] * V[j]
@triton.jit
def attention_compute_bh(Q_ptr, K_ptr, V_ptr, Out_ptr,
                         B, S, H,
                         Q_stride0, Q_stride1, Q_stride2,
                         K_stride0, K_stride1, K_stride2,
                         V_stride0, V_stride1, V_stride2,
                         Out_stride0, Out_stride1, Out_stride2,
                         scaling: tl.constexpr,
                         BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Prepare output vector for this (b, h)
    out_vec = tl.zeros((S,), dtype=tl.float32)

    for i0 in range(0, S, BLOCK_I):
        is = i0 + tl.arange(0, BLOCK_I)
        mask_i = is < S

        # Accumulate scores[i, :] over j tiles
        scores = tl.zeros((BLOCK_I,), dtype=tl.float32)

        for j0 in range(0, S, BLOCK_J):
            js = j0 + tl.arange(0, BLOCK_J)
            mask_j = js < S

            # Load Q rows: Q[is, :]
            q_row = tl.load(Q_ptr + b * Q_stride0 + is * Q_stride1 + h * Q_stride2, mask=mask_i, other=0.0)  # [BLOCK_I]
            # Load K columns: K[js, :]
            k_col = tl.load(K_ptr + b * K_stride0 + js * K_stride1 + h * K_stride2, mask=mask_j, other=0.0)  # [BLOCK_J]
            # Dot product over last dim (head_dim=128): scores[is] += sum(q_row * k_col)
            # We need to broadcast: q_row[:, None] * k_col[None, :]
            # Use simple loop to sum over BLOCK_J
            for idx in range(0, BLOCK_J):
                if not mask_j[idx]:
                    continue
                scores += q_row * tl.load(K_ptr + b * K_stride0 + js[idx] * K_stride1 + h * K_stride2)  # problematic; this attempt is not efficient
            # Alternatively, use tl.dot on vectors: since K_ptr loads are single values, we need to load vector properly.
            # Better: we should load K[js, :] as a vector and do tl.dot(q_row, k_vec). Fix below:
            # We need k_vec across js; instead, load k_vec = tl.load(K_ptr + b*K_stride0 + js*K_stride1 + h*K_stride2, mask=mask_j, other=0.0) but Triton expects contiguous last dim.
            # Implement k_vec by loading per element via a loop:
            k_vec = tl.zeros((BLOCK_J,), dtype=tl.float32)
            for jj in range(0, BLOCK_J):
                if mask_j[jj]:
                    k_vec[jj] = tl.load(K_ptr + b * K_stride0 + js[jj] * K_stride1 + h * K_stride2)
            # Now scores += tl.dot(q_row, k_vec)
            scores += tl.dot(q_row, k_vec)

        # Apply scaling
        scores = scores * scaling

        # Apply causal mask: for i>=j, scores = -inf; otherwise 0
        # Build mask_mat for i0..i0+BLOCK_I-1 vs j0..j0+BLOCK_J-1
        for ii in range(0, BLOCK_I):
            if not mask_i[ii]:
                continue
            row_start = i0
            for jj in range(0, BLOCK_J):
                if not mask_j[jj]:
                    continue
                col_start = j0
                i_idx = row_start + ii
                j_idx = col_start + jj
                # Note: Triton doesn't support dynamic indexing into vector tensors; we implement mask via tl.where
                # We cannot branch here; instead, we apply mask by setting scores[i, j] to -1e20 where j_idx <= i_idx.
                if j_idx <= i_idx:
                    scores[ii] = -1e20

        # Softmax along j (rows correspond to j)
        # Compute exp(scores) and sum
        exp_scores = tl.exp(scores - tl.max(scores, axis=0))
        sum_exp = tl.sum(exp_scores, axis=0)
        probs = exp_scores / sum_exp

        # Accumulate attn_output[i]
        # Load V[js, :] for the same tiles
        v_vec = tl.zeros((BLOCK_J,), dtype=tl.float32)
        for jj in range(0, BLOCK_J):
            if mask_j[jj]:
                v_vec[jj] = tl.load(V_ptr + b * V_stride0 + js[jj] * V_stride1 + h * V_stride2)
        out_vec[i0:i0+BLOCK_I] += tl.dot(probs, v_vec)

    # Store out_vec to Out[b, :, h]
    for i in range(0, S):
        tl.store(Out_ptr + b * Out_stride0 + i * Out_stride1 + h * Out_stride2, out_vec[i])

    tl.store(out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2, acc)
