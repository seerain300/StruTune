    inv_sqrt_hd = 1.0 / tl.sqrt(128.0)  # constant

    out_vec = tl.zeros([S], dtype=tl.float32)

    # Loop over i in tiles
    for i0 in range(0, S, BLOCK_S):
        i = i0 + tl.arange(0, BLOCK_S)
        mask_i = i < S

        # Load Q_out[i, h] vector
        q_vec = tl.zeros([BLOCK_S], dtype=tl.float32)
        for d in range(0, 128, 64):
            offs_d = d + tl.arange(0, 64)
            mask_d = offs_d < 128
            # Construct 2D mask combining i and d
            # We need to load q[i, d] per d
            for ii in range(0, BLOCK_S):
                # Scalar load for each i
                q_val = tl.load(QOut_ptr + b * QOut_stride0 + i[ii] * QOut_stride1 + h * QOut_stride2 + offs_d, mask=mask_d, other=0.0)
                q_vec[ii] = q_val[0]  # Not applicable; we'll load as a scalar loop below.

        # Simpler scalar loop per i to load q[i, d] fully
        for ii in range(0, BLOCK_S):
            if mask_i[ii]:
                for d0 in range(0, 128, 64):
                    offs_d = d0 + tl.arange(0, 64)
                    mask_d = offs_d < 128
                    q_val = tl.load(QOut_ptr + b * QOut_stride0 + i[ii] * QOut_stride1 + h * QOut_stride2 + offs_d, mask=mask_d, other=0.0)
                    # q_vec[ii] = q_val  # Triton doesn't allow vector assignment here; keep as scalar accumulator

        # Compute scores[i, j] for j in tiles
        for j0 in range(0, S, BLOCK_S):
            j = j0 + tl.arange(0, BLOCK_S)
            mask_j = j < S
            # scores[i, j] = sum_k q[i, k] * k[j, k] * inv_sqrt_hd
            scores = tl.zeros([BLOCK_S, BLOCK_S], dtype=tl.float32)
            for d0 in range(0, 128, 64):
                d = d0 + tl.arange(0, 64)
                mask_d = d < 128
                for ii in range(0, BLOCK_S):
                    if mask_i[ii]:
                        qk = 0.0
                        for di in range(0, 64):
                            # scalar load q[i, di]
                            qk += tl.load(QOut_ptr + b * QOut_stride0 + i[ii] * QOut_stride1 + h * QOut_stride2 + d0 + di)
                            # scalar load k[j, di]
                            for jj in range(0, BLOCK_S):
                                if mask_j[jj]:
                                    k_val = tl.load(KOutR_ptr + b * KOutR_stride0 + j[jj] * KOutR_stride1 + h * KOutR_stride2 + d0 + di)
                                    qk += q_val * k_val  # q_val loaded from q_vec[ii]
                                    scores[ii, jj] = qk * inv_sqrt_hd
            # Apply causal mask: for j > i, scores[i, j] = -inf
            # Note: scores is 2D; implement causal mask elementwise
            for ii in range(0, BLOCK_S):
                if mask_i[ii]:
                    for jj in range(0, BLOCK_S):
                        if mask_j[jj] and j[jj] > (i[ii]):
                            scores[ii, jj] = -1e20

            # Softmax along j axis
            for ii in range(0, BLOCK_S):
                if mask_i[ii]:
                    exp_row = tl.exp(scores[ii, :])
                    sum_row = tl.sum(exp_row, axis=0)
                    soft_row = exp_row / sum_row
                    for jj in range(0, BLOCK_S):
                        if mask_j[jj]:
                            out_vec[i[ii]] += soft_row[jj] * tl.load(VOut_ptr + b * VOut_stride0 + j[jj] * VOut_stride1 + h * VOut_stride2)

    # Store output vector
    for i in range(0, S):
        tl.store(Output_ptr + b * Output_stride0 + i * Output_stride1 + h * Output_stride2, out_vec[i])
