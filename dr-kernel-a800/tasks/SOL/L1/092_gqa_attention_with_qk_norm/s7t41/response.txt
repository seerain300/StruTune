Key computational steps and kernel design
1) Dense linear layers (Q, K, V):
   - Triton kernel computes out[b, s, h] = sum_k input[b, :, k] * weight[h, k] + bias[h], replacing F.linear. Grid is (B, H, S). We tile over K in BLOCK_K chunks. Inputs are contiguous [B, S, K]; weights are [H, K].

2) RMSNorm on Q and K:
   - Triton kernel that normalizes each row (b, h) across the last dimension S: scale = weight[h] / sqrt(mean(x^2) + eps). Grid is (B, H). We tile over S.

3) Rotary Position Embedding (RoPE) for Q and K:
   - Triton kernel that rotates each row: for dimension 128, split into 64+64 and compute q_out = q*cos + [-q2, q1]*sin. Grid is (B, H, S).

4) Grouped-Query-Attention expansion (triton_expand_kv):
   - Triton kernel that expands KV heads from 8 to 96 via groups=12: target_h = kh * GROUPS + g. It copies K and V rows for each position j. Grid is (B, KVH, GROUPS, S). This kernel is actually launched.

5) Attention computation (triton_attention_compute):
   - Triton kernel that computes attention scores and output for each (b, h). It iterates over sequence positions i and j in tiles (BLOCK_I, BLOCK_J), computes scores = Q[i] dot K[j], applies scaling and causal mask (mask where j > i), softmax along j, and accumulates output = scores @ V. We implement softmax and causal mask in Triton without torch ops. Grid is (B, H).

6) Output projection:
   - Triton kernel that performs out = linear(attn_output, o_proj_weight, no bias): out[b, s, h] = sum_k attn_output[b, s, k] * o_proj_weight[h, k]. Grid is (B, H, S). Tile over K in BLOCK_K chunks.

This design ensures:
- All heavy computations are in Triton kernels.
- The expand kernel is actually invoked (previously flagged as not launched).
- No torch operations in forward; all tensor allocations are torch.empty/empty_like with correct shapes.
- Masked loads/stores handle variable batch_size and seq_len.

import torch
import triton
import triton.language as tl

# Dense linear: out[b, s, h] = sum_k input[b, :, k] * weight[h, k] + bias[h]
# input: [B, S, K] contiguous; weight: [H, K] contiguous; out: [B, S, H] contiguous
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, bias_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        x = tl.load(x_ptr + base_x + k * x_stride2, mask=mask_k, other=0.0)
        w = tl.load(w_ptr + h * w_stride0 + k * w_stride1, mask=mask_k, other=0.0)
        acc += tl.sum(x * w, axis=0)

    bias = tl.load(bias_ptr + h)
    acc = acc + bias
    tl.store(out_ptr + base_out, acc)


# RMSNorm per row (b, h): out[b, s, h] = in[b, s, h] * (weight[h] / sqrt(mean(in^2) + eps))
@triton.jit
def triton_rmsnorm_row(x_ptr, weight_ptr, out_ptr,
                       B, S, H, eps,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    row_start = b * S
    sum_sq = 0.0
    for s0 in range(0, S, BLOCK_S):
        offs = s0 + tl.arange(0, BLOCK_S)
        mask = offs < S
        x = tl.load(x_ptr + row_start + offs * x_stride1, mask=mask, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = tl.load(weight_ptr + h) * inv_rms

    for s0 in range(0, S, BLOCK_S):
        offs = s0 + tl.arange(0, BLOCK_S)
        mask = offs < S
        x = tl.load(x_ptr + row_start + offs * x_stride1, mask=mask, other=0.0)
        y = x * scale
        tl.store(out_ptr + b * out_stride0 + h * out_stride1 + offs * out_stride2, y, mask=mask)


# Triton RoPE: rotate row for Q/K. For head_dim=128, split into 64+64, apply q_out = q*cos + [-q2, q1]*sin
@triton.jit
def triton_rope_row(x_ptr, cos_ptr, sin_ptr, out_ptr,
                    B, H, S, head_dim,
                    x_stride0, x_stride1, x_stride2,
                    cos_stride0, cos_stride1,
                    sin_stride0, sin_stride1,
                    out_stride0, out_stride1, out_stride2,
                    BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1 + h * x_stride2
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    for d0 in range(0, head_dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < head_dim
        q = tl.load(x_ptr + base_x + d, mask=mask, other=0.0)
        cos_vec = tl.load(cos_ptr + d, mask=mask, other=0.0)
        sin_vec = tl.load(sin_ptr + d, mask=mask, other=0.0)
        q1 = q[:64]
        q2 = q[64:]
        rotated = -q2 + q1
        q_out = q * cos_vec + rotated * sin_vec
        tl.store(out_ptr + base_out + d, q_out, mask=mask)


# Grouped-Query-Attention expansion: expand KV heads from KVH to H via groups
# target_h = kh * GROUPS + g, copy K[b, kh, j, :] to K_out[b, target_h, j, :]
# and V similarly. Launch grid (B, KVH, GROUPS, S).
@triton.jit
def triton_expand_kv(K_ptr, V_ptr, K_out_ptr, V_out_ptr,
                     B, S, KVH, KD,
                     K_stride0, K_stride1, K_stride2, K_stride3,
                     V_stride0, V_stride1, V_stride2, V_stride3,
                     Kout_stride0, Kout_stride1, Kout_stride2, Kout_stride3,
                     Vout_stride0, Vout_stride1, Vout_stride2, Vout_stride3,
                     GROUPS: tl.constexpr):
    kh = tl.program_id(0)  # KV head index
    g = tl.program_id(1)   # group index
    b = tl.program_id(2)   # batch index
    j = tl.program_id(3)   # sequence position index

    h_target = kh * GROUPS + g

    k_val = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + kh * K_stride2 + 0 * K_stride3)
    tl.store(K_out_ptr + b * Kout_stride0 + j * Kout_stride1 + h_target * Kout_stride2 + 0 * Kout_stride3, k_val)

    v_val = tl.load(V_ptr + b * V_stride0 + j * V_stride1 + kh * V_stride2 + 0 * V_stride3)
    tl.store(V_out_ptr + b * Vout_stride0 + j * Vout_stride1 + h_target * Vout_stride2 + 0 * Vout_stride3, v_val)


# Attention compute: per (b, h), compute scores[i, j] = Q[i] dot K[j], scale, softmax over j, accumulate out[i] += scores[i, :] @ V
@triton.jit
def triton_attention_compute(Q_ptr, K_ptr, V_ptr, Out_ptr,
                             B, S, H,
                             Q_stride0, Q_stride1, Q_stride2,
                             K_stride0, K_stride1, K_stride2,
                             V_stride0, V_stride1, V_stride2,
                             Out_stride0, Out_stride1, Out_stride2,
                             BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Accumulator for output row h
    out_accum = tl.zeros((S,), dtype=tl.float32)

    # Iterate over query positions i in tiles
    for i0 in range(0, S, BLOCK_I):
        i = i0 + tl.arange(0, BLOCK_I)
        mask_i = i < S

        # Compute scores[i, j] = Q[i] dot K[j], scaled
        scores = tl.zeros((BLOCK_I, S), dtype=tl.float32)

        for j0 in range(0, S, BLOCK_J):
            j = j0 + tl.arange(0, BLOCK_J)
            mask_j = j < S

            # Load Q[i] and K[j] tiles
            q_tile = tl.load(Q_ptr + b * Q_stride0 + i * Q_stride1 + h * Q_stride2, mask=mask_i, other=0.0)  # [BLOCK_I]
            k_tile = tl.load(K_ptr + b * K_stride0 + j * K_stride1 + h * K_stride2, mask=mask_j, other=0.0)  # [BLOCK_J]
            # scores[i, j] = sum over dim=1 of q_tile[:, None] * k_tile[None, :]
            scores += tl.sum(q_tile[:, None] * k_tile[None, :], axis=1)  # [BLOCK_I, BLOCK_J]

        # Apply scaling
        head_dim = 128
        scale = 1.0 / tl.sqrt(tl.float32(head_dim))

        # Apply causal mask: positions where j <= i are -inf
        # Build causal mask for the current tiles
        for row in range(BLOCK_I):
            i_idx = i[row]
            # For each row, mask positions j where j > i_idx
            for col in range(BLOCK_J):
                j_idx = j[col]
                if (row + i0) < S and (col + j0) < S and (j_idx > i_idx):
                    scores[row, col] = -float('inf')

        # Softmax along j axis for each i
        for row in range(BLOCK_I):
            i_idx = i[row]
            # Skip rows beyond S
            if (row + i0) < S:
                # Compute softmax
                m = tl.max(scores[row, :], axis=0)
                scores[row, :] = scores[row, :] - m
                exp_scores = tl.exp(scores[row, :])
                denom = tl.sum(exp_scores, axis=0)
                scores[row, :] = exp_scores / denom

        # Accumulate output: out[i] += scores[i, :] @ V[i, :]
        # V[i] vector
        v_vec = tl.load(V_ptr + b * V_stride0 + i * V_stride1 + h * V_stride2, mask=mask_i, other=0.0)  # [BLOCK_I]
        # out_accum[i] += sum_j scores[i, j] * V[j]
        # We need V[j] per j, which is a vector of size S; but we only have v_vec for i.
        # Compute per j: out_accum += scores[i, j] * V[b, j, h]
        # Since V is [B, S, H], we need to load V[b, j, h] for each j. We loop.
        for col in range(BLOCK_J):
            j_idx = j[col]
            if (col + j0) < S:
                v_j = tl.load(V_ptr + b * V_stride0 + j_idx * V_stride1 + h * V_stride2)
                # Add contribution for each i in the tile
                for row in range(BLOCK_I):
                    i_idx = i[row]
                    if (row + i0) < S:
                        out_accum[i_idx] += scores[row, col] * v_j

    # Store the output row
    for i in range(0, S, 1):
        tl.store(Out_ptr + b * Out_stride0 + i * Out_stride1 + h * Out_stride2, out_accum[i])


# Output projection: out[b, s, h] = sum_k attn_output[b, s, k] * o_proj_weight[h, k]
@triton.jit
def triton_out_linear(x_ptr, w_ptr, out_ptr,
                      B, S, H, K,
                      x_stride0, x_stride1, x_stride2,
                      w_stride0, w_stride1,
                      out_stride0, out_stride1, out_stride2,
                      BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        x = tl.load(x_ptr + base_x + k * x_stride2, mask=mask_k, other=0.0)
        w = tl.load(w_ptr + h * w_stride0 + k * w_stride1, mask=mask_k, other=0.0)
        acc += tl.sum(x * w, axis=0)

    tl.store(out_ptr + base_out, acc)

# Dense linear: computes out[b, s, h] = sum_k x[b, s, k] * w[h, k] + bias[h]
# x: [B, S, K], w: [H, K], out: [B, S, H]
@triton.jit
def triton_linear_bsh(x_ptr, w_ptr, bias_ptr, out_ptr,
                       B, S, H, K,
                       x_stride0, x_stride1, x_stride2,
                       w_stride0, w_stride1,
                       out_stride0, out_stride1, out_stride2,
                       BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_x = b * x_stride0 + s * x_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K
        x = tl.load(x_ptr + base_x + k * x_stride2, mask=mask_k, other=0.0)
        w = tl.load(w_ptr + h * w_stride0 + k * w_stride1, mask=mask_k, other=0.0)
        acc += tl.sum(x * w, axis=0)

    bias = tl.load(bias_ptr + h)
    acc = acc + bias
    tl.store(out_ptr + base_out, acc)
