全面的诊断与修复

- 问题根源：
  - 上次提交中有 kernel 定义但没有被调用（decoy），评分会认为未满足 Triton-only，并标记为错误。
  - 有 kernel 的调用存在 grid/bounds/mask 问题，导致 runtime errors。例如 attention score matmul kernel 循环不正确，softmax mask kernel 的指针算法规则错误等。
  - 一些 kernel 没有被正确 launch 或者 launch 参数不对，导致 runtime error 和 correctness fail。

- 修复策略：
  - 严格只提供被 forward 调用的 Triton kernel，并且确保每个 kernel 都有正确的 grid 和参数。
  - 重点实现并调用以下 Triton kernels：
    - Linear projection (Q, K, V) 三个 kernel
    - RMSNorm for Q and K
    - Q/K rotation (RoPE): cosine/sine rotation with split
    - 注意力分数计算 matmul kernel：compute attn_scores[b, qh, l, t] = Q[b, qh, l] * K[b, qh, t]
    - Softmax with causal mask in Triton: per-row softmax with mask m[t] = 0 if t >= l else -inf
    - Output matmul kernel: attn_output[b, qh, l] = sum_t attn_scores_masked[b, qh, l, t] * V[b, qh, t]
    - Final linear projection: final_out[b, l, :] = attn_output_flat[b, l, :] @ o_proj_weight^T (no bias)
  - forward 仅进行 reshape/view/contiguous/empty/empty_like 和 Triton kernel 的 launch，不做任何 torch 的 compute。

- 特别注意：
  - Softmax必须是行-wise，dim=-1 of [num_heads, L, L] per (b), 并且上三角 (t < l) 应该被屏蔽为 -inf，在 kernel 内实现稳定性 (subtract max, exp, sum)。
  - output matmul kernel 从 attn_scores_masked[b, qh, l, t] 和 V[b, qh, t] 计算 output[b, qh, l]，没有 bias。
  - Q/K rotation kernel 实现 h1[:64] 乘 cos, h2[64:] 乘 sin(- for K, + for Q), 并拼接回 head_dim=128。

- 性能：
  - 本文不追求最高性能，但保证 Triton kernels 被正确调用并完成工作，确保正确性通过后再考虑优化。使用适当的 BLOCK_K/BLOCK_T 和 num_warps 设置来避免超时。实际注意力分数 matmul 和 softmax 的 Triton 实现会比 torch fast，但首要任务是确保正确性和合规性。

@triton.jit
def linear_proj_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H_in]
    w_ptr,           # *f16/f32/bf16, [N_out, H_in]
    bias_ptr,        # *f32, [N_out] (ignored if None)
    y_ptr,           # *f32, [B, L, N_out]
    B, L, H_in, N_out,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_in, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_in

        # x[b, l, offs_k]
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0).to(tl.float32)

        # w[n, offs_k]
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0).to(tl.float32)

        acc += tl.sum(x_vals[:, None] * w_vals[None, :], axis=0)

    # store y[b, l, n]
    y_ptrs = y_ptr + b * y_bs0 + l * y_bs1 + n * y_bs2

# Proper masked softmax kernel with l as runtime scalar:
@triton.jit
def softmax_masked_kernel(
    attn_ptr,        # *f32, [B, heads, L, L] pre-masked
    out_ptr,         # *f32, [B, heads, L, L]
    B, heads, L, l,  # l is scalar int
    attn_bs0, attn_bs1, attn_bs2, attn_bs3,
    out_bs0, out_bs1, out_bs2, out_bs3,
    BLOCK_T: tl.constexpr,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    # l is provided as scalar

    # Compute row max over t >= l only; else set to -inf
    max_val = -float('inf')
    for t0 in range(0, L, BLOCK_T):
        offs_t = t0 + tl.arange(0, BLOCK_T)
        mask_t = (offs_t < L) & (offs_t >= l)
        attn_ptrs = attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2 + offs_t * attn_bs3
        scores = tl.load(attn_ptrs, mask=mask_t, other=-float('inf'))
        tile_max = tl.max(scores, axis=0)
        max_val = tl.maximum(max_val, tile_max)

    sum_exp = 0.0
    for t0 in range(0, L, BLOCK_T):
        offs_t = t0 + tl.arange(0, BLOCK_T)
        mask_t = (offs_t < L) & (offs_t >= l)
        attn_ptrs = attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2 + offs_t * attn_bs3
        scores = tl.load(attn_ptrs, mask=mask_t, other=-float('inf'))
        exps = tl.exp(scores - max_val)
        sum_exp += tl.sum(exps, axis=0)

    for t0 in range(0, L, BLOCK_T):
        offs_t = t0 + tl.arange(0, BLOCK_T)
        mask_t = (offs_t < L) & (offs_t >= l)
        attn_ptrs = attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2 + offs_t * attn_bs3
        scores = tl.load(attn_ptrs, mask=mask_t, other=-float('inf'))
        exps = tl.exp(scores - max_val) / sum_exp
        out_ptrs = out_ptr + b * out_bs0 + qh * out_bs1 + l * out_bs2 + offs_t * out_bs3
        tl.store(out_ptrs, exps, mask=mask_t)


# Implement output matmul kernel:
@triton.jit
def output_matmul_kernel(
    attn_ptr,        # *f32, [B, heads, L, L] softmaxed
    V_ptr,           # *f32, [B, heads, L, H] (same H=128)
    out_ptr,         # *f32, [B, heads, L, 1] (we store per row; more realistically a 2D [B, heads, L])
    B, heads, L, H,
    attn_bs0, attn_bs1, attn_bs2, attn_bs3,
    V_bs0, V_bs1, V_bs2, V_bs3,
    out_bs0, out_bs1, out_bs2, out_bs3,
    BLOCK_T: tl.constexpr,
):
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for t0 in range(0, L, BLOCK_T):
        offs_t = t0 + tl.arange(0, BLOCK_T)
        mask_t = offs_t < L

        attn_ptrs = attn_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2 + offs_t * attn_bs3
        attn_vals = tl.load(attn_ptrs, mask=mask_t, other=0.0).to(tl.float32)

        V_ptrs = V_ptr + b * V_bs0 + qh * V_bs1 + offs_t * V_bs2 + l * V_bs3
        V_vals = tl.load(V_ptrs, mask=mask_t, other=0.0).to(tl.float32)

        acc += tl.sum(attn_vals * V_vals, axis=0)

    out_ptrs = out_ptr + b * out_bs0 + qh * out_bs1 + l * out_bs2
    tl.store(out_ptrs, acc)
