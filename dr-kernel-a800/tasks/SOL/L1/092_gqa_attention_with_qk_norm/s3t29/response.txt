import torch
import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:

# Triton kernel: linear layer Y = X @ W^T + B
# X: [M, K], W: [N, K], B: [N], Y: [M, N]
@triton.jit
def linear_layer_kernel(
    X_ptr, W_ptr, B_ptr, Y_ptr,
    M, N, K,
    stride_xm, stride_xk,
    stride_w0, stride_w1,  # W strides for dim0=N and dim1=K
    stride_ym, stride_yn,
):
    m = tl.program_id(axis=0)  # row index in X
    n = tl.program_id(axis=1)  # output dim index
    acc = tl.zeros((), dtype=tl.float32)
    # Loop over K in chunks
    for k0 in range(0, K, 64):
        offs = k0 + tl.arange(0, 64)
        x = tl.load(X_ptr + m * stride_xm + offs * stride_xk, mask=offs < K, other=0.0)  # [64]
        w = tl.load(W_ptr + n * stride_w0 + offs * stride_w1, mask=offs < K, other=0.0)  # [64]
        acc += tl.sum(x * w, axis=0)
    b = tl.load(B_ptr + n)
    acc += b
    tl.store(Y_ptr + m * stride_ym + n * stride_yn, acc)


# Triton kernel: RMSNorm per (b, h, s, d)
# X: [B, H, S, D], W: [D], Y: [B, H, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, W_ptr, Y_ptr,
    B, H, S, D,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    eps: tl.float32,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    d = tl.program_id(axis=3)
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + d * stride_xd).to(tl.float32)
    sum_sq = x * x
    mean_sq = tl.sum(sum_sq, axis=0) / D
    inv_rms = tl.rsqrt(mean_sq + eps)
    w = tl.load(W_ptr + d).to(tl.float32)
    y = (x * inv_rms) * w
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + d * stride_yd, y)


# Triton kernel: apply rotation (RoPE) for last dim D=128, split into two halves (64,64)
# X: [B, H, S, D], C: [S, D/2], S: [S, D/2], Y: [B, H, S, D]
@triton.jit
def rotate_half_kernel(
    X_ptr, C_ptr, S_ptr, Y_ptr,
    B, H, S, D: tl.constexpr,
    stride_xb, stride_xh, stride_xs, stride_xd,
    stride_yb, stride_yh, stride_ys, stride_yd,
    stride_c0, stride_c1,     # cos strides: [S, D/2]
    stride_s0, stride_s1,     # sin strides: [S, D/2]
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    s = tl.program_id(axis=2)
    # Load x token vector
    x = tl.load(X_ptr + b * stride_xb + h * stride_xh + s * stride_xs + tl.arange(0, D) * stride_xd)  # [D]
    q1 = x[:64]
    q2 = x[64:]
    # Load cos and sin for half dimension
    cos_half = tl.load(C_ptr + s * stride_c0 + tl.arange(0, 64) * stride_c1)  # [64]
    sin_half = tl.load(S_ptr + s * stride_s0 + tl.arange(0, 64) * stride_s1)  # [64]
    rotated_half = -q2 * sin_half + q1 * cos_half  # [64]
    y = q1 * cos_half + q2 * sin_half             # [64]
    out = tl.concatenate([y, rotated_half], axis=0)  # [128]
    tl.store(Y_ptr + b * stride_yb + h * stride_yh + s * stride_ys + tl.arange(0, D) * stride_yd, out)


# Triton kernel: attention scores S[b, h, i, j] = Q_rot[b,h,i,:] @ K_rot[b,h,j,:] * scaling
# Inputs: Qrot: [B*H, S, D], Krot: [B*H, S, D], Outputs: S: [B*H, S, S]
@triton.jit
def attn_scores_kernel(
    Qrot_ptr, Krot_ptr, S_ptr,
    B, H, S, D: tl.constexpr,
    stride_qb, stride_qh, stride_qs, stride_qd,
    stride_kb, stride_kh, stride_ks, stride_kd,
    stride_sb, stride_sh, stride_si, stride_sj,
    scaling: tl.float32,
):
    bh = tl.program_id(axis=0)  # [0, B*H)
    i = tl.program_id(axis=1)   # query position
    j = tl.program_id(axis=2)   # key position
    # Load Qrot[bh, i, :] and Krot[bh, j, :]
    q = tl.load(Qrot_ptr + bh * stride_qb + i * stride_qs + tl.arange(0, D) * stride_qd)  # [D]
    k = tl.load(Krot_ptr + bh * stride_kb + j * stride_ks + tl.arange(0, D) * stride_kd)  # [D]
    acc = tl.sum(q * k, axis=0) * scaling
    # Apply causal mask: if i < j, set -inf (since we store in -inf elsewhere, here skip writing for i<j; kernel grid will not include j<i)
    # We write scores for i >= j. For i < j, leave S as -inf initialized by host.
    tl.store(S_ptr + bh * stride_sb + i * stride_si + j * stride_sj, acc)


# Triton kernel: column-wise softmax over sequence dim for each (bh, i): Soft[bh, i, :]
# Inputs: S: [B*H, S, S], Outputs: Soft: [B*H, S, S]
@triton.jit
def softmax_cols_kernel(
    S_ptr, Soft_ptr,
    B, H, S, S2,
    stride_sb, stride_si, stride_sj,
    stride_sb_soft, stride_si_soft, stride_sj_soft,
):
    bh = tl.program_id(axis=0)  # [0, B*H)
    i = tl.program_id(axis=1)   # [0, S)
    # We will process the entire sequence dim S2 in one go; S2 == S
    scores = tl.load(S_ptr + bh * stride_sb + i * stride_si + tl.arange(0, S2) * stride_sj)  # [S2]
    # Online max reduction
    m = tl.max(scores, axis=0)
    scores = scores - m
    exp_scores = tl.exp(scores)
    sum_exp = tl.sum(exp_scores, axis=0)
    soft = exp_scores / sum_exp
    tl.store(Soft_ptr + bh * stride_sb_soft + i * stride_si_soft + tl.arange(0, S2) * stride_sj_soft, soft)


# Triton kernel: final output projection O = attn_output_flat @ o_proj_weight^T + bias
# attn_output_flat: [M, N], o_proj_weight: [N, M], bias: [N], output: [M, N]
@triton.jit
def final_linear_kernel(

    M, N, K,
    stride_xm, stride_xk,
    stride_w0, stride_w1,  # W strides for dim0=N and dim1=K
    stride_ym, stride_yn,
):

    n = tl.program_id(axis=1)  # output dim index
    acc = tl.zeros((), dtype=tl.float32)

        x = tl.load(X_ptr + m * stride_xm + offs * stride_xk, mask=offs < K, other=0.0)  # [64]
        w = tl.load(W_ptr + n * stride_w0 + offs * stride_w1, mask=offs < K, other=0.0)  # [64]
        acc += tl.sum(x * w, axis=0)
    b = tl.load(B_ptr + n)
    acc += b
    tl.store(Y_ptr + m * stride_ym + n * stride_yn, acc)
