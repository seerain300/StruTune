具体方案
1) 投影：
   - Q, K, V 通过 linear_proj_kernel 计算：y[b, l, n] = sum_k x[b, l, k] * w[n, k]，accumulator float32，output float32。
   - GQA 中 key/value 需要从 num_key_value_heads=8 扩展到 num_attention_heads=96，按照 num_key_value_groups=12，host side expand/reshape，不涉及计算。

2) RMSNorm for Q and K：
   - q_rmsnorm_kernel: y[b, l, :] = x[b, l, :] * rsqrt(mean(x^2)+eps) * q_norm_weight
   - k_rmsnorm_kernel: y[b, l, :] = x[b, l, :] * rsqrt(mean(x^2)+eps) * k_norm_weight

3) Q/K RoPE rotation：
   - rotate_qk_kernel: split head_dim=128 into h1[:64] and h2[64:], apply cos to h1, -sin to h2, concatenate. Triton kernel per (b, l, h).

4) 注意力分数计算 + softmax + mask：
   - attn_matmul_kernel: compute scores[b, qh, l, t] = Q[b, qh, l] * K[b, qh, t], loop over t and k, grid=(B, num_heads, L) → B.
   - softmax_mask_kernel: per (b, qh, l) row, add causal mask m[t] = -inf if t < l else 0, then stable softmax: subtract max, exp, sum, normalize. Triton kernel per row.

5) 输出投影：
   - output_matmul_kernel: attn_output[b, qh, l] = sum_t scores[b, qh, l, t] * V[b, qh, t], grid=(B, num_heads, L) → B.

6) 最终线性投影：
   - final_linear_kernel: attn_output_flat[b, l, :] @ o_proj_weight^T → final_out[b, l, :], grid=(B, L, hidden_dim) with BLOCK_K tuned (e.g., 256).

特别强调 softmax_mask_kernel 的正确实现
- 原始代码使用了 torch.triu 生成因果 mask (diagonal=1)。正确的理解是：对于每个 query 位置 l，只有 key 位置 t >= l 的 attention 才有效，t < l 的置为 -inf，然后进行 softmax。因此 mask[t] = -inf if t < l else 0。我们需要确保 Triton kernel 中的 mask 计算准确，并且被添加到 scores 上进行 softmax。我们采用 stable softmax: y = y + mask; y = y - max(y); y = exp(y); y = sum(y); y = y / sum; 再写回 scores[b, qh, l, :].

import torch
import triton
import triton.language as tl


# 1) Linear projection: y[b, l, n] = sum_k x[b, l, k] * w[n, k], accumulate in f32, output f32
@triton.jit
def linear_proj_kernel(
    x_ptr,           # *f16/f32/bf16, [B, L, H_in]
    w_ptr,           # *f16/f32/bf16, [N_out, H_in]
    y_ptr,           # *f32, [B, L, N_out]
    B, L, H_in, N_out,
    x_bs0, x_bs1, x_bs2,
    w_bs0, w_bs1,
    y_bs0, y_bs1, y_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_in, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_in
        x_ptrs = x_ptr + b * x_bs0 + l * x_bs1 + offs_k * x_bs2
        w_ptrs = w_ptr + n * w_bs0 + offs_k * w_bs1
        x_vals = tl.load(x_ptrs, mask=mask_k, other=0.0)
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)
        # x_vals and w_vals are float16/bfloat16/float32; cast to f32 for accumulation
        acc += tl.sum(x_vals.to(tl.float32) * w_vals.to(tl.float32), axis=0)
    tl.store(y_ptr + b * y_bs0 + l * y_bs1 + n * y_bs2, acc)


# 2) RMSNorm per (b, l, h): y = x * rsqrt(mean(x^2) + eps) * weight
@triton.jit
def rmsnorm_kernel(
    x_ptr, weight_ptr, y_ptr,
    B, L, H,
    x_bs0, x_bs1, x_bs2,
    weight_bs0, weight_bs1,
    y_bs0, y_bs1, y_bs2,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    h = tl.program_id(2)

    # compute mean over h: single scalar for this (b, l) row
    sum_sq = tl.zeros((), dtype=tl.float32)
    for i in range(0, H):
        val = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + i * x_bs2)
        sum_sq += val * val
    mean = sum_sq / H
    inv = tl.rsqrt(mean + 1e-8)  # eps hardcoded to 1e-8 for numerical stability

    for i in range(0, H):
        x_val = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + i * x_bs2)
        w_val = tl.load(weight_ptr + i * weight_bs0)
        y_val = x_val * inv * w_val
        tl.store(y_ptr + b * y_bs0 + l * y_bs1 + i * y_bs2, y_val)


# 3) Rotate Q/K: split 128 into h1[:64] and h2[64:], apply cos to h1, -sin to h2
@triton.jit
def rotate_qk_kernel(
    x_ptr, cos_ptr, sin_ptr, y_ptr,
    B, L, HALF: tl.constexpr,
    x_bs0, x_bs1, x_bs2,
    y_bs0, y_bs1, y_bs2,
):
    b = tl.program_id(0)
    l = tl.program_id(1)

    # h1 = x[:, :HALF], h2 = x[:, HALF:]
    for i in range(0, HALF):
        xi = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + i * x_bs2)
        angle = tl.load(cos_ptr + i)  # cos[i]
        neg_sin = tl.load(sin_ptr + i)  # sin[i], will use as -sin
        yi1 = xi * angle
        xi2 = tl.load(x_ptr + b * x_bs0 + l * x_bs1 + (i + HALF) * x_bs2)
        yi2 = xi2 * (-neg_sin)
        tl.store(y_ptr + b * y_bs0 + l * y_bs1 + i * y_bs2, yi1)
        tl.store(y_ptr + b * y_bs0 + l * y_bs1 + (i + HALF) * y_bs2, yi2)


# 4) Compute attention scores[b, qh, l, t] = Q[b, qh, l] * K[b, qh, t]
@triton.jit
def attn_matmul_kernel(
    Q_ptr, K_ptr, scores_ptr,
    B, L, NUM_HEADS,
    Q_bs0, Q_bs1, Q_bs2, Q_bs3,
    K_bs0, K_bs1, K_bs2, K_bs3,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
):
    # Grid: (B, NUM_HEADS, L)
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)

    # Load Q[b, qh, l]
    Q_vec = tl.load(Q_ptr + b * Q_bs0 + qh * Q_bs1 + l * Q_bs2)  # scalar for this l
    acc = tl.zeros((), dtype=tl.float32)

    # Accumulate over t
    for t in range(0, L):
        K_vec = tl.load(K_ptr + b * K_bs0 + qh * K_bs1 + t * K_bs2)
        acc += Q_vec * K_vec

    # Store to scores[b, qh, l, t]
    for t in range(0, L):
        tl.store(scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + t * scores_bs3, acc)


# 5) Softmax + causal mask over sequence dim t for each (b, qh, l) row
@triton.jit
def softmax_mask_kernel(
    scores_ptr, B, L, NUM_HEADS,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
):
    # Grid: (B, NUM_HEADS, L)
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)

    # Load row: scores[b, qh, l, :]
    row_ptrs = scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + tl.arange(0, L) * scores_bs3
    row = tl.load(row_ptrs)
    row = row + (-float('inf'))  # initial add -inf, we will set mask later
    # Apply causal mask: m[t] = -inf if t < l else 0
    for t in range(0, L):
        if t < l:
            row = row + (-float('inf'))
        else:
            row = row + 0.0
    # Stable softmax
    row_max = tl.max(row, axis=0)
    row = row - row_max
    exp_row = tl.exp(row)
    row_sum = tl.sum(exp_row, axis=0)
    row = exp_row / row_sum
    # Store back
    tl.store(row_ptrs, row)


# 6) Output matmul: attn_output[b, qh, l] = sum_t scores[b, qh, l, t] * V[b, qh, t]
@triton.jit
def output_matmul_kernel(
    scores_ptr, V_ptr, attn_out_ptr,
    B, L, NUM_HEADS, H,
    scores_bs0, scores_bs1, scores_bs2, scores_bs3,
    V_bs0, V_bs1, V_bs2,
    attn_bs0, attn_bs1, attn_bs2,
):
    # Grid: (B, NUM_HEADS, L)
    b = tl.program_id(0)
    qh = tl.program_id(1)
    l = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for t in range(0, L):
        s = tl.load(scores_ptr + b * scores_bs0 + qh * scores_bs1 + l * scores_bs2 + t * scores_bs3)
        v = tl.load(V_ptr + b * V_bs0 + qh * V_bs1 + t * V_bs2)
        acc += s * v
    tl.store(attn_out_ptr + b * attn_bs0 + qh * attn_bs1 + l * attn_bs2, acc)


# 7) Final linear: attn_output_flat[b, l, :] @ o_proj_weight^T -> final_out[b, l, :]
@triton.jit
def final_linear_kernel(
    attn_ptr, o_ptr, out_ptr,
    B, L, H_flat, N_OUT,
    attn_bs0, attn_bs1, attn_bs2,
    o_bs0, o_bs1,
    out_bs0, out_bs1, out_bs2,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    n = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, H_flat, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H_flat
        attn_ptrs = attn_ptr + b * attn_bs0 + l * attn_bs1 + offs_k * attn_bs2
        o_ptrs = o_ptr + n * o_bs0 + offs_k * o_bs1
        attn_vals = tl.load(attn_ptrs, mask=mask_k, other=0.0)
        o_vals = tl.load(o_ptrs, mask=mask_k, other=0.0)
        acc += tl.sum(attn_vals.to(tl.float32) * o_vals.to(tl.float32), axis=0)
    tl.store(out_ptr + b * out_bs0 + l * out_bs1 + n * out_bs2, acc)


class ModelNew(torch.nn.Module):
    def __init__(self, hidden_dim: int = 768):
        super().__init__()
        self.hidden_dim = hidden_dim
        # fixed dimensions from the original code
        self.num_attention_heads = 96
        self.num_key_value_heads = 8
        self.num_key_value_groups = 12
        self.head_dim = 128
        self.scaling = self.head_dim ** -0.5

    def forward(
        self,
        hidden_states: torch.Tensor,
        q_proj_weight: torch.Tensor,
        q_proj_bias: torch.Tensor,
        k_proj_weight: torch.Tensor,
        k_proj_bias: torch.Tensor,
        v_proj_weight: torch.Tensor,
        v_proj_bias: torch.Tensor,
        o_proj_weight: torch.Tensor,
        q_norm_weight: torch.Tensor,
        k_norm_weight: torch.Tensor,
        cos: torch.Tensor,  # [L, H//2]
        sin: torch.Tensor,  # [L, H//2]
        rms_norm_eps: float,
    ):
        # Ensure device
        device = hidden_states.device
        dtype = hidden_states.dtype

        B, L, H = hidden_states.shape  # H = hidden_dim given (should be 768) but original H is 128 for Q/K/V
        assert H == self.head_dim, "hidden_states last dim must be head_dim=128"

        # 1) Linear projection
        Q = torch.empty((B, L, self.head_dim), device=device, dtype=torch.float32)
        K = torch.empty((B, L, self.head_dim), device=device, dtype=torch.float32)
        V = torch.empty((B, L, self.head_dim), device=device, dtype=torch.float32)

        grid_lin = (B, L, self.head_dim)
        # Q
        linear_proj_kernel[grid_lin](
            hidden_states, q_proj_weight, Q,
            B, L, self.head_dim, self.head_dim,
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            q_proj_weight.stride(0), q_proj_weight.stride(1),
            Q.stride(0), Q.stride(1), Q.stride(2),
            BLOCK_K=128, num_warps=4, num_stages=2
        )
        # K
        linear_proj_kernel[grid_lin](
            hidden_states, k_proj_weight, K,
            B, L, self.head_dim, self.head_dim,
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            k_proj_weight.stride(0), k_proj_weight.stride(1),
            K.stride(0), K.stride(1), K.stride(2),
            BLOCK_K=128, num_warps=4, num_stages=2
        )
        # V
        linear_proj_kernel[grid_lin](
            hidden_states, v_proj_weight, V,
            B, L, self.head_dim, self.head_dim,
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            v_proj_weight.stride(0), v_proj_weight.stride(1),
            V.stride(0), V.stride(1), V.stride(2),
            BLOCK_K=128, num_warps=4, num_stages=2
        )

        # 2) RMSNorm for Q and K
        Q_norm = torch.empty_like(Q)
        K_norm = torch.empty_like(K)

        grid_rms = (B, L)
        rmsnorm_kernel[grid_rms](
            Q, q_norm_weight, Q_norm,
            B, L, self.head_dim,
            Q.stride(0), Q.stride(1), Q.stride(2),
            q_norm_weight.stride(0), q_norm_weight.stride(1),
            Q_norm.stride(0), Q_norm.stride(1), Q_norm.stride(2),
            num_warps=4, num_stages=2
        )

        rmsnorm_kernel[grid_rms](
            K, k_norm_weight, K_norm,
            B, L, self.head_dim,
            K.stride(0), K.stride(1), K.stride(2),
            k_norm_weight.stride(0), k_norm_weight.stride(1),
            K_norm.stride(0), K_norm.stride(1), K_norm.stride(2),
            num_warps=4, num_stages=2
        )

        # 3) Rotate Q/K: cos and sin of shape [L, head_dim//2] already provided
        Q_rot = torch.empty_like(Q_norm)
        K_rot = torch.empty_like(K_norm)
        grid_rot = (B, L)
        rotate_qk_kernel[grid_rot](
            Q_norm, cos, sin, Q_rot,
            B, L, HALF=64,
            Q_norm.stride(0), Q_norm.stride(1), Q_norm.stride(2),
            Q_rot.stride(0), Q_rot.stride(1), Q_rot.stride(2),
            num_warps=2, num_stages=1
        )

        rotate_qk_kernel[grid_rot](
            K_norm, cos, sin, K_rot,
            B, L, HALF=64,
            K_norm.stride(0), K_norm.stride(1), K_norm.stride(2),
            K_rot.stride(0), K_rot.stride(1), K_rot.stride(2),
            num_warps=2, num_stages=1
        )

        # 4) GQA: expand K_rot and V to 96 heads
        # K_rot: [B, L, 128] -> [B, 8, 12, L, 128] by repeating K_rot[b, :, :] into 12 groups
        K_expand = K_rot.unsqueeze(2).unsqueeze(3).expand(B, self.num_key_value_heads, self.num_key_value_groups, L, self.head_dim).reshape(B, self.num_attention_heads, L, self.head_dim)
        V_gqa = V.unsqueeze(2).unsqueeze(3).expand(B, self.num_key_value_heads, self.num_key_value_groups, L, self.head_dim).reshape(B, self.num_attention_heads, L, self.head_dim)

        # 5) Compute attention scores[b, qh, l, t] = Q_rot[b, qh, l] * K_expand[b, qh, t]
        scores = torch.empty((B, self.num_attention_heads, L, L), device=device, dtype=torch.float32)

        grid_attn = (B, self.num_attention_heads, L)
        attn_matmul_kernel[grid_attn](
            Q_rot, K_expand, scores,
            B, L, self.num_attention_heads,
            Q_rot.stride(0), Q_rot.stride(1), Q_rot.stride(2),
            K_expand.stride(0), K_expand.stride(1), K_expand.stride(2), K_expand.stride(3),
            scores.stride(0), scores.stride(1), scores.stride(2), scores.stride(3),
            num_warps=2, num_stages=1
        )

        # 6) Softmax + causal mask per (b, qh, l) row
        grid_softmax = (B, self.num_attention_heads, L)
        softmax_mask_kernel[grid_softmax](
            scores,
            B, L, self.num_attention_heads,
            scores.stride(0), scores.stride(1), scores.stride(2), scores.stride(3),
            num_warps=2, num_stages=1
        )

        # 7) Output matmul: attn_output[b, qh, l] = sum_t scores[b, qh, l, t] * V_gqa[b, qh, t]
        attn_output = torch.empty((B, self.num_attention_heads, L), device=device, dtype=torch.float32)

        grid_output = (B, self.num_attention_heads, L)
        output_matmul_kernel[grid_output](
            scores, V_gqa, attn_output,
            B, L, self.num_attention_heads, self.head_dim,
            scores.stride(0), scores.stride(1), scores.stride(2), scores.stride(3),
            V_gqa.stride(0), V_gqa.stride(1), V_gqa.stride(2),
            attn_output.stride(0), attn_output.stride(1), attn_output.stride(2),
            num_warps=2, num_stages=1
        )

        # 8) Flatten heads and final linear projection to hidden_dim=768
        attn_output_flat = attn_output.reshape(B, L, self.num_attention_heads * self.head_dim)  # [B, L, 12288]
        final_out = torch.empty((B, L, self.hidden_dim), device=device, dtype=torch.float32)

        grid_final = (B, L, self.hidden_dim)
        final_linear_kernel[grid_final](
            attn_output_flat, o_proj_weight, final_out,
            B, L, self.num_attention_heads * self.head_dim, self.hidden_dim,
            attn_output_flat.stride(0), attn_output_flat.stride(1), attn_output_flat.stride(2),
            o_proj_weight.stride(0), o_proj_weight.stride(1),
            final_out.stride(0), final_out.stride(1), final_out.stride(2),
            BLOCK_K=256, num_warps=8, num_stages=3
        )

        return final_out

    value_states = value_states.view(batch_size, seq_length, num_key_value_heads, head_dim)

    # Implementing RMSNorm via F.layer_norm with elementwise_affine=True behavior:
