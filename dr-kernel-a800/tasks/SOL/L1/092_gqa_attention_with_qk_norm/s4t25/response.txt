# A corrected, functional linear_bias kernel with exact H_out output:
@triton.jit
def linear_bias_kernel_fixed(
    X_ptr, W_ptr, B_ptr, Out_ptr,
    Bsz, Ssz, H_in, H_out,
    stride_x_b, stride_x_s, stride_x_h,

):

    oh = tl.program_id(2)  # output head index [0..H_out)

    # We will compute the entire output vector for oh as a 1D vector of length H_out
    # For simplicity, we set BLOCK_I and BLOCK_H to match H_in and H_out (assuming they are small and typical).
    # Load bias
    bias = tl.load(B_ptr + oh * stride_b_o)
    acc = tl.zeros((H_out,), dtype=tl.float32)

    # Loop over input H_in

        xi = tl.load(X_ptr + b * stride_x_b + s * stride_x_s + ih * stride_x_h)

        acc += xi * wi

    acc += bias
    out_off = b * stride_out_b + s * stride_out_s + oh * stride_out_h
    # Store acc as a scalar to Out[b, s, oh]
    tl.store(Out_ptr + out_off, acc)


# Kernel: RMSNorm over last dim D=128: y = x * rsqrt(mean(x^2) + eps) * weight
# Input x: [B, S, D], weight: [D] -> Output y: [B, S, D]
@triton.jit
def rmsnorm_kernel(
    X_ptr, Weight_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_x_b, stride_x_s, stride_x_d,
    stride_w_d, stride_out_b, stride_out_s, stride_out_d,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    # We process one row (b, s) and normalize across D=128
    sum_sq = 0.0
    for i in range(0, D):
        x = tl.load(X_ptr + b * stride_x_b + s * stride_x_s + i * stride_x_d)
        sum_sq += x * x
    mean = sum_sq / D
    inv_rms = tl.rsqrt(mean + 1.0e-8)  # use a small eps for numerical stability (original uses 0.0 but safer here)
    # weight per dim
    w = tl.load(Weight_ptr + i * stride_w_d)
    # Write normalized and scaled output
    for i in range(0, D):
        x = tl.load(X_ptr + b * stride_x_b + s * stride_x_s + i * stride_x_d)
        y = x * inv_rms * w
        tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + i * stride_out_d, y)


# Kernel: Rotate half of last 128 dims: for d>=64, q1, q2 = q[d-64], q[d], q_rot = q1*cos - q2*sin
@triton.jit
def rotate_half_kernel(
    Q_ptr, Sin_ptr, Cos_ptr, Out_ptr,
    Bsz, Ssz, D,  # D=128
    stride_q_b, stride_q_s, stride_q_d,
    stride_out_b, stride_out_s, stride_out_d,
    stride_s_d, stride_c_d,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    d = tl.program_id(2)  # d over 0..D-1

    # Load q vector element
    q = tl.load(Q_ptr + b * stride_q_b + s * stride_q_s + d * stride_q_d)
    # Load sin and cos for this d
    sin_val = tl.load(Sin_ptr + d * stride_s_d)
    cos_val = tl.load(Cos_ptr + d * stride_c_d)

    # If d >= 64, rotate: q1=q[d-64], q2=q[d]
    if d >= 64:
        q1 = tl.load(Q_ptr + b * stride_q_b + s * stride_q_s + (d - 64) * stride_q_d)
        q2 = q  # current q[d]
        q_rot = q1 * cos_val - q2 * sin_val
        tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + d * stride_out_d, q_rot)
    else:
        # For d < 64, just copy q
        tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + d * stride_out_d, q)


# Kernel: Compute attention scores Q @ K^T for one (b, s, h): output [S]
@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_q_b, stride_q_h, stride_q_s, stride_q_d,
    stride_k_b, stride_k_h, stride_k_s, stride_k_d,
    stride_out_b, stride_out_h, stride_out_s,
):
    b = tl.program_id(0)
    s = tl.program_id(1)  # query position
    h = tl.program_id(2)  # attention head

    # Accumulator for attention scores over K-length S
    acc = tl.zeros((Ssz,), dtype=tl.float32)

    # Loop over K positions
    for k in range(0, Ssz):
        q_vec = tl.zeros((D,), dtype=tl.float32)
        k_vec = tl.zeros((D,), dtype=tl.float32)
        # Load Q[b, h, s, :]
        for d in range(0, D):
            q_off = b * stride_q_b + h * stride_q_h + s * stride_q_s + d * stride_q_d
            q_vec[d] = tl.load(Q_ptr + q_off)
        # Load K[b, h, k, :]
        for d in range(0, D):
            k_off = b * stride_k_b + h * stride_k_h + k * stride_k_s + d * stride_k_d
            k_vec[d] = tl.load(K_ptr + k_off)
        # Dot product
        prod = 0.0
        for d in range(0, D):
            prod += q_vec[d] * k_vec[d]
        acc[k] = prod

    # Store to Out[b, h, s]
    out_off = b * stride_out_b + h * stride_out_h + s * stride_out_s
    tl.store(Out_ptr + out_off, acc[0])  # we store scalar per (b,h,s)


# Kernel: Softmax with causal mask: input logits [S], output probs [S]
# Causal mask: if col>k then -inf, else logits
@triton.jit
def softmax_mask_kernel(
    In_ptr, Out_ptr,
    Ssz,
    stride_in_s, stride_out_s,
):
    s = tl.program_id(0)
    # Load logits
    logits = tl.load(In_ptr + s * stride_in_s)
    # Compute row-wise max for numerical stability
    max_val = -float('inf')
    for i in range(0, Ssz):
        li = tl.load(In_ptr + i * stride_in_s)
        if li > max_val:
            max_val = li
    # Apply mask: if i > s, set to -inf
    exp_sum = 0.0
    for i in range(0, Ssz):
        li = tl.load(In_ptr + i * stride_in_s)
        if i > s:
            li = -float('inf')
        exp_val = tl.exp(li - max_val)
        exp_sum += exp_val
    # Normalize
    for i in range(0, Ssz):
        li = tl.load(In_ptr + i * stride_in_s)
        if i > s:
            li = -float('inf')
        pi = tl.exp(li - max_val) / exp_sum
        tl.store(Out_ptr + i * stride_out_s, pi)


# Kernel: Compute attention output: probs @ V for one (b, s, h): output [D]
@triton.jit
def matmul_attn_kernel(
    Probs_ptr, V_ptr, Out_ptr,
    Bsz, Ssz, D,
    stride_p_b, stride_p_h, stride_p_s,
    stride_v_b, stride_v_h, stride_v_s, stride_v_d,
    stride_out_b, stride_out_h, stride_out_s, stride_out_d,
):
    b = tl.program_id(0)
    s = tl.program_id(1)  # output position
    h = tl.program_id(2)

    # Accumulator for V vector of length D
    acc = tl.zeros((D,), dtype=tl.float32)

    # Loop over K positions (S), each contributes prob * V[b, h, k, :]
    for k in range(0, Ssz):
        prob = tl.load(Probs_ptr + b * stride_p_b + h * stride_p_h + k * stride_p_s)
        v_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            v_off = b * stride_v_b + h * stride_v_h + k * stride_v_s + d * stride_v_d
            v_vec[d] = tl.load(V_ptr + v_off)
        acc += prob * v_vec

    # Store to Out[b, h, s, :]
    out_off = b * stride_out_b + h * stride_out_h + s * stride_out_s
    for d in range(0, D):
        tl.store(Out_ptr + out_off + d * stride_out_d, acc[d])

            K_ptr, Krep_ptr,
            Bsz, Ssz, H_in, D,
            stride_k_b, stride_k_s, stride_k_h, stride_k_d,
            stride_krep_b, stride_krep_h, stride_krep_s, stride_krep_d,
            h, g,

            s = tl.program_id(1)
            d = tl.program_id(2)  # d over 0..D-1
