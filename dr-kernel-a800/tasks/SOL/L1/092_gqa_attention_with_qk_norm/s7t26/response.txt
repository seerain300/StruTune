# Triton kernel: dense linear over (B, S, H). Computes out[b, s, h] = sum_k input[b, s, k] * weight[h, k] + bias[h]
# Assumes input has shape [B, S, K] and weight has shape [H, K], output [B, S, H]
@triton.jit
def triton_linear_bsh(input_ptr, weight_ptr, bias_ptr, out_ptr,
                      B, S, H, K,
                      in_stride0, in_stride1, in_stride2,
                      wt_stride0, wt_stride1,
                      out_stride0, out_stride1, out_stride2,
                      BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_in = b * in_stride0 + s * in_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask = k < K
        x = tl.load(input_ptr + base_in + k * in_stride2, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + h * wt_stride0 + k * wt_stride1, mask=mask, other=0.0).to(tl.float32)
        acc += tl.sum(x * w, axis=0)

    bval = tl.load(bias_ptr + h).to(tl.float32)
    tl.store(out_ptr + base_out, acc + bval)


# Triton RMSNorm per row over last dim S: out[b, h, :] = x * (weight[h] / sqrt(mean(x^2) + eps))
# x is [B, S, H], strides over dims are provided; we run grid (B, H)
@triton.jit
def triton_rmsnorm_row(x_ptr, weight_ptr, out_ptr,
                       B, S, H,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       eps: tl.constexpr,
                       BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    base_x = b * x_stride0 + h * x_stride1
    base_out = b * out_stride0 + h * out_stride2

    sum_sq = 0.0
    for s0 in range(0, S, BLOCK_S):
        s = s0 + tl.arange(0, BLOCK_S)
        mask = s < S
        x = tl.load(x_ptr + base_x + s * x_stride2, mask=mask, other=0.0).to(tl.float32)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = inv_rms * tl.load(weight_ptr + h).to(tl.float32)

    for s0 in range(0, S, BLOCK_S):
        s = s0 + tl.arange(0, BLOCK_S)
        mask = s < S
        x = tl.load(x_ptr + base_x + s * x_stride2, mask=mask, other=0.0).to(tl.float32)
        y = x * scale
        tl.store(out_ptr + base_out + s * out_stride2, y, mask=mask)


# Triton kernel: compute attention for each (b, h) over query positions i and key positions j.
# Computes scores[b, i, j] = Q[b, i, :] @ K[b, j, :].T * scaling, applies causal mask, softmax over j, and accumulates
# output[b, i, :] += sum_j attn[b, i, j] * V[b, j, :].
# Inputs: Q, K, V shapes [B, S, H], output Out [B, S, H].
@triton.jit
def attention_tiled(Q_ptr, K_ptr, V_ptr, Out_ptr,
                    B, S, H,
                    Q_stride0, Q_stride1, Q_stride2,
                    K_stride0, K_stride1, K_stride2,
                    V_stride0, V_stride1, V_stride2,
                    Out_stride0, Out_stride1, Out_stride2,
                    scaling: tl.constexpr,
                    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)

    base_Q = b * Q_stride0 + h * Q_stride1
    base_K = b * K_stride0 + h * K_stride1
    base_V = b * V_stride0 + h * V_stride1
    base_Out = b * Out_stride0 + h * Out_stride2

    for i0 in range(0, S, BLOCK_I):
        i = i0 + tl.arange(0, BLOCK_I)
        mask_i = i < S
        # initialize output accumulator for this (b,h) row
        out_row = tl.zeros([BLOCK_I], dtype=tl.float32)

        for j0 in range(0, S, BLOCK_J):
            j = j0 + tl.arange(0, BLOCK_J)
            mask_j = j < S

            # Load Q row segment [BLOCK_I, 1]
            q_row = tl.zeros([BLOCK_I], dtype=tl.float32)
            for ii in range(0, BLOCK_I):
                q_row[ii] = tl.load(Q_ptr + base_Q + (i[ii] * Q_stride2), mask=mask_i[ii], other=0.0).to(tl.float32)

            # Load K segment [1, BLOCK_J]
            k_col = tl.zeros([BLOCK_J], dtype=tl.float32)
            for jj in range(0, BLOCK_J):
                k_col[jj] = tl.load(K_ptr + base_K + (j[jj] * K_stride2), mask=mask_j[jj], other=0.0).to(tl.float32)

            # Compute scores[i, j] = q_row @ k_col (outer product per element)
            scores = q_row[:, None] * k_col[None, :]  # [BLOCK_I, BLOCK_J]
            scores = scores * scaling  # float32 scalar

            # Causal mask: block future (i < j)
            for ii in range(0, BLOCK_I):
                for jj in range(0, BLOCK_J):
                    if i[ii] < j[jj]:
                        scores[ii, jj] = -float('inf')

            # Softmax along j axis
            exp_scores = tl.exp(scores)
            denom = tl.sum(exp_scores, axis=1)  # [BLOCK_I]
            attn = exp_scores / denom[:, None]  # [BLOCK_I, BLOCK_J]

            # Accumulate output: out_row += sum_j attn[i, j] * V[b, j, :]
            for jj in range(0, BLOCK_J):
                v_j = tl.load(V_ptr + base_V + (j[jj] * V_stride2), mask=mask_j[jj], other=0.0).to(tl.float32)
                out_row += tl.sum(attn[:, jj], axis=0) * v_j

        # Store result
        for ii in range(0, BLOCK_I):
            tl.store(Out_ptr + base_Out + (i[ii] * Out_stride2), out_row[ii], mask=mask_i[ii])

2) RMSNorm for Q and K:

3) Attention computation:
   - Triton kernel attention_compute(Q, K, V, Out, B, S, H, Q/K/V strides, Out strides, scaling): For each (b, h), loop over i (query positions) and j (key positions) in tiles (BLOCK_I, BLOCK_J). Compute scores = Q[b, i, :] @ K[b, j, :].T * scaling; apply causal mask (lower triangular, diagonal=1); softmax along j; accumulate output[b, i, :] += sum_j attn[b, i, j] * V[b, j, :]. Grid (B, H). This avoids torch matmul/softmax in forward.

4) Output projection:
   - Triton kernel triton_linear_bsh_no_bias(input, weight, out, B, S, H, K, input strides, weight strides, out strides, BLOCK_K): out[b, s, h] = sum_k input[b, s, k] * weight[h, k]. Launch to project attn_output to final output using o_proj_weight (no bias).

Important shape assumptions
- The original code uses hidden_states of shape [B, S, H*head_dim] where H=num_attention_heads and head_dim=128, and the projection weights have shape [H, head_dim]. In the simplified Triton implementation, we adapt to general K = hidden_states.shape[-1]. For Triton linear_bsh, the weight shape must be [H, K]. The evaluator’s inputs should align accordingly (typically K=128 or H*128). To avoid the previous “not launched” issues, we implement linear_bsh robustly with grid (B, H, S) and tiling over K. This kernel is actually launched and performs the dense linear projection.

Final code

```python
import torch
import triton
import triton.language as tl

# Triton kernel: dense linear over (B, S, H). Computes out[b, s, h] = sum_k input[b, s, k] * weight[h, k] + bias[h]
# Input: [B, S, K], Weight: [H, K], Output: [B, S, H]
@triton.jit
def triton_linear_bsh(input_ptr, weight_ptr, bias_ptr, out_ptr,
                      B, S, H, K,
                      in_stride0, in_stride1, in_stride2,
                      wt_stride0, wt_stride1,
                      out_stride0, out_stride1, out_stride2,
                      BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s = tl.program_id(2)

    base_in = b * in_stride0 + s * in_stride1
    base_out = b * out_stride0 + s * out_stride1 + h * out_stride2

    acc = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask = k < K
        x = tl.load(input_ptr + base_in + k * in_stride2, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + h * wt_stride0 + k * wt_stride1, mask=mask, other=0.0).to(tl.float32)
        acc += tl.sum(x * w, axis=0)

    bval = tl.load(bias_ptr + h).to(tl.float32)
    tl.store(out_ptr + base_out, acc + bval)


# Triton RMSNorm per row over last dim S: out[b, h, :] = x * (weight[h] / sqrt(mean(x^2) + eps))
# x is [B, S, H], strides over dims are provided; we run grid (B, H)
@triton.jit
def triton_rmsnorm_row(x_ptr, weight_ptr, out_ptr,
                       B, S, H,
                       x_stride0, x_stride1, x_stride2,
                       out_stride0, out_stride1, out_stride2,
                       eps: tl.constexpr,
                       BLOCK_S: tl.constexpr):
    b = tl.program_id(0)
    h = tl.program_id(1)
    base_x = b * x_stride0 + h * x_stride1
    base_out = b * out_stride0 + h * out_stride2

    sum_sq = 0.0
    for s0 in range(0, S, BLOCK_S):
        s = s0 + tl.arange(0, BLOCK_S)
        mask = s < S
        x = tl.load(x_ptr + base_x + s * x_stride2, mask=mask, other=0.0).to(tl.float32)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_sq / S
    inv_rms = 1.0 / tl.sqrt(mean + eps)
    scale = inv_rms * tl.load(weight_ptr + h).to(tl.float32)

    for s0 in range(0, S, BLOCK_S):
        s = s0 + tl.arange(0, BLOCK_S)
        mask = s < S
        x = tl.load(x_ptr + base_x + s * x_stride2, mask=mask, other=0.0).to(tl.float32)
        y = x * scale
        tl.store(out_ptr + base_out + s * out_stride2, y, mask=mask)


# Triton attention: compute scores, causal mask, softmax, and accumulate output
# Inputs: Q, K, V shapes [B, S, H], Output Out [B, S, H]
# Grid: (B, H)
@triton.jit
def attention_compute(Q_ptr, K_ptr, V_ptr, Out_ptr,
                      B, S, H,
                      Q_stride0, Q_stride1, Q_stride2,
                      K_stride0, K_stride1, K_stride2,
                      V_stride0, V_stride1, V_stride2,
                      Out_stride0, Out_stride
