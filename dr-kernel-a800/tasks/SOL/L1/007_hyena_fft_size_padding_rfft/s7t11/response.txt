import torch

# Triton import and availability check
TRITON_AVAILABLE = False
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: compute zero-padded real FFT and write real/imag parts
# We treat input length N_in = 2*seqlen and zero-pad for n >= seqlen.
# Output length L_out = N_in//2 + 1 = seqlen + 1. We only compute and store the first L_out elements.
@triton.jit
def _rfft_zero_pad_direct_kernel(
    x_ptr,           # *const float32, input flattened
    real_out_ptr,    # *float32, output real flattened
    imag_out_ptr,    # *float32, output imag flattened
    N_in: tl.int32,  # total length we treat input as (2 * seqlen)
    L_out: tl.int32, # output length (seqlen + 1)
    scale: tl.float32,
    BLOCK_N: tl.constexpr,  # tile size for reduction over n
):
    # One program per output frequency index k in [0, L_out)
    k = tl.program_id(0)
    # If k >= L_out, exit (grid is set to L_out, so this is fine)
    # Compute y[k] = sum_{n=0}^{N_in-1} x[n] * exp(-2πi k n / N_in)
    # Initialize accumulators
    acc_real = tl.zeros((), dtype=tl.float32)
    acc_imag = tl.zeros((), dtype=tl.float32)

    # Loop over n in tiles
    n_start = 0
    while n_start < N_in:
        n = n_start + tl.arange(0, BLOCK_N)
        mask = n < N_in
        # For n >= seqlen, x[n] = 0 (zero padding); we'll rely on masked loads with other=0.0 below
        # Load x[n]; we pass x_ptr as float32, so masked load fills with zeros
        x_vals = tl.load(x_ptr + n, mask=mask, other=0.0)  # shape: (BLOCK_N,)
        # Compute exp(-2πi k n / N_in) for this tile
        # Use float32 math: N_in is int, convert to float for division
        # factor = 2.0 * π * k / N_in, but Triton doesn't have π constant, use 3.141592653589793
        factor = (2.0 * 3.141592653589793) * (k) / N_in  # scalar
        # broadcast n to float
        n_f = n.to(tl.float32)
        # exp(-j * 2π * k * n / N_in) = cos(...) - j sin(...)
        angle = -factor * n_f
        # Triton trig: tl.cos and tl.sin expect vector (we can compute per element)
        cos_angle = tl.cos(angle)
        sin_angle = tl.sin(angle)
        # Multiply x_vals (real) with complex exponent; since x_vals is real, this is:
        # x * cos(angle) contributes to real part, and -x * sin(angle) contributes to imag part
        contrib_real = x_vals * cos_angle
        contrib_imag = -x_vals * sin_angle
        # Sum across the tile
        # Reduce vector to scalar
        # Triton allows tl.sum over axis=0; if not, emulate via loop
        sum_real = tl.sum(contrib_real, axis=0)
        sum_imag = tl.sum(contrib_imag, axis=0)
        acc_real += sum_real
        acc_imag += sum_imag
        n_start += BLOCK_N

    # Apply normalization
    acc_real = acc_real * scale
    acc_imag = acc_imag * scale

    # Store results (k in [0, L_out))
    # Flatten output pointers; since L_out == seqlen+1, k is valid index into flattened array
    tl.store(real_out_ptr + k, acc_real)
    tl.store(imag_out_ptr + k, acc_imag)


def _ceil_div(a: int, b: int) -> int:
    return (a + b - 1) // b

            x_freq = x_freq / (2.0 * x.shape[-1])

        # Input x: (batch, channels, seqlen)
        batch, channels, seqlen = x.shape

        x_f32 = x.to(torch.float32)
        x_flat = x_f32.view(-1)

        # Padded length and output length as per rfft semantics
        N_in = 2 * seqlen  # zero-pad to this length
        # Output length for rfft(input_length=seqlen, n=N_in) is L_out = N_in // 2 + 1, which equals seqlen + 1
        L_out = N_in // 2 + 1  # equals seqlen + 1

        scale = 1.0 / (2.0 * seqlen)
