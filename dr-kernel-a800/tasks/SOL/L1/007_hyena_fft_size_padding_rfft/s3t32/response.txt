@triton.jit
def pad_to_2L_kernel(
    x_ptr,               # *f32, input tensor (B, C, L), contiguous
    x_padded_ptr,        # *f32, output tensor (B, C, 2*L), contiguous
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,         # original seqlen
    stride_x_b: tl.int32,  # input stride for batch
    stride_x_c: tl.int32,  # input stride for channel
    stride_x_l: tl.int32,  # input stride for last dim (should be 1)
    stride_pad_b: tl.int32, # padded input stride for batch
    stride_pad_c: tl.int32, # padded input stride for channel
    stride_pad_l: tl.int32, # padded input stride for last dim (should be 1)
    BLOCK_N: tl.constexpr,  # tile size for copying
):
    # One Triton program per (batch, channel) slice
    pid = tl.program_id(axis=0)
    b = pid // channels
    c = pid % channels

    base_in = b * stride_x_b + c * stride_x_c
    base_out = b * stride_pad_b + c * stride_pad_c
    twoL = 2 * L

    # Copy original L elements into first L positions
    start = 0
    while start < L:
        j = start + tl.arange(0, BLOCK_N)
        mask = j < L
        vals = tl.load(x_ptr + base_in + j * stride_x_l, mask=mask, other=0.0)
        tl.store(x_padded_ptr + base_out + j * stride_pad_l, vals, mask=mask)
        start += BLOCK_N

    # Fill remaining positions with zeros
    zero_vec = tl.zeros([BLOCK_N], dtype=tl.float32)
    start = L
    while start < twoL:
        j = start + tl.arange(0, BLOCK_N)
        mask = j < twoL
        tl.store(x_padded_ptr + base_out + j * stride_pad_l, zero_vec, mask=mask)
        start += BLOCK_N


@triton.jit
def rfft_direct_kernel(
    x_padded_ptr,        # *f32, input padded tensor (B, C, 2*L), contiguous
    real_out_ptr,        # *f32, output real part (B, C, L+1), contiguous
    imag_out_ptr,        # *f32, output imag part (B, C, L+1), contiguous
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,         # original seqlen
    out_stride_b: tl.int32,  # output stride for batch (should be L+1 in last dim)
    out_stride_c: tl.int32,  # output stride for channel
    out_stride_l: tl.int32,  # output stride for last dim (should be 1)
    BLOCK_K: tl.constexpr,   # tile size for k (freq index)
    BLOCK_N: tl.constexpr,   # tile size for j (time index)
):
    # One Triton program per (batch, channel) slice
    pid = tl.program_id(axis=0)
    b = pid // channels
    c = pid % channels

    base_pad = b * x_padded_ptr.stride(0) + c * x_padded_ptr.stride(1)
    base_out = b * real_out_ptr.stride(0) + c * real_out_ptr.stride(1)

    # Accumulators for real and imag parts
    real_acc = tl.zeros([1], dtype=tl.float32)
    imag_acc = tl.zeros([1], dtype=tl.float32)

    twoL = 2 * L
    norm = 1.0 / (2.0 * L)

    # Loop over k = 0..L (we will store k=0..L -> output length L+1)
    k_start = 0
    while k_start <= L:
        k_vec = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_vec <= L

        # Accumulate over j = 0..2*L-1
        j = 0
        while j < twoL:
            j_vec = j + tl.arange(0, BLOCK_N)
            mask_j = j_vec < twoL

            # Load x_padded[b, c, j] for this j_vec
            vals = tl.load(x_padded_ptr + base_pad + j_vec * x_padded_ptr.stride(2), mask=mask_j, other=0.0)

            # Compute angles: theta = pi * k * j / L
            # Note: j_vec is int32, L is int32, result is float32 due to vals dtype promotion
            theta = tl.pi * (k_vec[:, None].to(tl.float32)) * (j_vec[None, :].to(tl.float32)) / L

            cos_t = tl.cos(theta)
            sin_t = tl.sin(theta)

            # Multiply-accumulate: sum_j vals[j] * (cos + i*sin)
            # For each k lane, sum across j lanes
            # We need to reduce across axis=1 (j dimension).
            prod_real = vals * cos_t
            prod_imag = vals * sin_t

            real_acc += tl.sum(prod_real, axis=1)
            imag_acc += tl.sum(prod_imag, axis=1)

            j += BLOCK_N

        # Store results at output position k
        # mask_k is vector [BLOCK_K], but out index is scalar k_start; store each valid k
        # If k_start exceeds L, we skip (mask_k would be False). We need to ensure we only store valid k.
        # Use tl.where to avoid writing for invalid k.
        # We can store one by one for simplicity; Triton supports scalar stores.
        k_scalar = k_start
        # For each lane in k_vec, if it's valid, store to output
        # This unrolled loop is acceptable for BLOCK_K being a small constexpr.
        for kk in range(BLOCK_K):
            if mask_k[kk]:
                out_idx = k_scalar
                # Store as contiguous last-dim offset out_idx
                tl.store(real_out_ptr + base_out + out_idx * out_stride_l, real_acc[kk] * norm, mask=True)
                tl.store(imag_out_ptr + base_out + out_idx * out_stride_l, imag_acc[kk] * norm, mask=True)

        k_start += BLOCK_K

            while j < twoL:
                val = tl.load(y_ptr + base_y + j * stride_y_l)
                # (-1)^j: parity check
                parity = (j % 2).to(tl.int32)
                # if j is even, add; else subtract
                total += val if parity == 0 else (-val)
                j += 1

            # Write real part at index L, imag part = 0
            tl.store(real_out_ptr + base_out + L * out_stride_l, total[0])
            tl.store(imag_out_ptr + base_out + L * out_stride_l, 0.0)

        # Launch the tiny kernel to set the Nyquist term correctly
        grid_nyq = (batch * channels,)
        compute_x_at_L_kernel[grid_nyq](
            y,
            real_out,
            batch, channels, L,
            y.stride(0), y.stride(1), y.stride(2),
            real_out.stride(0), real_out.stride(1), real_out.stride(2),
            imag_out.stride(0), imag_out.stride(1), imag_out.stride(2),
            twoL,
            num_warps=1,
        )

        return real_out, imag_out
