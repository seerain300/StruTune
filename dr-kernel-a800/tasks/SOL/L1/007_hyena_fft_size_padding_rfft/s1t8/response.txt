import triton
import triton.language as tl


@triton.jit
def _real_to_complex_rfft_triton_kernel(
    x_ptr,                # *float32, input pointer to x (B, C, L) contiguous
    out_real_ptr,         # *float32, output pointer to real part (B, C, L+1)
    out_imag_ptr,         # *float32, output pointer to imag part (B, C, L+1)
    B: tl.constexpr,      # batch size (unused in kernel but passed for clarity)
    C: tl.constexpr,      # channels (unused in kernel but passed for clarity)
    N: tl.constexpr,      # total length of real input (2 * L)
    L: tl.constexpr,      # original seqlen
    stride_b: tl.constexpr,   # stride for batch in x
    stride_c: tl.constexpr,   # stride for channel in x
    stride_t: tl.constexpr,   # stride for time in x (for contiguous this is 1)
    out_stride_b: tl.constexpr, # stride for batch in outputs
    out_stride_c: tl.constexpr, # stride for channel in outputs
    out_stride_m: tl.constexpr  # stride for m (last dim) in outputs
):
    # One program per (b, c) slice
    pid = tl.program_id(axis=0)
    b = pid // C
    c = pid % C

    # Compute base pointers for this (b, c) slice
    # x layout: x[b, c, t] -> offset = b*stride_b + c*stride_c + t*stride_t
    # out layout: out[b, c, m] -> offset = b*out_stride_b + c*out_stride_c + m*out_stride_m
    # We will compute m = 0..L
    # We avoid vectorized 2D indexing to prevent runtime errors; use scalar j loop.

    # Scale for normalization: 1 / (2 * L)
    scale = 1.0 / (2.0 * L)

    # We'll compute j from 0 to L (M-1) and store to out_real/out_imag
    # For each j, compute re_j and im_j by summing over t in [0, N)
    # re_j = scale * sum_{t=0..N-1} x[t] * cos(2*pi*j*t/N)
    # im_j = scale * sum_{t=0..N-1} x[t] * sin(2*pi*j*t/N)

    # We use scalar loop over j and t to keep it simple and robust.
    # Triton supports scalar for loops; this minimizes risk of runtime issues.
    for j in range(0, L + 1):
        sum_re = 0.0
        sum_im = 0.0

        # Sum over t = 0..N-1
        for t in range(0, N):
            # Load x[b, c, t]
            x_off = b * stride_b + c * stride_c + t * stride_t
            x_val = tl.load(x_ptr + x_off)

            # Compute cos and sin with float32
            angle = 2.0 * 3.141592653589793 * (j * t) / N
            sum_re += x_val * tl.cos(angle)
            sum_im += x_val * tl.sin(angle)

        # Apply normalization
        sum_re = sum_re * scale
        sum_im = sum_im * scale

        # Store to outputs: out_real[b, c, j], out_imag[b, c, j]
        out_off_real = b * out_stride_b + c * out_stride_c + j * out_stride_m
        out_off_imag = b * out_stride_b + c * out_stride_c + j * out_stride_m

        # Triton will cast to pointer dtype if needed; store as float32
        tl.store(out_real_ptr + out_off_real, sum_re)
        tl.store(out_imag_ptr + out_off_imag, sum_im)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        Compute fused FFT size padding and real FFT computation for Hyena convolution,
        but entirely in Triton. Returns real and imaginary parts separately.

        Input x: (batch, channels, seqlen), float32 or float16; we will cast to float32 inside.
        Output: (batch, channels, seqlen+1) for both real and imaginary parts.
        """
        # Ensure x is contiguous and cast to float32 for numerical stability
        x = x.contiguous()
        B, C, L = x.shape
        N = 2 * L

        # Allocate outputs
        out_real = torch.empty((B, C, L + 1), device=x.device, dtype=torch.float32)
        out_imag = torch.empty((B, C, L + 1), device=x.device, dtype=torch.float32)

        # Get strides (in elements)
        stride_b, stride_c, stride_t = x.stride()
        out_stride_b, out_stride_c, out_stride_m = out_real.stride()

        # Launch one program per (b, c) slice
        grid = (B * C,)

        # Run Triton kernel: all computation inside Triton, no torch ops
        _real_to_complex_rfft_triton_kernel[grid](
            x, out_real, out_imag,
            B, C, N, L,
            stride_b, stride_c, stride_t,
            out_stride_b, out_stride_c, out_stride_m,
            num_warps=1,  # conservative launch config; avoids overcommit for simple scalar loops
            num_stages=1
        )

        # Return outputs (already normalized)
        return out_real, out_imag
