import torch
import triton
import triton.language as tl


@triton.jit
def pad_to_2L_kernel(
    x_ptr,          # *f32, input x (B, C, L)
    x_padded_ptr,   # *f32, output padded x (B, C, 2L), contiguous
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,    # original seqlen
    stride_b: tl.int32,
    stride_c: tl.int32,
    stride_l: tl.int32,
    pad_stride_b: tl.int32,
    pad_stride_c: tl.int32,
    pad_stride_l: tl.int32,
    BLOCK_J: tl.constexpr,
):
    # One program per (batch, channel) slice
    pid = tl.program_id(axis=0)
    b = pid // channels
    c = pid % channels

    base_in = b * stride_b + c * stride_c
    base_out = b * pad_stride_b + c * pad_stride_c

    twoL = 2 * L
    # Copy original L elements into first half
    start = 0
    while start < L:
        j = start + tl.arange(0, BLOCK_J)
        mask = j < L
        vals = tl.load(x_ptr + base_in + j * stride_l, mask=mask, other=0.0)
        # Store into padded first half
        tl.store(x_padded_ptr + base_out + j * pad_stride_l, vals, mask=mask)
        start += BLOCK_J

    # Fill remaining positions with zeros
    zero_vec = tl.zeros([BLOCK_J], dtype=tl.float32)
    start = L
    while start < twoL:
        j = start + tl.arange(0, BLOCK_J)
        mask = j < twoL
        tl.store(x_padded_ptr + base_out + j * pad_stride_l, zero_vec, mask=mask)
        start += BLOCK_J


@triton.jit
def rfft_direct_kernel(
    x_padded_ptr,   # *f32, input padded tensor (B, C, 2*L)
    real_out_ptr,   # *f32, output real part (B, C, L+1), contiguous
    imag_out_ptr,   # *f32, output imag part (B, C, L+1), contiguous
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,    # original seqlen
    out_stride_b: tl.int32,
    out_stride_c: tl.int32,
    out_stride_l: tl.int32,
    pad_stride_b: tl.int32,
    pad_stride_c: tl.int32,
    pad_stride_l: tl.int32,
    BLOCK_J: tl.constexpr,
):
    # One program per (batch, channel) slice
    pid = tl.program_id(axis=0)
    b = pid // channels
    c = pid % channels

    base_out = b * out_stride_b + c * out_stride_c
    base_pad = b * pad_stride_b + c * pad_stride_c
    twoL = 2 * L

    # Accumulators for real and imag parts for each k
    real_acc = tl.zeros([L + 1], dtype=tl.float32)
    imag_acc = tl.zeros([L + 1], dtype=tl.float32)

    # Loop over k = 0..L
    k = 0
    while k <= L:
        # Accumulate over j = 0..2*L-1
        acc_real = 0.0
        acc_imag = 0.0
        start = 0
        while start < twoL:
            j = start + tl.arange(0, BLOCK_J)
            mask = j < twoL
            xj = tl.load(x_padded_ptr + base_pad + j * pad_stride_l, mask=mask, other=0.0)
            # theta = pi * k * j / (2*L)
            theta = 3.141592653589793 * k * j / twoL
            cosv = tl.cos(theta)
            sinv = tl.sin(theta)
            prod_real = xj * cosv
            prod_imag = xj * sinv
            # Reduce vector to scalars
            acc_real += tl.sum(prod_real, axis=0)
            acc_imag += tl.sum(prod_imag, axis=0)
            start += BLOCK_J
        # Scale by 1/(2*L) and store
        scale = 1.0 / (2.0 * L)
        real_acc[k] = acc_real * scale
        imag_acc[k] = acc_imag * scale
        k += 1

    # Store results (contiguous along last dim)
    k = 0
    while k <= L:
        tl.store(real_out_ptr + base_out + k * out_stride_l, real_acc[k])
        tl.store(imag_out_ptr + base_out + k * out_stride_l, imag_acc[k])
        k += 1
