import torch
import triton
import triton.language as tl


@triton.jit
def rfft_real_bins_kernel(x_ptr, out_real_ptr, N, seqlen, BLOCK_K: tl.constexpr):
    """
    Compute real part of rfft bins for a single (batch, channel) row:
      out_real[j] = sum_{k=0..N-1} x[k] * cos(2*pi*j*k/N) / N, for j in 0..seqlen.
    x_ptr: float32 pointer to padded input vector of length N.
    out_real_ptr: float32 pointer to output real vector (will contain only first seqlen+1 elements).
    """
    b = tl.program_id(0)
    c = tl.program_id(1)
    # j is the bin index
    j = tl.program_id(2)  # grid dimension for bins

    # Only compute for j in [0, seqlen]
    # Note: we assume grid is set to (batch, channels, seqlen+1) so j ranges correctly.
    acc = 0.0

    # Loop over k in chunks
    k = 0
    while k < N:
        offs = tl.arange(0, BLOCK_K)
        idx = k + offs
        mask = idx < N

        # Load x[idx]; out-of-range masked as 0
        x_vals = tl.load(x_ptr + idx, mask=mask, other=0.0)

        # Compute cos(2*pi*j*k/N) for each k in the chunk
        # Use float32 for accuracy
        theta = 2.0 * 3.141592653589793 * (j * idx / N)
        cos_vals = tl.cos(theta)

        # Accumulate
        acc += tl.sum(x_vals * cos_vals, axis=0)

        k += BLOCK_K

    # Normalize by N and store
    acc = acc / N
    tl.store(out_real_ptr + j, acc)


@triton.jit
def rfft_imag_bins_kernel(x_ptr, out_imag_ptr, N, seqlen, BLOCK_K: tl.constexpr):
    """
    Compute imaginary part of rfft bins for a single (batch, channel) row:
      out_imag[j] = sum_{k=0..N-1} x[k] * sin(2*pi*j*k/N) / N, for j in 1..seqlen-1.
    x_ptr: float32 pointer to padded input vector of length N.
    out_imag_ptr: float32 pointer to output imag vector (will contain only first seqlen+1 elements).
    """

    c = tl.program_id(1)

        Triton-only implementation: compute rfft(x, n=2*seqlen), normalize by 2*seqlen,
        and return real and imaginary parts separately as float32 tensors of shape
        (batch, channels, seqlen+1). No torch operations in forward; Triton kernels are launched.

        # focuses on forward and kernel launches; we keep forward free of torch compute.

        # Launch grid: one program per (batch, channel, bin). Note that we set grid's z dimension
        # equal to seqlen+1 for real, and for imag we use seqlen (since j=1..seqlen-1).
        # However, Triton does not support per-program dynamic index handling in Python grids.
        # Instead, we launch two separate calls: one for real j=0..seqlen, and one for imag j=1..seqlen-1.
