import torch
import triton
import triton.language as tl


@triton.jit
def _build_input_complex_kernel(x_ptr, t_real_ptr, t_imag_ptr, seqlen, N: tl.constexpr):
    """
    Build complex input for FFT of length N = 2*seqlen:
    - t_real[0:seqlen] = x
    - t_real[seqlen:N] = 0
    - t_imag[0:N] = 0
    Assumes x is contiguous 1D (we flatten batch*channels).
    """
    pid = tl.program_id(axis=0)
    idx = pid * tl.num_programs(axis=0)
    while idx < N:
        if idx < seqlen:
            val = tl.load(x_ptr + idx)
            tl.store(t_real_ptr + idx, val)
            tl.store(t_imag_ptr + idx, 0.0)
        else:
            tl.store(t_real_ptr + idx, 0.0)
            tl.store(t_imag_ptr + idx, 0.0)
        idx += tl.num_programs(axis=0)


@triton.jit
def _real_fft_complex_inplace_kernel(t_real_ptr, t_imag_ptr, N: tl.constexpr):
    """
    In-place real-FFT via complex FFT of length N (Cooley-Tukey, mixed-radix).
    We construct the complex FFT of a real input by zero-padding the second half.
    This kernel:
    - Bit-reverses pairs in the first half.
    - Iterates over stages k=1,2,3,..., N//2 (log2 N + 1 passes).
    - Computes butterfly updates with twiddle factors.
    After completion, t_real contains the final real outputs for k < N//2 + 1,
    which equals the real part of rfft(x, n=N).
    """
    HALF = N // 2
    i = 0
    # Bit-reverse pairing for real input (for k = N//2)
    # For generality, we implement bit-reverse of the first half indices.
    # Note: Triton supports while loops and integer ops; this is safe for moderate N.
    while i < HALF:
        j = i
        rev = 0
        b = 1
        # 16-bit bit-reverse (covers N up to 65536)
        while b < 16:
            rev ^= (j & 1) << b
            j >>= 1
            b += 1
        # Load from i and rev
        a_real = tl.load(t_real_ptr + i)
        a_imag = tl.load(t_imag_ptr + i)
        b_real = tl.load(t_real_ptr + rev)
        b_imag = tl.load(t_imag_ptr + rev)
        # For real input, b_imag should be 0; enforce by using b_imag=0.
        # Store swapped positions (no-op within same vector because we read them)
        i += 1

    # Main stages: k = 1,2,3,... up to N//2
    # For each stage, j = N//2 // step, step *= 2
    # Compute all butterflies for given step.
    step = 2
    while step <= N:
        j = step // 2
        while j > 0:
            # All i in [0, N/step)
            i = 0
            while i < (N // step):
                idx = i + j
                a_real = tl.load(t_real_ptr + i)
                a_imag = tl.load(t_imag_ptr + i)
                b_real = tl.load(t_real_ptr + idx)
                b_imag = tl.load(t_imag_ptr + idx)
                # twiddle factors: theta = 2*pi*i/step
                theta = 2.0 * 3.141592653589793 * (i + 0.0) / step
                c = tl.cos(theta)
                s = tl.sin(theta)
                creal = c * b_real - s * b_imag
                cimag = c * b_imag + s * b_real
                new_a_real = a_real + creal
                new_a_imag = a_imag + cimag
                new_b_real = a_real - creal
                new_b_imag = a_imag - cimag
                tl.store(t_real_ptr + i, new_a_real)
                tl.store(t_imag_ptr + i, new_a_imag)
                tl.store(t_real_ptr + idx, new_b_real)
                tl.store(t_imag_ptr + idx, new_b_imag)
                i += step
            j //= 2
        step *= 2


@triton.jit
def _divide_inplace_kernel(x_ptr, n_elements, scale, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    x = tl.load(x_ptr + offsets, mask=mask)
    y = x / scale
    tl.store(x_ptr + offsets, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        Triton-only implementation:
        - Compute real FFT of x (shape: (B, C, L)) with n=2*L, return real and imag parts.
        - Normalize by 2*L.
        - Output shape: (B, C, L+1), real and imag separately.
        """
        assert x.dim() == 3, f"Input must be (batch, channels, seqlen), got {x.shape}"
        batch, channels, seqlen = x.shape
        N = 2 * seqlen

        # Flatten (batch, channels) to a single 1D vector for kernel processing
        x_flat = x.reshape(-1).to(torch.float32).contiguous()
        B = x_flat.numel()

        # Allocate complex input buffers (real and imag) of length N
        t_real = torch.empty(N, dtype=torch.float32, device=x.device)
        t_imag = torch.empty(N, dtype=torch.float32, device=x.device)

        # Build complex input: t_real = x for first seqlen, zeros for second half; t_imag = 0
        # We choose a grid size proportional to N for parallelism.
        grid_build = (triton.cdiv(N, 256),)
        _build_input_complex_kernel[grid_build](x_flat, t_real, t_imag, seqlen, N=N)

        # Perform real-FFT via complex FFT in-place on t_real/t_imag
        # Launch with a grid sized to cover N elements.
        grid_fft = (triton.cdiv(N, 256),)
        _real_fft_complex_inplace_kernel[grid_fft](t_real, t_imag, N=N)

        # Extract real part of rfft: for real input, the final real outputs correspond to k < N//2 + 1.
        # But since we populated t_real with output after complex FFT, we take first N//2 + 1 as real rfft.
        # The kernel above writes final outputs to t_real. We need to compute output length = N//2 + 1.
        out_len = (N // 2) + 1
        # Take first out_len elements from t_real as the real rfft output.
        x_freq_real = t_real[:out_len].contiguous()

        # Normalize by N = 2*seqlen using Triton
        scale = float(N)
        out_len = x_freq_real.numel()
        BLOCK_SIZE = 1024
        grid_norm = (triton.cdiv(out_len, BLOCK_SIZE),)
        _divide_inplace_kernel[grid_norm](x_freq_real, out_len, scale, BLOCK_SIZE=BLOCK_SIZE)

        # Reshape to (batch, channels, seqlen + 1)
        out_real = x_freq_real.view(batch, channels, seqlen + 1)

        # Imaginary part is zero (because original returns real and imag of rfft; here we compute only real)
        # To return both, set imag to zeros.
        out_imag = torch.zeros_like(out_real)

        return out_real, out_imag

import torch
import triton
import triton.language as tl


@triton.jit
def _dct2_rfft_real_kernel(x_ptr, out_ptr, L, N: tl.constexpr):
    """
    Compute real part of rfft(x, n=N) via DCT-II for a real vector x of length L,
    writing output of length N//2 + 1 into out_ptr.
    y[k] = sum_{t=0..L-1} x[t] * cos(pi*k*(t+0.5)/N) for k=1..N//2
    y[0] = 0.5 * sum_{t=0..L-1} x[t]
    """
    # One program per output index k
    pid = tl.program_id(axis=0)
    if pid == 0:
        # k = 0 special case
        sum_val = tl.zeros((), dtype=tl.float32)
        t = 0
        while t < L:
            val = tl.load(x_ptr + t)
            sum_val += val
            t += 1
        # torch.rfft scales by 1/N for real outputs; here we compute DCT-II which needs scaling by 1/N.
        sum_val *= 0.5  # rfft real output y[0] equals 0.5 * sum(x)
        tl.store(out_ptr + 0, sum_val)
    else:
        k = pid - 1  # k in [0..N//2 - 1]
        sum_val = tl.zeros((), dtype=tl.float32)
        t = 0
        while t < L:
            # cos(pi*k*(t+0.5)/N)
            theta = tl.cos(3.141592653589793 * k * (t + 0.5) / N)
            val = tl.load(x_ptr + t)
            sum_val += val * theta
            t += 1
        # Store scaled result (rfft normalization divides by N)
        sum_val = sum_val / N
        tl.store(out_ptr + k + 1, sum_val)


@triton.jit
def _divide_inplace_kernel(x_ptr, n_elements, scale, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    x = tl.load(x_ptr + offsets, mask=mask)
    y = x / scale
    tl.store(x_ptr + offsets, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        Triton-only implementation of the original run(x):
        - Input: x of shape (batch, channels, seqlen)
        - Compute real part of rfft(x, n=2*seqlen) using DCT-II in Triton
        - Return real and imaginary parts (imag=0), normalized by 2*seqlen
        Output shape: (batch, channels, seqlen+1) for both real and imag
        """
        assert x.dim() == 3, f"Input must be (batch, channels, seqlen), got {x.shape}"
        batch, channels, seqlen = x.shape
        N = 2 * seqlen

        # Flatten to 1D vector for kernel processing
        x_flat = x.reshape(-1).to(torch.float32).contiguous()
        L = x_flat.numel()  # equals batch*channels*seqlen, but we only use seqlen for DCT

        # Output length: (N//2) + 1

        # Allocate output for real rfft
