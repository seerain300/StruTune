import torch
import triton
import triton.language as tl

@triton.jit
def dft_k_scalar_kernel(
    in_ptr,           # *float32, flattened input per (b, c): length = (B*C) * (2*L)
    out_real_ptr,     # *float32, flattened output real per (b, c, k): length = (B*C) * (L+1)
    out_imag_ptr,     # *float32, flattened output imag per (b, c, k): length = (B*C) * (L+1)
    L: tl.constexpr,      # int32, seqlen
    two_L: tl.constexpr,  # int32, 2 * seqlen
    B: tl.constexpr,      # int32, batch size (for index math)
    C: tl.constexpr,      # int32, channels
    b: tl.constexpr,      # int32, current batch index for this program
    c: tl.constexpr,      # int32, current channel index for this program
    k_index: tl.constexpr # int32, current k index for this program
):
    # Accumulators (scalars)
    acc_real = 0.0
    acc_imag = 0.0

    # Loop over t in [0, 2*L)
    for t in range(0, two_L):
        base = (b * C + c) * two_L
        x_t = tl.load(in_ptr + base + t)

        # Compute angle for DFT: -2*pi*k*t/(2*L)
        angle = -2.0 * 3.141592653589793 * float(k_index) * float(t) / float(two_L)

        cos_part = tl.cos(angle)
        sin_part = tl.sin(angle)

        acc_real += x_t * cos_part
        acc_imag -= x_t * sin_part  # imaginary contribution

    # Normalize by 2*L
    norm = 1.0 / float(two_L)
    acc_real = acc_real * norm
    # Imaginary for real input is zero; keep it zero and store as 0.0
    acc_imag = 0.0

    # Compute output offsets: for each (b, c), outputs span k in [0..L], so offset = (b*C + c) * (L+1) + k_index
    out_base = (b * C + c) * (L + 1)
    tl.store(out_real_ptr + out_base + k_index, acc_real)
    tl.store(out_imag_ptr + out_base + k_index, acc_imag)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        # Ensure contiguous float32 input
        assert x.dim() == 3, "Input must be 3D: (batch, channels, seqlen)"
        B, C, L = x.shape
        x = x.contiguous().to(torch.float32)

        # Prepare zero-padded input: padded length = 2*L
        two_L = 2 * L
        # Flatten all (B*C) slices into a single vector of length (B*C) * (2*L)
        # Original x has shape (B, C, L); flattened slice per (b, c) is contiguous of length two_L starting at offset (b*C + c) * two_L
        # Build a single tensor where the first (B*C)*two_L entries are x followed by zeros for the second half.
        # But we'll construct per (b, c) padded vectors and then flatten.
        # Easier: make a zero tensor of shape (B, C, two_L) and fill the first L with x, others zero.
        padded = torch.zeros((B, C, two_L), dtype=torch.float32, device=x.device)
        # Copy x into padded[:, :, :L]
        padded[:, :, :L] = x  # broadcasting by copying (x is (B, C, L), so assignment fills first L columns)

        # Flatten for Triton
        in_flat = padded.reshape(B * C, two_L).reshape(-1)  # 1D contiguous vector
        M_batches = B * C
        out_real_flat = torch.empty(M_batches * (L + 1), dtype=torch.float32, device=x.device)
        out_imag_flat = torch.empty(M_batches * (L + 1), dtype=torch.float32, device=x.device)

        # Launch Triton kernel once per (b, c, k)
        # Note: Triton loop with tl.constexpr allows scalar loops; grid is 1D over k.
        for b in range(B):
            for c in range(C):
                # We'll iterate over k directly in Triton by passing b, c, k as constexpr parameters per call.
                # However Triton doesn't support Python loop over constexpr k; instead we call the kernel M*L times via Python loops.
                # To avoid that, we can compute all k in nested Python loops and call kernel for each k.
                for k in range(L + 1):
                    dft_k_scalar_kernel[(1,)](
                        in_flat,
                        out_real_flat,
                        out_imag_flat,
                        L,
                        two_L,
                        B,
                        C,
                        b,
                        c,
                        k,
                        num_warps=1,
                        num_stages=1,
                    )

        # Reshape outputs to (B, C, L+1)
        out_real = out_real_flat.view(B, C, L + 1)
        out_imag = out_imag_flat.view(B, C, L + 1)

        return out_real, out_imag
