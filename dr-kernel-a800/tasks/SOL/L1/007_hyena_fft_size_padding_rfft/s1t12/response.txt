import triton
import triton.language as tl

def _rfft_real_imag_triton_kernel(

    out_real_ptr,           # *float32, output pointer to real part (B, C, L+1)

    C: tl.constexpr,        # channels    (unused but passed for clarity)
    L: tl.constexpr,        # seqlen

    x_stride_c,             # stride for channel in x
    x_stride_l,             # stride for last dim in x

    out_stride_c,           # stride for channel in outputs
    out_stride_l,           # stride for last dim in outputs

    M,                      # int, output length (L+1)

    # Compute base offset for this slice in x and outputs.
    # Since we pass x as contiguous (B, C, L), the strides are provided.
    # We can decode b and c by integer division/mod, but we don't need them.
    # We'll compute offsets using pid and strides.

    # Triton supports runtime while loops. We'll use them.
    j = 0
    while j < M:
        # Accumulators for this j
        acc_re = 0.0
        acc_im = 0.0

        # Loop over t from 0 to N-1
        t = 0
        while t < N:
            # Load x[t] for this (b, c) slice.
            # The pointer arithmetic: x_ptr + b*x_stride_b + c*x_stride_c + t*x_stride_l
            # We don't have b,c directly; we can infer from pid using integer division/mod.
            # But since grid = (B*C,), and we made x contiguous as (B, C, L), each program
            # corresponds to a unique (b, c). We need to find which (b, c) this pid maps to.
            # To keep things simple and robust, we assume host launches grid=(B*C,) and
            # the tensor x is contiguous with shape (B, C, L). Triton can't decode shapes here,
            # so we rely on the host to pass correct strides and layout.
            # We'll compute the base offset for this pid by assuming x is flattened except
            # last dim: base = pid * L + t. But to be precise, we need (b, c). Instead,
            # we rely on the fact that we launch grid=(B*C,) and each program handles exactly
            # one (b, c) slice: base_x = pid * L.

            # To ensure correctness, we assume x is contiguous in (B, C, L) and we compute
            # offset for each t as: offset = (b*C + c)*L + t. Since we don't have b,c here,
            # we instead pass x as contiguous (B, C, L) and rely on the host to set grid=(B*C,).

        # grid=(B*C,), so we can compute base_out = pid * (L+1) + j.
        base_out = pid * (L + 1) + j
