    - t[:N] = x_flat

@triton.jit
def bitreverse_pairs_kernel(t_ptr, M: tl.constexpr):
    """
    Bit-reverse pairing for a time-domain buffer of length M (power-of-two).
    For i in [0, M//2), swap t[i] with t[bitrev(i)], and t[M + i] with t[M + bitrev(i)].
    """
    HALF = M // 2
    i = tl.program_id(axis=0)
    while i < HALF:
        rev = 0
        j = 0
        # Compute bit-reverse of i
        while j < 32:
            b = (i >> (31 - j)) & 1
            rev ^= b << j
            j += 1
        # Swap t[i] with t[rev], and t[M + i] with t[M + rev]
        tmp_i = tl.load(t_ptr + i)
        tmp_iM = tl.load(t_ptr + M + i)
        tmp_rev = tl.load(t_ptr + rev)
        tmp_revM = tl.load(t_ptr + M + rev)
        tl.store(t_ptr + i, tmp_rev)
        tl.store(t_ptr + rev, tmp_i)
        tl.store(t_ptr + M + i, tmp_revM)
        tl.store(t_ptr + M + rev, tmp_iM)
        i += 1


@triton.jit
def real_fft_stages_kernel(t_ptr, M: tl.constexpr):
    """
    Iterative Cooley-Tukey stages (radix-2) for complex FFT on buffer t of length 2*M.
    Assumes t[0:M] are real input (first half initialized by caller), t[M:2*M] are real outputs.
    For k = 1, 2, 4, ..., M:
      For p = 0..k//2-1:
        j = i + k//2
        dist = M - k
        tw_real = +cos(2*pi*p/k), tw_imag = -sin(2*pi*p/k)
        y1 = t[M + i], y2 = t[M + j]
        t[M + i] = y1 * tw_real - y2 * tw_imag
        t[M + j] = y2 * tw_real + y1 * tw_imag
    We compute these updates in-place on t[M + i] and t[M + j], using masks to avoid out-of-bounds.
    """
    k = 1
    while k < M:
        halfk = k // 2
        kk = k
        p = 0
        while p < halfk:
            # For each i in [0, M) with j = i + k, only process i < M - k (so j < M)
            # We process all i, but updates only affect pairs (i, j) where j < M.
            i = tl.program_id(axis=0)
            j = i + kk
            # mask: j < M

            tw_imag = -tl.sin(theta)

            y1 = tl.load(t_ptr + M + i, mask=mask, other=0.0)
            y2 = tl.load(t_ptr + M + j, mask=mask, other=0.0)
            # Update in-place
            new_i = y1 * tw_real - y2 * tw_imag
            new_j = y2 * tw_real + y1 * tw_imag
            tl.store(t_ptr + M + i, new_i, mask=mask)
            tl.store(t_ptr + M + j, new_j, mask=mask)
            p += 1
        k *= 2


@triton.jit
def normalize_divide_real_kernel(inp_ptr, out_ptr, n_elements, scale, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise division by scale. Writes out_ptr = inp_ptr / scale.
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    x = tl.load(inp_ptr + offsets, mask=mask, other=0.0)
    y = x / scale
    tl.store(out_ptr + offsets, y, mask=mask)
