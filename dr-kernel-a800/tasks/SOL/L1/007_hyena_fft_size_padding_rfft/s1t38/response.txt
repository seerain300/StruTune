    stride_out_b, stride_out_c, stride_out_l,  # strides for out
    N: tl.constexpr,        # nfft = 2 * L (runtime int)
):

    b = pid // C
    c = pid % C

    out_base = out_real_ptr + b * stride_out_b + c * stride_out_c

        sum_imag = 0.0
