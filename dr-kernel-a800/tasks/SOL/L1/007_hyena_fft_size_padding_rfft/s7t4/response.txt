import torch
import triton
import triton.language as tl

# Required Triton kernel: computes normalized real FFT of a length N_in input with zero-padding beyond seqlen,
# and writes out the first seqlen+1 complex coefficients (matching torch.rfft(x, n=2*seqlen).real/imag).
# One program per output frequency index k in [0, seqlen + 1).
@triton.jit
def _real_rfft_pad_to_2N_kernel(x_ptr, real_out_ptr, imag_out_ptr, seqlen, scale):
    """
    Triton kernel to emulate torch.fft.rfft(x, n=2*seqlen) with zero-padding and write normalized real/imag parts.
    We conceptually zero-pad to N_in = 2*seqlen; output length is L_out = N_in//2 + 1 = seqlen + 1.
    This kernel computes only the first L_out outputs (which equals seqlen + 1) and normalizes by 2*seqlen.
    """
    # pid = k index in output
    k = tl.program_id(axis=0)

    # Accumulators for real and imaginary parts
    acc_real = 0.0
    acc_imag = 0.0

    # N_in = 2 * seqlen
    N_in = 2 * seqlen

    # Direct summation over n = 0..N_in-1 with zero-padding beyond seqlen
    # y[k] = sum_{n=0}^{N_in-1} x[n] * (cos(2*pi*k*n/N_in) + i*sin(2*pi*k*n/N_in))
    # Because k <= seqlen (since we launch grid=(seqlen+1,)), sin/cos are safe and small.
    for n in range(0, N_in):
        # Zero-pad: for n >= seqlen, x[n] is 0
        x_n = tl.load(x_ptr + n, mask=(n < seqlen), other=0.0)
        angle = 2.0 * tl.math.pi * k * n / N_in

        sin_term = tl.sin(angle)

    acc_imag = acc_imag * scale

    # Store outputs; k in [0, seqlen + 1)
    tl.store(real_out_ptr + k, acc_real)
    tl.store(imag_out_ptr + k, acc_imag)
