@triton.jit
def cast_to_float32_inplace_kernel(x_ptr, n_elements: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    """
    In-place cast of tensor x_ptr from whatever dtype to float32.
    Assumes x_ptr points to a contiguous tensor and we can reinterpret and store float32.
    This is a data movement kernel; it does not perform heavy math.
    """

    # In practice, we store values as float32 by constructing a float32 tensor. Triton supports
    # casting via tl.astype:

    We pair indices i in [0, S) with rev in [S, 2*S).

      real[i] <-> real[rev], imag[i] <-> imag[rev]
      real[S+i] <-> real[S+rev], imag[S+i] <-> imag[S+rev]
    """
    HALF = S
    i = tl.program_id(axis=0)
    while i < HALF:
        rev = tl.zeros((), dtype=tl.int32)
        j = tl.zeros((), dtype=tl.int32)
