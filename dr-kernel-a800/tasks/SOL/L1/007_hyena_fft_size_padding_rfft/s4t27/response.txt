import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded_kernel(x_ptr, out_ptr, L: tl.int32, N: tl.int32):
    # Each program handles one (bc) row: pid_bc in [0, B*C)
    pid_bc = tl.program_id(0)
    # Copy x[pid_bc, :L] into out[pid_bc, 0:L]
    # N is even and >= L
    for i in tl.static_range(0, L):
        val = tl.load(x_ptr + pid_bc * L + i)
        tl.store(out_ptr + pid_bc * N + i, val)
    # Fill the rest with zeros
    for i in tl.static_range(L, N):
        tl.store(out_ptr + pid_bc * N + i, 0.0)


@triton.jit
def _direct_rfft_flat_kernel(padded_ptr, out_real_ptr, out_imag_ptr, L: tl.int32, N: tl.int32, BC: tl.int32):
    # This kernel computes for each (bc, k) the sum over j of padded[j] * cos(2*pi*k*j/N)
    # and padded[j] * sin(2*pi*k*j/N), where k = program_id(1) in [0, L].
    pid_bc = tl.program_id(0)
    k = tl.program_id(1)

    acc_real = 0.0
    acc_imag = 0.0

    # Sum over j=0..N-1 in chunks for better performance
    BLOCK_J = 256
    for j0 in tl.static_range(0, N, BLOCK_J):
        j = j0 + tl.arange(0, BLOCK_J)
        valid = j < N
        vals = tl.load(padded_ptr + pid_bc * N + j, mask=valid, other=0.0)
        # angle = 2*pi*k*j/N
        angle = 2.0 * 3.141592653589793 * (k * j) / N
        cosv = tl.cos(angle)
        sinv = tl.sin(angle)
        vals = vals * valid  # ensure valid elements contribute
        acc_real += tl.sum(vals * cosv, axis=0)
        acc_imag += tl.sum(vals * sinv, axis=0)

    # Normalize by N
    inv_N = 1.0 / N
    acc_real = acc_real * inv_N
    acc_imag = acc_imag * inv_N

    # Store results at index k in (L+1) output
    # out_real/imag are laid out as (BC, L+1) contiguous
    tl.store(out_real_ptr + pid_bc * (L + 1) + k, acc_real)
    tl.store(out_imag_ptr + pid_bc * (L + 1) + k, acc_imag)


@triton.jit
def _divide_by_scalar_kernel(in_ptr, out_ptr, numel: tl.int32, scale: tl.float32, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    x = x / scale
    tl.store(out_ptr + offs, x, mask=mask)

        padded_x = torch.empty((BC, N), dtype=torch.float32, device=x.device)

        grid_copy = (BC,)
        _copy_row_to_padded_kernel[grid_copy](x.view(BC, L), padded_x, L, N)

        # Allocate outputs: (BC, L+1) float32
        out_real = torch.empty((BC, L + 1), dtype=torch.float32, device=x.device)
        out_imag = torch.empty((BC, L + 1), dtype=torch.float32, device=x.device)

        _direct_rfft_flat_kernel[grid_rfft](padded_x, out_real, out_imag, L, N, BC)

        # Reshape back to (B, C, L+1)
        out_real = out_real.view(B, C, L + 1)
        out_imag = out_imag.view(B, C, L + 1)

        # Normalize by N (2*L) as done in the original code
        # Use Triton kernel to divide (in-place on a temporary)
        tmp_real = torch.empty_like(out_real)
        tmp_imag = torch.empty_like(out_imag)
        numel_real = out_real.numel()
        numel_imag = out_imag.numel()

        grid_norm = (triton.cdiv(numel_real, BLOCK),)

        _divide_by_scalar_kernel[grid_norm](out_imag, tmp_imag, numel_imag, float(N), BLOCK=BLOCK)

        # Assign results
        out_real = tmp_real
        out_imag = tmp_imag

        return out_real, out_imag

    acc_imag = 0.0
