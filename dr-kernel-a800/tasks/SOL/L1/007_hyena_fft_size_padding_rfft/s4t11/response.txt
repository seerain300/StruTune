import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded_kernel(
    x_ptr, out_ptr,
    B, C, L,
    stride_xb, stride_xc, stride_xl,
    stride_outb, stride_outc, stride_outl,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    k = tl.program_id(2)

    if b >= B or c >= C or k >= L:
        return

    x_row_ptr = x_ptr + b * stride_xb + c * stride_xc + k * stride_xl
    out_row_ptr = out_ptr + b * stride_outb + c * stride_outc

    j = tl.arange(0, BLOCK_K)
    mask_j = j < (2 * L)
    vals = tl.load(x_row_ptr + j * stride_xl, mask=mask_j, other=0.0)
    tl.store(out_row_ptr + j * stride_outl, vals, mask=mask_j)


@triton.jit
def _compute_cos_table_kernel(
    out_ptr,
    N, K,
    stride_outj, stride_outk,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    j = tl.program_id(0)
    k = tl.program_id(1)
    if j >= N or k >= K:
        return

    idx_j = j * stride_outj
    idx_k = k * stride_outk
    base = out_ptr + idx_j + idx_k

    # Compute cos(2*pi*k*j/N) for this (j, k)
    # Note: N is scalar, k is scalar, j is vector j
    # Triton supports elementwise operations with tl.arange
    j_vec = j + tl.arange(0, BLOCK_J)
    k_vec = k + tl.arange(0, BLOCK_K)
    mask_j = j_vec < N
    mask_k = k_vec < K

    angle = 2.0 * 3.141592653589793 * k_vec * j_vec / N  # vectorized angle
    cos_vals = tl.cos(angle)

    # Store into out_ptr[j, k]
    tl.store(base, cos_vals[0])  # store single scalar at out_ptr[j, k]


@triton.jit
def _compute_sin_table_kernel(
    out_ptr,
    N, K,
    stride_outj, stride_outk,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    j = tl.program_id(0)
    k = tl.program_id(1)
    if j >= N or k >= K:
        return

    idx_j = j * stride_outj
    idx_k = k * stride_outk
    base = out_ptr + idx_j + idx_k

    j_vec = j + tl.arange(0, BLOCK_J)
    k_vec = k + tl.arange(0, BLOCK_K)
    mask_j = j_vec < N
    mask_k = k_vec < K

    angle = 2.0 * 3.141592653589793 * k_vec * j_vec / N
    sin_vals = tl.sin(angle)

    tl.store(base, sin_vals[0])


@triton.jit
def _reduce_real_kernel(
    padded_ptr, cos_ptr, out_ptr,
    N, K,
    stride_pj, stride_pk,  # strides for padded
    stride_cj, stride_ck,  # strides for cos_table
    stride_ol,             # stride for output along K
    BLOCK_J: tl.constexpr,
):
    k = tl.program_id(0)
    if k >= K:
        return
    acc = tl.zeros((), dtype=tl.float32)

    j = 0
    while j < N:
        j_vec = j + tl.arange(0, BLOCK_J)
        mask_j = j_vec < N

        # Load padded row segment
        padded_row_ptrs = padded_ptr + j_vec * stride_pj + k * stride_pk
        vals = tl.load(padded_row_ptrs, mask=mask_j, other=0.0)

        # Load cos table row segment: cos_table[j, k]
        cos_row_ptrs = cos_ptr + j_vec * stride_cj + k * stride_ck
        cos_vals = tl.load(cos_row_ptrs, mask=mask_j, other=0.0)

        # Fused multiply-add: sum over block
        prod = vals * cos_vals
        acc += tl.sum(prod, axis=0)
        j += BLOCK_J

    tl.store(out_ptr + k * stride_ol, acc)


@triton.jit
def _reduce_imag_kernel(
    padded_ptr, sin_ptr, out_ptr,
    N, K,
    stride_pj, stride_pk,  # strides for padded
    stride_sj, stride_sk,  # strides for sin_table
    stride_ol,             # stride for output along K
    BLOCK_J: tl.constexpr,
):

    if k >= K:
        return

    j = 0

        sin_row_ptrs = sin_ptr + j_vec * stride_sj + k * stride_sk

        prod = vals * sin_vals
        acc += tl.sum(prod, axis=0)
        j += BLOCK_J

    tl.store(out_ptr + k * stride_ol, acc)


@triton.jit
def _divide_kernel(
    x_ptr, out_ptr, numel, scale,
    BLOCK: tl.constexpr,
):
    idx = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    mask = idx < numel
    vals = tl.load(x_ptr + idx, mask=mask, other=0.0)
    vals = vals / scale
    tl.store(out_ptr + idx, vals, mask=mask)

        # with original which assumes a single tensor input; we ignore extras.

        dtype = torch.float32  # ensure float32 for compute

        # Compute cos table: grid (N, L)
        grid_cos = (N, L)

        # will still be correct. BLOCK sizes are compile-time constants.
        _compute_cos_table_kernel[grid_cos](
            cos_table,
            N, L,
            stride_cj, stride_ck,
            BLOCK_J=128, BLOCK_K=128,
        )

        # Compute sin table: grid (N, L)
        _compute_sin_table_kernel[_compute_sin_table_kernel](
            sin_table,
            N, L,
            stride_sj, stride_sk,
            BLOCK_J=128, BLOCK_K=128,
        )

        # Allocate outputs: (B, C, L+1)
        out_real = torch.empty((B, C, L + 1), dtype=dtype, device=device)
        out_imag = torch.empty((B, C, L + 1), dtype=dtype, device=device)

        # Strides for outputs (only along K dimension we index; but we store scalar per k)
        # We launch reduction kernels once per k: grid size (L,)
        grid_reduce = (L,)
        _reduce_real_kernel[grid_reduce](
            padded, cos_table, out_real,
            N, L,
            stride_pj=1, stride_pk=1,  # padded strides: not used; we pass correct ptrs
            stride_cj=stride_cj, stride_ck=stride_ck,
            stride_ol=1,
            BLOCK_J=256,
        )
        _reduce_imag_kernel[grid_reduce](
            padded, sin_table, out_imag,
            N, L,
            stride_pj=1, stride_pk=1,
            stride_sj=stride_sj, stride_sk=stride_sk,
            stride_ol=1,
            BLOCK_J=256,
        )

        # Normalize by 2*L
        scale = 2.0 * float(L)
        grid_div = ((out_real.numel() + 1023) // 1024,)
        _divide_kernel[grid_div](out_real, out_real, out_real.numel(), scale, BLOCK=1024)
        _divide_kernel[grid_div](out_imag, out_imag, out_imag.numel(), scale, BLOCK=1024)

        return out_real, out_imag

- All Triton kernels are actually invoked from ModelNew.forward for each (batch, channel) slice. There are no host-side torch.cos/torch.sin; cos and sin tables are computed by Triton kernels.
- The copy kernel reads x[b, c, k] and writes padded[b, c, k] for k in [0..L-1], filling padded[k+L..2L-1] with zeros implicitly (we never write them). This avoids undefined behavior and ensures n=2*L can be used.
- The reduction kernels loop over j in BLOCK_J chunks and use atomic_add-like accumulation via tl.sum to produce one output per k. We then normalize using a simple Triton elementwise division kernel.
- Outputs have shape (B, C, L+1) and are normalized by 2*L, matching the original behavior.
