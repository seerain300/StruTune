import torch
import triton
import triton.language as tl

# Triton kernel: build zero-padded flattened input for (B, C, L) slices.
# We flatten all (B*C) slices into a single vector of length M = (B*C) * (2*L).
@triton.jit
def pad_kernel(
    x_ptr,         # *float32, input x of shape (B, C, L) contiguous
    out_ptr,       # *float32, output flattened padded vector of length M
    B: tl.constexpr,
    C: tl.constexpr,
    L: tl.constexpr,
    two_L: tl.constexpr,  # = 2*L
    stride_x_bc: tl.constexpr,  # = L (since x is (B,C,L) contiguous, bc-slice is length L)
    stride_out_bc: tl.constexpr, # = 2*L (each (b,c) slice occupies 2*L in flattened out)
):
    pid = tl.program_id(0)  # one program per (b, c)
    # Map pid to (b, c)
    b = pid // C
    c = pid % C

    # Base offsets for input and output
    base_x = b * C * L + c * L
    base_out = (b * C + c) * two_L

    # Copy first L elements (actual data)
    # We'll do a scalar loop to be safe with Triton addressing constraints.
    for t in range(0, L):
        val = tl.load(x_ptr + base_x + t)
        tl.store(out_ptr + base_out + t, val)

    # Zero pad the remaining L elements
    for t in range(0, L):
        tl.store(out_ptr + base_out + L + t, 0.0)


# Triton kernel: compute real DFT over zero-padded input of length two_L,
# store normalized real and imaginary parts for k in [0..L] (output length = L+1).
@triton.jit
def real_dft_kernel(
    x_ptr,           # *float32, flattened padded input of length M = (B*C)*two_L
    out_real_ptr,    # *float32, output real part of length M_out = (B*C)*(L+1)
    out_imag_ptr,    # *float32, output imag part of length M_out = (B*C)*(L+1)
    B: tl.constexpr,
    C: tl.constexpr,
    L: tl.constexpr,        # actual data length
    two_L: tl.constexpr,    # = 2*L, total padded length
    inv_two_L,              # normalization factor = 1.0 / (2*L), float32
):
    b = tl.program_id(0)  # b in [0, B)
    c = tl.program_id(1)  # c in [0, C)
    k = tl.program_id(2)  # k in [0, L]

    # Compute base indices
    bc = b * C + c
    base_in = bc * two_L   # starts reading padded input at this offset
    base_out = bc * (L + 1)

    # Accumulator for real and imag parts
    accum_real = 0.0
    accum_imag = 0.0

    # Compute DFT sum: X[k] = sum_{t=0}^{2*L-1} x[t] * exp(-2πi k t / (2*L))
    # Only t in [0, 2*L-1] matters. Input is zero-padded to two_L length.
    for t in range(0, two_L):
        x_t = tl.load(x_ptr + base_in + t)
        # angle = -2*pi*k*t / (2*L)
        angle = -2.0 * 3.141592653589793 * float(k) * float(t) * inv_two_L
        # cos and sin contributions
        cosv = tl.cos(angle)
        sinv = tl.sin(angle)
        accum_real += x_t * cosv
        accum_imag += x_t * sinv

    # Normalize
    accum_real *= inv_two_L
    accum_imag *= inv_two_L

    # Store outputs (real/imag parts). For real inputs, imag should be 0, but we store normalized result.
    tl.store(out_real_ptr + base_out + k, accum_real)
    tl.store(out_imag_ptr + base_out + k, accum_imag)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):

        assert x.ndim == 3, "Input must be a 3D tensor (B, C, L)"

        two_L = 2 * L

        # Ensure contiguous float32
        x = x.contiguous().to(torch.float32)

        # 1) Prepare flattened zero-padded input for Triton
        # Total elements M = (B*C) * two_L
        M = (B * C) * two_L
        padded = torch.empty(M, dtype=torch.float32, device=x.device)

        # Launch pad kernel: one program per (b, c)
        grid_pad = (B * C,)
        pad_kernel[grid_pad](
            x, padded,
            B, C, L, two_L,
            L,  # stride_x_bc
            two_L,  # stride_out_bc
            num_warps=4,
        )

        # 2) Allocate outputs for real/imag parts, flattened of length M_out = (B*C)*(L+1)
        M_out = (B * C) * (L + 1)
        out_real = torch.empty(M_out, dtype=torch.float32, device=x.device)
        out_imag = torch.empty(M_out, dtype=torch.float32, device=x.device)

        # 3) Launch real DFT kernel over grid (B, C, L)
        grid_dft = (B, C, L)
        inv_two_L = 1.0 / float(two_L)
        real_dft_kernel[grid_dft](
            padded, out_real, out_imag,
            B, C, L, two_L,
            inv_two_L,
            num_warps=4,
        )

        # 4) Reshape to (B, C, L+1) and return
        x_freq_real = out_real.view(B, C, L + 1)
        x_freq_imag = out_imag.view(B, C, L + 1)
        return x_freq_real, x_freq_imag
