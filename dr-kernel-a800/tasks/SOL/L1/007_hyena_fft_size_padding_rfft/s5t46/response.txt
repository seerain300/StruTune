import torch
import triton
import triton.language as tl


@triton.jit
def rfft_real_kernel(x_ptr, out_ptr,
                      N, seqlen, invN,
                      BLOCK_K: tl.constexpr):
    """
    Compute real part of rfft for one row:
      real_out[j] = sum_{k=0..N-1} x[k] * cos(2*pi*j*k/N) * invN,  j in 0..seqlen.
    x_ptr points to the padded input vector of length N (float32).
    out_ptr points to output float32 vector of length seqlen+1 (we will write j=0..seqlen).
    invN = 1.0 / N.
    """
    # One program per row
    pid = tl.program_id(0)
    # j loop
    for j in range(0, seqlen + 1):  # j goes 0..seqlen
        acc = 0.0
        # accumulate over k in chunks
        for k in range(0, N, BLOCK_K):
            k_offsets = k + tl.arange(0, BLOCK_K)
            mask = k_offsets < N
            x_vals = tl.load(x_ptr + pid * N + k_offsets, mask=mask, other=0.0)
            # angle = 2*pi*j*k/N
            angle = (2.0 * 3.141592653589793 * j * k_offsets) / N
            cos_term = tl.cos(angle)
            acc += tl.sum(x_vals * cos_term, axis=0)
        acc *= invN
        tl.store(out_ptr + pid * (seqlen + 1) + j, acc)

def rfft_imag_kernel(x_ptr, out_ptr,
                      N, seqlen, invN,
                      BLOCK_K: tl.constexpr):

    Compute imaginary part of rfft for one row:
      imag_out[j] = sum_{k=0..N-1} x[k] * sin(2*pi*j*k/N) * invN,  j in 1..seqlen-1.
