import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded_kernel(x_ptr, padded_ptr, L: tl.int32):
    # Each program handles one (b, c) row: x_ptr points to the start of the row, length L
    # padded_ptr points to the start of the corresponding padded row, length N=2*L
    # We only copy the first L elements; padded[0:L] gets x[0:L], padded[L:] remains zero (host pre-zeroed)
    pid = tl.program_id(0)  # one program per (b, c) slice
    for i in tl.static_range(0, L):
        val = tl.load(x_ptr + i)
        tl.store(padded_ptr + i, val)


@triton.jit
def _accumulate_real_kernel(padded_ptr, out_real_ptr, L: tl.int32, N: tl.int32, K: tl.int32):
    # Accumulate y_real[K] = sum_{j=0}^{N-1} padded[j] * cos(2*pi*K*j/N)
    # This kernel runs with program_id(0) == K, i.e., one program per k
    inv_N = 1.0 / N
    acc = 0.0
    for j in tl.static_range(0, N):
        v = tl.load(padded_ptr + j)
        angle = 2.0 * 3.141592653589793 * (K * j) / N
        c = tl.cos(angle)
        acc += v * c
    acc = acc * inv_N  # normalize by N
    tl.store(out_real_ptr + K, acc)
