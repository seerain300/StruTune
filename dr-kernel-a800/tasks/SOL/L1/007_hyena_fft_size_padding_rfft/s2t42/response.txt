import torch
import triton
import triton.language as tl

# Triton kernel: build zero-padded flattened input for all (b, c) slices.
# out_ptr: flattened output buffer of length M = (B*C) * (2*L)
# x_ptr: input x of shape (B, C, L), contiguous float32
@triton.jit
def pad_kernel(
    x_ptr,         # *float32, input tensor as 1D contiguous
    out_ptr,       # *float32, flattened output buffer
    B: tl.constexpr,
    C: tl.constexpr,
    L: tl.constexpr,
    two_L: tl.constexpr,
):
    pid = tl.program_id(0)  # one program per (b, c)
    # Map pid to (b, c)
    b = pid // C
    c = pid % C

    # Base offset for this (b, c) slice in the flattened output
    base = (b * C + c) * two_L

    # Copy x[b, c, :] into out_ptr[base : base + L]
    # Scalar loop is fine and robust
    for t in range(0, L):
        # address in x_ptr for x[b, c, t] is (b*C + c)*L + t
        x_addr = (b * C + c) * L + t
        val = tl.load(x_ptr + x_addr)
        tl.store(out_ptr + base + t, val)

    # Write zeros for the next L positions
    for t in range(0, L):
        tl.store(out_ptr + base + L + t, 0.0)


# Triton kernel: compute DFT for each (b, c, k), k in [0..L], using the padded input.
# out_real_ptr, out_imag_ptr: flattened output buffers of length M = (B*C)*(L+1)
@triton.jit
def dft_real_kernel(
    out_ptr,               # *float32, flattened padded input of length (B*C) * (2*L)
    out_real_ptr,          # *float32, flattened output real of length (B*C) * (L+1)
    out_imag_ptr,          # *float32, flattened output imag of length (B*C) * (L+1)
    B: tl.constexpr,
    C: tl.constexpr,
    L: tl.constexpr,
    two_L: tl.constexpr,
):
    # Grid dimension is set to B*C*k_max; one program per (b, c, k)
    pid = tl.program_id(0)

    # Derive (b, c, k) from pid
    # Note: pid ranges over 0 .. (B*C*L) - 1
    k = pid % (L + 1)
    tmp = pid // (L + 1)
    c = tmp % C
    b = tmp // C

    # Accumulate X[k] = sum_{t=0}^{2*L-1} x[t] * exp(-2πi k t / two_L)
    acc_real = 0.0
    acc_imag = 0.0

    # We can compute cos/sin per t and accumulate
    for t in range(0, two_L):
        # Load x[t] for this (b, c)
        base = (b * C + c) * two_L
        x_t = tl.load(out_ptr + base + t)
        # angle = -2π * k * t / two_L
        angle = -2.0 * 3.141592653589793 * (k * t) / two_L
        coss = tl.cos(angle)
        sins = tl.sin(angle)
        # Accumulate in complex form
        acc_real += x_t * coss
        acc_imag += x_t * (-sins)

    # Normalize by two_L (original code divides by 2*L)
    norm = 1.0 / two_L
    acc_real = acc_real * norm
    acc_imag = acc_imag * norm

    # Store results to flattened output at index pid
    tl.store(out_real_ptr + pid, acc_real)
    tl.store(out_imag_ptr + pid, acc_imag)
