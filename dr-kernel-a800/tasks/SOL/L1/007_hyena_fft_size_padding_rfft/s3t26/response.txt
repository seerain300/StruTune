import torch
import triton
import triton.language as tl


@triton.jit
def pad_to_2L_kernel(
    x_ptr,                # *f32, input (B, C, L)
    x_padded_ptr,         # *f32, output (B, C, 2L)
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,
    stride_b: tl.int32,
    stride_c: tl.int32,
    stride_l: tl.int32,
    out_stride_b: tl.int32,
    out_stride_c: tl.int32,
    out_stride_l: tl.int32,
    BLOCK_N: tl.constexpr,
):
    # One Triton program per (batch, channel)
    pid = tl.program_id(axis=0)
    b = pid // channels
    c = pid % channels

    base_in = b * stride_b + c * stride_c
    base_out = b * out_stride_b + c * out_stride_c
    twoL = 2 * L

    # Copy first L elements
    start = 0
    while start < L:
        j = start + tl.arange(0, BLOCK_N)
        mask = j < L
        vals = tl.load(x_ptr + base_in + j * stride_l, mask=mask, other=0.0)
        tl.store(x_padded_ptr + base_out + j * out_stride_l, vals, mask=mask)
        start += BLOCK_N

    # Fill remaining positions with zeros
    zero_vec = tl.zeros([BLOCK_N], dtype=tl.float32)
    start = L
    while start < twoL:
        j = start + tl.arange(0, BLOCK_N)
        mask = j < twoL
        tl.store(x_padded_ptr + base_out + j * out_stride_l, zero_vec, mask=mask)
        start += BLOCK_N


@triton.jit
def rfft_direct_kernel(
    x_padded_ptr,         # *f32, (B, C, 2L) input
    real_out_ptr,         # *f32, (B, C, L+1) output for real part
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,
    twoL: tl.int32,       # 2 * L
    out_stride_b: tl.int32,
    out_stride_c: tl.int32,
    out_stride_l: tl.int32,
    BLOCK_K: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    # 2D grid: axis 0 over (B*C), axis 1 over k-block
    pid0 = tl.program_id(axis=0)
    pid1 = tl.program_id(axis=1)
    b = pid0 // channels
    c = pid0 % channels

    base_in = b * out_stride_b + c * out_stride_c
    base_out = b * out_stride_b + c * out_stride_c  # real_out strides

    # k indices for this block
    k0 = pid1 * BLOCK_K
    k = k0 + tl.arange(0, BLOCK_K)
    k_mask = k < L

    # Accumulator for real part
    acc = tl.zeros([BLOCK_K], dtype=tl.float32)

    # Sum over j = 0..2L-1 in tiles
    j0 = 0
    while j0 < twoL:
        j = j0 + tl.arange(0, BLOCK_J)
        j_mask = j < twoL
        # Load padded input values for this (b, c) slice
        vals = tl.load(x_padded_ptr + base_in + j * out_stride_l, mask=j_mask, other=0.0)  # shape [BLOCK_J], f32

        # Compute angles: angle = 2*pi*k[j_tile]*j/(2L)
        # Note: j is vector, k is vector [BLOCK_K], broadcasting works
        angle = (2.0 * 3.141592653589793) * k[:, None] * j[None, :] / twoL  # shape [BLOCK_K, BLOCK_J]

        # cos/sin
        cos_t = tl.cos(angle)
        sin_t = tl.sin(angle)

        # Contribution: sum over j tile
        # For each k, sum over j of vals[j] * (cos(angle) + i*sin(angle)), but since input is real, imag is zero.
        # We only accumulate real part: vals * cos(angle)
        # Reduction over j axis:
        # We need to multiply each vals[j] with the corresponding cos_t[k, j] and sum over j
        contrib = tl.sum(vals[None, :] * cos_t, axis=1)  # reduce over j -> shape [BLOCK_K]
        acc += contrib

        j0 += BLOCK_J

    # Divide by normalization factor (2*L) per original code
    acc = acc / twoL

    # Store results to real_out at positions k (0..L-1)
    # real_out has shape (B, C, L+1), we only use first L positions
    out_base = b * out_stride_b + c * out_stride_c
    k_store = k  # valid k < L
    tl.store(real_out_ptr + out_base + k_store * out_stride_l, acc, mask=k_mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):

        L = seqlen

            x_padded.stride(0), x_padded.stride(1), x_padded.stride(2),
            BLOCK_N=1024, num_warps=4,
        )

        # Allocate real output: (B, C, L+1)
        real_out = torch.empty((batch, channels, L + 1), dtype=torch.float32, device=x.device)

        # Launch direct rfft kernel:
        # Grid: axis 0 over (B*C), axis 1 over k-blocks of size BLOCK_K
        BLOCK_K = 128
        BLOCK_J = 256
        grid_rfft = (batch * channels, triton.cdiv(L, BLOCK_K))

            BLOCK_K=BLOCK_K, BLOCK_J=BLOCK_J,
            num_warps=4,
        )

        # Imaginary part is zero for real-input rfft; return zeros tensor with correct shape
        imag_out = torch.zeros((batch, channels, L + 1), dtype=torch.float32, device=x.device)

        return real_out, imag_out
