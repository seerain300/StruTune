import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded_kernel(
    x_ptr,              # *f32, input row pointer, shape [L]
    out_ptr,            # *f32, output padded pointer, shape [n]
    L, n,               # int32
    BLOCK_J: tl.constexpr,
):
    # One program copies one element from x to out[0..L-1], leaves out[L..n-1] as zeros (we'll pre-fill zeros).
    # But since we pass out as zeros, we just copy and Triton will not write beyond L; padding remains zeros.
    pid = tl.program_id(axis=0)  # we launch exactly L programs
    # Load from x
    # Note: x_ptr is a 1D pointer, and we access by index directly
    x_val = tl.load(x_ptr + pid)
    # Store to out at index pid
    tl.store(out_ptr + pid, x_val)


@triton.jit
def _compute_cos_table_kernel(
    out_ptr,            # *f32, output cos table pointer, shape [n, L+1]
    n, L,               # int32
    BLOCK_K: tl.constexpr,
):
    # 2D grid: axis 0 over n, axis 1 over L+1
    j = tl.program_id(axis=0)
    k = tl.program_id(axis=1)
    # Guard: if k > L, skip (grid is set to L+1, so no need)
    # Compute cos(2*pi*k*j / n)
    arg = 2.0 * 3.141592653589793 * k * j / n
    cos_val = tl.cos(arg)
    tl.store(out_ptr + j * (L + 1) + k, cos_val)


@triton.jit
def _compute_sin_table_kernel(
    out_ptr,            # *f32, output sin table pointer, shape [n, L+1]
    n, L,               # int32
    BLOCK_K: tl.constexpr,
):
    # 2D grid: axis 0 over n, axis 1 over L+1
    j = tl.program_id(axis=0)
    k = tl.program_id(axis=1)
    arg = 2.0 * 3.141592653589793 * k * j / n
    sin_val = tl.sin(arg)
    tl.store(out_ptr + j * (L + 1) + k, sin_val)


@triton.jit
def _reduce_real_kernel(
    padded_ptr,         # *f32, padded input, shape [n]
    cos_ptr,            # *f32, cos table pointer, shape [n, L+1]
    out_ptr,            # *f32, output real accumulation, shape [L+1]
    n, L,               # int32
    BLOCK_J: tl.constexpr,
):
    # Each program handles one k in [0..L]
    k = tl.program_id(axis=0)
    acc = 0.0
    # Loop over j in blocks
    for start in range(0, n, BLOCK_J):
        j_offsets = start + tl.arange(0, BLOCK_J)
        mask = j_offsets < n
        j = j_offsets
        xj = tl.load(padded_ptr + j, mask=mask, other=0.0)
        cjk = tl.load(cos_ptr + j * (L + 1) + k, mask=mask, other=0.0)
        # Multiply and reduce within the block
        acc += tl.sum(xj * cjk, axis=0)
    # Atomic add into output at index k
    tl.atomic_add(out_ptr + k, acc)


@triton.jit
def _reduce_imag_kernel(
    padded_ptr,         # *f32, padded input, shape [n]
    sin_ptr,            # *f32, sin table pointer, shape [n, L+1]
    out_ptr,            # *f32, output imag accumulation, shape [L+1]
    n, L,               # int32
    BLOCK_J: tl.constexpr,
):
    # Each program handles one k in [0..L]
    k = tl.program_id(axis=0)
    acc = 0.0
    for start in range(0, n, BLOCK_J):
        j_offsets = start + tl.arange(0, BLOCK_J)
        mask = j_offsets < n
        j = j_offsets
        xj = tl.load(padded_ptr + j, mask=mask, other=0.0)
        sjk = tl.load(sin_ptr + j * (L + 1) + k, mask=mask, other=0.0)
        acc += tl.sum(xj * sjk, axis=0)
    tl.atomic_add(out_ptr + k, acc)


@triton.jit
def _normalize_kernel(
    in_ptr,             # *f32, input vector to normalize, shape [L+1]
    out_ptr,            # *f32, output vector, shape [L+1]
    L,                  # int32
    scale,              # f32
    BLOCK: tl.constexpr,
):
    size = L + 1
    grid = (triton.cdiv(size, BLOCK),)
    for i in range(0, size, BLOCK):
        idx = i + tl.arange(0, BLOCK)
        mask = idx < size
        val = tl.load(in_ptr + idx, mask=mask, other=0.0)
        val = val / scale
        tl.store(out_ptr + idx, val, mask=mask)


@triton.jit
def _store_output_kernel(
    real_ptr,           # *f32, real vector, shape [L+1]
    imag_ptr,           # *f32, imag vector, shape [L+1]
    out_real_ptr,       # *f32, final out real, shape [B, C, L+1] with stride
    out_imag_ptr,       # *f32, final out imag, shape [B, C, L+1] with stride
    L,                  # int32
    batch: tl.constexpr,
    channels: tl.constexpr,
    stride_b: tl.constexpr,
    stride_c: tl.constexpr,
    stride_l: tl.constexpr,
):
    # grid over (batch, channel, l)
    b = tl.program_id(axis=0)
    c = tl.program_id(axis=1)
    l = tl.program_id(axis=2)
    # Only l in [0..L] is meaningful for the output
    if l <= L:
        base = b * stride_b + c * stride_c
        off = base + l * stride_l
        tl.store(out_real_ptr + off, tl.load(real_ptr + l))
        tl.store(out_imag_ptr + off, tl.load(imag_ptr + l))

        # 5) Normalize by 2*L
        real_norm = torch.empty((L + 1,), dtype=torch.float32, device=device)
        imag_norm = torch.empty((L + 1,), dtype=torch.float32, device=device)
        scale = 2.0 * float(L)
        _normalize_kernel[(1,)](real_acc, real_norm, L, scale, BLOCK=256)
        _normalize_kernel[(1,)](imag_acc, imag_norm, L, scale, BLOCK=256)

        out_real = torch.empty((B, C, L + 1), dtype=torch.float32, device=device)
        out_imag = torch.empty((B, C, L + 1), dtype=torch.float32, device=device)
