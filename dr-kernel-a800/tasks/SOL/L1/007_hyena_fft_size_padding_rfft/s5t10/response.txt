import torch
import triton
import triton.language as tl


@triton.jit
def cosine_sum_kernel(x_ptr, out_ptr, j, N, stride_row):
    """
    Compute real_rfft[j] for j in [0, M], where M=seqlen and N=2*seqlen.
    real_out[j] = (1/N) * sum_{k=0..N-1} x[k] * cos(2*pi*j*k/N)
    out_ptr is a flattened buffer with per-(b,c) row starting at base offset = b*stride_row + c*(M+1).
    We write result to out_ptr[base + j].
    """
    # acc scalar
    acc = 0.0
    # Sum over k = 0..N-1 (x_ptr has length N with zeros already)
    for k in range(0, N):
        xk = tl.load(x_ptr + k)
        angle = 2.0 * 3.141592653589793 * j * k / N
        acc += xk * tl.cos(angle)
    # Normalize by N (original code divides by 2*seqlen; here N == 2*seqlen)
    acc *= 1.0 / N
    # Store into out_ptr at index base + j. We need to provide base; Triton receives stride_row and
    # will compute per (b,c) row. Since out_ptr is flattened to (B*C, M+1), the caller will pass
    # the correct base offset for each (b,c) row.
    # We'll assume out_ptr points to the start of (b,c) row and we write at offset j.
    # Triton expects the pointer with the base offset; the caller must pass the correct pointer per row.
    # Here, we store at out_ptr + j. The caller will set out_ptr to point at row start, so this is valid.
    # Triton doesn't have 'base' as a runtime parameter; we rely on the caller passing the correct out_ptr for each row.
    tl.store(out_ptr + j, acc)


@triton.jit
def sine_sum_kernel(x_ptr, out_ptr, j, N, stride_row):
    """
    Compute imag_rfft[j] for j in [1, M-1], where M=seqlen and N=2*seqlen.
    imag_out[j] = (1/N) * sum_{k=0..N-1} x[k] * sin(2*pi*j*k/N)
    Writes to out_ptr at index base + j.
    """
    acc = 0.0
    for k in range(0, N):
        xk = tl.load(x_ptr + k)
        angle = 2.0 * 3.141592653589793 * j * k / N
        acc += xk * tl.sin(angle)
    acc *= 1.0 / N
    tl.store(out_ptr + j, acc)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        Triton-only implementation of:
            x_freq = torch.fft.rfft(x, n=2*seqlen)  # pads to 2*seqlen internally
            x_freq = x_freq / (2*seqlen)
            return real(x_freq), imag(x_freq)

            real_out: (B, C, seqlen+1), float32
            imag_out: (B, C, seqlen+1), float32
        """

        B, C, M = x.shape  # batch, channels, seqlen
        N = 2 * M  # padded length

        # Cast to float32 for numerical stability and consistency with original code
        x_in = x.to(torch.float32)

        # Place original data at the beginning
        x_padded[:, :, :M] = x_in

        # Allocate outputs (B, C, M+1), zero-initialized
        real_out = torch.zeros((B, C, M + 1), dtype=torch.float32, device=x.device)
        imag_out = torch.zeros((B, C, M + 1), dtype=torch.float32, device=x.device)

        # Flatten pointers for per-(b,c) row writes
        # We'll launch kernels per row: total rows = B * C
        # stride_row = M + 1 is the number of columns per row
        rows = B * C
        stride_row = M + 1

        # Launch cosine sum kernels for j in 0..M-1 (inclusive)
        for j in range(M + 1):  # j = 0..seqlen
            # Compute base pointer: for row index r in [0, rows), base = r * (M+1)
            # We'll pass out_ptr pointing to the start of row r in real_out and imag_out.
            # However, Triton expects a single pointer and we wrote at offset j; here we pass real_out.view(-1) and imag_out.view(-1).
            # To keep correct row mapping, we compute row index as integer division:
            # But Triton kernels here only write to a single offset per call, not per row. We need a per-row write.
            # Therefore, we need to pass the correct out_ptr for each row. Triton does not provide per-thread row id, but we can emulate by launching with a grid that maps to rows.
            # To do this, we call kernels explicitly per row in Python.
            for r in range(rows):
                b = r // C
                c = r % C
                # Compute base offset for this (b,c) row in flattened outputs
                base_real = b * C * stride_row + c * stride_row  # equivalent to b*(C*(M+1)) + c*(M+1)
                base_imag = b * C * stride_row + c * stride_row

                # Prepare per-row pointers: since out tensor is (B*C, M+1), we can flatten and set out_ptr to base + offset j
                out_ptr_real = real_out.view(-1) + base_real
                out_ptr_imag = imag_out.view(-1) + base_imag

                # x_ptr per row: x_padded has shape (B, C, N); we index by (b, c, :)
                x_row_ptr = x_padded[b, c, :]

                # Cast stride_row to int for Triton
                cosine_sum_kernel[(1,)](x_row_ptr, out_ptr_real, j, N, stride_row)

                if 1 <= j <= M - 1:

        # Normalize by N = 2 * seqlen (original code divides by 2*seqlen)
        real_out = real_out / float(N)
        imag_out = imag_out / float(N)

        return real_out, imag_out
