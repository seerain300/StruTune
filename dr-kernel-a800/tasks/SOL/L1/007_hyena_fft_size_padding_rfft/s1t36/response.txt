import triton
import triton.language as tl


@triton.jit
def _rfft_real_imag_triton_kernel(
    x_ptr,                  # *const float32, input pointer to x (B, C, L) contiguous
    out_real_ptr,           # *float32, output pointer to real part (B, C, L+1)
    out_imag_ptr,           # *float32, output pointer to imag part (B, C, L+1)
    B: tl.int32,            # batch size (runtime int, used in pointer arithmetic)
    C: tl.int32,            # channels (runtime int, used in pointer arithmetic)
    L: tl.int32,            # seqlen (runtime int)
    SCALE: tl.float32,      # 1.0 / (2 * L) normalization factor
):
    # Each program handles one (b, c) slice
    pid = tl.program_id(axis=0)
    # Decode b, c from linear pid
    # We assume pid ranges exactly over B*C.
    b = pid // C
    c = pid % C

    # Compute N and M
    N = 2 * L
    M = L + 1

    # Base offsets for this (b, c) slice
    # x is laid out as (B, C, L) contiguous => index = ((b*C + c) * L) + t
    # out tensors are (B, C, M) contiguous => index = ((b*C + c) * M) + j
    # We keep base offsets as int32 for pointer arithmetic
    base_in = (b * C + c) * L
    base_out = (b * C + c) * M

    # Loop over j = 0 .. M-1, compute re_j and im_j
    j = 0
    while j < M:
        sum_real = 0.0
        sum_imag = 0.0

        # Loop over t = 0 .. N-1
        t = 0
        while t < N:
            # Load x[t] as float32
            x_val = tl.load(x_ptr + base_in + t)
            x_val = x_val.to(tl.float32)

            # Compute cos and sin with scaling by 1/N = SCALE
            # angle = 2*pi*j*t/N = 2*pi * j * t / (2*L) = pi * j * t / L
            angle = 3.141592653589793 * j * t * SCALE  # since SCALE = 1 / (2*L), and cos/sin need 2*pi factor: 2*pi = 6.283185307...

            # Note: Triton supports tl.cos and tl.sin; ensure constants are float32
            angle = angle  # already float32
            cos_term = tl.cos(angle)
            sin_term = tl.sin(angle)

            # Accumulate
            sum_real += x_val * cos_term
            sum_imag += x_val * sin_term

            t += 1

        # Normalize by (2*L)
        sum_real *= SCALE
        sum_imag *= SCALE

        # Store results to out_real and out_imag at index (b, c, j)
        # out tensors are contiguous in last dim (M)
        tl.store(out_real_ptr + base_out + j, sum_real)
        tl.store(out_imag_ptr + base_out + j, sum_imag)

        j += 1
