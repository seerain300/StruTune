@triton.jit
def bitreverse_complex_kernel(treal_ptr, tim_ptr, N: tl.constexpr, HALF: tl.constexpr):
    """
    Bit-reverse pairs for complex time-domain vector t of length N = 2*seqlen.
    t is stored as interleaved [real, imag]: t[2*i] = real[i], t[2*i+1] = imag[i].
    We bit-reverse the first half indices i in [0, HALF) by swapping with their
    reversed partners at indices (N-1 - i), which lies in [HALF, N).
    """
    i = tl.program_id(axis=0)
    while i < HALF:

        rev = tl.zeros((), dtype=tl.int32)
        j = tl.zeros((), dtype=tl.int32)

            rev ^= b << j
            j += 1

        imag_i = tl.load(tim_ptr + (2 * i + 1))
