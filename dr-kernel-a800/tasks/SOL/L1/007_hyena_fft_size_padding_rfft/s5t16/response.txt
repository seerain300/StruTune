import torch
import triton
import triton.language as tl


@triton.jit
def rfft_real_kernel(x_ptr, out_ptr,
                      B, C, seqlen, n_iters,
                      BLOCK_K: tl.constexpr):
    """
    Compute real part of rfft for each (b, c) row:
      real_out[j] = sum_{k=0..2*seqlen-1} x[k] * cos(2*pi*j*k/(2*seqlen)) / (2*seqlen)
      for j in 0..seqlen.
    x_ptr points to the flattened vector of length B*C*seqlen, considered as:
      for each row r in [0, B*C), offset_base = r * seqlen, then x[offset_base + k].
    out_ptr points to the flattened vector of length B*C*(seqlen+1), considered as:
      for each row r, offset_base = r * (seqlen+1), then out[offset_base + j].
    """
    # Each program handles one (b, c) row
    row_id = tl.program_id(0)
    # Map row_id to (b, c) and compute base offsets
    b = row_id // C
    c = row_id % C
    # Base offsets for input and output
    base_in = row_id * seqlen
    base_out = row_id * (seqlen + 1)

    # Precompute normalization
    norm = 1.0 / (2.0 * seqlen)

    # Loop over j = 0 .. seqlen
    j = 0
    while j <= seqlen:
        # Accumulator for real part
        acc = 0.0

        # Loop over k = 0 .. 2*seqlen - 1 in chunks of BLOCK_K
        k_start = 0
        while k_start < (2 * seqlen):
            k_vec = k_start + tl.arange(0, BLOCK_K)
            # Mask for in-bounds k
            mask_k = k_vec < (2 * seqlen)

            # Load x[k] (handle padding: beyond seqlen is zeros)
            # If k < seqlen, index is base_in + k; else zero
            x_val = tl.load(x_ptr + base_in + tl.where(k_vec < seqlen, k_vec, 0), mask=mask_k, other=0.0)

            # Compute cos(2*pi*j*k/(2*seqlen))
            # For masked elements, cos is computed but multiplied by 0 (since x_val=0), so fine.
            angle = 2.0 * 3.141592653589793 * j * k_vec / (2.0 * seqlen)
            cosv = tl.cos(angle)

            # Accumulate sum
            acc += tl.sum(x_val * cosv, axis=0)

            k_start += BLOCK_K

        # Normalize and store
        out_val = acc * norm
        tl.store(out_ptr + base_out + j, out_val)
        j += 1


@triton.jit
def rfft_imag_kernel(x_ptr, out_ptr,
                      B, C, seqlen, n_iters,
                      BLOCK_K: tl.constexpr):
    """
    Compute imaginary part of rfft for each (b, c) row:
      imag_out[j] = sum_{k=0..2*seqlen-1} x[k] * sin(2*pi*j*k/(2*seqlen)) / (2*seqlen)
      for j in 1..seqlen-1 (imag_out[0] = 0, imag_out[seqlen] = 0).
    Similar indexing as rfft_real_kernel.
    """
    row_id = tl.program_id(0)
    b = row_id // C
    c = row_id % C
    base_in = row_id * seqlen
    base_out = row_id * (seqlen + 1)

    norm = 1.0 / (2.0 * seqlen)

    j = 1
    while j < seqlen:
        acc = 0.0
        k_start = 0
        while k_start < (2 * seqlen):
            k_vec = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_vec < (2 * seqlen)

            # Load x[k] (zeros beyond seqlen)
            x_val = tl.load(x_ptr + base_in + tl.where(k_vec < seqlen, k_vec, 0), mask=mask_k, other=0.0)

        # Launch kernels: one program per (b, c) row
        grid = (B * C,)

        BLOCK_K = 256

        imag_out[B * C * seqlen] = 0.0

            B, C, seqlen, 0,  # n_iters unused

            num_warps=2
        )

        x_freq_imag = imag_out.view(B, C, seqlen + 1)

        return x_freq_real, x_freq_imag

            seqlen, BLOCK_K=BLOCK_K,
            num_warps=4
        )

            x_padded, imag_out,
            seqlen, BLOCK_K=BLOCK_K,
            num_warps=4
        )

        x_freq_real = real_out.view(B, C, seqlen + 1)
        x_freq_imag = imag_out.view(B, C, seqlen + 1)

        return x_freq_real, x_freq_imag
