import math
import torch
import triton
import triton.language as tl


@triton.jit
def _rfft_real_imag_triton_kernel(
    x_ptr,                  # *const float32, input pointer to x of shape (B*C, L) contiguous
    out_real_ptr,           # *float32, output pointer to real part of shape (B*C, L+1)
    out_imag_ptr,           # *float32, output pointer to imag part of shape (B*C, L+1)
    L: tl.constexpr,        # original seqlen
    invN: tl.constexpr,     # 1.0 / (2*L), normalization factor
    NC: tl.constexpr,       # number of (batch, channel) slices = B*C
):
    # Each program handles one (b, c) slice; slice index is program_id(0)
    pid = tl.program_id(0)
    # If pid >= NC, nothing to do (grid size should be NC, so no out-of-bounds)
    # Compute base offset for this slice in the flattened (B*C, L) input
    base_in = pid * L
    # Output length M = L + 1
    M = L + 1

    # Vector of j indices we will compute for this slice; set BLOCK_J = 64
    j_vec = tl.arange(0, 64)
    # Loop over j in chunks to cover M = L + 1
    # Note: Triton supports for-loops with constant bounds.
    for start in range(0, M, 64):
        j_idx = start + j_vec  # vector of j candidates in this chunk
        # Mask for valid j indices: j < M
        mask_j = j_idx < M

        # Accumulators for real and imag parts (float32)
        re_sum = tl.zeros([64], dtype=tl.float32)
        im_sum = tl.zeros([64], dtype=tl.float32)

        # Iterate over t from 0 to 2*L - 1 (N = 2*L, zero-padded)
        for t in range(0, 2 * L):
            # Load x[pid, t] with mask to avoid OOB if pid >= NC (but grid ensures pid < NC)
            x_val = tl.load(x_ptr + base_in + t)  # pid < NC ensures base_in + t in range for t < 2*L

            # Compute angle = 2*pi*j*t/(2*L) = pi*j*t/L
            # Use float32 for math
            # j is vector, t is scalar: broadcast t across j_vec
            angle = (j_idx * t) * (math.pi * invN)  # invN = 1/(2*L), angle = j * (pi * t / L)

            # cos and sin contributions
            cos_term = tl.cos(angle)
            sin_term = tl.sin(angle)

            re_sum += x_val * cos_term
            im_sum += x_val * sin_term

        # Scale by normalization factor 1/(2*L) = invN
        re_sum = re_sum * invN
        im_sum = im_sum * invN

        # Store results for valid j
        # Output is flattened as (B*C, M). We compute out pointers accordingly.
        out_base_real = out_real_ptr + pid * M
        out_base_imag = out_imag_ptr + pid * M

        tl.store(out_base_real + j_idx, re_sum, mask=mask_j)
        tl.store(out_base_imag + j_idx, im_sum, mask=mask_j)
