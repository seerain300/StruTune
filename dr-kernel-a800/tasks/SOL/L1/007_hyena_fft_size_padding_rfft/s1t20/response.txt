import triton.language as tl


@triton.jit
def _rfft_real_imag_triton_kernel(
    x_ptr,                  # *const float32, input pointer to x (B, C, L) contiguous
    out_real_ptr,           # *float32, output pointer to real part (B, C, L+1)
    out_imag_ptr,           # *float32, output pointer to imag part (B, C, L+1)

    L: tl.constexpr,        # seqlen
    N: tl.constexpr,        # n = 2 * seqlen (length for rfft)
    M: tl.constexpr,        # M = L + 1 (output length)
):

    b = tl.program_id(0)
    c = tl.program_id(1)

    # Compute base pointers for this (b, c)
    # x layout: (B, C, L) contiguous => linear index = ((b*C + c) * L) + t
    # out layout: (B, C, M) contiguous => linear index = ((b*C + c) * M) + j

    base_out = (b * C + c) * M

    # We'll use a small BLOCK_J (e.g., 64) and iterate over j from 0 to M-1
    # We compute re and im for each j in the chunk and store.
    # Note: We use scalar t loop for simplicity and robustness.
    # We will iterate j from 0 to M-1 by manual increments to avoid unsupported Triton reductions.
    # However, Triton requires loops with compile-time bounds; we can simulate by unrolled scalar computation.
    # To keep it simple and correct, we compute one j at a time.

    # We'll compute re and im for j = 0..M-1
    # Since M is passed as constexpr, Triton can unroll these loops at compile time.
    # For each j, we loop over t = 0..N-1 and accumulate.
    for j in range(0, M):
        re_acc = 0.0
        im_acc = 0.0
        # Accumulate over t

            # Load x[t] as float32
            # Pointer arithmetic: x_ptr + base_x + t
            x_val = tl.load(x_ptr + base_x + t)
            # Compute cos and sin

            cos_term = tl.cos(angle)
            sin_term = tl.sin(angle)
            # Accumulate
            re_acc += x_val * cos_term
            im_acc += x_val * sin_term
        # Normalize by N
        re_acc = re_acc / float(N)
        im_acc = im_acc / float(N)

        tl.store(out_real_ptr + base_out + j, re_acc)
        tl.store(out_imag_ptr + base_out + j, im_acc)
