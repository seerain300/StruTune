import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded_kernel(x_row_ptr, padded_ptr, L, n, BLOCK: tl.constexpr):
    """
    Copy input row of length L into padded buffer of length n, padding with zeros.
    x_row_ptr: *float32, length L
    padded_ptr: *float32, length n
    """
    pid = tl.program_id(axis=0)
    j = pid
    if j < L:
        tl.store(padded_ptr + j, tl.load(x_row_ptr + j))


@triton.jit
def _compute_cos_table_kernel(padded_ptr, cos_table_ptr, L, n, BLOCK_J: tl.constexpr):
    """
    Compute cos table: cos_table[j, k] = cos(2*pi*k*j / n), j in [0..n-1], k in [0..L]
    Store into cos_table_ptr with shape (L+1, n), row-major contiguous.
    """
    # Outer dimension: k, inner dimension: j
    k = tl.program_id(axis=0)
    if k > L:
        return
    for jj in range(0, n, BLOCK_J):
        j_vec = jj + tl.arange(0, BLOCK_J)
        mask_j = j_vec < n
        angle = 2.0 * 3.141592653589793 * k * j_vec * (1.0 / n)
        cosv = tl.cos(angle)
        tl.store(cos_table_ptr + k * n + j_vec, cosv, mask=mask_j)


@triton.jit
def _compute_sin_table_kernel(padded_ptr, sin_table_ptr, L, n, BLOCK_J: tl.constexpr):
    """
    Compute sin table: sin_table[j, k] = sin(2*pi*k*j / n), j in [0..n-1], k in [0..L]
    Store into sin_table_ptr with shape (L+1, n), row-major contiguous.
    """
    k = tl.program_id(axis=0)
    if k > L:
        return
    for jj in range(0, n, BLOCK_J):
        j_vec = jj + tl.arange(0, BLOCK_J)
        mask_j = j_vec < n
        angle = 2.0 * 3.141592653589793 * k * j_vec * (1.0 / n)
        sinv = tl.sin(angle)
        tl.store(sin_table_ptr + k * n + j_vec, sinv, mask=mask_j)


@triton.jit
def _reduce_real_kernel(padded_ptr, cos_table_ptr, out_real_ptr, L, n, BLOCK_J: tl.constexpr):
    """
    Accumulate y_real[k] = sum_{j=0..n-1} padded[j] * cos_table[j, k]
    out_real_ptr: *float32, length (L+1), will be written by atomic_add
    """
    k = tl.program_id(axis=0)
    if k > L:
        return
    acc = 0.0
    for jj in range(0, n, BLOCK_J):
        j_vec = jj + tl.arange(0, BLOCK_J)
        mask_j = j_vec < n
        xj = tl.load(padded_ptr + j_vec, mask=mask_j, other=0.0)
        cos_row = tl.load(cos_table_ptr + k * n + j_vec, mask=mask_j, other=0.0)
        prod = xj * cos_row
        local_sum = tl.zeros((), dtype=tl.float32)
        for t in range(0, BLOCK_J):
            local_sum += prod[t]
        acc += local_sum
    tl.atomic_add(out_real_ptr + k, acc)


@triton.jit
def _reduce_imag_kernel(padded_ptr, sin_table_ptr, out_imag_ptr, L, n, BLOCK_J: tl.constexpr):
    """
    Accumulate y_imag[k] = sum_{j=0..n-1} padded[j] * sin_table[j, k]
    out_imag_ptr: *float32, length (L+1), will be written by atomic_add
    """
    k = tl.program_id(axis=0)
    if k > L:
        return
    acc = 0.0
    for jj in range(0, n, BLOCK_J):
        j_vec = jj + tl.arange(0, BLOCK_J)
        mask_j = j_vec < n
        xj = tl.load(padded_ptr + j_vec, mask=mask_j, other=0.0)
        sin_row = tl.load(sin_table_ptr + k * n + j_vec, mask=mask_j, other=0.0)
        prod = xj * sin_row
        local_sum = tl.zeros((), dtype=tl.float32)
        for t in range(0, BLOCK_J):
            local_sum += prod[t]
        acc += local_sum
    tl.atomic_add(out_imag_ptr + k, acc)


@triton.jit
def _normalize_divide_kernel(in_ptr, out_ptr, numel, scale, BLOCK: tl.constexpr):
    """
    Elementwise division: out[i] = in[i] / scale
    """
    pid = tl.program_id(axis=0)
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < numel
    val = tl.load(in_ptr + idx, mask=mask, other=0.0)
    val = val / scale
    tl.store(out_ptr + idx, val, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # Expect input of shape (batch, channels, seqlen)
        assert len(args) == 1, "ModelNew expects a single input tensor"
        x = args[0]
        assert x.ndim == 3, "Input must be 3D (batch, channels, seqlen)"
        B, C, L = x.shape

        device = x.device
        # Ensure float32
        x = x.to(torch.float32)

        n = 2 * L

        # Output tensors: (B, C, L+1)
        out_real = torch.empty((B, C, L + 1), dtype=torch.float32, device=device)
        out_imag = torch.empty((B, C, L + 1), dtype=torch.float32, device=device)

        # Iterate over batch and channels; compute rFFT per (b, c) slice
        # Use BLOCK sizes that cover n in one pass for performance
        BLOCK_J = n  # since n <= 2*seqlen, typical values like 2048/4096 are fine; we pick n to keep it simple
        for b in range(B):
            for c in range(C):
                # Prepare padded buffer of length n
                padded = torch.empty(n, dtype=torch.float32, device=device)
                # Kernel 1: copy input row to padded
                x_row = x[b, c, :]
                grid_copy = (L,)
                _copy_row_to_padded_kernel[grid_copy](x_row, padded, L, n, BLOCK=1)

                # Allocate cos/sin tables: shape (L+1, n), contiguous row-major
                cos_table = torch.empty((L + 1, n), dtype=torch.float32, device=device)
                sin_table = torch.empty((L + 1, n), dtype=torch.float32, device=device)

                # Kernel 2: compute cos table
                grid_cos = (L + 1,)
                _compute_cos_table_kernel[grid_cos](padded, cos_table, L, n, BLOCK_J=BLOCK_J)

                # Kernel 3: compute sin table
                grid_sin = (L + 1,)
                _compute_sin_table_kernel[grid_sin](padded, sin_table, L, n, BLOCK_J=BLOCK_J)

                # Initialize accumulators to zeros
                out_real[b, c, :] = torch.zeros(L + 1, dtype=torch.float32, device=device)
                out_imag[b, c, :] = torch.zeros(L + 1, dtype=torch.float32, device=device)

                # Kernel 4: reduce real
                _reduce_real_kernel[grid_cos](padded, cos_table, out_real[b, c, :], L, n, BLOCK_J=BLOCK_J)

                # Kernel 5: reduce imag
                _reduce_imag_kernel[grid_sin](padded, sin_table, out_imag[b, c, :], L, n, BLOCK_J=BLOCK_J)

                # Kernel 6: normalize by 2*L
                numel = L + 1
                grid_norm = (numel + 255) // 256  # 1D grid
                _normalize_divide_kernel[grid_norm](out_real[b, c, :], out_real[b, c, :], numel, 2.0 * float(L), BLOCK=256)
                _normalize_divide_kernel[grid_norm](out_imag[b, c, :], out_imag[b, c, :], numel, 2.0 * float(L), BLOCK=256)

        # Return real and imaginary parts (both shape (B, C, L+1))
        return out_real, out_imag
