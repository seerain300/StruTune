import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded_kernel(
    x_ptr, padded_ptr,
    L,
    stride_xb, stride_xc, stride_xl,
    stride_pbj, stride_pbc, stride_pbl,
    b, c,
    BLOCK_L: tl.constexpr,
):
    # Each program handles one k in [0..L)
    k = tl.program_id(0)
    if k >= L:
        return

    x_row_ptr = x_ptr + b * stride_xb + c * stride_xc + k * stride_xl
    out_row_ptr = padded_ptr + b * stride_pbj + c * stride_pbc + k * stride_pbl

    # Load and store single element; padded tail is zeros in host (we allocate zeros)
    val = tl.load(x_row_ptr)
    tl.store(out_row_ptr, val)


@triton.jit
def _compute_cos_table_kernel(
    out_ptr,
    N, K,
    stride_outj, stride_outk,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    j = tl.program_id(0)
    k = tl.program_id(1)
    if j >= N or k >= K:
        return

    j_vec = j + tl.arange(0, BLOCK_J)
    k_vec = k + tl.arange(0, BLOCK_K)
    mask_j = j_vec < N
    mask_k = k_vec < K

    angle = 2.0 * 3.141592653589793 * k_vec * j_vec / N
    cos_vals = tl.cos(angle)

    # Store per-k vector into row j: out_ptr[j, k_vec]
    tl.store(out_ptr + j * stride_outj + k_vec * stride_outk, cos_vals, mask=mask_k)


@triton.jit
def _compute_sin_table_kernel(
    out_ptr,
    N, K,
    stride_outj, stride_outk,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    j = tl.program_id(0)
    k = tl.program_id(1)
    if j >= N or k >= K:
        return

    j_vec = j + tl.arange(0, BLOCK_J)
    k_vec = k + tl.arange(0, BLOCK_K)
    mask_j = j_vec < N
    mask_k = k_vec < K

    angle = 2.0 * 3.141592653589793 * k_vec * j_vec / N
    sin_vals = tl.sin(angle)

    tl.store(out_ptr + j * stride_outj + k_vec * stride_outk, sin_vals, mask=mask_k)


@triton.jit
def _reduce_real_kernel(
    padded_ptr, cos_ptr, out_ptr,
    N, K,
    stride_pj, stride_pk,  # strides for padded
    stride_cj, stride_ck,  # strides for cos_table
    stride_ol,             # stride for output along K
    BLOCK_J: tl.constexpr,
):
    # One program per k (k in [0..K))
    k = tl.program_id(0)
    if k >= K:
        return
    acc = tl.zeros((), dtype=tl.float32)

    j = 0
    while j < N:
        j_vec = j + tl.arange(0, BLOCK_J)
        mask_j = j_vec < N

        # Load padded[j_vec] (vector of BLOCK_J)
        p_ptr = padded_ptr + j_vec * stride_pj  # along N dimension
        vals = tl.load(p_ptr, mask=mask_j, other=0.0)

        # Load cos_table[j_vec, k] (vector)
        c_ptr = cos_ptr + j_vec * stride_cj + k * stride_ck
        cos_vec = tl.load(c_ptr, mask=mask_j, other=0.0)

        # Multiply and reduce (sum across BLOCK_J)
        prod = vals * cos_vec
        acc += tl.sum(prod, axis=0)

        j += BLOCK_J

    # Store accumulated real part
    out_ptr_k = out_ptr + k * stride_ol
    tl.store(out_ptr_k, acc)


@triton.jit
def _reduce_imag_kernel(
    padded_ptr, sin_ptr, out_ptr,
    N, K,
    stride_pj, stride_pk,
    stride_sj, stride_sk,
    stride_ol,
    BLOCK_J: tl.constexpr,
):
    k = tl.program_id(0)
    if k >= K:
        return
    acc = tl.zeros((), dtype=tl.float32)

    j = 0
    while j < N:
        j_vec = j + tl.arange(0, BLOCK_J)
        mask_j = j_vec < N

        p_ptr = padded_ptr + j_vec * stride_pj
        vals = tl.load(p_ptr, mask=mask_j, other=0.0)

        s_ptr = sin_ptr + j_vec * stride_sj + k * stride_sk
        sin_vec = tl.load(s_ptr, mask=mask_j, other=0.0)

        prod = vals * sin_vec
        acc += tl.sum(prod, axis=0)

        j += BLOCK_J

    out_ptr_k = out_ptr + k * stride_ol
    tl.store(out_ptr_k, acc)


@triton.jit
def _divide_kernel(
    in_ptr, out_ptr, numel, divisor, BLOCK: tl.constexpr
):
    idx = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    mask = idx < numel
    vals = tl.load(in_ptr + idx, mask=mask, other=0.0)
    vals = vals / divisor
    tl.store(out_ptr + idx, vals, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        # x: (B, C, L)
        B, C, L = x.shape
        n = 2 * L
        device = x.device

        # Ensure float32
        x = x.to(torch.float32)

        # Allocate padded input buffer (B, C, n) zeros
        padded = torch.zeros((B, C, n), dtype=torch.float32, device=device)

        # Launch copy kernel: copy x[:, :, :] into padded[:, :, :L]
        # One program per (b, c, k)
        # Note: we pass strides for x and padded
        for b in range(B):
            for c in range(C):
                _copy_row_to_padded_kernel[(L,)](
                    x, padded,
                    L,
                    x.stride(0), x.stride(1), x.stride(2),
                    padded.stride(0), padded.stride(1), padded.stride(2),
                    b, c,
                    BLOCK_L=1,
                )

        # Allocate tables (N, K) on device
        N = n
        K = L
        cos_table = torch.empty((N, K), dtype=torch.float32, device=device)
        sin_table = torch.empty((N, K), dtype=torch.float32, device=device)

        # Launch kernels to fill cos/sin tables
        BLOCK_J = 128
        BLOCK_K = 64
        grid_tables = (triton.cdiv(N, BLOCK_J), triton.cdiv(K, BLOCK_K))
        _compute_cos_table_kernel[grid_tables](cos_table, N, K, cos_table.stride(0), cos_table.stride(1), BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K)
        _compute_sin_table_kernel[grid_tables](sin_table, N, K, sin_table.stride(0), sin_table.stride(1), BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K)

        # Allocate outputs (B, C, L+1) initialized to zeros
        out_real = torch.zeros((B, C, L + 1), dtype=torch.float32, device=device)
        out_imag = torch.zeros((B, C, L + 1), dtype=torch.float32, device=device)

        # Reduce real and imag per (b,c) over j in chunks
        for b in range(B):
            for c in range(C):
                # Strides for padded (B,C,n)
                stride_pj = padded.stride(2)  # typically 1
                stride_pk = 1  # since we iterate K dimension; but padded is (B,C,n), K=L, so we use stride along L (padded.stride(2)) and keep k index as offset. For reduction, we index along N dimension: stride along N is padded.stride(2), which is 1 here. We'll use padded.stride(2) for j.
                # Strides for cos/sin tables (N,K)
                stride_cj = cos_table.stride(0)  # N
                stride_ck = cos_table.stride(1)  # K
                stride_sj = sin_table.stride(0)  # N
                stride_sk = sin_table.stride(1)  # K

                # Real part reduction
                _reduce_real_kernel[(K,)](
                    padded, cos_table,
                    out_real[b, c, :],
                    N, K,
                    stride_pj, padded.stride(2),  # padded.stride(2) is 1
                    stride_cj, stride_ck,
                    1,  # stride for out along K; since out_real is contiguous (B,C,L+1), stride along last dim is 1
                    BLOCK_J=128,
                )
                # Imag part reduction
                _reduce_imag_kernel[(K,)](
                    padded, sin_table,
                    out_imag[b, c, :],
                    N, K,
                    stride_pj, padded.stride(2),
                    stride_sj, stride_sk,
                    1,
                    BLOCK_J=128,
                )

        # Normalize by 2*L
        numel_real = out_real.numel()
        numel_imag = out_imag.numel()
        _divide_kernel[(triton.cdiv(numel_real, 256),)](out_real, out_real, numel_real, 2.0 * float(L), BLOCK=256)
        _divide_kernel[(triton.cdiv(numel_imag, 256),)](out_imag, out_imag, numel_imag, 2.0 * float(L), BLOCK=256)

        return out_real, out_imag
