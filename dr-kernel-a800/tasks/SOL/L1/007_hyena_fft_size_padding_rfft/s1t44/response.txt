```python
import triton
import triton.language as tl


@triton.jit
def _rfft_element_kernel(
    x_ptr,           # *const float32, input pointer to x (B, C, L) contiguous in linear order
    out_real_ptr,    # *float32, output pointer to real part (B, C, L+1) contiguous
    out_imag_ptr,    # *float32, output pointer to imag part (B, C, L+1) contiguous
    B: tl.constexpr, # batch size
    C: tl.constexpr, # channels
    L: tl.constexpr, # seqlen (input length)
    N: tl.constexpr, # N = 2 * L
    M: tl.constexpr, # M = L + 1
    J: tl.constexpr, # output index j for this program (0..M-1)
):
    # Grid is (B, C, M). We map program_id(0) -> b, program_id(1) -> c, program_id(2) -> j.
    b = tl.program_id(0)
    c = tl.program_id(1)
    j = tl.program_id(2)

    # Compute base linear offset for (b, c) slice in x: base_x = b * (C * L) + c * L
    base_x = b * (C * L) + c * L

    # Compute base linear offset for output (b, c, j): base_out = b * (C * M) + c * M + j
    # Note: out tensors are (B, C, M), so linear index is computed similarly.
    base_out = b * (C * M) + c * M + j

    # Accumulate real and imaginary parts
    re = 0.0
    im = 0.0

    # Iterate over t = 0..N-1 (N is constexpr -> Triton can unroll or compile this loop)
    for t in range(0, N):
        # Load x[b, c, t]
        x_val = tl.load(x_ptr + base_x + t)  # x is contiguous along last dim within (b, c)

        # Compute cos and sin with scaling by 1/N (normalization)
        angle = 2.0 * 3.141592653589793 * j * t / N
        re += x_val * tl.cos(angle)
        im += x_val * tl.sin(angle)

    # Normalize by N
    re = re / N
    im = im / N

    # Store to outputs
    tl.store(out_real_ptr + base_out, re)
    tl.store(out_imag_ptr + base_out, im)

    def forward(self, x: torch.Tensor):
        """

        for each (batch, channel) slice, normalized by 2*seqlen, and return
        real and imaginary parts separately.

        N = 2 * L
        M = L + 1

        # Triton expects pointers; ensure dtype float32 and contiguity
        x_f32 = x.contiguous().to(torch.float32)
        device = x_f32.device

        # Allocate outputs (B, C, M) float32
        out_real = torch.empty((B, C, M), dtype=torch.float32, device=device)
        out_imag = torch.empty((B, C, M), dtype=torch.float32, device=device)

        # Launch kernel with grid (B, C, M). N and M are constexpr (compile-time) for the kernel.
        grid = (B, C, M)
        _rfft_element_kernel[grid](
            x_f32, out_real, out_imag,
            B, C, L, N, M, M  # pass M as J; each program handles one j in 0..M-1
        )
        return out_real, out_imag
```
