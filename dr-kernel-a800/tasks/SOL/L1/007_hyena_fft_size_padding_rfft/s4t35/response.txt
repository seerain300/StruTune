import torch
import triton
import triton.language as tl


@triton.jit
def _copy_row_to_padded(x_ptr, padded_ptr, L: tl.int32, N: tl.constexpr):
    """
    Copy x[row, 0:L] into padded[row, 0:L]. padded[row, L:N] is assumed to be zeros already.
    Grid: (rows,)
    """
    row = tl.program_id(0)
    # Each row has length L
    for i in tl.static_range(0, L):
        val = tl.load(x_ptr + row * L + i)
        tl.store(padded_ptr + row * N + i, val)


@triton.jit
def _compute_rfft_real_per_k(padded_ptr, out_real_ptr, N: tl.constexpr):
    """
    For each (row, k) where k = program_id(1) in [0..L], compute y_real[k]:
      y_real[k] = (1/N) * sum_{j=0}^{N-1} padded[row, j] * cos(2*pi*k*j/N)
    Normalize by 1/N inside the kernel; write to out_real[row, k].
    Grid: (rows, L+1)
    """
    row = tl.program_id(0)
    k = tl.program_id(1)
    acc = 0.0
    # Loop over j=0..N-1 (N is constexpr, so this is unrolled)
    for j in tl.static_range(0, N):
        val = tl.load(padded_ptr + row * N + j)
        angle = 2.0 * 3.141592653589793 * (k * j) / N
        cosv = tl.cos(angle)
        acc += val * cosv
    # Normalize by 1/N
    acc *= (1.0 / N)
    tl.store(out_real_ptr + row * (L + 1) + k, acc)


@triton.jit
def _compute_rfft_imag_per_k(padded_ptr, out_imag_ptr, N: tl.constexpr):
    """
    For each (row, k) where k = program_id(1) in [0..L], compute y_imag[k]:
      y_imag[k] = (1/N) * sum_{j=0}^{N-1} padded[row, j] * sin(2*pi*k*j/N)
    Normalize by 1/N inside the kernel; write to out_imag[row, k].
    Grid: (rows, L+1)
    """
    row = tl.program_id(0)
    k = tl.program_id(1)
    acc = 0.0
    for j in tl.static_range(0, N):
        val = tl.load(padded_ptr + row * N + j)
        angle = 2.0 * 3.141592653589793 * (k * j) / N
        sinv = tl.sin(angle)
        acc += val * sinv
    acc *= (1.0 / N)
    tl.store(out_imag_ptr + row * (L + 1) + k, acc)


@triton.jit
def _normalize_div(ptr, factor: tl.float32, M: tl.int32, K: tl.constexpr):
    """
    Divide each element in ptr (length M) by factor. K is the number of elements per row.
    Grid: (rows,)
    Note: Here M = (L+1) * rows, but we handle each row separately.
    For simplicity, we launch grid (rows,) and process K elements per row.
    """
    row = tl.program_id(0)
    base = row * (L + 1)
    # Process up to K elements per row
    for i in tl.static_range(0, K):
        val = tl.load(ptr + base + i)
        val = val * factor
        tl.store(ptr + base + i, val)


@triton.jit
def _normalize_div_vector(ptr, factor: tl.float32, length: tl.int32):
    """
    Generic elementwise division by factor for a 1D vector of given length.
    Grid: (1,)
    """
    # Single program divides the entire vector
    for i in tl.static_range(0, length):
        val = tl.load(ptr + i)
        val = val * factor  # multiply by reciprocal (factor = 1/N)
        tl.store(ptr + i, val)


def _triton_run(x: torch.Tensor):
    """
    Triton-only implementation of the original run function:
    - Pads to n=2*L and computes real/imag parts of rfft, normalized by n, using Triton kernels.
    - Returns (B, C, L+1) tensors for real and imaginary parts.
    """
    B, C, L = x.shape
    # We will handle padded length N as a constexpr up to 65536 (covers max 2*seqlen=65536 for the provided workloads).
    N = 2 * L
    assert N <= 65536, "This Triton implementation supports padded length up to 65536."

    device = x.device
    rows = B * C

    # Ensure float32 and contiguous per row
    x_f32 = x.to(torch.float32).contiguous()  # shape (B, C, L)
    # Allocate padded buffer: (rows, N), initialize zeros
    padded = torch.zeros((rows, N), dtype=torch.float32, device=device)
    # Copy each row to padded
    _copy_row_to_padded[(rows,)](x_f32.view(rows, L), padded, L, N=N, num_warps=1)

    # Allocate outputs: real and imag, shape (rows, L+1)
    out_real = torch.empty((rows, L + 1), dtype=torch.float32, device=device)
    out_imag = torch.empty((rows, L + 1), dtype=torch.float32, device=device)

    # Launch rfft real/imag computation: grid = (rows, L+1)
    grid = (rows, L + 1)
    _compute_rfft_real_per_k[grid](padded, out_real, N=N, num_warps=1)
    _compute_rfft_imag_per_k[grid](padded, out_imag, N=N, num_warps=1)

    # Normalize by n = 2*L (i.e., divide by N). Use Triton to perform elementwise division.
    inv_N = 1.0 / float(N)
    # Normalize out_real: length = rows * (L+1)
    # Launch a single program that divides the entire vector; grid (rows,) will be fine since K=(L+1) is small per row.
    # For safety, use vectorized elementwise division for simplicity.
    # We can call _normalize_div with grid (rows,) and K=(L+1) per row, but simpler is to divide the flattened output.
    # Flatten for clarity
    out_real_flat = out_real.view(-1)
    out_imag_flat = out_imag.view(-1)
    total_len = out_real_flat.numel()
    # Elementwise division by inv_N (i.e., multiply by inv_N). We can do this in Triton with a single program.
    _normalize_div_vector[(1,)](out_real_flat, inv_N, total_len, num_warps=1)
    _normalize_div_vector[(1,)](out_imag_flat, inv_N, total_len, num_warps=1)
    # Reshape back
    out_real = out_real_flat.view(rows, L + 1)
    out_imag = out_imag_flat.view(rows, L + 1)

    # Reshape to (B, C, L+1)
    out_real = out_real.view(B, C, L + 1)
    out_imag = out_imag.view(B, C, L + 1)
    return out_real, out_imag


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # Expect single tensor input of shape (batch, channels, seqlen)
        x = args[0] if len(args) == 1 else args[0]

@triton.jit
def _copy_row_to_padded(x_ptr, padded_ptr, L: tl.int32, TWO_L: tl.constexpr):
    """
    Copy x[row, 0:L] into padded[row, 0:L]. padded[row, L:TWO_L] is assumed zeros.
    Grid: (rows,)
    """

        tl.store(padded_ptr + row * TWO_L + i, val)

      y_real[k] = (1/(2*L)) * sum_{j=0}^{2*L-1} padded[row, j] * cos(2*pi*k*j/(2*L))
    Grid: (rows, L+1)

    k = tl.program_id(1)
    acc = 0.0
    # Loop over j=0..2*L-1 (constexpr)
