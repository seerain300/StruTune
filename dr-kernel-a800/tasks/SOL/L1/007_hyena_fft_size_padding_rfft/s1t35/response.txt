import triton.language as tl

def _rfft_real_imag_triton_kernel(
    x_ptr,                # *const float32, input pointer to x with shape (B, C, L), contiguous
    out_real_ptr,         # *float32, output pointer to real part (B, C, L+1), contiguous
    out_imag_ptr,         # *float32, output pointer to imag part (B, C, L+1), contiguous
    L,                    # int32, seqlen
    B,                    # int32, batch size (used to compute stride for batch)
    stride_b,             # int32, stride between batches in x (elements)
    stride_c,             # int32, stride between channels in x (elements)
):

    pid = tl.program_id(axis=0)
    # Derive b and c from pid; axis=0 equals B*C, so b = pid // C, c = pid % C
    C = tl.num_programs(axis=0) // B  # Triton provides program id but not B; we need to pass C or reconstruct

    # We can infer b and c using the grid size B*C and passing B as an argument.
    b = pid // C
    c = pid % C

    # Base offsets for this (b, c) slice in x
    x_base = b * stride_b + c * stride_c

    N = 2 * L
    M = L + 1

    invN = 1.0 / N

    # For j in 0..M-1, compute re and im and store
    j = 0
    while j < M:
        re_sum = 0.0
        im_sum = 0.0
        t = 0

            x_val = tl.load(x_ptr + x_base + t)

            sinj = tl.sin(angle)

            im_sum += x_val * sinj * invN

        tl.store(out_real_ptr + b * stride_b + c * stride_b + j, re_sum)
        tl.store(out_imag_ptr + b * stride_b + c * stride_b + j, im_sum)
