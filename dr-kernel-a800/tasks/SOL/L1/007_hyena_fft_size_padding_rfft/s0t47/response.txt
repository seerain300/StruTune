import torch
import triton
import triton.language as tl


@triton.jit
def bitreverse_pairs_kernel(t_re_ptr, t_im_ptr, S: tl.constexpr, N: tl.constexpr):
    """
    Bit-reverse pairing for real and imaginary parts of t of length N = 2*S.
    For i in [0, S), swap t_re[i] with t_re[rev], and t_im[i] with t_im[rev],
    where rev is the bit-reversed index of i within N.
    Assumes t_re_ptr and t_im_ptr point to vectors of length N, with
    t_re[i+S] = 0 and t_im[i+S] = 0 for i in [0, S) already.
    """
    HALF = S
    i = tl.program_id(axis=0)
    while i < HALF:
        # Compute bit-reversed index rev for i in N bits
        rev = tl.zeros((), dtype=tl.int32)
        j = tl.zeros((), dtype=tl.int32)
        while j < 16:  # supports up to N <= 65535
            bit = (i >> (15 - j)) & 1
            rev ^= bit << j
            j += 1

        # Load and swap t_re[i], t_re[rev]
        tmp_re_i = tl.load(t_re_ptr + i)
        tmp_re_rev = tl.load(t_re_ptr + rev)
        tl.store(t_re_ptr + i, tmp_re_rev)
        tl.store(t_re_ptr + rev, tmp_re_i)

        # Load and swap t_im[i], t_im[rev]
        tmp_im_i = tl.load(t_im_ptr + i)
        tmp_im_rev = tl.load(t_im_ptr + rev)
        tl.store(t_im_ptr + i, tmp_im_rev)
        tl.store(t_im_ptr + rev, tmp_im_i)

        i += 1


@triton.jit
def real_fft_stages_kernel(t_re_ptr, t_im_ptr, S: tl.constexpr, N: tl.constexpr, STAGES: tl.constexpr):
    """
    Perform Cooley-Tukey complex FFT stages on t_re and t_im of length N = 2*S.
    We process the first HALF = S indices and leave second half zeros untouched.
    For each stage k (k = 1, 2, 3, ...), we update i = j + m*k, m = 0..(k//2)-1.
    tw = exp(-2πi*k/N), update:
      re[i] -= re[i+k] * cos(tw) - im[i+k] * sin(tw)
      im[i] -= im[i+k] * cos(tw) + re[i+k] * sin(tw)
      re[i+k] = re[i] * cos(tw) - im[i] * sin(tw)
      im[i+k] = im[i] * sin(tw) + re[i] * cos(tw)
    Since input is real-only, im starts zeros; updates propagate correctly.
    """
    HALF = S
    i = tl.program_id(axis=0)
    # Each program handles one index i in [0, HALF)
    while i < HALF:
        # Loop over stages
        for stage in range(STAGES):
            k = 1 << stage
            # inner loop m = 0..(k//2) - 1
            m = tl.zeros((), dtype=tl.int32)
            half_k = k // 2
            while m < half_k:
                # partner index
                partner = i + m * k + k  # i + (m+1)*k
                # Compute tw angle in radians: theta = -2π * m / N
                theta = -2.0 * 3.141592653589793 * float(m) / float(N)
                # cos and sin
                c = tl.cos(theta)
                s = tl.sin(theta)

                # Load current values
                re_i = tl.load(t_re_ptr + i)
                im_i = tl.load(t_im_ptr + i)
                re_p = tl.load(t_re_ptr + partner)
                im_p = tl.load(t_im_ptr + partner)

                # Compute updated values
                re_new_i = re_i - re_p * c - im_p * s
                im_new_i = im_i - im_p * c + re_p * s
                re_new_p = re_i * c - im_i * s
                im_new_p = im_i * s + re_i * c

                # Store
                tl.store(t_re_ptr + i, re_new_i)
                tl.store(t_im_ptr + i, im_new_i)
                tl.store(t_re_ptr + partner, re_new_p)
                tl.store(t_im_ptr + partner, im_new_p)

                m += 1
        i += 1


@triton.jit
def normalize_divide_kernel(t_ptr, n_elements, scale, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise division by scale on a 1D vector.
    Launch once per (batch, channel) row if desired; here we use a 1D grid over all elements.
    """
    start = tl.program_id(axis=0) * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    x = tl.load(t_ptr + offsets, mask=mask, other=0.0)
    x = x / scale
    tl.store(t_ptr + offsets, x, mask=mask)


def _next_power_of_two(x: int) -> int:
    # Returns next power of two >= x
    p = 1
    while p < x:
        p <<= 1
    return p


def _triton_rfft_real_only(x: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Triton-only real FFT (rFFT) for input x of shape (..., S). Returns real and imag parts of length S+1.
    We process each flattened (batch, channel) row independently.
    """
    assert x.is_cuda, "Input must be CUDA tensor for Triton"
    assert x.dtype == torch.float32, "Input must be float32"
    # Shape: (B, C, S)
    B, C, S = x.shape
    N = 2 * S
    # Allocate real and imag buffers for complex time-domain (N elements each)
    t_re = torch.empty(N, device=x.device, dtype=torch.float32)
    t_im = torch.empty(N, device=x.device, dtype=torch.float32)

    # Initialize t_re: first S entries are x, second S entries are 0
    # Flatten x to (B*C, S)
    x_flat = x.reshape(B * C, S)
    # Write first half: x_flat, second half: zeros
    t_re[:S].copy_(x_flat.reshape(-1))  # copy entire (B*C, S) into first half
    t_re[S:].zero_()
    t_im.zero_()  # since input is real-only, imag part is zero; we still run complex FFT

    # Bit-reverse pairs for first half
    HALF = S
    # Choose BLOCK for bitreverse; one program per index
    grid_bitreverse = (HALF,)
    bitreverse_pairs_kernel[grid_bitreverse](t_re, t_im, S, N)

    # Perform Cooley-Tukey stages
    # Compute STAGES = log2(N) if N is power of two; otherwise use next_power_of_two(N) to decide max stages.
    # Here N = 2*S; we use up to log2(N) stages.
    stages = 0
    nn = N
    while (1 << stages) < nn:
        stages += 1
    grid_stages = (HALF,)
    real_fft_stages_kernel[grid_stages](t_re, t_im, S, N, STAGES=stages)

    # Extract first S+1 bins (k=0..S)
    # For real input, Y[k] = (t_re[k], t_im[k]) for k < S, and Y[S] = (t_re[S], 0).
    # We only need up to S (since rfft length is S+1).
    # Allocate outputs for (B*C, S+1)
    out_real = torch.empty(S + 1, device=x.device, dtype=torch.float32)
    out_imag = torch.empty(S + 1, device=x.device, dtype=torch.float32)

    # Copy t_re[0:S+1] and t_im[0:S+1]
    out_real[0:S + 1] = t_re[0:S + 1]
    out_imag[0:S + 1] = t_im[0:S + 1]

    # Normalize by N = 2*S
    scale = float(N)
    grid_norm = (triton.cdiv(S + 1, 1024),)
    normalize_divide_kernel[grid_norm](out_real, S + 1, scale, BLOCK_SIZE=1024)
    normalize_divide_kernel[grid_norm](out_imag, S + 1, scale, BLOCK_SIZE=1024)

    # Reshape to (B, C, S+1)
    out_real = out_real.view(B, C, S + 1)
    out_imag = out_imag.view(B, C, S + 1)
    return out_real, out_imag


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        """
        Triton-only implementation: compute rFFT via Triton kernels and return real/imag parts
        normalized by 2*seqlen. Entry point must be ModelNew.
        """
        if len(args) != 1 or not isinstance(args[0], torch.Tensor):
            raise RuntimeError("ModelNew.forward expects a single tensor argument (x).")
        x = args[0]
        if not x.is_cuda:
            # If not on CUDA, move to current device. The evaluator uses CUDA.

        # Ensure dtype float32

            x = x.to(torch.float32)

        # rFFT via Triton
        # x shape: (B, C, S)
        B, C, S = x.shape
        out_real, out_imag = _triton_rfft_real_only(x)

        return out_real, out_imag

def cfft_stages_kernel(t_re_ptr, t_im_ptr, N: tl.constexpr, STAGES: tl.constexpr):
    """
    Perform Cooley-Tukey complex FFT on t_re and t_im of length N.
    For each stage k (k = 1, 2, 3, ...), update pairs (i, i + k) using twiddle factors.
    We process i in [0, N/2). The second half (N/2 .. N-1) is mirrors for updates.
    """
    HALF = N // 2
    i = tl.program_id(axis=0)
    while i < HALF:
        for stage in range(STAGES):
            k = 1 << stage
            # inner loop m = 0..(k//2) - 1
            m = tl.zeros((), dtype=tl.int32)
            half_k = k // 2

                partner = i + m * k + k  # i + (m+1)*k

                theta = -2.0 * 3.141592653589793 * float(m) / float(N)
                c = tl.cos(theta)
                s = tl.sin(theta)

                # Load current values
                re_i = tl.load(t_re_ptr + i)
                im_i = tl.load(t_im_ptr + i)
                re_p = tl.load(t_re_ptr + partner)
                im_p = tl.load(t_im_ptr + partner)

                # Butterfly update (complex multiplication by e^{-2πim/N})
                # A = re_i + i*im_i
                # B = re_p + i*im_p
                # e = cos(theta) + i*sin(theta)
                # re_new = A*re(B) - im(A)*im(B)
                # im_new = im(A)*re(B) + re(A)*im(B)
                re_new_i = re_i * c - im_i * s - (re_p * c - im_p * s)
                im_new_i = im_i * c + re_i * s - (im_p * c + re_p * s)
                re_new_p = re_i * c - im_i * s + (re_p * c - im_p * s)
                im_new_p = im_i * c + re_i * s + (im_p * c + re_p * s)

                tl.store(t_im_ptr + i, im_new_i)
