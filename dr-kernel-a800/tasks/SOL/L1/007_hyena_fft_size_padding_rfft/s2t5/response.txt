import torch
import triton
import triton.language as tl


@triton.jit
def pad_kernel(
    x_ptr,             # *const float, input x of shape (B, C, L)
    padded_ptr,        # *float, output flattened padded vector length P = 2*L
    B: tl.constexpr,   # batch size
    C: tl.constexpr,   # channels
    L,                 # actual seqlen (int)
    P: tl.constexpr,   # padded length = 2*L (int)
):
    # One program per (b, c)
    pid = tl.program_id(0)
    c = pid % C
    b = pid // C
    base = b * C + c  # unused but kept for clarity

    # Compute base linear offset in padded_ptr for this (b, c)
    # Flattened indexing: for each (b,c), write P elements at offset (b*C + c) * P
    offset = (b * C + c) * P

    # Load x[b, c, :] into padded_ptr[offset : offset + L]
    # x_ptr is laid out as contiguous (B, C, L): stride across L is 1 when x is contiguous.
    x_bc_ptr = x_ptr + (b * C + c) * L  # pointer to start of this (b,c) slice
    # Copy first L elements
    for t in range(0, L):
        val = tl.load(x_bc_ptr + t)
        tl.store(padded_ptr + offset + t, val)

    # Zero-pad the remaining P - L positions
    for t in range(L, P):
        tl.store(padded_ptr + offset + t, 0.0)


@triton.jit
def real_dft_kernel(
    padded_ptr,         # *const float, padded input vector of length P = 2*L
    out_real_ptr,       # *float, output real part vector of length L+1
    out_imag_ptr,       # *float, output imag part vector of length L+1
    B: tl.constexpr,
    C: tl.constexpr,
    L,                  # actual seqlen (int)
    P: tl.constexpr,    # padded length = 2*L (int)
    inv_P,              # float32, 1.0 / P
):
    # Grid over (B, C, L+1)
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    k = tl.program_id(2)  # frequency index in [0, L]

    # Compute base pointer for this (b, c)
    bc = pid_b * C + pid_c
    base = bc * (L + 1)

    # Accumulator as float32
    acc = 0.0

    # Compute DFT: X[k] = sum_{t=0}^{P-1} x[t] * exp(-2πi k t / P)
    # Since Triton doesn't have complex types, we compute as real sin/cos:
    # exp(-i theta) = cos(theta) - i sin(theta). We accumulate real part into acc.
    # Imaginary part is stored explicitly using sin; but for real rfft, imag should be zero for k>0.
    for t in range(0, P):
        x_t = tl.load(padded_ptr + t)
        theta = -2.0 * 3.141592653589793 * float(k) * float(t) * inv_P
        # Real contribution: x_t * cos(theta)
        acc += x_t * tl.cos(theta)

    # Normalize by P
    acc *= inv_P

    # Store real and imaginary parts
    tl.store(out_real_ptr + base + k, acc)
    # Imaginary part: for real input, rfft produces purely real outputs except possible numerical noise; we store 0.
    tl.store(out_imag_ptr + base + k, 0.0)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        Triton-optimized version:
        - Pads seqlen to 2*L and computes real DFT via Triton kernels.
        - Returns real and imaginary parts of shape (B, C, L+1), normalized by 2*L.
        """
        assert x.is_cuda, "Input must be on CUDA device for Triton kernels."
        assert x.dtype == torch.float32, "Input must be float32."
        x = x.contiguous()
        B, C, L = x.shape
        P = 2 * L  # padded length

        # Flatten pointers for Triton kernels
        # 1) Build zero-padded vectors for all (b, c) in a single flattened buffer of length M = (B*C)*P
        M = (B * C) * P
        padded = torch.empty(M, dtype=torch.float32, device=x.device)

        grid_pad = (B * C,)
        pad_kernel[grid_pad](
            x, padded,
            B, C, L, P,
            num_warps=4,
        )

        # 2) Compute real DFT for k in [0, L], normalized by P
        out_real = torch.empty(M, dtype=torch.float32, device=x.device)
        out_imag = torch.empty(M, dtype=torch.float32, device=x.device)

        grid_dft = (B, C, L + 1)

            padded, out_real, out_imag,

        x_freq_real = out_real.view(B, C, L + 1)
        x_freq_imag = out_imag.view(B, C, L + 1)

        return x_freq_real, x_freq_imag
