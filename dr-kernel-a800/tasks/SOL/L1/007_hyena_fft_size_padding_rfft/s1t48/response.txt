import triton
import triton.language as tl


@triton.jit
def _rfft_real_imag_kernel(
    x_ptr,            # *const float32, input pointer to x with shape (B, C, L)
    out_real_ptr,     # *float32, output pointer to real part (B, C, L+1) flattened
    out_imag_ptr,     # *float32, output pointer to imag part (B, C, L+1) flattened
    B: tl.constexpr,  # batch size
    C: tl.constexpr,  # channels
    L: tl.constexpr,  # seqlen
    N: tl.constexpr,  # N = 2 * L
    M: tl.constexpr,  # M = L + 1
):
    # One program per (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)

    # Base pointer offset for this (b, c) slice
    # x is laid out as (B, C, L) contiguous, so linear index = ((b*C + c)*L) + t
    # We'll compute base pointers and iterate t from 0 to N-1.

    # Compute the base linear offsets for output (B, C, M) flattened:
    # For a given (b, c), the contiguous range over M is linear indices:
    # base_out = ((b*C + c)*M)
    base_out = (b * C + c) * M

    # Loop over j in chunks of 1 (robust scalar accumulation)
    # We'll compute re and im for j = 0..M-1
    # Initialize accumulators as scalars
    j = 0
    while j < M:
        # Accumulators for this j
        acc_real = tl.zeros((), dtype=tl.float32)
        acc_imag = tl.zeros((), dtype=tl.float32)

        # Sum over t = 0..N-1
        t = 0
        while t < N:
            # Load x[b, c, t] as scalar
            # x_ptr is a 1D pointer to contiguous data; index = ((b*C + c)*L) + t
            # Here we linearize (b,c) into a single dimension and use t as the last dim.
            # Since Triton program_id(1) already gives c, we can compute:
            # index = (b*C + c)*L + t
            index = (b * C + c) * L + t
            x_val = tl.load(x_ptr + index)

            # Compute angle = 2*pi*j*t/N
            # Note: j is scalar; N,M are const; use float32 math
            angle = 2.0 * 3.141592653589793 * (j * t) / N

            # Accumulate
            acc_real += x_val * tl.cos(angle)
            acc_imag += x_val * tl.sin(angle)

            t += 1

        # Normalize by 1/N
        inv_N = 1.0 / N
        acc_real *= inv_N
        acc_imag *= inv_N

        # Store results to out[b, c, j]
        # out is (B, C, M) contiguous flattened: offset = base_out + j
        tl.store(out_real_ptr + base_out + j, acc_real)
        tl.store(out_imag_ptr + base_out + j, acc_imag)

        j += 1

        )

        out_real = out_real.view(B, C, M)
        out_imag = out_imag.view(B, C, M)

        return out_real, out_imag
