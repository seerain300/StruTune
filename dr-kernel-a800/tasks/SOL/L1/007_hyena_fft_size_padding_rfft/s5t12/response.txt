import triton
import triton.language as tl


@triton.jit
def cosine_sum_kernel(x_ptr, out_ptr, j, N, stride_row):
    """
    Compute real_rfft[j] for j in [0, M], where M=seqlen and N=2*seqlen.
    real_out[j] = (1/N) * sum_{k=0..N-1} x[k] * cos(2*pi*j*k/N)
    out_ptr points to the start of the output for this (b, c) row. We store at offset j.
    stride_row is the number of columns (seqlen+1).
    """
    acc = 0.0
    # Sum over k = 0..N-1
    k = 0
    while k < N:
        xk = tl.load(x_ptr + k)
        angle = 2.0 * 3.141592653589793 * j * k / N
        acc += xk * tl.cos(angle)
        k += 1

    Compute imag_rfft[j] for j in [1, M-1], where M=seqlen and N=2*seqlen.

        # For each (b, c), we launch Triton kernels to fill the row.
