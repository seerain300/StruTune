import triton
import triton.language as tl


# Triton kernel: compute real DFT over a zero-padded input vector for each (b, c).
# One program handles one (b, c), and iterates k from 0 to L, accumulating X[k].
# The input padded vector has length two_L = 2 * L; the first L entries are x,
# the remaining entries are zeros.
@triton.jit
def real_dft_kernel(
    x_ptr,          # *float32, pointer to input x[b, c, :] (already zero-padded flattened view)
    out_real_ptr,   # *float32, pointer to output real part (flattened buffer of length B*C*(L+1))
    out_imag_ptr,   # *float32, pointer to output imag part (flattened buffer of length B*C*(L+1))
    B: tl.constexpr,  # batch size (constexpr in this simple case)
    C: tl.constexpr,  # channels (constexpr in this simple case)
    L,                 # seqlen (runtime int)
    two_L,             # 2 * L (runtime int)
    stride_bc,         # stride between (b, c) slices in flattened buffers (runtime int)
):
    # program id = linear index over (b, c)
    pid = tl.program_id(0)
    # compute b, c from pid (assuming pid in [0, B*C))
    b = pid // C
    c = pid % C

    # base offsets for output tensors
    bc_offset = b * C + c
    base_real = bc_offset * (L + 1)
    base_imag = bc_offset * (L + 1)

    # accumulate X[k] for k in [0, L]
    # For real inputs, output imaginary part is zero, but we compute it to be general.
    for k in range(0, L + 1):
        sum_real = 0.0
        sum_imag = 0.0
        # sum over t = 0..(2*L - 1)
        for t in range(0, two_L):
            # load x[t]; vectorized indexing assumed contiguous
            val = tl.load(x_ptr + t)  # x_ptr points to a contiguous vector of length 2*L
            # exp(-2*pi*i*k*t / (2*L)) = cos(...) + i*sin(...)

            cos_a = tl.cos(angle)
            sin_a = tl.sin(angle)
            # accumulate
            sum_real += val * cos_a
            sum_imag += val * sin_a

        # normalize by 2*L
        norm = 1.0 / two_L
        sum_real *= norm
        sum_imag *= norm

        # store real and imaginary parts at position (b, c, k)
        out_real_idx = base_real + k
        out_imag_idx = base_imag + k
        tl.store(out_real_ptr + out_real_idx, sum_real)
        tl.store(out_imag_ptr + out_imag_idx, sum_imag)


# Helper: flatten x[b, c, :] to a contiguous vector of length 2*L for each (b, c)
# This is done using PyTorch to form the tensor (forward cannot use torch.rfft).
def pad_to_twoL(x_bc: torch.Tensor, two_L: int) -> torch.Tensor:
    """
    x_bc: 1D tensor of length L
    Returns: 1D tensor of length two_L, first L entries = x_bc, next two_L-L entries = 0
    """
    assert x_bc.numel() == x_bc.shape[0]
    pad = torch.zeros(two_L - x_bc.shape[0], device=x_bc.device, dtype=x_bc.dtype)
    return torch.cat([x_bc, pad])

        Compute real FFT-like behavior without torch.rfft, using Triton kernels.
        Input: x (B, C, L) float32
        Output: (B, C, L+1) tensors, real and imag parts (imag is zero for real inputs)
        """

        B, C, L = x.shape
        two_L = 2 * L

        # Ensure float32 contiguous
        x_f32 = x.contiguous().to(torch.float32)

        # Prepare flattened input for Triton: for each (b, c), we have a zero-padded vector of length 2*L
        # We'll pack them into a single buffer and compute offsets in the kernel.
        # However, Triton kernel expects a pointer to a contiguous vector of length 2*L per (b, c).
        # Easiest: flatten x to (B*C, L), then pad to 2*L, and store as list of tensors. But Triton can handle
        # a single contiguous buffer of size B*C*2*L. We'll create that buffer on the fly.
        # Build the buffer: for each (b, c), place x[b, c, :] into positions [c_offset, c_offset+L], zeros elsewhere.
        # But since we don't know c_offset at runtime, we'll create a contiguous buffer by iterating (b,c) and
        # placing each x[b,c,:] into consecutive slots of length 2*L. This requires a host-side loop; however,
        # Triton can still operate on this buffer as a single tensor.

        # Allocate the input buffer for Triton: [B*C, 2*L] flattened as 1D
        x_flat = torch.empty((B * C) * two_L, device=x_f32.device, dtype=torch.float32)

        # Fill the buffer: for each (b, c), copy x[b, c, :] into x_flat[pid * two_L : (pid+1) * two_L]
        # We need to flatten x_bc to 1D and pad to 2*L.

        for bc in range(B * C):

            c = bc % C

        # torch.zero_ is fine and ensures correctness. If needed, replace with Triton kernel.

            x_flat,              # input buffer (B*C*2*L) contiguous
            out_real.view(-1),   # flatten to 1D for simple linear addressing: size = B*C*(L+1)
            out_imag.view(-1),   # flatten to 1D
            B, C, L, two_L, (L + 1),  # stride_bc is (L+1) since outputs are (L+1) per (b, c)
        )

        # but zero_ is already efficient and correct.
        return out_real, out_imag

# Output: real and imag parts of length L+1, normalized by 2*L.
@triton.jit

    x_ptr,          # *float32, pointer to input vector (length = 2*L), laid out as B*C contiguous chunks
    out_real_ptr,   # *float32, pointer to output real part flattened as 1D (length = B*C*(L+1))
    out_imag_ptr,   # *float32, pointer to output imag part flattened as 1D (length = B*C*(L+1))
    B: tl.constexpr,
    C: tl.constexpr,
    L,                 # runtime int
    two_L,             # runtime int
    stride_bc,         # stride between (b, c) in flattened outputs: L+1
):
    pid = tl.program_id(0)  # linear id over (b, c)
    b = pid // C
    c = pid % C

    base_real = (b * C + c) * (L + 1)
    base_imag = (b * C + c) * (L + 1)

    # Compute DFT for k in [0..L]
    for k in range(0, L + 1):
        sum_real = 0.0
        sum_imag = 0.0
        for t in range(0, two_L):
            val = tl.load(x_ptr + t)  # x_ptr points to a contiguous vector of length 2*L
            angle = -2.0 * 3.141592653589793 * k * t / two_L  # imag part uses sin
            cos_a = tl.cos(angle)
            sin_a = tl.sin(angle)
            sum_real += val * cos_a
            sum_imag += val * sin_a

        sum_real *= norm
        sum_imag *= norm

    Returns: 1D tensor of length two_L, first L entries = x_bc, next two_L - L entries = 0

    pad = torch.zeros(two_L - x_bc.shape[0], device=x_bc.device, dtype=x_bc.dtype)
    return torch.cat([x_bc, pad])

    def forward(self, x: torch.Tensor):
