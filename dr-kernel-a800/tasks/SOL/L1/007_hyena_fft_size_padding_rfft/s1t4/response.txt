import triton
import triton.language as tl


@triton.jit
def copy_to_real_kernel(y_ptr, out_real_ptr, N_elements, BLOCK_SIZE: tl.constexpr):
    """
    Copy the real part from y_ptr (length N_elements) to out_real_ptr (length N_elements).
    Assumes y_ptr is a contiguous vector of complex output; we only copy real part.
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N_elements

    # In practice, after computing y = rfft, we extract .real as a float tensor and pass that to kernel.
    # This kernel just copies that float tensor.
    vals = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    tl.store(out_real_ptr + offsets, vals, mask=mask)

import torch
import triton
import triton.language as tl


@triton.jit
def _compute_real_part_kernel(
    x_ptr, out_real_ptr,
    B, C, N, M,
    stride_b, stride_c, stride_t,
    out_stride_b, out_stride_c, out_stride_m,
    BLOCK_T: tl.constexpr,
):
    # Each program handles one (b, c) slice
    b = tl.program_id(0)
    c = tl.program_id(1)

    # Base offsets for this (b, c) slice
    base_x = b * stride_b + c * stride_c
    base_out = b * out_stride_b + c * out_stride_c

    # Compute re_j = (1/N) * sum_{t=0}^{N-1} x[t] * cos(2*pi*j*t/N)
    # Iterate j from 0 to M-1
    for j in range(0, M):
        acc = 0.0
        # Vectorize over t in chunks
        for t_start in range(0, N, BLOCK_T):
            t_offsets = t_start + tl.arange(0, BLOCK_T)
            mask = t_offsets < N
            # Load x[b, c, t]
            x_vals = tl.load(x_ptr + base_x + t_offsets * stride_t, mask=mask, other=0.0)
            # cos(2*pi*j*t/N)
            angle = 2.0 * 3.141592653589793 * j * t_offsets * (1.0 / N)
            cosv = tl.cos(angle)
            # multiply and reduce
            acc += tl.sum(x_vals * cosv, axis=0)
        # normalize by N
        re = acc * (1.0 / N)
        # store to out_real[b, c, j]
        tl.store(out_real_ptr + base_out + j * out_stride_m, re)
