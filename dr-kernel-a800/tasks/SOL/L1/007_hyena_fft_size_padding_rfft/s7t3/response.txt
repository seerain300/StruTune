@triton.jit
def _rfft_zero_pad_masked_kernel(x_ptr, real_out_ptr, imag_out_ptr, seqlen, N_in, scale):
    """
    Triton kernel that emulates torch.fft.rfft(x, n=N_in) for real input x of length seqlen,
    by zero-padding to N_in = 2 * seqlen and computing the exact real FFT via direct summation.
    We produce only the first seqlen + 1 outputs (matching torch.rfft output length for n=2*seqlen).
    Inputs:
      x_ptr: pointer to flattened input of length effectively seqlen (we load with mask n < seqlen).
      real_out_ptr, imag_out_ptr: pointers to outputs of length seqlen + 1.
      seqlen: original sequence length.
      N_in: padded length = 2 * seqlen.
      scale: normalization factor = 1.0 / (2.0 * seqlen).
    """
    k = tl.program_id(axis=0)  # each program handles one output frequency index k in [0, seqlen + 1)

    # Accumulators for real and imaginary parts
    acc_real = 0.0
    acc_imag = 0.0

    # Direct summation over padded length N_in, zero-padding beyond seqlen via masked loads
    n = 0
    while n < N_in:
        valid = n < seqlen
        x_n = tl.load(x_ptr + n, mask=valid, other=0.0)
        angle = 2.0 * math.pi * k * n / N_in
        cos_term = tl.cos(angle)
        sin_term = tl.sin(angle)
        acc_real += x_n * cos_term
        acc_imag += x_n * sin_term
        n += 1

    # Normalize by 2*seqlen
    y_real = acc_real * scale
    y_imag = acc_imag * scale

    # Store results (k in [0, seqlen + 1))
    tl.store(real_out_ptr + k, y_real)
    tl.store(imag_out_ptr + k, y_imag)


def _ceil_div(a: int, b: int) -> int:
    return (a + b - 1) // b
