import torch
import triton
import triton.language as tl

# 1) Copy input row x[b, c, :] (length L) into padded buffer of length n=2*L, fill tail with zeros
@triton.jit
def _copy_row_to_padded_kernel(
    in_ptr,                # *const float
    out_ptr,               # *float
    L: tl.constexpr,       # int
    n: tl.constexpr,       # int, n = 2 * L
    BLOCK: tl.constexpr,   # block size along L
):
    pid = tl.program_id(0)  # one program per (b, c) element, we'll pass L/C via grid
    # We derive b, c from pid in host: grid = (B*C,) so pid is index
    # But since we launch one program per (b,c), host code controls mapping.
    # Compute offsets for j in [0, L)
    j = tl.arange(0, BLOCK)
    mask = j < L
    # Load x[pid, j]
    x = tl.load(in_ptr + pid * L + j, mask=mask, other=0.0)
    # Store to out[pid, j] and zero pad to n
    # For j >= L, value should be 0; we'll pre-zero out_ptr and only store j < L
    # However, to keep it simple, we store only valid j. Host zeros the rest.
    tl.store(out_ptr + pid * n + j, x, mask=mask)
    # Tail is already zeroed in host before launching this kernel.

# 2) Compute cos_table[j, k] for j in [0..n-1], k in [0..L]
@triton.jit
def _compute_cos_table_kernel(
    out_ptr,               # *float, shape (n, L+1)
    n: tl.constexpr,       # int
    L: tl.constexpr,       # int
    BLOCK_K: tl.constexpr, # block size along k
):
    j = tl.program_id(0)   # each program handles one j
    k = tl.arange(0, BLOCK_K)
    k_mask = k < (L + 1)
    # Compute cos(2*pi*k*j / n)
    # For j == 0, cos == 1 for all k
    denom = 2.0 * 3.141592653589793 * k * (j if j != 0 else 0) / n  # j==0 guarded by value later
    # We should avoid division by zero; since j ranges 0..n-1, only j==0 is special:
    cos_val = tl.where(j == 0, tl.full([BLOCK_K], 1.0, tl.float32), tl.cos(denom))
    tl.store(out_ptr + j * (L + 1) + k, cos_val, mask=k_mask)

# 3) Compute sin_table[j, k] for j in [0..n-1], k in [0..L]
@triton.jit
def _compute_sin_table_kernel(
    out_ptr,               # *float, shape (n, L+1)
    n: tl.constexpr,       # int
    L: tl.constexpr,       # int
    BLOCK_K: tl.constexpr, # block size along k
):
    j = tl.program_id(0)   # each program handles one j
    k = tl.arange(0, BLOCK_K)
    k_mask = k < (L + 1)
    # Compute sin(2*pi*k*j / n); j==0 => sin==0
    denom = 2.0 * 3.141592653589793 * k * (j if j != 0 else 0) / n
    sin_val = tl.where(j == 0, tl.full([BLOCK_K], 0.0, tl.float32), tl.sin(denom))
    tl.store(out_ptr + j * (L + 1) + k, sin_val, mask=k_mask)

# 4) Reduce real: y_real[k] = sum_j padded[j] * cos_table[j, k], k in [0..L]
@triton.jit
def _reduce_real_kernel(
    padded_ptr,            # *const float, shape (B*C, n)
    cos_ptr,               # *const float, shape (n, L+1)
    out_ptr,               # *float, shape (B*C, L+1)
    L: tl.constexpr,       # int
    n: tl.constexpr,       # int
    BLOCK_J: tl.constexpr, # block size along j
):
    pid = tl.program_id(0)  # per (b,c) slice
    k = tl.program_id(1)    # per k
    acc = tl.zeros((), dtype=tl.float32)
    # Sum over j in blocks
    for jj in range(0, n, BLOCK_J):
        j_idx = jj + tl.arange(0, BLOCK_J)
        mask_j = j_idx < n
        # Load padded[j]
        xj = tl.load(padded_ptr + pid * n + j_idx, mask=mask_j, other=0.0)
        # Load cos_table[j, k]
        cos_val = tl.load(cos_ptr + j_idx * (L + 1) + k, mask=mask_j, other=0.0)
        # Accumulate
        acc += tl.sum(xj * cos_val, axis=0)
    tl.atomic_add(out_ptr + pid * (L + 1) + k, acc)

# 5) Reduce imag: y_imag[k] = sum_j padded[j] * sin_table[j, k], k in [0..L]
@triton.jit
def _reduce_imag_kernel(
    padded_ptr,            # *const float, shape (B*C, n)
    sin_ptr,               # *const float, shape (n, L+1)
    out_ptr,               # *float, shape (B*C, L+1)
    L: tl.constexpr,       # int
    n: tl.constexpr,       # int
    BLOCK_J: tl.constexpr, # block size along j
):
    pid = tl.program_id(0)  # per (b,c) slice
    k = tl.program_id(1)    # per k
    acc = tl.zeros((), dtype=tl.float32)
    # Sum over j in blocks
    for jj in range(0, n, BLOCK_J):
        j_idx = jj + tl.arange(0, BLOCK_J)
        mask_j = j_idx < n
        # Load padded[j]
        xj = tl.load(padded_ptr + pid * n + j_idx, mask=mask_j, other=0.0)
        # Load sin_table[j, k]
        sin_val = tl.load(sin_ptr + j_idx * (L + 1) + k, mask=mask_j, other=0.0)
        # Accumulate
        acc += tl.sum(xj * sin_val, axis=0)
    tl.atomic_add(out_ptr + pid * (L + 1) + k, acc)

# 6) Normalize by scale = 2*L
@triton.jit
def _divide_by_scale_kernel(
    in_ptr,                # *float, shape (B*C, L+1)
    out_ptr,               # *float, shape (B*C, L+1)
    scale,                 # float32
    L: tl.constexpr,       # int
    BLOCK: tl.constexpr,   # block size along (B*C)*(L+1)
):
    pid = tl.program_id(0)  # linear index over (B*C)*(L+1)
    # We launch grid=(B*C, L+1); use program_id(0)=bc, program_id(1)=k
    # But this kernel is 1D; emulate via pid
    num_k = L + 1
    bc = pid // num_k
    k = pid % num_k
    val = tl.load(in_ptr + bc * (num_k) + k)
    val = val / scale
    tl.store(out_ptr + bc * (num_k) + k, val)

# 7) Store output: real/imag parts into (B, C, L+1) tensors
@triton.jit
def _store_output_kernel(
    in_ptr,                # *float, shape (B*C, L+1)
    out_ptr,               # *float, shape (B, C, L+1)
    L: tl.constexpr,       # int
    BLOCK: tl.constexpr,   # block size
):
    bc = tl.program_id(0)  # per (b,c)
    k = tl.program_id(1)   # per k in [0..L]
    val = tl.load(in_ptr + bc * (L + 1) + k)
    # Write to out[b, c, k]
    tl.store(out_ptr + bc * (L + 1) + k, val)

class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        x: (B, C, L) float32 tensor
        Returns:
            x_freq_real: (B, C, L+1) float32
            x_freq_imag: (B, C, L+1) float32
        """
        assert x.ndim == 3, "Input must be (B, C, L)"
        B, C, L = x.shape
        n = 2 * L

        # 1) Prepare padded buffer: (B*C, n)
        x_contig = x.contiguous()
        padded = torch.zeros((B * C, n), dtype=torch.float32, device=x.device)

        # Launch copy kernel to fill first L entries
        BLOCK = 256
        grid_copy = (B * C,)
        _copy_row_to_padded_kernel[grid_copy](
            x_contig.reshape(-1, L),  # input row pointer for each (b,c)
            padded,                    # output padded buffer
            L, n, BLOCK,
        )

        # 2) Compute cos and sin tables: shape (n, L+1)
        cos_table = torch.empty((n, L + 1), dtype=torch.float32, device=x.device)
        sin_table = torch.empty((n, L + 1), dtype=torch.float32, device=x.device)
        BLOCK_K = 256
        grid_cos = (n,)
        _compute_cos_table_kernel[grid_cos](
            cos_table, n, L, BLOCK_K
        )
        grid_sin = (n,)
        _compute_sin_table_kernel[grid_sin](
            sin_table, n, L, BLOCK_K
        )

        # 3) Allocate outputs for real/imag (B*C, L+1)
        out_real = torch.zeros((B * C, L + 1), dtype=torch.float32, device=x.device)
        out_imag = torch.zeros((B * C, L + 1), dtype=torch.float32, device=x.device)

        # 4) Reduction kernels: compute real and imag
        BLOCK_J = 256
        grid_reduce = (B * C, L + 1)
        _reduce_real_kernel[grid_reduce](padded, cos_table, out_real, L, n, BLOCK_J)
        _reduce_imag_kernel[grid_reduce](padded, sin_table, out_imag, L, n, BLOCK_J)

        # 5) Normalize by 2*L
        scale = 2.0 * float(L)
        out_real_norm = torch.empty_like(out_real, dtype=torch.float32, device=x.device)
        out_imag_norm = torch.empty_like(out_imag, dtype=torch.float32, device=x.device)
        grid_div = (B * C * (L + 1),)
        _divide_by_scale_kernel[grid_div](out_real, out_real_norm, scale, L, 128)
        _divide_by_scale_kernel[grid_div](out_imag, out_imag_norm, scale, L, 128)

        # 6) Store outputs into (B, C, L+1) using _store_output_kernel (ensures kernel is used)
        # Create empty outputs
        x_freq_real = torch.empty((B, C, L + 1), dtype=torch.float32, device=x.device)
        x_freq_imag = torch.empty((B, C, L + 1), dtype=torch.float32, device=x.device)
        # Launch store kernel
        grid_store = (B * C, L + 1)
        _store_output_kernel[grid_store](out_real_norm, x_freq_real, L, 128)
        _store_output_kernel[grid_store](out_imag_norm, x_freq_imag, L, 128)

        return x_freq_real, x_freq_imag
