import triton
import triton.language as tl


@triton.jit
def _rfft_real_imag_triton_kernel(
    x_ptr,                  # *const float32, input pointer to x (B, C, L) contiguous
    out_real_ptr,           # *float32, output pointer to real part (B, C, L+1)
    out_imag_ptr,           # *float32, output pointer to imag part (B, C, L+1)
    B: tl.constexpr,        # batch size (for indexing output)
    C: tl.constexpr,        # channels (for indexing output)
    L: tl.constexpr,        # seqlen
    N: tl.constexpr,        # n = 2 * seqlen (length of rfft)
):
    # One program per (b, c) slice
    pid = tl.program_id(axis=0)  # total programs = B*C
    b = pid // C
    c = pid % C

    # Precompute 2*pi / N as a float32
    two_pi_over_N = 2.0 * 3.14159265358979323846 / N  # float32

    # Vector of j indices for this program: we process all j in one go
    # Choose BLOCK_J = 256; mask ensures j < M where M = N//2 + 1
    M = N // 2 + 1
    j_offsets = tl.arange(0, 256)
    mask_j = j_offsets < M

    # Accumulators for real and imaginary parts, vectorized over j
    re = tl.zeros([256], dtype=tl.float32)
    im = tl.zeros([256], dtype=tl.float32)

    # Scalar loop over t = 0 .. N-1
    # For each t, accumulate into re[j] and im[j] for all j
    for t in range(N):
        # Load x[t] for this (b, c) slice. x is contiguous along last dim.
        # Address: ((b*C + c)*L + t) is the offset for (b, c, t)
        x_offset = (b * C + c) * L + t
        # Load as float32
        xt = tl.load(x_ptr + x_offset)
        # Compute cos and sin terms for all j in vector
        # angle = j * (2*pi*t / N)
        angle = j_offsets * two_pi_over_N * t
        ct = tl.cos(angle)
        st = tl.sin(angle)
        # Accumulate: re += xt * cos, im += xt * sin
        re += xt * ct
        im += xt * st

    # Normalize by N
    re = re * (1.0 / N)
    im = im * (1.0 / N)

    # Store results to out_real[b, c, j] and out_imag[b, c, j] for j in [0, M)
    # Output is contiguous (B, C, M). We compute linear index as ((b*C + c)*M + j)

            out_imag_offset = (b * C + c) * M + jj
            tl.store(out_real_ptr + out_real_offset, re[jj])
            tl.store(out_imag_ptr + out_imag_offset, im[jj])
