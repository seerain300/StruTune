```python
import torch
import triton
import triton.language as tl


@triton.jit
def normalize_divide_kernel(in_ptr, out_ptr, n_elements: tl.constexpr, scale: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise divide in_ptr by 'scale' and write to out_ptr.
    n_elements: total number of elements in the tensor.
    scale: scalar to divide by (2*seqlen).
    BLOCK_SIZE: Triton block size for parallelism.
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    y = x / scale
    tl.store(out_ptr + offsets, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor):
        """
        Fused pad and real FFT computation with Triton normalization.
        Input: x of shape (batch, channels, seqlen), float32
        Output: x_freq_real, x_freq_imag of shape (batch, channels, seqlen+1)
        """
        batch, channels, seqlen = x.shape
        # Cast to float32 for numerical stability
        x_f32 = x.to(torch.float32)

        # Compute rFFT with implicit zero-padding to n = 2*seqlen
        n = 2 * seqlen
        x_freq = torch.fft.rfft(x_f32, n=n)

        # Normalize by n (i.e., 2*seqlen)
        scale = float(n)

        # Extract real and imaginary parts
        x_freq_real = x_freq.real.contiguous()
        x_freq_imag = x_freq.imag.contiguous()

        # Flatten for Triton elementwise division
        n_real = x_freq_real.numel()
        n_imag = x_freq_imag.numel()
        # Allocate output tensors
        out_real = torch.empty_like(x_freq_real)
        out_imag = torch.empty_like(x_freq_imag)

        # Choose block size and grid
        BLOCK_SIZE = 1024
        grid_real = (triton.cdiv(n_real, BLOCK_SIZE),)
        grid_imag = (triton.cdiv(n_imag, BLOCK_SIZE),)

        # Launch Triton normalization kernels
        normalize_divide_kernel[grid_real](x_freq_real, out_real, n_real, scale, BLOCK_SIZE=BLOCK_SIZE)
        normalize_divide_kernel[grid_imag](x_freq_imag, out_imag, n_imag, scale, BLOCK_SIZE=BLOCK_SIZE)

        # Reshape back to (batch, channels, seqlen+1)
        out_real = out_real.view(batch, channels, seqlen + 1)
        out_imag = out_imag.view(batch, channels, seqlen + 1)

        return out_real, out_imag
```
