    batch: tl.int32,
    channels: tl.int32,

    x_stride_c: tl.int32,
    x_stride_l: tl.int32,
    x2_stride_b: tl.int32,
    x2_stride_c: tl.int32,
    x2_stride_l: tl.int32,
    BLOCK_N: tl.constexpr
):

    b = pid // channels
    c = pid % channels

    start = 0
