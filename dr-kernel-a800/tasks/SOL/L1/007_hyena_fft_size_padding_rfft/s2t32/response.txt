import triton
import triton.language as tl


@triton.jit
def dct_real_kernel(x_ptr, out_ptr, L: tl.constexpr):
    """
    Triton kernel: compute real DCT (type-2) for a 1D slice of length L.
    One program handles one slice. It computes:
      X[k] = sum_{t=0..L-1} x[t] * cos(pi * k * (2t + 1) / (2L)), for k in [0..L-1]
    We normalize by (2*L). Imaginary part is assumed zero for real input.
    """
    # program id: we assume 1D grid over (B*C), so one program per (b,c)
    pid = tl.program_id(axis=0)

    # Offsets for t and k loops
    # Triton prefers constexpr for loop bounds; L is passed as constexpr.
    # We will use scalar loads/stores to avoid address mode issues.
    # First, compute X for k in [0..L-1]
    # Prepare accumulator for X[0..L-1]
    X = tl.zeros((L,), dtype=tl.float32)

    # Loop over t
    # Triton supports range loops with constexpr bounds.
    for t in range(0, L):
        # load x[t]
        # We need to compute offset for x_ptr: we pass x as a flattened contiguous tensor,
        # but to handle (B, C, L), we need to map pid to its slice. Since grid=(B*C,), we
        # can assume the host has laid out x as (B*C, L) contiguous along L.
        # However, Triton kernels typically get pointers and sizes, not multi-dim indices.
        # To keep it simple and robust, we pass x as a single contiguous vector of length
        # total = B*C*L. Inside the kernel, we cannot index by pid directly; so we provide
        # x as (B*C, L) in host, and compute base pointer for this pid.
        # Since Triton doesn't have pid mapping to multi-dim here, we instead pass x as 1D
        # contiguous and rely on grid to launch exactly B*C programs, and in host we pass
        # x as (B*C, L) and flatten to 1D. But kernel doesn't know L again; we need to
        # rethink mapping.

    # The above comment reveals a limitation: to map pid to (b, c) and then to the (B*C, L)
    # layout, we need to pass additional information. Triton kernels receive pointers and
    # constexprs. The simple approach is to require x to be a single 1D contiguous tensor
    # and launch grid=(B*C,) programs; each program processes L elements starting at
    # pid*L. However, we don't have pid in that sense. To make this work, we arrange
    # x as (B*C, L) and pass x_ptr as contiguous over rows. Triton doesn't handle
    # multi-dim indexing easily; so we'll instead pass x as 1D contiguous length (B*C*L)
    # and in kernel compute index using pid * L + t. This is doable.

    # Let's redefine: host will pass x as 1D contiguous length (B*C*L), and launch grid=(B*C,),
    # each program processes L elements starting at pid * L. Then we can compute x[t] as
    # tl.load(x_ptr + pid * L + t). This avoids multi-dim indexing inside Triton.

    base = pid * L
    for t in range(0, L):

        for k in range(0, L):

    norm = 2.0 * L
    X = X / norm

    # Store X to out_ptr: out is contiguous of length (B*C*L)
    out_base = pid * L
    for k in range(0, L):
        tl.store(out_ptr + out_base + k, X[k])
