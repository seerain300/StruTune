import torch
import triton
import triton.language as tl


@triton.jit
def histogram_kernel(original_ptr, counts_ptr, M: tl.int32, N: tl.int32, BLOCK: tl.constexpr):
    # Count occurrences of each value in [0..N-1]
    for start in range(0, N):  # N is small (e.g., 256) — one iteration
        for i in range(0, M, BLOCK):
            idx = i + tl.arange(0, BLOCK)
            valid = idx < M
            vals = tl.load(original_ptr + idx, mask=valid, other=0)
            # Accumulate counts per lane
            # Note: Triton atomics on int32 are supported for pointer addressing
            # Here we use scalar loads to avoid vector atomics: iterate per lane
            # But to keep performance, we use vectorized atomic adds via tl.atomic_add
            # Convert vector to scalar contribution: add 1 per valid & (vals==start)
            eq = vals == start
            # Sum up eq (int1) per element
            count_contrib = tl.sum((eq & valid).to(tl.int32), axis=0)
            # Atomic add to counts[start]
            # counts_ptr[start] += count_contrib
            tl.atomic_add(counts_ptr + start, count_contrib)


@triton.jit
def cumsum_kernel(counts_ptr, prefix_ptr, N: tl.int32, BLOCK: tl.constexpr):
    # Compute inclusive prefix sum of counts for N values (int32)
    # We implement it in a single block, iterating per element with scalar loads/stores.
    # counts_ptr[0..N-1], prefix_ptr[0..N-1]
    for i in range(N):
        val = tl.load(counts_ptr + i)
        tl.store(prefix_ptr + i, val)
        # Update subsequent positions: prefix[j] += prefix[j-1]
        for j in range(i + 1, N):
            prev = tl.load(prefix_ptr + (j - 1))
            curr = tl.load(prefix_ptr + j)
            tl.store(prefix_ptr + j, curr + prev)


@triton.jit
def assemble_offsets_kernel(prefix_ptr, offsets_ptr, N: tl.int32, BLOCK: tl.constexpr):
    # offsets[0] = 0
    tl.store(offsets_ptr + 0, 0)
    if N > 1:

        prev = tl.load(prefix_ptr + 0)
        tl.store(offsets_ptr + 1, prev)
        for i in range(2, N + 1):

            tl.store(offsets_ptr + i, prev)
    else:
        tl.store(offsets_ptr + 1, 0)

    MAX_M = 8192  # upper bound for M in provided workloads; avoids dynamic loops in Triton
    for v in range(N):

        number_of_less = 0
        if v > 0:

            original_i = tl.load(original_ptr + i, mask=valid_i, other=0)

            neqb = 0
