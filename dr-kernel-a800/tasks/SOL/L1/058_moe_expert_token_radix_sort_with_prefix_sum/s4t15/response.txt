    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: parallel histogram of expert ids across a flat tensor
# Each program processes BLOCK elements and performs atomic adds per expert bin.
@triton.jit
def _histogram_kernel(flat_ptr, N, counts_ptr, num_experts: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask = offsets < N

    # Load flattened indices and ensure int32
    idx = tl.load(flat_ptr + offsets, mask=mask, other=0)
    idx = idx.to(tl.int32)

    # Build counts vector for each expert id (num_experts is a constexpr here)
    counts_vec = tl.zeros((num_experts,), dtype=tl.int32)
    for i in range(num_experts):
        matches = idx == i
        # Convert boolean to int32 and reduce across the vector to get the count for this bin
        # For masked-out positions, matches is False and contributes 0.
        cnt = tl.sum((matches & mask).to(tl.int32), axis=0)
        counts_vec = counts_vec + (cnt == 1) * 0  # placeholder to keep vector form; see below
        # Note: Triton does not support dynamic Python loops with runtime-dependent bounds well.
        # However, since num_experts is a tl.constexpr, this loop is unrolled at compile time.
        # We need to accumulate 'cnt' into the counts_ptr via atomic add.
        # The above pattern is a common way to compute vectorized reductions in Triton.

    # Atomic add the counts for each bin into counts_ptr
    for i in range(num_experts):
        # Retrieve the count for expert i (scalar) from counts_vec[i] by reduction trick:
        # Since we cannot index a Triton vector with a Python variable directly, we recompute cnt here
        # using the same boolean mask. This is fine because the loop is unrolled.
        matches = idx == i
        cnt = tl.sum((matches & mask).to(tl.int32), axis=0)
        tl.atomic_add(counts_ptr + i, cnt)


# Triton kernel: compute inclusive cumulative offsets per expert (exclusive prefix sum + write inclusive)
# This kernel iterates through experts in static-sized chunks to compute local sums and carry.
@triton.jit
def _compute_offsets_kernel(counts_ptr, offsets_ptr, num_experts: tl.constexpr, CHUNK: tl.constexpr):
    # First pass: compute local inclusive sums per chunk and store them temporarily in offsets_ptr[1:]
    # Then, second pass: add carry (prefix sum from previous chunks) to each chunk and write final offsets.
    # We use a static-sized loop over chunks; num_experts is passed as constexpr to allow unrolling.
    # Initialize global carry to 0
    carry = 0

    # We will iterate chunk-by-chunk. For each chunk, compute inclusive sums and then a second pass with carry.
    # Since Triton doesn't support dynamic while loops with runtime num_experts, we unroll with tl.constexpr.
    # Loop structure:
    # For chunk_start in 0, CHUNK, 2*CHUNK, ... up to num_experts:
    #   local_sums = [sum(offsets[chunk_start + i] for i in [0..CHUNK-1])]
    #   Write local_sums to offsets_ptr[chunk_start + i + 1] (final positions)
    #   Second pass: for each element in the chunk, add carry and write into offsets_ptr[i + chunk_start + 1]
    #   Update carry += sum of counts in this chunk.
    # We need to compute sums. We'll do it by looping over CHUNK positions and adding counts.
    # Note: This approach keeps everything in Triton and avoids torch.cumsum.
    # To simplify, we use a loop with step CHUNK. Triton supports static ranges; we emulate chunk iteration
    # by using a vectorized reduction across CHUNK positions. However, implementing inclusive scan
    # inside a Triton kernel robustly requires a pattern. Here, we adopt a simple loop structure that
    # iterates chunk_start from 0 to num_experts with step CHUNK, and for each chunk, iterate j in range(CHUNK)
    # to compute local sums. This keeps the kernel simple and correct for num_experts up to CHUNK*loops.
    # Given num_experts=256 and CHUNK=32, we'll do it in three passes.

    # Implement chunked inclusive scan: not straightforward in Triton without extra memory, so we switch to
    # a simpler pattern that loops over all experts sequentially and writes offsets one-by-one. While not
    # chunked, it ensures correctness and avoids the need for a built-in cumsum in torch.
    # We will instead implement a sequential per-expert offset computation in Triton by iterating i from 0..num_experts-1.
    # Each program can process multiple i if we use a grid with a single program; however, Triton does not
    # support dynamic grid sizing based on num_experts. To keep it simple and correct, we will run a single
    # program and use a static loop over all experts (num_experts is constexpr). This avoids torch and host
    # reductions entirely.

    # Single-program kernel that writes offsets in sequence:
    # This kernel is launched with grid=(1,) to ensure only one program handles the sequential loop.
    # We'll keep the same signature but force grid to 1 in forward.
    i = 0
    # We need to know num_experts at compile time for this pattern. We pass num_experts as tl.constexpr
    # and Triton unrolls this loop.
    while i < num_experts:
        cnt = tl.load(counts_ptr + i)  # scalar load
        carry = carry + cnt
        tl.store(offsets_ptr + i + 1, carry)
        i += 1


def _run_triton_only(topk_idx: torch.Tensor):
    # Ensure CUDA and contiguous
    if not topk_idx.is_cuda:
        topk_idx = topk_idx.to(device="cuda")
    topk_idx = topk_idx.contiguous()

    # Flatten
    flat = topk_idx.reshape(-1)
    N = flat.numel()

    # Histogram counts (per expert)
    num_experts = 256
    counts = torch.zeros(num_experts, dtype=torch.int32, device=flat.device)

    # Triton histogram kernel: process BLOCK elements per program
    BLOCK = 1024
    grid = (triton.cdiv(N, BLOCK),)
    _histogram_kernel[grid](flat, N, counts, num_experts=num_experts, BLOCK=BLOCK, num_warps=4)

    # Compute expert offsets via Triton per-expert sequential accumulation
    offsets = torch.empty(num_experts + 1, dtype=torch.int32, device=flat.device)
    # We need grid=1 for the per-expert loop; set it accordingly.
    _compute_offsets_kernel[(1,)](counts, offsets, num_experts=num_experts, CHUNK=32)

    # Stable sort of flattened indices in PyTorch (data-independent on num_experts)
    sorted_token_indices = flat.sort(stable=True)[1].to(torch.int32)

    return sorted_token_indices, offsets
