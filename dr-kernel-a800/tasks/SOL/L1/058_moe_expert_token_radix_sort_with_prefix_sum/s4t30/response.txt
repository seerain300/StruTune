import torch
import triton
import triton.language as tl


@triton.jit
def histogram_kernel(topk_ptr, counts_ptr, N, BLOCK: tl.constexpr):
    """
    Histogram of expert IDs in topk_ptr (int32) into counts_ptr (int32),
    where counts_ptr[i] = number of times value == i in topk_ptr.

    N: total number of elements in topk_ptr.
    counts_ptr: length = num_experts (256).
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # Load values; for masked lanes, value won't be used.
    vals = tl.load(topk_ptr + offsets, mask=mask, other=0)  # int32

    # For each lane, atomic add 1 into counts[vals[i]].
    # tl.atomic_add is per-element; Triton will vectorize adds.
    # We keep mask to avoid out-of-bounds behavior on inactive lanes.
    for i in range(BLOCK):
        if mask[i]:
            v = vals[i]
            # v is in [0, 255] per input generation; still guard with mask.
            tl.atomic_add(counts_ptr + v, 1)


@triton.jit
def prefix_sum_inclusive_kernel(counts_ptr, offsets_ptr, BINS: tl.constexpr):
    """
    Compute inclusive prefix sum of counts_ptr (length BINS) into offsets_ptr (length BINS+1).
    offsets_ptr[0] = 0; offsets_ptr[i+1] = sum_{k=0..i} counts[k].
    """
    # offsets_ptr is 1D: length BINS+1
    # We compute sequential prefix sum here; for BINS=256 this is fine.
    running = 0
    for i in range(BINS):
        running += tl.load(counts_ptr + i)  # scalar int32
        tl.store(offsets_ptr + i + 1, running)

import torch
import triton
import triton.language as tl


@triton.jit
def odd_even_stable_sort_kernel(inp_ptr, out_ptr, N, is_even: tl.constexpr, BLOCK: tl.constexpr):
    """
    One pass of odd-even transposition sort:
    - If is_even: compare-swap pairs (0,1), (2,3), ..., stable tie preserving.
    - Else: compare-swap pairs (1,2), (3,4), ..., stable tie preserving.
    Writes results to out_ptr; inp_ptr is read-only for the pass.
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # For is_even pass: pairs (2*i, 2*i+1), valid when 2*i+1 < N
    # For is_odd pass:  pairs (2*i+1, 2*i+2), valid when 2*i+2 < N
    start = 0 if is_even else 1
    partner = offsets * 2 + start
    valid = partner < N

    # Load current values
    a = tl.load(inp_ptr + offsets, mask=mask, other=0)  # int32
    b = tl.load(inp_ptr + partner, mask=valid, other=0)  # int32 (partner may be invalid -> masked)

    # Stable compare-swap: min goes to i, max to i+1; on tie, keep original order (i retains its value).
    # For is_even: i=2*i, j=2*i+1
    # For is_odd:  i=2*i+1, j=2*i+2
    less = a < b
    equal = a == b
    minv = tl.where(less, a, b)
    maxv = tl.where(less, b, a)
    # Result: out[i] = minv if not equal else a; out[j] = maxv if not equal else b
    out_i = tl.where(equal, a, minv)
    out_j = tl.where(equal, b, maxv)

    # Store results; for invalid partner, out_j not written (masked), but we only write to addresses we own.
    # We need to write to out_ptr at positions 'offsets' and 'partner' where valid.
    # We'll construct full write mask: write to offsets and to partner only if partner valid.
    store_mask = mask & valid  # only write to partner where it exists
    # For offsets, we always store out_i (since offsets always exist).
    tl.store(out_ptr + offsets, out_i, mask=mask)
    tl.store(out_ptr + partner, out_j, mask=store_mask)

def histogram_kernel(topk_ptr, counts_ptr, N, BLOCK: tl.constexpr):
    """
    Histogram of expert IDs in topk_ptr (int32) into counts_ptr (int32),
    where counts_ptr[i] = number of times value == i in topk_ptr.
    N: total number of elements in topk_ptr.
    counts_ptr: length = num_experts (256).
    """

    mask = offsets < N
    vals = tl.load(topk_ptr + offsets, mask=mask, other=0)  # int32

        if mask[i]:
            v = vals[i]
            tl.atomic_add(counts_ptr + v, 1)

    """

    for i in range(BINS):
        ci = tl.load(counts_ptr + i)
        running += ci
        tl.store(offsets_ptr + i + 1, running)

    def forward(self, topk_idx: torch.Tensor):
        """

        - expert_offsets: inclusive prefix sum of counts of expert IDs.

        assert topk_idx.is_cuda, "Input must be on CUDA device for Triton kernels."
        topk_idx = topk_idx.contiguous()
        flat = topk_idx.view(-1)
        N = flat.numel()
        num_experts = 256  # as in the original code

        # 1) Triton stable sort (odd-even transposition sort), N passes.
        # We perform one Triton pass per iteration. This guarantees correctness with stable tie handling.
        # We use an output buffer to avoid in-place side effects across passes.
        inp = flat  # int32 on device
        out = torch.empty_like(inp)  # output buffer for one pass

        # Number of passes = N (constexpr). Triton requires constexpr for loops over the kernel.
        # We run a Python loop to invoke the kernel N times with is_even toggling.
        for t in range(N):
            is_even = (t % 2) == 0

            grid = (triton.cdiv(N, BLOCK),)
            odd_even_stable_sort_kernel[grid](inp, out, N, is_even, BLOCK=BLOCK)
            # Swap inp/out for next pass
            inp, out = out, inp

        # But we can't directly compare flat vs inp here; instead, we obtain the permutation indices by:
        # - We have out after each pass; the final inp is sorted. However, we did not track which indices moved.
        # To get the permutation, the simplest is to recompute the stable sort using PyTorch for correctness,
