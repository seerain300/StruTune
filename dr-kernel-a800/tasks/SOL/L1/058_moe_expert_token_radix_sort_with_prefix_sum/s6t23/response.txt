  - Compute exclusive prefix sums of counts to get offsets[v].

- Launch kernels in forward:

  - scan_exclusive_values_kernel: exclusive prefix sums for counts.
  - rank_atomic_kernel: local stable ranks.

import torch
import triton
import triton.language as tl


# Sort the flattened values using Triton counting sort (assumes values in [0, L-1], here L=256).
@triton.jit
def count_values_kernel(flat_ptr, counts_ptr, N, L: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
    # vals are in [0, L-1] as per get_inputs for num_experts=256. Increment counts for each non-masked val.
    for j in range(BLOCK_SIZE):
        v = vals[j]
        valid = mask[j]
        if valid:
            tl.atomic_add(counts_ptr + v, 1)


@triton.jit
def scan_exclusive_values_kernel(counts_ptr, offsets_ptr, L: tl.constexpr):
    # Single program computes exclusive prefix sum across counts[0..L-1].
    total = 0
    for i in range(L):
        prev = total
        total += tl.load(counts_ptr + i)
        tl.store(offsets_ptr + i, prev)


@triton.jit
def rank_atomic_kernel(flat_ptr, out_ptr, offsets_ptr, N, L: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    idxs = base + tl.arange(0, BLOCK_SIZE)
    mask = idxs < N
    vals = tl.load(flat_ptr + idxs, mask=mask, other=0)
    # For each block, compute local ranks stably: count elements j < idx with flat[j] == vals[j].
    for j in range(BLOCK_SIZE):
        i = base + j
        if mask[j]:
            v = vals[j]
            # local_rank = number of elements with value < i having same v
            rank = 0
            # compare all preceding positions in the block
            for k in range(BLOCK_SIZE):
                jk = base + k
                if mask[k] and jk < i and vals[k] == v:
                    rank += 1
            tl.atomic_add(out_ptr + i, rank)


@triton.jit
def placement_kernel(flat_ptr, out_ptr, offsets_ptr, N, L: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    idxs = base + tl.arange(0, BLOCK_SIZE)
    mask = idxs < N
    vals = tl.load(flat_ptr + idxs, mask=mask, other=0)
    # Compute local ranks to place stably
    for j in range(BLOCK_SIZE):
        i = base + j
        if mask[j]:
            v = vals[j]
            # local_rank already set by rank_atomic_kernel
            local_rank = tl.load(out_ptr + i)
            pos = tl.load(offsets_ptr + v) + local_rank
            # Place i at position pos
            tl.store(out_ptr + pos, i)  # out_ptr is permutation, but direct store is fine since indices are unique; use a temp buffer if needed.


# Triton histogram for expert offsets
@triton.jit
def histogram_kernel(flat_ptr, counts_exp_ptr, N, L: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
    for j in range(BLOCK_SIZE):
        v = vals[j]
        valid = mask[j]
        if valid:
            tl.atomic_add(counts_exp_ptr + v, 1)


# Triton inclusive scan for expert offsets: given counts_exp, compute prefix sums into offsets_exp
@triton.jit
def inclusive_scan_kernel(counts_exp_ptr, offsets_exp_ptr, L: tl.constexpr):
    # Single program computes inclusive prefix sums.
    total = 0
    for i in range(L):
        total += tl.load(counts_exp_ptr + i)
        tl.store(offsets_exp_ptr + i, total)
    # Last element is N; we set it explicitly
    tl.store(offsets_exp_ptr + L, N)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.L = 256  # num_experts
        # Choose a block size; N in provided workloads are small to moderate
        self.BLOCK = 1024

    def forward(self, topk_idx: torch.Tensor):
        """
        Returns:
          sorted_token_indices: int32 tensor of shape (N,) with permutation indices
          expert_offsets: int32 tensor of shape (num_experts+1,) where offsets[e] = inclusive count of elements with value < e; last is N
        """
        # Flatten and ensure int32
        flat = topk_idx.reshape(-1).to(torch.int32)
        N = flat.numel()
        device = flat.device

        # 1) Triton counting sort to produce sorted_token_indices
        # Allocate buffers
        out_perm = torch.empty(N, dtype=torch.int32, device=device)  # permutation of indices
        counts = torch.zeros(self.L, dtype=torch.int32, device=device)
        # Launch counting values
        grid_counts = (triton.cdiv(N, self.BLOCK),)
        count_values_kernel[grid_counts](flat, counts, N, self.L, self.BLOCK)
        # Exclusive prefix sums of counts to get offsets
        offsets = torch.empty(self.L, dtype=torch.int32, device=device)
        scan_exclusive_values_kernel[(1,)](counts, offsets, self.L)
        # Local ranks (stable)
        out_ranks = torch.zeros(N, dtype=torch.int32, device=device)
        rank_atomic_kernel[grid_counts](flat, out_ranks, offsets, N, self.L, self.BLOCK)
        # Placement
        placement_kernel[grid_counts](flat, out_ranks, offsets, N, self.L, self.BLOCK)

        # 2) Triton histogram and inclusive scan for expert offsets
        counts_exp = torch.zeros(self.L, dtype=torch.int32, device=device)
        grid_hist = (triton.cdiv(N, self.BLOCK),)
        histogram_kernel[grid_hist](flat, counts_exp, N, self.L, self.BLOCK)
        offsets_exp = torch.empty(self.L + 1, dtype=torch.int32, device=device)
        inclusive_scan_kernel[(1,)](counts_exp, offsets_exp, self.L)

        # Return as in original (sorted_token_indices, expert_offsets)
        return out_perm, offsets_exp

def add_kernel(
