import triton
import triton.language as tl


@triton.jit
def compute_cumsum_and_finalize_offsets_kernel(
    counts_ptr,        # *int32, length NUM_EXPERTS
    offsets_ptr,       # *int32, length (NUM_EXPERTS + 1)
    NUM_EXPERTS: tl.constexpr
):
    # Compute inclusive prefix sums into offsets[:NUM_EXPERTS]
    running = tl.zeros((), dtype=tl.int32)
    for e in tl.static_range(0, NUM_EXPERTS):
        cnt = tl.load(counts_ptr + e)
        running += cnt
        tl.store(offsets_ptr + e, running)
    # Final element is total count + 1 (host passes total_count separately and handles +1)
    # NOTE: The host already set offsets[NUM_EXPERTS] = total_count + 1 before launching this kernel.
    # This kernel only writes the per-expert prefix sums. We could have the host set the last element,
    # but to keep it self-contained, we rely on host setting the last element before calling this kernel.
    pass

    out_ptr,           # *int32, length M
    M: tl.constexpr
):
