import triton
import triton.language as tl


@triton.jit
def histogram_kernel(original_ptr, counts_ptr, M: tl.int32, NUM_VALUES: tl.int32, BLOCK: tl.constexpr):
    """
    Compute histogram of values in original_ptr (int32) into counts_ptr (int32).
    Values must be in [0, NUM_VALUES-1].
    We iterate over the original array in tiles of BLOCK and atomic_add into counts.
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    # Load a tile of original values
    vals = tl.load(original_ptr + offsets, mask=mask, other=0)  # int32
    # For each element in the tile, atomic add to its count bin
    # We use masked loads so out-of-range offsets do nothing
    for j in range(BLOCK):
        idx = offsets[j]
        if mask[j]:
            v = vals[j]
            tl.atomic_add(counts_ptr + v, 1)


@triton.jit
def inclusive_cumsum_kernel(counts_ptr, prefix_ptr, N: tl.int32, BLOCK: tl.constexpr):
    """
    Compute inclusive prefix sum of counts_ptr (int32) into prefix_ptr (int32).
    prefix[i] = sum_{k=0..i} counts[k], for i in 0..N-1.
    """
    pid = tl.program_id(0)
    # Each program handles a block of elements
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask = offsets < N

    # Compute local inclusive sum for the block
    local_sum = tl.zeros((), dtype=tl.int32)
    # We write the prefix for each i as the sum of all previous counts up to i
    for i in range(BLOCK):
        idx = start + i
        if mask[i]:
            # current count at idx
            c = tl.load(counts_ptr + idx)
            # accumulate local_sum
            local_sum += c
            # store inclusive sum at idx
            tl.store(prefix_ptr + idx, local_sum)
        # else: skip

        M = original.numel()

        NUM_VALUES = 256
