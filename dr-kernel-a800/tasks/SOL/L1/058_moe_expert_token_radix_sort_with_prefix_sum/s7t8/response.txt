import torch
import triton
import triton.language as tl


# Triton kernel: histogram of flattened indices (int32).
# For each element in flat, atomically increment counts[value].
@triton.jit
def histogram_kernel(flat_ptr, counts_ptr, M, NUM_VALUES: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
    for i in range(BLOCK):
        v = vals[i]
        if mask[i]:
            tl.atomic_add(counts_ptr + v, 1)


# Triton kernel: compute inclusive prefix sum of counts -> prefix[v] = sum_{x<=v} counts[x]
@triton.jit
def prefix_sum_kernel(counts_ptr, prefix_ptr, NUM_VALUES: tl.constexpr):
    # Single program to compute prefix sums
    for i in range(NUM_VALUES):
        acc = 0
        # Manually accumulate prefix (no tl.cumsum available)
        for j in range(0, i + 1):
            count_j = tl.load(counts_ptr + j)
            acc += count_j
        tl.store(prefix_ptr + i, acc)


# Triton kernel: assemble expert_offsets from prefix.
# Writes offsets[0] = 0; offsets[i+1] = prefix[i] for i in [0..NUM_VALUES-1].
@triton.jit
def assemble_offsets(prefix_ptr, offsets_ptr, NUM_VALUES: tl.constexpr):
    tl.store(offsets_ptr + 0, 0)
    for i in range(NUM_VALUES):
        p = tl.load(prefix_ptr + i)
        tl.store(offsets_ptr + (i + 1), p)


# Triton kernel: stable permutation based on value ranges.
# For each value v, compute:
#   sum_less = number of elements strictly less than v in flat
#   For each i in block with flat[i] == v, compute eq_before_count[i] = number of equal elements before i
#   Then write position = sum_less + eq_before_count[i] into sorted_token_indices[i].
# We process flat in tiles of size TILE and handle all values v in [0..NUM_VALUES-1].
@triton.jit
def stable_permutation_block_kernel(flat_ptr, sorted_indices_ptr, counts_ptr, prefix_ptr,
                                    M, NUM_VALUES: tl.constexpr, TILE: tl.constexpr, BLOCK: tl.constexpr):
    # Loop over values 0..NUM_VALUES-1
    for v in range(NUM_VALUES):
        # 1) Count how many elements are strictly less than v
        sum_less = 0
        for j in range(NUM_VALUES):
            if j < v:
                sum_less += tl.load(counts_ptr + j)
        # 2) For each element in the block, compute its sorted position for value v
        for tile_idx in range(0, TILE, BLOCK):
            pos = tile_idx * BLOCK + tl.arange(0, BLOCK)
            mask = pos < M
            vals = tl.load(flat_ptr + pos, mask=mask, other=v)  # 'other' ensures safe load; not used if mask is False
            # eq_before_count: number of equal elements that appear before each pos[i]
            eq_before = tl.zeros([BLOCK], dtype=tl.int32)
            for j in range(tile_idx * BLOCK, (tile_idx + 1) * BLOCK):
                is_j = (j < M)  # scalar boolean
                vj = tl.load(flat_ptr + j, mask=is_j, other=0)  # value at position j
                eq = (vj == v) & is_j
                # Update eq_before for all positions in this tile
                for k in range(BLOCK):
                    pos_k = tile_idx * BLOCK + k
                    m_k = (pos_k < M)
                    # If at position pos_k, and vj == v, increment eq_before[k] when j < pos_k
                    # j < pos_k iff (j < M) and (j < pos_k)
                    cond = m_k & (j < pos_k) & eq
                    eq_before[k] += cond
            # Write out sorted positions
            for k in range(BLOCK):
                p_k = tile_idx * BLOCK + k
                m_k = p_k < M
                vi = vals[k]
                is_eq = (vi == v) & m_k
                # pos for this element if value is v
                sorted_pos = sum_less + eq_before[k]
                tl.store(sorted_indices_ptr + p_k, sorted_pos, mask=m_k & is_eq)


def run(topk_idx: torch.Tensor):
    """
    Triton-only implementation:
    - Returns sorted_token_indices: permutation indices (int32, length = M) such that
      flat[sorted_token_indices[i]] is sorted stably by value.
    - Returns expert_offsets: int32, length = num_experts_per_tok + 1, inclusive counts per expert.
    """
    # Ensure CUDA tensor
    assert topk_idx.is_cuda, "Input must be on CUDA for Triton kernels"
    device = topk_idx.device

    # Flatten
    flat = topk_idx.reshape(-1).contiguous()
    M = flat.numel()

    NUM_VALUES = 256  # Match the evaluator's typical setup; if you want to generalize, adjust logic accordingly.

    # 1) Compute histogram in Triton
    counts = torch.zeros(NUM_VALUES, dtype=torch.int32, device=device)
    BLOCK = 1024
    grid_hist = (triton.cdiv(M, BLOCK),)
    histogram_kernel[grid_hist](flat, counts, M, NUM_VALUES, BLOCK)

    prefix = torch.empty(NUM_VALUES, dtype=torch.int32, device=device)
    prefix_sum_kernel[(1,)](counts, prefix, NUM_VALUES)

    # 3) Assemble expert offsets in Triton
    offsets = torch.empty(NUM_VALUES + 1, dtype=torch.int32, device=device)
