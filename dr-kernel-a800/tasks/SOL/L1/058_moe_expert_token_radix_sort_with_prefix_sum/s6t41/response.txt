import torch
import triton
import triton.language as tl


# Triton kernel: Histogram of values in orig (int32) into counts (int32).
# We assume values are in [0, L-1], where L is num_experts (e.g., 256).
@triton.jit
def histogram_kernel(orig_ptr, counts_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    # Each program handles a block of elements and atomically increments
    # the counts for the corresponding values. This is O(N) with atomics.
    lane = tl.program_id(0)
    offsets = lane * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    vals = tl.load(orig_ptr + offsets, mask=mask, other=0)  # int32
    # Atomic add to counts[vals[i]] for valid lanes
    tl.atomic_add(counts_ptr + vals, 1, mask=mask)


# Triton kernel: Exclusive prefix sum across counts to produce offsets per value.
# out_ptr[e] = inclusive sum of counts for ids < e. out_ptr[L] = total N.
# We loop over e = 0..L-1 and maintain a running sum. This avoids torch.cumsum.
@triton.jit
def exclusive_scan_kernel(counts_ptr, out_ptr, L: tl.constexpr):
    # Single program performing exclusive scan across L values.
    running = 0
    for e in range(L):
        # counts_ptr[e] is int32 scalar
        c = tl.load(counts_ptr + e)
        out_ptr[e] = running
        running += c
    # total elements
    out_ptr[L] = running


# Triton kernel: Odd-even transposition sort on x (int32) of length N with stable tie order by original index.
# We store the sorted permutation into perm (int32) of length N. This is not O(N log N), but simple and correct.
@triton.jit
def odd_even_sort_kernel(x_ptr, perm_ptr, N: tl.constexpr):
    # Each element i performs compare-and-swap with its neighbor based on phase.
    # We use stable tie order by original index: i < j.
    for phase in range(N):  # phases: 0,1,2,...
        # Even phase: pairs (0,1), (2,3), ...
        # Odd phase:  pairs (1,2), (3,4), ...
        # For each i, only do work if i is start of a pair in this phase.
        # We process all i and use masks to avoid overlapping writes.
        for i in range(N):
            if (phase % 2 == 0):
                j = i + 1
                do_pair = ((i % 2) == 0) & (j < N)
            else:
                j = i + 1
                do_pair = ((i % 2) == 1) & (j < N)

            # If not doing this pair, skip
            if not do_pair:
                continue

            vi = tl.load(x_ptr + i)
            vj = tl.load(x_ptr + j)

            # Stable compare: ascending by value; tie-break by original index (i < j)
            swap = (vj < vi) | ((vj == vi) & (j < i))

            # Compute min and max values with stable order
            minv = tl.where(swap, vj, vi)
            maxv = tl.where(swap, vi, vj)

            # Write back to x_ptr positions i and j
            tl.store(x_ptr + i, minv)
            tl.store(x_ptr + j, maxv)

            # Also update perm_ptr to keep track of sorted indices
            pi = tl.load(perm_ptr + i)
            pj = tl.load(perm_ptr + j)
            # Stable swap of indices based on the same swap condition
            pmin = tl.where(swap, pj, pi)
            pmax = tl.where(swap, pi, pj)
            tl.store(perm_ptr + i, pmin)
            tl.store(perm_ptr + j, pmax)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # We keep no parameters; all work is on the device via Triton kernels.

    def forward(self, topk_idx: torch.Tensor):
        # Ensure dtype and shape are correct
        assert topk_idx.dtype == torch.int32, "topk_idx must be int32"
        device = topk_idx.device
        # Flatten to 1D
        flat = topk_idx.reshape(-1)
        N = flat.numel()
        num_experts = 256  # From the original context

        # 1) Compute sorted_token_indices via Triton odd-even sort
        # We sort the values themselves and return the permutation of indices [0..N-1].
        # Create a copy of flat to sort (we do not modify the original)
        x = flat.clone()
        perm = torch.arange(N, dtype=torch.int32, device=device)

        # Launch odd-even sort kernel: grid of 1 since N is passed as constexpr in Triton requires dynamic handling, but we set N for loop bounds. Triton allows runtime N with for loops; the kernel will run as long as N is defined. We use a grid of 1 and rely on loop over N.
        # Note: In Triton, loops must be based on runtime values, and we cannot use tl.constexpr for N. To make this work, we treat N as runtime and run the loops; however Triton requires compile-time for certain constructs. Here we keep it as simple: we run the loops with N, grid size is 1, and the kernel operates on the entire vector.
        # Since Triton does not allow dynamic grids bound to N inside kernel, we will implement a host-side loop. But the requirement is to use Triton kernels in forward. To keep it Triton-only, we can instead implement a single-program kernel by setting grid=(1,). The odd-even sort will perform enough iterations to sort N elements.
        # We will run the kernel with grid=(1,). For larger N, this may not be fastest, but correctness is the priority.

        odd_even_sort_kernel[(1,)](x, perm, N=N)

        sorted_token_indices = perm  # Shape: [N], int32

        # 2) Compute expert_offsets via Triton histogram + exclusive scan
        counts = torch.zeros(num_experts, dtype=torch.int32, device=device)
        # Choose a reasonable BLOCK for histogram; 1024 works well.
        BLOCK_HIST = 1024
        grid_hist = (triton.cdiv(N, BLOCK_HIST),)
        histogram_kernel[grid_hist](flat, counts, N, L=num_experts, BLOCK=BLOCK_HIST)

        offsets = torch.empty(num_experts + 1, dtype=torch.int32, device=device)
        # Exclusive scan over counts to produce offsets[e] = sum of counts for ids < e
        # grid=(1,) since we do a single program scan over num_experts (256)
        exclusive_scan_kernel[(1,)](counts, offsets, L=num_experts)

        return sorted_token_indices, offsets
