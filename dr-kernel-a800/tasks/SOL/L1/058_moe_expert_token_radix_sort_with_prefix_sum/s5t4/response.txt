import torch
import triton
import triton.language as tl


# Kernel 1: Compute per-expert counts (histogram) using one program per key.
@triton.jit
def _histogram_counts_kernel(flat_ptr, counts_ptr, M: tl.int32, NUM_EXPERTS: tl.constexpr):
    """
    For each key k in [0, NUM_EXPERTS), counts_ptr[k] = number of elements in flat_ptr equal to k.
    Launch with grid=(NUM_EXPERTS,)
    """
    k = tl.program_id(axis=0)
    cnt = tl.zeros((), dtype=tl.int32)
    for j in range(0, M):
        val = tl.load(flat_ptr + j)
        if val == k:
            cnt += 1
    tl.store(counts_ptr + k, cnt)


# Kernel 2: Stable argsort permutation using a bitonic sorting network on indices [0..M-1].
# We implement compare-exchange for pairs (i, j) in each bitonic stage. This kernel is called
# multiple times (total iterations = num_stages), each time updating out_perm and sorted_vals.
@triton.jit
def _stable_argsort_perm_kernel(flat_ptr, out_perm_ptr, sorted_vals_ptr, M: tl.int32, NUM_STAGES: tl.constexpr):
    """
    Each iteration performs one compare-exchange step of bitonic sort on the permutation out_perm_ptr.
    We read flat values by indexing into flat_ptr using current out_perm_ptr entries.
    We also update sorted_vals_ptr to keep the flattened values corresponding to out_perm_ptr.
    """
    i = tl.program_id(axis=0)  # which index i we process
    # For each iteration, compute partner j and direction
    for stage in range(0, NUM_STAGES):
        k = 1 << (stage + 1)  # size of current bitonic sequence
        j = i ^ k              # partner index
        up = (i & k) == 0      # ascending direction for this subsequence
        # Ensure we only process each pair once (avoid double-processing)
        if j > i:
            # Load current pair values
            idx_i = tl.load(out_perm_ptr + i)
            idx_j = tl.load(out_perm_ptr + j)
            val_i = tl.load(flat_ptr + idx_i)
            val_j = tl.load(flat_ptr + idx_j)
            # Stable compare: primary by value, then by index
            should_swap = (up and val_i > val_j) or ((not up) and val_i < val_j)
            tie_by_idx = (not up) and (val_i == val_j) and (idx_i > idx_j)
            swap = should_swap or tie_by_idx
            # Swap if needed
            new_i = idx_j if swap else idx_i
            new_j = idx_i if swap else idx_j
            # Write updated pair
            tl.store(out_perm_ptr + i, new_i)
            tl.store(out_perm_ptr + j, new_j)
            # Update sorted_vals_ptr accordingly: swap values in sorted_vals_ptr[i], sorted_vals_ptr[j]
            val_i_slot = tl.load(sorted_vals_ptr + i)
            val_j_slot = tl.load(sorted_vals_ptr + j)
            new_val_i = val_j_slot if swap else val_i_slot
            new_val_j = val_i_slot if swap else val_j_slot
