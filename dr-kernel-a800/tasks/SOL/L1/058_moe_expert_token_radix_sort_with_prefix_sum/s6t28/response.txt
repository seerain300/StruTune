```python
import torch
import triton
import triton.language as tl


@triton.jit
def histogram_kernel(orig_ptr, counts_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    """
    Counts occurrences of each value in orig_ptr (int32) into counts_ptr (int32),
    assuming values are in [0, L-1], with L=256.
    """
    lane = tl.program_id(0)
    offsets = lane * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    vals = tl.load(orig_ptr + offsets, mask=mask, other=0)
    # For valid lanes, count occurrences for each v in 0..L-1
    for v in range(L):
        eq = vals == v
        # Convert boolean mask to int and sum within this block; since masked lanes have other=0, eq is False there.
        increment = tl.where(mask & eq, 1, 0)
        # Atomic add to global counts_ptr[v]
        tl.atomic_add(counts_ptr + v, tl.sum(increment))


@triton.jit
def exclusive_scan_kernel(counts_ptr, offsets_ptr, num_exps: tl.constexpr, BLOCK: tl.constexpr):
    """
    Compute exclusive prefix sums across counts_ptr[0:num_exps] into offsets_ptr[0:num_exps].
    offsets[num_exps] is set to N (total elements).
    """
    running = 0
    N = tl.load(offsets_ptr + num_exps)  # we will write N here after loop, but keep a local variable
    # We don't need N for exclusive scan; we compute offsets[0:num_exps].
    for i in range(num_exps):
        count_i = tl.load(counts_ptr + i)
        running += count_i
        tl.store(offsets_ptr + i, running)
    # Write total N to the last entry
    tl.store(offsets_ptr + num_exps, N)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        """
        Compute expert_offsets using Triton kernels only. No torch operations in forward.

        Args:
            topk_idx: int32 tensor of shape (batch_size, seq_len, num_experts_per_tok)
        Returns:
            expert_offsets: int32 tensor of shape (256 + 1,)
        """
        # Ensure dtype is int32 and contiguous
        flat = topk_idx.to(torch.int32).reshape(-1).contiguous()
        N = flat.numel()
        L = 256  # num_experts

        # Allocate counts and offsets
        counts = torch.zeros(L, dtype=torch.int32, device=flat.device)
        offsets = torch.empty(L + 1, dtype=torch.int32, device=flat.device)

        # Launch histogram kernel
        # Grid size: number of programs = ceil_div(N, BLOCK)
        BLOCK = 1024
        grid = (triton.cdiv(N, BLOCK),)
        histogram_kernel[grid](flat, counts, N, L=L, BLOCK=BLOCK)

        # Launch exclusive scan kernel: we need to provide N for the last offset
        # We can set offsets[256] to N outside by writing after scan; but the kernel stores N at the end.
        # For simplicity, we perform a second step to set offsets[256] = N using a small kernel or torch.
        # However, to keep Triton-only, we can compute total N and set it via a tiny Triton kernel.
        # Here we compute N using flat (it's already in flat); we just pass N to the kernel via a scalar arg.
        # We'll set N in a separate tiny kernel to offsets[256].
        total_N = N
        # Write total N to offsets[256] using a tiny Triton kernel
        # One program is enough
        # offset writer kernel: writes a single int32 to a pointer at index 256
        # Note: We need to pass the address of offsets[256]. We can construct a simple 1-element buffer or write directly.
        # To avoid extra torch, we can simply do: offsets[256] = total_N via a small kernel that writes to a given pointer.
        # Triton can't index tensors in Python like offsets_ptr[256], so we use a dedicated kernel.

        # Define a tiny kernel that writes total_N to a provided pointer (offsets_ptr + 256)
        @triton.jit
        def write_last_offset_kernel(ptr, value):
            tl.store(ptr, value)

        # Launch it: pointer to offsets[256] is offsets_ptr + 256
        write_last_offset_kernel[(1,)](offsets + 256, total_N)

        # Now run exclusive scan to fill offsets[0:256]
        exclusive_scan_kernel[(1,)](counts, offsets, num_exps=L, BLOCK=32)  # num_exps is constexpr

        # Return expert_offsets
        return offsets
```
