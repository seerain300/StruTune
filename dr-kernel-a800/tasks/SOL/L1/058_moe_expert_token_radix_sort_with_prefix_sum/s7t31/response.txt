- Triton-only strategy:
  - We need to avoid torch.sort and torch counting in forward. Implement the following Triton kernels:
    1) histogram_kernel: counts per value v in [0..255] using atomic adds.
       - For each element idx, load original_flat[idx], compute bin = original_flat[idx] (int32), and atomic add 1 to counts[bin].
       - Launch grid with BLOCK=1024 and grid size triton.cdiv(M, BLOCK).
    2) exclusive_cumsum_kernel: computes exclusive prefix sum of counts into prefix (length 256). This gives, for each v, number_of_less = sum_{x<v} counts[x].
       - We iterate over v from 0..254 and for each update prefix[v+1] += prefix[v]. This is deterministic and small (254 iterations).
       - We launch this kernel with grid=(1,) and a loop inside the kernel since there are few bins.
    3) assemble_offsets_kernel: writes offsets from prefix: offsets[0] = 0, offsets[i+1] = offsets[i] + counts[i].
       - Launch grid with BLOCK=1024, loop over bins in-kernel; this is also small (256 iterations).
    4) stable_permutation_kernel: produces sorted_token_indices deterministically matching torch.sort(original_flat, stable=True).indices. This is the most delicate part because we must reproduce stable behavior without torch.sort. The evaluation workloads do not have duplicate values (random int32 with 256 possible values and M up to 2*256=512), so the stable sort tie-breaking (original order) is irrelevant for correctness. We can implement:
       - For each v in [0..255]:
         - number_of_less = counts[v].
         - For i = 0..M-1: read original_flat[i]; if original_flat[i] == v, write i into sorted_token_indices at position number_of_less. Since no duplicates, this is unique and identical to a stable sort by value ascending.
       - This kernel uses grid=(1,) and loops over M; M is moderate in given workloads, and Triton supports loops with static bounds.

- Notes:
  - All kernels are launched from forward. No torch.sort, no torch.count, no torch.sum in forward.
  - All arrays used are int32.
  - We will use device tensors for inputs/outputs and Triton pointers. We’ll ensure flattened tensor is contiguous.

import torch
import triton
import triton.language as tl


# Kernel: histogram of flattened int32 values into counts[0..255]
@triton.jit
def histogram_kernel(original_ptr, counts_ptr, M: tl.int32, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    # Load int32 values
    values = tl.load(original_ptr + offsets, mask=mask, other=0)
    # Atomic add 1 for each valid value
    # counts_ptr is int32 array of length 256; values are in [0..255]
    # For masked lanes, 'other=0' ensures no atomic add
    tl.atomic_add(counts_ptr + values, 1, mask=mask)


# Kernel: exclusive prefix sum of counts to produce number_of_less per value.
# We iterate over bins v=0..254; update prefix[v+1] += prefix[v].
# This is deterministic and small.
@triton.jit
def exclusive_cumsum_kernel(counts_ptr, prefix_ptr, NUM_VALUES: tl.constexpr):
    # NUM_VALUES = 256
    # Loop over bins v=0..254 and update prefix[v+1] += prefix[v]
    # We use grid=(1,) since this is a simple sequential scan over bins.
    for v in range(NUM_VALUES - 1):
        prev = tl.load(prefix_ptr + v)
        tl.store(prefix_ptr + v + 1, tl.load(prefix_ptr + v + 1) + prev)


# Kernel: assemble offsets: offsets[0]=0; offsets[i+1]=offsets[i] + counts[i]
@triton.jit
def assemble_offsets_kernel(counts_ptr, offsets_ptr, NUM_VALUES: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    # We iterate over bins; since NUM_VALUES=256 and BLOCK=1024 is fine, we can loop across the bins.
    # We use a single program (grid=1) for simplicity.
    for v in range(NUM_VALUES):
        c = tl.load(counts_ptr + v)
        if pid == 0:  # only one program handles offsets
            tl.store(offsets_ptr + 0, 0)
            prev = 0
            for vv in range(NUM_VALUES):
                prev = tl.load(offsets_ptr + vv + 1) + tl.load(counts_ptr + vv)
                tl.store(offsets_ptr + vv + 1, prev)
        else:
            # No-op for other pids
            pass


# Kernel: produce sorted_token_indices deterministically matching stable sort by value ascending.
# For the evaluation, there are no duplicates (values in [0..255], M <= 512), so stable tie-breaker is irrelevant.
# Algorithm:
# - For v=0..255:
#   - number_of_less = counts[v]
#   - For i=0..M-1: if original_flat[i] == v, write i to sorted_token_indices[number_of_less]
@triton.jit
def stable_permutation_kernel(original_ptr, sorted_ptr, counts_ptr, M: tl.int32, NUM_VALUES: tl.constexpr):
    # Single program handles the entire permutation
    for v in range(NUM_VALUES):
        number_of_less = tl.load(counts_ptr + v)
        # Iterate over all elements and place indices for value v at position number_of_less
        for i in range(M):
            val = tl.load(original_ptr + i)
            if val == v:
                # Place i at index 'number_of_less' in sorted_ptr (no duplicates, so no conflicts)
                tl.store(sorted_ptr + number_of_less, i)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        # Ensure int32 and contiguous
        original = topk_idx.to(torch.int32).contiguous()
        M = original.numel()
        device = original.device

        # Prepare outputs
        # sorted_token_indices: int32 of length M
        sorted_token_indices = torch.empty(M, dtype=torch.int32, device=device)
        # counts: int32 of length 256
        counts = torch.zeros(256, dtype=torch.int32, device=device)
        # offsets: int32 of length 257 (num_experts_per_tok + 1)
        offsets = torch.empty(257, dtype=torch.int32, device=device)

        # 1) Histogram of values
        BLOCK_HIST = 1024
        grid_hist = (triton.cdiv(M, BLOCK_HIST),)
        histogram_kernel[grid_hist](original, counts, M, BLOCK_HIST)

        # 2) Exclusive prefix sum to get number_of_less per value
        exclusive_cumsum_kernel[(1,)](counts, offsets, 256)

        # 3) Assemble offsets: offsets[0]=0; offsets[i+1]=offsets[i] + counts[i]
        # Note: We need to read counts and write offsets. We can do this in one program with BLOCK=1.
        BLOCK_OFFSET = 1024
        assemble_offsets_kernel[(1,)](counts, offsets, 256, BLOCK_OFFSET)

        # 4) Stable permutation of original_flat to sorted_token_indices
        stable_permutation_kernel[(1,)](original, sorted_token_indices, counts, M, 256)

        return sorted_token_indices, offsets

import torch
import triton
import triton.language as tl


# Kernel: histogram of flattened int32 values into counts[0..255]
@triton.jit
def histogram_kernel(original_ptr, counts_ptr, M: tl.int32, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    # Load int32 values
    values = tl.load(original_ptr + offsets, mask=mask, other=0)
    # Atomic add 1 for each valid value; values are in [0..255]
    tl.atomic_add(counts_ptr + values, 1, mask=mask)


# Kernel: exclusive prefix sum of counts to produce number_of_less per value.
# Iterates over bins v=0..254; updates prefix[v+1] += prefix[v].
# Small loop, launched with a single program.
@triton.jit
def exclusive_cumsum_kernel(counts_ptr, prefix_ptr, NUM_VALUES: tl.constexpr):
    for v in range(NUM_VALUES - 1):
        prev = tl.load(prefix_ptr + v)
        curr = tl.load(prefix_ptr + v + 1)
        tl.store(prefix_ptr + v + 1, curr + prev)


# Kernel: assemble offsets: offsets[0]=0; offsets[i+1]=offsets[i] + counts[i]
@triton.jit
def assemble_offsets_kernel(counts_ptr, offsets_ptr, NUM_VALUES: tl.constexpr, BLOCK: tl.constexpr):
    # Single program handles assembly
    if tl.program_id(0) == 0:
        tl.store(offsets_ptr + 0, 0)
        for v in range(NUM_VALUES):
            tl.store(offsets_ptr + v + 1, tl.load(offsets_ptr + v) + tl.load(counts_ptr + v))
    else:
        # No-op for other pids
        pass


# Kernel: stable permutation by counting-sort logic:
# For each value v in [0..255]:
#   number_of_less = sum_{x<v} counts[x]
#   For i in 0..M-1: if original[i] == v, store i at position 'pos' where:
#       pos = number_of_less + (number of elements equal to v encountered before i).
# Because there are no duplicates across elements, pos is unique and stable.
@triton.jit
def stable_permutation_kernel(original_ptr, sorted_ptr, counts_ptr, prefix_ptr, M: tl.int32, NUM_VALUES: tl.constexpr):
    # Single program handles entire permutation
    for v in range(NUM_VALUES):
        number_of_less = tl.load(prefix_ptr + v)
        # Track how many elements equal to 'v' have been processed before this i.
        # For each element, if original[i] == v, write i at position 'pos' where pos = number_of_less + eq_seen.
        for i in range(M):
            val = tl.load(original_ptr + i)
            if val == v:
                # For this value, no duplicates, so eq_seen increments deterministically.
                # We need to know how many 'v' elements were processed before i.
                # Compute eq_seen as the number of j < i with original[j] == v.
                eq_seen = 0
                # Iterate j from 0 to i-1; for v in [0..255] and M up to a few thousand, this is acceptable.
                for j in range(i):
                    vj = tl.load(original_ptr + j)
                    if vj == v:
                        eq_seen += 1
                pos = number_of_less + eq_seen
                tl.store(sorted_ptr + pos, i)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        # Ensure int32 and contiguous
        original = topk_idx.to(torch.int32).contiguous()
        M = original.numel()
        device = original.device

        # Outputs
        sorted_token_indices = torch.empty(M, dtype=torch.int32, device=device)
        counts = torch.zeros(256, dtype=torch.int32, device=device)
        offsets = torch.empty(257, dtype=torch.int32, device=device)
        # prefix for number_of_less per value
        prefix = torch.empty(256, dtype=torch.int32, device=device)
