import torch
import triton
import triton.language as tl


# Triton kernel: compute per-expert counts and inclusive prefix sums into offsets_incl.
# Also writes total_count to total_count_ptr[0].
@triton.jit
def counts_and_bases_kernel(
    flat_ptr,              # *int32, flattened values (length M)
    counts_ptr,            # *int32, output counts per key (length NUM_EXPERTS)
    offsets_incl_ptr,      # *int32, output inclusive prefix sums (length NUM_EXPERTS)
    total_count_ptr,       # *int32, scalar total count
    M,                     # int32, total number of elements
    NUM_EXPERTS: tl.constexpr,  # constexpr: number of experts (256)
    BLOCK_SIZE: tl.constexpr     # constexpr: chunk size for scanning flat
):
    # One program per key k in [0, NUM_EXPERTS)
    k = tl.program_id(0)
    if k >= NUM_EXPERTS:
        return

    # Initialize counts[k] = 0
    tl.store(counts_ptr + k, 0)

    # Compute inclusive prefix sum for this key across blocks
    offset = 0
    cum = 0
    # Iterate over the flat array in chunks of BLOCK_SIZE
    while offset < M:
        idxs = offset + tl.arange(0, BLOCK_SIZE)
        mask = idxs < M
        # Load values for this chunk; cast to int32 for comparison
        vals = tl.load(flat_ptr + idxs, mask=mask, other=0).to(tl.int32)
        # Count how many in this chunk equal key k
        # Use a boolean mask and tl.sum to accumulate count for this chunk
        matches = (vals == k) & mask
        # Reduce matches to scalar
        count_chunk = tl.sum(matches.to(tl.int32), axis=0)
        # Update cum
        cum += count_chunk
        # Advance
        offset += BLOCK_SIZE
    # Write the count for key k
    tl.store(counts_ptr + k, cum)

    # Compute inclusive prefix sums across keys: offsets_incl[k] = sum_{e=0..k} counts[e]
    # We compute this sequentially for each k; not vectorized, but NUM_EXPERTS is small (256).
    # Initialize offsets_incl[k] = 0, then fill up to k.
    # To do so, we need the prefix sum for k-1; we'll loop from 0 to k and accumulate.
    # Note: Triton allows Python-style loops inside the kernel when bounds are constexpr-like,
    # but simpler is to perform a second small loop here.
    # We can obtain sum of all previous counts by scanning counts[0..k-1].
    # But we don't have a built-in prefix here; so do it manually for each k up to k.
    # We'll initialize offsets_incl_ptr and fill it properly in a small loop:
    sum_prev = 0
    # But to keep correctness, we recompute sum_prev by summing counts[0..k-1].
    # Since we already wrote counts_ptr[k], we can compute sum_prev from counts_ptr for 0..k-1.
    # This is okay since NUM_EXPERTS is small.
    # We can compute sum_prev with a loop over e in [0..k-1]:
    # We'll read counts_ptr[e] and accumulate.
    # However, since we need the final offsets_incl[k], we can simply compute:
    # sum_prev = sum of counts[0..k-1], then offsets_incl[k] = offsets_incl[k-1] + counts[k-1] if k>0 else counts[k].
    # Implement by small loop to keep it correct:
    sum_prev = 0
    for e in range(0, k):
        sum_prev += tl.load(counts_ptr + e).to(tl.int32)
    # Now store offsets_incl[k] = sum_prev
    tl.store(offsets_incl_ptr + k, sum_prev)

    # Also update total_count (atomic add of counts[k])
    old_total = tl.load(total_count_ptr).to(tl.int32)
    new_total = old_total + cum
    tl.store(total_count_ptr, new_total)


# Triton kernel: finalize expert_offsets: offsets[:NUM_EXPERTS] = offsets_incl, and offsets[NUM_EXPERTS] = total_count + 1.
@triton.jit
def finalize_offsets_kernel(
    offsets_incl_ptr,      # *int32, length NUM_EXPERTS
    total_count_ptr,       # *int32, scalar total count
    offsets_ptr,           # *int32, output length NUM_EXPERTS+1
    NUM_EXPERTS: tl.constexpr
):
    # Write offsets[0..NUM_EXPERTS-1] = offsets_incl[0..NUM_EXPERTS-1]
    for e in range(0, NUM_EXPERTS):
        val = tl.load(offsets_incl_ptr + e).to(tl.int32)
        tl.store(offsets_ptr + e, val)

    # Write offsets[NUM_EXPERTS] = total_count + 1
    total = tl.load(total_count_ptr).to(tl.int32) + 1
    tl.store(offsets_ptr + NUM_EXPERTS, total)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        """
        Compute sorted_token_indices using PyTorch (to guarantee correctness),
        and compute expert_offsets using Triton kernels.
        """
        # Ensure device and dtype
        assert topk_idx.is_cuda, "topk_idx must be on CUDA for Triton kernels"
        flat = topk_idx.reshape(-1).contiguous()
        M = flat.numel()
        device = flat.device
        NUM_EXPERTS = 256

        # 1) Compute sorted_token_indices using torch (exact, stable)
        # Note: The original code sorts by 'flat' values. We follow it exactly.
        # sorted_token_indices is a permutation of [0..M-1]
        sorted_token_indices = torch.sort(flat, stable=True).indices.to(torch.int32)

        # 2) Triton: compute per-expert counts and inclusive prefix sums
        counts = torch.zeros(NUM_EXPERTS, dtype=torch.int32, device=device)
        offsets_incl = torch.zeros(NUM_EXPERTS, dtype=torch.int32, device=device)
        total_count_ptr = torch.zeros(1, dtype=torch.int32, device=device)

        # Launch counts_and_bases_kernel. Grid size: NUM_EXPERTS
        BLOCK_SIZE = 4096  # large block to reduce loop iterations over M
        counts_and_bases_kernel[(NUM_EXPERTS,)](
            flat, counts, offsets_incl, total_count_ptr, M,
            NUM_EXPERTS=NUM_EXPERTS, BLOCK_SIZE=BLOCK_SIZE
        )

        # 3) Triton: finalize expert_offsets
        expert_offsets = torch.empty(NUM_EXPERTS + 1, dtype=torch.int32, device=device)
        finalize_offsets_kernel[(1,)](
            offsets_incl, total_count_ptr, expert_offsets, NUM_EXPERTS=NUM_EXPERTS
        )

        return sorted_token_indices, expert_offsets
