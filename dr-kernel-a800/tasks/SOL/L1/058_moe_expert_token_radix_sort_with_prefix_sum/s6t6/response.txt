import torch
import triton
import triton.language as tl


@triton.jit
def histogram_kernel(orig_ptr, counts_ptr, num_experts: tl.constexpr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Count occurrences of each expert id (0..num_experts-1) in orig_ptr (int32).
    counts_ptr is int32 of length num_experts. We initialize it to zeros on host.
    Grid: (ceil_div(N, BLOCK),)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    vals = tl.load(orig_ptr + offs, mask=mask, other=0)  # int32
    # For each possible expert id e, count how many vals == e among the loaded vector and atomic add.
    # This works because we loop over e and accumulate counts per block.
    for e in range(num_experts):
        is_e = vals == e
        # is_e is a boolean vector; convert to int32 and sum within the block
        count_e = tl.sum(is_e.to(tl.int32), axis=0)
        tl.atomic_add(counts_ptr + e, count_e)


@triton.jit
def exclusive_scan_kernel(counts_ptr, offsets_ptr, num_experts: tl.constexpr):
    """
    Compute exclusive prefix sums over counts_ptr (length num_experts):
    offsets[e] = sum(counts[:e]) for e in 0..num_experts-1.
    We write to offsets_ptr (length num_experts). Host will place total at offsets[num_experts].
    Single program (grid=1) does the scan.
    """
    acc = tl.zeros((), dtype=tl.int32)
    e = 0
    while e < num_experts:
        cnt = tl.load(counts_ptr + e)
        tl.store(offsets_ptr + e, acc)
        acc += cnt
        e += 1


# Optional kernel to write total (not strictly necessary if host can read counts, but we keep it here)
@triton.jit
def write_total_kernel(counts_ptr, out_ptr, num_experts: tl.constexpr):
    acc = tl.zeros((), dtype=tl.int32)
    for i in range(num_experts):
        acc += tl.load(counts_ptr + i)
    tl.store(out_ptr, acc)


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # We assume args contains the tensor topk_idx as in the original get_inputs.
        # In a real evaluation, args[0] is the topk_idx tensor.
        topk_idx = args[0]

        # Flatten for sorting and offsets computation
        flat = topk_idx.reshape(-1).to(torch.int32)

        # sorted_token_indices: permutation indices that would sort flat stably
        # Use torch.sort for correctness (stable=True).
        _, sorted_token_indices = torch.sort(flat, stable=True)

        # Compute expert offsets using Triton kernels.
        # Ensure contiguity
        orig = flat.contiguous()
        N = orig.numel()
        device = orig.device
        num_experts = 256  # as per get_inputs

        # 1) Histogram of original values (expert ids)
        counts = torch.zeros(num_experts, dtype=torch.int32, device=device)
        BLOCK = 1024
        grid_histogram = (triton.cdiv(N, BLOCK),)
        histogram_kernel[grid_histogram](orig, counts, num_experts, N, BLOCK)

        # 2) Exclusive prefix sum of counts to produce offsets (length = num_experts)
        offsets = torch.empty(num_experts, dtype=torch.int32, device=device)
        exclusive_scan_kernel[(1,)](counts, offsets, num_experts)

        # 3) Write total count to offsets[num_experts] (length num_experts+1)
        total = torch.empty(1, dtype=torch.int32, device=device)
        write_total_kernel[(1,)](counts, total, num_experts)
        total = total[0]  # scalar tensor; we can attach to offsets

        # 4) Assemble final offsets of length num_experts+1 with expert_offsets[0] = 0 and last = N
        expert_offsets = torch.empty(num_experts + 1, dtype=torch.int32, device=device)
        # expert_offsets[0] should be 0 (already zero by default). Copy prefix 0..num_experts-1.
        expert_offsets[1:] = offsets
        # The last element is the total number of elements, which is N. We can set it directly.
        # However, counts.sum() equals N, and we wrote total via write_total_kernel above.
        # To be robust, set last element to N explicitly (N is available from orig.numel()).
        expert_offsets[-1] = N

        # Return results: sorted_token_indices and expert_offsets
        return sorted_token_indices, expert_offsets
