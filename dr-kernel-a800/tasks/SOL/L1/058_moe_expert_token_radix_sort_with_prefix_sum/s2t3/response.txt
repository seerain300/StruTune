import triton
import triton.language as tl


@triton.jit
def count_experts_kernel(vals_ptr, counts_ptr, N, num_experts: tl.constexpr):
    """
    Triton kernel: for each element in vals_ptr (int32), atomically increment counts_ptr[exp],
    where exp is the value at that position. Assumes vals_ptr holds expert indices in [0, num_experts).
    """

    BLOCK = 1024

    mask = offsets < N

def merge_experts(vals_ptr, ranks_ptr, sorted_ptr, N, num_experts: tl.constexpr):
    """
    Triton kernel to perform stable merge of sorted indices by ranks.
    Iteratively selects the minimum rank among remaining candidates and writes the
    corresponding original index into sorted_ptr at position t. This produces a stable
    order. ranks_ptr should contain ranks[i] (int32) and be initialized before the merge.
    """
    t = 0
    while t < N:
        min_rank = tl.full((), 0x7FFFFFFF, tl.int32)
        chosen = tl.zeros((), dtype=tl.int32)
        # Scan all i to find minimum rank
        i = 0
        while i < N:
            rank_i = tl.load(ranks_ptr + i)
            if rank_i < min_rank:
                min_rank = rank_i
                # tie-break: choose smallest i among equal ranks
                chosen = i
            i += 1

        tl.store(ranks_ptr + chosen, -1)
        # Write chosen index into sorted_ptr[t]
        tl.store(sorted_ptr + t, chosen)
        t += 1
