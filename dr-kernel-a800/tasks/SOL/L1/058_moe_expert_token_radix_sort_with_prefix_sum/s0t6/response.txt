import torch
import triton
import triton.language as tl


@triton.jit
def bitonic_sort_stable_int64(out_ptr, flat_ptr, N: tl.int32, LOGN: tl.int32):
    """
    Stable bitonic sort on flat_ptr (int64) producing sorted indices in out_ptr (int64).
    Uses a 2D grid: axis 0 covers elements (0..N-1), axis 1 covers stages (0..LOGN-1).
    """
    i = tl.program_id(axis=0)
    j = tl.program_id(axis=1)
    # Skip if i >= N
    if i >= N:
        return
    # For each stage j, partner = i ^ (1 << j)
    step = 1 << j
    partner = i ^ step
    # Only process each pair once
    do_pair = i < partner
    in_bounds = (i < N) & (partner < N) & do_pair
    # Load values
    a = tl.load(flat_ptr + i, mask=in_bounds, other=0)
    b = tl.load(flat_ptr + partner, mask=in_bounds, other=0)
    # Determine ascending/descending for this half of bitonic network
    asc = ((i & (1 << j)) == 0)
    # Stable tie-break: if equal, i < partner
    tie = a == b
    # Compare results
    a_lt_b = a < b
    a_gt_b = a > b
    # For ascending:
    #   if a < b: i=a, partner=b
    #   if a > b: i=b, partner=a
    #   if a == b: i=min(i, partner) by original index (stable)
    new_i_asc = tl.where(a_lt_b, a, b)
    new_p_asc = tl.where(a_lt_b, b, a)
    # For descending:
    #   if a < b: i=b, partner=a
    #   if a > b: i=a, partner=b
    #   if a == b: i=min(i, partner)
    new_i_desc = tl.where(a_gt_b, a, b)
    new_p_desc = tl.where(a_gt_b, b, a)
    # Select based on asc, with tie-break preferring i if equal
    # If tie, i < partner
    new_i_stage = tl.where(asc, new_i_asc, new_i_desc)
    new_p_stage = tl.where(asc, new_p_asc, new_p_desc)
    # Store results
    tl.store(out_ptr + i, new_i_stage, mask=in_bounds)
    tl.store(out_ptr + partner, new_p_stage, mask=in_bounds)


@triton.jit
def count_histogram_atomic_int64(offsets_ptr, flat_ptr, N: tl.int32, num_experts: tl.int32):
    """
    Histogram of flat_ptr (int64) into offsets_ptr (int64) where offsets[v+1] += 1.
    offsets_ptr length = num_experts + 1 (int64), offsets[0] unused.
    """
    pid = tl.program_id(axis=0)
    base = pid * 1024 + tl.arange(0, 1024)
    mask = base < N
    vals = tl.load(flat_ptr + base, mask=mask, other=0)
    # Atomic add 1 (int64) for each occurrence into offsets_ptr[val + 1]
    tl.atomic_add(offsets_ptr + (vals + 1), 1, mask=mask)


@triton.jit
def inclusive_scan_int32(out_ptr, inp_ptr, length: tl.int32):
    """
    Inclusive scan (prefix sum) on inp_ptr (int32) of length 'length' and write to out_ptr (int32).
    Implemented via iterative doubling scan in a single Triton program.
    """
    # Single program processes the scan. We assume 'length' is num_experts + 1.
    # Load the input array into registers
    idx = tl.arange(0, 1024)
    mask = idx < length
    data = tl.load(inp_ptr + idx, mask=mask, other=0)
    # Perform inclusive scan in-place in 'data'
    stride = 1
    while stride < length:
        j = idx ^ stride
        valid = (j < length) & (idx < length)
        tmp = tl.load(out_ptr + j, mask=valid, other=0)
        data = tl.where(idx >= j, data + tmp, data)
        tl.store(out_ptr + idx, data, mask=mask)
        stride *= 2
    # After loop, 'data' contains inclusive scan; write back
    tl.store(out_ptr + idx, data, mask=mask)

        - Computes expert_offsets (int32) via Triton histogram (atomic adds) and in-kernel inclusive scan.
        Returns:

            expert_offsets: torch.int32 tensor of shape (num_experts + 1,)
        """

        if not topk_idx.is_cuda:
            topk_idx = topk_idx.cuda()
        if topk_idx.dtype != torch.int32:
            topk_idx = topk_idx.to(torch.int32)

        # Flatten
        flat = topk_idx.reshape(-1)
        N = flat.numel()
        device = flat.device

        flat_i64 = flat.to(torch.int64)
        sorted_token_indices = torch.empty(N, dtype=torch.int64, device=device)
        LOGN = (N - 1).bit_length() if N > 0 else 0
        grid = (N, LOGN)
        bitonic_sort_stable_int64[grid](sorted_token_indices, flat_i64, N, LOGN)

        num_experts = 256
        counts_i64 = torch.zeros(num_experts + 1, dtype=torch.int64, device=device)
        BLOCK = 1024
        grid_count = (BLOCK,)
        count_histogram_atomic_int64[grid_count](counts_i64, flat_i64, N, num_experts)
