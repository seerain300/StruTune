    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
    # vals are assumed to be in [0, NUM_VALUES-1]; other mask ensures safe load for out-of-range offsets.
