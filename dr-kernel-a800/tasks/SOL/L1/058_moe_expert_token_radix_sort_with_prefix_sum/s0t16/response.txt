      Stability tie-break: if equal, lower index i comes first.
    """

    j = tl.program_id(axis=1)
    # Compute partner for this stage j
    # partner = i ^ (1 << j)
    partner = i ^ (1 << j)
    # Only process once per pair

    # Bounds mask

    # If out of bounds, exit early

    # Load current values at i and partner

    b = tl.load(flat_ptr + partner)
    # Load original indices (initialized to i for all positions, will be overwritten in final positions)

    idx_b = tl.load(out_idx_ptr + partner)

    cond = (a > b) if asc else (a < b)

    new_idx_b = idx_a if cond else idx_b

    # However, Triton requires each program to own its own index updates; so each program that owns i (i.e., this program) writes out_idx_ptr[i] = new_idx_a, and the partner program writes out_idx_ptr[partner] = new_idx_b.

    # But partner > i implies partner program writes for partner; here we don't write for i. Therefore, only pair programs (i < partner) write both.
    # Implement this logic: write only for pair programs for i position.
    if do_pair:
        tl.store(out_idx_ptr + i, new_idx_a)
