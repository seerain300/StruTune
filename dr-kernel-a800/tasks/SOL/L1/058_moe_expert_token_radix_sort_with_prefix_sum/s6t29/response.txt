```python
import torch
import triton
import triton.language as tl


# Kernel 1: Count occurrences of each value in flat (int32) into counts (int32).
# We assume values are in [0, 255] (num_experts=256). L is compile-time constant (256).
@triton.jit
def histogram_kernel(flat_ptr, counts_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # Load flat values (int32). Masked loads use other=0 to avoid contributing to counts.
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)

    # For each value v in [0, L-1], atomically add 1 for each occurrence where vals == v.
    for v in range(L):
        eq = vals == v
        increment = tl.where(mask & eq, 1, 0)  # int32 vector
        # Atomically add to counts[v]. counts_ptr[v] is a scalar memory location.
        tl.atomic_add(counts_ptr + v, tl.sum(increment))


# Kernel 2: Write stable ranks into sorted_token_indices at computed positions.
# For each original position j, rank = number of elements < flat[j] (stable).
# We do a two-phase: compute rank for each j, then write j at that rank in a second kernel.
# Here we compute ranks and store them in rank_ptr[j].
@triton.jit
def compute_rank_kernel(flat_ptr, counts_ptr, rank_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # Load flat values at positions offsets.
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)

    # Compute stable rank for each j: rank = sum_{v < vals[j]} counts[v]
    running = tl.zeros([BLOCK], dtype=tl.int32)
    # Unrolled loop over L values; add counts[v] if v < vals[j]
    for v in range(L):
        # counts[v] is scalar; broadcast to vector
        less = (vals > v) & mask  # stable: only increment when v < value, so ties keep original order
        increment = tl.where(less, 1, 0)
        running += increment

    # Store rank for each j
    tl.store(rank_ptr + offsets, running, mask=mask)


# Kernel 2b: Write original indices j into sorted_token_indices at their computed ranks.
@triton.jit
def write_permutation_kernel(rank_ptr, indices_ptr, sorted_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    ranks = tl.load(rank_ptr + offsets, mask=mask, other=0)
    # Load original indices 0..N-1
    j = offsets  # int32 vector
    # Store j at position ranks; ensure masked stores
    tl.store(sorted_ptr + ranks, j, mask=mask & (ranks < N))


# Kernel 3: Exclusive prefix sum of counts to produce offsets[e] = inclusive sum of counts for ids < e.
# We implement a simple loop scan per element e; grid=1, num_warps=1. counts_ptr length = L+1 where counts_ptr[L] unused.
@triton.jit
def exclusive_scan_kernel(counts_ptr, offsets_ptr, L: tl.constexpr):
    running = 0
    # Write offsets for e in [0..L-1]
    for e in range(L):
        # counts_ptr[e] is a scalar; read it
        cnt = tl.load(counts_ptr + e)
        running += cnt
        # Store inclusive sum before e: offsets[e] = sum_{k < e} counts[k]
        tl.store(offsets_ptr + e, running - cnt)  # running - cnt = sum_{k < e} counts[k]
    # The last element should be N; we leave it untouched by this kernel (host sets it).
    # However, to be explicit, set offsets[L] = N via host code after this kernel returns.


def run_triton(topk_idx: torch.Tensor):
    """
    Triton-only implementation that:
    - Computes sorted_token_indices (stable permutation) via Triton counting sort.
    - Computes expert_offsets via Triton histogram + prefix sum.
    """
    assert topk_idx.is_cuda, "topk_idx must be on CUDA device for Triton kernels."
    flat = topk_idx.reshape(-1).contiguous()  # 1D of length N
    N = flat.numel()
    L = 256  # num_experts

    # 1) Histogram of values in [0..255]
    counts = torch.zeros(L, dtype=torch.int32, device=flat.device)
    BLOCK_HIST = 256
    grid_hist = (triton.cdiv(N, BLOCK_HIST),)
    histogram_kernel[grid_hist](flat, counts, N, L, BLOCK_HIST, num_warps=4)

    # 2) Compute stable ranks and write permutation
    # Allocate rank buffer
    rank_buf = torch.empty(N, dtype=torch.int32, device=flat.device)
    BLOCK_RANK = 1024
    grid_rank = (triton.cdiv(N, BLOCK_RANK),)
    compute_rank_kernel[grid_rank](flat, counts, rank_buf, N, L, BLOCK_RANK, num_warps=4)

    # Now write original indices j into sorted_token_indices at computed ranks
    sorted_token_indices = torch.empty(N, dtype=torch.int32, device=flat.device)
    write_permutation_kernel[grid_rank](rank_buf, flat, sorted_token_indices, N, L, BLOCK_RANK, num_warps=4)

    # 3) Expert offsets: inclusive count of elements less than each expert id
    # We need offsets length L+1. We'll compute inclusive sums and then adjust in host.
    offsets = torch.empty(L + 1, dtype=torch.int32, device=flat.device)
    # exclusive_scan_kernel expects counts length >= L+1? We pass counts length L and only read up to L-1. For simplicity, we do a two-step:
    # First zero offsets, then compute exclusive sum into offsets[0:L], and set offsets[L] = N.
    # However, to keep exclusive_scan simple, we pass a counts array of length L+1 and leave the last element unused.
    # So we create a counts_plus1 buffer to hold scanned result and ignore last element.
    # But we already have counts of length L. We can compute exclusive scan directly into offsets[0:L] and set offsets[L] to N manually.
    # Implementing scan here with a loop in Triton kernel using grid=(1,):
    # We need to read counts[0..L-1] and write running sums into offsets[0..L-1] as described.
    # Since Triton requires a launch grid, we launch grid=(1,) and iterate L.
    # Note: This kernel uses grid=(1,) since it's a single-program scan.
    # Call exclusive_scan_kernel to populate offsets[0:L] via the pattern described above.
    # Launch with num_warps=1; counts pointer points to counts buffer of length L; offsets pointer points to offsets buffer length L+1.
    # We will read counts[e] and write into offsets[e] = running - cnt. Then host sets offsets[L] = N.
    exclusive_scan_kernel[(1,)](counts, offsets, L, num_warps=1)

    # Set last element to N (total number of elements)
    offsets[L] = N

    return sorted_token_indices, offsets


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        # Ensure Triton is used: we invoke at least one Triton kernel (and multiple here).
        # Note: get_inputs returns CPU tensors; move to CUDA for Triton.

            topk_idx = topk_idx.to('cuda')
        return run_triton(topk_idx)
```
