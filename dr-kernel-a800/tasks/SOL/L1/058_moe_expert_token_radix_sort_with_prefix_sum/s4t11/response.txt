import torch
import triton
import triton.language as tl


@triton.jit
def _histogram_kernel(flat_ptr, N, counts_ptr, BLOCK: tl.constexpr):
    """
    Histogram of 8-bit integer ids in flat_ptr into counts_ptr[0..255].
    Each program handles BLOCK elements; for each id in 0..255:
      - build mask (flat == id), sum it, atomic_add to counts[id].
    This minimizes atomic operations per program.
    """
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    # Load flat values; assume dtype int32 for flat
    x = tl.load(flat_ptr + offs, mask=mask, other=0)

    # Process each expert id 0..255
    # Note: num_experts is 256 in this problem; using a for-loop here is fine and compiles.
    for i in range(256):
        # Equality compare; x is int32, i is int32 scalar
        eq = x == i
        # Mask out-of-range lanes to avoid counting invalid elements
        eq = eq & mask
        # Reduce across the block to get count for this id
        cnt = tl.sum(eq.to(tl.int32), axis=0)
        # Atomic add count to global counts[i]
        tl.atomic_add(counts_ptr + i, cnt)


@triton.jit
def _inclusive_prefix_offsets_kernel(
    counts_ptr,        # int32[num_experts]
    offsets_ptr,       # int32[num_experts+1], initialized as [0, ...]
    left_ptr,          # int32[num_experts]
    right_ptr,         # int32[num_experts]
    carry_ptr,         # int32[1], holds carry between bins
    num_experts: tl.constexpr,  # here 256
):
    """
    Compute inclusive prefix offsets for counts_ptr into offsets_ptr[1..],
    using a two-pass scan per bin:
      - left[i] = counts[i] - carry; carry = left[i]
      - right[i] = left[i] + (i == 0 ? 0 : left[i-1])
      - offsets[i+1] = offsets[i] + right[i]
    All arrays are length num_experts.
    """
    i = tl.program_id(axis=0)
    if i >= num_experts:
        return

    # Pass 1: left and right across the array, carry updated per element
    # Load current counts[i]
    curr = tl.load(counts_ptr + i)
    # Load previous carry
    prev_carry = tl.load(carry_ptr)
    # Compute left = counts - carry
    left_i = curr - prev_carry
    # Store left[i]
    tl.store(left_ptr + i, left_i)
    # Update carry = left[i]
    tl.store(carry_ptr, left_i)

    # For right[i]: depends on prev left[i-1]
    # We need left[i-1]; if i == 0, right[i] = left[i] + 0; else add prev_left
    prev_left = 0
    if i > 0:
        prev_left = tl.load(left_ptr + (i - 1))
    right_i = left_i + (prev_left if i > 0 else 0)
    tl.store(right_ptr + i, right_i)

    # Pass 2: offsets[i+1] = offsets[i] + right[i]
    prev_offset = tl.load(offsets_ptr + i)
    new_offset = prev_offset + right_i
    tl.store(offsets_ptr + (i + 1), new_offset)


class ModelNew(torch.nn.Module):
    """
    Triton-optimized version:
    - Uses a Triton histogram kernel to count per-expert tokens.
    - Computes per-expert inclusive offsets with Triton via a two-pass scan.
    - Keeps stable sorting in PyTorch to match original behavior precisely.
    """

    def __init__(self):
        super().__init__()

    def forward(self, topk_idx: torch.Tensor):
        # Ensure tensor is on CUDA and contiguous
        if not topk_idx.is_cuda:
            topk_idx = topk_idx.to(device="cuda")
        topk_idx = topk_idx.contiguous()

        # Flatten to 1D
        flat = topk_idx.reshape(-1)
        N = flat.numel()

        # Triton histogram per expert id (0..255). Ensure dtype int32 for equality
        flat_i32 = flat.to(torch.int32)

        # counts buffer
        counts = torch.zeros(256, dtype=torch.int32, device=flat.device)

        # Kernel launch configuration
        BLOCK = 1024  # process 1024 elements per program
        grid = (triton.cdiv(N, BLOCK),)

        _histogram_kernel[grid](flat_i32, N, counts, BLOCK=BLOCK, num_warps=4)

        # Compute per-expert offsets using Triton (inclusive prefix sum)
        num_experts = 256
        offsets = torch.zeros(num_experts + 1, dtype=torch.int32, device=flat.device)
        # left and right temporary buffers
        left = torch.empty(num_experts, dtype=torch.int32, device=flat.device)
        right = torch.empty(num_experts, dtype=torch.int32, device=flat.device)
        # carry buffer (single int32)
        carry = torch.empty(1, dtype=torch.int32, device=flat.device)

        # Run per-expert scan; one program per bin (simple, reliable)
        # Note: We could do a multi-program scan with block reductions, but
        # since num_experts is small (256), this simple approach is efficient and robust.
        for i in range(num_experts):
            _inclusive_prefix_offsets_kernel[(1,)](
                counts, offsets, left, right, carry, num_experts=num_experts
            )

        # Stable sort of flattened indices (PyTorch), matching original behavior
        sorted_token_indices = flat.sort(stable=True)[1].to(torch.int32)

        return sorted_token_indices, offsets

import triton.language as tl

    Each program handles BLOCK elements; for each id in 0..255:

    This minimizes atomic operations per program.
    """

    mask = offs < N

    x = tl.load(flat_ptr + offs, mask=mask, other=0)

    for i in range(256):
        eq = (x == i) & mask
        # Reduce across the block to get count for this id
        cnt = tl.sum(eq.to(tl.int32), axis=0)
        # Atomic add count to global counts[i]
        tl.atomic_add(counts_ptr + i, cnt)


@triton.jit
def _compute_offset_for_bin(counts_ptr, offsets_ptr, bin_idx, carry_ptr):
    """
    Helper to compute offsets[bin_idx+1] given counts[bin_idx] and current carry.
    We update offsets[bin_idx+1] in-place. This is a single-bin scan with carry.
    """
    # Load current counts[bin_idx]
    curr = tl.load(counts_ptr + bin_idx)
    # Load previous carry (sum of counts[0..bin_idx-1])
    prev_carry = tl.load(carry_ptr)
    # left = counts - carry
    left_i = curr - prev_carry
    # Update offsets[bin_idx+1] = prev_offset + left
    prev_offset = tl.load(offsets_ptr + bin_idx)
    new_offset = prev_offset + left_i
    tl.store(offsets_ptr + (bin_idx + 1), new_offset)
    # Update carry for next bins: sum of processed bins so far
    new_carry = prev_carry + left_i
    tl.store(carry_ptr, new_carry)
