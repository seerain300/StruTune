                new_v_i = tl.where(swap, v_p, v_i)
                new_v_p = tl.where(swap, v_i, v_p)
                new_o_i = tl.where(swap, o_p, o_i)
                new_o_p = tl.where(swap, o_i, o_p)

        stride = stride // 2
