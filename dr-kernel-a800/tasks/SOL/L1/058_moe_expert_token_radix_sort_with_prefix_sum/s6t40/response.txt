    offsets = lane * BLOCK + tl.arange(0, BLOCK)

import torch
import triton
import triton.language as tl


# Triton kernel: bitonic stable sort on a list of pairs (value, index), ascending by value.
# We sort an array of length N over the first dimension (value) stably using a bitonic network.
# The array is laid out as [0..N-1] in memory; each element is a 2-int32 pair: [value, original_index].
# We operate with BLOCK = next power of two >= N, padding with +inf values (represented as large int32) for simplicity.
# For pairs, we implement compare-swap: if a > b, swap; for equal values, swap if index_a > index_b to enforce stability.
@triton.jit
def bitonic_sort_pairs_kernel(data_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    # Each program handles one position i in the block
    i = pid
    k = 2
    while k <= BLOCK:
        j = k // 2
        while j > 0:
            partner = i ^ j
            # Load current pair (i) and partner pair (partner)
            base_i = i * 2
            base_p = partner * 2
            ai = tl.load(data_ptr + base_i + 0)  # value at i
            ii = tl.load(data_ptr + base_i + 1)  # original index at i
            ap = tl.load(data_ptr + base_p + 0)  # value at partner
            ip = tl.load(data_ptr + base_p + 1)  # original index at partner

            # Determine if a should precede b in ascending order (stable)
            asc = ( (i & k) == 0 )
            cmp = ai > ap  # ascending comparison
            # For ties, prefer smaller original index: stable=True for equal values
            tie = ai == ap
            stable_swap = tie & (ii > ip)

            do_swap = cmp ^ asc  # XOR: ascending -> swap on ai>ap, descending -> swap on ai<ap
            do_swap = do_swap | (stable_swap & asc)  # in ascending passes, tie-break by index

            # If do_swap, swap the pairs (i, partner)
            # Compute swapped values
            ai_new = tl.where(do_swap, ap, ai)
            ii_new = tl.where(do_swap, ip, ii)
            ap_new = tl.where(do_swap, ai, ap)
            ip_new = tl.where(do_swap, ii, ip)

            # Write back
            tl.store(data_ptr + base_i + 0, ai_new)
            tl.store(data_ptr + base_i + 1, ii_new)
            tl.store(data_ptr + base_p + 0, ap_new)
            tl.store(data_ptr + base_p + 1, ip_new)

            j //= 2
        k *= 2

        num_experts = 256

        # Length must be a power of two for bitonic sort; pad if necessary.
        # Here N is small in workloads (e.g., up to 4096). We choose BLOCK as next power of two >= N.
        def next_power_of_two(x: int) -> int:
            return 1 << (x - 1).bit_length()

        BLOCK = next_power_of_two(N)
        # Allocate data for pairs: [value, index] of length BLOCK
        data = torch.empty(BLOCK * 2, dtype=torch.int32, device=flat.device)
        # Initialize valid positions with (flat[i], i), pad invalid positions with large value (e.g., 1<<30) to push to end
        # We'll set padded values to a large number so they sort to the end.
        INF = 1 << 30  # large int32 sentinel
        # Fill data: for i < N, store (flat[i], i); for i >= N, store (INF, i)
        for i in range(BLOCK):
            if i < N:
                val = flat[i]
            else:
                val = INF
            data[i * 2 + 0] = val
            data[i * 2 + 1] = i

        grid = (BLOCK,)  # one program per position in the block
