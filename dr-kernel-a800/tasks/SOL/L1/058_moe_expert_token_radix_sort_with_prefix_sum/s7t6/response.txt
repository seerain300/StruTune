# For each element in flat, atomically increment counts[value].
@triton.jit
def histogram_kernel(flat_ptr, counts_ptr, M, NUM_VALUES: tl.constexpr, BLOCK: tl.constexpr):

    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
    for i in range(BLOCK):

        if mask[i]:

@triton.jit
def prefix_sum_kernel(counts_ptr, prefix_ptr, NUM_VALUES: tl.constexpr):

    i = 0
    while i < NUM_VALUES:
        count_i = tl.load(counts_ptr + i)
        tl.store(prefix_ptr + i, running)
        running += count_i
        i += 1
    # handle the last element (store running after adding count[NUM_VALUES-1])
    # we already stored each i: the loop above is a simple scan; running has final value.

        # For each i, count how many elements equal to v appear before it in original order.
        # We do this in blocks and write positions to sorted_token_indices[i] = sum_less + eq_before_count.

            # Initialize eq_before_count vector

            # For each j in the block, count how many true in eq_mask at positions < j
            for j in range(BLOCK):
                # j is scalar here
                # Build mask for "positions < j" within this block
                less_j_mask = (offsets_less < j) & mask  # but we need to consider only positions actually in range and equal to v
                # eq_mask is per element, less_j_mask is per element; build per-element eq & less

                # Implement by scanning k and accumulating
                acc = tl.zeros((), dtype=tl.int32)
                for k in range(0, BLOCK):
                    # Only consider k within this block's actual mask
                    mkk = (k < M) & mask[k]
                    # But we must stay within BLOCK range; M may exceed BLOCK, but our for-loop structure ensures k is within M

                    contrib = eqk_i * less_k.to(tl.int32)

            for j in range(BLOCK):

                    # sorted_token_indices should be of length M and int32

        sum_less = tl.zeros((), dtype=tl.int32)

            mask = idxs < M

            is_less = (vals < v) & mask
            block_sum = tl.sum(is_less.to(tl.int32), axis=0)
            sum_less += block_sum

        for start in range(0, M, BLOCK):

            mask = idxs < M

            eq_mask = (vals == v) & mask  # vector of booleans

                    acc += contrib
