import torch
import triton
import triton.language as tl


@triton.jit
def count_experts_kernel(vals_ptr, counts_ptr, N, num_experts: tl.constexpr):
    """
    Triton kernel: for each token i in 0..N-1, if vals[i] == e, atomic add 1 to counts[e].
    vals_ptr: *int32, length N
    counts_ptr: *int32, length num_experts
    N: number of tokens (runtime int)
    num_experts: number of experts (constexpr, e.g., 256)
    Each program processes a block of tokens sequentially to avoid excessive atomic ops.
    """
    # Use a 1D launch grid; each program handles BLOCK tokens at a time
    BLOCK = 1024
    pid = tl.program_id(0)
    # Starting index for this program
    start = pid * BLOCK
    # Loop over tokens in chunks
    for offset in range(0, N, BLOCK):
        idx = start + offset + tl.arange(0, BLOCK)
        mask = idx < N
        # Load vals for this chunk
        vals = tl.load(vals_ptr + idx, mask=mask, other=0)
        # For each expert e, count occurrences in this chunk and atomic add to global counts[e]
        # num_experts is constexpr, so this loop is unrolled
        for e in range(0, num_experts):
            # mask_e = (vals == e) only for valid idx
            eq = vals == e  # int1 mask
            # Convert mask to int32 by casting and reduce over BLOCK dimension
            # We use tl.sum on the vector; for invalid positions, vals are 0 and eq is False
            cnt = tl.sum(tl.where(mask, eq, False).to(tl.int32))
            # Atomic add to counts[e]
            tl.atomic_add(counts_ptr + e, cnt)


@triton.jit
def inclusive_scan_kernel(counts_ptr, out_ptr, N: tl.constexpr):
    """
    Triton kernel: inclusive prefix sum over counts_ptr (length N), store to out_ptr.
    We implement a sequential scan in a single program. N is small (256).
    """
    acc = 0
    # For i in 0..N-1
    for i in range(0, N):
        acc += tl.load(counts_ptr + i)
        tl.store(out_ptr + i, acc)
