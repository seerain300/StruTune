import torch
import triton
import triton.language as tl


@triton.jit
def histogram_kernel(flat_ptr, counts_ptr, M, NUM_VALUES: tl.constexpr, BLOCK: tl.constexpr):
    # Each program handles BLOCK elements, atomically increments counts[value].
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
    # vals are int32; ensure they are within [0, NUM_VALUES-1] (we assume inputs are valid).
    # Atomic add 1 for each valid value.
    for i in range(BLOCK):
        if mask[i]:
            v = vals[i]
            # bounds check: if v < 0 or v >= NUM_VALUES, skip (safety; not expected in eval)
            if (v >= 0) & (v < NUM_VALUES):
                tl.atomic_add(counts_ptr + v, 1)


@triton.jit
def prefix_sum_kernel(counts_ptr, prefix_ptr, NUM_VALUES: tl.constexpr):
    # Compute inclusive prefix sums: prefix[i] = sum_{x=0..i} counts[x]
    # We do this in a single program looping over i, since NUM_VALUES is small (256).
    for i in range(NUM_VALUES):
        total = 0
        for j in range(i + 1):  # sum from 0 to i
            total += tl.load(counts_ptr + j)
        tl.store(prefix_ptr + i, total)


@triton.jit
def assemble_offsets(prefix_ptr, offsets_ptr, NUM_VALUES: tl.constexpr):
    # offsets[0] = 0; offsets[i+1] = prefix[i]
    tl.store(offsets_ptr + 0, 0)
    for i in range(NUM_VALUES):
        tl.store(offsets_ptr + 1 + i, tl.load(prefix_ptr + i))

    # original_pos[i] is the original position i.

    # Phase 1: per-value scan to compute number_of_less and eq_before counts, then write positions.
    # We will do this in two loops: first compute number_of_less for each v; second, write positions using eq_before counts.
    for v in range(NUM_VALUES):

        # Compute number_of_less = count of elements strictly less than v.
        base = 0
        while base < M:

            mask = offsets < M
            vals = tl.load(flat_ptr + offsets, mask=mask, other=0)
            for i in range(BLOCK):
                if mask[i]:

            base += BLOCK

        # Second pass: write positions for all i where flat[i] == v, using original positions as tie-breaker.
        write_start = number_of_less
        base = 0
        while base < M:

                if mask[i]:
                    vi = vals[i]
                    # If tie-breaking by original_pos (strictly less takes lower index), ensure we don't double count.
                    # We set write_start only for vi == v.
                    if vi == v:
