import math
import torch
import triton
import triton.language as tl


@triton.jit
def stable_bitonic_sort_kernel(vals_ptr, out_idx_ptr, N, LOGN: tl.constexpr):
    """
    Stable bitonic sort on int32 vals_ptr. Writes sorted original indices (int64) into out_idx_ptr.
    Grid:
      axis 0: programs per element (N)
      axis 1: programs per bitonic stage (LOGN)
    For each stage j (axis 1), each element i (axis 0) compares with partner = i ^ (1 << j).
    Ascending if ((i & (1 << (j+1))) == 0), else descending. Stability: tie-break by i < partner.
    """
    i = tl.program_id(axis=0)  # element index
    j = tl.program_id(axis=1)  # stage index

    # Compute bit lengths
    LOGN_CONST = LOGN  # passed as constexpr
    # Compute partner index for this stage
    partner = i ^ (1 << j)

    # Load current values
    val_i = tl.load(vals_ptr + i)
    val_p = tl.load(vals_ptr + partner)

    # Determine direction for this stage
    asc = ((i & (1 << (j + 1))) == 0)

    # Load original indices
    idx_i = tl.load(out_idx_ptr + i)
    idx_p = tl.load(out_idx_ptr + partner)

    # Compare and decide swap, with stability tie-breaker
    swap = tl.zeros((), dtype=tl.int1)
    gt = val_i > val_p
    lt = val_i < val_p
    eq = ~(gt | lt)
    # If ascending: swap if val_i > val_p; if descending: swap if val_i < val_p
    if asc:
        swap = gt
    else:
        swap = lt
    # Stability: if equal, lower original index comes first
    stable_swap = eq & (i < partner)

    do_pair = i < partner  # process each pair once

    new_val_i = val_i
    new_val_p = val_p
    new_idx_i = idx_i
    new_idx_p = idx_p

    # If we are the 'i' side and we need to swap, take partner's values; else take own values
    # Also, if stability requires swap when equal, override based on i < partner.
    # This block is executed in the context of program i; program partner will compute the complementary state.
    # We need to ensure both sides perform the same swap. To avoid double writes, we only perform swap when do_pair.
    # Triton will execute both sides, but only one will store because both load current and will write if swap.
    # We can force only one side to write by using do_pair. The partner side with i >= partner will not store.
    if do_pair:
        if swap:
            new_val_i = val_p
            new_val_p = val_i
            new_idx_i = idx_p
            new_idx_p = idx_i
        else:
            new_val_i = val_i
            new_val_p = val_p
            new_idx_i = idx_i
            new_idx_p = idx_p
        # Stability override: if equal and i < partner, ensure lower index comes first
        if eq & stable_swap:
            new_val_i = val_p
            new_val_p = val_i
            new_idx_i = idx_p
            new_idx_p = idx_i

    # Store results back to positions i and partner, but only the 'i' side stores to avoid conflicts.
    # Triton kernel grid ensures each program handles one position; partner's program handles the other.
    # We can store here safely as partner != i for do_pair; otherwise, i stores for non-pair or other stage.
    # Note: We store for all axis0 programs; the partner's program also stores to the same positions.
    # This is fine because both sides compute the same swap decision; only one side will write due to do_pair.
    # To avoid any race, we make partner's program write only if i >= partner by using do_pair and (i < partner).
    # However, Triton doesn't support if on axis1; we rely on do_pair to ensure only one writes.
    # Since we have axis1 as LOGN, and each element i runs for all j, both sides will run and both will attempt to write.
    # To prevent conflicting writes, we structure the kernel so that only one side performs the store.
    # Triton doesn't allow per-element condition on axis1, so we instead restructure: only write when i < partner.
    # We'll store only from the i side:
    # Since both sides compute, but only one writes, we must ensure single writer per pair.
    # Triton allows scalar if based on runtime condition; we will store unconditionally because both sides will load current vals.
    # The partner side will not write due to do_pair being false there. Therefore, we store only when do_pair is true for this side.

    # We store both positions using do_pair mask:
    # If do_pair is true for this program i, then it writes both i and partner. The partner program will also set do_pair True and write partner and i symmetrically.
    # Because partner != i for do_pair, there is no conflict.

    # Store current indices for safety in case no swap occurs
    tl.store(out_idx_ptr + i, new_idx_i)
    tl.store(out_idx_ptr + partner, new_idx_p)

def count_histogram_kernel(vals_ptr, counts_ptr, N, num_experts: tl.constexpr):
    """
    Compute histogram of int32 vals (length N) into counts_ptr (int32), length num_experts.
    Each program processes BLOCK elements and atomically adds 1 for each occurrence.
    """
    pid = tl.program_id(axis=0)
    BLOCK = 1024
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    vals = tl.load(vals_ptr + offs, mask=mask, other=0)  # int32 values in [0, num_experts-1]
    tl.atomic_add(counts_ptr + vals, 1, mask=mask)


@triton.jit
def prefix_sum_kernel(counts_ptr, offsets_ptr, num_experts: tl.constexpr):
    """
    Compute inclusive prefix sum of counts_ptr (int64, length num_experts) into offsets_ptr (int64).
    offsets_ptr[0] should be set to 0 by the caller. We write offsets[1..].
    """
    pid = tl.program_id(axis=0)
    acc = tl.zeros((), dtype=tl.int64)
    # Triton loop over num_experts, constexpr
    for i in range(0, num_experts):
        acc += tl.load(counts_ptr + i)
        tl.store(offsets_ptr + (i + 1), acc)
