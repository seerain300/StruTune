import triton
import triton.language as tl


@triton.jit
def _hist_kernel(topk_ptr, counts_ptr, N, NUM_EXPERTS: tl.constexpr, BLOCK_SIZE: tl.constexpr):

    pid = tl.program_id(0)

    mask = offs < N

    # This avoids per-element atomics; we aggregate per-bin.

        # Compute bin index
        bin_idx = val  # safe because val is in [0, NUM_EXPERTS-1]
        # Mask to only count valid lanes
        valid = mask[j]
        # Accumulate: if valid and bin_idx < NUM_EXPERTS, increment
        # We can't branch per-lane; instead, we use a masked vectorized add.
        # We build a vector 'ones' for bin_idx and sum it. But better: directly use scalar add.

        # For each bin, create a boolean vector 'is_equal' and reduce with tl.sum.
        # But since bin_idx is scalar, we need to vectorize across bins.
        # The clean approach is: for each j, find bin_idx and increment local_counts[bin_idx] by 1 if valid.
        # Triton doesn't support per-lane scalar vector indexing in a simple way, so we use a vectorized per-lane increment via a temporary vector:
        # We prepare a vector of size NUM_EXPERTS: only element at bin_idx increments by 1 for this j.

        # local_counts[bin_idx] += 1. This is supported when bin_idx is a scalar int.

    for b in range(NUM_EXPERTS):
