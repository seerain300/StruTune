import triton.language as tl


@triton.jit
def histogram_kernel(orig_ptr, counts_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    # Each program counts occurrences for a block of indices.
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    vals = tl.load(orig_ptr + offsets, mask=mask, other=0)  # int32
    # Count occurrences of each value v in [0..L-1]
    # For masked lanes, load as 0 so they don't contribute.
    for v in range(L):
        eq = (vals == v) & mask
        # Sum eq across the vector and atomic add to counts[v]
        tl.atomic_add(counts_ptr + v, tl.sum(eq.to(tl.int32)))


@triton.jit
def stable_rank_write_kernel(orig_ptr, sorted_ptr, counts_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    # For each position j, compute its stable rank and write j at that rank.
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # Load original values at positions offsets
    v = tl.load(orig_ptr + offsets, mask=mask, other=0)  # int32
    running = tl.zeros([BLOCK], dtype=tl.int32)

    # Compute stable rank: for each value v_scan < v, add counts[v_scan]; for v_scan == v, do not add (preserve order)
    for v_scan in range(L):
        count_v = tl.load(counts_ptr + v_scan)  # int32 scalar
        # If v_scan < v, increment running; if v_scan == v, do nothing (stable)
        running += (v_scan < v) * count_v

    # Store: write position 'offsets' at stable_rank = running
    # Since running is the rank, sorted_ptr[running] = offsets
    # Note: offsets is the original index j; we store j at position running.
    # Use mask to avoid OOB writes.

    tl.store(sorted_ptr + running, offsets, mask=mask & (running < N))

def exclusive_scan_kernel(counts_ptr, offsets_ptr, num_exps: tl.constexpr, BLOCK: tl.constexpr):
