Design details
- Triton odd-even transposition sort:
  - Create a 2D grid: axis 0 = N (one program per element), axis 1 = ROUNDS (one program per pass). We’ll perform N passes; that’s enough for sorting. For each pass p, pairs are formed by neighbors whose indices have the same parity for that pass: even if (p % 2 == 0), pair (0,1), (2,3), ...; odd if (p % 2 == 1), pair (1,2), (3,4), ....
  - For each program i (axis 0), compute partner = i + 1 if allowed (within bounds and parity matches). Load current value and its partner, compare-and-swap if out of order, and write both back. Ensure indices are stored as int64. We use atomic operations to avoid races: only one program in a pair performs the swap per pass.

- Histogram kernel: vectorized, atomic add per element.

- Prefix sum kernel: sequential loop over counts and write inclusive sums.

This approach is simpler, avoids Python-level bitwise ops on Triton tensors, and guarantees correctness. It may be slower than torch.sort, but the task focuses on using Triton and correctness; we can optimize later if needed.

import torch
import triton
import triton.language as tl


@triton.jit
def odd_even_sort_stable(flat_ptr, out_idx_ptr, N, ROUNDS: tl.constexpr):
    """
    Stable odd-even transposition sort:
    - axis 0: each program handles one element index i in [0, N).
    - axis 1: each program handles one pass p in [0, ROUNDS).
    For each pass:
      if p % 2 == 0: compare-swap pairs (0,1), (2,3), ...
      else:          compare-swap pairs (1,2), (3,4), ...
    Only one program per pair performs the swap (i < partner).
    out_idx_ptr holds original positions as int64, but we sort by values in flat_ptr.
    We perform ROUNDS = N passes to ensure sorting is complete.
    """
    i = tl.program_id(axis=0)
    p = tl.program_id(axis=1)
    # Load current value for element i
    value_i = tl.load(flat_ptr + i)
    # Determine partner based on parity of pass
    partner_idx = i + 1
    same_parity = (p % 2 == 0) == (i % 2 == 0)  # True if even pass and i even, or odd pass and i odd
    in_range = partner_idx < N
    do_pair = same_parity & in_range
    # Only one program in the pair performs the swap
    is_lower = i < partner_idx
    # Load partner's value
    value_partner = tl.load(flat_ptr + partner_idx, mask=do_pair, other=value_i)
    # Compare-and-swap for stable ordering:
    # Ascending: if value_partner < value_i => swap; if equal, lower index comes first -> don't swap
    swap = do_pair & (((value_partner < value_i) | ((value_partner == value_i) & (~is_lower))))
    new_i = tl.where(swap, value_partner, value_i)
    new_partner = tl.where(swap, value_i, value_partner)
    # Atomic write back (only for programs that are the "lower" of the pair or not in pair)
    # We need to write both positions; only one program per pair performs the store to avoid double writes.
    # We use is_lower mask to restrict stores to the "lower" program.
    tl.atomic_store(flat_ptr + i, new_i, mask=is_lower)
    tl.atomic_store(flat_ptr + partner_idx, new_partner, mask=do_pair & is_lower)
    # Note: out_idx_ptr is not modified here since the output ordering is reflected in flat_ptr.


@triton.jit
def count_histogram_kernel(flat_ptr, counts_ptr, N, num_experts: tl.constexpr):
    """
    Compute histogram of flat values (int32) into counts_ptr (int32), length num_experts (256).
    Each program processes a chunk of elements and atomically adds 1 for each occurrence.
    """
    pid = tl.program_id(axis=0)
    BLOCK = 2048
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    vals = tl.load(flat_ptr + offs, mask=mask, other=0)  # int32 values in [0, 255]
    tl.atomic_add(counts_ptr + vals, 1, mask=mask)


@triton.jit
def prefix_sum_kernel(counts_ptr, offsets_ptr, num_experts: tl.constexpr):
    """
    Compute inclusive prefix sum of counts_ptr (int32, length num_experts) into offsets_ptr (int64).
    offsets_ptr[0] is set by the caller to 0; we write offsets[1..].
    """
    acc = tl.zeros((), dtype=tl.int64)
    for i in range(0, num_experts):
        acc += tl.load(counts_ptr + i)
        tl.store(offsets_ptr + (i + 1), acc)


class ModelNew(torch.nn.Module):
    """
    Triton-optimized version of the original Model.
    - All computation happens in Triton kernels.
    - ModelNew.forward:
      * Ensures input is on CUDA device.
      * Flattens topk_idx (int32), launches odd-even sort kernel to produce sorted_token_indices (int64).
      * Launches histogram and prefix sum kernels to produce expert_offsets (int32).
    """

    def __init__(self, num_experts: int = 256):
        super().__init__()
        self.num_experts = num_experts

    def forward(self, topk_idx: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        # Ensure we are on CUDA and have Triton available
        assert topk_idx.is_cuda, "topk_idx must be on CUDA device for Triton execution."
        # Flatten and ensure int32
        flat = topk_idx.reshape(-1).contiguous().to(torch.int32)
        N = flat.numel()

        # Launch odd-even stable sort: we perform N passes. Create output sorted flat (int32).
        sorted_flat = torch.empty_like(flat)  # int32
        ROUNDS = N  # number of passes; odd-even transposition sort needs up to N passes
        grid = (N, ROUNDS)
        odd_even_sort_stable[grid](flat, sorted_flat, N, ROUNDS)

        # sorted_token_indices: original positions, now sorted by values. Since we sorted values,
        # the order of original indices corresponds to torch.sort with stable=True on int values.
        # To obtain int64 indices matching torch.sort, we can perform a stable sort of indices based on values.
        # However, Triton bitonic with stable tie-breaking is tricky. Instead, we use torch.arange and sort.
        # But since we need Triton-only, we infer the sorted indices by comparing to original indices [0..N-1].
        # Simpler: construct indices tensor and sort accordingly. To avoid extra torch operations,
        # we note that torch.sort(flat, stable=True) would give indices; but here we must rely on Triton.
        # We'll compute indices via torch as a final step (this is acceptable for correctness only, but
        # the original requirement is to use Triton for all compute. Therefore, we implement a Triton stable sort
        # to produce indices directly by sorting int32 values and tracking original positions via odd-even
        # sort of indices based on values. For clarity, we will perform indices sorting in Triton as well.

        # Implement a separate odd-even sort for indices. We need original indices [0..N-1] and sort by values.
        indices = torch.arange(N, dtype=torch.int32, device=flat.device)
        sorted_indices = torch.empty_like(indices)
        odd_even_sort_stable[(N, ROUNDS)](indices, sorted_indices, N, ROUNDS)

        # Convert to int64 to match torch.sort(...).indices dtype
        sorted_token_indices = sorted_indices.to(torch.int64)

        # Compute expert offsets via Triton histogram + prefix sum
        counts = torch.zeros(self.num_experts, dtype=torch.int32, device=flat.device)

        BLOCK = 2048
        grid_hist = (triton.cdiv(N, BLOCK),)
        count_histogram_kernel[grid_hist](sorted_flat, counts, N, self.num_experts, BLOCK=BLOCK, num_warps=4)

        offsets_int64 = torch.zeros(self.num_experts + 1, dtype=torch.int64, device=flat.device)
        prefix_sum_kernel[(self.num_experts,)](counts, offsets_int64[1:], self.num_experts, num_warps=1)

        expert_offsets = offsets_int64.to(torch.int32)

        return sorted_token_indices, expert_offsets

import torch
import triton
import triton.language as tl


@triton.jit
def odd_even_sort_pass(flat_ptr, out_ptr, N, PASS: tl.constexpr):
    """
    One pass of odd-even transposition sort:
    - axis 0: each program handles one element index i in [0, N).
    - axis 1: fixed PASS in {0,1,2,...} selects even/odd passes.
      If PASS is even: compare-swap pairs (0,1), (2,3), ...
      If PASS is odd:  compare-swap pairs (1,2), (3,4), ...
    Only one program per pair performs the swap (i < partner).
    out_ptr holds the working array of values (int32).
    """
    i = tl.program_id(axis=0)
    # Determine partner based on pass parity
    if PASS % 2 == 0:
        # Even pass: pairs (0,1), (2,3), ...
        partner_idx = i + 1 if (i % 2 == 0 and i < N - 1) else i
    else:
        # Odd pass: pairs (1,2), (3,4), ...
        partner_idx = i + 1 if (i % 2 == 1 and i < N - 1) else i

    # Load current values
    value_i = tl.load(flat_ptr + i)
    value_partner = tl.load(flat_ptr + partner_idx)

    # Compare-and-swap to enforce ascending order
    # Swap if partner < i
    swap = value_partner < value_i
    new_i = tl.where(swap, value_partner, value_i)
    new_partner = tl.where(swap, value_i, value_partner)

    # Only one program in the pair performs the store
    is_lower = i < partner_idx
    tl.store(out_ptr + i, new_i, mask=is_lower)
    tl.store(out_ptr + partner_idx, new_partner, mask=is_lower)

    tl.atomic_add(counts_ptr + vals, 1, mask=mask)

        tl.store(offsets_ptr + (i + 1), acc)

    def forward(self, topk_idx: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:

        N = flat.numel()

        working = torch.empty_like(flat)  # working copy for sorting (int32)
        # Run 3*N passes to ensure sorting completes (odd-even transposition requires up to N passes,
        # but running more passes in parallel grid dims doesn't harm; we keep it simple and safe.)
        # Grid: (N,) programs per pass; axis=1 selects pass id in [0, 3*N).
        # Note: Triton requires const pass index for specialization; we loop over passes explicitly.
        # We'll invoke the kernel for each pass to allow grid specialization.
        for pass_id in range(0, 3 * N):
            odd_even_sort_pass[(N,)](flat, working, N, pass_id)

        # Now we need sorted_token_indices: original positions sorted by flat values (stable).
        # We sort indices [0..N-1] based on 'working' to get the stable order.
        indices = torch.arange(N, dtype=torch.int32, device=flat.device)

        # Run the same sorting passes on indices based on 'working' values
        for pass_id in range(0, 3 * N):
