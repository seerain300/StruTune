import torch
import triton
import triton.language as tl


# Triton histogram kernel: counts occurrences of each integer id in [0, L-1]
# Inputs:
#   orig_ptr: pointer to int32 flat tensor of length N
#   counts_ptr: pointer to int32 output histogram length L
#   N: number of elements in orig
#   L: number of possible expert ids (e.g., 256)
@triton.jit
def histogram_kernel(orig_ptr, counts_ptr, N, L: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    # Load values; cast to int32 explicitly
    vals = tl.load(orig_ptr + offsets, mask=mask, other=0).to(tl.int32)
    # For each possible value v in [0, L), accumulate counts
    for v in range(L):
        # Create a boolean mask where vals == v and within bounds
        eq = vals == v
        increment = tl.where(mask & eq, 1, 0)
        # Accumulate per lane, then atomic add once per lane
        # Note: Triton's atomic_add expects a pointer and a value; we sum increments to reduce per-lane work.
        tl.atomic_add(counts_ptr + v, tl.sum(increment, axis=0))


# Triton exclusive scan kernel to compute per-expert offsets from counts
# Inputs:
#   counts_ptr: pointer to int32 counts array of length L
#   out_ptr: pointer to int32 output offsets array of length L+1
#   num_exps: L (number of experts), passed as constexpr for loop unrolling
@triton.jit
def exclusive_scan_kernel(counts_ptr, out_ptr, num_exps: tl.constexpr, BLOCK: tl.constexpr):
    running = 0
    # out_ptr[0] = 0
    # Compute offsets[i] = running before i; store at out_ptr[i+1] to leave out_ptr[0] as zero
    # For each expert i in [0..num_exps-1], running += counts[i]; out_ptr[i+1] = running before adding counts[i]
    for i in range(num_exps):
        # running is int32 scalar; counts_ptr[i] is loaded as scalar
        c = tl.load(counts_ptr + i)
        out_val = running
        running += c
        tl.store(out_ptr + i + 1, out_val)
