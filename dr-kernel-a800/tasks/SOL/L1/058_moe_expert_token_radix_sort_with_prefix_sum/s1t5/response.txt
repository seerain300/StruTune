import torch
import triton
import triton.language as tl


@triton.jit
def histogram_atomic_kernel(flat_ptr, counts_ptr, N, BLOCK: tl.constexpr):
    """
    Histogram of values in flat_ptr (int32), one pass with atomic adds.
    flat_ptr: *int32, shape [N]
    counts_ptr: *int32, shape [num_experts]
    N: int, total number of elements
    Each program processes BLOCK elements; masked loads prevent OOB.
    """
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    for i in range(BLOCK):
        idx = start + i
        m = idx < N  # scalar mask
        if m:
            id_val = tl.load(flat_ptr + idx)
            tl.atomic_add(counts_ptr + id_val, 1)


@triton.jit
def compute_le_counts(counts_ptr, le_ptr, num_experts: tl.int32):
    """
    Compute inclusive prefix sum of counts_ptr into le_ptr for k in [0, num_experts-1].
    counts_ptr: *int32, shape [num_experts]
    le_ptr: *int32, shape [num_experts]
    """
    running = tl.zeros((), dtype=tl.int32)
    # Loop over k; num_experts is passed as tl.constexpr via meta in launch, but we use runtime int for generality.
    for k in range(0, num_experts):
        c = tl.load(counts_ptr + k)
        running += c
        tl.store(le_ptr + k, running)


@triton.jit
def compute_lt_counts(le_ptr, counts_ptr, lt_ptr, num_experts: tl.int32):
    """
    Compute lt_counts[k] = le_ptr[k] - counts_ptr[k] for k in [0, num_experts-1].
    le_ptr: *int32, shape [num_experts]
    counts_ptr: *int32, shape [num_experts]
    lt_ptr: *int32, shape [num_experts]
    """
    for k in range(0, num_experts):
        le_k = tl.load(le_ptr + k)
        cnt_k = tl.load(counts_ptr + k)
        tl.store(lt_ptr + k, le_k - cnt_k)


@triton.jit
def compute_out_pos(flat_ptr, out_pos_ptr, le_ptr, lt_ptr, N, num_experts: tl.int32):
    """
    Compute stable argsort permutation out_pos[i] = i-th element of flat sorted by value.
    out_pos_ptr: *int32, output permutation of length N.
    flat_ptr: *int32, input flattened of length N.
    le_ptr: *int32, length num_experts, inclusive counts
    lt_ptr: *int32, length num_experts, lt_counts = le - counts
    """
    # Single-program grid; each program_id lane handles one element safely with mask.
    pid = tl.program_id(axis=0)
    # We launch grid=(N,), so pid ranges [0, N-1]. Use mask to ensure safety.
    m = pid < N
    if m:
        id_val = tl.load(flat_ptr + pid)
        # le_counts for this value, then subtract 1 for ties to preserve original order
        le_id = tl.load(le_ptr + id_val)
        lt_id = tl.load(lt_ptr + id_val)
        pos = le_id - (lt_id > 0)
        # out_pos stores the original index i at position pos; values are indices [0..N-1]
        tl.store(out_pos_ptr + pos, pid)


@triton.jit
def prefix_scan_counts(counts_ptr, offsets_ptr, num_experts: tl.int32):
    """
    Inclusive prefix sum of counts_ptr into offsets_ptr[1:], offsets_ptr[0]=0.
    counts_ptr: *int32, shape [num_experts]
    offsets_ptr: *int32, shape [num_experts+1]
    """
    # offsets_ptr[0] must be set by host as 0
    running = tl.zeros((), dtype=tl.int32)
    for k in range(0, num_experts):
        cnt = tl.load(counts_ptr + k)
        running += cnt
        tl.store(offsets_ptr + k + 1, running)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        """
        Triton-only forward that returns:
          - sorted_token_indices: stable argsort permutation of flattened topk_idx (indices of positions).
          - expert_offsets: inclusive prefix counts per expert (num_experts+1).
        """
        # Ensure on CUDA and int32; behavior assumes num_experts=256 (same as original code)
        assert topk_idx.is_cuda, "Input must be on CUDA device"
        assert topk_idx.dtype == torch.int32, "Input must be int32"
        flat = topk_idx.reshape(-1)  # 1D of length N
        N = flat.numel()
        num_experts = 256

        # 1) Histogram of expert IDs
        counts = torch.zeros(num_experts, dtype=torch.int32, device=flat.device)
        BLOCK_HIST = 256  # elements per program
        grid_hist = (triton.cdiv(N, BLOCK_HIST),)
        histogram_atomic_kernel[grid_hist](flat, counts, N, BLOCK_HIST)

        # 2) Compute le_counts (inclusive prefix sum) and lt_counts (exclusive less-than)
        le_counts = torch.empty(num_experts, dtype=torch.int32, device=flat.device)
        lt_counts = torch.empty(num_experts, dtype=torch.int32, device=flat.device)
        compute_le_counts[(1,)](counts, le_counts, num_experts)
        compute_lt_counts[(1,)](le_counts, counts, lt_counts, num_experts)

        # 3) Compute stable argsort permutation via Triton: out_pos[i] = i
        sorted_token_indices = torch.empty(N, dtype=torch.int32, device=flat.device)
        compute_out_pos[(N,)](flat, sorted_token_indices, le_counts, lt_counts, N, num_experts)

        # 4) Compute expert_offsets via Triton inclusive prefix sum over counts
        expert_offsets = torch.empty(num_experts + 1, dtype=torch.int32, device=flat.device)
        expert_offsets[0] = 0  # host sets first to zero
        prefix_scan_counts[(1,)](counts, expert_offsets, num_experts)

        return sorted_token_indices, expert_offsets
