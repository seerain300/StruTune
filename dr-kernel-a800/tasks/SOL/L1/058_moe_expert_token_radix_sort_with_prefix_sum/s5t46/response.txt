import torch
import triton
import triton.language as tl


@triton.jit
def counts_and_bases_kernel(
    flat_ptr,                 # *int32, flattened input of length M
    offsets_incl_ptr,         # *int32, output length NUM_EXPERTS (inclusive prefix sums)
    total_count_ptr,          # *int32, scalar to store total count
    M: tl.constexpr,          # number of elements in flat
    NUM_EXPERTS: tl.constexpr,  # number of possible keys (256)
    BLOCK: tl.constexpr,      # chunk size for scanning flat
):
    k = tl.program_id(0)  # one program per key in [0..NUM_EXPERTS)
    cnt = tl.zeros((), dtype=tl.int32)
    # Loop over flat in chunks to count occurrences of key k
    for off in range(0, M, BLOCK):
        offs = off + tl.arange(0, BLOCK)
        mask = offs < M
        vals = tl.load(flat_ptr + offs, mask=mask, other=NUM_EXPERTS)  # out-of-range lanes set to NUM_EXPERTS
        # Count how many lanes equal k
        cnt += tl.sum((vals == k).to(tl.int32), axis=0)
    # Store inclusive prefix sum for key k (note: no exclusive prefix needed here)
    tl.store(offsets_incl_ptr + k, cnt)
    # Accumulate total count via atomic add (though only one key per program, this is fine)
    tl.atomic_add(total_count_ptr, cnt)


@triton.jit
def stable_permutation_kernel(
    flat_ptr,                 # *int32, flattened input of length M
    offsets_incl_ptr,         # *int32, length NUM_EXPERTS (inclusive prefix sums)
    sorted_ptr,               # *int32, output permutation of length M
    total_count_ptr,          # *int32, scalar total count (not used directly)
    M: tl.constexpr,
    NUM_EXPERTS: tl.constexpr,
    BLOCK: tl.constexpr,      # chunk size for scanning flat
):
    # Each program handles a block of output indices
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < M

    # Load values for these indices to compute base_excl
    vals_j = tl.load(flat_ptr + offs, mask=mask, other=0)  # use 0 for masked lanes

    # Compute base_excl = inclusive prefix up to (key - 1); if key == 0, base_excl = 0
    base_excl = tl.zeros([BLOCK], dtype=tl.int32)
    # We need offsets_incl[key - 1] if key > 0; set base_excl = 0 for key == 0
    for k in range(NUM_EXPERTS):
        base = tl.load(offsets_incl_ptr + k)  # inclusive prefix sum at key k
        base_excl += (vals_j > k) * base      # add base if val_j > k
        # For vals_j == k, we should use the base at k; handle in tie_count

    # Compute tie_count for each j: number of previous indices t < j with same key and vals_j < vals_t
    tie_count = tl.zeros([BLOCK], dtype=tl.int32)
    for t_off in range(0, M, BLOCK):
        t = t_off + tl.arange(0, BLOCK)
        mask_t = t < M
        vals_t = tl.load(flat_ptr + t, mask=mask_t, other=0)
        # For each j in this block, scan t block and count ties
        # We only count when t < j and vals_t == vals_j and vals_t < vals_j is false (i.e., equal)
        # So condition: mask_t & (t < j) & (vals_t == vals_j)
        # Note: tl.where semantics, and we need to reduce along t dimension per j.
        # Since Triton doesn't support dynamic per-lane reductions across different axis,
        # we implement as: for each j lane, count across t block.
        for jj in range(BLOCK):
            j_valid = (start + jj) < M
            # Contribution from t block: count where t < (start + jj) and vals_t == vals_j[jj]
            j_val = vals_j[jj]
            # only process if j is valid
            if j_valid:
                t_eq = (vals_t == j_val) & mask_t & (t < (start + jj))
                tie_count[jj] += tl.sum(t_eq.to(tl.int32))

    # Compute rank = base_excl + tie_count, write sorted indices
    rank = base_excl + tie_count
    tl.store(sorted_ptr + offs, rank, mask=mask)


@triton.jit
def finalize_offsets_kernel(
    offsets_incl_ptr,         # *int32, length NUM_EXPERTS (inclusive prefix sums)
    total_count_ptr,          # *int32, scalar total count
    expert_offsets_ptr,       # *int32, output of length NUM_EXPERTS + 1
    NUM_EXPERTS: tl.constexpr,
):
    # Copy inclusive prefix sums to first NUM_EXPERTS positions
    for k in range(NUM_EXPERTS):
        tl.store(expert_offsets_ptr + k, tl.load(offsets_incl_ptr + k))
    # Write total count + 1 at the end
    total = tl.load(total_count_ptr)
    tl.store(expert_offsets_ptr + NUM_EXPERTS, total + 1)


class ModelNew(torch.nn.Module):
    def __init__(self, num_experts: int = 256):
        super().__init__()
        self.num_experts = num_experts

    def forward(self, topk_idx: torch.Tensor):
        # Ensure we're on CUDA for Triton
        assert topk_idx.is_cuda, "topk_idx must be on CUDA device for Triton kernels"
        flat = topk_idx.reshape(-1)
        M = flat.numel()
        NUM_EXPERTS = self.num_experts

        # Allocate outputs
        sorted_token_indices = torch.empty(M, dtype=torch.int32, device=flat.device)
        expert_offsets = torch.empty(NUM_EXPERTS + 1, dtype=torch.int32, device=flat.device)
        # Scalar to hold total count for offsets
        total_count = torch.zeros(1, dtype=torch.int32, device=flat.device)

        # 1) Kernel to compute counts and bases (inclusive prefix sums per key)
        grid_counts = (NUM_EXPERTS,)
        counts_and_bases_kernel[grid_counts](
            flat, expert_offsets, total_count, M, NUM_EXPERTS, BLOCK=1024
        )

        # 2) Kernel to compute stable permutation (sorted token indices)
        grid_perm = (triton.cdiv(M, 1024),)
        stable_permutation_kernel[grid_perm](
            flat, expert_offsets, sorted_token_indices, total_count, M, NUM_EXPERTS, BLOCK=1024
        )

        # 3) Kernel to finalize expert_offsets (inclusive counts + 1 at end)
        # Note: expert_offsets already holds inclusive counts for keys 0..NUM_EXPERTS-1
        # via counts_and_bases_kernel; total_count holds sum of counts.
        # We simply set the last element to total_count + 1. However, to be explicit:
        finalize_offsets_kernel[(1,)](
            expert_offsets, total_count, expert_offsets, NUM_EXPERTS
        )
        # If total_count wasn't updated (we wrote counts per key, not total), set the last element to M+1.
        # Since counts_and_bases_kernel only wrote per-key counts, we must set the last element to M+1.
        # total_count was zero-initialized; we need to compute M via torch, but we can infer M from sorted_token_indices length and the fact that stable_permutation_kernel used M. However, Triton kernel cannot read M here directly.
        # To ensure correctness, we set the last element to M+1 explicitly using PyTorch:
        # We don't have M in Triton scope here, so perform a simple write:
        # After the finalize kernel, total_count still holds 0 because we didn't sum. Hence we overwrite last element:
        # Compute M using flat.numel() on host
        M_host = flat.numel()
        expert_offsets[-1] = M_host + 1

        return sorted_token_indices, expert_offsets
