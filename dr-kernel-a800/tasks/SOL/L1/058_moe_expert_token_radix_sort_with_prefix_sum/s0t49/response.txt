def bitonic_sort_kernel(flat_ptr, out_idx_ptr, N, LOGN: tl.constexpr):
    """
    Stable bitonic sort on flat_ptr (int32). out_idx_ptr holds int64 positions 0..N-1.
    This kernel uses nested static loops over k and j to implement a full bitonic sort.
    Launch with grid=(N,) and LOGN set appropriately (e.g., LOGN=12 for N<=4096).
    """
    i = tl.program_id(axis=0)
    # Bitonic sort network using static loops
    for k in range(0, LOGN):

                    idx_p = tmp
                tl.store(out_idx_ptr + i, idx_i)
                tl.store(out_idx_ptr + partner, idx_p)
