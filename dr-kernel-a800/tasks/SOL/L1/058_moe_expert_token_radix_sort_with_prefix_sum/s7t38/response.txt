import triton
import triton.language as tl


@triton.jit
def histogram_counts_kernel(original_ptr, counts_ptr, M, NUM_VALUES: tl.constexpr, BLOCK: tl.constexpr):
    """
    Compute counts of each int32 value in [0..NUM_VALUES-1] from original_ptr[0:M].
    counts_ptr[v] = number of occurrences of v in original_ptr.
    We only count up to NUM_VALUES; values outside are ignored.
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M

    # Load a tile of original values
    # For masked-off elements, set to 0, then ensure we only count within mask.
    vals = tl.load(original_ptr + offsets, mask=mask, other=0)

    # Initialize counts to zero
    # counts_ptr is assumed to be pre-zeroed by host
    # We'll atomically add for each valid element.
    # Create a local vector of counts for each tile, then atomic_add per element.
    # Since Triton's atomic_add target must be a scalar for per-element atomics, we loop over the tile:
    for i in range(BLOCK):
        if mask[i]:
            val = vals[i]
            # Only count if 0 <= val < NUM_VALUES
            in_range = (val >= 0) & (val < NUM_VALUES)
            if in_range:
                # Atomic add 1 to counts[val]
                tl.atomic_add(counts_ptr + val, 1)


@triton.jit
def inclusive_prefix_sum_kernel(counts_ptr, prefix_ptr, NUM_VALUES: tl.constexpr):
    """
    Compute inclusive prefix sums of counts_ptr[0:NUM_VALUES] into prefix_ptr[0:NUM_VALUES].
    prefix[i] = sum_{k=0..i} counts[k].
    """
    # Each program handles one element i
    i = tl.program_id(axis=0)
    if i < NUM_VALUES:
        total = tl.zeros((), dtype=tl.int32)
        for j in range(i + 1):
            total += tl.load(counts_ptr + j)
        tl.store(prefix_ptr + i, total)


@triton.jit
def assemble_offsets_kernel(prefix_ptr, offsets_ptr, NUM_VALUES: tl.constexpr):
    """
    Assemble offsets: offsets[0] = 0; offsets[i+1] = prefix[i] for i in 0..NUM_VALUES-1.
    """
    i = tl.program_id(axis=0)
    if i == 0:
        tl.store(offsets_ptr + 0, 0)
    elif i <= NUM_VALUES:
        tl.store(offsets_ptr + i, tl.load(prefix_ptr + (i - 1)))
