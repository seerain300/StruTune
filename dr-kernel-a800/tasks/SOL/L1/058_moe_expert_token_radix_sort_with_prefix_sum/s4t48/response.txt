@triton.jit
def _hist_kernel(values_ptr, counts_ptr, NUM_TOKS: tl.constexpr, NUM_EXPERTS: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    """
    Triton kernel: histogram over NUM_EXPERTS bins for int32 values in 'values_ptr' of length NUM_TOKS.
    Each program instance processes BLOCK_SIZE elements and performs atomic adds to counts_ptr.
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < NUM_TOKS
    # Load a block of values
    vals = tl.load(values_ptr + offsets, mask=mask, other=0)
    # For each expert id in [0, NUM_EXPERTS), count occurrences in vals
    for j in range(NUM_EXPERTS):
        # Build a vector compare: vals == j
        cond = vals == j
        # Mask out invalid lanes
        cond = cond & mask
        # Sum of true in this vector -> number of matches for this j within the block
        # Triton allows tl.sum on vectors of 32-bit ints (casting cond to int32)
        matches = tl.sum(cond.to(tl.int32))
        # Atomic add to counts[j]
        tl.atomic_add(counts_ptr + j, matches)

@triton.jit
def _prefix_sum_inclusive(counts_ptr, offsets_ptr, NUM_EXPERTS: tl.constexpr):
    """
    Triton kernel: inclusive prefix sum over 'counts_ptr' of length NUM_EXPERTS.
    Writes result to 'offsets_ptr' at index 1..NUM_EXPERTS; offsets_ptr[0] = 0 in host.
    """
    # We perform a simple sequential scan. NUM_EXPERTS is small (256).
    acc = tl.zeros((), dtype=tl.int32)
    for i in range(NUM_EXPERTS):
        ci = tl.load(counts_ptr + i)
        acc += ci
        tl.store(offsets_ptr + 1 + i, acc)
