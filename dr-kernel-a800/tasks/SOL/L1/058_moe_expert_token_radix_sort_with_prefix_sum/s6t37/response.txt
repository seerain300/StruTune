@triton.jit

    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    vals = tl.load(orig_ptr + offs, mask=mask, other=0)
    # For each possible value v in [0, L), count occurrences among vals
    for v in range(L):
        eq = (vals == v) & mask
        # atomic add 1 for each match
        tl.atomic_add(counts_ptr + v, tl.sum(eq))

        # Store base = running (exclusive sum)
        tl.store(base_ptr + i, running)
        # Update running += cnt
