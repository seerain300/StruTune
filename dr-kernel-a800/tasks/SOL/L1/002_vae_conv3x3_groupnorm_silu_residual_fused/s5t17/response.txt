I reviewed the failures and realized the main issue: the Triton GroupNorm kernels I used previously did not handle arbitrary H and W correctly. Specifically:
- group_norm_reduce_kernel and group_norm_apply_kernel assumed the entire spatial plane could be indexed via a single linear offset with no mask. When H or W isn’t divisible by certain block sizes (e.g., H=131), that approach breaks. Triton requires masked loads/stores for out-of-bounds elements; otherwise it leads to runtime errors or incorrect results.
- Additionally, prior conv3x3 kernels assumed H and W are contiguous and didn’t guard padding correctly with masks, which could also cause out-of-bounds accesses.

To ensure correctness across all provided workloads, I rewrote:
- The conv3x3 kernels to use explicit 3D grids over (B, C_out, H) and a second grid dimension over W in tiles, with masked loads and vectorized accumulation. This avoids per-pixel scalar loops and correctly handles padding and arbitrary H, W.
- GroupNorm using two Triton kernels:
  1) group_norm_reduce_kernel: grid over (B, num_groups, channels_per_group). For each channel in the group, we loop over tiles of HW using N_TILES as a tl.constexpr and load vectors with masks to compute sum and sum of squares for mean/rstd. This avoids host-side loops and works for any H, W.
  2) group_norm_apply_kernel: grid over (B, num_groups, channels_per_group, N_TILES). Each program handles one tile of the spatial plane for a specific (n, group, channel), applies normalization using precomputed mean/rstd and per-channel scale/bias, and computes SiLU. Masked loads/stores ensure correctness for all shapes.
- Residual addition implemented as a Triton elementwise kernel with masked loads/stores.

This design keeps all math in Triton, eliminates Python loops around Triton, and uses masks to prevent out-of-bounds access, which should fix the runtime errors and produce correct outputs for varied H/W.

import torch
import triton
import triton.language as tl


# Conv3x3: y = x * w, stride=1, padding=1, no bias
# Grid: (B, C_out, H), second grid dimension tiles W with BLOCK_W and uses masks
@triton.jit
def conv3x3_kernel_hw(
    x_ptr,           # *f32, input [B, C_in, H, W]
    w_ptr,           # *f32, weight [C_out, C_in, 3, 3]
    y_ptr,           # *f32, output [B, C_out, H, W]
    B: tl.constexpr, C_in: tl.constexpr, C_out: tl.constexpr,
    H: tl.constexpr, W: tl.constexpr,
    BLOCK_W: tl.constexpr,
):
    n = tl.program_id(0)  # batch
    c_out = tl.program_id(1)  # output channel
    h = tl.program_id(2)  # output row index
    tile_w = tl.program_id(3)  # tile index over width

    w_start = tile_w * BLOCK_W
    offs_w = w_start + tl.arange(0, BLOCK_W)  # vector of width indices for this tile
    mask_w = offs_w < W

    acc = tl.zeros([BLOCK_W], dtype=tl.float32)

    # Accumulate over input channels and 3x3 neighborhood, with masks for padding
    for c_in in range(0, C_in):
        for dh in range(-1, 2):
            h_in = h + dh  # padded
            for dw in range(-1, 2):
                w_in = offs_w + dw  # vector over this tile
                mask_hw = mask_w & (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)

                # Compute input offsets for vectorized load
                # Input index: (((n * C_in) + c_in) * H + h_in) * W + w_in
                x_base = ((n * C_in) + c_in) * H * W
                x_idx = x_base + h_in * W + w_in
                x_vals = tl.load(x_ptr + x_idx, mask=mask_hw, other=0.0)

                # Compute weight scalar for this (c_out, c_in, dh, dw)
                w_off = (((c_out * C_in) + c_in) * 9) + (dh + 1) * 3 + (dw + 1)
                w_val = tl.load(w_ptr + w_off)

                acc += x_vals * w_val

    # Store output vector for this tile
    y_base = ((n * C_out) + c_out) * H * W
    y_idx = y_base + h * W + offs_w
    tl.store(y_ptr + y_idx, acc, mask=mask_w)


# GroupNorm reduction: per-channel (n,group) compute mean and rstd over all HW
# Grid: (B * num_groups, channels_per_group)
@triton.jit
def group_norm_reduce_kernel(
    x_ptr,           # *f32, input [B, C, H, W]
    mean_ptr,        # *f32, output [C]
    rstd_ptr,        # *f32, output [C]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    num_groups: tl.constexpr, eps: tl.constexpr,
    N_TILES: tl.constexpr,  # number of tiles along HW
    BLOCK_HW: tl.constexpr, # tile size per pass
):
    # We will use a global reduction: for each (n, group, c) loop over tiles and accumulate.
    # Each program handles one channel c in one group for all batches. Compute n and group via id.
    pid = tl.program_id(0)
    # map pid -> (batch, group, channel_in_group)
    # Note: pid ranges over B * num_groups * channels_per_group
    ch_per_group = C // num_groups
    groups_per_batch = num_groups
    b = pid // (groups_per_batch * ch_per_group)
    g = (pid // ch_per_group) % groups_per_batch
    c_rel = pid % ch_per_group
    c = g * ch_per_group + c_rel

    # Initialize accumulators
    total_sum = tl.zeros((), dtype=tl.float32)
    total_sumsq = tl.zeros((), dtype=tl.float32)

    for t in range(0, N_TILES):
        offs = t * BLOCK_HW + tl.arange(0, BLOCK_HW)
        mask = offs < (H * W)
        # Compute n, h, w
        h_idx = offs // W
        w_idx = offs % W

        # Base for this (b, c)
        x_base = ((b * C) + c) * H * W
        x_idx = x_base + h_idx * W + w_idx
        x_vals = tl.load(x_ptr + x_idx, mask=mask, other=0.0)

        total_sum += tl.sum(x_vals, axis=0)
        total_sumsq += tl.sum(x_vals * x_vals, axis=0)

    mean = total_sum / (H * W)
    var = total_sumsq / (H * W) - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    tl.store(mean_ptr + c, mean)
    tl.store(rstd_ptr + c, rstd)


# GroupNorm apply: normalize per channel using precomputed mean/rstd, then affine + SiLU
# Grid: (B * num_groups, channels_per_group, N_TILES)
@triton.jit
def group_norm_apply_kernel(
    x_ptr,           # *f32, input [B, C, H, W]
    mean_ptr,        # *f32, [C]
    rstd_ptr,        # *f32, [C]
    gamma_ptr,       # *f32, scale [C] (norm1_weight)
    beta_ptr,        # *f32, bias [C] (norm1_bias)
    y_ptr,           # *f32, output [B, C, H, W]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    num_groups: tl.constexpr,
    N_TILES: tl.constexpr,  # number of tiles along HW
    BLOCK_HW: tl.constexpr, # tile size per pass
):
    pid = tl.program_id(0)
    ch_per_group = C // num_groups
    groups_per_batch = num_groups
    b = pid // (groups_per_batch * ch_per_group)
    g = (pid // ch_per_group) % groups_per_batch
    c_rel = pid % ch_per_group
    c = g * ch_per_group + c_rel

    mean = tl.load(mean_ptr + c)
    rstd = tl.load(rstd_ptr + c)
    gamma = tl.load(gamma_ptr + c)
    beta = tl.load(beta_ptr + c)

    # Loop over tiles
    for t in range(0, N_TILES):
        offs = t * BLOCK_HW + tl.arange(0, BLOCK_HW)
        mask = offs < (H * W)
        h_idx = offs // W
        w_idx = offs % W

        x_base = ((b * C) + c) * H * W
        x_idx = x_base + h_idx * W + w_idx
        x_vals = tl.load(x_ptr + x_idx, mask=mask, other=0.0)

        # Normalize + affine
        norm = (x_vals - mean) * rstd
        # SiLU: x * sigmoid(x) = x / (1 + exp(-x))
        sig = 1.0 / (1.0 + tl.exp(-norm))
        y_vals = norm * sig * gamma + beta

        tl.store(y_ptr + x_idx, y_vals, mask=mask)


# Elementwise residual add: out = a + b, both (B, C, H, W)
@triton.jit
def residual_add_kernel(
    a_ptr, b_ptr, out_ptr,
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
):
    n = tl.program_id(0)
    c = tl.program_id(1)
    hw = tl.program_id(2)

    h = hw // W
    w = hw % W

    idx = (((n * C) + c) * H + h) * W + w
    a_val = tl.load(a_ptr + idx)
    b_val = tl.load(b_ptr + idx)
    tl.store(out_ptr + idx, a_val + b_val)


class ModelNew(torch.nn.Module):
    def __init__(self, num_groups: int = 32, eps: float = 1e-5):
        super().__init__()
        self.num_groups = num_groups
        self.eps = eps

    def forward(self, x: torch.Tensor,
                conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor):
        """
        x: (B, C, H, W), conv weights: (C_out, C_in, 3, 3), norm scales/bias: (C,)
        Returns: (B, C_out2, H, W)
        """
        assert x.is_cuda, "Inputs must be on CUDA for Triton kernels"
        assert x.dim() == 4, "x must be (B, C, H, W)"
        B, C_in, H, W = x.shape

        # Cast to float32 for stable compute
        x_f32 = x.contiguous().to(torch.float32)
        conv1_w_f32 = conv1_weight.contiguous().to(torch.float32)  # (C_out1, C_in, 3, 3)
        conv2_w_f32 = conv2_weight.contiguous().to(torch.float32)  # (C_out2, C_out1, 3, 3)
        norm1_weight_f32 = norm1_weight.contiguous().to(torch.float32)
        norm1_bias_f32 = norm1_bias.contiguous().to(torch.float32)
        norm2_weight_f32 = norm2_weight.contiguous().to(torch.float32)
        norm2_bias_f32 = norm2_bias.contiguous().to(torch.float32)

        # First conv: out1 = conv3x3(x)
        C_out1 = conv1_w_f32.shape[0]
        out1 = torch.empty((B, C_out1, H, W), device=x.device, dtype=torch.float32)

        # Launch conv kernel over tiles of width, with grid (B, C_out1, H, tiles)
        BLOCK_W = 64  # tile size along width; 64 works well for typical sizes
        tiles_w = (W + BLOCK_W - 1) // BLOCK_W
        grid = (B, C_out1, H, tiles_w)
        conv3x3_kernel_hw[grid](
            x_f32, conv1_w_f32, out1,
            B=B, C_in=C_in, C_out=C_out1,
            H=H, W=W,
            BLOCK_W=BLOCK_W,
            num_warps=4,
            num_stages=2,
        )

        # GroupNorm + SiLU for first block
        mean1 = torch.empty(C_out1, device=x.device, dtype=torch.float32)
        rstd1 = torch.empty(C_out1, device=x.device, dtype=torch.float32)
        assert C_out1 % self.num_groups == 0, "C_out1 must be divisible by num_groups"
        ch_per_group = C_out1 // self.num_groups

        N_TILES = (H * W + 1024 - 1) // 1024  # tiles over HW for reduction (BLOCK_HW=1024)
        BLOCK_HW = 1024
        grid_reduce = (B * self.num_groups * ch_per_group,)
        group_norm_reduce_kernel[grid_reduce](
            out1, mean1, rstd1,
            B=B, C=C_out1, H=H, W=W,
            num_groups=self.num_groups, eps=self.eps,
            N_TILES=N_TILES, BLOCK_HW=BLOCK_HW,
            num_warps=4,
            num_stages=2,
        )

        out1_norm = torch.empty_like(out1)
        grid_apply = (B * self.num_groups * ch_per_group,)
        group_norm_apply_kernel[grid_apply](
            out1, mean1, rstd1, norm1_weight_f32, norm1_bias_f32, out1_norm,
            B=B, C=C_out1, H=H, W=W,
            num_groups=self.num_groups,
            N_TILES=N_TILES, BLOCK_HW=BLOCK_HW,
            num_warps=4,
            num_stages=2,
        )

        # Second conv: out2 = conv3x3(out1_norm)
        C_in2 = C_out1
        C_out2 = conv2_w_f32.shape[0]
        out2 = torch.empty((B, C_out2, H, W), device=x.device, dtype=torch.float32)

        tiles_w2 = (W + BLOCK_W - 1) // BLOCK_W
        grid2 = (B, C_out2, H, tiles_w2)
        conv3x3_kernel_hw[grid2](
            out1_norm, conv2_w_f32, out2,
            B=B, C_in=C_in2, C_out=C_out2,
            H=H, W=W,
            BLOCK_W=BLOCK_W,
            num_warps=4,
            num_stages=2,
        )

        # GroupNorm + SiLU for second block
        mean2 = torch.empty(C_out2, device=x.device, dtype=torch.float32)
        rstd2 = torch.empty(C_out2, device=x.device, dtype=torch.float32)
        assert C_out2 % self.num_groups == 0, "C_out2 must be divisible by num_groups"
        ch_per_group2 = C_out2 // self.num_groups

        grid_reduce2 = (B * self.num_groups * ch_per_group2,)
        group_norm_reduce_kernel[grid_reduce2](
            out2, mean2, rstd2,
            B=B, C=C_out2, H=H, W=W,
            num_groups=self.num_groups, eps=self.eps,
            N_TILES=N_TILES, BLOCK_HW=BLOCK_HW,
            num_warps=4,
            num_stages=2,
        )

        out2_norm = torch.empty_like(out2)
        grid_apply2 = (B * self.num_groups * ch_per_group2,)
        group_norm_apply_kernel[grid_apply2](
            out2, mean2, rstd2, norm2_weight_f32, norm2_bias_f32, out2_norm,
            B=B, C=C_out2, H=H, W=W,
            num_groups=self.num_groups,
            N_TILES=N_TILES, BLOCK_HW=BLOCK_HW,
            num_warps=4,
            num_stages=2,
        )

        # Final residual add: out2_norm + x
        out = torch.empty_like(out2_norm)
        grid_add = (B, C_out2, H * W)
        residual_add_kernel[grid_add](
            out2_norm, x_f32, out,
            B=B, C=C_out2, H=H, W=W,
            num_warps=4,
            num_stages=2,
        )

        return out

    beta = tl.load(beta_ptr + c)

            num_warps=4,
            num_stages=2,
        )

        # GroupNorm for second block (PyTorch), then Triton affine + SiLU
        out2_gn = torch.nn.functional.group_norm(out2, self.num_groups, weight=None, bias=None, eps=self.eps)
        out2_norm = torch.nn.functional.group_norm(out2, self.num_groups, norm2_weight_f32, norm2_bias_f32, eps=self.eps)

        grid_aff2 = (B, C_out2, H * W)
        affine_silu_kernel[grid_aff2](
            out2_norm, norm2_weight_f32, norm2_bias_f32, out2_affine,
            B=B, C=C_out2, H=H, W=W,
            num_warps=4,
            num_stages=2,
        )

        # Final residual add: out2_affine + x
        out = torch.empty_like(out2_affine)
        grid_add = (B, C_out2, H * W)
        residual_add_kernel[grid_add](
            out2_affine, x_f32, out,
            B=B, C=C_out2, H=H, W=W,
            num_warps=4,
            num_stages=2,
        )

        return out

    H: tl.constexpr, W: tl.constexpr,
):

    c_out = tl.program_id(1)  # output channel
    h_out = tl.program_id(2)  # output row
    w_out = tl.program_id(3)  # output col

    acc = tl.zeros((), dtype=tl.float32)

    # Accumulate over input channels and 3x3 neighborhood, with masks for padding
    for c_in in range(0, C_in):
        for dh in range(-1, 2):

                w_in = w_out + dw
                in_bounds = (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)
                # Input offset: (((n * C_in) + c_in) * H + h_in) * W + w_in
