import torch
import triton
import triton.language as tl

# Kernel 1: conv2d 3x3 stride=1 padding=1 bias=None
@triton.jit
def conv3x3_stride1_pad1_biasNone(
    x_ptr,              # *f16 or *f32 input: [N, C_in, H, W]
    w_ptr,              # *f16 or *f32 weight: [C_out, C_in, 3, 3]
    y_ptr,              # *f16 or *f32 output: [N, C_out, H, W]
    N, C_in, H, W, C_out,
    # no strides here; we rely on linear indexing
    BLOCK_OC: tl.constexpr,
):
    # grid = (N, ceil_div(C_out, BLOCK_OC))
    n = tl.program_id(0)
    oc_block_id = tl.program_id(1)
    oc_start = oc_block_id * BLOCK_OC
    oc_offsets = oc_start + tl.arange(0, BLOCK_OC)
    oc_mask = oc_offsets < C_out

    # Accumulator for each output channel in the tile
    acc = tl.zeros((BLOCK_OC,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel taps
    for cin in range(C_in):
        for kh in range(3):
            for kw in range(3):
                # For each output pixel, compute input coordinates (ih, iw) with padding=1
                # We will loop over output positions vectorized: flatten oh, ow
                OHW = H * W
                for p in range(OHW):
                    oh = p // W
                    ow = p % W
                    ih = oh + kh - 1
                    iw = ow + kw - 1
                    valid = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
                    # Input linear index: ((n * C_in + cin) * H + ih) * W + iw
                    in_index = ((n * C_in + cin) * H + ih) * W + iw
                    x_val = tl.load(x_ptr + in_index, mask=valid, other=0.0)
                    # Weight linear index: ((oc * C_in + cin) * 9) + (kh * 3 + kw)
                    for j in range(BLOCK_OC):
                        if oc_mask[oc_offsets[j]]:
                            w_index = ((oc_offsets[j] * C_in + cin) * 9) + (kh * 3 + kw)
                            w_val = tl.load(w_ptr + w_index)
                            acc[j] += x_val * w_val

    # Store results to y[n, oc, oh, ow] for all positions; we can store using flattened (oh, ow)
    # To store, we need to write per (oh, ow). Since we accumulated acc for all oc, we compute for each p
    for p in range(OHW):
        oh = p // W
        ow = p % W
        base = n * C_out * OHW + oh * W + ow  # row base for all oc at this (n, oh, ow)
        for j in range(BLOCK_OC):
            if oc_mask[oc_offsets[j]]:
                y_index = base + oc_offsets[j]
                tl.store(y_ptr + y_index, acc[j])


# Kernel 2: GroupNorm with affine, num_groups fixed (here 32), per sample n
@triton.jit
def group_norm_affine(
    y_in_ptr,           # *f16 or *f32 input after conv: [N, C, H, W]
    weight_ptr,         # *f16 or *f32 per-channel scale: [C]
    bias_ptr,           # *f16 or *f32 per-channel bias: [C]
    y_out_ptr,          # *f16 or *f32 output normalized + affine: [N, C, H, W]
    N, C, H, W,         # shapes
    num_groups: tl.constexpr,  # here 32
    eps: tl.constexpr,
):
    # Grid: (N, num_groups)
    n = tl.program_id(0)
    g = tl.program_id(1)
    # Elements per group
    channels_in_group = (C + num_groups - 1) // num_groups * num_groups  # just C in practice (C % num_groups == 0)
    channels_in_group = C // num_groups
    group_size = channels_in_group * H * W

    # First pass: compute sum and sumsq per group
    total_sum = tl.zeros((), dtype=tl.float32)
    total_sumsq = tl.zeros((), dtype=tl.float32)

    # Iterate over channels in the group
    for c_local in range(channels_in_group):
        c = g * channels_in_group + c_local
        base = (n * C + c) * (H * W)
        # Iterate over H*W
        for p in range(H * W):
            lin = base + p
            x = tl.load(y_in_ptr + lin)
            total_sum += x
            total_sumsq += x * x

    mean = total_sum / group_size
    var = total_sumsq / group_size - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for c_local in range(channels_in_group):
        c = g * channels_in_group + c_local
        base = (n * C + c) * (H * W)
        scale = tl.load(weight_ptr + c)
        beta = tl.load(bias_ptr + c)
        for p in range(H * W):
            lin = base + p
            x = tl.load(y_in_ptr + lin)
            y_norm = (x - mean) * inv_std
            y = y_norm * scale + beta
            tl.store(y_out_ptr + lin, y)


# Kernel 3: SiLU activation elementwise: y = x * sigmoid(x)
@triton.jit
def silu(
    x_ptr,              # *f16 or *f32 input
    y_ptr,              # *f16 or *f32 output
    N, C, H, W,
):
    # Grid: (N, C, H, W)
    n = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    w = tl.program_id(3)
    base = (n * C + c) * (H * W) + h * W + w
    x = tl.load(x_ptr + base)
    # sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + base, y)


# Kernel 4: Residual add elementwise: y = y + x
@triton.jit
def add_residual(
    y_ptr,              # *f16 or *f32 output to which residual will be added
    x_ptr,              # *f16 or *f32 residual input
    N, C, H, W,
):
    # Grid: (N, C, H, W)
    n = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    w = tl.program_id(3)
    base = (n * C + c) * (H * W) + h * W + w
    y = tl.load(y_ptr + base)
    x = tl.load(x_ptr + base)
    y = y + x
    tl.store(y_ptr + base, y)

    def __init__(self):
        super().__init__()

    def forward(self, x: torch.Tensor,
                conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor,
                eps: float):
        """
        Fused residual block: Conv3x3 -> GroupNorm -> SiLU -> Conv3x3 -> GroupNorm -> SiLU -> Add
        All heavy ops are Triton kernels; no torch ops are used in forward (except potential .contiguous()).
        """
        assert x.is_cuda and conv1_weight.is_cuda and norm1_weight.is_cuda and norm1_bias.is_cuda \
               and conv2_weight.is_cuda and norm2_weight.is_cuda and norm2_bias.is_cuda, "All tensors must be on CUDA."
        N, C, H, W = x.shape
        # Enforce GroupNorm constraint: C % 32 == 0
        num_groups = 32
        if C % num_groups != 0:
            raise ValueError(f"GroupNorm requires channels divisible by num_groups. Got C={C}, num_groups={num_groups}.")

        # 1) Conv1 (Triton)
        C_in = C  # input channels equal output channels for conv1 due to weight shape (C, C, 3, 3)
        C_out = C
        x0 = x  # save residual
        y1 = torch.empty((N, C_out, H, W), device=x.device, dtype=x.dtype)
        BLOCK_OC = 16  # small tile for output channels; fine for typical C
        grid1 = (N, (C_out + BLOCK_OC - 1) // BLOCK_OC)

            x0, conv1_weight, y1, N, C_in, H, W, C_out,
            BLOCK_OC=BLOCK_OC,
        )

        # 2) GroupNorm1 (Triton)
        y2 = torch.empty_like(y1)
        grid_gn1 = (N, num_groups)
        group_norm_affine[grid_gn1](
            y1, norm1_weight, norm1_bias, y2, N, C, H, W,
            num_groups=32, eps=eps,
        )

        # 3) SiLU1 (Triton)
        y3 = torch.empty_like(y2)
        grid_silu1 = (N, C, H, W)
        silu[grid_silu1](y2, y3, N, C, H, W)

        # 4) Conv2 (Triton)
        y4 = torch.empty((N, C_out, H, W), device=x.device, dtype=x.dtype)
        conv3x3_stride1_pad1_biasNone[grid1](
            y3, conv2_weight, y4, N, C_in, H, W, C_out,
            BLOCK_OC=BLOCK_OC,
        )

        # 5) GroupNorm2 (Triton)
        y5 = torch.empty_like(y4)
        group_norm_affine[grid_gn1](
            y4, norm2_weight, norm2_bias, y5, N, C, H, W,
            num_groups=32, eps=eps,
        )

        # 6) SiLU2 (Triton)
        y6 = torch.empty_like(y5)
        silu[grid_silu1](y5, y6, N, C, H, W)

        # 7) Add residual (Triton)
        y_out = torch.empty_like(y6)
        add_residual[grid_silu1](y6, x0, N, C, H, W)

        return y_out
