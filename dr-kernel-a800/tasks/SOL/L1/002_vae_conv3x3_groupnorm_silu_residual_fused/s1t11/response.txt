import torch
import triton
import triton.language as tl


@triton.jit
def conv3x3_nchw_fp32(
    x_ptr,         # *const float
    w_ptr,         # *const float
    y_ptr,         # *float
    B: tl.constexpr, C_in: tl.constexpr, H_in: tl.constexpr, W_in: tl.constexpr,
    C_out: tl.constexpr, K: tl.constexpr,
    BLOCK_IN: tl.constexpr,
):
    # program id maps to (n, c_out, h_out, w_out)
    pid = tl.program_id(0)
    HW = H_in * W_in
    N = B
    C_OUT = C_out

    n = pid // (C_OUT * HW)
    rem = pid % (C_OUT * HW)
    c_out = rem // HW
    rem2 = rem % HW
    h_out = rem2 // W_in
    w_out = rem2 % W_in

    # initialize accumulator
    acc = 0.0
    # loop over input channels in chunks
    for cin_base in range(0, C_in, BLOCK_IN):
        offs = tl.arange(0, BLOCK_IN)
        cin = cin_base + offs
        mask_c = cin < C_in

        # accumulator for the chunk
        acc_chunk = tl.zeros([BLOCK_IN], dtype=tl.float32)

        # loop over 3x3 kernel window with padding=1
        for kh in range(0, 3):
            for kw in range(0, 3):
                # compute input spatial coordinate
                h_in = h_out + kh - 1
                w_in = w_out + kw - 1

                # mask for valid input coordinates (0 <= h_in < H_in) and (0 <= w_in < W_in)
                mask_hw = (h_in >= 0) & (h_in < H_in) & (w_in >= 0) & (w_in < W_in)

                # compute linear offsets
                # x[n, cin, h_in, w_in] flattened index:
                x_off = n * (C_in * H_in * W_in) + cin * (H_in * W_in) + h_in * W_in + w_in
                x_mask = mask_c & mask_hw

                # load x with mask (masked elements as 0.0)
                x_vals = tl.load(x_ptr + x_off, mask=x_mask, other=0.0)

                # weight scalar w[cin, c_out, kh, kw]
                w_off = cin * (K * K * C_OUT) + kh * (K * C_OUT) + kw * C_OUT + c_out
                w_vals = tl.load(w_ptr + w_off, mask=mask_c, other=0.0)

                # FMA: acc_chunk += x_vals[:, None] * w_vals[None, :]
                acc_chunk += x_vals * w_vals

        # reduce acc_chunk into scalar
        acc += tl.sum(acc_chunk, axis=0)

    # store y[n, c_out, h_out, w_out]
    y_off = n * (C_out * H_in * W_in) + c_out * (H_in * W_in) + h_out * W_in + w_out
    tl.store(y_ptr + y_off, acc)


@triton.jit
def group_norm_affine_fp32(
    x_ptr,          # *const float
    weight_ptr,     # *const float (C,)
    bias_ptr,       # *const float (C,)
    y_ptr,          # *float
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    num_groups: tl.constexpr,
    eps: tl.float32,
    BLOCK_HW: tl.constexpr,
):
    # one program per (n, group)
    pid = tl.program_id(0)
    n = pid // num_groups
    g = pid % num_groups

    # channels per group
    CPG = C // num_groups

    # compute sum and sum of squares over group (channels=g*CPG:(g+1)*CPG) and all H*W
    sum_val = 0.0
    sum_sq = 0.0

    for c_start in range(0, CPG):
        c = g * CPG + c_start
        # iterate over H*W in chunks
        for hw_start in range(0, H * W, BLOCK_HW):
            offs = hw_start + tl.arange(0, BLOCK_HW)
            mask = offs < (H * W)
            h = offs // W
            w = offs % W

            x_off = n * (C * H * W) + c * (H * W) + h * W + w
            x_vals = tl.load(x_ptr + x_off, mask=mask, other=0.0)

            # reduction over chunk
            sum_val += tl.sum(x_vals, axis=0)
            sum_sq += tl.sum(x_vals * x_vals, axis=0)

    m = CPG * (H * W)
    mean = sum_val / m
    var = sum_sq / m - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: normalize and apply affine
    for c_start in range(0, CPG):
        c = g * CPG + c_start
        for hw_start in range(0, H * W, BLOCK_HW):
            offs = hw_start + tl.arange(0, BLOCK_HW)
            mask = offs < (H * W)
            h = offs // W
            w = offs % W

            x_off = n * (C * H * W) + c * (H * W) + h * W + w
            x_vals = tl.load(x_ptr + x_off, mask=mask, other=0.0)

            # affine: y = ((x - mean) * inv_std) * weight[c] + bias[c]
            w_val = tl.load(weight_ptr + c, mask=True, other=1.0)  # weight_ptr is 1D
            b_val = tl.load(bias_ptr + c, mask=True, other=0.0)

            y_vals = ((x_vals - mean) * inv_std) * w_val + b_val

            y_off = n * (C * H * W) + c * (H * W) + h * W + w
            tl.store(y_ptr + y_off, y_vals)


@triton.jit
def silu_kernel_fp32(x_ptr, y_ptr, total: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    # sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + offs, y)


@triton.jit

    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    out = a + b
    tl.store(out_ptr + offs, out)
