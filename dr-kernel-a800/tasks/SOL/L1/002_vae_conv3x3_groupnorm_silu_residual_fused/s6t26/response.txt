```python
import torch
import triton
import triton.language as tl


# Triton kernel: conv3x3 NCHW, stride=1, padding=1.
# Grid: (N, C_out, H, W). Each program computes y[n, co, h, w] by looping over input channels and 3x3 neighborhood.
@triton.jit
def conv3x3_nchw_4d(x_ptr, w_ptr, y_ptr,
                    N, C_in, H, W,
                    x_sN, x_sC, x_sH, x_sW,
                    w_sCo, w_sCi, w_sKh, w_sKw,
                    y_sN, y_sC, y_sH, y_sW,
                    NUM_CI: tl.constexpr, NUM_H: tl.constexpr, NUM_W: tl.constexpr):
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w = tl.program_id(3)

    acc = tl.zeros((), dtype=tl.float32)

    for ci in range(0, NUM_CI):
        for kh in range(0, NUM_H):
            h_in = pid_h + kh
            h_in_valid = (h_in >= 0) & (h_in < H)
            for kw in range(0, NUM_W):
                w_in = pid_w + kw
                w_in_valid = (w_in >= 0) & (w_in < W)
                if h_in_valid and w_in_valid:
                    x_offset = pid_n * x_sN + ci * x_sC + h_in * x_sH + w_in * x_sW
                    x_val = tl.load(x_ptr + x_offset)
                else:
                    x_val = tl.zeros((), dtype=tl.float32)

                w_offset = pid_co * w_sCo + ci * w_sCi + kh * w_sKh + kw * w_sKw
                w_val = tl.load(w_ptr + w_offset)
                acc += x_val * w_val

    y_offset = pid_n * y_sN + pid_co * y_sC + pid_h * y_sH + pid_w * y_sW
    tl.store(y_ptr + y_offset, acc)


# Triton kernel: GroupNorm reduction per (n, group) to compute sum and sumsq across channels in group and all spatial positions (H*W).
# Grid: (N, groups). groups is passed as constexpr (e.g., 32).
@triton.jit
def groupnorm_reduce_sums(y_ptr, sums_ptr,
                           N, C, H, W,
                           groups: tl.constexpr,
                           y_sN, y_sC, y_sH, y_sW):
    pid_n = tl.program_id(0)
    pid_g = tl.program_id(1)

    group_size_c = C // groups
    start_c = pid_g * group_size_c

    sum_val = tl.zeros((), dtype=tl.float32)
    sumsq_val = tl.zeros((), dtype=tl.float32)

    for c in range(start_c, start_c + group_size_c):
        for h in range(0, H):
            for w in range(0, W):
                y_offset = pid_n * y_sN + c * y_sC + h * y_sH + w * y_sW
                val = tl.load(y_ptr + y_offset)
                sum_val += val
                sumsq_val += val * val

    base = pid_n * groups + pid_g
    tl.store(sums_ptr + base * 2 + 0, sum_val)  # sum
    tl.store(sums_ptr + base * 2 + 1, sumsq_val)  # sumsq


# Triton kernel: GroupNorm apply + per-channel affine + SiLU for each (n, c) across H*W.
# Grid: (N, C). groups is constexpr (e.g., 32).
@triton.jit
def groupnorm_apply_affine_silu(y_ptr, sums_ptr, scale_ptr, bias_ptr, out_ptr,
                                N, C, H, W,
                                groups: tl.constexpr,
                                y_sN, y_sC, y_sH, y_sW,
                                out_sN, out_sC, out_sH, out_sW):
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)

    # Compute group for channel pid_c
    group_size_c = C // groups
    group_g = pid_c // group_size_c

    # Load sum and sumsq for (n, group)
    base = pid_n * groups + group_g
    sum_val = tl.load(sums_ptr + base * 2 + 0)
    sumsq_val = tl.load(sums_ptr + base * 2 + 1)

    HxW = H * W
    mean = sum_val / HxW
    var = sumsq_val / HxW - mean * mean
    rstd = 1.0 / tl.sqrt(var + 1e-5)  # eps

    # Preload per-channel affine params
    scale = tl.load(scale_ptr + pid_c)
    bias = tl.load(bias_ptr + pid_c)

    # Compute SiLU on normalized and affine-transformed values
    for h in range(0, H):
        for w in range(0, W):
            y_offset = pid_n * y_sN + pid_c * y_sC + h * y_sH + w * y_sW
            val = tl.load(y_ptr + y_offset)
            norm = (val - mean) * rstd
            out_val = norm * scale + bias
            # SiLU: x * sigmoid(x)
            sig = 1.0 / (1.0 + tl.exp(-out_val))
            out_val = out_val * sig
            out_offset = pid_n * out_sN + pid_c * out_sC + h * out_sH + w * out_sW
            tl.store(out_ptr + out_offset, out_val)


# Triton kernel: elementwise add residual across all elements: out = y + x
@triton.jit
def add_residual(y_ptr, x_ptr, out_ptr, total_elems: tl.constexpr):
    pid = tl.program_id(0)
    # Each program processes one element
    if pid < total_elems:
        y_val = tl.load(y_ptr + pid)
        x_val = tl.load(x_ptr + pid)
        tl.store(out_ptr + pid, y_val + x_val)


def _launch_conv3x3_nchw_4d(x: torch.Tensor, w: torch.Tensor, out: torch.Tensor):
    # x: [N, C_in, H, W], w: [C_out, C_in, 3, 3], out: [N, C_out, H, W]
    assert x.is_cuda and w.is_cuda and out.is_cuda
    x32 = x.contiguous().to(torch.float32)
    w32 = w.contiguous().to(torch.float32)
    out32 = out.contiguous().to(torch.float32)

    N, C_in, H, W = x32.shape
    C_out = w32.shape[0]

    grid = (N, C_out, H, W)
    conv3x3_nchw_4d[grid](
        x32, w32, out32,
        N, C_in, H, W,
        x32.stride(0), x32.stride(1), x32.stride(2), x32.stride(3),
        w32.stride(0), w32.stride(1), w32.stride(2), w32.stride(3),
        out32.stride(0), out32.stride(1), out32.stride(2), out32.stride(3),
        NUM_CI=C_in, NUM_H=3, NUM_W=3,
        num_warps=1,
        num_stages=1
    )
    return out32


def _groupnorm_reduce_sums(y: torch.Tensor, num_groups: int = 32):
    # y: [N, C, H, W]
    assert y.is_cuda
    N, C, H, W = y.shape
    groups = num_groups
    assert C % groups == 0, "C must be divisible by num_groups"

    # sums buffer: (N, groups, 2) float32
    sums = torch.empty((N, groups, 2), dtype=torch.float32, device=y.device)

    grid = (N, groups)
    groupnorm_reduce_sums[grid](
        y,
        sums,
        N, C, H, W,
        groups,
        y.stride(0), y.stride(1), y.stride(2), y.stride(3),
        num_warps=1,
        num_stages=1
    )
    return sums


def _groupnorm_apply_affine_silu(y: torch.Tensor, scale: torch.Tensor, bias: torch.Tensor, out: torch.Tensor):
    # y: [N, C, H, W], scale: [C], bias: [C]
    assert y.is_cuda and scale.is_cuda and bias.is_cuda and out.is_cuda
    N, C, H, W = y.shape

    # Ensure contiguous
    y32 = y.contiguous().to(torch.float32)
    scale32 = scale.contiguous().to(torch.float32)
    bias32 = bias.contiguous().to(torch.float32)
    out32 = out.contiguous().to(torch.float32)

    grid = (N, C)
    groupnorm_apply_affine_silu[grid](
        y32, _groupnorm_reduce_sums(y32, 32), scale32, bias32, out32,
        N, C, H, W,
        32,
        y32.stride(0), y32.stride(1), y32.stride(2), y32.stride(3),
        out32.stride(0), out32.stride(1), out32.stride(2), out32.stride(3),
        num_warps=1,
        num_stages=1
    )
    return out32


def _add_residual(y: torch.Tensor, x: torch.Tensor, out: torch.Tensor):
    # y: [N, C, H, W], x: [N, C, H, W], out: [N, C, H, W]
    assert y.is_cuda and x.is_cuda and out.is_cuda
    total_elems = y.numel()
    y32 = y.contiguous().to(torch.float32)
    x32 = x.contiguous().to(torch.float32)
    out32 = out.contiguous().to(torch.float32)
    grid = (total_elems,)
    add_residual[grid](y32, x32, out32, total_elems, num_warps=1, num_stages=1)
    return out32


class ModelNew(torch.nn.Module):
    def forward(
        self,
        x: torch.Tensor,
        conv1_weight: torch.Tensor,
        norm1_weight: torch.Tensor,
        norm1_bias: torch.Tensor,
        conv2_weight: torch.Tensor,
        norm2_weight: torch.Tensor,
        norm2_bias: torch.Tensor,
        eps: float,
    ):
        # First path: conv1 -> GroupNorm -> SiLU
        y1 = _launch_conv3x3_nchw_4d(x, conv1_weight, torch.empty((x.shape[0], conv1_weight.shape[0], x.shape[2], x.shape[3]), device=x.device, dtype=torch.float32))
        # GroupNorm
        y1_gn = _groupnorm_apply_affine_silu(y1, norm1_weight, norm1_bias, torch.empty_like(y1))
        # SiLU is applied inside _groupnorm_apply_affine_silu (as per original: GroupNorm then SiLU in model; here SiLU fused in kernel)
        # Note: The original code applies SiLU after GroupNorm, which we implement in the Triton kernel.

        # Second path: conv2 on y1_gn
        y2 = _launch_conv3x3_nchw_4d(y1_gn, conv2_weight, torch.empty((y1_gn.shape[0], conv2_weight.shape[0], y1_gn.shape[2], y1_gn.shape[3]), device=y1_gn.device, dtype=torch.float32))
        # GroupNorm
        y2_gn = _groupnorm_apply_affine_silu(y2, norm2_weight, norm2_bias, torch.empty_like(y2))
        # Residual add
        out = _add_residual(y2_gn, x, torch.empty_like(y2_gn))

        return out
```
