import triton
import triton.language as tl


@triton.jit
def conv3x3_no_bias_nchw_nosum(
    x_ptr,            # *f32, input (B, C, H, W)
    w_ptr,            # *f32, weight (C_out, C_in, 3, 3) but here C_in == C == C_out
    out_ptr,          # *f32, output (B, C_out, H_out, W_out)
    B: tl.constexpr,  # int
    C: tl.constexpr,  # int, input channels == output channels
    H: tl.constexpr,  # int
    W: tl.constexpr,  # int
    H_out: tl.constexpr,  # int
    W_out: tl.constexpr,  # int
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    # program ids
    n = tl.program_id(0)
    co = tl.program_id(1)
    h_out = tl.program_id(2)
    w_out = tl.program_id(3)

    # accumulate in fp32
    acc = tl.zeros((), dtype=tl.float32)

    # loop over input channels (C_in == C)
    for ci in range(C):
        # loop over 3x3 neighborhood
        for dh in range(3):
            in_h = h_out * 1 + dh  # since h_out is index, add dh
            for dw in range(3):
                in_w = w_out * 1 + dw

                # pad: if out-of-bounds, set to 0
                in_bounds = (in_h >= 0) & (in_h < H) & (in_w >= 0) & (in_w < W)

                # compute input element address: ((n*C + ci)*H + in_h)*W + in_w
                in_idx = ((n * C + ci) * H + in_h) * W + in_w
                x_val = tl.load(x_ptr + in_idx, mask=in_bounds, other=0.0)

                # load weight for (co, ci, dh, dw): ((co*C + ci)*9 + dh*3 + dw)
                w_offset = (co * C + ci) * 9 + dh * 3 + dw
                w_val = tl.load(w_ptr + w_offset)

                acc += x_val * w_val

    # store output: ((n*C_out + co)*H_out + h_out)*W_out + w_out
    out_idx = ((n * C + co) * H_out + h_out) * W_out + w_out
    tl.store(out_ptr + out_idx, acc)


@triton.jit
def group_norm_triton(
    y_ptr,            # *f32, input tensor (B, C, H_out, W_out), normalized result will be written here
    weight_ptr,       # *f32, (C,) per-channel scale
    bias_ptr,         # *f32, (C,) per-channel bias
    B,                # int
    C,                # int
    H_out,            # int
    W_out,            # int
    num_groups,       # int, 32
    eps,              # float
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    n = tl.program_id(0)
    group = tl.program_id(1)

    channels_per_group = C // num_groups
    group_start_c = group * channels_per_group
    M = H_out * W_out  # per-channel spatial elements
    total = channels_per_group * M  # total elements in this group for sample n

    # First pass: compute per-channel sum and sum of squares across the group
    sum_c = tl.zeros((channels_per_group,), dtype=tl.float32)
    sumsq_c = tl.zeros((channels_per_group,), dtype=tl.float32)

    for ch in range(channels_per_group):
        c = group_start_c + ch
        base = (n * C + c) * M
        for h in range(H_out):
            for w in range(W_out):
                idx = base + h * W_out + w
                x = tl.load(y_ptr + idx)
                sum_c[ch] += x
                sumsq_c[ch] += x * x

    # Mean and variance per channel (per channel variance across group)
    mean = sum_c / float(M)
    var = sumsq_c / float(M) - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for ch in range(channels_per_group):
        c = group_start_c + ch
        scale = tl.load(weight_ptr + c)
        bias = tl.load(bias_ptr + c)
        base = (n * C + c) * M
        for h in range(H_out):
            for w in range(W_out):
                idx = base + h * W_out + w
                x = tl.load(y_ptr + idx)
                y = (x - mean[ch]) * rstd[ch]
                y = y * scale + bias
                tl.store(y_ptr + idx, y)


@triton.jit
def silu_triton(x_ptr, y_ptr, N, num_warps: tl.constexpr, num_stages: tl.constexpr):
    """
    Elementwise SiLU: y = x * sigmoid(x) over flat tensors of length N.
    """
    pid = tl.program_id(0)
    offsets = pid * 1024 + tl.arange(0, 1024)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    s = 1.0 / (1.0 + tl.exp(-x))  # sigmoid
    y = x * s
    tl.store(y_ptr + offsets, y, mask=mask)


@triton.jit
def add_residual_triton(x_ptr, y_ptr, N, num_warps: tl.constexpr, num_stages: tl.constexpr):
    """
    Elementwise add: y = y + x over flat tensors of length N.
    """
    pid = tl.program_id(0)
    offsets = pid * 1024 + tl.arange(0, 1024)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    y = y + x
    tl.store(y_ptr + offsets, y, mask=mask)
