```python
        )

        # conv2: F.conv2d (no bias), stride=1, padding=1
        y2_pre = torch.nn.functional.conv2d(y1_grouped, conv2_weight, bias=None, stride=1, padding=1)

        # Triton GroupNorm + affine + SiLU for conv2
        y2_grouped = torch.empty_like(y2_pre)
        grid_gn2 = (B, self.num_groups)
        group_norm_affine_silu_kernel[grid_gn2](
            y2_pre, norm2_weight, norm2_bias, y2_grouped,
            B, C, y2_pre.shape[2], y2_pre.shape[3], self.num_groups, self.eps,
            y2_pre.stride(0), y2_pre.stride(1), y2_pre.stride(2), y2_pre.stride(3),
            y2_grouped.stride(0), y2_grouped.stride(1), y2_grouped.stride(2), y2_grouped.stride(3),
            BLOCK=1024,
        )

        # Final residual add: out = y2_grouped + x
        out = torch.empty_like(x)
        N = B * C * y2_grouped.shape[2] * y2_grouped.shape[3]
        grid_add = (triton.cdiv(N, 1024),)
        add_residual_kernel[grid_add](y2_grouped, x, out, N, BLOCK=1024)
        return out
```

Notes

- This implementation ensures Triton kernels are invoked: GroupNorm + affine + SiLU twice (after each conv) and a residual add at the end.
- Convolutions are handled by PyTorch’s F.conv2d to avoid indexing pitfalls and ensure correctness. The original code uses no bias, stride=1, padding=1, so this matches exactly.
- The GroupNorm kernel assumes num_groups=32 and C divisible by 32 (as in the original). It performs per-(batch, group) reductions, then normalizes and applies per-channel scale and bias, followed by SiLU activation.
- The residual addition kernel is elementwise and runs over all elements, ensuring out = y2_grouped + x.

This should eliminate the prior runtime errors and achieve correctness across the evaluation workloads. If performance tuning is later required, we can revisit a fused Triton conv; for now, correctness and robust Triton usage are prioritized.
