import torch
import triton
import triton.language as tl


@triton.jit
def silu_kernel(inp_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    x = tl.load(inp_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(out_ptr + offsets, y, mask=mask)


@triton.jit
def add_residual_kernel(inp_ptr, res_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    y = tl.load(inp_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    x = tl.load(res_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    z = y + x
    tl.store(out_ptr + offsets, z, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self, eps: float = 1e-5, num_groups: int = 32):
        super().__init__()
        self.eps = eps
        self.num_groups = num_groups

    def forward(self, x: torch.Tensor,
                conv1_weight: torch.Tensor,
                norm1_weight: torch.Tensor,
                norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor,
                norm2_weight: torch.Tensor,
                norm2_bias: torch.Tensor,
                eps: float):
        # Compute first conv
        conv1 = torch.nn.functional.conv2d(x, conv1_weight, bias=None, stride=1, padding=1)

        # First GroupNorm
        group1 = torch.nn.functional.group_norm(conv1, num_groups=self.num_groups,
                                                weight=norm1_weight, bias=norm1_bias, eps=eps)

        # SiLU1 using PyTorch
        silu1 = group1 * torch.sigmoid(group1)

        # Second conv
        conv2 = torch.nn.functional.conv2d(silu1, conv2_weight, bias=None, stride=1, padding=1)

        # Second GroupNorm
        group2 = torch.nn.functional.group_norm(conv2, num_groups=self.num_groups,
                                                weight=norm2_weight, bias=norm2_bias, eps=eps)

        # SiLU2 using PyTorch
        silu2 = group2 * torch.sigmoid(group2)

        # Residual add using Triton
        B, C, H, W = x.shape
        N = silu2.numel()
        out = torch.empty_like(silu2)

        # Ensure float32 and contiguous for Triton
        silu2_f = silu2.contiguous()
        x_f = x.contiguous()
        grid = (triton.cdiv(N, 1024),)
        add_residual_kernel[grid](silu2_f, x_f, out, N, BLOCK=1024)

        return out
