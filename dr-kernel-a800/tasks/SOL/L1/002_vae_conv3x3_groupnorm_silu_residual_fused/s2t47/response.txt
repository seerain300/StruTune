import torch
import triton
import triton.language as tl


# Triton kernel: Conv3x3 NCHW, stride=1, padding=1, no bias
# Computes out[n, co, h, w] for a single (n, co, h), vectorizing across w.
# We pass H and W as constexpr to avoid dynamic loops in Triton.
@triton.jit
def conv3x3_nchw_kernel_row(
    x_ptr, w_ptr, out_ptr,
    B, C, C_OUT, H, W,
    BLOCK_W: tl.constexpr,  # number of w positions to compute per program; here 1, but we can generalize
):
    # Grid: (B, C_OUT, H, 1)
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_h = tl.program_id(2)
    # We compute a single w position (BLOCK_W=1); can be extended to a vector if needed.
    h = pid_h
    w = 0  # for this program, we handle a single w; other w handled by separate programs.

    # Accumulator
    acc = tl.float32(0.0)

    # Loop over input channels
    for ci in range(0, C):
        # Accumulate over 3x3 neighborhood
        for dh in range(-1, 2):
            for dw in range(-1, 2):
                ih = h + dh
                iw = w + dw
                # Bounds check for input read
                in_bounds = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
                # Input pointer for x[n, ci, ih, iw]
                x_idx = ((pid_n * C + ci) * H + ih) * W + iw
                x_val = tl.load(x_ptr + x_idx, mask=in_bounds, other=0.0)
                # Weight pointer for w[co, ci, 1+dh, 1+dw]; padding=1 so index is 1+dh, 1+dw
                w_idx = (pid_co * C + ci) * 9 + (1 + dh) * 3 + (1 + dw)
                w_val = tl.load(w_ptr + w_idx)
                acc += x_val * w_val

    # Store output: out[n, co, h, w]
    out_idx = ((pid_n * C_OUT + pid_co) * H + h) * W + w
    tl.store(out_ptr + out_idx, acc)


# Triton kernel: compute sum and sum of squares per (n, group) across all channels in the group and all H*W elements
# Assumes H and W are constexpr known at launch; loops are static.
@triton.jit
def groupnorm_sums_kernel(
    x_ptr, sums_ptr, sumsq_ptr,
    B, C, num_groups,
    C_PER_GROUP: tl.constexpr,  # must be C/num_groups (here 2)
    H: tl.constexpr, W: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_idx = n * num_groups + g

    start_ci = g * C_PER_GROUP
    s = tl.float32(0.0)
    s2 = tl.float32(0.0)

    # Loop over channels in the group and spatial positions statically
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        for h in range(0, H):
            for w in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w
                x_val = tl.load(x_ptr + idx)
                s += x_val
                s2 += x_val * x_val

    tl.store(sums_ptr + out_idx, s)
    tl.store(sumsq_ptr + out_idx, s2)


# Triton kernel: compute inverse std per (n, group) from sums and sumsq
@triton.jit
def groupnorm_invstd_kernel(
    sums_ptr, sumsq_ptr, invstd_ptr,
    B, C, num_groups,
    C_PER_GROUP: tl.constexpr,
    H: tl.constexpr, W: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_idx = n * num_groups + g

    s = tl.load(sums_ptr + out_idx)
    s2 = tl.load(sumsq_ptr + out_idx)
    group_size = C_PER_GROUP * H * W
    mean = s / group_size
    var = s2 / group_size - mean * mean
    invstd = 1.0 / tl.sqrt(var + 1e-5)  # epsilon for numerical stability
    tl.store(invstd_ptr + out_idx, invstd)


# Triton kernel: apply normalization + affine (norm_weight, norm_bias) + SiLU per (n, group)
@triton.jit
def groupnorm_silu_apply_kernel(
    x_ptr, norm_w_ptr, norm_b_ptr, out_ptr, invstd_ptr,
    B, C, num_groups,
    C_PER_GROUP: tl.constexpr,
    H: tl.constexpr, W: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_idx = n * num_groups + g
    invstd = tl.load(invstd_ptr + out_idx)

    start_ci = g * C_PER_GROUP
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        w = tl.load(norm_w_ptr + ci)   # GroupNorm scale
        b = tl.load(norm_b_ptr + ci)   # GroupNorm bias
        for h in range(0, H):
            for w2 in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w2
                x_val = tl.load(x_ptr + idx)
                norm = (x_val - 0.0) * invstd * w + b  # GroupNorm affine
                # SiLU: x * sigmoid(x)
                silu = norm * (1.0 / (1.0 + tl.exp(-norm)))
                tl.store(out_ptr + idx, silu)


def _assert_divisible(dividend: int, divisor: int):
    if dividend % divisor != 0:
        raise ValueError(f"{dividend} is not divisible by {divisor}")


def _to_dtype_for_triton(x: torch.Tensor, dtype: torch.dtype) -> torch.Tensor:
    if x.dtype != dtype:
        return x.to(dtype)
    return x


def _conv3x3_triton_nchw(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """
    Triton-based Conv3x3 (stride=1, padding=1), NCHW layout, no bias.
    Computes out[b, co, h, w] for all b, co, h, w.
    """
    B, C, H, W = x.shape
    _, Co, _, _ = weight.shape
    assert Co == C, "Weight out_channels must equal input channels for this implementation"

    # Prepare pointers and outputs in float32 for numerical stability
    x_f32 = _to_dtype_for_triton(x, torch.float32).contiguous()
    w_f32 = _to_dtype_for_triton(weight, torch.float32).contiguous()
    out = torch.empty((B, Co, H, W), dtype=torch.float32, device=x.device)

    # Launch Triton kernel over grid (B, Co, H, 1); each program handles one (b, co, h) and one w
    grid = (B, Co, H, 1)
    # Note: passing H and W as constexpr-like via kernel signature; Triton expects compile-time constants for loops.
    conv3x3_nchw_kernel_row[grid](
        x_f32, w_f32, out,
        B, C, Co, H, W,
        BLOCK_W=1,
    )
    return out


def _groupnorm_silu_triton(x: torch.Tensor, num_groups: int, weight: torch.Tensor, bias: torch.Tensor, eps: float) -> torch.Tensor:
    """
    Triton implementation of GroupNorm + SiLU per sample and per group:
    1) Compute sums and sumsq per (n, group).
    2) Compute invstd per (n, group).
    3) Apply normalization + affine + SiLU and write to output.
    """
    B, C, H, W = x.shape
    assert C % num_groups == 0, f"C ({C}) must be divisible by num_groups ({num_groups})"
    C_PER_GROUP = C // num_groups

    # Compute sums and sumsq
    sums = torch.empty((B * num_groups,), dtype=torch.float32, device=x.device)
    sumsq = torch.empty((B * num_groups,), dtype=torch.float32, device=x.device)

    groupnorm_sums_kernel[(B * num_groups,)](
        x, sums, sumsq,
        B, C, num_groups,
        C_PER_GROUP=C_PER_GROUP,
        H=H, W=W,
    )

    # Compute invstd
    invstd = torch.empty((B * num_groups,), dtype=torch.float32, device=x.device)
    groupnorm_invstd_kernel[(B * num_groups,)](
        sums, sumsq, invstd,
        B, C, num_groups,
        C_PER_GROUP=C_PER_GROUP,
        H=H, W=W,
    )

    # Apply normalization + affine + SiLU and store
    # We normalize in-place into out (same shape as x)
    out = torch.empty_like(x, dtype=torch.float32, device=x.device)
    groupnorm_silu_apply_kernel[(B * num_groups,)](
        x, weight, bias, out, invstd,
        B, C, num_groups,
        C_PER_GROUP=C_PER_GROUP,
        H=H, W=W,
    )
    return out


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x: torch.Tensor, conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor, conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor, eps: float):
        """
        Fused residual block:
        Conv3x3 -> GroupNorm -> SiLU
        Conv3x3 -> GroupNorm -> SiLU
        Add residual

        All heavy work is performed inside Triton kernels.
        """
        # Validate C and num_groups
        _assert_divisible(x.shape[1], 32)  # original code uses num_groups=32

        # Stage 1: Conv1 -> GroupNorm -> SiLU
        y1 = _conv3x3_triton_nchw(x, conv1_weight)  # Triton conv
        y1 = _groupnorm_silu_triton(y1, 32, norm1_weight, norm1_bias, eps)  # Triton GroupNorm + SiLU

        # Stage 2: Conv2 -> GroupNorm -> SiLU
        y2 = _conv3x3_triton_nchw(y1, conv2_weight)  # Triton conv
        y2 = _groupnorm_silu_triton(y2, 32, norm2_weight, norm2_bias, eps)  # Triton GroupNorm + SiLU

        # Residual add
        out = y2 + x
        return out
