import torch
import triton
import triton.language as tl


def _assert_divisible(dividend: int, divisor: int):
    if dividend % divisor != 0:
        raise ValueError(f"{dividend} must be divisible by {divisor}")


# Triton kernel: Conv3x3 NCHW, stride=1, padding=1, no bias
# Computes out[n, co, h, w] = sum_{ci=0..C-1} sum_{dh=-1..1} sum_{dw=-1..1} x[n, ci, h+dh, w+dw] * w[co, ci, 1+dh, 1+dw]
@triton.jit
def conv3x3_nchw_kernel(
    x_ptr, w_ptr, out_ptr,
    B, C, H, W, C_OUT,
    # constexprs below are compile-time constants for Triton
    KH: tl.constexpr, KW: tl.constexpr,  # 3x3
    PAD_H: tl.constexpr, PAD_W: tl.constexpr,  # 1
    BLOCK_W: tl.constexpr,                 # tile size along W, e.g., 64
):
    # Grid: (B, C_OUT, H, ceil_div(W, BLOCK_W))
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w_blk = tl.program_id(3)

    h = pid_h
    w_start = pid_w_blk * BLOCK_W
    w_offsets = w_start + tl.arange(0, BLOCK_W)
    mask_w = w_offsets < W

    # Accumulator for output row
    acc = tl.zeros([BLOCK_W], dtype=tl.float32)

    # Loop over input channels (dynamic range; Triton supports python loops)
    for ci in range(0, C):
        # Accumulate over 3x3 neighborhood with padding=1
        for dh in range(-1, 2):  # 3 rows: -1,0,1
            h_tmp = h + dh + 1  # since original image has padded zeros, we index at 1+dh
            h_valid = (h_tmp >= 0) and (h_tmp < H)
            for dw in range(-1, 2):  # 3 cols: -1,0,1
                w_tmp = w_offsets + dw  # vectorized across BLOCK_W
                w_valid = (w_tmp >= 0) & (w_tmp < W)
                valid = mask_w & h_valid & w_valid

                # Input pointer: ((n*C + ci) * H + h_tmp) * W + w_tmp
                x_idx = ((pid_n * C + ci) * H + h_tmp) * W + w_tmp
                x_val = tl.load(x_ptr + x_idx, mask=valid, other=0.0)

                # Weight pointer: ((co*C_OUT + ci) * KH + (1+dh)) * KW + (1+dw)
                # KH=3, KW=3, PAD_H=PAD_W=1 (stride=1)
                w_idx = ((pid_co * C + ci) * KH + (1 + dh)) * KW + (1 + dw)
                w_val = tl.load(w_ptr + w_idx)  # scalar

                acc += x_val * w_val

    # Store results for this block of W
    out_idx = ((pid_n * C_OUT + pid_co) * H + h) * W + w_offsets
    tl.store(out_ptr + out_idx, acc, mask=mask_w)


# Triton: compute sum and sum of squares per (n, group) across all channels in group and all H*W elements
@triton.jit
def groupnorm_sums_kernel(
    x_ptr, sums_ptr, sumsq_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,            # 64 // 32 = 2
):
    # One program per (n, group)
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    group_start_ci = g * C_PER_GROUP

    s = tl.float32(0.0)
    s2 = tl.float32(0.0)

    # Accumulate over channels in the group and all spatial positions
    for ci in range(0, C_PER_GROUP):
        ci_abs = group_start_ci + ci
        for h in range(0, H):
            for w in range(0, W):
                idx = ((n * C + ci_abs) * H + h) * W + w
                x_val = tl.load(x_ptr + idx)
                s += x_val
                s2 += x_val * x_val

    tl.store(sums_ptr + pid, s)
    tl.store(sumsq_ptr + pid, s2)


# Triton: compute inverse std per (n, group) from sums and sumsq
@triton.jit
def groupnorm_invstd_kernel(
    sums_ptr, sumsq_ptr, invstd_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_idx = n * num_groups + g

    s = tl.load(sums_ptr + out_idx)
    s2 = tl.load(sumsq_ptr + out_idx)
    group_size = C_PER_GROUP * H * W
    mean = s / group_size
    var = s2 / group_size - mean * mean
    invstd = 1.0 / tl.sqrt(var + 1e-5)  # epsilon for stability
    tl.store(invstd_ptr + out_idx, invstd)


# Triton: normalize + affine + SiLU per (n, group), using precomputed invstd
@triton.jit
def groupnorm_silu_apply_kernel(
    x_ptr, norm_w_ptr, norm_b_ptr, out_ptr, invstd_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_idx = n * num_groups + g
    invstd = tl.load(invstd_ptr + out_idx)

    group_start_ci = g * C_PER_GROUP
    for ci in range(0, C_PER_GROUP):
        ci_abs = group_start_ci + ci
        w = tl.load(norm_w_ptr + ci_abs)
        b = tl.load(norm_b_ptr + ci_abs)
        for h in range(0, H):
            for w_idx in range(0, W):
                idx = ((n * C + ci_abs) * H + h) * W + w_idx
                x_val = tl.load(x_ptr + idx)
                # normalize
                y = (x_val - x_val * 0.0)  # dummy to keep scope; we recompute below
                y = (x_val * invstd)  # gamma=1 handled via w=1, but we use norm_w
                # SiLU: y * sigmoid(y) = y / (1 + exp(-y))
                sig = 1.0 / (1.0 + tl.exp(-y))
                y = y * sig + b  # affine bias
                tl.store(out_ptr + idx, y)


# Forward function using Triton kernels
@torch.no_grad()
def run_t(
    x: torch.Tensor,
    conv1_weight: torch.Tensor,
    norm1_weight: torch.Tensor,
    norm1_bias: torch.Tensor,
    conv2_weight: torch.Tensor,
    norm2_weight: torch.Tensor,
    norm2_bias: torch.Tensor,
    eps: float,
):
    """
    Fused residual block: Conv3x3 -> GroupNorm -> SiLU -> Conv3x3 -> GroupNorm -> SiLU -> Add
    Triton-only implementation for convs and GroupNorm+SiLU. Residual addition done in PyTorch.
    """
    # Assumptions based on original code:
    # - C = 64, num_groups = 32
    # - x is NCHW, contiguous, float32
    B, C, H, W = x.shape
    C_OUT = conv1_weight.shape[0]  # for both convs, weight is (C, C, 3, 3)
    if C != 64:
        raise ValueError(f"Expected C=64, got C={C}")
    if conv1_weight.shape != (C_OUT, C, 3, 3) or conv2_weight.shape != (C_OUT, C, 3, 3):
        raise ValueError("conv weights must be of shape (C_OUT, C, 3, 3)")
    if norm1_weight.shape != (C,) or norm1_bias.shape != (C,) or norm2_weight.shape != (C,) or norm2_bias.shape != (C,):
        raise ValueError("GroupNorm weights/biases must be of shape (C,)")

    # Ensure contiguity and dtype
    x = x.contiguous()
    x32 = x.to(torch.float32)

    # Allocate outputs for convs
    # conv1 output
    out1 = torch.empty((B, C_OUT, H, W), device=x.device, dtype=torch.float32)
    # conv2 output
    out2 = torch.empty((B, C_OUT, H, W), device=x.device, dtype=torch.float32)

    # Launch conv1 Triton kernel
    BLOCK_W = 64  # tile size along W; works for W=64/128/256/293/1024 with masking
    grid1 = (B, C_OUT, H, triton.cdiv(W, BLOCK_W))
    conv3x3_nchw_kernel[grid1](
        x32, conv1_weight.contiguous().to(torch.float32), out1,
        B, C, H, W, C_OUT,
        KH=3, KW=3, PAD_H=1, PAD_W=1,
        BLOCK_W=BLOCK_W,
    )

    # GroupNorm + SiLU for out1
    num_groups = 32
    _assert_divisible(C, num_groups)
    C_PER_GROUP = C // num_groups

    out1_gn = torch.empty_like(out1)  # we will write GroupNorm result into this
    # Compute sums and sumsq per (n, group)
    sums = torch.empty(B * num_groups, device=x.device, dtype=torch.float32)
    sumsq = torch.empty(B * num_groups, device=x.device, dtype=torch.float32)
    grid_sums = (B * num_groups,)
    groupnorm_sums_kernel[grid_sums](
        out1, sums, sumsq,
        B, C, H, W, num_groups,
        C_PER_GROUP=C_PER_GROUP,
    )

    # Compute invstd per (n, group)
    invstd = torch.empty(B * num_groups, device=x.device, dtype=torch.float32)
    grid_inv = (B * num_groups,)
    groupnorm_invstd_kernel[grid_inv](
        sums, sumsq, invstd,
        B, C, H, W, num_groups,
        C_PER_GROUP=C_PER_GROUP,
    )

    # Apply normalization + affine + SiLU
    grid_apply = (B * num_groups,)
    groupnorm_silu_apply_kernel[grid_apply](
        out1, norm1_weight.contiguous().to(torch.float32), norm1_bias.contiguous().to(torch.float32),
        out1_gn, invstd,
        B, C, H, W, num_groups,
        C_PER_GROUP=C_PER_GROUP,
    )

    # conv2 Triton kernel: conv1_gn is input for conv2
    grid2 = (B, C_OUT, H, triton.cdiv(W, BLOCK_W))
    conv3x3_nchw_kernel[grid2](
        out1_gn, conv2_weight.contiguous().to(torch.float32), out2,
        B, C_OUT, H, W, C_OUT,
        KH=3, KW=3, PAD_H=1, PAD_W=1,
        BLOCK_W=BLOCK_W,
    )

    # GroupNorm + SiLU for out2
    sums2 = torch.empty(B * num_groups, device=x.device, dtype=torch.float32)
    sumsq2 = torch.empty(B * num_groups, device=x.device, dtype=torch.float32)
    groupnorm_sums_kernel[(B * num_groups,)](
        out2, sums2, sumsq2,
        B, C, H, W, num_groups,
        C_PER_GROUP=C_PER_GROUP,
    )

    invstd2 = torch.empty(B * num_groups, device=x.device, dtype=torch.float32)
    groupnorm_invstd_kernel[(B * num_groups,)](
        sums2, sumsq2, invstd2,
        B, C, H, W, num_groups,
        C_PER_GROUP=C_PER_GROUP,
    )

    out2_gn = torch.empty_like(out2)
    groupnorm_silu_apply_kernel[(B * num_groups,)](
        out2, norm2_weight.contiguous().to(torch.float32), norm2_bias.contiguous().to(torch.float32),
        out2_gn, invstd2,
        B, C, H, W, num_groups,
        C_PER_GROUP=C_PER_GROUP,
    )

    # Residual add (PyTorch)
    out = out2_gn + x32

    # Cast back to original dtype if needed (here x is float32 by default)
    if x.dtype != torch.float32:
        out = out.to(x.dtype)

    return out


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # Expect input and six tensors as in the original run function.
        # We implement everything in Triton: convs and GroupNorm+SiLU, residual addition in PyTorch.
        return run_t(*args)

# Triton: compute sum and sum of squares per (n, group) across all channels in group and all H*W elements
@triton.jit
def groupnorm_sums_kernel(
    x_ptr, sums_ptr, sumsq_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,  # 64 // 32 = 2
):
    # One program per (n, group)
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    group_start_ci = g * C_PER_GROUP

    s = tl.float32(0.0)
    s2 = tl.float32(0.0)

    # Accumulate over channels in the group and all spatial positions
    for ci in range(0, C_PER_GROUP):
        ci_abs = group_start_ci + ci
        for h in range(0, H):

                s += x_val
                s2 += x_val * x_val

    out_idx = n * num_groups + g
    tl.store(sums_ptr + out_idx, s)
    tl.store(sumsq_ptr + out_idx, s2)
