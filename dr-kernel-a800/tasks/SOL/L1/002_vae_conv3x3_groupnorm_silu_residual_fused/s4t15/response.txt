    pid_co = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w = tl.program_id(3)

    # bounds check
    if (pid_b >= B) or (pid_co >= C_out) or (pid_h >= H_out) or (pid_w >= W_out):
