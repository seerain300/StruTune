import torch
import triton
import triton.language as tl


# Triton kernel: Conv3x3 (stride=1, padding=1, no bias), NCHW layout, float32 compute
@triton.jit
def conv3x3_nchw_fp32(x_ptr, w_ptr, y_ptr,
                      B, C_in, H_in, W_in, C_out, H_out, W_out,
                      BLOCK_IN: tl.constexpr):
    # program ids: (n, c_out, h_out, w_out)
    pid = tl.program_id(axis=0)
    hw_total = H_out * W_out
    n = pid // (C_out * hw_total)
    rem = pid % (C_out * hw_total)
    c_out = rem // hw_total
    rem2 = rem % hw_total
    h_out = rem2 // W_out
    w_out = rem2 % W_out

    # Accumulate output
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over input channels in chunks
    for c_start in range(0, C_in, BLOCK_IN):
        offs_c = c_start + tl.arange(0, BLOCK_IN)
        mask_c = offs_c < C_in

        # Accumulate over 3x3 neighborhood with padding=1
        for kh in range(0, 3):
            h_in = h_out * 1 + kh - 1  # since stride=1, padding=1: h_out = (h_in + 2*pad - d)/stride + 1 -> h_out = h_in for pad=1
            for kw in range(0, 3):
                w_in = w_out * 1 + kw - 1

                # valid mask for input coordinates
                valid_hw = (h_in >= 0) & (h_in < H_in) & (w_in >= 0) & (w_in < W_in)

                # Build input pointers for this (kh, kw)
                # Pointer: x[n, offs_c, h_in, w_in]
                # Use masks for channel and spatial validity
                x_ptrs = x_ptr + n * (C_in * H_in * W_in) + offs_c[:, None] * (H_in * W_in) + h_in * W_in + w_in
                x_mask = mask_c[:, None] & valid_hw
                x_vals = tl.load(x_ptrs, mask=x_mask, other=0.0)  # [BLOCK_IN, 1]
                x_vals = tl.reshape(x_vals, (BLOCK_IN,))
                w_ptrs = w_ptr + c_out * (C_in * 3 * 3) + offs_c * (3 * 3) + kh * 3 + kw
                w_mask = mask_c
                w_vals = tl.load(w_ptrs, mask=w_mask, other=0.0)  # [BLOCK_IN]

                # FMA: acc += sum_c (x[n, c, h_in, w_in] * w[c_out, c, kh, kw])
                acc += tl.sum(x_vals * w_vals, axis=0)

    # Store output: y[n, c_out, h_out, w_out]
    y_ptr_offset = y_ptr + n * (C_out * H_out * W_out) + c_out * (H_out * W_out) + h_out * W_out + w_out
    tl.store(y_ptr_offset, acc)


# Triton kernel: GroupNorm (num_groups fixed, affine) per (n, group), NCHW layout
@triton.jit
def groupnorm_affine_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                             B, C, H, W, NUM_GROUPS, EPS,
                             BLOCK_HW: tl.constexpr):
    # program ids: axis 0 over batch, axis 1 over groups
    pid_n = tl.program_id(axis=0)
    pid_g = tl.program_id(axis=1)

    channels_per_group = C // NUM_GROUPS
    c_start = pid_g * channels_per_group

    # Pass 1: compute sum and sum of squares over group channels and all H*W
    sum_val = tl.zeros((), dtype=tl.float32)
    sumsq_val = tl.zeros((), dtype=tl.float32)
    for c in range(c_start, c_start + channels_per_group):
        # Iterate over spatial positions in chunks
        for start in range(0, H * W, BLOCK_HW):
            hw_offsets = start + tl.arange(0, BLOCK_HW)
            mask_hw = hw_offsets < (H * W)
            # Map hw_offsets to (h, w)
            h = hw_offsets // W
            w = hw_offsets % W

            x_ptrs = x_ptr + pid_n * (C * H * W) + c * (H * W) + h * W + w
            x_vals = tl.load(x_ptrs, mask=mask_hw, other=0.0)  # [BLOCK_HW]
            sum_val += tl.sum(x_vals, axis=0)
            sumsq_val += tl.sum(x_vals * x_vals, axis=0)

    hw_total = H * W
    mean = sum_val / (channels_per_group * hw_total)
    var = sumsq_val / (channels_per_group * hw_total) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Pass 2: normalize and apply affine, write to y
    for c in range(c_start, c_start + channels_per_group):
        for start in range(0, H * W, BLOCK_HW):
            hw_offsets = start + tl.arange(0, BLOCK_HW)
            mask_hw = hw_offsets < (H * W)
            h = hw_offsets // W
            w = hw_offsets % W

            x_ptrs = x_ptr + pid_n * (C * H * W) + c * (H * W) + h * W + w
            x_vals = tl.load(x_ptrs, mask=mask_hw, other=0.0)  # [BLOCK_HW]

            norm_vals = (x_vals - mean) * inv_std
            # affine: y = norm * weight[c] + bias[c]
            w_scale = tl.load(weight_ptr + c)
            b_bias = tl.load(bias_ptr + c)
            y_vals = norm_vals * w_scale + b_bias

            y_ptrs = y_ptr + pid_n * (C * H * W) + c * (H * W) + h * W + w
            tl.store(y_ptrs, y_vals, mask=mask_hw)


# Triton kernel: SiLU elementwise over flattened tensor (fp32)
@triton.jit
def silu_kernel(x_ptr, y_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    x_vals = tl.load(x_ptr + offs, mask=mask, other=0.0)
    # sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x_vals))
    y_vals = x_vals * sig
    tl.store(y_ptr + offs, y_vals, mask=mask)


# Triton kernel: elementwise add: y = x1 + x2 (fp32), flat 1D
@triton.jit
def add_residual_fp32(x1_ptr, x2_ptr, y_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    a = tl.load(x1_ptr + offs, mask=mask, other=0.0)
    b = tl.load(x2_ptr + offs, mask=mask, other=0.0)
    tl.store(y_ptr + offs, a + b, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self, conv1_weight, norm1_weight, norm1_bias, conv2_weight, norm2_weight, norm2_bias, eps):
        super().__init__()
        self.conv1_weight = torch.nn.Parameter(conv1_weight)  # not used for autograd, but stored for forward
        self.norm1_weight = torch.nn.Parameter(norm1_weight)
        self.norm1_bias = torch.nn.Parameter(norm1_bias)
        self.conv2_weight = torch.nn.Parameter(conv2_weight)
        self.norm2_weight = torch.nn.Parameter(norm2_weight)
        self.norm2_bias = torch.nn.Parameter(norm2_bias)
        self.eps = eps
        self.num_groups = 32

    def forward(self, x: torch.Tensor):
        # x: (B, C, H, W), assume float32 input
        B, C, H, W = x.shape
        device = x.device

        # Ensure float32 and contiguous
        x0 = x.contiguous()
        dtype = x0.dtype

        # 1) First conv: y1 = conv3x3(x)
        # Prepare input and weight tensors
        x_for_conv1 = x0.contiguous().to(torch.float32)
        w1 = self.conv1_weight.contiguous().to(torch.float32)  # (C, C, 3, 3)
        # Output tensor: (B, C, H, W)
        y1 = torch.empty((B, C, H, W), device=device, dtype=torch.float32)

        C_in = C
        C_out = C  # output channels equal to input channels for this conv
        H_out = H
        W_out = W
        # Launch conv kernel: one program per output element
        total_out = B * C_out * H_out * W_out
        conv3x3_nchw_fp32[(total_out,)](
            x_for_conv1, w1, y1,
            B, C_in, H, W, C_out, H_out, W_out,
            BLOCK_IN=32,  # loop over input channels in chunks; 32 covers typical C=64 in provided workloads
        )

        # 2) GroupNorm on y1 (num_groups=32), affine
        # Output tensor y1_gn: (B, C, H, W)
        y1_gn = torch.empty_like(y1, device=device, dtype=torch.float32)
        # norm1 params
        norm1_weight = self.norm1_weight.contiguous().to(torch.float32)  # (C,)
        norm1_bias = self.norm1_bias.contiguous().to(torch.float32)     # (C,)
        # Launch groupnorm kernel: axis 0 = batch, axis 1 = groups
        grid_gn = (B, self.num_groups)
        groupnorm_affine_kernel[grid_gn](
            y1, y1_gn, norm1_weight, norm1_bias,
            B, C, H, W, self.num_groups, self.eps,
            BLOCK_HW=1024,
        )

        # 3) SiLU on y1_gn
        y1_silu = torch.empty_like(y1_gn, device=device, dtype=torch.float32)
        total_elems = y1_silu.numel()
        silu_kernel[(triton.cdiv(total_elems, 1024),)](
            y1_gn, y1_silu, total_elems, BLOCK=1024
        )

        # 4) Second conv: y2 = conv3x3(y1_silu)
        x_for_conv2 = y1_silu.contiguous()
        w2 = self.conv2_weight.contiguous().to(torch.float32)  # (C, C, 3, 3)
        # Output tensor: (B, C, H, W)
        y2 = torch.empty((B, C, H, W), device=device, dtype=torch.float32)

        C_in2 = C
        C_out2 = C
        H_out2 = H
        W_out2 = W
        total_out2 = B * C_out2 * H_out2 * W_out2
        conv3x3_nchw_fp32[(total_out2,)](
            x_for_conv2, w2, y2,
            B, C_in2, H_out2, W_out2, C_out2, H_out2, W_out2,
            BLOCK_IN=32,
        )

        # 5) GroupNorm on y2 (num_groups=32), affine
        y2_gn = torch.empty_like(y2, device=device, dtype=torch.float32)
        norm2_weight = self.norm2_weight.contiguous().to(torch.float32)  # (C,)
        norm2_bias = self.norm2_bias.contiguous().to(torch.float32)     # (C,)
        grid_gn2 = (B, self.num_groups)
        groupnorm_affine_kernel[grid_gn2](
            y2, y2_gn, norm2_weight, norm2_bias,
            B, C, H, W, self.num_groups, self.eps,
            BLOCK_HW=1024,
        )

        # 6) SiLU on y2_gn
        y2_silu = torch.empty_like(y2_gn, device=device, dtype=torch.float32)
        total_elems2 = y2_silu.numel()
        silu_kernel[(triton.cdiv(total_elems2, 1024),)](
            y2_gn, y2_silu, total_elems2, BLOCK=1024
        )

        # 7) Residual add: out = y2_silu + x0 (elementwise, shapes match: (B, C, H, W))
        # Ensure x0 is float32 and contiguous
        x0_f32 = x0.contiguous().to(torch.float32)
        out = torch.empty_like(x0_f32, device=device, dtype=torch.float32)
        total_elems_add = x0_f32.numel()
        add_residual_fp32[(triton.cdiv(total_elems_add, 1024),)](
            x0_f32, y2_silu, out, total_elems_add, BLOCK=1024
        )

        # Return result; original model returns float32 (default), so we keep dtype as float32
        return out
