import triton
import triton.language as tl


# Triton kernels for GroupNorm (reduce then apply), and residual add.
# We focus on robustness: Triton loops are constexpr (N_TILES) and vectorized over tiles.

# GroupNorm reduce: compute per-(n, c) mean and rstd over spatial plane (H*W) for given num_groups, channels_per_group
# Grid: (B, num_groups, channels_per_group)
# Each program reduces one channel within one group over all spatial tiles.
@triton.jit
def group_norm_reduce_kernel(
    x_ptr,                # *f32, input [B, C, H, W]
    mean_ptr,             # *f32, per-channel mean [C]
    rstd_ptr,             # *f32, per-channel rstd [C]
    B: tl.constexpr,      # int
    C: tl.constexpr,      # int
    H: tl.constexpr,      # int
    W: tl.constexpr,      # int
    num_groups: tl.constexpr,  # int
    N_TILES: tl.constexpr,      # int, number of tiles over HW
    BLOCK_HW: tl.constexpr,     # int, tile size for HW
):
    n = tl.program_id(0)
    group = tl.program_id(1)
    c = tl.program_id(2)  # channel index within the group

    # channels per group
    channels_per_group = C // num_groups
    # starting channel index for this group
    c_start = group * channels_per_group
    # absolute channel index
    # We set c = c_start + c (host guarantees grid sets c accordingly), but we use c_start here.
    # For Triton grid, we launch with c in [c_start, c_start + channels_per_group).

    # Accumulators for sum and sumsq over spatial HW
    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)

    # Flatten spatial index space HW
    HW = H * W

    # Loop over tiles (constexpr), each tile processes BLOCK_HW elements
    for t in range(N_TILES):
        tile_start = t * BLOCK_HW
        idxs = tile_start + tl.arange(0, BLOCK_HW)
        mask = idxs < HW

        # Compute h, w from flattened idxs
        h = idxs // W
        w = idxs % W

        # Base offset for batch n
        base = n * C * HW

        # Per-channel offset for this c
        chan_offset = c * HW

        # Load values for this channel across the tile
        ptrs = x_ptr + base + chan_offset + idxs
        vals = tl.load(ptrs, mask=mask, other=0.0)

        # Accumulate sum and sum of squares
        sum_val += tl.sum(vals, axis=0)
        sum_sq += tl.sum(vals * vals, axis=0)

    # Compute mean and rstd
    hw_total = HW  # int
    mean = sum_val / hw_total
    var = sum_sq / hw_total - mean * mean
    # Numerical stability: clamp variance
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    # Store mean and rstd for this channel
    tl.store(mean_ptr + c, mean)
    tl.store(rstd_ptr + c, rstd)


# GroupNorm apply: y = ((x - mean[c]) * rstd[c]) * scale[c] + bias[c], then SiLU
# SiLU(z) = z * sigmoid(z)
# Grid: (B, num_groups, channels_per_group, N_TILES)
@triton.jit
def group_norm_apply_kernel(
    x_ptr,           # *f32, input [B, C, H, W]
    mean_ptr,        # *f32, mean [C]
    rstd_ptr,        # *f32, rstd [C]
    scale_ptr,       # *f32, scale [C]
    bias_ptr,        # *f32, bias [C]
    y_ptr,           # *f32, output [B, C, H, W]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    num_groups: tl.constexpr, N_TILES: tl.constexpr, BLOCK_HW: tl.constexpr,
):
    n = tl.program_id(0)
    group = tl.program_id(1)
    c = tl.program_id(2)  # absolute channel index
    t = tl.program_id(3)  # tile index over HW

    channels_per_group = C // num_groups
    c_start = group * channels_per_group

    # Load per-channel stats and affine params
    mean_c = tl.load(mean_ptr + c)
    rstd_c = tl.load(rstd_ptr + c)
    scale_c = tl.load(scale_ptr + c)
    bias_c = tl.load(bias_ptr + c)

    HW = H * W
    tile_start = t * BLOCK_HW
    idxs = tile_start + tl.arange(0, BLOCK_HW)
    mask = idxs < HW

    h = idxs // W
    w = idxs % W

    base = n * C * HW
    chan_offset = c * HW

    x_ptrs = x_ptr + base + chan_offset + idxs
    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)

    # Normalize and affine
    norm = (x_vals - mean_c) * rstd_c
    y_vals = norm * scale_c + bias_c

    # SiLU
    # sigmoid(z) = 1 / (1 + exp(-z))
    sig = 1.0 / (1.0 + tl.exp(-y_vals))
    y_vals = y_vals * sig

    y_ptrs = y_ptr + base + chan_offset + idxs
    tl.store(y_ptrs, y_vals, mask=mask)


# Triton elementwise residual add: out = y + x
@triton.jit
def residual_add_kernel(
    y_ptr, x_ptr, out_ptr,
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
):

    c = tl.program_id(1)
    hw = tl.program_id(2)

    HW = H * W
    idxs = hw + tl.arange(0, 1)
    mask = hw < HW  # single index per program

    base = n * C * HW
    chan_offset = c * HW

    out_ptrs = out_ptr + base + chan_offset + hw

    y_val = tl.load(y_ptrs, mask=mask, other=0.0)
    x_val = tl.load(x_ptrs, mask=mask, other=0.0)
    tl.store(out_ptrs, y_val + x_val, mask=mask)

        self.eps = eps

        conv2_weight: (C_out2, C_out1, 3, 3)
        norm2_weight, norm2_bias: (C_out2,)
        Returns: (B, C_out2, H, W)
        """

        assert x.dim() == 4, "x must be (B, C, H, W)"

            num_groups=self.num_groups, N_TILES=N_TILES1, BLOCK_HW=BLOCK_HW,
            num_warps=4, num_stages=2,
        )

        # Second conv

                                          bias=None, stride=1, padding=1)

        mean2 = torch.empty(C_out2, device=x.device, dtype=torch.float32)

            num_warps=4, num_stages=2,
        )

        return out
