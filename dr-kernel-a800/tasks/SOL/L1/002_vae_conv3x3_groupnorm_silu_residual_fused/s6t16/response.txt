import torch
import triton
import triton.language as tl


@triton.jit
def conv3x3_nchw_4d(x_ptr, w_ptr, y_ptr,
                    N, C_in, C_out, H, W,
                    x_sN, x_sC, x_sH, x_sW,
                    w_sCo, w_sCi, w_sKh, w_sKw,
                    y_sN, y_sC, y_sH, y_sW,
                    NUM_CI: tl.constexpr):
    # Grid: (N, C_out, H, W) one program per output element (n, co, h, w)
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w = tl.program_id(3)

    acc = tl.zeros((), dtype=tl.float32)

    # Sum over input channels and 3x3 neighborhood
    for ci in range(NUM_CI):
        for kh in range(3):
            for kw in range(3):
                h_in = pid_h + kh
                w_in = pid_w + kw
                # bounds check for padding
                h_in_valid = (h_in >= 0) & (h_in < H)
                w_in_valid = (w_in >= 0) & (w_in < W)
                if h_in_valid and w_in_valid:
                    x_offset = pid_n * x_sN + ci * x_sC + h_in * x_sH + w_in * x_sW
                    x_val = tl.load(x_ptr + x_offset)
                else:
                    x_val = 0.0

                w_offset = pid_co * w_sCo + ci * w_sCi + kh * w_sKh + kw * w_sKw
                w_val = tl.load(w_ptr + w_offset)
                acc += x_val * w_val

    y_offset = pid_n * y_sN + pid_co * y_sC + pid_h * y_sH + pid_w * y_sW
    tl.store(y_ptr + y_offset, acc)


@triton.jit
def groupnorm_reduce_sums(y_ptr, sums_ptr,
                           N, C, H, W, groups,
                           y_sN, y_sC, y_sH, y_sW):
    # Grid over (N, groups). Compute per-(n, group) sum and sumsq over all channels in group and all spatial positions.
    pid_n = tl.program_id(0)
    pid_g = tl.program_id(1)

    group_size_c = C // groups
    start_c = pid_g * group_size_c

    sum_val = tl.zeros((), dtype=tl.float32)
    sumsq_val = tl.zeros((), dtype=tl.float32)

    for c in range(start_c, start_c + group_size_c):
        for h in range(0, H):
            for w in range(0, W):
                y_offset = pid_n * y_sN + c * y_sC + h * y_sH + w * y_sW
                val = tl.load(y_ptr + y_offset)
                sum_val += val
                sumsq_val += val * val

    # Store [sum, sumsq] for (n, g)
    base = pid_n * groups + pid_g
    tl.store(sums_ptr + base * 2 + 0, sum_val)
    tl.store(sums_ptr + base * 2 + 1, sumsq_val)


@triton.jit
def groupnorm_apply_affine_silu(y_ptr, out_ptr,
                                scale_ptr, bias_ptr,
                                N, C, H, W,
                                sums_ptr,
                                y_sN, y_sC, y_sH, y_sW,
                                groups,
                                eps):
    # Grid over (N, C). For each (n, c), compute group index, normalize, apply affine, SiLU.
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)

    group_size_c = C // groups
    group_id = pid_c // group_size_c

    # Load sum and sumsq for (n, group_id)
    base = pid_n * groups + group_id
    sum_val = tl.load(sums_ptr + base * 2 + 0)
    sumsq_val = tl.load(sums_ptr + base * 2 + 1)

    group_size = group_size_c
    M = H * W
    mean = sum_val / M
    var = sumsq_val / M - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Preload scale and bias for this channel
    scale = tl.load(scale_ptr + pid_c)
    bias = tl.load(bias_ptr + pid_c)

    # Process all spatial positions for this channel
    for h in range(0, H):
        for w in range(0, W):
            y_offset = pid_n * y_sN + pid_c * y_sC + h * y_sH + w * y_sW
            val = tl.load(y_ptr + y_offset)
            norm = (val - mean) * rstd
            # Affine
            af = norm * scale + bias
            # SiLU: x * sigmoid(x) = x / (1 + exp(-x))
            sig = 1.0 / (1.0 + tl.exp(-af))
            out_val = af * sig
            out_offset = pid_n * y_sN + pid_c * y_sC + h * y_sH + w * y_sW  # same layout
            tl.store(out_ptr + out_offset, out_val)


@triton.jit
def add_residual_kernel(y_ptr, x_ptr, out_ptr, total_elems):
    pid = tl.program_id(0)
    # 1D elementwise add: out = y + x
    val_y = tl.load(y_ptr + pid)
    val_x = tl.load(x_ptr + pid)
    val_out = val_y + val_x
    tl.store(out_ptr + pid, val_out)

        grid_apply1 = (N, C)

            N, C, H, W,

            N, C2, H, W, groups,

            y2_sN, y2_sC, y2_sH, y2_sW,
            groups, eps

        # Residual add: out = y2_norm + x32
        total_elems = N * C2 * H * W
        out = torch.empty_like(y2_norm)
        grid_add = (total_elems,)
        add_residual_kernel[grid_add](
            y2_norm, x32, out, total_elems
        )

        return out
