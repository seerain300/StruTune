import torch
import triton
import triton.language as tl


# Triton kernel: conv3x3 stride=1, padding=1, no bias
# Input x: [B, C_in, H, W], contiguous NCHW
# Weights w: [C_out, C_in, 3, 3], contiguous in w[co, ci, kh, kw]
# Output out: [B, C_out, H, W], contiguous NCHW
@triton.jit
def conv3x3_nobias_tiled(
    x_ptr,      # *const float
    w_ptr,      # *const float
    out_ptr,    # *float
    B: tl.constexpr,
    C_in: tl.constexpr,
    H: tl.constexpr,
    W: tl.constexpr,
    C_out: tl.constexpr,
    H_out: tl.constexpr,
    W_out: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_W: tl.constexpr,
    x_stride_n: tl.constexpr, x_stride_c: tl.constexpr, x_stride_h: tl.constexpr, x_stride_w: tl.constexpr,
    out_stride_n: tl.constexpr, out_stride_c: tl.constexpr, out_stride_h: tl.constexpr, out_stride_w: tl.constexpr,
):
    pid_n = tl.program_id(0)  # batch
    pid_co = tl.program_id(1) # output channel
    pid_hblk = tl.program_id(2)  # tile over H_out
    pid_wblk = tl.program_id(3)  # tile over W_out

    h_start = pid_hblk * BLOCK_H
    w_start = pid_wblk * BLOCK_W

    h_offsets = h_start + tl.arange(0, BLOCK_H)
    w_offsets = w_start + tl.arange(0, BLOCK_W)

    # masks for output tile bounds
    h_mask = h_offsets < H_out
    w_mask = w_offsets < W_out

    # Accumulator for [BLOCK_H, BLOCK_W]
    acc = tl.zeros((BLOCK_H, BLOCK_W), dtype=tl.float32)

    # Iterate over input channels and 3x3 kernel
    for ci in tl.static_range(C_in):
        for kh in tl.static_range(3):
            for kw in tl.static_range(3):
                # input coordinates for 3x3 neighborhood
                h_in = h_offsets - 1 + kh  # padding=1
                w_in = w_offsets - 1 + kw

                # in-bounds masks
                in_h = (h_in >= 0) & (h_in < H)
                in_w = (w_in >= 0) & (w_in < W)
                in_bounds = in_h[:, None] & in_w[None, :]

                # x base pointer for (n, ci, h_in, w_in) with broadcasting to [BLOCK_H, BLOCK_W]
                base_x = pid_n * x_stride_n + ci * x_stride_c
                ptr_x = x_ptr + base_x + (h_in[:, None] * x_stride_h) + (w_in[None, :] * x_stride_w)
                x_vals = tl.load(ptr_x, mask=in_bounds, other=0.0).to(tl.float32)

                # weight scalar: w[co, ci, kh, kw]
                w_idx = pid_co * (C_in * 9) + ci * 9 + (kh * 3 + kw)
                w_val = tl.load(w_ptr + w_idx).to(tl.float32)

                # accumulate
                acc += x_vals * w_val

    # Store results with output bounds masks
    out_base = pid_n * out_stride_n + pid_co * out_stride_c
    out_ptrs = out_ptr + out_base + (h_offsets[:, None] * out_stride_h) + (w_offsets[None, :] * out_stride_w)
    out_mask = (h_offsets[:, None] < H_out) & (w_offsets[None, :] < W_out)
    tl.store(out_ptrs, acc, mask=out_mask)


# Triton GroupNorm: two-pass per (n, group), no bias in first pass
# Input: input_ptr flattened, N = B*C*H*W
# gamma_ptr: per-channel weight of length C (applied later)
# beta_ptr: per-channel bias of length C (applied later)
@triton.jit
def group_norm_two_pass(
    input_ptr,       # *const float, input flattened
    gamma_ptr,       # *const float, length C
    beta_ptr,        # *const float, length C
    output_ptr,      # *float, output flattened
    N: tl.constexpr,  # total number of elements
    C: tl.constexpr,  # channels
    H: tl.constexpr,  # height
    W: tl.constexpr,  # width
    G: tl.constexpr,  # num_groups
    eps: tl.float32,
    BLOCK: tl.constexpr,
):
    # program id: combine batch and group
    pid = tl.program_id(0)
    n = pid // G
    g = pid % G
    group_size = (C // G) * H * W
    start = n * (C * H * W) + g * group_size

    # First pass: compute sum and sum of squares over the group
    total_sum = 0.0
    total_sumsq = 0.0
    # loop over the group elements in chunks
    for off in tl.static_range(0, group_size, BLOCK):
        idx = start + off
        in_range = idx < start + group_size
        x = tl.load(input_ptr + idx, mask=in_range, other=0.0).to(tl.float32)
        total_sum += tl.sum(x, axis=0)
        total_sumsq += tl.sum(x * x, axis=0)

    mean = total_sum / group_size
    var = total_sumsq / group_size - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine gamma/beta
    for off in tl.static_range(0, group_size, BLOCK):
        idx = start + off
        in_range = idx < start + group_size
        x = tl.load(input_ptr + idx, mask=in_range, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        # channel index for affine per channel
        ch_per_group = C // G
        hw = (group_size // ch_per_group)
        chan = g * ch_per_group + ((idx - start) // hw)
        gamma = tl.load(gamma_ptr + chan).to(tl.float32)
        beta = tl.load(beta_ptr + chan).to(tl.float32)
        y = y * gamma + beta
        tl.store(output_ptr + idx, y, mask=in_range)


# Triton elementwise SiLU: y = x * sigmoid(x)
@triton.jit
def silu_kernel(in_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(in_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    # sigmoid(x) = 1 / (1 + exp(-x))
    s = 1.0 / (1.0 + tl.exp(-x))
    y = x * s
    tl.store(out_ptr + offs, y, mask=mask)


# Triton elementwise residual add: out = in + residual
@triton.jit
def add_residual_kernel(in_ptr, res_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(in_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    r = tl.load(res_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    y = x + r
    tl.store(out_ptr + offs, y, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self, num_groups=32, eps=1e-5):
        super().__init__()
        self.num_groups = num_groups
        self.eps = eps

    def forward(self, x: torch.Tensor, conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor):
        # Ensure device and dtype
        device = x.device
        dtype = torch.float32
        x = x.to(device=device, dtype=dtype).contiguous()
        conv1_weight = conv1_weight.to(device=device, dtype=dtype).contiguous()
        conv2_weight = conv2_weight.to(device=device, dtype=dtype).contiguous()

        B, C, H, W = x.shape
        assert C % self.num_groups == 0, "C must be divisible by num_groups (32)"
        C_out1 = conv1_weight.shape[0]
        C_out2 = conv2_weight.shape[0]
        # First convolution: conv3x3 stride=1, padding=1, no bias
        out1 = torch.empty((B, C_out1, H, W), device=device, dtype=dtype)
        grid_conv1 = (B, C_out1, triton.cdiv(H, 8), triton.cdiv(W, 8))
        conv3x3_nobias_tiled[grid_conv1](
            x, conv1_weight, out1,
            B, C, H, W, C_out1, H, W,
            8, 8,
            x.stride(0), x.stride(1), x.stride(2), x.stride(3),
            out1.stride(0), out1.stride(1), out1.stride(2), out1.stride(3),
            num_warps=4,
        )

        # GroupNorm1 on out1
        out1_flat = out1.view(-1)
        gamma1 = norm1_weight.to(device=device, dtype=dtype).contiguous()
        beta1 = norm1_bias.to(device=device, dtype=dtype).contiguous()
        out1_gn = torch.empty_like(out1)
        # Each program handles one (n, group)
        grid_gn1 = (B * self.num_groups,)
        group_norm_two_pass[grid_gn1](
            out1_flat, gamma1, beta1, out1_gn.view(-1),
            out1_flat.numel(), C_out1, H, W, self.num_groups, self.eps,
            BLOCK=4096,
            num_warps=4,
        )
        out1_gn = out1_gn.view(B, C_out1, H, W)

        # SiLU1
        out1_silu = torch.empty_like(out1_gn)
        N1 = out1_gn.numel()
        grid_silu1 = (triton.cdiv(N1, 4096),)
        silu_kernel[grid_silu1](out1_gn.view(-1), out1_silu.view(-1), N1, BLOCK=4096, num_warps=8)

        # Second convolution
        out2_pre = torch.empty((B, C_out2, H, W), device=device, dtype=dtype)
        grid_conv2 = (B, C_out2, triton.cdiv(H, 8), triton.cdiv(W, 8))
        conv3x3_nobias_tiled[grid_conv2](
            out1_silu, conv2_weight, out2_pre,
            B, C_out1, H, W, C_out2, H, W,
            8, 8,
            out1_silu.stride(0), out1_silu.stride(1), out1_silu.stride(2), out1_silu.stride(3),
            out2_pre.stride(0), out2_pre.stride(1), out2_pre.stride(2), out2_pre.stride(3),
            num_warps=4,
        )

        # GroupNorm2 on out2_pre
        out2_pre_flat = out2_pre.view(-1)
        gamma2 = norm2_weight.to(device=device, dtype=dtype).contiguous()
        beta2 = norm2_bias.to(device=device, dtype=dtype).contiguous()
        out2_gn = torch.empty_like(out2_pre)
        grid_gn2 = (B * self.num_groups,)
        group_norm_two_pass[grid_gn2](
            out2_pre_flat, gamma2, beta2, out2_gn.view(-1),
            out2_pre_flat.numel(), C_out2, H, W, self.num_groups, self.eps,
            BLOCK=4096,
            num_warps=4,
        )
        out2_gn = out2_gn.view(B, C_out2, H, W)

        # SiLU2
        out2_silu = torch.empty_like(out2_gn)
        N2 = out2_gn.numel()
        grid_silu2 = (triton.cdiv(N2, 4096),)
        silu_kernel[grid_silu2](out2_gn.view(-1), out2_silu.view(-1), N2, BLOCK=4096, num_warps=8)

        # Residual add: out2_silu + x
        out = torch.empty_like(out2_silu)
        Nfinal = out2_silu.numel()
        grid_add = (triton.cdiv(Nfinal, 4096),)
        add_residual_kernel[grid_add](out2_silu.view(-1), x.view(-1), out.view(-1), Nfinal, BLOCK=4096, num_warps=8)

        return out
