# Grid: (B, C_out, tiles over W_out). Each program handles one (b, oc) and a vector of w_out across all h_out.
@triton.jit

    x_ptr, w_ptr, y_ptr,
    B, C_in, H, W, C_out, H_out, W_out,
    x_stride_b, x_stride_c, x_stride_h, x_stride_w,
    w_stride_cin, w_stride_cout, w_stride_kh, w_stride_kw,
    y_stride_b, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_OC: tl.constexpr,  # e.g., 32
    BLOCK_W: tl.constexpr,   # e.g., 128
):
    b = tl.program_id(0)
    oc_base = tl.program_id(1)
    tile_w = tl.program_id(2)

    # vector of width positions for this tile
    w_vec = tile_w * BLOCK_W + tl.arange(0, BLOCK_W)
    w_mask = w_vec < W_out

    # accumulator for BLOCK_OC output channels
    acc = tl.zeros((BLOCK_OC, BLOCK_W), dtype=tl.float32)

    # loop over output height positions
    for h_out in range(0, H_out):
        # loop over input channels in blocks
        for ic_base in range(0, C_in, BLOCK_OC):
            oc_vec = oc_base + tl.arange(0, BLOCK_OC)
            oc_mask = oc_vec < C_out

            # initialize acc for this (ic, h_out, w_vec) to zero
            acc = tl.zeros((BLOCK_OC, BLOCK_W), dtype=tl.float32)

            # loop over 3x3 neighborhood
            for kh in range(0, 3):
                h_in = h_out + (kh - 1)  # padding=1
                for kw in range(0, 3):
                    w_in_vec = w_vec + (kw - 1)

                    # compute x pointers for this (ic, h_in, w_in_vec) and sum over ic-block
                    for ic_full in range(0, BLOCK_OC):
                        ic = ic_base + ic_full
                        ic_valid = ic < C_in
                        # if ic is invalid, skip
                        if not ic_valid:
                            continue

                        x_ptrs = x_ptr \
                                 + b * x_stride_b \
                                 + ic * x_stride_c \
                                 + h_in * x_stride_h \
                                 + w_in_vec * x_stride_w
                        x_vals = tl.load(x_ptrs, mask=w_mask, other=0.0)  # [BLOCK_W]

                        # load weights for this (ic, oc_vec, kh, kw)
                        w_ptrs = w_ptr \
                                 + ic * w_stride_cin \
                                 + oc_vec * w_stride_cout \
                                 + kh * w_stride_kh \
                                 + kw * w_stride_kw
                        w_vals = tl.load(w_ptrs, mask=oc_mask, other=0.0)  # [BLOCK_OC]

                        # accumulate: outer product w_vals[:, None] * x_vals[None, :]
                        # Use masking: w_mask and oc_mask
                        acc += w_vals[:, None] * x_vals[None, :]

            # store accumulated results for this (b, oc_vec, h_out, w_vec)

                     + b * y_stride_b \
                     + oc_vec * y_stride_c \
                     + h_out * y_stride_h \
                     + w_vec * y_stride_w
            # mask combines oc and w
            store_mask = oc_mask[:, None] & w_mask[None, :]
            tl.store(y_ptrs, acc, mask=store_mask)

@triton.jit
def group_norm_affine_silu(
    x_ptr, weight_ptr, bias_ptr, y_ptr,
    B, C, H, W, num_groups: tl.constexpr,  # 32
    eps: tl.constexpr,
    x_stride_b, x_stride_c, x_stride_h, x_stride_w,

    BLOCK: tl.constexpr,  # reduction tile
):

    g = tl.program_id(1)

    group_elements = group_channels * H * W

                 + b * x_stride_b \
                 + c * x_stride_c \
                 + h * x_stride_h \
                 + w * x_stride_w
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)

        sum_val += tl.sum(x_vals, axis=0)
        sum_sq += tl.sum(x_vals * x_vals, axis=0)

        h = sp // W
        w = sp % W
