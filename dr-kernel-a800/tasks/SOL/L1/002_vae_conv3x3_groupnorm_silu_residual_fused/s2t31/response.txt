import triton
import triton.language as tl


def _assert_divisible(dividend: int, divisor: int):
    if dividend % divisor != 0:
        raise ValueError(f"{dividend} is not divisible by {divisor}")


# Triton kernel: Conv3x3 NCHW, stride=1, padding=1, no bias
# Specialized for C == C_OUT == BLOCK_C == 64. Computes out[n, co, h, w].
@triton.jit
def conv3x3_nchw_kernel(
    x_ptr, w_ptr, out_ptr,
    B, C, H, W, C_OUT,
    BLOCK_W: tl.constexpr,  # tile size along W
    BLOCK_C: tl.constexpr,  # must be 64 for this specialization
):
    # Grid: (B, C_OUT, H, ceil_div(W, BLOCK_W))
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w_blk = tl.program_id(3)

    h = pid_h
    w_start = pid_w_blk * BLOCK_W
    w_offsets = w_start + tl.arange(0, BLOCK_W)
    mask_w = w_offsets < W

    # Accumulator for output row
    acc = tl.zeros([BLOCK_W], dtype=tl.float32)

    # We specialize for C == BLOCK_C == 64
    # Loop over input channels ci = 0..BLOCK_C-1
    for ci in range(0, BLOCK_C):
        # Accumulate over 3x3 neighborhood with padding=1
        # Note: indices may be out of bounds (mask handles), but we keep code compact.
        for dh in range(-1, 1 + 1):
            for dw in range(-1, 1 + 1):
                hi = h + dh
                # mask for valid h
                mask_h = (hi >= 0) & (hi < H)
                for w_offset in w_offsets:
                    wi = w_offset + dw
                    mask_wi = (wi >= 0) & (wi < W)
                    # Compute input pointer: (((n * C + ci) * H + hi) * W + wi)
                    x_idx = (((pid_n * C + ci) * H + hi) * W + wi)
                    # Combine masks: mask_wi is a scalar; broadcast with mask_w as vector
                    load_mask = mask_w & mask_h & mask_wi
                    x_val = tl.load(x_ptr + x_idx, mask=load_mask, other=0.0)
                    # Compute weight pointer: (((co * C + ci) * 3 + (1+dh)) * 3 + (1+dw))
                    w_idx = (((pid_co * C + ci) * 3 + (1 + dh)) * 3 + (1 + dw))
                    w_val = tl.load(w_ptr + w_idx)
                    acc += x_val * w_val

    # Store output: out_idx = (((n * C_OUT + co) * H + h) * W + w)
    for i, w_off in enumerate(w_offsets):
        out_idx = (((pid_n * C_OUT + pid_co) * H + h) * W + w_off)
        tl.store(out_ptr + out_idx, acc[i], mask=mask_w[i])


# Triton kernels for GroupNorm (per sample and per group):
# 1) Compute sum and sum of squares across all elements in the group.
@triton.jit
def groupnorm_sums_kernel(
    x_ptr, sums_ptr, sumsq_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,
):
    # One program per (n, group)
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    start_ci = g * C_PER_GROUP
    s = tl.float32(0.0)
    s2 = tl.float32(0.0)
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        for h in range(0, H):
            for w in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w
                x_val = tl.load(x_ptr + idx)
                s += x_val
                s2 += x_val * x_val
    tl.store(sums_ptr + pid, s)
    tl.store(sumsq_ptr + pid, s2)


# 2) Compute inverse std from sums and sumsq (host computes mean/var; Triton computes invstd).
@triton.jit
def groupnorm_invstd_kernel(
    sums_ptr, sumsq_ptr, invstd_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    s = tl.load(sums_ptr + pid)
    s2 = tl.load(sumsq_ptr + pid)
    group_size = C_PER_GROUP * H * W
    mean = s / group_size
    var = s2 / group_size - mean * mean
    invstd = 1.0 / tl.sqrt(var + 1e-5)
    tl.store(invstd_ptr + pid, invstd)


# 3) Apply GroupNorm + affine + SiLU: normalize, then y = silu(y * invstd + b)
@triton.jit
def groupnorm_silu_apply_kernel(
    x_ptr, norm_w_ptr, norm_b_ptr, out_ptr, invstd_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    invstd = tl.load(invstd_ptr + pid)

    start_ci = g * C_PER_GROUP
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        w = tl.load(norm_w_ptr + ci)
        b = tl.load(norm_b_ptr + ci)
        for h in range(0, H):
            for w_idx in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w_idx
                x_val = tl.load(x_ptr + idx)
                norm = x_val * invstd + b
                # SiLU: x * sigmoid(x) = x / (1 + exp(-x))
                sig = 1.0 / (1.0 + tl.exp(-norm))
                y = norm * sig
                tl.store(out_ptr + idx, y)
