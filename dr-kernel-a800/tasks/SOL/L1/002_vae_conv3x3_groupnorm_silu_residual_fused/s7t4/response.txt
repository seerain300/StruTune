import torch
import triton
import triton.language as tl


@triton.jit
def conv3x3_stride1_pad1_kernel(
    x_ptr,          # *float32 input tensor (B, C_in, H, W)
    w_ptr,          # *float32 weight tensor (C_out, C_in, 3, 3)
    y_ptr,          # *float32 output tensor (B, C_out, H, W)
    N, C_in, H, W, C_out,
    BLOCK_OC: tl.constexpr,
):
    # Grid: (N, ceil_div(C_out, BLOCK_OC))
    n = tl.program_id(0)
    oc_block_id = tl.program_id(1)
    oc_start = oc_block_id * BLOCK_OC
    oc_offsets = oc_start + tl.arange(0, BLOCK_OC)
    oc_mask = oc_offsets < C_out

    # Accumulator for output channels in this tile
    acc = tl.zeros((BLOCK_OC,), dtype=tl.float32)

    # Loop over input channels and 3x3 taps
    for cin in range(C_in):
        for kh in range(3):
            for kw in range(3):
                for oh in range(H):
                    ih = oh + kh - 1
                    for ow in range(W):
                        iw = ow + kw - 1
                        valid = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
                        # Input linear index: (((n * C_in + cin) * H + ih) * W + iw)
                        x_index = (((n * C_in + cin) * H + ih) * W + iw)
                        x_val = tl.load(x_ptr + x_index, mask=valid, other=0.0)
                        # Weight linear index: (((oc * C_in + cin) * 9) + (kh * 3 + kw))
                        w_index_vec = (((oc_offsets * C_in + cin) * 9) + (kh * 3 + kw))
                        # Load weights for this oc tile
                        w_vals = tl.load(w_ptr + w_index_vec, mask=oc_mask, other=0.0)
                        # FMA accumulate
                        acc += w_vals * x_val

    # Store results to output y[n, oc, h, w] for all h,w. We'll write acc to each row across H*W.
    # We don't have h/w indices here; the outer loops handle all positions. After the loops,
    # we can write acc to y at the appropriate (n, oc) slice.
    # To make it explicit, we recompute per (oh, ow):
    # We need to write acc[oc] into y[n, oc, oh, ow] for all oh, ow. We do this by launching
    # a second grid over H*W, but Triton doesn't allow nested grid launches here. Instead,
    # we store after finishing accumulation by iterating oh and ow again:
    for oh in range(H):
        for ow in range(W):
            for j in range(BLOCK_OC):
                if oc_mask[oc_offsets[j]]:
                    y_index = (((n * C_out + oc_offsets[j]) * H + oh) * W + ow)
                    tl.store(y_ptr + y_index, acc[j])


@triton.jit
def group_norm_affine_kernel(
    y_in_ptr,       # *float32 input tensor after conv (B, C, H, W)
    scale_ptr,      # *float32 GroupNorm weight (C,)
    bias_ptr,       # *float32 GroupNorm bias (C,)
    y_out_ptr,      # *float32 output tensor (B, C, H, W)
    N, C, H, W,     # dims
    num_groups: tl.constexpr,  # e.g., 32
    eps: tl.float32,
):
    # Grid: (N, num_groups)
    n = tl.program_id(0)
    g = tl.program_id(1)

    channels_per_group = C // num_groups
    group_start_channel = g * channels_per_group
    group_size = channels_per_group * H * W

    # First pass: compute mean and variance
    total_sum = 0.0
    total_sumsq = 0.0

    for c_local in range(channels_per_group):
        c = group_start_channel + c_local
        base = (n * C + c) * H * W
        for oh in range(H):
            for ow in range(W):
                idx = oh * W + ow
                x = tl.load(y_in_ptr + base + idx)
                total_sum += x
                total_sumsq += x * x

    mean = total_sum / (channels_per_group * H * W)
    var = total_sumsq / (channels_per_group * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for c_local in range(channels_per_group):
        c = group_start_channel + c_local
        base_in = (n * C + c) * H * W
        scale = tl.load(scale_ptr + c)
        beta = tl.load(bias_ptr + c)
        base_out = (n * C + c) * H * W
        for oh in range(H):
            for ow in range(W):
                idx = oh * W + ow
                x = tl.load(y_in_ptr + base_in + idx)
                y = (x - mean) * inv_std
                y = y * scale + beta
                tl.store(y_out_ptr + base_out + idx, y)


@triton.jit
def silu_kernel(x_ptr, y_ptr, N, C, H, W, num_warps: tl.constexpr, num_stages: tl.constexpr):
    # Elementwise SiLU: y = x * sigmoid(x)
    for n in range(N):
        for c in range(C):
            base = (n * C + c) * H * W
            for i in range(H * W):
                x = tl.load(x_ptr + base + i)
                s = 1.0 / (1.0 + tl.exp(-x))
                y = x * s
                tl.store(y_ptr + base + i, y)


@triton.jit
def add_residual_kernel(y_ptr, x_ptr, out_ptr, N, C, H, W, num_warps: tl.constexpr, num_stages: tl.constexpr):
    # out = y + x, elementwise
    for n in range(N):
        for c in range(C):
            base = (n * C + c) * H * W
            for i in range(H * W):
                a = tl.load(y_ptr + base + i)
                b = tl.load(x_ptr + base + i)
                c = a + b
                tl.store(out_ptr + base + i, c)

import torch
import triton
import triton.language as tl


@triton.jit
def conv3x3_stride1_pad1_kernel(
    x_ptr,          # *float32 input tensor (B, C_in, H, W)
    w_ptr,          # *float32 weight tensor (C_out, C_in, 3, 3)
    y_ptr,          # *float32 output tensor (B, C_out, H, W)
    N, C_in, H, W, C_out,
    BLOCK_OC: tl.constexpr,
):
    # Grid: (N, ceil_div(C_out, BLOCK_OC))
    n = tl.program_id(0)
    oc_block_id = tl.program_id(1)
    oc_start = oc_block_id * BLOCK_OC
    oc_offsets = oc_start + tl.arange(0, BLOCK_OC)
    oc_mask = oc_offsets < C_out

    # Loop over output spatial positions to compute full output for this tile
    for oh in range(H):
        for ow in range(W):
            # Accumulator for this (n, oc_tile) at fixed (oh, ow)
            acc = tl.zeros((BLOCK_OC,), dtype=tl.float32)
            # Loop over input channels and 3x3 taps
            for cin in range(C_in):
                for kh in range(3):
                    for kw in range(3):
                        ih = oh + kh - 1
                        iw = ow + kw - 1
                        valid = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
                        # Input linear index: (((n * C_in + cin) * H + ih) * W + iw)
                        x_index = (((n * C_in + cin) * H + ih) * W + iw)
                        x_val = tl.load(x_ptr + x_index, mask=valid, other=0.0)
                        # Weight linear index: (((oc * C_in + cin) * 9) + (kh * 3 + kw))
                        w_index_vec = (((oc_offsets * C_in + cin) * 9) + (kh * 3 + kw))
                        # Load weights for this oc tile (vector)
                        w_vals = tl.load(w_ptr + w_index_vec, mask=oc_mask, other=0.0)
                        # FMA accumulate
                        acc += w_vals * x_val
            # Store results for each oc in this tile at (n, oc, oh, ow)
            for j in range(BLOCK_OC):
                if oc_mask[oc_offsets[j]]:
                    y_index = (((n * C_out + oc_offsets[j]) * H + oh) * W + ow)

    eps: tl.float32,
):

    group_elems = channels_per_group * H * W

        c = group_start_channel + c_local
