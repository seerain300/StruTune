- Convolution kernel: One Triton program computes one output pixel for a given (n, c_out). It loops over input channels and 3x3 neighborhood (padding=1), accumulating into a register and storing. - GroupNorm:

- Dtypes: All tensors are promoted to float32 for numerical stability. The original returns float32 since inputs are typically float32. If different dtypes are provided, we convert to float32.

# Conv3x3: y = x * w, stride=1, padding=1, no bias
# Grid: (B, C_out, H*W). Each program computes one output pixel.
@triton.jit
def conv3x3_kernel(
    x_ptr,            # *f32, input [B, C_in, H, W]
    w_ptr,            # *f32, weight [C_out, C_in, 3, 3]
    y_ptr,            # *f32, output [B, C_out, H, W]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    C_in: tl.constexpr, C_out: tl.constexpr,
):
    n = tl.program_id(0)
    co = tl.program_id(1)
    hw = tl.program_id(2)
    h_out = hw // W
    w_out = hw % W

    acc = 0.0
    # Loop over input channels
    for ci in range(C_in):
        # 3x3 neighborhood
        for dh in range(3):
            for dw in range(3):
                h_in = h_out + dh - 1  # padding=1
                w_in = w_out + dw - 1
                # compute input index
                idx_x = ((n * C + ci) * H + h_in) * W + w_in
                # compute weight index: [co, ci, dh+1, dw+1]
                idx_w = ((co * C_in + ci) * 9) + (dh + 1) * 3 + (dw + 1)
                x_val = tl.load(x_ptr + idx_x)
                w_val = tl.load(w_ptr + idx_w)
                acc += x_val * w_val

    # store output
    idx_y = ((n * C_out + co) * H + h_out) * W + w_out
    tl.store(y_ptr + idx_y, acc)
