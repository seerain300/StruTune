import torch
import triton
import triton.language as tl


# Triton conv3x3 stride=1, padding=1, no bias: one program per (n, co, h_out, w_out)
@triton.jit
def conv3x3_nobias_one_elem(
    x_ptr,         # *const float, input [B, C_in, H, W]
    w_ptr,         # *const float, weights [C_out, C_in, 3, 3], flattened to [C_out, C_in, 9]
    out_ptr,       # *float, output [B, C_out, H, W]
    B: tl.constexpr,
    C_in: tl.constexpr,
    H: tl.constexpr,
    W: tl.constexpr,
    C_out: tl.constexpr,
    H_out: tl.constexpr,
    W_out: tl.constexpr,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    out_stride_n, out_stride_c, out_stride_h, out_stride_w,
):
    # program ids
    pid_n = tl.program_id(0)  # batch
    pid_co = tl.program_id(1) # output channel
    pid_h = tl.program_id(2)  # output height index
    pid_w = tl.program_id(3)  # output width index

    h_out = pid_h
    w_out = pid_w

    # accumulate
    acc = 0.0

    # loop over input channels and 3x3 kernel
    for ci in tl.static_range(C_in):
        for kh in tl.static_range(3):
            for kw in tl.static_range(3):
                h_in = h_out - 1 + kh  # padding=1
                w_in = w_out - 1 + kw
                in_bounds = (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)
                # base pointer for x at (n, ci, h_in, w_in)
                ptr_x = x_ptr + pid_n * x_stride_n + ci * x_stride_c + h_in * x_stride_h + w_in * x_stride_w
                x_val = tl.load(ptr_x, mask=in_bounds, other=0.0)
                # weight index: (co * (C_in * 9)) + (ci * 9) + (kh * 3 + kw)
                w_idx = pid_co * (C_in * 9) + ci * 9 + (kh * 3 + kw)
                w_val = tl.load(w_ptr + w_idx)
                acc += x_val * w_val

    # store output
    out_ptr_elt = out_ptr + pid_n * out_stride_n + pid_co * out_stride_c + h_out * out_stride_h + w_out * out_stride_w
    tl.store(out_ptr_elt, acc)


# Triton GroupNorm over groups: two-pass per (n, group)
@triton.jit
def group_norm_two_pass(
    out_ptr,       # *const float, input [B, C, H, W]
    gamma_ptr,     # *const float, per-channel scale [C]
    beta_ptr,      # *const float, per-channel bias [C]
    out_norm_ptr,  # *float, output normalized [B, C, H, W]
    B: tl.constexpr,
    C: tl.constexpr,
    H: tl.constexpr,
    W: tl.constexpr,
    num_groups: tl.constexpr,
    eps: tl.constexpr,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,  # we use the same strides for out norm
):
    # Grid: (B, num_groups)
    pid_n = tl.program_id(0)  # sample
    pid_g = tl.program_id(1)  # group id

    group_size = C // num_groups
    group_start_c = pid_g * group_size

    # First pass: compute sum and sumsq over the group for sample pid_n
    sum_val = 0.0
    sum_sq = 0.0
    # Iterate channels in this group
    for c_off in tl.static_range(group_size):
        c = group_start_c + c_off
        # iterate H*W elements for this channel
        for h in tl.static_range(H):
            for w in tl.static_range(W):
                ptr = out_ptr + pid_n * x_stride_n + c * x_stride_c + h * x_stride_h + w * x_stride_w
                x_val = tl.load(ptr)
                sum_val += x_val
                sum_sq += x_val * x_val

    # Compute mean and variance
    M = group_size * H * W
    mean = sum_val / M
    var = sum_sq / M - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine, then store
    for c_off in tl.static_range(group_size):
        c = group_start_c + c_off
        gamma = tl.load(gamma_ptr + c)
        beta = tl.load(beta_ptr + c)
        for h in tl.static_range(H):
            for w in tl.static_range(W):
                ptr_in = out_ptr + pid_n * x_stride_n + c * x_stride_c + h * x_stride_h + w * x_stride_w
                x_val = tl.load(ptr_in)
                y = (x_val - mean) * inv_std
                y = y * gamma + beta
                ptr_out = out_norm_ptr + pid_n * x_stride_n + c * x_stride_c + h * x_stride_h + w * x_stride_w
                tl.store(ptr_out, y)


# Triton SiLU: elementwise on flattened buffer
@triton.jit
def silu_kernel(
    inp_ptr,       # *const float
    out_ptr,       # *float
    N,             # int
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(inp_ptr + offs, mask=mask, other=0.0)
    # SiLU(x) = x * sigmoid(x) = x / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(out_ptr + offs, y, mask=mask)


# Triton residual add: out = out + inp (elementwise)
@triton.jit
def add_residual_kernel(
    out_ptr,       # *const float
    inp_ptr,       # *const float
    N,             # int
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    out_val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    inp_val = tl.load(inp_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, out_val + inp_val, mask=mask)


# ModelNew entry point
class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.num_groups = 32
        self.eps = 1e-5

    def forward(self, x: torch.Tensor,
                conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor):
        # Ensure contiguous and float32
        device = x.device
        B, C, H, W = x.shape
        # First conv: conv1_weight: [C, C, 3, 3] -> output [B, C, H, W]
        out1 = torch.empty((B, C, H, W), device=device, dtype=torch.float32)

        grid_conv1 = (B, C, H, W)
        conv3x3_nobias_one_elem[grid_conv1](
            x, conv1_weight, out1,
            B=B, C_in=C, H=H, W=W, C_out=C, H_out=H, W_out=W,
            x_stride_n=x.stride(0), x_stride_c=x.stride(1), x_stride_h=x.stride(2), x_stride_w=x.stride(3),
            out_stride_n=out1.stride(0), out_stride_c=out1.stride(1), out_stride_h=out1.stride(2), out_stride_w=out1.stride(3),
        )

        # GroupNorm1 (num_groups=32)
        out1_gn = torch.empty_like(out1)

        grid_gn1 = (B, self.num_groups)
        group_norm_two_pass[grid_gn1](
            out1, norm1_weight, norm1_bias, out1_gn,

        )

        # SiLU1
        out1_silu = torch.empty_like(out1_gn)
        N1 = out1_gn.numel()
        grid_silu1 = (triton.cdiv(N1, 1024),)
        silu_kernel[grid_silu1](out1_gn, out1_silu, N1, BLOCK=1024)

        # Second conv: conv2_weight: [C, C, 3, 3] -> output [B, C, H, W]
        out2_pre = torch.empty((B, C, H, W), device=device, dtype=torch.float32)

        grid_conv2 = (B, C, H, W)

            out1_silu, conv2_weight, out2_pre,
            B=B, C_in=C, H=H, W=W, C_out=C, H_out=H, W_out=W,
            x_stride_n=out1_silu.stride(0), x_stride_c=out1_silu.stride(1), x_stride_h=out1_silu.stride(2), x_stride_w=out1_silu.stride(3),
            out_stride_n=out2_pre.stride(0), out_stride_c=out2_pre.stride(1), out_stride_h=out2_pre.stride(2), out_stride_w=out2_pre.stride(3),
        )

        # GroupNorm2 (num_groups=32)
        out2_gn = torch.empty_like(out2_pre)

        grid_gn2 = (B, self.num_groups)
        group_norm_two_pass[grid_gn2](
            out2_pre, norm2_weight, norm2_bias, out2_gn,
            B=B, C=C, H=H, W=W, num_groups=self.num_groups, eps=self.eps,
            x_stride_n=out2_gn.stride(0), x_stride_c=out2_gn.stride(1), x_stride_h=out2_gn.stride(2), x_stride_w=out2_gn.stride(3),
        )

        # SiLU2
        out2_silu = torch.empty_like(out2_gn)
        N2 = out2_gn.numel()
        grid_silu2 = (triton.cdiv(N2, 1024),)
        silu_kernel[grid_silu2](out2_gn, out2_silu, N2, BLOCK=1024)

        # Add residual x
        out = torch.empty_like(out2_silu)
        Nfinal = out2_silu.numel()
        grid_add = (triton.cdiv(Nfinal, 1024),)
        add_residual_kernel[grid_add](out2_silu, x, Nfinal, BLOCK=1024)

        return out

- Si defines group_norm_two_pass(..., num_groups: tl.constexpr, eps: tl.constexpr, ...) en Triton, en forward hay que poner group_norm_two_pass(..., 32, 1e-5, ...). No usar variables. No fallar.
- No uses torch ops en forward. Si lo haces, el evaluador lo rechaza. Forward puede hacer allocations, hacer x = x.contiguous(), pasarlo a float32, y llamar kernels. No F.conv2d, no F.group_norm, no F.silu.

En cuanto a performance: se evalúa speed (geomean). En general, conv3x3 es costoso; evitarte implementar conv en Triton puede ayudarte a tener éxito y a centrarse en GroupNorm y SiLU, que se pueden hacer en Triton y seguir siendo correctos. Aun así, intentaremos mantenerlo simple y robusto, y volveremos a conv en PyTorch si es necesario, aunque recomiendo implementar conv en Triton después (y no es trivial, y falla en tiempo). Prioridad es que no falles en tiempo y quede correcto. Prostitutas y muestra cómo hacerlo.

Si se mantiene estricto, vamos a implementar conv en Triton: un kernel simple, robusto, y con num_groups constexpr.

Preparémonos a implementar un kernel de conv3x3 en Triton y un kernel de GroupNorm forward, y ambos con num_groups constexpr=32. No uses runtime args for num_groups. GroupNorm forward two-pass: first pass compute sum and sumsq for each (n, group); second pass normalize + affine (gamma, beta). Operate on float32, ensure inputs are contiguous.

Empecemos. Vamos a implementar:
- Triton conv3x3 no bias, stride=1, padding=1, one output element per program (n, co, h_out, w_out). For-loops over ci in [0..C_in-1] and kh,kw in [0..2], with padding. Load weight as w_ptr[co, ci, kh, kw].
- Triton GroupNorm forward two-pass, constexpr num_groups=32, eps constexpr.
- Triton SiLU elementwise, constexpr block size, grid over elements.
- Triton residual add elementwise.

Forward in ModelNew:
- x = x.contiguous().to(torch.float32)
- Allocate out1, do Triton conv1.
- Allocate out1_gn, do Triton GroupNorm1.
- Allocate out1_silu, do Triton SiLU.
- Allocate out2_pre, do Triton conv2 (same as conv1).
- Allocate out2_gn, do Triton GroupNorm2.
- Allocate out2_silu, do Triton SiLU.
- Allocate out, do Triton residual add.
- Return out.

Fixa num_groups constexpr en 32. No lo calcules en runtime. Si falla en algunos workloads, será porque C no es múltiplo de 32. En ese caso, GroupNorm no es compatible (PyTorch GroupNorm requires divisible channels by num_groups). Pero en el código original, GroupNorm is applied with num_groups=32, implying C must be divisible by 32; thus, for our workloads, C % 32 == 0. We will enforce x.channels divisible by 32. If not, fallback to PyTorch GroupNorm (but we cannot use torch ops in forward; so we will guard and raise an error, but since we cannot raise from Triton, we will ensure inputs are correct by construction). Still, to be robust, we’ll keep num_groups=32 constexpr and rely on provided workloads satisfying C % 32 == 0.

Now, code. Triton kernels first, then ModelNew.

Triton kernels:

1) conv3x3_nobias_one_elem
- args:
  - x_ptr, w_ptr, out_ptr
  - B, C_in, H, W, C_out, H_out, W_out (all constexpr because Triton requires them known at JIT time)
  - x_stride_n, x_stride_c, x_stride_h, x_stride_w
  - out_stride_n, out_stride_c, out_stride_h, out_stride_w
- grid: (B, C_out, H_out, W_out)
- For each program id corresponding to (n, co, h_out, w_out), loop ci in [0..C_in-1], kh,kw in [0..2], compute h_in, w_in with padding=1, masked load from x, load weight scalar from w_ptr using linear index (co * (C_in * 9)) + (ci * 9 + kh*3 + kw), accumulate, store to out_ptr.

2) group_norm_forward_two_pass
- args:
  - out_ptr: input tensor [B, C, H, W]
  - gamma_ptr: [C]
  - beta_ptr: [C]
  - out_norm_ptr: output normalized [B, C, H, W]
  - B, C, H, W
  - num_groups: tl.constexpr (32)
  - eps: tl.constexpr (e.g., 1e-5)
  - x_stride_n, x_stride_c, x_stride_h, x_stride_w (same strides used for input and output)
- grid: (B, num_groups)
- In first pass: compute sum and sumsq over channels in group and all H*W. In second pass: normalize using mean/var and apply gamma/beta, store to out_norm_ptr.
- Note: We loop over H and W using tl.static_range, which Triton requires. Since H and W are passed as tl.constexpr, this is fine. This two-pass approach avoids using runtime-dependent loop bounds inside Triton.

3) silu_kernel
- args:
  - inp_ptr: *const float
  - out_ptr: *float
  - N: int
  - BLOCK: tl.constexpr (e.g., 1024)
- grid: (ceil_div(N, BLOCK),)
- Elementwise SiLU: y = x * sigmoid(x) = x / (1 + exp(-x)). Use tl.exp and tl.store.

4) add_residual_kernel
- args:
  - out_ptr: *float
  - inp_ptr: *const float (residual x)
  - N: int
  - BLOCK: tl.constexpr
- grid: (ceil_div(N, BLOCK),)
- Elementwise out = out + inp.
