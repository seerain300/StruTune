import torch
import triton
import triton.language as tl


@triton.jit
def conv3x3_stride1_pad1_kernel(
    x_ptr,                # *float32 input tensor (B, C_in, H, W)
    w_ptr,                # *float32 weights tensor (C_out, C_in, 3, 3)
    y_ptr,                # *float32 output tensor (B, C_out, H, W)
    N, C_in, C_out, H, W,             # sizes
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    w_stride_oc, w_stride_ic, w_stride_kh, w_stride_kw,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_OC: tl.constexpr,           # tile of output channels per program
    BLOCK_HW: tl.constexpr,           # spatial tile
):
    # program ids
    n = tl.program_id(0)
    oc_block = tl.program_id(1)
    hw_block = tl.program_id(2)

    # oc tile
    oc_start = oc_block * BLOCK_OC
    oc_offsets = oc_start + tl.arange(0, BLOCK_OC)
    oc_mask = oc_offsets < C_out

    # spatial tile
    hw_start = hw_block * BLOCK_HW
    hw_offsets = hw_start + tl.arange(0, BLOCK_HW)
    hw_mask = hw_offsets < (H * W)
    h = hw_offsets // W
    w = hw_offsets % W

    # accumulator for output channels in tile
    acc = tl.zeros((BLOCK_OC,), dtype=tl.float32)

    # loop over input channels
    for ic in range(C_in):
        # loop over 3x3 taps
        for kh in range(3):
            for kw in range(3):
                # input coordinates with padding
                ih = h + kh - 1  # shape: (BLOCK_HW,)
                iw = w + kw - 1  # shape: (BLOCK_HW,)
                in_mask = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W) & hw_mask

                # input index: (((n * C_in + ic) * H + ih) * W + iw)
                in_base = (n * C_in + ic) * H
                in_index = in_base + ih * W + iw

                # load x values (masked)
                x_val = tl.load(x_ptr + in_index, mask=in_mask, other=0.0)

                # load weights for this oc tile: w[oc, ic, kh, kw]
                for j in range(BLOCK_OC):
                    oc_idx = oc_offsets[j]
                    if oc_mask[j]:
                        w_index = oc_idx * w_stride_oc + ic * w_stride_ic + kh * w_stride_kh + kw * w_stride_kw
                        w_val = tl.load(w_ptr + w_index)
                        acc[j] += x_val * w_val

    # store results y[n, oc, h, w] for all oc in tile
    # y index: n*y_stride_n + oc*y_stride_c + h*y_stride_h + w*y_stride_w
    for j in range(BLOCK_OC):

        if oc_mask[j]:
            y_ptrs = n * y_stride_n + oc_idx * y_stride_c + h * y_stride_h + w * y_stride_w
            # store acc[j] to each h,w in the tile
            tl.store(y_ptr + y_ptrs, acc[j], mask=hw_mask)


@triton.jit
def group_norm_affine_kernel(
    x_ptr,                # *float32 input tensor (N, C, H, W)
    weight_ptr,           # *float32 scale tensor (C,)
    bias_ptr,             # *float32 bias tensor (C,)
    y_ptr,                # *float32 output tensor (N, C, H, W)
    N, C, H, W,           # sizes
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    weight_stride_c, bias_stride_c,
    num_groups: tl.constexpr,          # 32
    eps,                              # epsilon
    BLOCK_HW: tl.constexpr,            # spatial tile
):
    # one program per (n, group)
    n = tl.program_id(0)
    g = tl.program_id(1)

    channels_per_group = C // num_groups
    group_start = g * channels_per_group

    # pass 1: compute sum and sumsq over group
    sum_g = 0.0
    sumsq_g = 0.0

    c = group_start
    while c < group_start + channels_per_group:
        c_local = c
        # loop over spatial
        hw = 0
        while hw < H * W:
            hw_idx = hw + tl.arange(0, BLOCK_HW)
            mask_hw = hw_idx < (H * W)
            h = hw_idx // W
            w = hw_idx % W
            x_ptrs = n * x_stride_n + c_local * x_stride_c + h * x_stride_h + w * x_stride_w
            x_vals = tl.load(x_ptr + x_ptrs, mask=mask_hw, other=0.0)
            sum_g += tl.sum(x_vals, axis=0)
            sumsq_g += tl.sum(x_vals * x_vals, axis=0)
            hw += BLOCK_HW
        c += 1

    mean = sum_g / (channels_per_group * H * W)
    var = sumsq_g / (channels_per_group * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # pass 2: normalize and apply affine
    c = group_start
    while c < group_start + channels_per_group:
        c_local = c
        weight_val = tl.load(weight_ptr + c_local * weight_stride_c)
        bias_val = tl.load(bias_ptr + c_local * bias_stride_c)
        hw = 0
        while hw < H * W:
            hw_idx = hw + tl.arange(0, BLOCK_HW)
            mask_hw = hw_idx < (H * W)
            h = hw_idx // W
            w = hw_idx % W
            x_ptrs = n * x_stride_n + c_local * x_stride_c + h * x_stride_h + w * x_stride_w
            x_vals = tl.load(x_ptr + x_ptrs, mask=mask_hw, other=0.0)
            y_vals = (x_vals - mean) * inv_std
            y_vals = y_vals * weight_val + bias_val
            y_ptrs = n * y_stride_n + c_local * y_stride_c + h * y_stride_h + w * y_stride_w
            tl.store(y_ptr + y_ptrs, y_vals, mask=mask_hw)
            hw += BLOCK_HW
        c += 1

@triton.jit
def silu_kernel_simple(N, C, H, W, total, y_ptr, x_ptr):
    n = tl.program_id(0)
    i = 0
    while i < total:
        # For contiguous tensors, linear index i maps to x_ptr[i], y_ptr[i]
        x_val = tl.load(x_ptr + i)
        # sigmoid
        sig = 1.0 / (1.0 + tl.exp(-x_val))
        y_val = x_val * sig
        tl.store(y_ptr + i, y_val)
        i += 1
    return


@triton.jit

    w_ptr,                # *float32 weights tensor (C_out, C_in, 3, 3)
    y_ptr,                # *float32 output tensor (B, C_out, H, W)
    N, C_in, C_out, H, W,             # sizes
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    w_stride_oc, w_stride_ic, w_stride_kh, w_stride_kw,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_OC: tl.constexpr,           # tile of output channels per program
    BLOCK_HW: tl.constexpr,           # spatial tile (e.g., 4096)
):
    # program ids
    n = tl.program_id(0)
    oc_block = tl.program_id(1)
    hw_block = tl.program_id(2)

    # oc tile
    oc_start = oc_block * BLOCK_OC
    oc_offsets = oc_start + tl.arange(0, BLOCK_OC)
    oc_mask = oc_offsets < C_out

    # spatial tile
    hw_start = hw_block * BLOCK_HW
    hw_offsets = hw_start + tl.arange(0, BLOCK_HW)
    hw_mask = hw_offsets < (H * W)
    h = hw_offsets // W
    w = hw_offsets % W

    # accumulator for output channels in tile
    acc = tl.zeros((BLOCK_OC,), dtype=tl.float32)

    # loop over input channels
    for ic in range(C_in):
        # loop over 3x3 taps
        for kh in range(3):
            for kw in range(3):
                # input coordinates with padding: ih=oh+kh-1, iw=ow+kw-1
                ih = h + kh - 1  # shape: (BLOCK_HW,)
                iw = w + kw - 1  # shape: (BLOCK_HW,)
                in_mask = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W) & hw_mask

                # input index: (((n * C_in + ic) * H + ih) * W + iw)
                in_base = (n * C_in + ic) * H
                in_index = in_base + ih * W + iw

                # load x values (masked)
                x_val = tl.load(x_ptr + in_index, mask=in_mask, other=0.0)

                # load weights for this oc tile: w[oc, ic, kh, kw]
                for j in range(BLOCK_OC):
                    oc_idx = oc_offsets[j]
                    if oc_mask[j]:
                        w_index = oc_idx * w_stride_oc + ic * w_stride_ic + kh * w_stride_kh + kw * w_stride_kw
                        w_val = tl.load(w_ptr + w_index)
                        acc[j] += x_val * w_val

    # store results y[n, oc, h, w] for all oc in tile
    # y index: n*y_stride_n + oc*y_stride_c + h*y_stride_h + w*y_stride_w
    for j in range(BLOCK_OC):
        oc_idx = oc_offsets[j]
        if oc_mask[j]:
            y_ptrs = n * y_stride_n + oc_idx * y_stride_c + h * y_stride_h + w * y_stride_w
            # store acc[j] to each h,w in the tile
            tl.store(y_ptr + y_ptrs, acc[j], mask=hw_mask)


@triton.jit
def group_norm_affine_kernel(
    x_ptr,                # *float32 input tensor (N, C, H, W)
    weight_ptr,           # *float32 scale tensor (C,)
    bias_ptr,             # *float32 bias tensor (C,)
    y_ptr,                # *float32 output tensor (N, C, H, W)
    N, C, H, W,           # sizes
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    weight_stride_c, bias_stride_c,
    num_groups: tl.constexpr,          # 32
    eps,                              # epsilon
    BLOCK_HW: tl.constexpr,            # spatial tile
):
    # one program per (n, group)
    n = tl.program_id(0)
    g = tl.program_id(1)

    channels_per_group = C // num_groups
    group_start = g * channels_per_group

    # pass 1: compute sum and sumsq over group
    sum_g = 0.0
    sumsq_g = 0.0

    c = group_start
    while c < group_start + channels_per_group:
        c_local = c
        hw = 0
        while hw < H * W:
            hw_idx = hw + tl.arange(0, BLOCK_HW)
            mask_hw = hw_idx < (H * W)
            h = hw_idx // W
            w = hw_idx % W
            x_ptrs = n * x_stride_n + c_local * x_stride_c + h * x_stride_h + w * x_stride_w
            x_vals = tl.load(x_ptr + x_ptrs, mask=mask_hw, other=0.0)
            sum_g += tl.sum(x_vals, axis=0)
            sumsq_g += tl.sum(x_vals * x_vals, axis=0)
            hw += BLOCK_HW
        c += 1

    mean = sum_g / (channels_per_group * H * W)
    var = sumsq_g / (channels_per_group * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # pass 2: normalize and apply affine
    c = group_start
    while c < group_start + channels_per_group:
        c_local = c
        weight_val = tl.load(weight_ptr + c_local * weight_stride_c)
        bias_val = tl.load(bias_ptr + c_local * bias_stride_c)
        hw = 0
        while hw < H * W:
            hw_idx = hw + tl.arange(0, BLOCK_HW)
            mask_hw = hw_idx < (H * W)
            h = hw_idx // W
            w = hw_idx % W
            x_ptrs = n * x_stride_n + c_local * x_stride_c + h * x_stride_h + w * x_stride_w
            x_vals = tl.load(x_ptr + x_ptrs, mask=mask_hw, other=0.0)
            y_vals = (x_vals - mean) * inv_std
            y_vals = y_vals * weight_val + bias_val
            y_ptrs = n * y_stride_n + c_local * y_stride_c + h * y_stride_h + w * y_stride_w
            tl.store(y_ptr + y_ptrs, y_vals, mask=mask_hw)
            hw += BLOCK_HW
        c += 1
