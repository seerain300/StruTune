import torch
import triton
import triton.language as tl


# Triton kernel: Conv3x3 (stride=1, padding=1, no bias), NCHW layout, float32 compute
@triton.jit
def conv3x3_nchw_fp32(x_ptr, w_ptr, y_ptr,
                      B, C_in, H_in, W_in, C_out, H_out, W_out,
                      BLOCK_IN: tl.constexpr):
    # program ids: (n, c_out, h_out, w_out)
    pid = tl.program_id(axis=0)
    hw_total = H_out * W_out
    n = pid // (C_out * hw_total)
    rem = pid % (C_out * hw_total)
    c_out = rem // hw_total
    w_out = rem % hw_total

    # Initialize output accumulator
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over input channels in chunks
    for ic_start in range(0, C_in, BLOCK_IN):
        c_offsets = ic_start + tl.arange(0, BLOCK_IN)
        mask_c = c_offsets < C_in

        # Accumulate convolution over 3x3 window with padding
        for kh in range(3):
            for kw in range(3):
                # Input coordinates with padding
                h_in = h_out + kh - 1  # padding=1
                w_in = w_out + kw - 1
                # Valid region mask for input coordinates
                valid_in = (h_in >= 0) & (h_in < H_in) & (w_in >= 0) & (w_in < W_in)

                # Compute base offsets for x and w
                # x offset: ((n * C_in + c_offsets) * H_in + h_in) * W_in + w_in
                # w offset: ((c_out * C_in + c_offsets) * 3 + kh) * 3 + kw
                # Build 2D offsets for x: shape [BLOCK_IN]
                x_offsets = (n * C_in + c_offsets) * H_in + h_in
                x_offsets = x_offsets * W_in + w_in
                # Build 2D offsets for w: shape [BLOCK_IN]
                w_offsets = (c_out * C_in + c_offsets) * 9 + (kh * 3 + kw)

                # Combined mask: valid input AND valid channel
                mask = mask_c & valid_in

                # Load x values for this (kh, kw) over input channels chunk
                x_vals = tl.load(x_ptr + x_offsets, mask=mask, other=0.0)
                # Load w values for this output channel over input channels chunk
                w_vals = tl.load(w_ptr + w_offsets, mask=mask_c, other=0.0)

                # Accumulate
                acc += tl.sum(x_vals * w_vals, axis=0)

    # Store result to y[n, c_out, h_out, w_out]
    y_offset = n * (C_out * H_out * W_out) + c_out * (H_out * W_out) + h_out * W_out + w_out
    tl.store(y_ptr + y_offset, acc)


# Triton kernel: GroupNorm with affine per (batch, group), float32 compute
@triton.jit
def groupnorm_affine_kernel(x_ptr, y_ptr, gn_weight_ptr, gn_bias_ptr,
                             B, C, H, W, num_groups, eps,
                             BLOCK_HW: tl.constexpr):
    # Each program handles one (n, group)
    pid = tl.program_id(axis=0)
    n = pid // num_groups
    g = pid % num_groups

    channels_per_group = C // num_groups
    c_start = g * channels_per_group

    # First pass: compute sum and sum of squares over group and all H*W
    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)

    for c in range(channels_per_group):
        c_idx = c_start + c
        for hw_start in range(0, H * W, BLOCK_HW):
            hw_offsets = hw_start + tl.arange(0, BLOCK_HW)
            mask_hw = hw_offsets < (H * W)
            # Linearize (c_idx, h, w) to 1D index for NCHW contiguous layout
            idx = ((n * C + c_idx) * (H * W)) + hw_offsets
            x_vals = tl.load(x_ptr + idx, mask=mask_hw, other=0.0)
            sum_val += tl.sum(x_vals, axis=0)
            sum_sq += tl.sum(x_vals * x_vals, axis=0)

    hw_total = H * W
    mean = sum_val / (channels_per_group * hw_total)
    var = sum_sq / (channels_per_group * hw_total) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine, store to y
    for c in range(channels_per_group):
        c_idx = c_start + c
        for hw_start in range(0, H * W, BLOCK_HW):
            hw_offsets = hw_start + tl.arange(0, BLOCK_HW)
            mask_hw = hw_offsets < (H * W)
            idx = ((n * C + c_idx) * (H * W)) + hw_offsets
            x_vals = tl.load(x_ptr + idx, mask=mask_hw, other=0.0)
            norm_vals = (x_vals - mean) * inv_std
            # Load affine parameters (per-channel)
            gamma = tl.load(gn_weight_ptr + c_idx)
            beta = tl.load(gn_bias_ptr + c_idx)
            y_vals = norm_vals * gamma + beta
            tl.store(y_ptr + idx, y_vals, mask=mask_hw)


# Triton kernel: SiLU (Swish) elementwise over flat memory
@triton.jit
def silu_kernel(x_ptr, y_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < total_elems
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    # SiLU: x * sigmoid(x), sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + offsets, y, mask=mask)


# Triton kernel: elementwise addition: y = x1 + x2 (both fp32, same shape)
@triton.jit
def add_residual_kernel(x1_ptr, x2_ptr, y_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < total_elems
    a = tl.load(x1_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(x2_ptr + offsets, mask=mask, other=0.0)
    tl.store(y_ptr + offsets, a + b, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self, conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                 conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor, eps: float):
        super().__init__()
        self.conv1_weight = conv1_weight
        self.norm1_weight = norm1_weight
        self.norm1_bias = norm1_bias
        self.conv2_weight = conv2_weight
        self.norm2_weight = norm2_weight
        self.norm2_bias = norm2_bias
        self.eps = eps

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        device = x.device
        dtype = x.dtype

        # Ensure inputs are contiguous and cast to float32 for Triton kernels
        B, C, H, W = x.shape
        C1 = self.conv1_weight.shape[0]
        C2 = self.conv2_weight.shape[0]

        # First conv
        x1 = x.contiguous()
        x1 = x1.to(torch.float32)
        y1 = torch.empty((B, C1, H, W), device=device, dtype=torch.float32)
        grid1 = (B * C1 * H * W,)
        conv3x3_nchw_fp32[grid1](
            x1, self.conv1_weight.contiguous().to(torch.float32),
            y1,
            B, C, H, W, C1, H, W,
            BLOCK_IN=64,
        )

        # First GroupNorm
        y1_gn = torch.empty_like(y1, device=device, dtype=torch.float32)
        num_groups = 32
        grid_gn1 = (B * num_groups,)
        groupnorm_affine_kernel[grid_gn1](
            y1, y1_gn,
            self.norm1_weight.contiguous().to(torch.float32),
            self.norm1_bias.contiguous().to(torch.float32),
            B, C1, H, W, num_groups, self.eps,
            BLOCK_HW=1024,
        )

        # First SiLU
        y1_silu = torch.empty_like(y1_gn, device=device, dtype=torch.float32)
        total1 = y1_gn.numel()
        grid_silu1 = (triton.cdiv(total1, 1024),)
        silu_kernel[grid_silu1](y1_gn, y1_silu, total1, BLOCK=1024)

        # Second conv
        y2 = torch.empty((B, C2, H, W), device=device, dtype=torch.float32)
        grid2 = (B * C2 * H * W,)
        conv3x3_nchw_fp32[grid2](
            y1_silu, self.conv2_weight.contiguous().to(torch.float32),
            y2,
            B, C1, H, W, C2, H, W,
            BLOCK_IN=64,
        )

        # Second GroupNorm
        y2_gn = torch.empty_like(y2, device=device, dtype=torch.float32)
        grid_gn2 = (B * num_groups,)
        groupnorm_affine_kernel[grid_gn2](
            y2, y2_gn,
            self.norm2_weight.contiguous().to(torch.float32),
            self.norm2_bias.contiguous().to(torch.float32),
            B, C2, H, W, num_groups, self.eps,
            BLOCK_HW=1024,
        )

        # Second SiLU
        y2_silu = torch.empty_like(y2_gn, device=device, dtype=torch.float32)
        total2 = y2_gn.numel()
        grid_silu2 = (triton.cdiv(total2, 1024),)
        silu_kernel[grid_silu2](y2_gn, y2_silu, total2, BLOCK=1024)

        # Residual addition: add original input x (cast to fp32 and same shape) to final output
        x_add = x.contiguous().to(torch.float32)
        out_add = torch.empty_like(x_add, device=device, dtype=torch.float32)
        total_add = x_add.numel()
        grid_add = (triton.cdiv(total_add, 1024),)
        add_residual_kernel[grid_add](x_add, y2_silu, out_add, total_add, BLOCK=1024)

        return out_add
