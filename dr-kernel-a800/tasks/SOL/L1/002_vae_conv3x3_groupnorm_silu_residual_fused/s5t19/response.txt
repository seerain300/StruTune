Design and plan
- Conv3x3: implement a Triton kernel conv3x3_kernel_hw_tiled that computes convolution stride=1, padding=1, no bias. Grid is (B, C_out, H, N_W_TILES). Each program computes a vector across W tiles for a given (n, c_out, h_out). It loops over input channels and 3x3 neighborhood, loads with masks for out-of-bounds (padding), accumulates, and stores.
- GroupNorm + SiLU:
  - group_norm_reduce_kernel: grid over (B, num_groups, channels_per_group). It loops over spatial tiles (N_TILES is a tl.constexpr) using masked loads to compute sum and sum of squares per (n, group, channel).
  - group_norm_apply_kernel: grid over (B, num_groups, channels_per_group, N_TILES). It applies normalization with precomputed mean and rstd, per-channel affine scale and bias, then SiLU.
- Residual add: elementwise Triton kernel residual_add_kernel that adds two tensors (out and x) with masks. It is launched for the final output.

All tensors are cast to float32 and made contiguous before launching. num_groups is fixed at 32, matching the original code. This should compile and run across varied H, W, B.

```python
import torch
import triton
import triton.language as tl


# Conv3x3: y = x * w, stride=1, padding=1, no bias
# Grid: (B, C_out, H, N_W_TILES)
@triton.jit
def conv3x3_kernel_hw_tiled(
    x_ptr,           # *f32, input [B, C_in, H, W]
    w_ptr,           # *f32, weight [C_out, C_in, 3, 3]
    y_ptr,           # *f32, output [B, C_out, H, W]
    B: tl.constexpr, C_in: tl.constexpr, C_out: tl.constexpr,
    H: tl.constexpr, W: tl.constexpr, N_W_TILES: tl.constexpr,
):
    n = tl.program_id(0)
    c_out = tl.program_id(1)
    h_out = tl.program_id(2)
    tile = tl.program_id(3)  # which W tile

    # Vector of W indices for this tile
    w_offsets = tile * tl.full((N_W_TILES,), 1, tl.int32) + tl.arange(0, N_W_TILES)
    mask_w = w_offsets < W

    # Accumulator for this vector across N_W_TILES
    acc = tl.zeros((N_W_TILES,), dtype=tl.float32)

    # Loop over input channels and 3x3 neighborhood with padding masks
    for c_in in range(0, C_in):
        for dh in range(-1, 2):
            h_in = h_out + dh
            valid_h = (h_in >= 0) & (h_in < H)
            for dw in range(-1, 2):
                w_in = w_offsets + dw
                valid = mask_w & (w_in >= 0) & (w_in < W) & valid_h
                # Input offsets for the vector
                x_off = (((n * C_in) + c_in) * H + h_in) * W + w_in
                x_val = tl.load(x_ptr + x_off, mask=valid, other=0.0)
                # Weight scalar for this (c_out, c_in, dh, dw)
                w_off = (((c_out * C_in) + c_in) * 9) + (dh + 1) * 3 + (dw + 1)
                w_val = tl.load(w_ptr + w_off)
                acc += x_val * w_val

    # Store results for this tile
    y_off = (((n * C_out) + c_out) * H + h_out) * W + w_offsets
    tl.store(y_ptr + y_off, acc, mask=mask_w)


# Triton kernel to compute per-channel (n, group, channel) mean and rstd over spatial HW
# Grid: (B * num_groups * channels_per_group,)
@triton.jit
def group_norm_reduce_kernel(
    x_ptr,           # *f32, input [B, C, H, W]
    mean_ptr,        # *f32, output [C]
    rstd_ptr,        # *f32, output [C]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    num_groups: tl.constexpr,
):
    pid = tl.program_id(0)
    # Map pid to (n, group, c)
    ch_per_group = C // num_groups
    n = pid // (num_groups * ch_per_group)
    g = (pid // ch_per_group) % num_groups
    c = pid % ch_per_group + g * ch_per_group

    # Accumulate sum and sum of squares over HW
    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)

    N_TILES = (H * W + 31) // 32  # tile count for HW over 32 elements per tile
    for t in range(0, N_TILES):
        start = t * 32
        idxs = start + tl.arange(0, 32)
        mask = idxs < (H * W)
        h = idxs // W
        w = idxs % W
        x_off = (((n * C) + c) * H + h) * W + w
        x_vec = tl.load(x_ptr + x_off, mask=mask, other=0.0)
        sum_val += tl.sum(x_vec, axis=0)
        sum_sq += tl.sum(x_vec * x_vec, axis=0)

    size = H * W
    mean = sum_val / size
    var = sum_sq / size - mean * mean
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    tl.store(mean_ptr + c, mean)
    tl.store(rstd_ptr + c, rstd)


# Triton kernel to apply GroupNorm affine and SiLU to y using precomputed mean/rstd
# Grid: (B * num_groups * channels_per_group, N_TILES)
@triton.jit
def group_norm_apply_kernel(
    y_ptr,           # *f32, input/output [B, C, H, W] (we normalize in-place)
    x_ptr,           # *f32, input [B, C, H, W] (unused, kept for symmetry)
    mean_ptr,        # *f32, [C]
    rstd_ptr,        # *f32, [C]
    gamma_ptr,       # *f32, [C] (norm scale)
    beta_ptr,        # *f32, [C] (norm bias)
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    num_groups: tl.constexpr,
):
    pid = tl.program_id(0)  # across (n, group, c)
    tile = tl.program_id(1)  # across tiles of HW

    ch_per_group = C // num_groups
    n = pid // (num_groups * ch_per_group)
    g = (pid // ch_per_group) % num_groups
    c = pid % ch_per_group + g * ch_per_group

    mean = tl.load(mean_ptr + c)
    rstd = tl.load(rstd_ptr + c)
    gamma = tl.load(gamma_ptr + c)
    beta = tl.load(beta_ptr + c)

    start = tile * 32
    idxs = start + tl.arange(0, 32)
    mask = idxs < (H * W)
    h = idxs // W
    w = idxs % W

    # Load current y vector for this (n, c, h, w) tile
    y_off = (((n * C) + c) * H + h) * W + w
    y_vec = tl.load(y_ptr + y_off, mask=mask, other=0.0)

    # Normalize + affine
    norm = (y_vec - mean) * rstd
    norm = norm * gamma + beta

    # SiLU: x * sigmoid(x)
    sig = 1.0 / (1.0 + tl.exp(-norm))
    out_vec = norm * sig

    tl.store(y_ptr + y_off, out_vec, mask=mask)


# Triton elementwise residual add: out = a + b
@triton.jit
def residual_add_kernel(
    a_ptr, b_ptr, out_ptr,
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
):
    n = tl.program_id(0)
    c = tl.program_id(1)
    hw = tl.program_id(2)
    h = hw // W
    w = hw % W
    idx = (((n * C) + c) * H + h) * W + w
    a_val = tl.load(a_ptr + idx)
    b_val = tl.load(b_ptr + idx)
    tl.store(out_ptr + idx, a_val + b_val)


class ModelNew(torch.nn.Module):
    def __init__(self, num_groups: int = 32, eps: float = 1e-5):
        super().__init__()
        self.num_groups = num_groups
        self.eps = eps

    def forward(self, x: torch.Tensor,
                conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor):
        """
        x: (B, C, H, W), conv weights: (C_out, C_in, 3, 3), norm scales/bias: (C,)
        Returns: (B, C_out2, H, W)
        """
        # Ensure CUDA and float32 for Triton
        assert x.is_cuda, "Inputs must be on CUDA for Triton kernels"
        B, C_in, H, W = x.shape

        x_f32 = x.contiguous().to(torch.float32)
        # First conv: C_in -> C_out1
        C_out1 = conv1_weight.shape[0]
        out1 = torch.empty((B, C_out1, H, W), device=x.device, dtype=torch.float32)
        grid1 = (B, C_out1, H, (W + 31) // 32)  # one tile of 32 columns per program along W
        conv3x3_kernel_hw_tiled[grid1](
            x_f32, conv1_weight.contiguous().to(torch.float32), out1,
            B=B, C_in=C_in, C_out=C_out1, H=H, W=W, N_W_TILES=32,
            num_warps=4, num_stages=2,
        )

        # GroupNorm + SiLU for first block
        mean1 = torch.empty(C_out1, device=x.device, dtype=torch.float32)
        rstd1 = torch.empty(C_out1, device=x.device, dtype=torch.float32)
        assert C_out1 % self.num_groups == 0, "C_out1 must be divisible by num_groups"
        ch_per_group1 = C_out1 // self.num_groups
        grid_reduce1 = (B * self.num_groups * ch_per_group1,)
        group_norm_reduce_kernel[grid_reduce1](
            out1, mean1, rstd1,
            B=B, C=C_out1, H=H, W=W, num_groups=self.num_groups,
            num_warps=4, num_stages=2,
        )
        out1_norm = torch.empty_like(out1)
        grid_apply1 = (B * self.num_groups * ch_per_group1, (H * W + 31) // 32)
        group_norm_apply_kernel[grid_apply1](
            out1, out1, mean1, rstd1, norm1_weight.contiguous().to(torch.float32), norm1_bias.contiguous().to(torch.float32),
            B=B, C=C_out1, H=H, W=W, num_groups=self.num_groups,
            num_warps=4, num_stages=2,
        )

        # Second conv: C_out1 -> C_out2
        C_out2 = conv2_weight.shape[0]
        out2 = torch.empty((B, C_out2, H, W), device=x.device, dtype=torch.float32)
        grid2 = (B, C_out2, H, (W + 31) // 32)
        conv3x3_kernel_hw_tiled[grid2](
            out1_norm, conv2_weight.contiguous().to(torch.float32), out2,
            B=B, C_in=C_out1, C_out=C_out2, H=H, W=W, N_W_TILES=32,
            num_warps=4, num_stages=2,
        )

        # GroupNorm + SiLU for second block
        mean2 = torch.empty(C_out2, device=x.device, dtype=torch.float32)
        rstd2 = torch.empty(C_out2, device=x.device, dtype=torch.float32)
        assert C_out2 % self.num_groups == 0, "C_out2 must be divisible by num_groups"
        ch_per_group2 = C_out2 // self.num_groups
        grid_reduce2 = (B * self.num_groups * ch_per_group2,)
        group_norm_reduce_kernel[grid_reduce2](
            out2, mean2, rstd2,
            B=B, C=C_out2, H=H, W=W, num_groups=self.num_groups,
            num_warps=4, num_stages=2,
        )
        out2_norm = torch.empty_like(out2)
        grid_apply2 = (B * self.num_groups * ch_per_group2, (H * W + 31) // 32)
        group_norm_apply_kernel[grid_apply2](
            out2, out2, mean2, rstd2, norm2_weight.contiguous().to(torch.float32), norm2_bias.contiguous().to(torch.float32),
            B=B, C=C_out2, H=H, W=W, num_groups=self.num_groups,
            num_warps=4, num_stages=2,
        )

        # Final residual add: out2_norm + x
        out = torch.empty_like(out2_norm)
        residual_add_kernel[(B, C_out2, H * W)](
            out2_norm, x_f32, out,
            B=B, C=C_out2, H=H, W=W,
            num_warps=4, num_stages=2,
        )

        return out
```
