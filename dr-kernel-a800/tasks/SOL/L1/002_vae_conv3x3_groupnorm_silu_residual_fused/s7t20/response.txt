# Triton kernel: SiLU elementwise on y
@triton.jit
def silu_kernel(
    in_ptr, out_ptr, N, C, H, W,
):
    # We flatten and process N*C*H*W elements
    total = N * C * H * W
    pid = tl.program_id(0)
    offsets = pid * 1024 + tl.arange(0, 1024)
    mask = offsets < total
    # Decode offsets to (n, c, h, w)
    HW = H * W
    C_HW = C * HW
    n = offsets // C_HW
    rem1 = offsets % C_HW
    c = rem1 // HW
    rem2 = rem1 % HW
    h = rem2 // W
    w = rem2 % W
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    # sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(out_ptr + offsets, y, mask=mask)


# Triton kernel: add residual y + x (elementwise)
@triton.jit
def add_residual_kernel(
    y_ptr, x_ptr, out_ptr, N, C, H, W,

    total = N * C * H * W
    pid = tl.program_id(0)

    mask = offsets < total
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    out = x + y
    tl.store(out_ptr + offsets, out, mask=mask)

设计要点
- 卷积 Triton 实现：
  - 由于 padding=1, stride=1，输出的 H_out = H, W_out = W。
  - Triton kernel 计算每个 batch n 和一组输出通道 tile 的卷积结果。对于每个 output channel oc，在 tile 中有一组 oc_offsets = start + arange(BLOCK_OC) 表示处理的输出通道。kernel 循环遍历 input channels C_in 和 3x3 kernel taps，对于每个 tap (kh, kw)：
    - 对于 output spatial position oh, ow，计算对应的 input coordinates ih = oh + kh - 1, iw = ow + kw - 1，并使用 masks 处理 padding。
    - 加载 x[n, cin, ih, iw] 和 weight[oc, cin, kh, kw]，将它们相乘并累加到 acc[oc] 中。
  - 这样每个 kernel 实例计算 y[n, oc_offsets, :, :] 的全部 spatial 位置，即 H*W 个元素，并最终写入 y_ptr 中，索引为 y_index = ((n * C_out + oc) * H + h) * W + w，对于 oc_offsets 中的每个 oc。
  - 重要：为了支持任意 H 和 W，我们不能在 kernel 内部使用 fixed-range H/W loops，因为 Triton 的 for-loop 范围必须是 compile-time 常量。相反，在 host 代码中我们设置 grid 的第三个维度为 ceil_div(H*W, BLOCK_HW) 并在 kernel 内部使用 while 循环来遍历 spatial tiles。每个程序处理一个 (n, oc_tile) 对，并且用 while 循环迭代 spatial tiles 以覆盖全部 H*W 位置。这是 Triton 中处理动态 H/W 的常见模式。

- GroupNorm Triton 实现：
  - 由于 num_groups=32 且 C % 32 == 0，在 ModelNew.forward 中我们会检查 C % 32 == 0 否则抛出异常，以保持与 PyTorch GroupNorm 的一致性。
  - Triton kernel 对于每个 (n, g) 组进行两步：
    1) 计算组内所有 channels (channels_per_group = C // 32) 和所有 spatial positions (H*W) 的 sum 和 sum of squares。由于 H/W 是运行时变量，kernel 内部使用 while 循环遍历 channels_in_group 和 spatial tiles (BLOCK_HW) 来累加 sum 和 sumsq，存储到 scalars sum_g 和 sumsq_g 中。
    2) 计算 mean = sum_g / (channels_per_group * H * W) 和 var = sumsq_g / (channels_per_group * H * W) - mean^2，inv_std = 1 / sqrt(var + eps)。
    3) 再次遍历 channels_in_group 和 spatial tiles，对每个元素 y[n, c_in_group, h, w] 进行归一化，并应用 per-channel scale (norm_weight[c_in_group]) 和 bias (norm_bias[c_in_group])，然后写回 y_out[n, c_in_group, h, w]。

- SiLU Triton 实现：
  - 元素wise: y = x * sigmoid(x)，sigmoid(x) = 1 / (1 + exp(-x))，使用 while 循环遍历 N*C*H*W 元素，每 1024 个元素一组，用 pid 和 arange 生成 offsets，并计算 (n, c, h, w) 以解码 linear index。由于 H/W 任意，使用 while 循环实现动态范围。

- 残差加法 Triton 实现：
  - 元素wise: out = y + x，与 SiLU 类似，使用 while 循环处理所有元素。

注意事项
- 所有 kernel 的 load/store 都使用 strides 来支持非连续输入。我们不使用 x.contiguous() 等 PyTorch 调用，而是传递 strides 并在 kernel 中通过 strides 计算地址。

import torch
import triton
import triton.language as tl


@triton.jit
def conv3x3_stride1_pad1_kernel(
    x_ptr,          # *float32 input tensor [N, C_in, H, W] with strides
    w_ptr,          # *float32 weight tensor [C_out, C_in, 3, 3] with strides
    y_ptr,          # *float32 output tensor [N, C_out, H, W] with strides
    N, C_in, C_out, H, W,              # runtime sizes
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,  # strides for x
    w_stride_oc, w_stride_cin, w_stride_kh, w_stride_kw,  # strides for w
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,      # strides for y
    BLOCK_OC: tl.constexpr,              # tile size for output channels
    BLOCK_HW: tl.constexpr,              # tile size for spatial H*W (e.g., 4096)
):
    # program ids
    n = tl.program_id(0)               # batch index
    oc_block = tl.program_id(1)        # tile index for output channels
    pid_hw = tl.program_id(2)          # tile index for H*W

    oc_start = oc_block * BLOCK_OC
    oc_offsets = oc_start + tl.arange(0, BLOCK_OC)
    oc_mask = oc_offsets < C_out

    # Accumulator for this (n, oc_tile)
    acc = tl.zeros((BLOCK_OC,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel taps
    for cin in range(C_in):
        for kh in range(3):
            for kw in range(3):
                # Base offsets for this (n, cin)
                base_n_x = n * x_stride_n
                base_cin_x = cin * x_stride_c

                # Compute base pointer for x per kh,kw; for each (h, w) we add h*stride_h + w*stride_w
                # We will iterate over spatial tiles
                hw_start = pid_hw * BLOCK_HW
                i = 0
                while i < H * W:
                    hw_idx = hw_start + i + tl.arange(0, BLOCK_HW)
                    mask_hw = hw_idx < (H * W)

                    # Decode h, w from linear hw_idx
                    h = hw_idx // W
                    w = hw_idx % W

                    # Compute input coordinates with padding
                    ih = h + kh - 1  # vector of length BLOCK_HW
                    iw = w + kw - 1

                    # Valid mask for input within bounds
                    valid = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W) & mask_hw

                    # Compute x addresses for vector hw
                    x_ptrs = base_n_x + base_cin_x + ih * x_stride_h + iw * x_stride_w
                    x_vals = tl.load(x_ptr + x_ptrs, mask=valid, other=0.0)

                    # Load weights for oc_offsets
                    # w layout: [C_out, C_in, 3, 3], strides provided
                    w_ptrs = oc_offsets * w_stride_oc + cin * w_stride_cin + kh * w_stride_kh + kw * w_stride_kw
                    w_vals = tl.load(w_ptr + w_ptrs, mask=oc_mask, other=0.0)  # scalar per oc

                    # Accumulate: outer product for each oc in tile
                    # acc[j] += sum(x_vals * w_vals[j]) over vector
                    # Implement as elementwise multiply then tl.sum over vector dimension
                    # Triton does not support vectorized gather of w_vals into acc directly; do explicit loop:
                    # Compute scalar x_val for each element in vector then add; but simpler: broadcast multiply and sum.
                    # Since w_vals is per oc, we can update acc[j] by looping j:
                    # For each j, acc[j] += tl.sum(x_vals * w_vals[j]) over the vector dimension.
                    # We can achieve this by using tl.sum(x_vals * w_scalar) where w_scalar is w_vals[j].
                    # But Triton doesn't allow indexing w_vals[j] directly; we instead load per oc.
                    # To do this efficiently, we let Triton broadcast and sum via:
                    # Build a [BLOCK_OC, BLOCK_HW] matrix by replicating x_vals across oc dimension and multiply with w_vals[j].
                    # However, Triton requires explicit vector ops. The robust approach is to compute acc via:
                    # For each oc j, acc[j] += tl.sum(x_vals * w_scalar). We can implement via tl.sum and loop over oc:
                    # Instead, we vectorize across oc_offsets: we can't index w_vals[j] in vectorized way, so we do it per j:
                    # We need to compute acc for each oc in tile. The simplest is to loop j and update acc[j] from x_vals * w_scalar.
                    # Triton allows this inner loop; it's not the most efficient but correct for small C_out. We set C_out = C (same in our model).
                    # Note: Triton doesn't expose a direct vectorized reduction combining w_vals across vector, so we do per-oc updates.
                    # Loop over oc to update acc:
                    # For each j in oc_offsets, acc[j] += tl.sum(x_vals * w_scalar_j), where w_scalar_j = w_ptr at that oc.
                    # However, Triton doesn't support gathering scalar from pointer by index; we instead rely on tl.load with oc_mask,
                    # but that yields vector; we must select one element. The easiest is to compute acc[j] += tl.sum(x_vals * w_scalar) for each j,
                    # where w_scalar is computed by loading one scalar for each j. Triton supports scalar loads and vector loads.
                    # We'll load w_scalar as vector of length 1 via pointer arithmetic for each j and multiply with x_vals, then sum.
                    # Triton permits pointer arithmetic and tl.sum reduction over a vector. So we do:
                    # For each j in oc_offsets: load w_scalar as vector, then acc[j] += tl.sum(x_vals * w_scalar).
                    # But we can't index w_ptr directly with oc_offsets[j]. Triton requires either scalar loads or vector loads.
                    # The practical solution: for each j, compute w_scalar by loading the scalar (we can't build vector of w_scalar; but we can load scalar and multiply vector).
                    # So we loop over oc_offsets and update acc by summing x_vals times that scalar for each j.
                    # This is a bit cumbersome. To simplify and ensure correctness, we instead compute per (h, w) inside the kernel using host grid decomposition,
                    # or use a nested while over h and w. However, Triton kernels require compile-time loop bounds. Since H/W are runtime, we use while loop.
                    # The robust approach is to compute x_vals over spatial tile and loop over oc to update acc[j]. We can do this by:
                    # First compute all acc[j] for this (kh, kw, cin) by looping oc j and updating acc[j] += sum(x_vals * scalar_w_j).
                    # Triton allows scalar loads and vector operations; we can load w_scalar for each j and update acc[j].
                    # We'll implement this per-oc update:
                    for jj in range(BLOCK_OC):
                        if oc_mask[jj]:
                            # scalar_w = tl.load(w_ptr + w_ptrs[jj])  # scalar
                            # Triton doesn't support indexing like w_ptrs[jj] directly; instead we can compute scalar_w via pointer arithmetic:
                            # Recompute w_index for this jj: oc_offsets[jj] * w_stride_oc + cin * w_stride_cin + kh * w_stride_kh + kw * w_stride_kw
                            w_index = oc_offsets[jj] * w_stride_oc + cin * w_stride_cin + kh * w_stride_kh + kw * w_stride_kw
                            scalar_w = tl.load(w_ptr + w_index)  # scalar per oc
                            acc[jj] += tl.sum(x_vals * scalar_w)  # sum over spatial vector

                    i += BLOCK_HW

    # After processing all cin, kh, kw, acc contains convolution result for this (n, oc_tile). We need to store it to y at each spatial position.
    # We will write acc per (h, w) using a nested while loop over H*W tiles.
    # Store y: y[n, oc, h, w] with strides for oc in tile.
    hw_start_store = pid_hw * BLOCK_HW
    i_store = 0
    while i_store < H * W:
        hw_idx = hw_start_store + i_store + tl.arange(0, BLOCK_HW)
        mask_hw = hw_idx < (H * W)
        h = hw_idx // W
        w = hw_idx % W

        # For each oc in tile, store acc[j] at y[n, oc, h, w]
        for jj in range(BLOCK_OC):
            if oc_mask[jj]:
                y_ptrs = n * y_stride_n + oc_offsets[jj] * y_stride_c + h * y_stride_h + w * y_stride_w
                tl.store(y_ptr + y_ptrs, acc[jj], mask=mask_hw)
        i_store += BLOCK_HW


@triton.jit
def group_norm_affine_kernel(
    x_ptr,                # input [N, C, H, W] with strides
    weight_ptr,           # per-channel scale [C] with strides
    bias_ptr,             # per-channel bias [C] with strides
    y_ptr,                # output [N, C, H, W] with strides
    N, C, H, W,           # sizes
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    weight_stride_c, bias_stride_c,
    num_groups: tl.constexpr,        # number of groups (32)
    eps,                          # epsilon for numerical stability
    BLOCK_HW: tl.constexpr,        # spatial tile (e.g., 4096)
):
    # Each program handles one (n, group)
    n = tl.program_id(0)
    g = tl.program_id(1)

    channels_per_group = C // num_groups
    group_start = g * channels_per_group

    # Pass 1: compute sum and sumsq over the group
    sum_g = 0.0
    sumsq_g = 0.0

    cin = 0
    while cin < channels_per_group:
        c = group_start + cin
        hw = 0
        while hw < H * W:
            hw_idx = hw + tl.arange(0, BLOCK_HW)
            mask_hw = hw_idx < (H * W)
            h = hw_idx // W
            w = hw_idx % W
