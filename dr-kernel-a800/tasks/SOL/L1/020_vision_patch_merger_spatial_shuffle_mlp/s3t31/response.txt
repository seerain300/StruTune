import torch
import math
import triton
import triton.language as tl


@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32

    One program per row. Compute mean and variance (fp32), normalize, apply affine, store bfloat16.
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: compute mean and variance (fp32)
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


@triton.jit
def _gelu_tanh_kernel(x_ptr, y_ptr, N, C, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise GELU using tanh approximation:
    gelu(x) ≈ 0.5 * x * (1 + tanh(√(2/π) * (x + 0.044715 * x^3)))
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C]
    """
    row = tl.program_id(0)
    if row >= N:
        return
    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        x3 = x * x * x
        c = 0.7978845608028654  # sqrt(2/pi)
        t = c * (x + 0.044715 * x3)
        gelu = 0.5 * x * (1.0 + tl.tanh(t))
        tl.store(y_row_ptr + offs, gelu.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


@triton.jit
def _linear_row_gemm_kernel(x_ptr, w_ptr, b_ptr, out_ptr,
                             N, C_in, C_out, K,
                             BLOCK_K: tl.constexpr):
    """
    Compute out[i, j] = dot(x[i, :], W[j, :]) + b[j] for all i in [0, N) and j in [0, C_out).
    This kernel launches one program per (i, j). For simplicity, N and C_out are runtime,
    but we use while loops; for performance we can set N and C_out via meta-params if constexpr.
    Here we keep dynamic N and C_out. The workload sizes in the provided JSON are typical,
    and this structure ensures correctness. Note: For best perf, consider constexpr N and C_out.

    x_ptr: *bf16, shape [N, C_in]
    w_ptr: *bf16, shape [C_out, C_in]
    b_ptr: *bf16, shape [C_out]
    out_ptr: *bf16, shape [N, C_out]
    """
    i = tl.program_id(0)  # row in x/out
    j = tl.program_id(1)  # output feature index
    if i >= N or j >= C_out:
        return

    # Accumulate in fp32
    acc = 0.0
    # dot over K dimension
    k = 0
    while k < K:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # Load x[i, offs_k]
        x_vec = tl.load(x_ptr + i * C_in + offs_k, mask=mask_k, other=0.0).to(tl.float32)
        # Load W[j, offs_k]
        w_vec = tl.load(w_ptr + j * C_in + offs_k, mask=mask_k, other=0.0).to(tl.float32)
        # acc += sum(x_vec * w_vec)
        acc += tl.sum(x_vec * w_vec, axis=0)
        k += BLOCK_K

    # Add bias
    bj = tl.load(b_ptr + j).to(tl.float32)
    y = acc + bj
    # Store
    tl.store(out_ptr + i * C_out + j, y.to(tl.bfloat16))


def _run_triton_layer_norm(hidden: torch.Tensor,
                            ln_weight: torch.Tensor,
                            ln_bias: torch.Tensor,
                            eps: float,
                            block_size: int = 1024) -> torch.Tensor:
    """
    Launch Triton LayerNorm kernel on hidden [N, C], return normalized + affine.
    """
    N, C = hidden.shape
    y = torch.empty_like(hidden)
    grid = (N,)
    _layer_norm_kernel[grid](
        hidden, y, ln_weight, ln_bias,
        N, C, eps,
        BLOCK_SIZE=block_size
    )
    return y


def _run_triton_gelu(hidden: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Launch Triton GELU kernel on hidden [N, C], returns GELU output.
    """
    N, C = hidden.shape
    y = torch.empty_like(hidden)
    grid = (N,)
    _gelu_tanh_kernel[grid](
        hidden, y,
        N, C,
        BLOCK_SIZE=block_size
    )
    return y
