import math
import torch
import triton
import triton.language as tl


@triton.jit
def _layer_norm_affine_kernel(
    x_ptr,         # *bf16, input [M, K]
    weight_ptr,    # *bf16, [K]
    bias_ptr,      # *bf16, [K]
    y_ptr,         # *bf16, output [M, K]
    M, K, eps,     # int32, float32
    BLOCK: tl.constexpr,
):
    # One program per row
    row = tl.program_id(axis=0)
    if row >= M:
        return
    # First pass: compute mean and variance in fp32
    acc1 = tl.zeros((), dtype=tl.float32)
    acc2 = tl.zeros((), dtype=tl.float32)
    for k0 in range(0, K, BLOCK):
        offs = k0 + tl.arange(0, BLOCK)
        mask = offs < K
        x = tl.load(x_ptr + row * K + offs, mask=mask, other=0.0).to(tl.float32)
        acc1 += tl.sum(x, axis=0)
        acc2 += tl.sum(x * x, axis=0)
    mean = acc1 / K
    var = acc2 / K - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for k0 in range(0, K, BLOCK):
        offs = k0 + tl.arange(0, BLOCK)
        mask = offs < K
        x = tl.load(x_ptr + row * K + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_ptr + row * K + offs, y.to(tl.bfloat16), mask=mask)


@triton.jit
def _gemm_bias_kernel(
    A_ptr,         # *bf16, input [M, K]
    W_ptr,         # *bf16, weight [N, K] (row-major: N is fast dim)
    B_ptr,         # *bf16, bias [N]
    C_ptr,         # *bf16, output [M, N]
    M, N, K,       # int32 sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    m0 = pid_m * BLOCK_M
    n0 = pid_n * BLOCK_N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k0 in range(0, K, BLOCK_K):
        k_offs = k0 + tl.arange(0, BLOCK_K)
        # Load A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + m0 * K + tl.arange(0, BLOCK_M)[:, None] * K + k_offs[None, :]
        a_mask = (m0 + tl.arange(0, BLOCK_M))[:, None] < M
        a_mask = a_mask & (k_offs[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load W tile as [BLOCK_K, BLOCK_N] by indexing W[n, k]
        w_ptrs = W_ptr + n0 * K + tl.arange(0, BLOCK_N)[None, :] * K + k_offs[:, None]
        w_mask = (n0 + tl.arange(0, BLOCK_N))[None, :] < N
        w_mask = w_mask & (k_offs[:, None] < K)
        w = tl.load(w_ptrs, mask=w_mask, other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, w)

    # Add bias
    bias = tl.load(B_ptr + n0 + tl.arange(0, BLOCK_N), mask=(n0 + tl.arange(0, BLOCK_N)) < N, other=0.0).to(tl.float32)
    acc += bias[None, :]

    # Store result to C
    c_ptrs = C_ptr + m0 * N + tl.arange(0, BLOCK_N)[None, :] * M + tl.arange(0, BLOCK_M)[:, None]
    c_mask = (m0 + tl.arange(0, BLOCK_M))[:, None] < M
    c_mask = c_mask & ((n0 + tl.arange(0, BLOCK_N))[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.bfloat16), mask=c_mask)


@triton.jit
def _gelu_kernel(
    x_ptr,         # *bf16, input [M, N]
    y_ptr,         # *bf16, output [M, N]
    M, N,          # int32
    BLOCK_N: tl.constexpr,
):
    row = tl.program_id(axis=0)
    col_blk = tl.program_id(axis=1)
    offs = col_blk * BLOCK_N + tl.arange(0, BLOCK_N)
    mask = offs < N
    x = tl.load(x_ptr + row * N + offs, mask=mask, other=0.0).to(tl.float32)
    # GELU tanh approximation
    # gelu(x) = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 x^3)))
    c0 = 0.7978845608028654  # sqrt(2/pi)
    c1 = 0.044715
    x3 = x * x * x
    gelu = 0.5 * x * (1.0 + tl.tanh(c0 * (x + c1 * x3)))
    tl.store(y_ptr + row * N + offs, gelu.to(tl.bfloat16), mask=mask)


def _launch_layer_norm_affine(x, ln_weight, ln_bias, out):
    M = x.shape[0]
    K = x.shape[1]
    BLOCK = 256
    grid = (M,)
    _layer_norm_affine_kernel[grid](
        x, ln_weight, ln_bias, out,
        M, K, 1e-6,
        BLOCK=BLOCK, num_warps=4, num_stages=2
    )

    N = W.shape[0]
    K = A.shape[1]
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    N = X.shape[1]
    BLOCK_N = 256
    grid = (M, triton.cdiv(N, BLOCK_N))

        M, N,
        BLOCK_N=BLOCK_N, num_warps=4, num_stages=2
    )
