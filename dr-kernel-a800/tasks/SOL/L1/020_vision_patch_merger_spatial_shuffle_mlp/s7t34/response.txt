```python
import math
import torch
import triton
import triton.language as tl


# Triton LayerNorm + affine: one program per row (patch), compute mean/var over features, normalize, affine
@triton.jit
def layernorm_affine_kernel(
    hidden_ptr,       # *bf16, [num_patches, hidden_size]
    ln_weight_ptr,    # *bf16, [hidden_size]
    ln_bias_ptr,      # *bf16, [hidden_size]
    out_ptr,          # *bf16, [num_patches, hidden_size]
    num_patches: tl.int32,
    hidden_size: tl.int32,
    eps: tl.float32,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # row base pointers
    hidden_row_ptr = hidden_ptr + pid * hidden_size
    out_row_ptr = out_ptr + pid * hidden_size

    # compute sum and sum of squares in fp32
    sum_val = 0.0
    sum_sq = 0.0
    for c in range(0, hidden_size, BLOCK_C):
        offs = c + tl.arange(0, BLOCK_C)
        mask = offs < hidden_size
        x = tl.load(hidden_row_ptr + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=0)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)

    mean = sum_val / hidden_size
    var = sum_sq / hidden_size - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # write normalized + affine
    for c in range(0, hidden_size, BLOCK_C):
        offs = c + tl.arange(0, BLOCK_C)
        mask = offs < hidden_size
        x = tl.load(hidden_row_ptr + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        norm = (x_f32 - mean) * inv_std
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = norm * w + b
        tl.store(out_row_ptr + offs, y.to(tl.bfloat16), mask=mask)


# Triton spatial reorder (2x2 merge) from LayerNorm output to first linear input
@triton.jit
def spatial_shuffle_to_fc1_kernel(
    ln_out_ptr,       # *bf16, [num_patches, hidden_size]
    grid_thw_ptr,     # *int64, [num_grids, 3]
    out_ptr,          # *bf16, [num_merged_patches, hidden_size_expanded]
    num_patches: tl.int32,
    hidden_size: tl.int32,
    hidden_size_expanded: tl.int32,
    merge_size: tl.int32,  # should be 2
    BLOCK_C: tl.constexpr,
):
    grid_id = tl.program_id(axis=0)
    t = tl.load(grid_thw_ptr + grid_id * 3 + 0).to(tl.int32)
    h = tl.load(grid_thw_ptr + grid_id * 3 + 1).to(tl.int32)
    w = tl.load(grid_thw_ptr + grid_id * 3 + 2).to(tl.int32)

    h_merged = h // merge_size
    w_merged = w // merge_size

    # For each original patch in this grid and each feature
    for p in range(0, t * h * w):
        base = ln_out_ptr + p * hidden_size
        for c in range(0, hidden_size, BLOCK_C):
            offs = c + tl.arange(0, BLOCK_C)
            mask = offs < hidden_size
            x = tl.load(base + offs, mask=mask, other=0.0).to(tl.bfloat16)
            # merged coordinates
            # i0 = p // w, j0 = p % w
            i0 = (p // w).to(tl.int32)
            j0 = (p % w).to(tl.int32)
            i_merged = i0 // merge_size
            j_merged = j0 // merge_size
            target_row = t * h_merged * w_merged + i_merged * w_merged + j_merged
            tl.store(out_ptr + target_row * hidden_size_expanded + offs, x, mask=mask)


# Triton GEMM with bias epilogue: A [M, K], B [K, N], bias [N]
@triton.jit
def matmul_bias_kernel(
    A_ptr,        # *bf16, [M, K]
    B_ptr,        # *bf16, [K, N]
    bias_ptr,     # *bf16, [N]
    C_ptr,        # *bf16, [M, N]
    M: tl.int32, N: tl.int32, K: tl.int32,
    stride_am: tl.int32, stride_ak: tl.int32,
    stride_bk: tl.int32, stride_bn: tl.int32,
    stride_cm: tl.int32, stride_cn: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = A_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (k + offs_k[None, :] < K),
            other=0.0,
        ).to(tl.float32)
        b = tl.load(
            b_ptrs,
            mask=(k + offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        ).to(tl.float32)
        acc += tl.dot(a, b)

    c_ptrs = C_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # add bias
    bias = tl.load(bias_ptr + offs_n, mask=offs_n < N, other=0.0).to(tl.float32)
    acc += bias[None, :]
    tl.store(
        c_ptrs,
        acc.to(tl.bfloat16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# Triton elementwise GELU (tanh approximation)
@triton.jit
def gelu_tanh_kernel(
    inp_ptr,        # *bf16, [M, N]
    out_ptr,        # *bf16, [M, N]
    M: tl.int32, N: tl.int32,
    stride_im: tl.int32, stride_in: tl.int32,
    stride_om: tl.int32, stride_on: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    inp_ptrs = inp_ptr + offs_m[:, None] * stride_im + offs_n[None, :] * stride_in
    out_ptrs = out_ptr + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on

    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    x = tl.load(inp_ptrs, mask=mask, other=0.0).to(tl.float32)
    # tanh approximation for GELU:
    # y = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    c0 = 0.7978845608028654  # sqrt(2/pi)
    c1 = 0.044715
    x3 = x * x * x
    inner = c0 * (x + c1 * x3)
    y = 0.5 * x * (1.0 + tl.tanh(inner))
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(
        self,
        hidden: torch.Tensor,
        grid_thw: torch.Tensor,
        ln_weight: torch.Tensor,
        ln_bias: torch.Tensor,
        fc1_weight: torch.Tensor,
        fc1_bias: torch.Tensor,
        fc2_weight: torch.Tensor,
        fc2_bias: torch.Tensor,
        eps: float,
    ):
        """
        hidden: [num_patches, hidden_size] bf16
        grid_thw: [num_grids, 3] int64
        ln_weight: [hidden_size] bf16
        ln_bias: [hidden_size] bf16
        fc1_weight: [hidden_size_expanded, hidden_size_expanded] bf16
        fc1_bias: [hidden_size_expanded] bf16
        fc2_weight: [out_hidden_size, hidden_size_expanded] bf16
        fc2_bias: [out_hidden_size] bf16
        eps: float
        Returns: [num_merged_patches, out_hidden_size] bf16
        """
        assert hidden.is_cuda and grid_thw.is_cuda and ln_weight.is_cuda and ln_bias.is_cuda \
               and fc1_weight.is_cuda and fc1_bias.is_cuda and fc2_weight.is_cuda and fc2_bias.is_cuda, \
               "All tensors must be on CUDA device for Triton kernels."

        device = hidden.device
        num_patches = hidden.shape[0]
        hidden_size = hidden.shape[1]
        hidden_size_expanded = fc1_weight.shape[0]
        out_hidden_size = fc2_weight.shape[0]
        num_merged_patches = num_patches  # as per provided workloads

        # 1) LayerNorm + affine in Triton
        hidden_norm = torch.empty_like(hidden, dtype=torch.bfloat16, device=device)
        layernorm_affine_kernel[(num_patches,)](
            hidden, ln_weight, ln_bias, hidden_norm,
            num_patches=num_patches,
            hidden_size=hidden_size,
            eps=eps,
            BLOCK_C=128,
            num_warps=4,
        )

        # 2) Spatial reorder (2x2 merge) into first linear input A_fc1
        A_fc1 = torch.empty((num_merged_patches, hidden_size_expanded), dtype=torch.bfloat16, device=device)
        spatial_shuffle_to_fc1_kernel[(grid_thw.shape[0],)](
            hidden_norm, grid_thw, A_fc1,
            num_patches=num_patches,
            hidden_size=hidden_size,
            hidden_size_expanded=hidden_size_expanded,
            merge_size=2,
            BLOCK_C=128,
            num_warps=4,
        )

        # 3) First Linear: GEMM with bias (A_fc1 is [num_merged_patches, hidden_size_expanded], B = fc1_weight.T)
        G_fc1 = torch.empty((num_merged_patches, hidden_size_expanded), dtype=torch.bfloat16, device=device)
        fc1_B = fc1_weight.T  # [hidden_size_expanded, hidden_size_expanded]
        grid1 = (triton.cdiv(num_merged_patches, 128), triton.cdiv(hidden_size_expanded, 128))
        matmul_bias_kernel[grid1](
            A_fc1, fc1_B, fc1_bias, G_fc1,
            M=num_merged_patches, N=hidden_size_expanded, K=hidden_size_expanded,
            stride_am=hidden_size_expanded, stride_ak=1,
            stride_bk=hidden_size_expanded, stride_bn=1,
            stride_cm=hidden_size_expanded, stride_cn=1,
            BLOCK_M=128, BLOCK_N=128, BLOCK_K=64,
            num_warps=4,
        )

        # 4) GELU activation in Triton
        G_fc1_after_gelu = torch.empty_like(G_fc1)
        gelu_tanh_kernel[grid1](
            G_fc1, G_fc1_after_gelu,
            M=num_merged_patches, N=hidden_size_expanded,
            stride_im=hidden_size_expanded, stride_in=1,
            stride_om=hidden_size_expanded, stride_on=1,
            BLOCK_M=128, BLOCK_N=128,
            num_warps=4,
        )

        # 5) Second Linear: GEMM with bias
        output = torch.empty((num_merged_patches, out_hidden_size), dtype=torch.bfloat16, device=device)
        fc2_B = fc2_weight.T  # [hidden_size_expanded, out_hidden_size]
        grid2 = (triton.cdiv(num_merged_patches, 128), triton.cdiv(out_hidden_size, 128))
        matmul_bias_kernel[grid2](
            G_fc1_after_gelu, fc2_B, fc2_bias, output,
            M=num_merged_patches, N=out_hidden_size, K=hidden_size_expanded,
            stride_am=out_hidden_size, stride_ak=1,
            stride_bk=hidden_size_expanded, stride_bn=1,
            stride_cm=out_hidden_size, stride_cn=1,
            BLOCK_M=128, BLOCK_N=128, BLOCK_K=64,
            num_warps=4,
        )

        return output
```
