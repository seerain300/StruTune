):

    # Guard

        return

    sum_val = 0.0
    sum_sq = 0.0
    for col in range(0, C, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(hidden_ptr + row * C + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

@triton.jit
def matmul_bias_kernel(
    A_ptr,     # *bf16, [M, K]
    B_ptr,     # *bf16, [K, N]
    Bias_ptr,  # *bf16, [N]
    C_ptr,     # *bf16, output [M, N]
    M, K, N,   # int sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # One program handles the full matrix (for given M,N,K), with tiling
    # Here we choose BLOCK_M=M, BLOCK_N=N, BLOCK_K=K so that it covers entire matrix in one go.
    # This is fine for the given sizes (<=8192) and avoids complexity of multiple blocks.
    # Accumulate in fp32.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in chunks of BLOCK_K (even though we set BLOCK_K=K, we keep the loop for generality)
    for k0 in range(0, K, BLOCK_K):
        # X tile [BLOCK_M, BLOCK_K]
        # We set BLOCK_M=BLOCK_N=M,N and K=K, so pointers are straightforward:
        A_tile_ptr = A_ptr + (tl.arange(0, BLOCK_M)[:, None] * K + (k0 + tl.arange(0, BLOCK_K))[None, :])
        # Masks for boundaries (though we chose full tiles)
        mask_A = (tl.arange(0, BLOCK_M)[:, None] < M) & ((k0 + tl.arange(0, BLOCK_K))[None, :] < K)
        A = tl.load(A_tile_ptr, mask=mask_A, other=0.0).to(tl.float16)

        # W tile [BLOCK_K, BLOCK_N]
        B_tile_ptr = B_ptr + ((k0 + tl.arange(0, BLOCK_K))[:, None] * N + tl.arange(0, BLOCK_N)[None, :])
        mask_B = ((k0 + tl.arange(0, BLOCK_K))[:, None] < K) & (tl.arange(0, BLOCK_N)[None, :] < N)
        B = tl.load(B_tile_ptr, mask=mask_B, other=0.0).to(tl.float16)

        # Accumulate
        acc += tl.dot(A, B)

    # Add bias
    bias = tl.load(Bias_ptr + tl.arange(0, BLOCK_N), mask=(tl.arange(0, BLOCK_N) < N), other=0.0).to(tl.float32)
    acc += bias[None, :]

    # Store back as bf16
    C_tile_ptr = C_ptr + (tl.arange(0, BLOCK_M)[:, None] * N + tl.arange(0, BLOCK_N)[None, :])
    mask_C = (tl.arange(0, BLOCK_M)[:, None] < M) & (tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(C_tile_ptr, acc.to(tl.bfloat16), mask=mask_C)


@triton.jit
def gelu_tanh_kernel(
    X_ptr,      # *bf16, input [M, N] flattened logically
    Y_ptr,      # *bf16, output [M, N] flattened
    M, N,       # int sizes
):
    pid = tl.program_id(axis=0)
    total = M * N
    if pid >= total:
        return
    x = tl.load(X_ptr + pid).to(tl.float32)
    # GELU tanh approximation:
    # y = 0.5 * x * (1 + tanh( sqrt(2/pi) * (x + 0.044715 * x^3) ))
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    inner = c * (x + 0.044715 * x3)
    y = 0.5 * x * (1.0 + tl.tanh(inner))
    tl.store(Y_ptr + pid, y.to(tl.bfloat16))

    mask_C = (tl.arange(0, BLOCK_M)[:, None] < M) & (tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.bfloat16), mask=mask_C)


@triton.jit
def gelu_tanh_kernel(
    X_ptr,      # *bf16, input flattened logically
    Y_ptr,      # *bf16, output flattened
    total_elems,  # int total number of elements
):
    pid = tl.program_id(axis=0)
    if pid >= total_elems:
        return
    x = tl.load(X_ptr + pid).to(tl.float32)
    # GELU tanh approximation: y = 0.5 * x * (1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    inner = c * (x + 0.044715 * x3)
    y = 0.5 * x * (1.0 + tl.tanh(inner))
    tl.store(Y_ptr + pid, y.to(tl.bfloat16))
