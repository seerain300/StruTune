I will provide a full Triton implementation that moves all numerical computation into @triton.jit kernels. The entry point class is ModelNew, and its forward method only allocates tensors, computes shapes/strides, and launches Triton kernels. No torch operations are used in the host code for math.

What is computed:
- LayerNorm: per-row normalization over the last dimension, with fp32 reductions for mean/var, normalization, and applying ln_weight/ln_bias in fp32, storing bfloat16.
- SpatialShuffle: exact mapping to merge 2x2 spatial positions for each patch. The output has shape [total_num_merged_patches, 4*hidden_size], where total_num_merged_patches = sum over grids of t * (h // 2) * (w // 2). The Triton kernel decodes (t, h, w) from grid_thw, iterates over all original patches within each grid, and writes the four 2x2 values to contiguous columns of the output.
- GELU: elementwise activation implemented in Triton using erf: y = 0.5 * x * (1 + erf(x / sqrt(2))).
- Linear layers: two row-wise GEMM Triton kernels:
  - Linear1: y1 = x @ fc1_weight^T + fc1_bias, where x has shape [num_merged_patches, 6144], fc1_weight has shape [6144, 6144], output y1 has shape [num_merged_patches, 6144].
  - Linear2: y = y1 @ fc2_weight^T + fc2_bias, where fc2_weight has shape [3584, 6144], output y has shape [num_merged_patches, 3584].

All tensors are created on the device, but no math is done with torch in the host. The Triton kernels perform the LayerNorm, shuffle, GELU, and both Linear layers.

Here is the complete code:

```python
import torch
import math
import triton
import triton.language as tl

# Triton kernel: LayerNorm per row (fp32 reduction, bfloat16 output)
@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: compute mean and variance in fp32
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


# Triton kernel: SpatialShuffle per grid. Produces output [TOTAL_ROWS, 4*C].
# Each program handles one grid entirely. Decode (t, h, w), iterate over all original
# patches in that grid, and write four 2x2 values to contiguous columns of the output.
@triton.jit
def _shuffle_2x2_per_grid_kernel(hidden_ptr, grid_thw_ptr, out_ptr,
                                 NUM_GRIDS, C,
                                 total_rows, start_row_offset,
                                 BLOCK_M: tl.constexpr):
    """
    hidden_ptr: *bf16, flattened per-grid: we will pass one pointer per grid, starting at the correct offset.
    grid_thw_ptr: *int64, shape [NUM_GRIDS, 3], each row is [t, h, w]
    out_ptr: *bf16, shape [total_rows, 4*C], row-major
    total_rows is the sum over all grids of t * (h//2) * (w//2).
    start_row_offset is the row offset in out_ptr where this grid's rows begin.
    """
    g = tl.program_id(0)
    if g >= NUM_GRIDS:
        return

    # Load grid dimensions
    t = tl.load(grid_thw_ptr + g * 3 + 0).to(tl.int32)
    h = tl.load(grid_thw_ptr + g * 3 + 1).to(tl.int32)
    w = tl.load(grid_thw_ptr + g * 3 + 2).to(tl.int32)

    h_merged = h // 2
    w_merged = w // 2
    num_merged_rows = t * h_merged * w_merged

    # Iterate over original patches in this grid
    m = 0
    while m < t * h * w:
        t_index = m // (h * w)
        rem = m % (h * w)
        h2 = rem // w
        w2 = rem % w

        out_row = start_row_offset + t_index * (h_merged * w_merged) + (h2 // 2) * w_merged + (w2 // 2)
        base_out = out_ptr + out_row * (4 * C)

        # Position (r=0,c=0) -> hidden[t_index, h2, w2]
        src_offset = t_index * (h * w) + h2 * w + w2
        val0 = tl.load(hidden_ptr + src_offset * C + 0, mask=True, other=0.0).to(tl.bfloat16)
        tl.store(base_out + 0 * C, val0)

        # Position (r=0,c=1) -> hidden[t_index, h2, w2+1]
        w2p1 = w2 + 1
        if w2p1 < w:
            val1 = tl.load(hidden_ptr + src_offset * C + 1, mask=True, other=0.0).to(tl.bfloat16)
            tl.store(base_out + 1 * C, val1)

        # Position (r=1,c=0) -> hidden[t_index, h2+1, w2]
        h2p1 = h2 + 1
        if h2p1 < h:
            val2 = tl.load(hidden_ptr + (t_index * (h * w) + h2p1 * w + w2) * C + 0, mask=True, other=0.0).to(tl.bfloat16)
            tl.store(base_out + 2 * C, val2)

        # Position (r=1,c=1) -> hidden[t_index, h2+1, w2+1]
        if h2p1 < h and w2p1 < w:
            val3 = tl.load(hidden_ptr + (t_index * (h * w) + h2p1 * w + w2p1) * C + 0, mask=True, other=0.0).to(tl.bfloat16)
            tl.store(base_out + 3 * C, val3)

        m += 1


# Triton kernel: Elementwise GELU. One program per row.
@triton.jit
def _gelu_row_kernel(x_ptr, y_ptr, N, C,
                     BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C]
    GELU: y = 0.5 * x * (1 + erf(x / sqrt(2)))
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
        y = 0.5 * x * (1.0 + tl.erf(x * inv_sqrt2))
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


# Triton kernel: Row-wise Linear (GEMM) - compute y[row] = x[row] @ W.T + b
# Assumes K is known at compile time. We implement two specializations:
# - Linear1: N is number of merged patches, C_in=6144, C_out=6144
# - Linear2: N is number of merged patches, C_in=6144, C_out=3584
@triton.jit
def _linear_row_kernel(x_ptr, W_ptr, b_ptr, y_ptr,
                        N, C_IN, C_OUT,
                        BLOCK_K: tl.constexpr, OUT_BLOCK: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C_IN], row-major (we pass one row per program)
    W_ptr: *bf16, shape [C_OUT, C_IN], row-major (each row is weight vector for an output unit)
    b_ptr: *bf16, shape [C_OUT]
    y_ptr: *bf16, shape [N, C_OUT], row-major
    """
    row = tl.program_id(0)
    if row >= N:
        return

    # Compute y[row, :] vector
    y_row_ptr = y_ptr + row * C_OUT

    # Initialize y
    col_out = 0
    while col_out < C_OUT:
        offs_out = col_out + tl.arange(0, OUT_BLOCK)
        mask_out = offs_out < C_OUT
        y_vec = tl.zeros([OUT_BLOCK], dtype=tl.float32)
        # Accumulate over K
        k0 = 0
        while k0 < C_IN:
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < C_IN
            # x[row, k] vector
            x_vec = tl.load(x_ptr + row * C_IN + offs_k, mask=mask_k, other=0.0).to(tl.float32)
            # W[k, offs_out] matrix: shape [BLOCK_K, OUT_BLOCK]
            W_sub = tl.load(W_ptr + offs_out * C_IN + offs_k, mask=(mask_out[:, None] & mask_k[None, :]), other=0.0).to(tl.float32)
            # dot: sum over K of x_vec * W_sub[i, :]
            dot_vec = tl.zeros([OUT_BLOCK], dtype=tl.float32)
            kk = 0
            while kk < BLOCK_K:
                k_idx = kk + tl.arange(0, BLOCK_K)
                mk = k_idx < C_IN
                x_sub = tl.load(x_ptr + row * C_IN + k_idx, mask=mk, other=0.0).to(tl.float32)
                W_sub2 = tl.load(W_ptr + offs_out * C_IN + k_idx, mask=(mask_out[:, None] & mk[None, :]), other=0.0).to(tl.float32)
                dot_vec += tl.sum(W_sub2 * x_sub[None, :], axis=1)
                kk += BLOCK_K
            y_vec += dot_vec
            k0 += BLOCK_K
        # add bias
        b_vec = tl.load(b_ptr + offs_out, mask=mask_out, other=0.0).to(tl.float32)
        y_vec += b_vec
        # store
        tl.store(y_row_ptr + offs_out, y_vec.to(tl.bfloat16), mask=mask_out)
        col_out += OUT_BLOCK


def _triton_layer_norm(hidden: torch.Tensor, ln_weight: torch.Tensor, ln_bias: torch.Tensor, device: torch.device) -> torch.Tensor:
    N, C = hidden.shape
    y = torch.empty_like(hidden)
    # Cast params to bf16 for load; compute in fp32
    ln_w = ln_weight.to(torch.bfloat16).to(device)
    ln_b = ln_bias.to(torch.bfloat16).to(device)
    # Choose block size as next power of 2 up to 1024
    BLOCK_SIZE = 1024
    _layer_norm_kernel[(N,)](hidden, y, ln_w, ln_b, N, C, 1e-6, BLOCK_SIZE)
    return y


def _triton_shuffle(hidden_norm: torch.Tensor, grid_thw: torch.Tensor, C: int, device: torch.device) -> torch.Tensor:
    """
    hidden_norm: [total_patches, C], bfloat16, contiguous
    grid_thw: [num_grids, 3], int64
    Returns: [total_num_merged_rows, 4*C], bfloat16
    """
    total_patches = hidden_norm.shape[0]
    num_grids = grid_thw.shape[0]
    # Compute total_num_merged_rows
    total_rows = 0
    for g in range(num_grids):
        t = int(grid_thw[g, 0].item())
        h = int(grid_thw[g, 1].item())
        w = int(grid_thw[g, 2].item())
        total_rows += t * (h // 2) * (w // 2)

    out = torch.empty((total_rows, 4 * C), dtype=torch.bfloat16, device=device)

    # Build start_row_offsets array to feed to kernel
    start_row_offsets = []
    curr = 0
    for g in range(num_grids):
        t = int(grid_thw[g, 0].item())
        h = int(grid_thw[g, 1].item())
        w = int(grid_thw[g, 2].item())
        num_merged_rows_g = t * (h // 2) * (w // 2)
        start_row_offsets.append(curr)
        curr += num_merged_rows_g
    start_row_offsets = torch.tensor(start_row_offsets, dtype=torch.int32, device=device)

    # Launch one program per grid
    _shuffle_2x2_per_grid_kernel[(num_grids,)](
        hidden_norm, grid_thw, out, num_grids, C, total_rows, start_row_offsets, 1
    )
    return out


def _triton_gelu(x: torch.Tensor) -> torch.Tensor:
    N, C = x.shape
    y = torch.empty_like(x)
    BLOCK_SIZE = 256
    _gelu_row_kernel[(N,)](x, y, N, C, BLOCK_SIZE)
    return y


def _triton_linear_row(x: torch.Tensor, W: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    x: [N, C_IN], bfloat16
    W: [C_OUT, C_IN], bfloat16
    returns y: [N, C_OUT], bfloat16
    """
    N, C_IN = x.shape
    C_OUT, _ = W.shape
    y = torch.empty((N, C_OUT), dtype=torch.bfloat16, device=x.device)
    # Choose blocks; given sizes, use BLOCK_K=1024, OUT_BLOCK=1024
    BLOCK_K = 1024
    OUT_BLOCK = 1024
    _linear_row_kernel[(N,)](x, W, b, y, N, C_IN, C_OUT, BLOCK_K, OUT_BLOCK)
    return y


class ModelNew(torch.nn.Module):
    def __init__(self, device: torch.device):
        super().__init__()
        self.device = device
        # Fixed sizes from the original implementation
        self.hidden_size = 1536
        self.hidden_size_expanded = 6144
        self.out_hidden_size = 3584
        self.merge_size = 2

    def forward(self, hidden: torch.Tensor,
                grid_thw: torch.Tensor,
                ln_weight: torch.Tensor,
                ln_bias: torch.Tensor,
                fc1_weight: torch.Tensor,
                fc1_bias: torch.Tensor,
                fc2_weight: torch.Tensor,
                fc2_bias: torch.Tensor,
                eps: float):
        """
        hidden: [num_patches, hidden_size], bfloat16
        grid_thw: [num_grids, 3], int64
        ln_weight, ln_bias: [hidden_size], bfloat16
        fc1_weight: [hidden_size_expanded, hidden_size_expanded] bfloat16
        fc1_bias: [hidden_size_expanded] bfloat16
        fc2_weight: [out_hidden_size, hidden_size_expanded] bfloat16
        fc2_bias: [out_hidden_size] bfloat16
        """
        device = self.device

        # Step 1: LayerNorm (pre-shuffle, on hidden_size dimension)
        hidden_norm = _triton_layer_norm(hidden, ln_weight, ln_bias, device)

        # Step 2: Spatial shuffle to merge 2x2 patches
        hidden_shuffled = _triton_shuffle(hidden_norm, grid_thw, self.hidden_size, device)

        # Step 3: Linear1 and GELU
        # Linear1: [num_merged_patches, hidden_size_expanded] @ [hidden_size_expanded, hidden_size_expanded].T
        y1 = _triton_linear_row(hidden_shuffled, fc1_weight, fc1_bias)
        y1_gelu = _triton_gelu(y1)

        # Step 4: Linear2
        out = _triton_linear_row(y1_gelu, fc2_weight, fc2_bias)
        return out
```

Notes:
- All computation is in Triton: layer_norm, shuffle, gelu, and both linear layers. The host code only allocates tensors, computes metadata (e.g., total_num_merged_rows), and launches the kernels.
- The Triton kernels are specialized for the given sizes. While row-wise GEMM may not be as fast as PyTorch’s optimized matmul, it satisfies the requirement to keep everything in Triton. If you later need speedups, we can implement 2D-tiled Triton GEMMs and/or fuse operations, but correctness must be ensured first.
