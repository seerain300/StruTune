@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32
    """

    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: compute mean and variance (fp32)
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE

    y_ptr: *bf16, shape [N, K_out], row-major

                    w_row = tl.load(w_block_ptr + tl.arange(0, BLOCK_K_OUT), mask=(tl.arange(0, BLOCK_K_OUT) < K_out), other=0.0).to(tl.float32)

                    x_val = x_vec[jj_col]

        k += BLOCK_K

    b = tl.load(b_ptr + k_out_offs, mask=mask_k_out, other=0.0).to(tl.float32)

                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):

    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32

    y_row_ptr = y_ptr + row * C

        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


# Triton kernel: Spatial shuffle for a single grid. Reconstruct original (t, h, w) and permute to
# (T, H//2, W//2, 2, 2, C) then flatten to [T*H//2*W//2, 4*C]. Only handles merge_size=2.
@triton.jit
def _shuffle_grid_kernel(hidden_ptr, out_ptr,
                          NUM_OUT_ROWS, NUM_OUT_COLS, C,
                          T, H, W,
                          grid_id,
                          BLOCK_ROWS: tl.constexpr):
    """
    hidden_ptr: *bf16, base pointer to a contiguous hidden tensor [N, C], N = T*H*W
    out_ptr: *bf16, base pointer to output [NUM_OUT_ROWS, NUM_OUT_COLS], contiguous row-major
    NUM_OUT_ROWS = T * (H//2) * (W//2)
    NUM_OUT_COLS = 4*C
    For each output row m in this grid:
      t = m // ((H//2)*(W//2))
      rem = m % ((H//2)*(W//2))
      h2 = rem // (W//2)
      w2 = rem % (W//2)
      -> out[m, :] = [hidden[t, h=2*h2, w=2*w2, 0], hidden[t, h, w+1], hidden[t, h+1, w], hidden[t, h+1, w+1]]
    """
    pid = tl.program_id(0)
    if pid >= BLOCK_ROWS:
        return

    H2 = H // 2
    W2 = W // 2
    m = pid  # row id in output for this grid

    if m >= NUM_OUT_ROWS:
        return

    # Compute (t, h2, w2)
    t = m // (H2 * W2)
    rem = m % (H2 * W2)
    h2 = rem // W2
    w2 = rem % W2

    h = h2 * 2
    w0 = w2 * 2

    # Output columns: 0*C, 1*C, 2*C, 3*C correspond to the four positions in the 2x2 patch
    # idx=0: (r=0,c=0) -> hidden[t, h, w0]
    # idx=1: (r=0,c=1) -> hidden[t, h, w0+1]
    # idx=2: (r=1,c=0) -> hidden[t, h+1, w0]
    # idx=3: (r=1,c=1) -> hidden[t, h+1, w0+1]
    idx = 0
    h_read = h
    w_read = w0
    val0 = tl.load(hidden_ptr + t * (H * W) + h_read * W + w_read).to(tl.bfloat16)
    tl.store(out_ptr + m * NUM_OUT_COLS + idx * C, val0)

    idx = 1
    h_read = h
    w_read = w0 + 1
    if w_read < W:
        val1 = tl.load(hidden_ptr + t * (H * W) + h_read * W + w_read).to(tl.bfloat16)
        tl.store(out_ptr + m * NUM_OUT_COLS + idx * C, val1)

    idx = 2
    h_read = h + 1
    w_read = w0
    if h_read < H and w_read < W:
        val2 = tl.load(hidden_ptr + t * (H * W) + h_read * W + w_read).to(tl.bfloat16)
        tl.store(out_ptr + m * NUM_OUT_COLS + idx * C, val2)

    idx = 3
    h_read = h + 1
    w_read = w0 + 1
    if h_read < H and w_read < W:
        val3 = tl.load(hidden_ptr + t * (H * W) + h_read * W + w_read).to(tl.bfloat16)
        tl.store(out_ptr + m * NUM_OUT_COLS + idx * C, val3)
