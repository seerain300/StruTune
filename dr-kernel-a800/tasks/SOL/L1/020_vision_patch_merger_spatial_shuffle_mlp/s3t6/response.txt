No torch.cat, torch.index_select, or any torch math will remain in the host code; all will be Triton kernels.

High-level plan:
1) LayerNorm per row:
   - One Triton kernel launches one program per row. It performs two passes over the row in chunks: first pass reduces sum and sum of squares in fp32 to compute mean and variance; second pass normalizes and applies ln_weight and ln_bias in fp32, then stores in bfloat16. This matches the original behavior.

2) Spatial shuffle (exact permutation of 2x2):
   - The original code concatenates per-grid patches into a single hidden_norm [num_patches, C] and then reshapes/permutes to [t, h_merged, w_merged, 2, 2, C] and flattens to [t * h_merged * w_merged, C * 4]. To implement this in Triton without relying on PyTorch index_select or cat, we:
     - First create an output buffer hidden_shuffled of shape [total_num_merged_patches, 4*C] using torch.empty (metadata only; we won’t fill it yet).
     - Launch a Triton kernel that writes exactly the permutation. For each grid, it iterates over all original patch indices m in [0, t*h*w). It decodes t_index, h2, w2, computes the out row index out_row = t_index * (h//2) * (w//2) + (h2//2) * (w//2) + (w2//2), and stores the four 2x2 positions into consecutive columns [0*C, 1*C, 2*C, 3*C] of that out_row. All indices are computed on the host and passed to Triton; the kernel uses simple while loops to avoid dynamic Python-side loops. This ensures exact matching of the original permutation.

3) GELU activation:
   - Triton kernel that applies GELU elementwise over hidden_shuffled (flattened). We use the standard tanh approximation for erf to keep computations in Triton. The kernel processes the vector in BLOCK-sized chunks.

4) Two Linear layers:
   - Implement row-wise GEMM-style kernels in Triton. Each kernel computes one output row (i.e., one sample) by iterating over the K dimension in BLOCK_K chunks. For Linear1: y = x @ fc1_weight.T + fc1_bias, with K_in=6144, K_out=6144. For GELU. For Linear2: y = x @ fc2_weight.T + fc2_bias, with K_in=6144, K_out=3584. We pass K_in, K_out as constexpr meta-parameters for Triton. The forward will launch these kernels sequentially to produce the final output.

import torch
import math
import triton
import triton.language as tl


@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major (we pass flattened with stride C)
    y_ptr: *bf16, shape [N, C]
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    Compute per-row LayerNorm in fp32: y = ((x - mean) / sqrt(var + eps)) * ln_weight + ln_bias
    eps is implicitly handled in the host (use 1e-6) or not needed here since no eps param.
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: reduce sum and sum of squares (fp32)
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + 1e-6)  # eps=1e-6 as in original

    # Pass 2: normalize and apply affine
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


@triton.jit
def _shuffle_2x2_per_grid_kernel(hidden_ptr, grid_thw_ptr, out_ptr,
                                 total_patches, C, NUM_GRIDS,
                                 H_MAX, W_MAX,  # host provides max dims to ensure masks
                                 BLOCK_M: tl.constexpr):
    """
    hidden_ptr: *bf16, flattened [total_patches, C]
    grid_thw_ptr: *int64, shape [NUM_GRIDS, 3], each row is [t, h, w]
    out_ptr: *bf16, flattened [total_merged_rows, 4*C]
    Each program handles one grid. It iterates over all original patch indices m in that grid and
    writes the 2x2 merged rows into out_ptr with exact mapping.
    """
    g = tl.program_id(0)
    if g >= NUM_GRIDS:
        return

    # Load grid dimensions for this grid
    t = tl.load(grid_thw_ptr + g * 3 + 0).to(tl.int32)
    h = tl.load(grid_thw_ptr + g * 3 + 1).to(tl.int32)
    w = tl.load(grid_thw_ptr + g * 3 + 2).to(tl.int32)

    h_merged = h // 2
    w_merged = w // 2
    num_merged_rows = t * h_merged * w_merged

    m = 0
    while m < t * h * w:
        t_index = m // (h * w)
        rem = m % (h * w)
        h2 = rem // w
        w2 = rem % w

        out_row = t_index * (h_merged * w_merged) + (h2 // 2) * w_merged + (w2 // 2)
        base_out = out_ptr + out_row * (4 * C)

        # idx=0: (r=0,c=0)
        # idx=1: (r=0,c=1)
        # idx=2: (r=1,c=0)
        # idx=3: (r=1,c=1)
        idx = 0
        h_read = h2
        w_read = w2
        src_offset = t_index * (h * w) + h_read * w + w_read
        val0 = tl.load(hidden_ptr + src_offset * C + 0, mask=(h2 < H_MAX) & (w2 < W_MAX), other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx * C, val0)

        idx = 1
        h_read = h2
        w_read = w2 + 1
        src_offset1 = t_index * (h * w) + h_read * w + w_read
        val1 = tl.load(hidden_ptr + src_offset1 * C + 0, mask=(w2 + 1 < w) & (h2 < H_MAX), other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx * C, val1)

        idx = 2
        h_read = h2 + 1
        w_read = w2
        src_offset2 = t_index * (h * w) + h_read * w + w_read
        val2 = tl.load(hidden_ptr + src_offset2 * C + 0, mask=(h2 + 1 < h) & (w2 < W_MAX), other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx * C, val2)

        idx = 3
        h_read = h2 + 1
        w_read = w2 + 1
        src_offset3 = t_index * (h * w) + h_read * w + w_read
        val3 = tl.load(hidden_ptr + src_offset3 * C + 0, mask=((h2 + 1 < h) & (w2 + 1 < w)), other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx * C, val3)

        m += 1


@triton.jit
def _gelu_kernel(x_ptr, y_ptr, N, C, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise GELU on y = x @ (.), then GELU. Here x_ptr is the linear output.
    We apply GELU: y = 0.5 * x * (1 + erf(x / sqrt(2)))
    Triton does not have erf, so use tanh approximation:
    erf(z) ≈ 1 - (a1 t + a2 t^2 + a3 t^3 + a4 t^4 + a5 t^5) * exp(-z^2), t = 1/(1 + p z), p=0.3275911
    a1=0.254829592, a2=-0.284496736, a3=1.421413741, a4=-1.453152027, a5=1.061405429
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        z = x * 0.7071067811865476  # 1/sqrt(2)
        # tanh approximation for erf
        p = 0.3275911
        a1 = 0.254829592
        a2 = -0.284496736
        a3 = 1.421413741
        a4 = -1.453152027
        a5 = 1.061405429
        sign = tl.where(z >= 0, 1.0, -1.0)
        az = tl.abs(z)
        t = 1.0 / (1.0 + p * az)
        poly = (((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t)
        erf_approx = 1.0 - poly * tl.exp(-az * az)
        gelu = 0.5 * x * (1.0 + erf_approx * sign)
        tl.store(y_row_ptr + offs, gelu.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


@triton.jit
def _linear_rowwise_kernel(x_ptr, w_ptr, b_ptr, y_ptr,
                            K_in, K_out, BLOCK_K: tl.constexpr):
    """
    Compute one output row: y[j] = sum_k x[k] * w[j, k] + b[j]
    x_ptr: *bf16, shape [1, K_in] (flattened, stride K_in)
    w_ptr: *bf16, shape [K_out, K_in] (row-major)
    b_ptr: *bf16, shape [K_out]
    y_ptr: *bf16, shape [1, K_out]
    """
    # One program per row (i=0 here, since we will launch N_row programs if needed).
    acc = tl.zeros([K_out], dtype=tl.float32)
    k = 0
    while k < K_in:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K_in
        x_k = tl.load(x_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)  # [BLOCK_K]
        # For each kk in BLOCK_K, accumulate into acc[j] += x_k[kk] * W[j, k+kk]
        for kk in range(BLOCK_K):
            k_curr = k + kk
            mkk = k_curr < K_in
            xk_val = tl.load(x_ptr + k_curr, mask=mkk, other=0.0).to(tl.float32)
            # Load W[:, k_curr] vector for all j
            wj = tl.load(w_ptr + k_curr * K_out + tl.arange(0, K_out), mask=(tl.arange(0, K_out) < K_out) & mkk, other=0.0).to(tl.float32)
            acc += xk_val * wj
        k += BLOCK_K

    # Add bias
    j = 0
    while j < K_out:
        offs_j = j + tl.arange(0, BLOCK_K)
        mask_j = offs_j < K_out
        bj = tl.load(b_ptr + offs_j, mask=mask_j, other=0.0).to(tl.float32)
        acc += bj
        j += BLOCK_K

    # Store y[0, :]
    j = 0
    while j < K_out:
        offs_j = j + tl.arange(0, BLOCK_K)
        mask_j = offs_j < K_out
        tl.store(y_ptr + offs_j, acc[offs_j].to(tl.bfloat16), mask=mask_j)
        j += BLOCK_K
