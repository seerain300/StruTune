```python
import torch
import math
import triton
import triton.language as tl

# Triton: LayerNorm per row (two passes). One program per row.
@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32
    """
    row = tl.program_id(0)
    if row >= N:
        return
    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: mean and variance in fp32
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


# Triton: Exact spatial 2x2 merge per grid, producing output [total_merged_rows, 4*C].
@triton.jit
def _shuffle_2x2_per_grid_kernel(hidden_ptr, grid_thw_ptr, out_ptr,
                                 C, NUM_GRIDS,
                                 MERGE_SIZE: tl.constexpr):
    # MERGE_SIZE must be 2 in this implementation.
    g = tl.program_id(0)
    if g >= NUM_GRIDS:
        return

    # Load grid dimensions
    t = tl.load(grid_thw_ptr + g * 3 + 0).to(tl.int32)
    h = tl.load(grid_thw_ptr + g * 3 + 1).to(tl.int32)
    w = tl.load(grid_thw_ptr + g * 3 + 2).to(tl.int32)

    # Only handle merge_size == 2
    h_merged = h // MERGE_SIZE
    w_merged = w // MERGE_SIZE
    num_merged_rows = t * h_merged * w_merged

    # For each original patch m in [0, t*h*w):
    # Decode (t_index, h2, w2), then out_row = t_index * (h_merged * w_merged) + (h2//2) * w_merged + (w2//2)
    m = 0
    total_patches = t * h * w
    while m < total_patches:
        t_index = m // (h * w)
        rem = m % (h * w)
        h2 = rem // w
        w2 = rem % w

        out_row = t_index * (h_merged * w_merged) + (h2 // 2) * w_merged + (w2 // 2)
        base_out = out_ptr + out_row * (MERGE_SIZE * MERGE_SIZE * C)

        # idx=0: (r=0,c=0) -> hidden[t_index, h2, w2]
        idx0 = 0
        col0 = (h2 * MERGE_SIZE + 0) * w_merged * C + (w2 * MERGE_SIZE + 0) * C
        src_offset0 = t_index * (h * w) + h2 * w + w2
        val0 = tl.load(hidden_ptr + src_offset0 * C + 0, mask=True, other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx0 * C, val0)

        # idx=1: (r=0,c=1) -> hidden[t_index, h2, w2+1]
        idx1 = 1
        col1 = (h2 * MERGE_SIZE + 0) * w_merged * C + (w2 * MERGE_SIZE + 1) * C
        src_offset1 = t_index * (h * w) + h2 * w + (w2 + 1)
        val1 = tl.load(hidden_ptr + src_offset1 * C + 0, mask=(w2 + 1) < w, other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx1 * C, val1)

        # idx=2: (r=1,c=0) -> hidden[t_index, h2+1, w2]
        idx2 = 2
        col2 = (h2 * MERGE_SIZE + 1) * w_merged * C + (w2 * MERGE_SIZE + 0) * C
        src_offset2 = t_index * (h * w) + (h2 + 1) * w + w2
        val2 = tl.load(hidden_ptr + src_offset2 * C + 0, mask=(h2 + 1) < h, other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx2 * C, val2)

        # idx=3: (r=1,c=1) -> hidden[t_index, h2+1, w2+1]
        idx3 = 3
        col3 = (h2 * MERGE_SIZE + 1) * w_merged * C + (w2 * MERGE_SIZE + 1) * C
        src_offset3 = t_index * (h * w) + (h2 + 1) * w + (w2 + 1)
        val3 = tl.load(hidden_ptr + src_offset3 * C + 0, mask=(h2 + 1) < h and (w2 + 1) < w, other=0.0).to(tl.bfloat16)
        tl.store(base_out + idx3 * C, val3)

        m += 1


# Triton: GELU elementwise using erf approximation (approximate='tanh' not used).
# We implement erf(z) via a high-accuracy polynomial approximation.
@triton.jit
def _gelu_erf_kernel(x_ptr, y_ptr, N, C,
                     BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C]
    GELU(x) = 0.5 * x * (1 + erf(x / sqrt(2)))
    Use Abramowitz-Stegun erf approximation for accuracy.
    """
    row = tl.program_id(0)
    if row >= N:
        return
    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        inv_sqrt2 = 0.7071067811865476
        z = x * inv_sqrt2
        # Abramowitz-Stegun erf approximation:
        # erf(z) ≈ sign(z) * (1 - (a1 t + a2 t^2 + a3 t^3 + a4 t^4 + a5 t^5) * exp(-z^2))
        # where t = 1 / (1 + p |z|)
        az = tl.abs(z)
        p = 0.3275911
        t = 1.0 / (1.0 + p * az)
        a1 = 0.254829592
        a2 = -0.284496736
        a3 = 1.421413741
        a4 = -1.453152027
        a5 = 1.061405429
        # Horner's method for polynomial in t
        poly = (((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t)
        erf_z = 1.0 - poly * tl.exp(-az * az)
        erf_z = tl.where(z >= 0, erf_z, -erf_z)
        y = 0.5 * x * (1.0 + erf_z)
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


# Triton: Linear row-wise GEMM: y_row = x_row @ W.T + b
# Specialized for K_in=6144, K_out=6144 (Linear1) and K_in=6144, K_out=3584 (Linear2).
@triton.jit
def _linear_row_gemm_kernel(x_ptr, w_ptr, b_ptr, y_ptr,
                            N, K_in, K_out,
                            BLOCK_K: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, K_in], row-major
    w_ptr: *bf16, shape [K_out, K_in], row-major (i.e., each row is a weight vector of length K_in)
    b_ptr: *bf16, shape [K_out]
    y_ptr: *bf16, shape [N, K_out], row-major
    One program computes one output row i.
    """
    row = tl.program_id(0)
    if row >= N:
        return
    x_row_ptr = x_ptr + row * K_in

    # Accumulator in fp32
    acc = tl.zeros((K_out,), dtype=tl.float32)

    k_base = 0
    while k_base < K_in:
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K_in
        # Load x_vec as fp32
        x_vec = tl.load(x_row_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)

        # For each output dim j, accumulate dot(x_vec, w[j, :])
        j = 0
        while j < K_out:
            w_row_ptr = w_ptr + j * K_in
            w_vec = tl.load(w_row_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)
            acc[j] += tl.sum(x_vec * w_vec, axis=0)
            j += 1
        k_base += BLOCK_K

    # Add bias and store bfloat16
    col = 0
    while col < K_out:
        bj = tl.load(b_ptr + col).to(tl.float32)
        yv = acc[col] + bj
        tl.store(y_ptr + row * K_out + col, yv.to(tl.bfloat16))
        col += 1


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, hidden: torch.Tensor,
                grid_thw: torch.Tensor,
                ln_weight: torch.Tensor,
                ln_bias: torch.Tensor,
                fc1_weight: torch.Tensor,
                fc1_bias: torch.Tensor,
                fc2_weight: torch.Tensor,
                fc2_bias: torch.Tensor,
                eps: float):
        """
        hidden: [num_patches, hidden_size] (bf16)
        grid_thw: [num_grids, 3] (int64) -> [t, h, w]
        ln_weight, ln_bias: [hidden_size] (bf16)
        fc1_weight: [hidden_size_expanded, hidden_size_expanded] (bf16)
        fc1_bias: [hidden_size_expanded] (bf16)
        fc2_weight: [out_hidden_size, hidden_size_expanded] (bf16)
        fc2_bias: [out_hidden_size] (bf16)
        eps: float
        """
        # Output tensors
        num_patches = hidden.shape[0]
        hidden_size = hidden.shape[1]
        hidden_exp = fc1_weight.shape[0]
        out_hidden_size = fc2_weight.shape[0]
        C = hidden_size
        # 1) LayerNorm in Triton
        hidden_norm = torch.empty_like(hidden)
        N = num_patches
        # Choose BLOCK_SIZE as a power of two up to C
        BLOCK_SIZE = 1024 if C >= 1024 else (512 if C >= 512 else 256)
        _layer_norm_kernel[(N,)](hidden, hidden_norm, ln_weight, ln_bias, N, C, eps, BLOCK_SIZE)

        # 2) Spatial shuffle (2x2 merge) in Triton: produce [num_merged_patches, 4*C]
        num_merged_patches = grid_thw[:, 0].sum().item() * ((grid_thw[:, 1] // 2).sum().item()) * ((grid_thw[:, 2] // 2).sum().item())
        # Allocate output for shuffle
        hidden_shuffled = torch.empty((num_merged_patches, 4 * C), dtype=torch.bfloat16, device=hidden.device)
        NUM_GRIDS = grid_thw.shape[0]
        MERGE_SIZE = 2
        # Launch one program per grid
        _shuffle_2x2_per_grid_kernel[(NUM_GRIDS,)](hidden_norm, grid_thw, hidden_shuffled, C, NUM_GRIDS, MERGE_SIZE)

        # 3) GELU in Triton
        hidden_gelu = torch.empty_like(hidden_shuffled)
        BLOCK_GELU = 1024
        _gelu_erf_kernel[(hidden_shuffled.shape[0],)](hidden_shuffled, hidden_gelu, hidden_shuffled.shape[0], 4 * C, BLOCK_GELU)

        # 4) Linear1: [num_merged_patches, hidden_exp] via Triton GEMM
        y1 = torch.empty((hidden_gelu.shape[0], hidden_exp), dtype=torch.bfloat16, device=hidden.device)
        # For K_in=6144 and K_out=6144 (Linear1), choose BLOCK_K=256
        BLOCK_K1 = 256
        _linear_row_gemm_kernel[(hidden_gelu.shape[0],)](hidden_gelu, fc1_weight, fc1_bias, y1, hidden_gelu.shape[0], 4 * C, hidden_exp, BLOCK_K1)

        # 5) GELU on y1 (optional): since original has GELU after fc1, apply Triton GELU again
        y1_gelu = torch.empty_like(y1)
        BLOCK_GELU2 = 1024
        _gelu_erf_kernel[(y1.shape[0],)](y1, y1_gelu, y1.shape[0], y1.shape[1], BLOCK_GELU2)

        # 6) Linear2: [num_merged_patches, out_hidden_size]
        output = torch.empty((y1_gelu.shape[0], out_hidden_size), dtype=torch.bfloat16, device=hidden.device)
        BLOCK_K2 = 256
        _linear_row_gemm_kernel[(y1_gelu.shape[0],)](y1_gelu, fc2_weight, fc2_bias, output, y1_gelu.shape[0], hidden_exp, out_hidden_size, BLOCK_K2)

        return output
```
