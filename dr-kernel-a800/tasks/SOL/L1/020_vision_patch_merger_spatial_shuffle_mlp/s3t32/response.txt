@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32
    One program per row (pid=0..N-1). Reduction and normalization done in fp32.
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: compute mean and variance in fp32
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


@triton.jit
def _gelu_tanh_kernel(x_ptr, y_ptr, N, C, BLOCK_SIZE: tl.constexpr):
    """
    GELU using tanh approximation: 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    x_ptr: *bf16, y_ptr: *bf16, shape [N, C]

        return
    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)

        y = 0.5 * x * (1.0 + tl.tanh(t))
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE


@triton.jit
def _linear_row_gemm_kernel(x_ptr, w_ptr, b_ptr, y_ptr,
                             N, K_in, K_out, eps,  # eps unused but can be passed
                             BLOCK_K: tl.constexpr):
    """
    Compute one output row y[i] = x[i] @ W.T + b, where:
      x_ptr: *bf16, shape [N, K_in], row-major (N is index dimension we loop, K_in is columns)
      w_ptr: *bf16, shape [K_out, K_in], row-major
      b_ptr: *bf16, shape [K_out]
      y_ptr: *bf16, shape [N, K_out], row-major
    We set N=index dimension of output, K_in=hidden_size_expanded, K_out=hidden_size_expanded or out_hidden_size.
    """
    row_out = tl.program_id(0)
    if row_out >= N:
        return

    # Accumulator for this output row
    acc = tl.zeros([K_out], dtype=tl.float32)

    k = 0
    while k < K_in:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K_in
        # Load x vector for this row (x is [N, K_in], row=row_out, columns offs_k)
        x_vec = tl.load(x_ptr + row_out * K_in + offs_k, mask=mask_k, other=0.0).to(tl.float32)

        # Load weight rows W[offs_k, :] (shape [BLOCK_K, K_out])
        w_rows = tl.load(w_ptr + offs_k[:, None] * K_in + tl.arange(0, K_out)[None, :], mask=mask_k[:, None], other=0.0).to(tl.float32)

        # Accumulate: dot = sum_j W[k,j] * x_vec[j]
        dot_vec = tl.sum(w_rows * x_vec[None, :], axis=1)  # [BLOCK_K]
        acc += dot_vec
        k += BLOCK_K

    # Add bias
    bias = tl.load(b_ptr + tl.arange(0, K_out), mask=None, other=0.0).to(tl.float32)
    acc += bias

    # Store result in bfloat16
    tl.store(y_ptr + row_out * K_out + tl.arange(0, K_out), acc.to(tl.bfloat16))

    hidden_norm_ptr: *bf16, shape [NUM_ROWS, C], row-major
    grid_thw_ptr: *int64, shape [NUM_GRIDS, 3], each row is [t, h, w]
    out_ptr: *bf16, shape [NUM_MERGED_ROWS, 4*C], row-major
    Triton kernel that implements the 2x2 spatial shuffle per grid.
    For each output row m in [0, NUM_MERGED_ROWS):
      decode grid g, t_index, h2, w2 via NUM_MERGED_ROWS and grid sizes
      compute source row in hidden_norm and copy 4 values into 4 contiguous columns of out[m, :]
    """
    m = tl.program_id(0)
    if m >= NUM_MERGED_ROWS:
        return

    # Decode (grid g, t_index, h2, w2) from m
    # Let total_m = sum_{g} t_g * (h_g//2) * (w_g//2) = NUM_MERGED_ROWS
    # We don't have per-grid prefix sums here; instead we decode using the structure of NUM_MERPED_ROWS.
    # A more robust approach would pass grid prefixes. To keep this simple and correct, we loop g
    # and compute contribution using t,h,w per grid; but Triton prefers vectorized static logic.
    # Implement by iterating over g until we find which grid m belongs to (simple in PyTorch but not in Triton).
    # Instead, we use a fixed grid index and compute via modulo and division; however, without per-grid
    # prefix sums, exact decoding is not possible. Therefore, we implement a helper in PyTorch to
    # compute NUM_MERGED_ROWS and place data. Since we must use Triton-only, we implement a fallback:
    # we compute NUM_MERGED_ROWS on host, and for Triton, we just write out using the host-computed mapping.
    # In practice, Triton cannot branch on dynamic NUM_MERGED_ROWS mapping without grid prefixes; thus
    # we keep PyTorch for exact mapping. But to adhere to Triton-only, we implement decoding inside Triton
    # using grid_thw and NUM_MERGED_ROWS by iterating g and computing contributions. Triton does not support
    # Python-level loops over NUM_GRIDS here; so we use a fixed number of iterations equal to NUM_GRIDS.
    # This is a pragmatic approach: for each m, try g=0..NUM_GRIDS-1, compute h_merged,w_merged, num_rows,
    # and check m < sum; if yes, decode; else skip. This works because Triton will try all g; only one g
    # will satisfy. We use a large sentinel to set g outside if none.

    # Initialize guessed g and contribution
    g = 0
    # We need to know t,h,w per grid to decode. Triton can load grid_thw[g].
    # We loop g from 0 to NUM_GRIDS-1 (constant at compile time since Triton doesn't support dynamic loops).
    for _ in range(NUM_GRIDS):
        t = tl.load(grid_thw_ptr + g * 3 + 0).to(tl.int32)
        h = tl.load(grid_thw_ptr + g * 3 + 1).to(tl.int32)
        w = tl.load(grid_thw_ptr + g * 3 + 2).to(tl.int32)
        h_merged = h // 2
        w_merged = w // 2
        num_rows = t * h_merged * w_merged

        # If m within this grid's contribution
        if m < num_rows:
            # Decode (t_index, h2, w2)
            t_index = m // (h_merged * w_merged)
            rem = m % (h_merged * w_merged)
            h2 = rem // w_merged
            w2 = rem % w_merged

            # Base offset in hidden_norm for this grid's contribution
            base_hidden = (t_index * (h * w))  # each original row has h*w elements

            # Compute 4 positions in 2x2 and write to out[m, :]
            # idx 0: (r=0,c=0) -> hidden_norm[t_index, h2, w2]
            idx0 = 0
            col0 = (h2 * 2 + 0) * (w_merged * C) + (w2 * 2 + 0) * C
            src0 = t_index * (h * w) + h2 * w + w2
            val0 = tl.load(hidden_norm_ptr + src0 * C + tl.arange(0, C), mask=None, other=0.0).to(tl.bfloat16)
            tl.store(out_ptr + m * (4 * C) + idx0 * C, val0, mask=None)

            # idx 1: (r=0,c=1) -> hidden_norm[t_index, h2, w2+1]
            idx1 = 1
            col1 = (h2 * 2 + 0) * (w_merged * C) + (w2 * 2 + 1) * C
            src1 = t_index * (h * w) + h2 * w + (w2 + 1)
            val1 = tl.load(hidden_norm_ptr + src1 * C + tl.arange(0, C), mask=None, other=0.0).to(tl.bfloat16)
            tl.store(out_ptr + m * (4 * C) + idx1 * C, val1, mask=None)

            # idx 2: (r=1,c=0) -> hidden_norm[t_index, h2+1, w2]
            idx2 = 2
            col2 = (h2 * 2 + 1) * (w_merged * C) + (w2 * 2 + 0) * C
            src2 = t_index * (h * w) + (h2 + 1) * w + w2
            val2 = tl.load(hidden_norm_ptr + src2 * C + tl.arange(0, C), mask=None, other=0.0).to(tl.bfloat16)
            tl.store(out_ptr + m * (4 * C) + idx2 * C, val2, mask=None)
