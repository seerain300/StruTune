import torch

import triton.language as tl

# Triton kernel: LayerNorm per row. One program per row.
@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32
    Each program handles one row.
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: compute mean and variance in fp32
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store bfloat16
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_row_ptr + offs, y.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE

# Triton kernel: Linear layer using row-wise GEMM: y[i_out,:] = x[i_out,:] @ W.T + b
# Specialized for K=6144, K_out=6144 (Linear1) and K_out=3584 (Linear2). One program per output row.
@triton.jit
def _linear_rowwise_kernel(x_ptr, W_ptr, b_ptr, y_ptr,
                            N, K, K_OUT,
                            BLOCK_K: tl.constexpr, BLOCK_K_OUT: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, K]
    W_ptr: *bf16, shape [K_OUT, K] (row-major: [k_out, k_in])
    b_ptr: *bf16, shape [K_OUT]
    y_ptr: *bf16, shape [N, K_OUT]
    Each program computes one output row i_out in [0, N).
    """
    i_out = tl.program_id(0)
    if i_out >= N:
        return

    y_row_ptr = y_ptr + i_out * K_OUT

    # We iterate over output columns in chunks of BLOCK_K_OUT
    k_out_start = 0
    while k_out_start < K_OUT:
        offs_k_out = k_out_start + tl.arange(0, BLOCK_K_OUT)
        mask_k_out = offs_k_out < K_OUT

        acc = tl.zeros([BLOCK_K_OUT], dtype=tl.float32)

        k_start = 0
        while k_start < K:
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # Load x[i_out, k] vector
            x_vec = tl.load(x_ptr + i_out * K + offs_k, mask=mask_k, other=0.0).to(tl.float32)
            # Load W[k_out, k] vector for current chunk
            W_vec = tl.load(W_ptr + offs_k_out * K + offs_k, mask=mask_k_out[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
            # Reduce over k dimension
            acc += tl.sum(W_vec * x_vec[None, :], axis=1)

            k_start += BLOCK_K

        # Add bias
        b_vec = tl.load(b_ptr + offs_k_out, mask=mask_k_out, other=0.0).to(tl.float32)
        acc = acc + b_vec

        tl.store(y_row_ptr + offs_k_out, acc.to(tl.bfloat16), mask=mask_k_out)

        k_out_start += BLOCK_K_OUT


# Triton kernel: GELU activation elementwise on [N, C] row-major, one program per row.
@triton.jit
def _gelu_kernel(x_ptr, y_ptr, N, C, BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, y_ptr: *bf16, shapes [N, C]
    Uses tanh-based GELU approximation.
    """
    row = tl.program_id(0)
    if row >= N:
        return
    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
        z = x * inv_sqrt2
        k = 0.7978845608028654  # sqrt(2/pi)
        t = k * (z + 0.044715 * z * z * z)
        gelu = 0.5 * x * (1.0 + tl.tanh(t))
        tl.store(y_row_ptr + offs, gelu.to(tl.bfloat16), mask=mask)
        col += BLOCK_SIZE

     - Linear1: [num_merged_patches, 6144] @ fc1_weight.T [6144, 6144] + fc1_bias
     - GELU activation
     - Linear2: [num_merged_patches, 3584] @ fc2_weight.T [3584, 6144] + fc2_bias
- Constraints:
  - No torch matmul, no torch.linear, no torch operations on tensors for math.
  - Triton-only: all kernels must be invoked from ModelNew.forward. We can use torch operations only for allocation and concatenation of per-grid outputs.

- Observations:
  - In many generator setups for such models, hidden rows are ordered so that per-grid rows are contiguous. The original code builds per-grid hidden tensors from hidden_norm without providing start indices; it simply uses t*h*w rows per grid. Given that, a reliable Triton-only approach for the shuffle is:
    - For each grid g, allocate out_grid [t * (h//2) * (w//2), 4*C].
    - Launch a Triton kernel per grid that treats hidden_norm as a source and decodes each row index m in [0, N) to (t_index, h2, w2) using t, h, w from grid_thw, and writes the 2x2 merge into out_grid. This mirrors the original behavior without needing per-grid start indices.
    - Concatenate all per-grid outputs to form hidden_shuffled. This preserves the order of outputs exactly as the original would have done since each grid’s outputs are already in the correct order within the Triton kernel.
- LayerNorm in Triton:
  - Implement a kernel that processes one row per program. Two-pass: first reduce to mean and variance in fp32, then normalize and apply ln_weight and ln_bias (affine) in fp32, store to bfloat16. This matches the original.
- MLP in Triton:
  - Implement row-wise GEMM kernels. Each program computes one output row vector:
    - For Linear1: iterate over K dimension in blocks, load weight vector chunk, multiply with input row chunk, reduce across K, add bias, store. Do this for K=6144 and K_OUT=6144 (specialized), then for K_OUT=3584.
    - For GELU: elementwise tanh-based GELU approximation in a Triton kernel, one program per row.
  - While a full 2D GEMM tiling can be faster, the row-wise approach keeps the code simple and robust for the given sizes and avoids Triton control-flow pitfalls. For K=6144, BLOCK_K=1024 leads to 6 iterations; performance is acceptable given the harness.

- Caveats and why prior attempts failed:
  - Implementing exact per-grid mapping without host-side slicing requires decoding row indices based on grid_thw inside Triton. Triton kernels cannot dynamically read arbitrary tensors (e.g., grid_thw) in ways that simplify per-grid slicing, and inter-kernel coordination is not available. The robust approach is to use Triton for the permutation itself by decoding indices inside the kernel. The earlier incorrect outputs were due to not matching the original per-grid layout exactly. The current approach decodes indices and writes per grid, then concatenates, which should match the original’s output order.

- Dtypes:
  - Use bfloat16 for inputs and outputs. Perform reductions and affine in float32 for numerical stability. Triton supports fp32 and bf16 operations; casting is handled via .to(tl.float32) and .to(tl.bfloat16) in kernels.

Final code
This defines ModelNew.forward that:
- Runs a Triton LayerNorm kernel on hidden to produce hidden_norm.
- Runs a Triton spatial-shuffle per-grid kernel to produce per-grid outputs, then concatenates them to hidden_shuffled using torch.cat (this is only for assembly, not math).
- Runs Triton row-wise linear kernels for Linear1 and Linear2, with Triton GELU in between.
- Returns the final output.

import math
import torch
import triton
import triton.language as tl


# Triton kernel: LayerNorm per row. One program per row.
@triton.jit
def _layer_norm_kernel(x_ptr, y_ptr, ln_weight_ptr, ln_bias_ptr,
                        N, C, eps,
                        BLOCK_SIZE: tl.constexpr):
    """
    x_ptr: *bf16, shape [N, C], row-major
    y_ptr: *bf16, shape [N, C], output
    ln_weight_ptr, ln_bias_ptr: *bf16, shape [C]
    eps: float32
    Each program handles one row: i in [0, N).
    """
    row = tl.program_id(0)
    if row >= N:
        return

    x_row_ptr = x_ptr + row * C
    y_row_ptr = y_ptr + row * C

    # Pass 1: compute mean and variance in fp32
    sum_val = 0.0
    sum_sq = 0.0
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_row_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        col += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Pass 2: normalize and apply affine, then store bfloat16
    col = 0
    while col < C:
        offs = col + tl.arange(0, BLOCK_SIZE
