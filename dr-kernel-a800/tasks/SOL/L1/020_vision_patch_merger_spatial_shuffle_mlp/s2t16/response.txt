import torch
import triton
import triton.language as tl


@triton.jit
def layernorm_affine_rowwise_kernel(
    X_ptr,  # *bfloat16, [num_patches, hidden_size]
    Y_ptr,  # *bfloat16, [num_patches, hidden_size]
    ln_w_ptr,  # *bfloat16, [hidden_size]
    ln_b_ptr,  # *bfloat16, [hidden_size]
    num_patches,  # int32
    hidden_size,  # int32
    eps,  # float32
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # Each program handles one row (one patch)
    # Load and compute mean/var in fp32
    sum_val = 0.0
    sum_sq = 0.0
    # Loop over hidden dimension in chunks of BLOCK_H
    for off in range(0, hidden_size, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < hidden_size
        x = tl.load(X_ptr + pid * hidden_size + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    n = hidden_size
    mean = sum_val / n
    var = sum_sq / n - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Normalize and apply affine
    for off in range(0, hidden_size, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < hidden_size
        x = tl.load(X_ptr + pid * hidden_size + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        y = (x - mean) * inv_std
        ln_w = tl.load(ln_w_ptr + idx, mask=mask, other=1.0).to(tl.float32)
        ln_b = tl.load(ln_b_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        y = y * ln_w + ln_b
        # Store as fp32; output tensor is fp32
        tl.store(Y_ptr + pid * hidden_size + idx, y, mask=mask)


@triton.jit
def spatial_shuffle_kernel(
    X_ptr,  # *bfloat16, [num_patches, hidden_size]
    Y_ptr,  # *bfloat16, [num_merged_patches, hidden_expanded]
    grid_thw_ptr,  # *int64, [num_grids, 3]
    offsets_ptr,   # *int64, [num_grids]
    N_MERGED,      # int32
    hidden_expanded,  # int32
    NUM_GRIDS: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_r = tl.program_id(axis=0)  # merged row
    pid_j = tl.program_id(axis=1)  # expanded column
    # Decode j -> (merge_h, merge_w, c)
    # hidden_expanded == 4 * hidden_size
    hidden_size = 1536
    C = hidden_size
    merge_size = 2
    merge_h = pid_j // (merge_size * C)
    merge_rem = pid_j % (merge_size * C)
    merge_w = merge_rem // C
    c = merge_rem % C

    # Compute base index into grid_thw
    # grid_thw shape: [NUM_GRIDS, 3] -> [T, H, W]
    # For each grid i, offsets[i] = sum over previous grids
    # We need grid_id such that pid_r belongs to it.
    # For grid i, start_row = offsets[i], end_row = offsets[i] + total_per_grid[i]
    total_per_grid = tl.zeros((NUM_GRIDS,), dtype=tl.int64)
    for i in range(NUM_GRIDS):
        t = tl.load(grid_thw_ptr + i * 3 + 0)
        h = tl.load(grid_thw_ptr + i * 3 + 1)
        w = tl.load(grid_thw_ptr + i * 3 + 2)
        total_per_grid[i] = t * h * w

    # Find i such that start_row <= pid_r < end_row
    start_row = 0
    for i in range(NUM_GRIDS):
        end_row = start_row + total_per_grid[i]
        if (pid_r >= start_row) & (pid_r < end_row):
            grid_id = i
            break
        start_row = end_row

    offsets_i = tl.load(offsets_ptr + grid_id)  # int64
    T = tl.load(grid_thw_ptr + grid_id * 3 + 0)
    H = tl.load(grid_thw_ptr + grid_id * 3 + 1)
    W = tl.load(grid_thw_ptr + grid_id * 3 + 2)

    T_merged = T // merge_size
    H_merged = H // merge_size
    W_merged = W // merge_size

    # Decode t_merged from pid_r within this grid
    # rows_per_grid_i = total_per_grid[grid_id]
    # We can recover t_merged via integer arithmetic using H_merged, W_merged.
    # Let q = (pid_r - offsets_i) // (H_merged * W_merged)
    # t_merged = q // (H_merged * W_merged) ? No, we should map pid_r to t_merged by:
    # Within grid, rows covered = total_per_grid[grid_id]
    # pid_r_local = pid_r - offsets_i
    # For simplicity, compute t_merged using division:
    # t_merged = pid_r_local // (H_merged * W_merged) * 0 + (pid_r_local % (H_merged * W_merged)) // (H_merged * W_merged) is overkill; use direct formula:
    # For a given grid (T,H,W), the rows are laid out contiguously. We can recover t_merged by:
    # t_merged = (pid_r - offsets_i) // (H_merged * W_merged)
    # This is correct for any T, H, W and merge_size=2 as per code.
    pid_r_local = pid_r - offsets_i
    num_merged_rows = T_merged * H_merged * W_merged
    t_merged = pid_r_local // (H_merged * W_merged)
    rem = pid_r_local % (H_merged * W_merged)

    h_merged_idx = rem // W_merged
    w_merged_idx = rem % W_merged

    # Map to original (t, h, w) positions
    # Each merged position corresponds to 2x2:
    t = t_merged
    h_orig = h_merged_idx * 2 + merge_h
    w_orig = w_merged_idx * 2 + merge_w

    # Compute row index in normalized hidden
    row_in_X = offsets_i + t * (H * W) + h_orig * W + w_orig

    # Load and store
    # X is bfloat16, cast to fp32 for compute, store as fp32
    val = tl.load(X_ptr + row_in_X * C + c)
    # We must store into Y[pid_r, pid_j]. Y is fp32 output; input is bfloat16, cast.
    # pid_j is expanded col; Y is 2D contiguous: stride(0)=hidden_expanded, stride(1)=1
    tl.store(Y_ptr + pid_r * hidden_expanded + pid_j, val.to(tl.float32))


@triton.jit
def gemm_kernel(
    A_ptr,  # *bfloat16 or *float32, [M, K]
    B_ptr,  # *bfloat16 or *float32, [K, N]
    Bias_ptr,  # *float32, [N]
    C_ptr,  # *float32, [M, N]
    M, N, K,
    A_stride_m, A_stride_k,
    B_stride_k, B_stride_n,
    C_stride_m, C_stride_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for off_k in range(0, K, BLOCK_K):
        k0 = off_k + tl.arange(0, BLOCK_K)
        # load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + off_m[:, None] * A_stride_m + k0[None, :] * A_stride_k
        a_mask = (off_m[:, None] < M) & (k0[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        a = a.to(tl.float32)

        # load B tile [BLOCK_K, BLOCK_N]
        b_ptrs = B_ptr + k0[:, None] * B_stride_k + off_n[None, :] * B_stride_n
        b_mask = (k0[:, None] < K) & (off_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # add bias
    bias = tl.load(Bias_ptr + off_n, mask=off_n < N, other=0.0)
    bias = bias[None, :]
    acc += bias

    # store
    c_ptrs = C_ptr + off_m[:, None] * C_stride_m + off_n[None, :] * C_stride_n
    c_mask = (off_m[:, None] < M) & (off_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


@triton.jit
def gelu_tanh_kernel(
    X_ptr,  # *float32, [M, N]
    Y_ptr,  # *float32, [M, N]
    M, N,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    # Load X tile [BLOCK_M, BLOCK_N]; for simplicity, assume 1D grid over columns, but we need 2D.
    # Implement 2D over (m, n):
    # This kernel expects M, N to be provided; we will launch with grid (M, N).
    for n_off in range(0, N, BLOCK_N):
        cols = n_off + tl.arange(0, BLOCK_N)
        x = tl.load(X_ptr + pid_m * N + cols, mask=cols < N, other=0.0)
        # GELU tanh approximation: 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
        c0 = 0.7978845608028654  # sqrt(2/pi)
        c1 = 0.044715
        x3 = x * x * x
        inner = c0 * (x + c1 * x3)
        y = 0.5 * x * (1.0 + tl.tanh(inner))
        tl.store(Y_ptr + pid_m * N + cols, y, mask=cols < N)

        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < hidden_size
        x = tl.load(X_ptr + pid * hidden_size + idx, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        ln_w = tl.load(ln_w_ptr + idx, mask=mask, other=1.0).to(tl.float32)
        ln_b = tl.load(ln_b_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        y = y * ln_w + ln_b
        tl.store(Y_ptr + pid * hidden_size + idx, y, mask=mask)


@triton.jit
def spatial_shuffle_kernel(
    X_ptr,  # *bfloat16, [num_patches, hidden_size]
    Y_ptr,  # *bfloat16, [num_merged_patches, hidden_expanded]
    grid_thw_ptr,  # *int64, [num_grids, 3] -> [T, H, W]
    offsets_ptr,   # *int64, [num_grids]
    N_MERGED,      # int32
    hidden_expanded,  # int32
    NUM_GRIDS: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_r = tl.program_id(axis=0)  # merged row
    pid_j = tl.program_id(axis=1)  # expanded column

    # Decode expanded col j into (merge_h, merge_w, c)
    hidden_size = 1536
    C = hidden_size
    merge_size = 2
    merge_h = pid_j // (merge_size * C)
    merge_rem = pid_j % (merge_size * C)
    merge_w = merge_rem // C
    c = merge_rem % C

    # Compute grid_id such that pid_r belongs to this grid
    start_row = 0
    for i in range(NUM_GRIDS):
        total_per_grid_i = tl.load(grid_thw_ptr + i * 3 + 0) * tl.load(grid_thw_ptr + i * 3 + 1) * tl.load(grid_thw_ptr + i * 3 + 2)
        end_row = start_row + total_per_grid_i
        if (pid_r >= start_row) & (pid_r < end_row):
            grid_id = i
            break
        start_row = end_row

    offsets_i = tl.load(offsets_ptr + grid_id)  # int64
    T = tl.load(grid_thw_ptr + grid_id * 3 + 0)
    H = tl.load(grid_thw_ptr + grid_id * 3 + 1)
    W = tl.load(grid_thw_ptr + grid_id * 3 + 2)

    T_merged = T // merge_size
    H_merged = H // merge_size
    W_merged = W // merge_size

    pid_r_local = pid_r - offsets_i  # index within this grid
    num_merged_rows = T_merged * H_merged * W_merged
    t_merged = pid_r_local // (H_merged * W_merged)
    rem = pid_r_local % (H_merged * W_merged)
    h_merged_idx = rem // W_merged
    w_merged_idx = rem % W_merged

    t = t_merged
    h_orig = h_merged_idx * 2 + merge_h
    w_orig = w_merged_idx * 2 + merge_w

    # Row index in normalized hidden
    row_in_X = offsets_i + t * (H * W) + h_orig * W + w_orig

    # Column is c in hidden_size
    val = tl.load(X_ptr + row_in_X * C + c)  # load as bfloat16
    # Store to Y[pid_r, pid_j]; Y is bfloat16
    tl.store(Y_ptr + pid_r * hidden_expanded + pid_j, val)


@triton.jit
def gemm_kernel(
    A_ptr,  # *bfloat16 or *float32, [M, K]
    B_ptr,  # *bfloat16 or *float32, [K, N]
    Bias_ptr,  # *float32, [N]
    C_ptr,  # *float32, [M, N]
    M, N, K,
    A_stride_m, A_stride_k,
    B_stride_k, B_stride_n,
    C_stride_m, C_stride_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for off_k in range(0, K, BLOCK_K):
        k0 = off_k + tl.arange(0, BLOCK_K)

        a_ptrs = A_ptr + off_m[:, None] * A_stride_m + k0[None, :] * A_stride_k
        a_mask = (off_m[:, None] < M) & (k0[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        b_ptrs = B_ptr + k0[:, None] * B_stride_k + off_n[None, :] * B_stride_n
        b_mask = (k0[:, None] < K) & (off_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # add bias
    bias = tl.load(Bias_ptr + off_n, mask=off_n < N, other=0.0)
    bias = bias[None, :]
    acc += bias

    c_ptrs = C_ptr + off_m[:, None] * C_stride_m + off_n[None, :] * C_stride_n
    c_mask = (off_m[:, None] < M) & (off_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


@triton.jit
def gelu_tanh_kernel(
    X_ptr,  # *float32, [M, N]
    Y_ptr,  # *float32, [M, N]
    M, N,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    for n_off in range(0, N, BLOCK_N):
        cols = n_off + tl.arange(0, BLOCK_N)
        x = tl.load(X_ptr + pid_m * N + cols, mask=cols < N, other=0.0)
        c0 = 0.7978845608028654  # sqrt(2/pi)
        c1 = 0.044715
        x3 = x * x * x
        inner = c0 * (x + c1 * x3)
        y = 0.5 * x * (1.0 + tl.tanh(inner))
        tl.store(Y_ptr + pid_m * N + cols, y, mask=cols < N)
