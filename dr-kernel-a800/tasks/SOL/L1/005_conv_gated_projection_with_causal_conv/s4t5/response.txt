import torch
import triton
import triton.language as tl

@triton.jit
def in_proj_linear_kernel(
    X_ptr,         # *const float, input x: (B, S, H)
    W_ptr,         # *const float, in_proj_weight: (I, H), I=3*H
    Out_ptr,       # *float, output BCx: (B, S, I)
    B: tl.int32,
    S: tl.int32,
    H: tl.int32,
    I: tl.int32,
    # strides for x
    x_b_stride: tl.int32, x_s_stride: tl.int32, x_h_stride: tl.int32,
    # strides for out
    out_b_stride: tl.int32, out_s_stride: tl.int32, out_i_stride: tl.int32,
    BLOCK_H: tl.constexpr,
):
    # grid = (B*S,)
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    x_base = X_ptr + b * x_b_stride + s * x_s_stride
    out_base = Out_ptr + b * out_b_stride + s * out_s_stride

    for i in range(0, I):
        acc = tl.zeros((), dtype=tl.float32)

        # Reduction over H
        for h in range(0, H, BLOCK_H):
            h_offsets = h + tl.arange(0, BLOCK_H)
            h_mask = h_offsets < H

            # Load x[b, s, h]
            x_ptrs = x_base + h_offsets * x_h_stride
            x_vals = tl.load(x_ptrs, mask=h_mask, other=0.0).to(tl.float32)

            # Load W[i, h] contiguous along h: stride for W is (I, H), so W_ptr + i*H + h
            w_ptrs = W_ptr + i * H + h_offsets
            w_vals = tl.load(w_ptrs, mask=h_mask, other=0.0).to(tl.float32)

            acc += tl.sum(x_vals * w_vals, axis=0)

        # Store out[b, s, i]
        out_ptr = out_base + i * out_i_stride
        tl.store(out_ptr, acc)


@triton.jit
def grouped_causal_conv1d_kernel(
    Bx_ptr,        # *const float, input after transpose + gating: shape (B, H, S)
    W_ptr,         # *const float, conv_weight: (H, 1, 4) flattened to (H, K)
    Bias_ptr,      # *const float, conv_bias: (H)
    Out_ptr,       # *float, output: (B, H, S)
    B: tl.int32,
    S: tl.int32,
    H: tl.int32,
    K: tl.constexpr,         # kernel_size = 4
    BLOCK_T: tl.constexpr,   # tile along S dimension
):
    # Each program handles one (b, g)
    pid = tl.program_id(axis=0)
    g = pid % H
    b = pid // H

    # Base pointers for this (b, g)
    Bx_base = Bx_ptr + b * (H * S) + g * S  # Bx layout is (B, H, S) contiguous
    Out_base = Out_ptr + b * (H * S) + g * S

    # We process t in tiles of BLOCK_T across S
    for t in range(0, S, BLOCK_T):
        t_offsets = t + tl.arange(0, BLOCK_T)
        t_mask = t_offsets < S

        acc = tl.zeros((BLOCK_T,), dtype=tl.float32)

        # Causal conv with kernel_size=K=4
        for k in range(0, K):
            pos = t_offsets + k - 1  # input position for t-k+1
            valid = (pos >= 0) & (pos < S) & t_mask

            # Load Bx[b, g, pos]
            bx_ptrs = Bx_base + pos
            bx_vals = tl.load(bx_ptrs, mask=valid, other=0.0).to(tl.float32)

            # Load conv_weight[g, k]
            w_val = tl.load(W_ptr + g * K + k).to(tl.float32)
            acc += bx_vals * w_val

        # Add bias[g]
        bias_val = tl.load(Bias_ptr + g).to(tl.float32)
        acc += bias_val

        # Store conv_out[b, g, t]
        out_ptrs = Out_base + t_offsets
        tl.store(out_ptrs, acc, mask=t_mask)


@triton.jit
def out_proj_linear_kernel(
    Y_ptr,         # *const float, input y: (B, S, H)
    W_ptr,         # *const float, out_proj_weight: (H, H)
    Out_ptr,       # *float, output: (B, S, H)
    B: tl.int32,
    S: tl.int32,
    H: tl.int32,
    # strides for y
    y_b_stride: tl.int32, y_s_stride: tl.int32, y_h_stride: tl.int32,
    # strides for W: (H, H)
    w_out_stride: tl.int32, w_in_stride: tl.int32,
    # strides for Out
    out_b_stride: tl.int32, out_s_stride: tl.int32, out_h_stride: tl.int32,
    BLOCK_H: tl.constexpr,
    BLOCK_T: tl.constexpr,  # BLOCK_T along S output; we produce H outputs per (b, s)
):
    # grid = (B*S,)
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    y_base = Y_ptr + b * y_b_stride + s * y_s_stride
    out_base = Out_ptr + b * out_b_stride + s * out_s_stride

    for h_out in range(0, H, BLOCK_H):
        h_out_offsets = h_out + tl.arange(0, BLOCK_H)
        h_out_mask = h_out_offsets < H

        acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

        # Reduction over input H
        for h_in in range(0, H, BLOCK_H):
            h_in_offsets = h_in + tl.arange(0, BLOCK_H)
            h_in_mask = h_in_offsets < H

            # y[b, s, h_in]
            y_ptrs = y_base + h_in_offsets * y_h_stride
            y_vals = tl.load(y_ptrs, mask=h_in_mask, other=0.0).to(tl.float32)

            # W[h_out, h_in]
            w_ptrs = W_ptr + h_out_offsets * w_out_stride + h_in_offsets * w_in_stride
            w_vals = tl.load(w_ptrs, mask=h_in_mask, other=0.0).to(tl.float32)

            acc += y_vals * w_vals

        # Store Out[b, s, h_out]
        out_ptrs = out_base + h_out_offsets * out_h_stride
        tl.store(out_ptrs, acc, mask=h_out_mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor,
                in_proj_weight: torch.Tensor,
                in_proj_bias: torch.Tensor,
                conv_weight: torch.Tensor,
                conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor,
                out_proj_bias: torch.Tensor):
        # x: (B, S, H), in_proj_weight: (I, H), I=3*H, conv_weight: (H, 1, 4), conv_bias: (H)
        # out_proj_weight: (H, H)
        device = x.device
        dtype = x.dtype

        B, S, H = x.shape
        I = 3 * H

        # Ensure float32 and contiguous for Triton kernels
        x_contig = x.contiguous().to(torch.float32)
        in_proj_weight_contig = in_proj_weight.contiguous().to(torch.float32)
        conv_weight_contig = conv_weight.contiguous().to(torch.float32)  # (H, 1, 4) => (H, 4)
        conv_bias_contig = conv_bias.contiguous().to(torch.float32)      # (H,)
        out_proj_weight_contig = out_proj_weight.contiguous().to(torch.float32)  # (H, H)

        # 1) Triton in_proj linear: compute BCx ∈ (B, S, I)
        BCx = torch.empty((B, S, I), device=device, dtype=torch.float32)
        grid_in = (B * S,)
        in_proj_linear_kernel[grid_in](
            x_contig, in_proj_weight_contig, BCx,
            B, S, H, I,
            x_b_stride=x_contig.stride(0), x_s_stride=x_contig.stride(1), x_h_stride=x_contig.stride(2),
            out_b_stride=BCx.stride(0), out_s_stride=BCx.stride(1), out_i_stride=BCx.stride(2),
            BLOCK_H=128,
            num_warps=4,
        )

        # 2) Slice/chunk BCx into B, C, x_proj: (B, S, H) each
        # BCx shape (B,S,I) with I=3H, chunk along last dim
        B_tensor = BCx[:, :, :H]           # (B, S, H)
        C_tensor = BCx[:, :, H:2*H]        # (B, S, H)
        x_proj_tensor = BCx[:, :, 2*H:]    # (B, S, H)

        # Elementwise gating: Bx = B * x_proj  (PyTorch for simplicity; non-heavy)
        # We will feed Bx to Triton conv by transposing to (B,H,S) for groups=H
        # First compute Bx in torch (elementwise), then pass it to Triton conv via view.
        Bx = B_tensor * x_proj_tensor  # (B, S, H)

        # Create transposed view for conv: (B, H, S)
        Bx_trans = Bx.transpose(1, 2).contiguous()  # shape (B, S, H) -> transpose to (B, H, S)

        # 3) Triton grouped causal conv: conv_out ∈ (B, H, S), kernel_size=4, groups=H
        conv_out = torch.empty((B, H, S), device=device, dtype=torch.float32)

        grid_conv = (B * H,)
        grouped_causal_conv1d_kernel[grid_conv](
            Bx_trans, conv_weight_contig.view(-1), conv_bias_contig,
            conv_out,
            B, S, H,
            K=4,
            BLOCK_T=128,
            num_warps=4,
        )

        # 4) Elementwise gating with C_tensor: y = C * conv_out
        # C_tensor: (B, S, H), conv_out: (B, H, S). We need elementwise multiply aligned.
        # Transpose C_tensor to (B, S, H) -> (B, H, S) for matching.
        C_trans = C_tensor.transpose(1, 2).contiguous()  # (B, H, S)
        y = C_trans * conv_out  # elementwise multiply in torch (non-heavy)

        # 5) Triton out_proj linear: y ∈ (B, S, H)
        # y is (B, S, H) after transpose back. We need to compute output = F.linear(y, out_proj_weight).
        # So we take y in original layout (B, S, H) by transposing y to (B, S, H).
        # Note: y is (B,H,S) from previous multiply; to get (B,S,H), transpose back.
        y_back = y.transpose(1, 2).contiguous()  # (B, S, H)
        output = torch.empty((B, S, H), device=device, dtype=torch.float32)

        grid_out = (B * S,)
        out_proj_linear_kernel[grid_out](
            y_back, out_proj_weight_contig,
            output,
            B, S, H,
            y_b_stride=y_back.stride(0), y_s_stride=y_back.stride(1), y_h_stride=y_back.stride(2),
            w_out_stride=out_proj_weight_contig.stride(0), w_in_stride=out_proj_weight_contig.stride(1),
            out_b_stride=output.stride(0), out_s_stride=output.stride(1), out_h_stride=output.stride(2),
            BLOCK_H=128,
            BLOCK_T=128,
            num_warps=4,
        )

        return output

        Bx = B_tensor * x_proj_tensor      # (B, S, H)

        # Transpose for conv input: (B, H, S)
        Bx_trans = Bx.transpose(1, 2).contiguous()  # (B, S, H) -> (B, H, S)

        # 3) Triton grouped causal conv: conv_out ∈ (B, H, S), kernel_size=4, groups=H
        conv_out = torch.empty((B, H, S), device=device, dtype=torch.float32)
        grid_conv = (B * H,)
        grouped_causal_conv1d_kernel[grid_conv](
            Bx_trans, conv_weight_contig, conv_bias_contig,
            conv_out,
            B, S, H,
            K=4,
            BLOCK_T=128,
            num_warps=4,
        )

        # 4) Elementwise gating with C_tensor: y = C * conv_out
        # C_tensor: (B, S, H), conv_out: (B, H, S). Transpose C to (B, H, S) then multiply.
        C_trans = C_tensor.transpose(1, 2).contiguous()  # (B, H, S)
        y = C_trans * conv_out  # elementwise multiply in torch (non-heavy)

        # 5) Triton out_proj linear: y ∈ (B, S, H)
        # y is (B, H, S) from previous multiply; to compute F.linear(y, out_proj_weight), we need y in (B, S, H).
        y_back = y.transpose(1, 2).contiguous()  # (B, S, H)
        output = torch.empty((B, S, H), device=device, dtype=torch.float32)

        grid_out = (B * S,)
        out_proj_linear_kernel[grid_out](
            y_back, out_proj_weight_contig,
            output,
            B, S, H,
            y_b_stride=y_back.stride(0), y_s_stride=y_back.stride(1), y_h_stride=y_back.stride(2),
            w_out_stride=out_proj_weight_contig.stride(0), w_in_stride=out_proj_weight_contig.stride(1),
            out_b_stride=output.stride(0), out_s_stride=output.stride(1), out_h_stride=output.stride(2),
            BLOCK_H=128,
            BLOCK_T=128,
            num_warps=4,
        )

        return output
