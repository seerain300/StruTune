import torch
import triton
import triton.language as tl

# 1) Triton kernel: linear projection producing (B, S, M) from x(B,S,H) and weight(M,H)
#    BCx[b, s, m] = sum_h x[b, s, h] * weight[m, h] + bias[m]
@triton.jit
def linear_bsh_kernel(
    x_ptr,          # *f32, (B, S, H)
    weight_ptr,     # *f32, (M, H) where M = 3*H
    bias_ptr,       # *f32, (M,)
    out_ptr,        # *f32, (B, S, M)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr, M: tl.constexpr,
    x_stride0, x_stride1, x_stride2,
    weight_stride0, weight_stride1,
    out_stride0, out_stride1, out_stride2,
    BLOCK_H: tl.constexpr,  # reduction tile over H
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    m = tl.program_id(2)  # channel index in [0, M)

    # Accumulator for this (b, s, m)
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over H in chunks
    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H

        # Load x[b, s, h_offsets]
        x_ptrs = x_ptr + b * x_stride0 + s * x_stride1 + h_offsets * x_stride2
        x_vals = tl.load(x_ptrs, mask=mask_h, other=0.0)  # (BLOCK_H,)

        # Load weight[m, h_offsets]
        w_ptrs = weight_ptr + m * weight_stride0 + h_offsets * weight_stride1
        w_vals = tl.load(w_ptrs, mask=mask_h, other=0.0)  # (BLOCK_H,)

        # Multiply-accumulate: sum over BLOCK_H
        acc += tl.sum(x_vals * w_vals, axis=0)

    # Add bias[m]
    bval = tl.load(bias_ptr + m)
    acc += bval

    # Store result to out[b, s, m]
    out_ptr_val = out_ptr + b * out_stride0 + s * out_stride1 + m * out_stride2
    tl.store(out_ptr_val, acc)


# 2) Elementwise multiplication kernel: out[b, h, s] = A[b, h, s] * B[b, h, s]
@triton.jit
def elemwise_mul_bsh_kernel(
    A_ptr, B_ptr, Out_ptr,
    B, H, S,
    A_stride0, A_stride1, A_stride2,
    B_stride0, B_stride1, B_stride2,
    Out_stride0, Out_stride1, Out_stride2,
    BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile = tl.program_id(2)
    s_offsets = tile * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    A_ptrs = A_ptr + b * A_stride0 + h * A_stride1 + s_offsets * A_stride2
    B_ptrs = B_ptr + b * B_stride0 + h * B_stride1 + s_offsets * B_stride2
    Out_ptrs = Out_ptr + b * Out_stride0 + h * Out_stride1 + s_offsets * Out_stride2

    A_vals = tl.load(A_ptrs, mask=mask_s, other=0.0)
    B_vals = tl.load(B_ptrs, mask=mask_s, other=0.0)
    tl.store(Out_ptrs, A_vals * B_vals, mask=mask_s)


# 3) Grouped causal 1D convolution: out[b, c, s] = sum_{k=0..3} Bx[b, c, s+k-1] * convW[c, c, k] + convB[c]
#    Input Bx: (B, S, H), convW: (H, H, 4), convB: (H,)
#    Output conv_out: (B, H, S)
@triton.jit
def grouped_causal_conv1d_bsh_to_bhs_kernel(
    Bx_ptr,         # *f32, (B, S, H)
    convW_ptr,      # *f32, (H, H, 4)
    convB_ptr,      # *f32, (H,)
    out_ptr,        # *f32, (B, H, S)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    Bx_stride0, Bx_stride1, Bx_stride2,
    convW_stride0, convW_stride1, convW_stride2,
    out_stride0, out_stride1, out_stride2,
    BLOCK_S: tl.constexpr
):
    b_idx = tl.program_id(0)   # batch
    c_idx = tl.program_id(1)   # channel
    tile_s = tl.program_id(2)  # tile along sequence

    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # For kernel_size=4 (default in original code), apply left padding of 3.
    for k in range(4):
        t_offsets = s_offsets + k - 1  # this is the index along the original S axis before padding
        valid = t_offsets >= 0  # True for positions that map to valid input indices
        Bx_ptrs = Bx_ptr + b_idx * Bx_stride0 + t_offsets * Bx_stride1 + c_idx * Bx_stride2
        Bx_vals = tl.load(Bx_ptrs, mask=mask_s & valid, other=0.0)
        w_ptr = convW_ptr + c_idx * convW_stride0 + c_idx * convW_stride1 + k * convW_stride2
        w_val = tl.load(w_ptr)  # scalar
        acc += Bx_vals * w_val

    # Add bias[c]
    bval = tl.load(convB_ptr + c_idx)
    acc += bval

    out_ptrs = out_ptr + b_idx * out_stride0 + c_idx * out_stride1 + s_offsets * out_stride2
    tl.store(out_ptrs, acc, mask=mask_s)


# 4) Triton kernel: final linear projection y_bhs[B,H,S] -> output[B,S,H]
#    output[b, s, h] = sum_c y[b, h, s] * out_proj_weight[h, c] + out_proj_bias[h]
@triton.jit
def linear_bhs_to_bsh_kernel(
    y_ptr,          # *f32, (B, H, S)
    out_proj_weight_ptr,  # *f32, (H, H)
    out_proj_bias_ptr,    # *f32, (H,)
    out_ptr,        # *f32, (B, S, H)
    B: tl.constexpr, H: tl.constexpr, S: tl.constexpr,
    y_stride0, y_stride1, y_stride2,
    out_proj_weight_stride0, out_proj_weight_stride1,
    out_stride0, out_stride1, out_stride2,
    BLOCK_H: tl.constexpr  # reduction tile over H
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)  # output hidden index in [0, H)

    acc = tl.zeros((), dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H

        # y[b, h_offsets, s]
        y_ptrs = y_ptr + b * y_stride0 + h_offsets * y_stride1 + s * y_stride2
        y_vals = tl.load(y_ptrs, mask=mask_h, other=0.0)  # (BLOCK_H,)

        # out_proj_weight[h, h_offsets] -> but in PyTorch we use y[H,S] and weight[H,H], we want weight[h, :] for each h to reduce over S, but here it's weight[H,H] with dim 0 = output channels (S), dim 1 = input channels (H). We need to read weight[:, h] and then multiply with y_vals.
        # However, we want output[b, s, h] = sum_c y[b, h, s] * out_proj_weight[h, c]. Note y has shape (B,H,S) and out_proj_weight is (H,H). The original PyTorch uses y (B,H,S) and weight (H,H) to produce (B,S,H). The correct mapping is: for each (b,s,h), sum over c: y[b,h,s] * weight[h,c] + bias[h]. That means we need out_proj_weight[h, :] as the reduction vector, not weight[:, h]. So correct access is weight[h, c_offsets], reducing over c.

        # We need to read out_proj_weight[h, c_offsets] where c_offsets correspond to H. The original uses out_proj_weight of shape (H,H). We'll treat weight[h, c] as the reduction coefficients. To implement, we build c_offsets and load weight[h, c_offsets], multiply with y_vals, and reduce.
        # But y_vals are (BLOCK_H,) for h_offsets, and we want to multiply with weight[h_offsets, c_offsets]. To keep it simple and correct, we'll use BLOCK_H=H in practice (H is a constexpr at launch). We'll set BLOCK_H to H to avoid partial tiles.
        c_offsets = h_start + tl.arange(0, BLOCK_H)
        mask_c = c_offsets < H

        w_ptrs = out_proj_weight_ptr + h_offsets[:, None] * out_proj_weight_stride0 + c_offsets[None, :] * out_proj_weight_stride1
        w_vals = tl.load(w_ptrs, mask=mask_h[:, None] & mask_c[None, :], other=0.0)  # (BLOCK_H, BLOCK_H)
        # Multiply and reduce over c: sum over axis=1
        prod = y_vals[:, None] * w_vals  # (BLOCK_H, BLOCK_H)
        acc += tl.sum(prod, axis=1)

    # Add bias[h]
    bval = tl.load(out_proj_bias_ptr + h)
    acc += bval

    # Store output[b, s, h]
    out_ptr_val = out_ptr + b * out_stride0 + s * out_stride1 + h * out_stride2
    tl.store(out_ptr_val, acc)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor,
                in_proj_weight: torch.Tensor, in_proj_bias: torch.Tensor,
                conv_weight: torch.Tensor, conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor, out_proj_bias: torch.Tensor):
        """
        Fused computation using Triton kernels:
        1) BCx = linear(x, in_proj_weight, in_proj_bias) -> (B, S, 3H)
        2) B = BCx[:, :, :H], C = BCx[:, :, H:2H], X = BCx[:, :, 2H:]
        3) Bx = B * X -> (B, H, S)
        4) conv_out = grouped causal conv1d(Bx, conv_weight, conv_bias) -> (B, H, S)
        5) y = C * conv_out -> (B, H, S)
        6) output = linear(y, out_proj_weight, out_proj_bias) -> (B, S, H)
        Returns tensor of dtype torch.float32 with shape (B, S, H).
        """
        # Ensure all inputs are float32 and contiguous
        device = x.device
        # Create outputs explicitly as float32
        B, S, H = x.shape
        M = 3 * H  # in_proj_weight is (3*H, H)

        # 1) Compute BCx: (B, S, 3H) = x @ in_proj_weight.T + in_proj_bias
        BCx = torch.empty((B, S, M), device=device, dtype=torch.float32)
        grid_linear = (B, S, M)
        linear_bsh_kernel[grid_linear](
            x, in_proj_weight, in_proj_bias, BCx,
            B, S, H, M,
            x.stride(0), x.stride(1), x.stride(2),
            in_proj_weight.stride(0), in_proj_weight.stride(1),
            BCx.stride(0), BCx.stride(1), BCx.stride(2),
            BLOCK_H=64,
            num_warps=4, num_stages=2
        )

        # 2) Slice into B, C, X: each (B, S, H)
        B_bsh = BCx[:, :, :H].contiguous()
        C_bsh = BCx[:, :, H:2*H].contiguous()
        X_bsh = BCx[:, :, 2*H:].contiguous()

        # 3) Element-wise gating: Bx = B * X -> (B, S, H)
        Bx = torch.empty((B, S, H), device=device, dtype=torch.float32)
        grid_mul1 = (B, H, (S + 128 - 1) // 128)
        elemwise_mul_bsh_kernel[grid_mul1](
            B_bsh, X_bsh, Bx,
            B, H, S,
            B_bsh.stride(0), B_bsh.stride(1), B_bsh.stride(2),
            X_bsh.stride(0), X_bsh.stride(1), X_bsh.stride(2),
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            BLOCK_S=128,
            num_warps=4, num_stages=2
        )

        # 4) Grouped causal 1D convolution: conv_out (B, H, S)
        conv_out = torch.empty((B, H, S), device=device, dtype=torch.float32)
        grid_conv = (B, H, (S + 128 - 1) // 128)
        grouped_causal_conv1d_bsh_to_bhs_kernel[grid_conv](
            Bx, conv_weight, conv_bias, conv_out,
            B, S, H,
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            conv_weight.stride(0), conv_weight.stride(1), conv_weight.stride(2),
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            BLOCK_S=128,
            num_warps=4, num_stages=2
        )

        # 5) Output gating: y = C * conv_out -> (B, H, S)
        y_bhs = torch.empty((B, H, S), device=device, dtype=torch.float32)
        grid_mul2 = (B, H, (S + 128 - 1) // 128)
        elemwise_mul_bsh_kernel[grid_mul2](
            C_bsh, conv_out, y_bhs,
            B, H, S,
            C_bsh.stride(0), C_bsh.stride(1), C_bsh.stride(2),
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            y_bhs.stride(0), y_bhs.stride(1), y_bhs.stride(2),
            BLOCK_S=128,
            num_warps=4, num_stages=2
        )

        # 6) Final linear projection: output (B, S, H)
        output = torch.empty((B, S, H), device=device, dtype=torch.float32)
        grid_final = (B, S, H)
        # For out_proj_weight (H,H) and y_bhs (B,H,S), output[b,s,h] = sum_c y[b,h,s] * weight[h,c] + bias[h]
        # We need to read weight[h, c] and reduce over c. The kernel linear_bhs_to_bsh_kernel implements this by loading weight[h, c_offsets], multiplying with y_vals[h_offsets, s], and reducing over c.
        linear_bhs_to_bsh_kernel[grid_final](
            y_bhs, out_proj_weight, out_proj_bias, output,
            B, H, S,
            y_bhs.stride(0), y_bhs.stride(1), y_bhs.stride(2),
            out_proj_weight.stride(0), out_proj_weight.stride(1),
            output.stride(0), output.stride(1), output.stride(2),
            BLOCK_H=H,  # we set BLOCK_H to H since H is small in typical workloads; this avoids partial tiles.
            num_warps=4, num_stages=2
        )

        return output
