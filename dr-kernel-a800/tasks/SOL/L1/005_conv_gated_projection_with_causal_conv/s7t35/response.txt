    b_ptr,                   # *float32, in_proj_bias (3H)

    stride_w_i, stride_w_h, stride_w_l,          # strides for w (3H, H, L)
    stride_BC_b, stride_BC_i, stride_BC_l        # strides for BCx (B, 3H, L)
):

    i_block = tl.program_id(0)  # over 3H
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    i_offsets = i_block * 64 + tl.arange(0, 64)     # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128)   # [128]

    mask_l = l_offsets < L

    h_i = i_offsets % H

    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)  # (64, 128)

# Input Bx: (B, H, L), conv_weight: (H, H, 4), conv_bias: (H)
# Output conv_out: (B, H, L)
@triton.jit
def conv1d_grouped_causal_kernel(
    Bx_ptr,                # *float32, input Bx (B, H, L) contiguous
    w_ptr,                 # *float32, weight (H, H, 4) contiguous
    bias_ptr,              # *float32, bias (H)
    out_ptr,               # *float32, output (B, H, L) contiguous
    B, L, H,               # sizes
    stride_Bx_b, stride_Bx_h, stride_Bx_l,   # strides for Bx (B, H, L)
    stride_w_o, stride_w_i, stride_w_k,      # strides for w (H, H, 4)
    stride_out_b, stride_out_h, stride_out_l  # strides for out (B, H, L)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)     # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128)   # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    acc = tl.zeros((64, 128), dtype=tl.float32)

    # For grouped conv with groups=H: each output channel h uses its own input channel h and kernel across l
    # Since Bx is contiguous (B, H, L), we can load along L dimension.
    for ho in range(0, H):
        # Load Bx[b, ho, l] as (1, 128), broadcast later
        Bx_ptrs = Bx_ptr + b * stride_Bx_b + ho * stride_Bx_h + l_offsets[None, :] * stride_Bx_l
        Bx_vals = tl.load(Bx_ptrs, mask=mask_l[None, :], other=0.0)  # (1, 128)

        # Sum over kernel positions k in [0, 4)
        for k in range(0, 4):
            # weight w[ho, ho, k] scalar
            w_ptrs = w_ptr + ho * stride_w_o + ho * stride_w_i + k * stride_w_k
            w_val = tl.load(w_ptrs)
            # Accumulate: out[b, ho, l] += w_val * Bx[b, ho, l - k]
            # For causal conv with padding, we need Bx at shifted indices; since we padded on host,
            # here we directly multiply with Bx at current l (the padded version is handled by host).
            # Note: This kernel assumes host has already applied causal padding via F.pad.
            acc += w_val * Bx_vals

    # Add bias
    bias_vals = tl.load(bias_ptr + h_offsets, mask=mask_h, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store to out[b, h, l]
    out_ptrs = out_ptr + b * stride_out_b + h_offsets[:, None] * stride_out_h + l_offsets[None, :] * stride_out_l
    tl.store(out_ptrs, acc, mask=mask)


# 5) Out-projection: y (B, L, H) -> out (B, L, H) with weight (H, L, H)
@triton.jit
def out_proj_kernel(
    y_ptr,                   # *float32, input y (B, L, H) contiguous
    w_ptr,                   # *float32, weight (H, L, H) contiguous
    b_ptr,                   # *float32, bias (H)
    out_ptr,                 # *float32, output (B, L, H) contiguous
    B, L, H,                 # sizes
    stride_y_b, stride_y_l, stride_y_h,       # strides for y (B, L, H)
    stride_w_o, stride_w_l, stride_w_i,       # strides for w (H, L, H)
    stride_out_b, stride_out_l, stride_out_h # strides for out (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)     # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128)   # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # Load y[b, l, h]
    y_ptrs = y_ptr + b * stride_y_b + l_offsets[None, :] * stride_y_l + h_offsets[:, None] * stride_y_h
    y_vals = tl.load(y_ptrs, mask=mask, other=0.0)  # (64, 128)

    # Accumulate acc = sum_h w[h, l, h] * y[b, l, h]
    acc = tl.zeros((64, 128), dtype=tl.float32)
    for ho in range(0, H):
        # weight scalar w[ho, l, ho]
        w_ptrs = w_ptr + ho * stride_w_o + l_offsets[None, :] * stride_w_l + ho * stride_w_i
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (1, 128), broadcast along ho
        acc += w_vals * y_vals  # broadcast y_vals along ho

    # Add bias
    bias_vals = tl.load(b_ptr + h_offsets, mask=mask_h, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store output
    out_ptrs = out_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + h_offsets[:, None] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask)

    i_block = tl.program_id(0)  # over 3H
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    i_offsets = i_block * 64 + tl.arange(0, 64)     # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128)   # [128]

    mask_i = i_offsets < (3 * H)
    mask_l = l_offsets < L
    mask = mask_i[:, None] & mask_l[None, :]

    # For each i in [0, 3H), compute BCx[b, i, l] = sum_h w[i, h, l] * x[b, l, h] + bias[i]
    # We'll do a simple dot per i across h (vectorized)
    # Note: w is (3H, H, L), x is (B, L, H), BCx is (B, 3H, L)
    for i in range(0, 64):
        # i may exceed 3H if i_block * 64 + i >= 3*H; use mask to guard
        if i_offsets[i] >= (3 * H):
            continue
        # Load x[b, l, h] for all h in a vector fashion: h = i_offsets % H
        h_i = i_offsets % H
        x_ptrs = x_ptr + b * stride_x_b + l_offsets[None, :] * stride_x_l + h_i[:, None] * stride_x_h
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)  # (64, 128), but only i-th row contributes

        # Load w[i, h, l]
        w_ptrs = w_ptr + i_offsets[i] * stride_w_i + h_i[None, :] * stride_w_h + l_offsets[None, :] * stride_w_l
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (1, 128), broadcast along rows

        # Compute out_vals = w_vals * x_vals + bias[i]
        bias_val = tl.load(b_ptr + i_offsets[i])  # scalar
        out_vals = w_vals * x_vals + bias_val

        # Store to BCx[b, i, l]
        BC_ptrs = BCx_ptr + b * stride_BC_b + i_offsets[i] * stride_BC_i + l_offsets[None, :] * stride_BC_l
        tl.store(BC_ptrs, out_vals, mask=mask_l[None, :])


# 2) Chunk BCx (B, 3H, L) into three parts (B, L, H): B_tensor, C_tensor, x_proj
@triton.jit
def chunk3_kernel(
    BCx_ptr,                 # *float32, input BCx (B, 3H, L)
    B_out_ptr,               # *float32, output B_tensor (B, L, H)
    C_out_ptr,               # *float32, output C_tensor (B, L, H)
    X_out_ptr,               # *float32, output x_proj (B, L, H)
    B, L, H,                 # sizes
    stride_BC_b, stride_BC_i, stride_BC_l,    # strides for BCx (B, 3H, L)
    stride_B_b, stride_B_l, stride_B_h,       # strides for B_out (B, L, H)
    stride_C_b, stride_C_l, stride_C_h,       # strides for C_out (B, L, H)
    stride_X_b, stride_X_l, stride_X_h        # strides for X_out (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # Copy x_proj from BCx[b, l, 0..H-1]
    for ho in range(0, H):
        BC_ptrs_x = BCx_ptr + b * stride_BC_b + l_offsets[None, :] * stride_BC_l + ho * stride_BC_i
        vals_x = tl.load(BC_ptrs_x, mask=mask, other=0.0)  # (64,128)
        X_out_ptrs_x = X_out_ptr + b * stride_X_b + l_offsets[None, :] * stride_X_l + ho * stride_X_h
        tl.store(X_out_ptrs_x, vals_x, mask=mask)

    # Copy B from BCx[b, l, H..2H-1]
    for ho in range(0, H):
        BC_ptrs_B = BCx_ptr + b * stride_BC_b + l_offsets[None, :] * stride_BC_l + (H + ho) * stride_BC_i
        vals_B = tl.load(BC_ptrs_B, mask=mask, other=0.0)  # (64,128)
        B_out_ptrs_B = B_out_ptr + b * stride_B_b + l_offsets[None, :] * stride_B_l + ho * stride_B_h
        tl.store(B_out_ptrs_B, vals_B, mask=mask)

    # Copy C from BCx[b, l, 2H..3H-1]
    for ho in range(0, H):
        BC_ptrs_C = BCx_ptr + b * stride_BC_b + l_offsets[None, :] * stride_BC_l + (2 * H + ho) * stride_BC_i
        vals_C = tl.load(BC_ptrs_C, mask=mask, other=0.0)  # (64,128)
        C_out_ptrs_C = C_out_ptr + b * stride_C_b + l_offsets[None, :] * stride_C_l + ho * stride_C_h
        tl.store(C_out_ptrs_C, vals_C, mask=mask)
