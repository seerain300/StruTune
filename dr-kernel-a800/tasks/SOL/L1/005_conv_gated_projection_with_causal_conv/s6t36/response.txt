    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


@triton.jit
def _elementwise_mul_1d(
    A_ptr, B_ptr, Out_ptr,
    M, N,
    stride_am, stride_an, stride_om,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < M * N
    a = tl.load(A_ptr + offs * stride_am, mask=mask, other=0.0)
    b = tl.load(B_ptr + offs * stride_an, mask=mask, other=0.0)
    out = a * b
    tl.store(Out_ptr + offs * stride_om, out, mask=mask)


@triton.jit
def _pad_left_kernel(
    In_ptr, Out_ptr,
    B, H, S, P,  # P = S + pad
    stride_ib, stride_ih, stride_is,   # input strides: (B, H, S)
    stride_ob, stride_oh, stride_os,   # output strides: (B, H, P)
    pad: tl.constexpr,
):
    # Each program handles one (b, h) row and writes to Out[b, h, :]
    b = tl.program_id(0)
    h = tl.program_id(1)
    in_base = In_ptr + b * stride_ib + h * stride_ih
    out_base = Out_ptr + b * stride_ob + h * stride_oh

    # Write zeros for the first 'pad' positions
    for j in range(pad):
        tl.store(out_base + j * stride_os, 0.0)

    # Copy middle region: start at index pad
    for s in range(S):
        tl.store(out_base + (s + pad) * stride_os, tl.load(in_base + s * stride_is))
