    W0_ptr, b0_ptr,   # (H, H), (H,)
    W1_ptr, b1_ptr,   # (H, H), (H,)
    W2_ptr, b2_ptr,   # (H, H), (H,)
    B_out_ptr,        # *f32, (B, S, H)
    C_out_ptr,        # *f32, (B, S, H)

    stride_xb, stride_xs, stride_xh,
    stride_w0n, stride_w0k,
    stride_w1n, stride_w1k,
    stride_w2n, stride_w2k,
    stride_bb, stride_bs, stride_bh,

    stride_xb, stride_xs, stride_xh,

    h_out = tl.program_id(1)
    s_block = tl.program_id(2)

def grouped_causal_conv1d_kernel(

                conv_w_ptr_k = conv_w_ptr + c * stride_wgn + c * stride_wgk + k2 * stride_wgk2
                conv_w_val = tl.load(conv_w_ptr_k)
                acc += y_val * conv_w_val

        acc += conv_b_val
        out_ptr_t = out_ptr + b * stride_ob + c * stride_on + t * stride_os

def final_linear_gemv_bsh_kernel(
    y_ptr,            # *f32, (B, S, H)
    wy_ptr,           # *f32, (H, H) out_proj_weight
    bb_ptr,           # *f32, (H,) out_proj_bias
    out_ptr,          # *f32, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_yb, stride_ys, stride_yh,
    stride_wyn, stride_wyk,        # wy is (H,H): n=channel (output feature), k=input feature
    stride_ob, stride_os, stride_oh,
    K_BLOCK: tl.constexpr,         # tile size over H
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    for h_out in range(0, H):
        acc = 0.0
        for k in range(0, H, K_BLOCK):
            k_offsets = k + tl.arange(0, K_BLOCK)
            mask_k = k_offsets < H
            y_ptrs = y_ptr + b * stride_yb + s * stride_ys + k_offsets * stride_yh
            y_vals = tl.load(y_ptrs, mask=mask_k, other=0.0)  # (K_BLOCK,)
            wy_ptrs = wy_ptr + h_out * stride_wyn + k_offsets * stride_wyk
            wy_vals = tl.load(wy_ptrs, mask=mask_k, other=0.0)
            # elementwise multiply and reduce
            for i in range(K_BLOCK):
                acc += y_vals[i] * wy_vals[i]
        bias_val = tl.load(bb_ptr + h_out * stride_oh)
        out_val = acc + bias_val
        out_ptr_hs = out_ptr + b * stride_ob + s * stride_os + h_out * stride_oh
        tl.store(out_ptr_hs, out_val)

        B_ptr = B_out_ptr + b * stride_bb + s * stride_bs + h_out * stride_bh
        C_ptr = C_out_ptr + b * stride_cb + s * stride_cs + h_out * stride_ch
        X_ptr = X_out_ptr + b * stride_xb + s * stride_xs + h_out * stride_xh
        tl.store(B_ptr, acc0)
        tl.store(C_ptr, acc1)
        tl.store(X_ptr, acc2)


@triton.jit
def grouped_causal_conv1d_kernel(
    y_in_ptr,         # *f32, (B, H, S) (pre-gated Bx)
    conv_w_ptr,       # *f32, (H, H, 4) weight per channel, kernel=4, groups=H
    conv_b_ptr,       # *f32, (H,)
    out_ptr,          # *f32, (B, H, S)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_yb, stride_yn, stride_ys,   # y_in strides
    stride_wgn, stride_wgk, stride_wgk2,  # conv_w strides (group n, channel k, k2 in {0..3})
    stride_ob, stride_on, stride_os,     # out strides
    BLOCK_S: tl.constexpr,
):
    # grid = (B, H, ceil(S/BLOCK_S))
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_block = tl.program_id(2)
    s_start = s_block * BLOCK_S
    for t in range(s_start, tl.minimum(s_start + BLOCK_S, S)):
        acc = 0.0
        # causal 1D with kernel_size=4: sum over k=0..3 of y[b, c, t+k-1] * conv_w[c, c, k]
        for k2 in range(0, 4):
            in_t = t + k2 - 1
            if in_t >= 0 and in_t < S:
                y_ptr = y_in_ptr + b * stride_yb + c * stride_yn + in_t * stride_ys
                y_val = tl.load(y_ptr)
                conv_w_ptr_k = conv_w_ptr + c * stride_wgn + c * stride_wgk + k2 * stride_wgk2
                conv_w_val = tl.load(conv_w_ptr_k)
                acc += y_val * conv_w_val
        # add bias
        conv_b_val = tl.load(conv_b_ptr + c)
        acc += conv_b_val
        out_ptr_t = out_ptr + b * stride_ob + c * stride_on + t * stride_os
        tl.store(out_ptr_t, acc)


@triton.jit
def final_linear_gemv_bsh_kernel(
    y_ptr,            # *f32, (B, S, H)
    wy_ptr,           # *f32, (H, H) out_proj_weight
    bb_ptr,           # *f32, (H,) out_proj_bias
    out_ptr,          # *f32, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_yb, stride_ys, stride_yh,
    stride_wyn, stride_wyk,        # wy is (H,H): n=channel (output feature), k=input feature
    stride_ob, stride_os, stride_oh,
    K_BLOCK: tl.constexpr,         # tile size over H
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    for h_out in range(0, H):
        acc = 0.0
        for k in range(0, H, K_BLOCK):
            k_offsets = k + tl.arange(0, K_BLOCK)
            mask_k = k_offsets < H
            y_ptrs = y_ptr + b * stride_yb + s * stride_ys + k_offsets * stride_yh
            y_vals = tl.load(y_ptrs, mask=mask_k, other=0.0)  # (K_BLOCK,)
            wy_ptrs = wy_ptr + h_out * stride_wyn + k_offsets * stride_wyk
            wy_vals = tl.load(wy_ptrs, mask=mask_k, other=0.0)
            # elementwise multiply and reduce
            for i in range(K_BLOCK):
                acc += y_vals[i] * wy_vals[i]
        bias_val = tl.load(bb_ptr + h_out * stride_oh)
        out_val = acc + bias_val
        out_ptr_hs = out_ptr + b * stride_ob + s * stride_os + h_out * stride_oh
        tl.store(out_ptr_hs, out_val)


# Triton elementwise kernels used for gating (Bx = A * B and y = C * conv_out)
@triton.jit
def elementwise_mul_bsh_kernel(
    a_ptr, b_ptr, out_ptr,
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_ab, stride_as, stride_ah,
    stride_bb, stride_bs, stride_bh,
    stride_ob, stride_os, stride_oh,
    BLOCK_S: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s_block = tl.program_id(2)
    s_start = s_block * BLOCK_S
    for s in range(s_start, tl.minimum(s_start + BLOCK_S, S)):
        a_ptr_s = a_ptr + b * stride_ab + s * stride_as + h * stride_ah
        b_ptr_s = b_ptr + b * stride_bb + s * stride_bs + h * stride_bh
        a_val = tl.load(a_ptr_s)
        b_val = tl.load(b_ptr_s)
        out_ptr_s = out_ptr + b * stride_ob + s * stride_os + h * stride_oh
        tl.store(out_ptr_s, a_val * b_val)

        # x: (B, S, H)
        x = x.contiguous()  # ensure predictable strides
        B = x.shape[0]
        S = x.shape[1]
        H = x.shape[2]

        b2 = in_proj_bias[2*H:3*H].contiguous()   # (H,)
