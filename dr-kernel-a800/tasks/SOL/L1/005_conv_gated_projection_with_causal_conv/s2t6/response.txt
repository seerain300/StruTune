import torch
import triton
import triton.language as tl


@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,        # *f32, (B, S, H)
    W0_ptr, b0_ptr,     # W0: (H, H), b0: (H,)
    W1_ptr, b1_ptr,     # W1: (H, H), b1: (H,)
    W2_ptr, b2_ptr,     # W2: (H, H), b2: (H,)
    B_out_ptr, C_out_ptr, X_out_ptr,  # outputs: (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_xb, stride_xs, stride_xh,
    stride_W0n, stride_W0k, stride_b0,
    stride_W1n, stride_W1k, stride_b1,
    stride_W2n, stride_W2k, stride_b2,
    stride_Bob, stride_Bos, stride_Boh,
    stride_Cob, stride_Cos, stride_Coh,
    stride_Xob, stride_Xos, stride_Xoh,
    BLOCK_H: tl.constexpr,
):
    # program ids
    b = tl.program_id(0)
    h_out_group = tl.program_id(1)  # 0 for B, 1 for C, 2 for x_proj
    s_block = tl.program_id(2)

    # tile over S
    s_start = s_block * BLOCK_H
    s_offsets = s_start + tl.arange(0, BLOCK_H)
    mask_s = s_offsets < S

    # loop over h_out
    for h in range(0, H):
        acc = tl.zeros((), dtype=tl.float32)

        # choose which weight/bias based on h_out_group
        if h_out_group == 0:
            W = W0_ptr
            b = b0_ptr
            stride_Wn = stride_W0n
            stride_Wk = stride_W0k
            stride_b = stride_b0
        elif h_out_group == 1:
            W = W1_ptr
            b = b1_ptr
            stride_Wn = stride_W1n
            stride_Wk = stride_W1k
            stride_b = stride_b1
        else:
            W = W2_ptr
            b = b2_ptr
            stride_Wn = stride_W2n
            stride_Wk = stride_W2k
            stride_b = stride_b2

        # load bias scalar
        bias = tl.load(b + h * stride_b)

        # accumulate over k
        for k in range(0, H):
            # x[b, s_offsets, k]
            x_ptrs = x_ptr + b * stride_xb + s_offsets * stride_xs + k * stride_xh
            x_vals = tl.load(x_ptrs, mask=mask_s, other=0.0)
            # W[h, k]
            w = tl.load(W + h * stride_Wn + k * stride_Wk)
            acc += tl.sum(x_vals * w, axis=0)

        out_val = acc + bias  # scalar

        # store to respective output
        if h_out_group == 0:
            out_ptr = B_out_ptr
            stride_ob = stride_Bob
            stride_os = stride_Bos
            stride_oh = stride_Boh
        elif h_out_group == 1:
            out_ptr = C_out_ptr
            stride_ob = stride_Cob
            stride_os = stride_Cos
            stride_oh = stride_Coh
        else:
            out_ptr = X_out_ptr
            stride_ob = stride_Xob
            stride_os = stride_Xos
            stride_oh = stride_Xoh

        out_ptrs = out_ptr + b * stride_ob + s_offsets * stride_os + h * stride_oh
        tl.store(out_ptrs, out_val, mask=mask_s)


@triton.jit
def grouped_causal_conv1d_kernel(
    Bx_ptr,        # *f32, (B, H, S) input to conv
    W_ptr, bb_ptr,     # W: (H, H, 4), bb: (H,)
    C_out_ptr,      # *f32, (B, H, S) output
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_bx_b, stride_bx_n, stride_bx_s,
    stride_Wn, stride_Wk, stride_Wt,     # W[n, k, t] strides
    stride_bb_n,                           # bb stride along n
    stride_co_b, stride_co_n, stride_co_s,
    BLOCK_S: tl.constexpr,
):
    # Grid over (B, H, S tiles)
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_block = tl.program_id(2)

    s_start = s_block * BLOCK_S
    t_offsets = s_start + tl.arange(0, BLOCK_S)  # output positions
    mask_s = t_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # loop over k in 0..3
    for k in range(0, 4):
        in_idx = t_offsets + k - 1
        mask_in = (in_idx >= 0) & (in_idx < S) & mask_s
        bx_ptrs = Bx_ptr + b * stride_bx_b + c * stride_bx_n + in_idx * stride_bx_s
        bx_vals = tl.load(bx_ptrs, mask=mask_in, other=0.0)
        w_ptrs = W_ptr + c * stride_Wn + c * stride_Wk + k * stride_Wt
        w_val = tl.load(w_ptrs)
        acc += bx_vals * w_val

    acc += tl.load(bb_ptr + c * stride_bb_n)  # bias per channel
    co_ptrs = C_out_ptr + b * stride_co_b + c * stride_co_n + t_offsets * stride_co_s
    tl.store(co_ptrs, acc, mask=mask_s)


@triton.jit
def final_linear_gemv_bsh_kernel(
    y_ptr,         # *f32, (B, S, H) input
    wy_ptr,        # *f32, (H, H) weight for final linear
    bb_ptr,        # *f32, (H,) bias
    out_ptr,       # *f32, (B, S, H) output
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_yb, stride_ys, stride_yh,
    stride_wyn, stride_wyk,  # wy[n, k]
    stride_ob, stride_os, stride_oh,
    BLOCK_H: tl.constexpr,
):
    # Grid over (B, S, H tiles)
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_block = tl.program_id(2)

    h_start = h_block * BLOCK_H
    h_out_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_out_offsets < H

    for h_out in range(0, BLOCK_H):
        h = h_out_offsets[h_out]
        if h >= H:
            break
        acc = 0.0
        for h_in in range(0, H):
            y_val = tl.load(y_ptr + b * stride_yb + s * stride_ys + h_in * stride_yh)
            wy_val = tl.load(wy_ptr + h * stride_wyn + h_in * stride_wyk)
            acc += y_val * wy_val
        bias = tl.load(bb_ptr + h * stride_wyn)  # note: bias stride equals wy stride along n
        out_ptr_hs = out_ptr + b * stride_ob + s * stride_os + h * stride_oh
        tl.store(out_ptr_hs, acc + bias)

            BLOCK_H=64,
            num_warps=4, num_stages=2
        )

        conv_out = torch.empty((B, H, S), device=x.device, dtype=x.dtype)

            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),

            num_warps=4, num_stages=2
        )

        # 4) Output gating: y = C_out * conv_out (Triton elementwise)
        y = torch.empty((B, S, H), device=x.device, dtype=x.dtype)

        @triton.jit
        def elementwise_mul_bsh_kernel(c_ptr, co_ptr, out_ptr, B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,

                                       stride_ob, stride_os, stride_oh,
                                       BLOCK_H: tl.constexpr):
            b = tl.program_id(0)
            s = tl.program_id(1)
            h_block = tl.program_id(2)
            h_start = h_block * BLOCK_H
            h_offsets = h_start + tl.arange(0, BLOCK_H)
            mask_h = h_offsets < H
            c_ptrs = c_ptr + b * stride_cb + s * stride_cs + h_offsets * stride_ch
            co_ptrs = co_ptr + b * stride_cob + s * stride_cos + h_offsets * stride_coh
            c_vals = tl.load(c_ptrs, mask=mask_h, other=0.0)
            co_vals = tl.load(co_ptrs, mask=mask_h, other=0.0)
            out_ptrs = out_ptr + b * stride_ob + s * stride_os + h_offsets * stride_oh
            tl.store(out_ptrs, c_vals * co_vals, mask=mask_h)

        elementwise_mul_bsh_kernel[(B, S, (H + 64 - 1) // 64)](
            C_out, conv_out.transpose(1, 2).contiguous(), y,
            B, S, H,
            C_out.stride(0), C_out.stride(1), C_out.stride(2),
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            y.stride(0), y.stride(1), y.stride(2),
            BLOCK_H=64,
            num_warps=4, num_stages=2
        )

        # 5) Final linear projection to output (B,S,H)
        out_final = torch.empty((B, S, H), device=x.device, dtype=x.dtype)

            out_final.stride(0), out_final.stride(1), out_final.stride(2),
            BLOCK_H=64,
            num_warps=4, num_stages=2
        )

import torch
import triton
import triton.language as tl


@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,        # *f32, (B, S, H)
    W0_ptr, b0_ptr,     # W0: (H, H), b0: (H,)
    W1_ptr, b1_ptr,     # W1: (H, H), b1: (H,)
    W2_ptr, b2_ptr,     # W2: (H, H), b2: (H,)
    B_out_ptr, C_out_ptr, X_out_ptr,  # outputs: (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_xb, stride_xs, stride_xh,
    stride_W0n, stride_W0k, stride_b0,
    stride_W1n, stride_W1k, stride_b1,
    stride_W2n, stride_W2k, stride_b2,
    stride_Bob, stride_Bos, stride_Boh,
    stride_Cob, stride_Cos, stride_Coh,
    stride_Xob, stride_Xos, stride_Xoh,
    BLOCK_H: tl.constexpr,
):
    # Grid: (B, 3, ceil(S/BLOCK_H))
    b = tl.program_id(0)
    group = tl.program_id(1)  # 0 -> B_out, 1 -> C_out, 2 -> X_out
    s_block = tl.program_id(2)
    s_start = s_block * BLOCK_H
    s_offsets = s_start + tl.arange(0, BLOCK_H)
    mask_s = s_offsets < S

    # choose weights and biases based on group
    if group == 0:
        W = W0_ptr
        b_vec = b0_ptr
        stride_Wn, stride_Wk = stride_W0n, stride_W0k
        stride_b = stride_b0
    elif group == 1:
        W = W1_ptr
        b_vec = b1_ptr
        stride_Wn, stride_Wk = stride_W1n, stride_W1k
        stride_b = stride_b1
    else:
        W = W2_ptr
        b_vec = b2_ptr
        stride_Wn, stride_Wk = stride_W2n, stride_W2k
        stride_b = stride_b2

    # accumulate output vector for s_offsets
    out_vec = tl.zeros((BLOCK_H,), dtype=tl.float32)

    for h in range(0, H):
        bias = tl.load(b_vec + h * stride_b)
        acc = 0.0
        for k in range(0, H):
            # load x[b, s_offsets, k]
            x_ptrs = x_ptr + b * stride_xb + s_offsets * stride_xs + k * stride_xh
            x_vals = tl.load(x_ptrs, mask=mask_s, other=0.0)  # (BLOCK_H,)
            # load W[h, k]
            w = tl.load(W + h * stride_Wn + k * stride_Wk)
            acc += tl.sum(x_vals * w, axis=0)
        out_vec += acc + bias

    # store to corresponding output tensor
    if group == 0:
        out_ptr = B_out_ptr
        stride_ob, stride_os, stride_oh = stride_Bob, stride_Bos, stride_Boh
    elif group == 1:
        out_ptr = C_out_ptr
        stride_ob, stride_os, stride_oh = stride_Cob, stride_Cos, stride_Coh
    else:
        out_ptr = X_out_ptr
        stride_ob, stride_os, stride_oh = stride_Xob, stride_Xos, stride_Xoh

    out_ptrs = out_ptr + b * stride_ob + s_offsets * stride_os + tl.arange(0, H) * stride_oh
    tl.store(out_ptrs, out_vec, mask=mask_s)


@triton.jit
def grouped_causal_conv1d_kernel(
    Bx_ptr,        # *f32, (B, H, S) input to conv
    W_ptr, bb_ptr,     # W: (H, H, 4), bb: (H,)
    C_out_ptr,      # *f32, (B, H, S) output
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_bx_b, stride_bx_n, stride_bx_s,
    stride_Wn, stride_Wk, stride_Wt,     # W[n, k, t] strides
    stride_bb_n,                           # bb stride along n
    stride_co_b, stride_co_n, stride_co_s,
    BLOCK_S: tl.constexpr,
):
    # Grid over (B, H, S tiles)
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_block = tl.program_id(2)

    s_start = s_block * BLOCK_S
    t_offsets = s_start + tl.arange(0, BLOCK_S)
    mask_s = t_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # loop over taps k in 0..3
    for k in range(0, 4):
        in_idx = t_offsets + k - 1
        mask_in = (in_idx >= 0) & (in_idx < S) & mask_s
        bx_ptrs = Bx_ptr + b * stride_bx_b + c * stride_bx_n + in_idx * stride_bx_s
        bx_vals = tl.load(bx_ptrs, mask=mask_in, other=0.0)
        w_ptrs = W_ptr + c * stride_Wn + c * stride_Wk + k * stride_Wt
        w_val = tl.load(w_ptrs)
        acc += bx_vals * w_val

    acc += tl.load(bb_ptr + c * stride_bb_n)  # per-channel bias
    co_ptrs = C_out_ptr + b * stride_co_b + c * stride_co_n + t_offsets * stride_co_s
    tl.store(co_ptrs, acc, mask=mask_s)


@triton.jit
def final_linear_gemv_bsh_kernel(
    y_ptr,         # *f32, (B, S, H) input
    wy_ptr,        # *f32, (H, H) weight for final linear
    bb_ptr,        # *f32, (H,) bias
    out_ptr,       # *f32, (B, S, H) output
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_yb, stride_ys, stride_yh,
    stride_wyn, stride_wyk,  # wy[n, k]
    stride_ob, stride_os, stride_oh,
    BLOCK_H: tl.constexpr,
):
    # Grid: (B, S, ceil(H/BLOCK_H))
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_block = tl.program_id(2)
    h_start = h_block * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H

    for h_out in range(0, BLOCK_H):
        h = h_offsets[h_out]
        if h >= H:
            break
        acc = 0.0
        for h_in in range(0, H):
            y_val = tl.load(y_ptr + b * stride_yb + s * stride_ys + h_in * stride_yh)
            wy_val = tl.load(wy_ptr + h * stride_wyn + h_in * stride_wyk)
            acc += y_val * wy_val
        bias = tl.load(bb_ptr + h * stride_wyn)  # bias stride equals wy stride along n
        out_ptr_hs = out_ptr + b * stride_ob + s * stride_os + h * stride_oh
        tl.store(out_ptr_hs, acc + bias, mask=mask_h)

def elementwise_mul_bsx_kernel(
    b_ptr, x_ptr, out_ptr,
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_bb, stride_bs, stride_bh,
    stride_xs, stride_xs, stride_xh,
    stride_ob, stride_os, stride_oh,
    BLOCK_H: tl.constexpr,
):

    b = tl.program_id(0)
    s = tl.program_id(1)
    h_block = tl.program_id(2)
    h_start = h_block * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H
    b_ptrs = b_ptr + b * stride_bb + s * stride_bs + h_offsets * stride_bh
    x_ptrs = x_ptr + b * stride_xs + s * stride_xs + h_offsets * stride_xh
    b_vals = tl.load(b_ptrs, mask=mask_h, other=0.0)
    x_vals = tl.load(x_ptrs, mask=mask_h, other=0.0)
    out_ptrs = out_ptr + b * stride_ob + s * stride_os + h_offsets * stride_oh
    tl.store(out_ptrs, b_vals * x_vals, mask=mask_h)
