import torch
import triton
import triton.language as tl


# Kernel A: Triple linear for B: out[b, s, h] = sum_h x[b, s, h] * in_proj_weight[h, h] + in_proj_bias[h]
@triton.jit
def triple_linear_B_kernel(
    x_ptr,                  # *f32, shape (B, S, H)
    in_proj_weight_ptr,     # *f32, shape (H, H)  # first H rows
    in_proj_bias_ptr,       # *f32, shape (H,)
    B_ptr,                 # *f32, shape (B, S, H)
    B, S, H,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_ho, stride_w_hi,  # in_proj_weight strides
    stride_b_b, stride_b_s, stride_b_h,
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    if (b >= B) or (s >= S) or (h >= H):
        return

    acc = 0.0
    for ci in range(0, H):
        x_val = tl.load(x_ptr + b * stride_x_b + s * stride_x_s + ci * stride_x_h)
        w_val = tl.load(in_proj_weight_ptr + h * stride_w_ho + ci * stride_w_hi)
        acc += x_val * w_val
    bias_val = tl.load(in_proj_bias_ptr + h)
    acc += bias_val

    tl.store(B_ptr + b * stride_b_b + s * stride_b_s + h * stride_b_h, acc)


# Kernel B: Triple linear for x_proj: out[b, s, h] = sum_h x[b, s, h] * in_proj_weight[h+H, h] + in_proj_bias[h+H]
@triton.jit
def triple_linear_xproj_kernel(
    x_ptr,                  # *f32, shape (B, S, H)
    in_proj_weight_ptr,     # *f32, shape (H, H)  # middle H rows
    in_proj_bias_ptr,       # *f32, shape (H,)
    xproj_ptr,              # *f32, shape (B, S, H)
    B, S, H,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_ho, stride_w_hi,
    stride_xp_b, stride_xp_s, stride_xp_h,
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    if (b >= B) or (s >= S) or (h >= H):
        return

    acc = 0.0
    for ci in range(0, H):
        x_val = tl.load(x_ptr + b * stride_x_b + s * stride_x_s + ci * stride_x_h)
        w_val = tl.load(in_proj_weight_ptr + (h + H) * stride_w_ho + ci * stride_w_hi)
        acc += x_val * w_val
    bias_val = tl.load(in_proj_bias_ptr + (h + H))
    acc += bias_val

    tl.store(xproj_ptr + b * stride_xp_b + s * stride_xp_s + h * stride_xp_h, acc)


# Kernel C: Triple linear for C: out[b, s, h] = sum_h x[b, s, h] * in_proj_weight[2H + h, h] + in_proj_bias[2H + h]
@triton.jit
def triple_linear_C_kernel(
    x_ptr,                  # *f32, shape (B, S, H)
    in_proj_weight_ptr,     # *f32, shape (H, H)  # last H rows
    in_proj_bias_ptr,       # *f32, shape (H,)
    C_ptr,                 # *f32, shape (B, S, H)
    B, S, H,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_ho, stride_w_hi,
    stride_c_b, stride_c_s, stride_c_h,
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    if (b >= B) or (s >= S) or (h >= H):
        return

    acc = 0.0
    for ci in range(0, H):
        x_val = tl.load(x_ptr + b * stride_x_b + s * stride_x_s + ci * stride_x_h)
        w_val = tl.load(in_proj_weight_ptr + (h + 2 * H) * stride_w_ho + ci * stride_w_hi)
        acc += x_val * w_val
    bias_val = tl.load(in_proj_bias_ptr + (h + 2 * H))
    acc += bias_val

    tl.store(C_ptr + b * stride_c_b + s * stride_c_s + h * stride_c_h, acc)


# Kernel 1: Gating mul Bx = B * x_proj
@triton.jit
def gating_mul_kernel(
    B_ptr, x_ptr, Out_ptr,
    B, S, H,
    stride_b_b, stride_b_s, stride_b_h,
    stride_x_b, stride_x_s, stride_x_h,
    stride_out_b, stride_out_s, stride_out_h,
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    if (b >= B) or (s >= S) or (h >= H):
        return

    b_val = tl.load(B_ptr + b * stride_b_b + s * stride_b_s + h * stride_b_h)
    x_val = tl.load(x_ptr + b * stride_x_b + s * stride_x_s + h * stride_x_h)
    out_val = b_val * x_val
    tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + h * stride_out_h, out_val)


# Kernel 2: Grouped causal 1D convolution (depthwise, kernel_size=4) on Bx
# Inputs: Bx (B, S, H), conv_weight (H, H, 4), conv_bias (H,)
# Output: conv_out (B, H, S)
@triton.jit
def causal_conv_groups_kernel(
    Bx_ptr,             # *f32, shape (B, S, H)
    conv_weight_ptr,    # *f32, shape (H, H, 4)
    conv_bias_ptr,      # *f32, shape (H,)
    conv_out_ptr,       # *f32, shape (B, H, S)
    B, S, H,            # ints
    stride_bx_b, stride_bx_s, stride_bx_h,  # for (b, s, h) in Bx
    stride_w_go, stride_w_gi, stride_w_k,   # conv_weight strides
    stride_co_b, stride_co_ci, stride_co_s, # for (b, ci, s) in conv_out
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    ci = tl.program_id(1)  # output/input channel

    if (b >= B) or (ci >= H) or (S <= 0):
        return

    # y[b, ci, t] = sum_{k=0..3} w[ci, ci, k] * x[b, ci, t + k] + bias[ci]
    for t in range(0, S):
        acc = 0.0
        for k in range(0, 4):
            x_pos = t + k
            # causal mask: if x_pos >= S, skip
            if x_pos < S:
                x_val = tl.load(Bx_ptr + b * stride_bx_b + x_pos * stride_bx_s + ci * stride_bx_h)
            else:
                x_val = 0.0
            w_val = tl.load(conv_weight_ptr + ci * stride_w_go + ci * stride_w_gi + k * stride_w_k)
            acc += x_val * w_val
        bias_val = tl.load(conv_bias_ptr + ci)
        acc += bias_val
        tl.store(conv_out_ptr + b * stride_co_b + ci * stride_co_ci + t * stride_co_s, acc)


# Kernel 3: Gating with C: y = C * conv_out
# We receive conv_out as (B, S, H) for convenience; kernel indexes it as (b, t, ci) but we map to (b, ci, t) via pointer arithmetic.
@triton.jit
def gating_mul_y_kernel(
    C_ptr,        # *f32, shape (B, S, H)
    conv_out_ptr, # *f32, shape (B, S, H) — actually conv_out[b, ci, t]
    y_ptr,        # *f32, shape (B, S, H) output
    B, S, H,
    stride_c_b, stride_c_s, stride_c_h,
    stride_co_b, stride_co_s, stride_co_ci,
    stride_y_b, stride_y_s, stride_y_h,
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    if (b >= B) or (s >= S) or (h >= H):
        return

    # conv_out[b, h, s] and C[b, s, h] -> y[b, s, h] = C * conv_out
    c_val = tl.load(C_ptr + b * stride_c_b + s * stride_c_s + h * stride_c_h)
    co_val = tl.load(conv_out_ptr + b * stride_co_b + s * stride_co_s + h * stride_co_ci)
    y_val = c_val * co_val
    tl.store(y_ptr + b * stride_y_b + s * stride_y_s + h * stride_y_h, y_val)


# Kernel 4: Final linear projection y -> out: out[b, s, h] = sum_h y[b, s, h] * out_proj_weight[h, h] + out_proj_bias[h]
@triton.jit
def linear_final_kernel(
    y_ptr,                  # *f32, shape (B, S, H)
    out_proj_weight_ptr,    # *f32, shape (H, H)
    out_proj_bias_ptr,      # *f32, shape (H,)
    out_ptr,                # *f32, shape (B, S, H)
    B, S, H,
    stride_y_b, stride_y_s, stride_y_h,
    stride_w_ho, stride_w_hi,
    stride_out_b, stride_out_s, stride_out_h,
    num_warps: tl.constexpr, num_stages: tl.constexpr,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)

    if (b >= B) or (s >= S) or (h >= H):
        return

    acc = 0.0
    for ci in range(0, H):
        y_val = tl.load(y_ptr + b * stride_y_b + s * stride_y_s + ci * stride_y_h)
        w_val = tl.load(out_proj_weight_ptr + h * stride_w_ho + ci * stride_w_hi)
        acc += y_val * w_val
    bias_val = tl.load(out_proj_bias_ptr + h)
    acc += bias_val

    tl.store(out_ptr + b * stride_out_b + s * stride_out_s + h * stride_out_h, acc)
