# Produces out of shape (B, S, N). Implements: out[b, s, n] = sum_k x[b, s, k] * weight[n, k] + bias[n]
@triton.jit

    x_ptr,         # *f32, (B, S, K)
    weight_ptr,    # *f32, (N, K)
    bias_ptr,      # *f32, (N,)

    s = tl.program_id(1)
    n_start = tl.program_id(2) * BLOCK_N

    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    bias_ptr,      # *f32, (N,)
    out_ptr,       # *f32, (B, S, N)

    BLOCK_N: tl.constexpr,   # tile over N (output channels)
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):

    s = tl.program_id(1)

):
    b = tl.program_id(0)
    s = tl.program_id(1)
    n_start = tl.program_id(2) * BLOCK_N

    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask = n_offsets < N

    A_vals = tl.load(A_ptr + b * A_stride_b + s * A_stride_s + n_offsets * A_stride_n, mask=mask, other=0.0)
    B_vals = tl.load(B_ptr + b * B_stride_b + s * B_stride_s + n_offsets * B_stride_n, mask=mask, other=0.0)
    Out_vals = A_vals * B_vals
    tl.store(Out_ptr + b * Out_stride_b + s * Out_stride_s + n_offsets * Out_stride_n, Out_vals, mask=mask)


# 3) Grouped causal 1D convolution with groups=H and kernel_size=4.
# Input Bx shape (B, H, S), conv_weight (H, H, 4), conv_bias (H).
# Output conv_out (B, H, S) where:
#   conv_out[b, c, t] = sum_{k=0..3} Bx[b, c, t+k-1] * conv_weight[c, c, k] + conv_bias[c]
@triton.jit
def grouped_causal_conv1d_depthwise_kernel(
    input_ptr,         # *f32, (B, H, S)
    convW_ptr,         # *f32, (H, H, 4)
    convB_ptr,         # *f32, (H,)
    out_ptr,           # *f32, (B, H, S)
    B, S, H,
    input_stride_b, input_stride_c, input_stride_s,
    convW_stride_n, convW_stride_c, convW_stride_k,
    out_stride_b, out_stride_c, out_stride_s,
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    # Grid: (B, H, tiles of S)
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_tile = tl.program_id(2)
    s_start = s_tile * 64
    s_offsets = s_start + tl.arange(0, 64)
    mask_s = s_offsets < S

    # Accumulator for conv_out at these s_offsets
    acc = tl.zeros((64,), dtype=tl.float32)

    # For each tap k in 0..3, sum input[b, c, t+k-1] * convW[c, c, k]
    for k in range(0, 4):
        t = s_offsets + k - 1  # causal index (potentially out of [0, S))
        # Valid positions are where 0 <= t < S
        mask_t = (t >= 0) & (t < S) & mask_s
        vals = tl.load(
            input_ptr + b * input_stride_b + c * input_stride_c + t * input_stride_s,
            mask=mask_t,
            other=0.0,
        )  # (64,)
        w = tl.load(convW_ptr + c * convW_stride_n + c * convW_stride_c + k * convW_stride_k)  # scalar
        acc += vals * w

    # Add bias
    bval = tl.load(convB_ptr + c)
    acc += bval

    # Store to out[b, c, s_offsets]
    tl.store(out_ptr + b * out_stride_b + c * out_stride_c + s_offsets * out_stride_s, acc, mask=mask_s)
