import torch
import triton
import triton.language as tl

# 1) Triple linear projection: x[B, S, H] -> three outputs (B,S,H) using in_proj_weight slices.
#    in_proj_weight shape: (3*H, H). We slice to three (H,H): W0, W1, W2, and biases (H): b0, b1, b2.
#    out[b, s, h] = sum_k x[b, s, k] * weight[h, k] + bias[h]
@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,         # *f32, (B, S, H) contiguous
    W0_ptr, b0_ptr,   # *f32, (H,H), (H,)
    W1_ptr, b1_ptr,   # *f32, (H,H), (H,)
    W2_ptr, b2_ptr,   # *f32, (H,H), (H,)
    out0_ptr, out1_ptr, out2_ptr,  # *f32, (B,S,H) contiguous
    B, S, H,
    x_stride_b, x_stride_s, x_stride_h,
    out0_stride_b, out0_stride_s, out0_stride_h,
    out1_stride_b, out1_stride_s, out1_stride_h,
    out2_stride_b, out2_stride_s, out2_stride_h,
    BLOCK_S: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Grid: (B, H, tiles of S)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s_block = tl.program_id(2)

    s_offsets = s_block * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    # Accumulator for current h across S tiles
    acc0 = tl.zeros([BLOCK_S], dtype=tl.float32)
    acc1 = tl.zeros([BLOCK_S], dtype=tl.float32)
    acc2 = tl.zeros([BLOCK_S], dtype=tl.float32)

    # Loop over input K dimension in chunks of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # Load x[b, s_offsets, k_offsets] as a 2D [BLOCK_S, BLOCK_K]
        x_ptrs = x_ptr + b * x_stride_b + s_offsets[:, None] * x_stride_s + k_offsets[None, :] * x_stride_h
        x_vals = tl.load(x_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)  # [BLOCK_S, BLOCK_K]

        # Load weights for current h
        # W0[h, k], W1[h, k], W2[h, k] as 1D vectors of length BLOCK_K
        w0 = tl.load(W0_ptr + h * (H) + k_offsets, mask=mask_k, other=0.0)  # (BLOCK_K,)
        w1 = tl.load(W1_ptr + h * (H) + k_offsets, mask=mask_k, other=0.0)
        w2 = tl.load(W2_ptr + h * (H) + k_offsets, mask=mask_k, other=0.0)

        # Accumulate: acc += sum_k x_vals[:, k] * w
        # x_vals is [BLOCK_S, BLOCK_K], w is [BLOCK_K]
        # We need a reduction over K. Triton supports tl.sum over axis.
        # We can use a loop over kk in BLOCK_K to perform per-k accumulation.
        for kk in range(BLOCK_K):
            k_valid = (k_start + kk) < H
            # If k_valid, accumulate; else zero
            x_col = x_vals[:, kk]  # [BLOCK_S]
            acc0 += x_col * w0[kk]
            acc1 += x_col * w1[kk]
            acc2 += x_col * w2[kk]

    # Add biases
    b0 = tl.load(b0_ptr + h)
    b1 = tl.load(b1_ptr + h)
    b2 = tl.load(b2_ptr + h)
    acc0 += b0
    acc1 += b1
    acc2 += b2

    # Store results
    out0_ptrs = out0_ptr + b * out0_stride_b + s_offsets * out0_stride_s + h * out0_stride_h
    out1_ptrs = out1_ptr + b * out1_stride_b + s_offsets * out1_stride_s + h * out1_stride_h
    out2_ptrs = out2_ptr + b * out2_stride_b + s_offsets * out2_stride_s + h * out2_stride_h
    tl.store(out0_ptrs, acc0, mask=mask_s)
    tl.store(out1_ptrs, acc1, mask=mask_s)
    tl.store(out2_ptrs, acc2, mask=mask_s)

# 2) Element-wise gating: Bx = B * X, output (B, S, H)
@triton.jit
def elemwise_mul_bsh_kernel(
    B_ptr, X_ptr, Out_ptr,
    B, S, H,
    B_stride_b, B_stride_s, B_stride_h,
    X_stride_b, X_stride_s, X_stride_h,
    Out_stride_b, Out_stride_s, Out_stride_h,
    BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    s_block = tl.program_id(2)
    s_offsets = s_block * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S
    B_ptrs = B_ptr + b * B_stride_b + s_offsets * B_stride_s + h * B_stride_h
    X_ptrs = X_ptr + b * X_stride_b + s_offsets * X_stride_s + h * X_stride_h
    Out_ptrs = Out_ptr + b * Out_stride_b + s_offsets * Out_stride_s + h * Out_stride_h
    B_vals = tl.load(B_ptrs, mask=mask_s, other=0.0)
    X_vals = tl.load(X_ptrs, mask=mask_s, other=0.0)
    Out_vals = B_vals * X_vals
    tl.store(Out_ptrs, Out_vals, mask=mask_s)

# 3) Grouped causal 1D convolution: Bx[B, S, H] -> conv_out[B, H, S], kernel_size=4, groups=H
#    conv_out[b, c, t] = sum_{k=0..3} Bx[b, c, t + (k - 1)] * conv_weight[c, c, k] + conv_bias[c]
@triton.jit
def grouped_causal_conv1d_kernel(
    Bx_ptr,         # *f32, (B, S, H) contiguous
    convW_ptr, convB_ptr,   # *f32, (H, H, 4), (H,)
    convOut_ptr,   # *f32, (B, H, S) contiguous
    B, S, H,
    Bx_stride_b, Bx_stride_s, Bx_stride_h,
    convW_stride0, convW_stride1, convW_stride2,  # strides for (C_in, C_out, K)
    convOut_stride_b, convOut_stride_c, convOut_stride_s,
    BLOCK_S: tl.constexpr
):
    # Grid: (B, H, tiles of S)
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_block = tl.program_id(2)
    t_offsets = s_block * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_t = t_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # k = 0..3
    for k in range(4):
        t_in = t_offsets + (k - 1)
        # Mask for valid indices t_in >= 0 and < S
        mask_in = (t_in >= 0) & (t_in < S) & mask_t

        # Load Bx[b, c, t_in]
        Bx_ptrs = Bx_ptr + b * Bx_stride_b + t_in * Bx_stride_s + c * Bx_stride_h
        vals = tl.load(Bx_ptrs, mask=mask_in, other=0.0)  # [BLOCK_S]

        # Load conv weight scalar convW[c, c, k]
        w_ptr = convW_ptr + c * convW_stride0 + c * convW_stride1 + k * convW_stride2
        w = tl.load(w_ptr)
        acc += vals * w

    # Add bias convB[c]
    b_ptr = convB_ptr + c
    bias = tl.load(b_ptr)
    acc += bias

    # Store conv_out[b, c, t]
    convOut_ptrs = convOut_ptr + b * convOut_stride_b + c * convOut_stride_c + t_offsets * convOut_stride_s
    tl.store(convOut_ptrs, acc, mask=mask_t)

# 4) Output gating: y = C * conv_out, (B, H, S)
@triton.jit
def gate_mul_bhs_kernel(
    C_ptr, convOut_ptr, y_ptr,
    B, S, H,
    C_stride_b, C_stride_s, C_stride_h,
    convOut_stride_b, convOut_stride_c, convOut_stride_s,
    y_stride_b, y_stride_s, y_stride_h,
    BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_block = tl.program_id(2)
    t_offsets = s_block * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_t = t_offsets < S

    C_ptrs = C_ptr + b * C_stride_b + t_offsets * C_stride_s + c * C_stride_h
    convOut_ptrs = convOut_ptr + b * convOut_stride_b + c * convOut_stride_c + t_offsets * convOut_stride_s
    y_ptrs = y_ptr + b * y_stride_b + t_offsets * y_stride_s + c * y_stride_h

    C_vals = tl.load(C_ptrs, mask=mask_t, other=0.0)
    convOut_vals = tl.load(convOut_ptrs, mask=mask_t, other=0.0)
    y_vals = C_vals * convOut_vals
    tl.store(y_ptrs, y_vals, mask=mask_t)

# 5) Final linear projection: y[B, S, H] -> output[B, S, H] using out_proj_weight (H, H), bias (H)
@triton.jit
def final_linear_bsh_kernel(
    y_ptr, outW_ptr, outB_ptr, out_ptr,
    B, S, H,
    y_stride_b, y_stride_s, y_stride_h,
    outW_stride0, outW_stride1,  # strides for (H, H): (row, col)
    out_stride_b, out_stride_s, out_stride_h,
    BLOCK_S: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Grid: (B, H, tiles of S)
    b = tl.program_id(0)
    h = tl.program_id(1)
    s_block = tl.program_id(2)
    s_offsets = s_block * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # Loop over K (input channels) in chunks
    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # Load y[b, s_offsets, k_offsets] as [BLOCK_S, BLOCK_K]
        y_ptrs = y_ptr + b * y_stride_b + s_offsets[:, None] * y_stride_s + k_offsets[None, :] * y_stride_h
        y_vals = tl.load(y_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)

        # Load outW[h, k] as vector [BLOCK_K]
        outW_ptrs = outW_ptr + h * outW_stride0 + k_offsets * outW_stride1
        outW_vec = tl.load(outW_ptrs, mask=mask_k, other=0.0)

        # Accumulate: for each kk, acc += sum_s y_vals[:, kk] * outW_vec[kk]
        for kk in range(BLOCK_K):
            k_valid = (k_start + kk) < H
            y_col = y_vals[:, kk]  # [BLOCK_S]
            acc += y_col * outW_vec[kk]

    # Add bias
    bias = tl.load(outB_ptr + h)
    acc += bias

    # Store output
    out_ptrs = out_ptr + b * out_stride_b + s_offsets * out_stride_s + h * out_stride_h
    tl.store(out_ptrs, acc, mask=mask_s)

# -----------------------
# ModelNew.forward
# -----------------------
class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor,
                in_proj_weight: torch.Tensor, in_proj_bias: torch.Tensor,
                conv_weight: torch.Tensor, conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor, out_proj_bias: torch.Tensor):
        """
        Triton-optimized forward that mirrors the original computation:
        - Triple linear projection: x -> (B, S, H) for B, C, x_proj
        - Element-wise gating: Bx = B * x_proj
        - Grouped causal 1D convolution: Bx -> conv_out (B,H,S), kernel_size=4, groups=H
        - Output gating: y = C * conv_out
        - Final linear projection: y -> output (B,S,H)
        """
        B, S, H = x.shape

        # Ensure all inputs are contiguous and float32 for Triton
        device = x.device
        x_c = x.contiguous().to(torch.float32)

        # Prepare slices for in_proj_weight
        W0 = in_proj_weight[:H, :].contiguous().to(torch.float32)  # (H, H)
        b0 = in_proj_bias[:H].contiguous().to(torch.float32)       # (H,)
        W1 = in_proj_weight[H:2 * H, :].contiguous().to(torch.float32)  # (H, H)
        b1 = in_proj_bias[H:2 * H].contiguous().to(torch.float32)       # (H,)
        W2 = in_proj_weight[2 * H:3 * H, :].contiguous().to(torch.float32)  # (H, H)
        b2 = in_proj_bias[2 * H:3 * H].contiguous().to(torch.float32)       # (H,)

        # Outputs for triple linear (B, S, H)
        B_out = torch.empty((B, S, H), device=device, dtype=torch.float32)
        C_out = torch.empty((B, S, H), device=device, dtype=torch.float32)
        X_out = torch.empty((B, S, H), device=device, dtype=torch.float32)

        # Launch triple linear kernel
        grid1 = (B, H, triton.cdiv(S, 128))
        triple_linear_bsh_kernel[grid1](
            x_c, W0, b0, W1, b1, W2, b2, B_out, C_out, X_out,
            B, S, H,
            x_c.stride(0), x_c.stride(1), x_c.stride(2),
            B_out.stride(0), B_out.stride(1), B_out.stride(2),
            C_out.stride(0), C_out.stride(1), C_out.stride(2),
            X_out.stride(0), X_out.stride(1), X_out.stride(2),
            BLOCK_S=128, BLOCK_K=64,
            num_warps=4, num_stages=2
        )

        # 2) Element-wise gating: Bx = B * X
        Bx = torch.empty((B, S, H), device=device, dtype=torch.float32)
        grid2 = (B, H, triton.cdiv(S, 128))
        elemwise_mul_bsh_kernel[grid2](
            B_out, X_out, Bx,
            B, S, H,
            B_out.stride(0), B_out.stride(1), B_out.stride(2),
            X_out.stride(0), X_out.stride(1), X_out.stride(2),
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            BLOCK_S=128,
            num_warps=4, num_stages=2
        )

        # 3) Grouped causal 1D convolution: conv_out (B,H,S)
        convW = conv_weight.contiguous().to(torch.float32)   # (H, H, 4)
        convB = conv_bias.contiguous().to(torch.float32)     # (H,)
        conv_out = torch.empty((B, H, S), device=device, dtype=torch.float32)
        grid3 = (B, H, triton.cdiv(S, 128))
        grouped_causal_conv1d_kernel[grid3](
            Bx, convW, convB, conv_out,
            B, S, H,
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            convW.stride(0), convW.stride(1), convW.stride(2),
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            BLOCK_S=128,
            num_warps=4, num_stages=2
        )

        # 4) Output gating: y = C * conv_out -> (B,H,S)
        y = torch.empty((B, H, S), device=device, dtype=torch.float32)
        grid4 = (B, H, triton.cdiv(S, 128))
        gate_mul_bhs_kernel[grid4](
            C_out, conv_out, y,
            B, S, H,
            C_out.stride(0), C_out.stride(1), C_out.stride(2),
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            y.stride(0), y.stride(1), y.stride(2),
            BLOCK_S=128,
            num_warps=4, num_stages=2
        )

        # 5) Final linear projection to output (B,S,H)
        out = torch.empty((B, S, H), device=device, dtype=torch.float32)
        # Prepare out_proj parameters
        outW = out_proj_weight.contiguous().to(torch.float32)  # (H,H)
        outB = out_proj_bias.contiguous().to(torch.float32)    # (H,)
        grid5 = (B, H, triton.cdiv(S, 128))
        final_linear_bsh_kernel[grid5](
            y, outW, outB, out,
            B, S, H,
            y.stride(0), y.stride(1), y.stride(2),
            outW.stride(0), outW.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            BLOCK_S=128, BLOCK_K=64,
            num_warps=4, num_stages=2
        )

        return out
