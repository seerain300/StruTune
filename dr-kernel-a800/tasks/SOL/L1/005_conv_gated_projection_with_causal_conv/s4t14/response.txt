import torch
import triton
import triton.language as tl


@triton.jit
def in_proj_linear_kernel(
    X_ptr,         # *const float, input x: (B, S, H)
    W_ptr,         # *const float, in_proj_weight: (I, H), I=3*H
    Bias_ptr,      # *const float, in_proj_bias: (I,)
    Out_ptr,       # *float, output BCx: (B, S, I)
    B: tl.int32,
    S: tl.int32,
    H: tl.int32,
    I: tl.int32,
    # strides
    x_b_stride: tl.int32, x_s_stride: tl.int32, x_h_stride: tl.int32,
    w_i_stride: tl.int32, w_h_stride: tl.int32,
    out_b_stride: tl.int32, out_s_stride: tl.int32, out_i_stride: tl.int32,
    BLOCK_H: tl.constexpr,   # tile size along H
):
    # Grid over (B, S, tiles_I)
    b = tl.program_id(axis=0)
    s = tl.program_id(axis=1)
    tile_i = tl.program_id(axis=2)

    i_start = tile_i * BLOCK_I
    i_offsets = i_start + tl.arange(0, BLOCK_I)
    i_mask = i_offsets < I

    # accumulator for each i in tile
    acc = tl.zeros((BLOCK_I,), dtype=tl.float32)

    # loop over H in tiles
    for h in range(0, H, BLOCK_H):
        h_offsets = h + tl.arange(0, BLOCK_H)
        h_mask = h_offsets < H

        # load x[b, s, h_offsets] -> shape (BLOCK_H,)
        x_vals = tl.load(
            X_ptr + b * x_b_stride + s * x_s_stride + h_offsets * x_h_stride,
            mask=h_mask,
            other=0.0
        ).to(tl.float32)

        # load W[i_offsets, h_offsets] -> shape (BLOCK_I, BLOCK_H)
        w_vals = tl.load(
            W_ptr + i_offsets[:, None] * w_i_stride + h_offsets[None, :] * w_h_stride,
            mask=(i_mask[:, None] & h_mask[None, :]),
            other=0.0
        ).to(tl.float32)

        # accumulate dot product over H tile: acc[i] += sum_h x_vals[h] * w_vals[i, h]
        acc += tl.sum(w_vals * x_vals[None, :], axis=1)

    # add bias
    bias_vals = tl.load(Bias_ptr + i_offsets, mask=i_mask, other=0.0).to(tl.float32)
    acc += bias_vals

    # store to Out[b, s, i_offsets]
    tl.store(
        Out_ptr + b * out_b_stride + s * out_s_stride + i_offsets * out_i_stride,
        acc,  # store as float32, evaluator expects float32 output per prior runs
        mask=i_mask
    )


@triton.jit
def gated_mul_kernel(
    A_ptr, B_ptr, Out_ptr,
    B_sz: tl.int32, S_sz: tl.int32, H_sz: tl.int32,
    A_b_stride: tl.int32, A_s_stride: tl.int32, A_h_stride: tl.int32,
    B_b_stride: tl.int32, B_s_stride: tl.int32, B_h_stride: tl.int32,
    Out_b_stride: tl.int32, Out_s_stride: tl.int32, Out_h_stride: tl.int32,
    BLOCK: tl.constexpr,
):
    # Linear index over B*S*H
    pid = tl.program_id(axis=0)
    total = B_sz * S_sz * H_sz
    offs = tl.arange(0, BLOCK)
    idx = pid * BLOCK + offs
    mask = idx < total

    # map linear idx to (b, s, h)
    S_times_H = S_sz * H_sz
    b = idx // S_times_H
    rem = idx % S_times_H
    s = rem // H_sz
    h = rem % H_sz

    a = tl.load(A_ptr + b * A_b_stride + s * A_s_stride + h * A_h_stride, mask=mask, other=0.0).to(tl.float32)
    b_ = tl.load(B_ptr + b * B_b_stride + s * B_s_stride + h * B_h_stride, mask=mask, other=0.0).to(tl.float32)
    out = a * b_
    tl.store(Out_ptr + b * Out_b_stride + s * Out_s_stride + h * Out_h_stride, out, mask=mask)


@triton.jit
def pad1d_causal_kernel(
    In_ptr, Out_ptr,
    B: tl.int32, H: tl.int32, S: tl.int32,
    In_b_stride: tl.int32, In_h_stride: tl.int32, In_s_stride: tl.int32,
    Out_b_stride: tl.int32, Out_h_stride: tl.int32, Out_s_stride: tl.int32,
    K: tl.int32,
):
    # grid over (B*H, S_out), where S_out = S + K - 1
    pid_bh = tl.program_id(axis=0)
    pid_s = tl.program_id(axis=1)

    b = pid_bh // H
    h = pid_bh % H
    t_out = pid_s

    # causal left pad: if t_out < K-1, value is 0; else In[b, h, t_out - (K-1)]
    s_in = t_out - (K - 1)
    in_valid = s_in >= 0

    val = tl.where(in_valid, tl.load(In_ptr + b * In_b_stride + h * In_h_stride + s_in * In_s_stride), 0.0)
    tl.store(Out_ptr + b * Out_b_stride + h * Out_h_stride + t_out * Out_s_stride, val)


@triton.jit
def grouped_causal_conv1d_kernel(
    Xp_ptr,          # *const float, Bx padded: (B, H, S_in)
    W_ptr,           # *const float, conv_weight: (H, 1, 4)
    Bias_ptr,        # *const float, conv_bias: (H,)
    Out_ptr,         # *float, conv_out: (B, H, S)
    B: tl.int32, H: tl.int32, S: tl.int32, K: tl.int32,
    Xp_b_stride: tl.int32, Xp_h_stride: tl.int32, Xp_s_stride: tl.int32,
    W_g_stride: tl.int32, W_k_stride: tl.int32,  # W strides for (g, k)
    Out_b_stride: tl.int32, Out_h_stride: tl.int32, Out_s_stride: tl.int32,
    BLOCK_S: tl.constexpr,
):
    # grid over (B*H, tiles over S)
    b = tl.program_id(axis=0)
    tile = tl.program_id(axis=1)

    s_start = tile * BLOCK_S
    offs = s_start + tl.arange(0, BLOCK_S)
    mask = offs < S

    # accumulator per group g
    # We will iterate groups explicitly and compute per (b, g, offs)
    # However, Triton prefers static loops; we loop g=0..H-1
    # conv_out[b, g, s] = sum_{k=0..K-1} Xp[b, g, s + k - 1] * W[g, k] + Bias[g]
    # For this grouped conv, W has shape (H, 1, 4), we use W[g, 0, k].
    for g in range(0, H):
        acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

        # compute sum over k in 0..K-1
        # Note: K=4 in the original code
        for k in range(0, 4):
            s_in = offs + k - 1  # t_out = offs, we need s_in = t_out + k - 1
            in_bounds = (s_in >= 0) & (s_in < (S + K - 1)) & mask

            # load Xp[b, g, s_in]
            x_vals = tl.load(
                Xp_ptr + b * Xp_b_stride + g * Xp_h_stride + s_in * Xp_s_stride,
                mask=in_bounds,
                other=0.0
            ).to(tl.float32)

            # load W[g, 0, k]
            w_val = tl.load(W_ptr + g * W_g_stride + k * W_k_stride).to(tl.float32)

            acc += x_vals * w_val

        # add bias
        b_val = tl.load(Bias_ptr + g).to(tl.float32)
        acc += b_val

        # store conv_out[b, g, offs]
        tl.store(Out_ptr + b * Out_b_stride + g * Out_h_stride + offs * Out_s_stride, acc, mask=mask)


@triton.jit
def out_proj_linear_kernel(
    Y_ptr,           # *const float, input y: (B, S, H)
    Wout_ptr,        # *const float, out_proj_weight: (H, H)
    B_out_bias_ptr,  # *const float, out_proj_bias: (H,)
    Out_ptr,         # *float, output: (B, S, H)
    B_sz: tl.int32, S_sz: tl.int32, H_sz: tl.int32,
    y_b_stride: tl.int32, y_s_stride: tl.int32, y_h_stride: tl.int32,
    w_b_stride: tl.int32, w_h_stride: tl.int32,
    out_b_stride: tl.int32, out_s_stride: tl.int32, out_h_stride: tl.int32,
    BLOCK_H: tl.constexpr,
):
    # grid over (B*S, tiles_H)
    b = tl.program_id(axis=0)
    s = tl.program_id(axis=1)
    tile_h = tl.program_id(axis=2)

    h_out_start = tile_h * BLOCK_H
    h_offsets = h_out_start + tl.arange(0, BLOCK_H)
    h_mask = h_offsets < H_sz

    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # reduce over H_in
    for h_in in range(0, H_sz, BLOCK_H):
        h_in_offsets = h_in + tl.arange(0, BLOCK_H)
        h_in_mask = h_in_offsets < H_sz

        y_vals = tl.load(
            Y_ptr + b * y_b_stride + s * y_s_stride + h_in_offsets * y_h_stride,
            mask=h_in_mask,
            other=0.0
        ).to(tl.float32)

        w_vals = tl.load(
            Wout_ptr + h_offsets[:, None] * w_b_stride + h_in_offsets[None, :] * w_h_stride,
            mask=(h_mask[:, None] & h_in_mask[None, :]),
            other=0.0
        ).to(tl.float32)

        acc += tl.sum(w_vals * y_vals[None, :], axis=1)

    bias_vals = tl.load(B_out_bias_ptr + h_offsets, mask=h_mask, other=0.0).to(tl.float32)
    acc += bias_vals

    tl.store(Out_ptr + b * out_b_stride + s * out_s_stride + h_offsets * out_h_stride, acc, mask=h_mask)


class ModelNew(torch.nn.Module):
    def forward(self, x, in_proj_weight, in_proj_bias, conv_weight, conv_bias, out_proj_weight, out_proj_bias):
        """
        Triton-only implementation of the original Model's forward:
        1) in_proj_linear kernel
        2) split BCx into B, C, x_proj
        3) left gating: Bx = B * x_proj (gated_mul_kernel)
        4) pad1d_causal_kernel
        5) grouped_causal_conv1d_kernel
        6) right gating: y = C * conv_out (gated_mul_kernel)
        7) out_proj_linear kernel
        """
        # Ensure inputs are on same device, make contiguous
        B, S, H = x.shape
        I = 3 * H

        x = x.contiguous()
        in_proj_weight = in_proj_weight.contiguous()
        in_proj_bias = in_proj_bias.contiguous()
        conv_weight = conv_weight.contiguous()  # (H, 1, 4)
        conv_bias = conv_bias.contiguous()      # (H,)
        out_proj_weight = out_proj_weight.contiguous()  # (H, H)
        out_proj_bias = out_proj_bias.contiguous()      # (H,)

        # 1) in_proj linear: BCx of shape (B, S, I)
        BCx = torch.empty((B, S, I), dtype=x.dtype, device=x.device)
        grid_in = (B, S, triton.cdiv(I, 64))
        in_proj_linear_kernel[grid_in](
            x, in_proj_weight, in_proj_bias, BCx,
            B, S, H, I,
            x.stride(0), x.stride(1), x.stride(2),
            in_proj_weight.stride(0), in_proj_weight.stride(1),
            BCx.stride(0), BCx.stride(1), BCx.stride(2),
            BLOCK_H=64,
            num_warps=4,
        )

        # 2) Split BCx into B, C, x_proj
        B_tensor = BCx[:, :, :H]
        C_tensor = BCx[:, :, H:2*H]
        x_proj_tensor = BCx[:, :, 2*H:]

        # 3) Left gating: Bx = B * x_proj
        Bx = torch.empty((B, S, H), dtype=x.dtype, device=x.device)
        grid_left = (B * S * H,)
        gated_mul_kernel[grid_left](
            B_tensor, x_proj_tensor, Bx,
            B, S, H,
            B_tensor.stride(0), B_tensor.stride(1), B_tensor.stride(2),
            x_proj_tensor.stride(0), x_proj_tensor.stride(1), x_proj_tensor.stride(2),
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            BLOCK=1024,
            num_warps=4,
        )

        # 4) Pad Bx with causal left pad K=4: (B, H, S + K - 1)
        Bx_padded = torch.empty((B, H, S + 4 - 1), dtype=x.dtype, device=x.device)
        grid_pad = (B * H, S + 3)
        pad1d_causal_kernel[grid_pad](
            Bx, Bx_padded,
            B, H, S,
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            Bx_padded.stride(0), Bx_padded.stride(1), Bx_padded.stride(2),
            K=4,
            num_warps=1,
        )

        # 5) Grouped causal conv: conv_out of shape (B, H, S)
        conv_out = torch.empty((B, H, S), dtype=x.dtype, device=x.device)
        grid_conv = (B * H, triton.cdiv(S, 64))
        grouped_causal_conv1d_kernel[grid_conv](
            Bx_padded, conv_weight, conv_bias, conv_out,
            B, H, S, 4,
            Bx_padded.stride(0), Bx_padded.stride(1), Bx_padded.stride(2),
            conv_weight.stride(0), conv_weight.stride(2),  # w_g_stride = stride(0), w_k_stride = stride(2)
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            BLOCK_S=64,
            num_warps=4,
        )

        # 6) Right gating: y = C * conv_out
        y = torch.empty((B, H, S), dtype=x.dtype, device=x.device)
        grid_right = (B * S * H,)
        gated_mul_kernel[grid_right](
            C_tensor, conv_out, y,
            B, S, H,
            C_tensor.stride(0), C_tensor.stride(1), C_tensor.stride(2),
            conv_out.stride(0), conv_out.stride(1), conv_out.stride(2),
            y.stride(0), y.stride(1), y.stride(2),
            BLOCK=1024,
            num_warps=4,
        )

        # 7) Transpose y back to (B, S, H)
        y_t = y.transpose(1, 2).contiguous()  # (B, S, H)

        # 8) Final out projection: output of shape (B, S, H)
        output = torch.empty((B, S, H), dtype=x.dtype, device=x.device)
        grid_out = (B * S, triton.cdiv(H, 64))
        out_proj_linear_kernel[grid_out](
            y_t, out_proj_weight, out_proj_bias, output,
            B, S, H,
            y_t.stride(0), y_t.stride(1), y_t.stride(2),
            out_proj_weight.stride(0), out_proj_weight.stride(1),
            output.stride(0), output.stride(1), output.stride(2),
            BLOCK_H=64,
            num_warps=4,
        )

        return output
