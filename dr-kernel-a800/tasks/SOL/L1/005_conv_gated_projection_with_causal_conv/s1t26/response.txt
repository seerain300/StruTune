# X: (B, S, H) contiguous flattened; W: (H, H) contiguous; bias: (H)
# Y: (B, S, H) contiguous flattened
@triton.jit
def TritonLinearProjKernel(
    X_ptr, W_ptr, bias_ptr, Y_ptr,
    Bsz, S, H,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    total = Bsz * S * H
    mask = offsets < total

    # Decode offsets into (b, s, h)
    b = offsets // (S * H)
    rem = offsets % (S * H)
    s = rem // H
    h = rem % H

    # Load X[b, s, h]
    X_ptrs = X_ptr + b * (S * H) + s * H + h
    x_vals = tl.load(X_ptrs, mask=mask, other=0.0)

    # Load W[h, h] -> we need a vector of weights indexed by h for the output dimension
    # Note: we load W[h, h] for each output h; inputs are X[b, s, i] where i iterates over H
    # To form the product, we need to iterate over H explicitly per block.
    # We'll compute Y per (b, s, h) using a loop over H:
    # Y[b, s, h] = sum_i x[b, s, i] * W[h, i] + bias[h]

    h_vec = h
    W_ptrs = W_ptr + h_vec * H + tl.arange(0, H)  # shape (H,)
    bias_ptrs = bias_ptr + h_vec                 # shape (1,)

    for i in range(0, H):

def TritonFinalProjKernel(
    Y_in_ptr, out_proj_weight_ptr, out_proj_bias_ptr, OUT_ptr,
    Bsz, S, H,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    total = Bsz * S * H
    mask = offsets < total

    b = offsets // (S * H)
    rem = offsets % (S * H)
    s = rem // H
    h = rem % H

    # Load Y_in[b, s, h]
    Y_in_ptrs = Y_in_ptr + b * (S * H) + s * H + h
    y_vals = tl.load(Y_in_ptrs, mask=mask, other=0.0)

    # Load out_proj_weight[h, h]
    w_ptrs = out_proj_weight_ptr + h * H + tl.arange(0, H)  # shape (H,)
    w_vals = tl.load(w_ptrs, mask=mask, other=0.0)

    # Dot product over H: output vector for this (b, s)
    acc = tl.zeros([BLOCK], dtype=tl.float32)
    for i in range(0, H):
        acc += y_vals[i] * w_vals[i]

    # Add bias
    bias_val = tl.load(out_proj_bias_ptr + h)
    acc += bias_val

    OUT_ptrs = OUT_ptr + b * (S * H) + s * H + h
    tl.store(OUT_ptrs, acc, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor,
                in_proj_weight: torch.Tensor, in_proj_bias: torch.Tensor,
                conv_weight: torch.Tensor, conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor, out_proj_bias: torch.Tensor):
        """
        x: (B, S, H)
        in_proj_weight: (3H, H), in_proj_bias: (3H)
        conv_weight: (H, 1, 4), conv_bias: (H)
        out_proj_weight: (H, H), out_proj_bias: (H)
        Returns: (B, S, H)
        """
        B, S, H = x.shape
        device = x.device

        # Ensure contiguous and float32 compute
        x_contig = x.contiguous().to(torch.float32)

        # 1) Three linear projections using Triton
        # Helper to launch linear projection kernel
        def _triton_linear(b, s, h, W, bvec, Y, BLOCK=1024):
            grid = (triton.cdiv(b * s * h, BLOCK),)
            TritonLinearProjKernel[grid](
                x_contig, W, bvec, Y,
                b, s, h,
                BLOCK=BLOCK,
            )

        # First projection: B = linear(x, in_proj_weight[:H, :], in_proj_bias[:H])
        W1 = in_proj_weight[:H, :].contiguous().to(torch.float32)
        b1 = in_proj_bias[:H].contiguous().to(torch.float32)
        B = torch.empty((B, S, H), dtype=torch.float32, device=device)
        _triton_linear(B, S, H, W1, b1, B, BLOCK=1024)

        # Second projection: C = linear(x, in_proj_weight[H:2H, :], in_proj_bias[H:2H])
        W2 = in_proj_weight[H:2 * H, :].contiguous().to(torch.float32)
        b2 = in_proj_bias[H:2 * H].contiguous().to(torch.float32)
        C = torch.empty((B, S, H), dtype=torch.float32, device=device)
        _triton_linear(B, S, H, W2, b2, C, BLOCK=1024)

        # Third projection: X_proj = linear(x, in_proj_weight[2H:3H, :], in_proj_bias[2H:3H])
        W3 = in_proj_weight[2 * H:3 * H, :].contiguous().to(torch.float32)
        b3 = in_proj_bias[2 * H:3 * H].contiguous().to(torch.float32)
        X_proj = torch.empty((B, S, H), dtype=torch.float32, device=device)
        _triton_linear(B, S, H, W3, b3, X_proj, BLOCK=1024)

        # 2) Element-wise gating Bx = B * X_proj
        def _triton_gate(b, s, h, B_ptr, X_ptr, OUT_ptr, BLOCK=1024):
            grid = (triton.cdiv(b * s * h, BLOCK),)
            TritonGateKernel[grid](
                B_ptr, X_ptr, OUT_ptr,
                b, s, h,
                BLOCK=BLOCK,
            )

        Bx = torch.empty((B, S, H), dtype=torch.float32, device=device)
        _triton_gate(B, S, H, B, X_proj, Bx, BLOCK=1024)

        # 3) Pad Bx by 3 for causal conv
        def _triton_pad_left(Bsz, S, H, PAD, Bx_ptr, OUT_ptr, BLOCK=1024):
            grid = (triton.cdiv(Bsz * H * (S + PAD), BLOCK),)
            TritonPadLeftKernel[grid](
                Bx_ptr, OUT_ptr,
                Bsz, S, H, PAD,
                BLOCK=BLOCK,
            )

        Bx_pad = torch.empty((B, H, S + 3), dtype=torch.float32, device=device)
        _triton_pad_left(B, S, H, 3, Bx, Bx_pad, BLOCK=1024)

        # 4) Grouped causal 1D convolution with groups=H, kernel_size=4
        def _triton_causal_conv(Bsz, S, H, PAD, KW, Bx_pad_ptr, conv_w_ptr, conv_b_ptr, OUT_ptr, BLOCK=1024):
            grid = (triton.cdiv(Bsz * H * S, BLOCK),)
            TritonCausalConvKernel[grid](
                Bx_pad_ptr, conv_w_ptr, conv_b_ptr, OUT_ptr,
                Bsz, S, H, PAD, KW=KW,
                BLOCK=BLOCK,
            )

        conv_weight_flat = conv_weight.contiguous().view(H, 4).to(torch.float32)  # (H, 4)
        conv_bias_t = conv_bias.contiguous().to(torch.float32)
        conv_out = torch.empty((B, H, S), dtype=torch.float32, device=device)
        _triton_causal_conv(B, S, H, 3, 4, Bx_pad, conv_weight_flat, conv_bias_t, conv_out, BLOCK=1024)

        # 5) Output gating: y = C * conv_out
        # C: (B, S, H), conv_out: (B, H, S). Multiply elementwise across (B, S, H) for each h and s.
        # We can compute y[b, s, h] = C[b, s, h] * conv_out[b, h, s].
        # Use PyTorch for this elementwise op to ensure correctness; it’s a small tensor.
        y = C * conv_out.transpose(1, 2)  # (B, S, H)
