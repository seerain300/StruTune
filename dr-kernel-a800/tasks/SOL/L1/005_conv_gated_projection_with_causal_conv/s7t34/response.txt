# 1) In-projection: compute BCx of shape (B, 3H, L) from x (B, L, H), weight (3H, H, L)
@triton.jit
def in_proj_kernel_B(
    x_ptr,               # *float32, input x (B, L, H)
    w_ptr,               # *float32, weight (3H, H, L)
    bias_ptr,            # *float32, bias (3H)
    BCx_ptr,             # *float32, output BCx (B, 3H, L)
    B, L, H,             # sizes
    stride_x_b, stride_x_l, stride_x_h,         # strides for x (B, L, H)
    stride_w_j, stride_w_h, stride_w_l,         # strides for w (3H, H, L)
    stride_BCx_b, stride_BCx_j, stride_BCx_l    # strides for BCx (B, 3H, L)
):
    # Grid: (J_tiles, L_tiles, B), where J = 3H
    j_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    j_offsets = j_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_j = j_offsets < (3 * H)
    mask_l = l_offsets < L
    mask = mask_j[:, None] & mask_l[None, :]

    # Accumulator per (j, l)
    acc = tl.zeros((64, 128), dtype=tl.float32)

    # Compute in-projection: for each j (0..3H-1)
    # w[j, h, l] * x[b, l, h] summed over h
    # We'll loop over h in chunks of 64 for efficiency
    for ho in range(0, H):
        # x[b, l, ho]
        x_ptrs = x_ptr + b * stride_x_b + l_offsets[None, :] * stride_x_l + ho * stride_x_h  # (1,128)
        x_vals = tl.load(x_ptrs, mask=mask_l[None, :], other=0.0)  # (1,128), broadcast along j

        # w[j, ho, l]
        w_ptrs = w_ptr + j_offsets[:, None] * stride_w_j + ho * stride_w_h + l_offsets[None, :] * stride_w_l
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (64,128)

        # Accumulate
        acc += w_vals * x_vals  # broadcast x_vals along j dimension

    # Add bias
    bias_vals = tl.load(bias_ptr + j_offsets, mask=mask_j, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store to BCx[b, j, l]
    BCx_ptrs = BCx_ptr + b * stride_BCx_b + j_offsets[:, None] * stride_BCx_j + l_offsets[None, :] * stride_BCx_l
    tl.store(BCx_ptrs, acc, mask=mask)

@triton.jit
def chunk3_kernel(
    inp_ptr,          # *float32, input BCx (B, 3H, L)
    outB_ptr,         # *float32, output B (B, L, H)
    outC_ptr,         # *float32, output C (B, L, H)
    outX_ptr,         # *float32, output X_proj (B, L, H)
    B, L, H,          # sizes
    stride_inp_b, stride_inp_j, stride_inp_l,     # strides for inp (B, 3H, L)
    stride_out_b, stride_out_l, stride_out_h      # strides for outputs (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # Copy B from inp[b, l, 0..H-1]
    for ho in range(0, H):
        in_ptrs_B = inp_ptr + b * stride_inp_b + l_offsets[None, :] * stride_inp_l + ho * stride_inp_j
        vals_B = tl.load(in_ptrs_B, mask=mask, other=0.0)  # (64,128)
        out_ptrs_B = outB_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_B, vals_B, mask=mask)

    # Copy C from inp[b, l, H..2H-1]
    for ho in range(0, H):
        in_ptrs_C = inp_ptr + b * stride_inp_b + l_offsets[None, :] * stride_inp_l + (H + ho) * stride_inp_j
        vals_C = tl.load(in_ptrs_C, mask=mask, other=0.0)  # (64,128)
        out_ptrs_C = outC_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_C, vals_C, mask=mask)

    # Copy X_proj from inp[b, l, 2H..3H-1]
    for ho in range(0, H):
        in_ptrs_X = inp_ptr + b * stride_inp_b + l_offsets[None, :] * stride_inp_l + (2 * H + ho) * stride_inp_j
        vals_X = tl.load(in_ptrs_X, mask=mask, other=0.0)  # (64,128)

@triton.jit
def gate_mul_kernel(
    a_ptr, b_ptr, out_ptr,   # pointers
    B, L, H,                 # sizes
    stride_a_b, stride_a_l, stride_a_h,    # strides for a (B, L, H)
    stride_b_b, stride_b_l, stride_b_h,    # strides for b (B, L, H)
    stride_out_b, stride_out_l, stride_out_h   # strides for out (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    a_ptrs = a_ptr + b * stride_a_b + l_offsets[None, :] * stride_a_l + h_offsets[:, None] * stride_a_h
    b_ptrs = b_ptr + b * stride_b_b + l_offsets[None, :] * stride_b_l + h_offsets[:, None] * stride_b_h
    out_ptrs = out_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + h_offsets[:, None] * stride_out_h

    b_vals = tl.load(b_ptrs, mask=mask, other=0.0)
    out_vals = a_vals * b_vals
    tl.store(out_ptrs, out_vals, mask=mask)


# 4) Grouped causal 1D convolution:
# Input Bx: (B, H, L), conv_weight: (H, H, 4), conv_bias: (H)
# Output conv_out: (B, H, L), stride=1, padding=K-1=3, groups=H (depthwise)
@triton.jit
def conv1d_grouped_causal_kernel(
    Bx_ptr,                # *float32, input Bx (B, H, L) contiguous
    w_ptr,                 # *float32, weight (H, H, 4) contiguous
    bias_ptr,              # *float32, bias (H)
    out_ptr,               # *float32, output (B, H, L) contiguous
    B, L, H,               # sizes
    stride_Bx_b, stride_Bx_h, stride_Bx_l,   # strides for Bx (B, H, L)
    stride_w_o, stride_w_i, stride_w_k,      # strides for w (H, H, 4)
    stride_out_b, stride_out_h, stride_out_l  # strides for out (B, H, L)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # For each output channel h, sum over its input channel and kernel positions
    for ho in range(0, H):
        acc = tl.zeros((64, 128), dtype=tl.float32)

        # Sum over k in [0, 4)
        for k in range(0, 4):
            # causal indexing: input index = l_offsets - (k - 1)
            l_in = l_offsets - (k - 1)
            # Ensure indices are within [0, L)
            valid_l = (l_in >= 0) & (l_in < L)
            # Cast to int32 for pointer arithmetic
            l_in_i32 = l_in.to(tl.int32)

            # Load Bx[b, ho, l_in]
            in_ptrs = Bx_ptr + b * stride_Bx_b + ho * stride_Bx_h + l_in_i32[None, :] * stride_Bx_l
            in_vals = tl.load(in_ptrs, mask=valid_l[None, :], other=0.0)  # (64,128)

            # Load weight w[ho, ho, k]
            w_ptrs = w_ptr + ho * stride_w_o + ho * stride_w_i + k * stride_w_k
            w_val = tl.load(w_ptrs)  # scalar
            acc += w_val * in_vals

        # Add bias
        b_val = tl.load(bias_ptr + ho)  # scalar
        acc += b_val

        # Store to out[b, ho, l]
        out_ptrs = out_ptr + b * stride_out_b + ho * stride_out_h + l_offsets[None, :] * stride_out_l
        tl.store(out_ptrs, acc, mask=mask)


# 5) Out-projection: y (B, L, H) -> out (B, L, H) with weight (H, L, H)
@triton.jit
def out_proj_kernel(
    y_ptr,               # *float32, input y (B, L, H)
    w_ptr,               # *float32, weight (H, L, H)
    b_ptr,               # *float32, bias (H)
    out_ptr,             # *float32, output (B, L, H)
    B, L, H,             # sizes
    stride_y_b, stride_y_l, stride_y_h,      # strides for y (B, L, H)
    stride_w_o, stride_w_l, stride_w_h,      # strides for w (H, L, H)
    stride_out_b, stride_out_l, stride_out_h # strides for out (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # For each output channel h, compute dot over L and H of y and w
    for ho in range(0, H):
        acc = tl.zeros((64, 128), dtype=tl.float32)

        # w[ho, l, h'] dot y[b, l, h']
        # We loop over h' in chunks (64)
        for hprime in range(0, H):
            # y[b, l, hprime]
            y_ptrs = y_ptr + b * stride_y_b + l_offsets[None, :] * stride_y_l + hprime * stride_y_h
            y_vals = tl.load(y_ptrs, mask=mask, other=0.0)  # (64,128)

            # w[ho, l, hprime]
            w_ptrs = w_ptr + ho * stride_w_o + l_offsets[None, :] * stride_w_l + hprime * stride_w_h
            w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (64,128)

            acc += w_vals * y_vals

        # Add bias
        b_val = tl.load(b_ptr + ho)  # scalar
        acc += b_val

        # Store to out[b, ho, l]
        out_ptrs = out_ptr + b * stride_out_b + ho * stride_out_h + l_offsets[None, :] * stride_out_l
        tl.store(out_ptrs, acc, mask=mask)

        # x[b, l, ho]
        x_ptrs = x_ptr + b * stride_x_b + l_offsets[None, :] * stride_x_l + ho * stride_x_h  # (1,128)
        x_vals = tl.load(x_ptrs, mask=mask_l[None, :], other=0.0)  # (1,128), broadcast along j

        # w[j, ho, l]
        w_ptrs = w_ptr + j_offsets[:, None] * stride_w_j + ho * stride_w_h + l_offsets[None, :] * stride_w_l
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (64,128)

        # Accumulate
        acc += w_vals * x_vals  # broadcast x_vals along j dimension

    # Add bias
    bias_vals = tl.load(bias_ptr + j_offsets, mask=mask_j, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store to BCx[b, j, l]
    BCx_ptrs = BCx_ptr + b * stride_BCx_b + j_offsets[:, None] * stride_BCx_j + l_offsets[None, :] * stride_BCx_l
    tl.store(BCx_ptrs, acc, mask=mask)


# 2) Chunk BCx (B, 3H, L) into three tensors (B, L, H): B, C, X_proj
@triton.jit
def chunk3_kernel(
    inp_ptr,          # *float32, input BCx (B, 3H, L)
    outB_ptr,         # *float32, output B (B, L, H)
    outC_ptr,         # *float32, output C (B, L, H)
    outX_ptr,         # *float32, output X_proj (B, L, H)
    B, L, H,          # sizes
    stride_inp_b, stride_inp_j, stride_inp_l,     # strides for inp (B, 3H, L)
    stride_out_b, stride_out_l, stride_out_h      # strides for outputs (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # Copy B from inp[b, l, 0..H-1]
    for ho in range(0, H):
        in_ptrs_B = inp_ptr + b * stride_inp_b + l_offsets[None, :] * stride_inp_l + ho * stride_inp_j
        vals_B = tl.load(in_ptrs_B, mask=mask, other=0.0)  # (64,128)
        out_ptrs_B = outB_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_B, vals_B, mask=mask)

    # Copy C from inp[b, l, H..2H-1]
    for ho in range(0, H):
        in_ptrs_C = inp_ptr + b * stride_inp_b + l_offsets[None, :] * stride_inp_l + (H + ho) * stride_inp_j
        vals_C = tl.load(in_ptrs_C, mask=mask, other=0.0)  # (64,128)
        out_ptrs_C = outC_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_C, vals_C, mask=mask)

    # Copy X_proj from inp[b, l, 2H..3H-1]
    for ho in range(0, H):
        in_ptrs_X = inp_ptr + b * stride_inp_b + l_offsets[None, :] * stride_inp_l + (2 * H + ho) * stride_inp_j
        vals_X = tl.load(in_ptrs_X, mask=mask, other=0.0)  # (64,128)
        out_ptrs_X = outX_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_X, vals_X, mask=mask)


# 3) Element-wise gating: out = a * b (vectorized)
@triton.jit
def gate_mul_kernel(
    a_ptr, b_ptr, out_ptr,   # pointers
    B, L, H,                 # sizes
    stride_a_b, stride_a_l, stride_a_h,    # strides for a (B, L, H)
    stride_b_b, stride_b_l, stride_b_h,    # strides for b (B, L, H)
    stride_out_b, stride_out_l, stride_out_h   # strides for out (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    a_ptrs = a_ptr + b * stride_a_b + l_offsets[None, :] * stride_a_l + h_offsets[:, None] * stride_a_h
    b_ptrs = b_ptr + b * stride_b_b + l_offsets[None, :] * stride_b_l + h_offsets[:, None] * stride_b_h
    out_ptrs = out_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + h_offsets[:, None] * stride_out_h

    a_vals = tl.load(a_ptrs, mask=mask, other=0.0)
    b_vals = tl.load(b_ptrs, mask=mask, other=0.0)
    out_vals = a_vals * b_vals
    tl.store(out_ptrs, out_vals, mask=mask)

    stride_out_b, stride_out_h, stride_out_l  # strides for out (B, H, L
