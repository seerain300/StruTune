import torch
import triton
import triton.language as tl


# Triton kernel: linear projection Y[B, S, H] = X[B, S, H] @ W[M, H].T + bias[M]
# Shapes:
#   X: (B, S, H) contiguous
#   W: (M, H) contiguous, where M is a range in {H, H, H}
#   bias: (M)
#   Y: (B, S, H) contiguous
@triton.jit
def TritonLinearKernel(
    X_ptr, W_ptr, bias_ptr, Y_ptr,
    B, S, M, H,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_m, stride_w_h,
    BLOCK_S: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_h = tl.program_id(2)

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_s = s_offsets < S
    mask_h = h_offsets < H

    # Accumulator for output tile
    acc = tl.zeros((BLOCK_S, BLOCK_H), dtype=tl.float32)

    # Perform simple outer-product accumulation over H
    # Note: X_ptr indexing for x[b, s, h] with contiguous layout (B,S,H)
    # Y_ptr indexing for y[b, s, h] with contiguous layout (B,S,H)
    for h_idx in range(0, H):
        x_ptrs = X_ptr + pid_b * (S * H) + s_offsets * H + h_idx
        x_vals = tl.load(x_ptrs, mask=mask_s, other=0.0).to(tl.float32)  # (BLOCK_S,)
        # Load weight vector for this output h-index
        w_ptrs = W_ptr + h_idx * stride_w_m  # scalar base, iterate over M in host
        w_val = tl.load(w_ptrs).to(tl.float32)  # scalar
        # Accumulate
        acc += x_vals[:, None] * w_val

    # Add bias
    b_ptrs = bias_ptr + h_offsets  # h_offsets iterates over M
    bias_vals = tl.load(b_ptrs, mask=mask_h, other=0.0).to(tl.float32)  # (BLOCK_H,)
    acc += bias_vals[None, :]

    # Store
    y_ptrs = Y_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]
    tl.store(y_ptrs, acc, mask=mask_s[:, None] & mask_h[None, :])


# Triton kernel: element-wise gating Bx = B * x_proj over (B, S, H)
@triton.jit
def TritonGateKernel(
    B_ptr, X_ptr, OUT_ptr,
    Bsz, S, H,
    BLOCK_S: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_h = tl.program_id(2)

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_s = s_offsets < S
    mask_h = h_offsets < H

    b_ptrs = B_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]
    x_ptrs = X_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]

    b_vals = tl.load(b_ptrs, mask=mask_s[:, None] & mask_h[None, :], other=0.0)
    x_vals = tl.load(x_ptrs, mask=mask_s[:, None] & mask_h[None, :], other=0.0)

    out_vals = b_vals * x_vals

    out_ptrs = OUT_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]
    tl.store(out_ptrs, out_vals, mask=mask_s[:, None] & mask_h[None, :])


# Triton kernel: left-pad along sequence by PAD for Bx, producing Bx_pad[B, H, S + PAD]
# Inputs: Bx [B, H, S], output: out_pad [B, H, S + PAD], PAD=3
@triton.jit
def TritonPadLeftKernel(
    Bx_ptr, out_ptr,
    B, H, S, PAD,
    stride_bx_b, stride_bx_h, stride_bx_s,
    stride_ob_b, stride_ob_h, stride_ob_s,
    BLOCK_S: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_sp = tl.program_id(2)  # tiling over S_out = S + PAD

    b = pid_b
    h = pid_h
    S_out = S + PAD

    s_out_start = pid_sp * BLOCK_S

    # write zeros at the first PAD columns
    for i in range(0, PAD):
        out_ptr_pos = out_ptr + b * stride_ob_b + h * stride_ob_h + i * stride_ob_s
        tl.store(out_ptr_pos, 0.0)

    # copy from Bx[:, :, :] into out[:, :, PAD:]
    for i in range(0, BLOCK_S):
        s_in = s_out_start + i
        if s_in < S:
            val = tl.load(Bx_ptr + b * stride_bx_b + h * stride_bx_h + s_in * stride_bx_s)
            tl.store(out_ptr + b * stride_ob_b + h * stride_ob_h + (s_in + PAD) * stride_ob_s, val)


# Triton kernel: grouped causal 1D convolution with groups=H, kernel_size=4
# Input: Bx_pad [B, H, S+PAD]; conv_weight [H, 1, 4]; conv_bias [H]
# Output: conv_out [B, H, S]
@triton.jit
def TritonGroupedCausalConvKernel(
    Bx_pad_ptr, conv_weight_ptr, conv_bias_ptr, out_ptr,
    B, H, S, PAD,
    stride_bx_b, stride_bx_h, stride_bx_s,
    stride_cw_h, stride_cw_k,  # conv_weight strides for (H, 1, 4) but we treat (H, 4)
    stride_out_b, stride_out_h, stride_out_s,
    BLOCK_S: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_s = tl.program_id(2)

    b = pid_b
    h = pid_h
    S_out = S + PAD

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # kernel_size = 4, causal padding handled by PAD
    for k in range(0, 4):
        s_in = s_offsets + k
        # s_in must be in [0, S_out)
        cond = (s_in >= 0) & (s_in < S_out) & mask_s
        vals = tl.load(Bx_pad_ptr + b * stride_bx_b + h * stride_bx_h + s_in * stride_bx_s, mask=cond, other=0.0)
        w_val = tl.load(conv_weight_ptr + h * stride_cw_h + k * stride_cw_k)
        acc += vals * w_val

    # add bias
    bias_val = tl.load(conv_bias_ptr + h)
    acc += bias_val

    # store to out[b, h, s]
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + s_offsets * stride_out_s
    tl.store(out_ptrs, acc, mask=mask_s)


# Triton kernel: final linear projection Y[B, S, H] = Z[B, H, S] @ OutW[H, H].T + out_bias[H]
# Note: Z comes from elementwise C * conv_out, with shapes (B, H, S).
@triton.jit
def TritonFinalLinearKernel(
    Z_ptr, OutW_ptr, out_bias_ptr, Y_ptr,
    B, S, H,
    stride_z_b, stride_z_h, stride_z_s,
    stride_ow_h, stride_ow_h2, stride_ow_h3,  # strides for OutW: (H, H) => (H, H)
    BLOCK_S: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_h = tl.program_id(2)

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_s = s_offsets < S
    mask_h = h_offsets < H

    acc = tl.zeros((BLOCK_S, BLOCK_H), dtype=tl.float32)

    for h_idx in range(0, H):
        # Z[b, h_idx, s]
        z_ptrs = Z_ptr + pid_b * (H * S) + h_idx * stride_z_h + s_offsets * stride_z_s
        z_vals = tl.load(z_ptrs, mask=mask_s, other=0.0).to(tl.float32)  # (BLOCK_S,)
        # OutW[h_idx, :] vector
        ow_ptrs = OutW_ptr + h_idx * stride_ow_h + h_offsets * stride_ow_h2
        ow_vals = tl.load(ow_ptrs, mask=mask_h, other=0.0).to(tl.float32)  # (BLOCK_H,)
        acc += z_vals[:, None] * ow_vals[None, :]

    # add bias
    bias_vals = tl.load(out_bias_ptr + h_offsets, mask=mask_h, other=0.0).to(tl.float32)
    acc += bias_vals[None, :]

    # store Y[b, s, h]
    y_ptrs = Y_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]
    tl.store(y_ptrs, acc, mask=mask_s[:, None] & mask_h[None, :])

        super().__init__()
        # We will rely on the input tensors passed to forward; no parameters here.

    def forward(self, x: torch.Tensor,
                in_proj_weight: torch.Tensor, in_proj_bias: torch.Tensor,
                conv_weight: torch.Tensor, conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor, out_proj_bias: torch.Tensor):
        """
        x: (B, S, H)
        in_proj_weight: (3H, H), in_proj_bias: (3H)
        conv_weight: (H, 1, 4), conv_bias: (H)
        out_proj_weight: (H, H), out_proj_bias: (H)
        Returns: (B, S, H)
        """

        B, S, H = x.shape

        # 1) Three linear projections using Triton
        # Ensure contiguous inputs for Triton kernels
        x_contig = x.contiguous()

        # B = linear(x, in_proj_weight[:H, :], in_proj_bias[:H])
        W1 = in_proj_weight[:H, :].contiguous()
        b1 = in_proj_bias[:H].contiguous()
        B = torch.empty((B, S, H), dtype=torch.float32, device=x.device)
        self._triton_linear_kernel(x_contig, W1, b1, B, B, S, H, H)

        # C = linear(x, in_proj_weight[H:2H, :], in_proj_bias[H:2H])
        W2 = in_proj_weight[H:2 * H, :].contiguous()
        b2 = in_proj_bias[H:2 * H].contiguous()
        C = torch.empty((B, S, H), dtype=torch.float32, device=x.device)
        self._triton_linear_kernel(x_contig, W2, b2, C, B, S, H, H)

        # x_proj = linear(x, in_proj_weight[2H:3H, :], in_proj_bias[2H:3H])
        W3 = in_proj_weight[2 * H:3 * H, :].contiguous()
        b3 = in_proj_bias[2 * H:3 * H].contiguous()
        X_proj = torch.empty((B, S, H), dtype=torch.float32, device=x.device)
        self._triton_linear_kernel(x_contig, W3, b3, X_proj, B, S, H, H)

        # 2) Element-wise gating Bx = B * X_proj
        Bx = torch.empty((B, S, H), dtype=torch.float32, device=x.device)
        self._triton_gate_kernel(B, X_proj, Bx, B, S, H)

        # 3) Left-pad Bx by 3 along sequence for causal conv
        Bx_pad = torch.empty((B, H, S + 3), dtype=torch.float32, device=x.device)
        self._triton_pad_left_kernel(Bx, Bx_pad, B, H, S, 3)

        # 4) Grouped causal 1D convolution with groups=H, kernel_size=4
        conv_out = torch.empty((B, H, S), dtype=torch.float32, device=x.device)
        conv_weight_flat = conv_weight.contiguous()  # (H, 1, 4) -> (H, 4) in practice
        self._triton_grouped_causal_conv_kernel(Bx_pad, conv_weight_flat, conv_bias.contiguous(), conv_out, B, H, S, 3)

        # 5) Output gating: y = C * conv_out
        y = C * conv_out  # elementwise (B, S, H)

        # 6) Final projection: F.linear(y, out_proj_weight, out_proj_bias)
        # Use Triton kernel to mimic F.linear
        y = y.contiguous()
        out_proj_bias_contig = out_proj_bias.contiguous()
        output = torch.empty((B, S, H), dtype=torch.float32, device=x.device)
        self._triton_final_linear_kernel(y, out_proj_weight.contiguous(), out_proj_bias_contig, output, B, S, H)

        return output

    @staticmethod
    def _triton_linear_kernel(X, W, bias, Y, *args, BLOCK_S=64, BLOCK_H=64):
        # Host-side launcher for TritonLinearKernel
        B, S, H = X.shape
        grid = (B, triton.cdiv(S, BLOCK_S), triton.cdiv(H, BLOCK_H))
        TritonLinearKernel[grid](
            X, W, bias, Y,
            B, S, H, H,  # here M=H
            X.stride(0), X.stride(1), X.stride(2),
            W.stride(0), W.stride(1),
            BLOCK_S=BLOCK_S, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

    @staticmethod
    def _triton_gate_kernel(B, X, OUT, Bsz, S, H, BLOCK_S=64, BLOCK_H=64):
        grid = (Bsz, triton.cdiv(S, BLOCK_S), triton.cdiv(H, BLOCK_H))
        TritonGateKernel[grid](
            B, X, OUT,
            Bsz, S, H,
            BLOCK_S=BLOCK_S, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

    @staticmethod
    def _triton_pad_left_kernel(Bx, out_pad, B, H, S, PAD, BLOCK_S=64):
        grid = (B, H, triton.cdiv(S + PAD, BLOCK_S))
        TritonPadLeftKernel[grid](
            Bx, out_pad,
            B, H, S, PAD,
            Bx.stride(0), Bx.stride(1), Bx.stride(2),
            out_pad.stride(0), out_pad.stride(1), out_pad.stride(2),
            BLOCK_S=BLOCK_S,
            num_warps=4, num_stages=2,
        )

    @staticmethod
    def _triton_grouped_causal_conv_kernel(Bx_pad, conv_weight, conv_bias, out, B, H, S, PAD, BLOCK_S=128):
        # conv_weight is (H, 1, 4) treated as (H, 4) in contiguous form

            out.stride(0), out.stride(1), out.stride(2),

            BLOCK_S=BLOCK_S, BLOCK_H=BLOCK_H,

        )

    out_vals = b_vals * x_vals

    out_ptrs = OUT_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]
    tl.store(out_ptrs, out_vals, mask=mask_s[:, None] & mask_h[None, :])


# Triton kernel: left-pad along sequence by PAD for Z, producing Z_pad[B, H, S + PAD]
# Inputs: Z [B, H, S], output: Z_pad [B, H, S + PAD], PAD=3
@triton.jit
def TritonPadLeftKernel(
    Z_ptr, out_ptr,
    B, H, S, PAD,
    stride_z_b, stride_z_h, stride_z_s,
    stride_ob_b, stride_ob_h, stride_ob_s,
    BLOCK_S: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_sp = tl.program_id(2)  # tiling over S_out = S + PAD

    b = pid_b
    h = pid_h
    S_out = S + PAD

    s_out_start = pid_sp * BLOCK_S

    # write zeros at the first PAD columns
    for i in range(0, PAD):
        out_ptr_pos = out_ptr + b * stride_ob_b + h * stride_ob_h + i * stride_ob_s
        tl.store(out_ptr_pos, 0.0)

    # copy from Z[:, :, :] into out[:, :, PAD:]
    for i in range(0, BLOCK_S):
        s_in = s_out_start + i
        if s_in < S:
            val = tl.load(Z_ptr + b * stride_z_b + h * stride_z_h + s_in * stride_z_s)
            tl.store(out_ptr + b * stride_ob_b + h * stride_ob_h + (s_in + PAD) * stride_ob_s, val)


# Triton kernel: grouped causal 1D convolution with groups=H, kernel_size=4
# Input: Z_pad [B, H, S+PAD]; conv_weight [H, 1, 4]; conv_bias [H]
# Output: conv_out [B, H, S]
@triton.jit
def TritonGroupedCausalConvKernel(
    Z_pad_ptr, conv_weight_ptr, conv_bias_ptr, out_ptr,
    B, H, S, PAD,
    stride_zb_b, stride_zb_h, stride_zb_s,
    stride_cw_h, stride_cw_k,  # conv_weight strides for (H, 4)
    stride_out_b, stride_out_h, stride_out_s,
    BLOCK_S: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_s = tl.program_id(2)

    b = pid_b
    h = pid_h
    S_out = S + PAD

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # kernel_size = 4, causal padding handled by PAD
    for k in range(0, 4):
        s_in = s_offsets + k
        cond = (s_in >= 0) & (s_in < S_out) & mask_s
        vals = tl.load(Z_pad_ptr + b * stride_zb_b + h * stride_zb_h + s_in * stride_zb_s, mask=cond, other=0.0)
        w_val = tl.load(conv_weight_ptr + h * stride_cw_h + k * stride_cw_k)
        acc += vals * w_val

    # add bias
    bias_val = tl.load(conv_bias_ptr + h)
    acc += bias_val

    # store to out[b, h, s]
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + s_offsets * stride_out_s
    tl.store(out_ptrs, acc, mask=mask_s)


# Triton kernel: final linear projection Y[B, S, H] = Z[B, H, S] @ OutW[H, H].T + out_bias[H]
@triton.jit
def TritonFinalLinearKernel(
    Z_ptr, OutW_ptr, out_bias_ptr, Y_ptr,
    B, S, H,
    stride_z_b, stride_z_h, stride_z_s,
    stride_ow_h, stride_ow_h2, stride_ow_h3,  # strides for OutW: (H, H)
    BLOCK_S: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_h = tl.program_id(2)

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_s = s_offsets < S
    mask_h = h_offsets < H

    acc = tl.zeros((BLOCK_S, BLOCK_H), dtype=tl.float32)

    for h_idx in range(0, H):
        # Z[b, h_idx, s]
        z_ptrs = Z_ptr + pid_b * (H * S) + h_idx * stride_z_h + s_offsets * stride_z_s
        z_vals = tl.load(z_ptrs, mask=mask_s, other=0.0).to(tl.float32)  # (BLOCK_S,)
        # OutW[h_idx, :] vector
        ow_ptrs = OutW_ptr + h_idx * stride_ow_h + h_offsets * stride_ow_h2
        ow_vals = tl.load(ow_ptrs, mask=mask_h, other=0.0).to(tl.float32)  # (BLOCK_H,)
        acc += z_vals[:, None] * ow_vals[None, :]

    # add bias
    bias_vals = tl.load(out_bias_ptr + h_offsets, mask=mask_h, other=0.0).to(tl.float32)
    acc += bias_vals[None, :]

    # store Y[b, s, h]
    y_ptrs = Y_ptr + pid_b * (S * H) + s_offsets[:, None] * H + h_offsets[None, :]
    tl.store(y_ptrs, acc, mask=mask_s[:, None] & mask_h[None, :])
