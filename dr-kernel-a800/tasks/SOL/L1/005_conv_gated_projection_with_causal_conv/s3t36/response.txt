    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr, Nproj: tl.constexpr,
    stride_x_b, stride_x_s, stride_x_h,
    stride_w_co, stride_w_ci,
    stride_bc_b, stride_bc_s, stride_bc_co,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    co = tl.program_id(2)  # output channel index in [0, Nproj)

    if b >= B or s >= S or co >= Nproj:
        return

    # Compute dot product over H: BCx[b, s, co] = sum_h x[b, s, h] * in_proj_weight[co, h] + in_proj_bias[co]
    acc = 0.0
    for h in range(0, H):
        x_val = tl.load(x_ptr + b * stride_x_b + s * stride_x_s + h * stride_x_h)
        w_val = tl.load(in_proj_weight_ptr + co * stride_w_co + h * stride_w_ci)
        acc += x_val * w_val
    bias_val = tl.load(in_proj_bias_ptr + co)
    acc += bias_val

    tl.store(BCx_ptr + b * stride_bc_b + s * stride_bc_s + co * stride_bc_co, acc)


# Kernel 2: Elementwise multiply Bx = B * x_proj
# B: (B, S, H) float32, X: (B, S, H) float32, Out: (B, S, H) float32
@triton.jit
def elementwise_mul_kernel(
    B_ptr: tl.pointer[tl.float32],
    X_ptr: tl.pointer[tl.float32],
    Out_ptr: tl.pointer[tl.float32],
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_b_b, stride_b_s, stride_b_h,
    stride_x_b, stride_x_s, stride_x_h,
    stride_out_b, stride_out_s, stride_out_h,
):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)
    if b >= B or s >= S or h >= H:
        return
    B_val = tl.load(B_ptr + b * stride_b_b + s * stride_b_s + h * stride_b_h)
    X_val = tl.load(X_ptr + b * stride_x_b + s * stride_x_s + h * stride_x_h)
    out_val = B_val * X_val
    tl.store(Out_ptr + b * stride_out_b + s * stride_out_s + h * stride_out_h, out_val)


# Kernel 3: Grouped causal 1D convolution (depthwise, kernel_size=4) on Bx
# Bx: conceptual (b, ci, t), actual tensor shape (B, S, H) indexed as (b, s, h)
# conv_weight: (H, H, 4), conv_bias: (H,)
# conv_out: (B, H, S), float32
@triton.jit
def causal_conv_groups_kernel(
    Bx_ptr: tl.pointer[tl.float32],             # *f32, shape (B, S, H)
    conv_weight_ptr: tl.pointer[tl.float32],    # *f32, shape (H, H, 4)
    conv_bias_ptr: tl.pointer[tl.float32],      # *f32, shape (H,)
    conv_out_ptr: tl.pointer[tl.float32],       # *f32, shape (B, H, S)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    stride_bx_b, stride_bx_s, stride_bx_h,      # strides for (b, s, h) on Bx
    stride_w_go, stride_w_gi, stride_w_k,       # strides for (go, gi, k) on conv_weight
    stride_out_b, stride_out_h, stride_out_s,   # strides for (b, h, s) on conv_out
):
    b = tl.program_id(0)
    ci = tl.program_id(1)  # output channel index (also input channel, groups=H)
    if b >= B or ci >= H or S <= 0:
        return

    # Initialize accumulator for this (b, ci)
    acc_vec = tl.zeros((S,), dtype=tl.float32)

    # Causal conv: y[b, ci, t] = sum_{k=0..3} w[ci, ci, k] * x[b, ci, t + k] + bias[ci]
    for k in range(0, 4):
        # dot product over t
        dot = 0.0
        for t in range(0, S):
            x_pos = t + k
            x_val = tl.load(Bx_ptr + b * stride_bx_b + ci * stride_bx_s + x_pos * stride_bx_h) if x_pos < S else 0.0
            w_val = tl.load(conv_weight_ptr + ci * stride_w_go + ci * stride_w_gi + k * stride_w_k)
            dot += x_val * w_val
        acc_vec += dot

    # Add bias
    bias_val = tl.load(conv_bias_ptr + ci)
    acc_vec += bias_val

    # Store acc_vec to conv_out[b, ci, :]
    # conv_out is (B, H, S) with strides (stride_out_b, stride_out_h, stride_out_s)
    for t in range(0, S):
        tl.store(conv_out_ptr + b * stride_out_b + ci * stride_out_h + t * stride_out_s, acc_vec[t])


# Kernel 4: Final linear projection y -> out
# y: (B, S, H), out_proj_weight: (H, H), out_proj_bias: (H,)
# out: (B, S, H), float32
@triton.jit
def final_linear_kernel(
    y_ptr: tl.pointer[tl.float32],              # *f32, shape (B, S, H)
    out_proj_weight_ptr: tl.pointer[tl.float32],# *f32, shape (H, H)
    out_proj_bias_ptr: tl.pointer[tl.float32],  # *f32, shape (H,)
    out_ptr: tl.pointer[tl.float32],            # *f32, shape (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr
