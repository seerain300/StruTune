High-level plan and design
- The original PyTorch flow:
  1) In-projection: BCx = F.linear(x, in_proj_weight, in_proj_bias) with x shape (B, L, H) and in_proj_weight shape (3H, H, L), producing BCx (B, 3H, L).
  2) Transpose and chunk: BCx (B, 3H, L) → (B, L, 3H), then split into B_tensor, C_tensor, x_proj each (B, L, H).
  3) Element-wise gating: Bx = B_tensor * x_proj.
  4) Grouped causal 1D convolution: conv1d on Bx (B, H, L) using conv_weight (H, H, 4) and bias (H), groups=H, kernel_size=4. We implement causal padding via a Triton kernel (pad_left_kernel) to avoid F.pad usage. The grouped causal conv is implemented in a Triton kernel (conv_group_causal_kernel).
  5) Output gating: y = C_tensor * conv_out.
  6) Final out-projection: output = F.linear(y, out_proj_weight, out_proj_bias) with y shape (B, L, H) and out_proj_weight shape (H, L, H), producing output (B, L, H).

- Triton kernels:
  1) in_proj_kernel_B: Computes BCx (B, 3H, L) from x (B, L, H) and in_proj_weight (3H, H, L).
  2) pad_left_kernel: Pads input tensor along the last dimension with zeros on the left, used to implement causal padding for conv.
  3) conv_group_causal_kernel: Grouped causal 1D convolution with kernel_size=4, stride=1, groups=H, padding=3 (so that input length becomes L+3). It operates on Bx padded and conv_weight (H, H, 4) to produce conv_out (B, H, L).
  4) gate_mul_kernel: Elementwise multiply a (B, L, H) by b (B, L, H) to produce out (B, L, H).
  5) out_proj_kernel: Implements F.linear on y (B, L, H) with weight (H, L, H) and bias (H), producing output (B, L, H).

- Host code:
  - ModelNew.forward calls each Triton kernel exactly once, with no torch heavy ops (no F.pad, no torch.chunk, no F.linear). We use only tensor metadata and allocations.
  - All tensors are made contiguous and float32 to ensure predictable strides.

- Correctness and masks:
  - We use conservative tiling (64x128) with masks to avoid out-of-bounds accesses. The grid covers full B dimension.
  - Padding: left_pad adds three zeros (since kernel_size=4) and then we skip the first three elements in the conv. This matches the original causal behavior with F.pad(..., (K-1, 0)).

import torch.nn as nn
import triton
import triton.language as tl


# 1) In-projection: BCx = F.linear(x, in_proj_weight, bias), output shape (B, 3H, L)
@triton.jit
def in_proj_kernel_B(
    x_ptr,                 # *float32, input x (B, L, H) contiguous
    w_ptr,                 # *float32, in_proj_weight (3H, H, L) contiguous
    b_ptr,                 # *float32, in_proj_bias (3H)
    BCx_ptr,               # *float32, output BCx (B, 3H, L) contiguous
    B, L, H,               # sizes
    stride_x_b, stride_x_l, stride_x_h,        # strides for x (B, L, H)
    stride_w_o, stride_w_i, stride_w_l,        # strides for w (3H, H, L): o=3H, i=H, l=L
    stride_BC_b, stride_BC_i, stride_BC_l      # strides for BCx (B, 3H, L)
):
    # Grid: (I_tiles, L_tiles, B)
    i_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    i_offsets = i_block * 64 + tl.arange(0, 64)   # [64] over 3H (input channels)
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128] over L

    mask_i = i_offsets < (3 * H)
    mask_l = l_offsets < L
    mask = mask_i[:, None] & mask_l[None, :]

    # Initialize accumulator
    acc = tl.zeros((64, 128), dtype=tl.float32)

    # Reduction over H (input features)
    for hi in range(0, H):
        # Load x[b, l, hi]
        x_ptrs = x_ptr + b * stride_x_b + l_offsets[None, :] * stride_x_l + hi * stride_x_h  # (1, 128) per hi
        x_vals = tl.load(x_ptrs, mask=mask_l[None, :], other=0.0)  # broadcast across i_offsets

        # Load w[i, hi, l]
        w_ptrs = w_ptr + i_offsets[:, None] * stride_w_o + hi * stride_w_i + l_offsets[None, :] * stride_w_l  # (64, 128)
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)

        acc += x_vals * w_vals  # (64, 128)

    # Add bias b[i]
    bias_vals = tl.load(b_ptr + i_offsets, mask=mask_i, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store to BCx[b, i, l]
    out_ptrs = BCx_ptr + b * stride_BC_b + i_offsets[:, None] * stride_BC_i + l_offsets[None, :] * stride_BC_l
    tl.store(out_ptrs, acc, mask=mask)


# 2) Pad left along last dimension: Input (B, D, T), pad_left zeros on last dim to produce (B, D, T+pad_left)
@triton.jit
def pad_left_kernel(
    in_ptr,            # *float32, input (B, D, T)
    out_ptr,           # *float32, output (B, D, T+pad_left)
    B, D, T, pad_left, # sizes
    stride_in_b, stride_in_d, stride_in_t,
    stride_out_b, stride_out_d, stride_out_t
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    t_out = tl.program_id(2)  # index along padded T

    t_in = t_out - pad_left
    # If t_in >= 0 and t_in < T, copy; else store 0
    valid = (t_in >= 0) & (t_in < T)
    in_ptrs = in_ptr + b * stride_in_b + d * stride_in_d + t_in * stride_in_t
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + t_out * stride_out_t

    val = tl.load(in_ptrs, mask=valid, other=0.0)
    tl.store(out_ptrs, val)

# Input Bx_pad: (B, H, Tpad), conv_weight: (H, H, 4), conv_bias: (H)
# Output conv_out: (B, H, T), where T = Tpad - (K-1)
@triton.jit
def conv_group_causal_kernel(
    Bx_pad_ptr,           # *float32, input Bx_pad (B, H, Tpad) contiguous
    w_ptr,                # *float32, weight (H, H, 4) contiguous
    bias_ptr,             # *float32, bias (H)
    out_ptr,              # *float32, output (B, H, T) contiguous
    B, T, H, Tpad,        # sizes
    stride_Bx_b, stride_Bx_h, stride_Bx_t,   # strides for Bx_pad (B, H, Tpad)

    t_block = tl.program_id(1)
    b = tl.program_id(2)

    acc = tl.zeros((64, 128), dtype=tl.float32)

    K = 4  # fixed kernel size
    # For each output position t, sum over kernel positions k=0..K-1

        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (64,)

        # Accumulate (grouped: input channel = output channel)
        acc += w_vals[:, None] * x_vals

    # Add bias
    b_vals = tl.load(bias_ptr + h_offsets, mask=mask_h, other=0.0)  # (64,)
    acc += b_vals[:, None]

    # Store conv_out[b, h, t]
    out_ptrs = out_ptr + b * stride_out_b + h_offsets[:, None] * stride_out_h + t_offsets[None, :] * stride_out_t
    tl.store(out_ptrs, acc, mask=mask)

    l_block = tl.program_id(1)
    b = tl.program_id(2)

    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_l = l_offsets < L

    a_ptrs = a_ptr + b * stride_a_b + l_offsets[None, :] * stride_a_l + h_offsets[:, None] * stride_a_h
    b_ptrs = b_ptr + b * stride_b_b + l_offsets[None, :] * stride_b_l + h_offsets[:, None] * stride_b_h
    out_ptrs = out_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + h_offsets[:, None] * stride_out_h

    b_vals = tl.load(b_ptrs, mask=mask, other=0.0)
    out_vals = a_vals * b_vals
    tl.store(out_ptrs, out_vals, mask=mask)

        # x[b, l, hi] -> (1, 128)
        x_ptrs = x_ptr + b * stride_x_b + l_offsets[None, :] * stride_x_l + hi * stride_x_h
        x_vals = tl.load(x_ptrs, mask=mask_l[None, :], other=0.0)

        # w[i, hi, l] -> (64, 128)
        w_ptrs = w_ptr + i_offsets[:, None] * stride_w_o + hi * stride_w_i + l_offsets[None, :] * stride_w_l
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)

        acc += x_vals * w_vals  # (64, 128)

    # Add bias b[i]
    bias_vals = tl.load(b_ptr + i_offsets, mask=mask_i, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store BCx[b, i, l]
    out_ptrs = BCx_ptr + b * stride_BC_b + i_offsets[:, None] * stride_BC_i + l_offsets[None, :] * stride_BC_l
    tl.store(out_ptrs, acc, mask=mask)


# 2) Pad left along last dimension: Input (B, D, T), pad_left zeros on last dim to produce (B, D, T+pad_left)
@triton.jit
def pad_left_kernel(
    in_ptr,            # *float32, input (B, D, T)
    out_ptr,           # *float32, output (B, D, T+pad_left)
    B, D, T, pad_left, # sizes
    stride_in_b, stride_in_d, stride_in_t,
    stride_out_b, stride_out_d, stride_out_t
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    t_out = tl.program_id(2)  # index along padded T

    t_in = t_out - pad_left
    valid = (t_in >= 0) & (t_in < T)

    in_ptrs = in_ptr + b * stride_in_b + d * stride_in_d + t_in * stride_in_t
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + t_out * stride_out_t

    val = tl.load(in_ptrs, mask=valid, other=0.0)
    tl.store(out_ptrs, val)


# 3) Grouped causal 1D convolution:
# Input Bx_pad: (B, H, Tpad), conv_weight: (H, H, 4), conv_bias: (H)
# Output conv_out: (B, H, T), where T = Tpad - (K-1)
@triton.jit
def conv_group_causal_kernel(
    Bx_pad_ptr,           # *float32, input Bx_pad (B, H, Tpad) contiguous
    w_ptr,                # *float32, weight (H, H, 4) contiguous
    bias_ptr,             # *float32, bias (H)
    out_ptr,              # *float32, output (B, H, T) contiguous
    B, T, H, Tpad,        # sizes
    stride_Bx_b, stride_Bx_h, stride_Bx_t,   # strides for Bx_pad (B, H, Tpad)
    stride_w_o, stride_w_i, stride_w_k,      # strides for w (H, H, 4)
    stride_out_b, stride_out_h, stride_out_t # strides for out (B, H, T)
):
    # Grid: (H_tiles, T_tiles, B)
    h_block = tl.program_id(0)
    t_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64] over H
    t_offsets = t_block * 128 + tl.arange(0, 128) # [128] over T

    mask_h = h_offsets < H
    mask_t = t_offsets < T
    mask = mask_h[:, None] & mask_t[None, :]

    acc = tl.zeros((64, 128), dtype=tl.float32)

    K = 4  # fixed kernel size
    # For each output position t, sum over kernel positions k=0..K-1
    for k in range(0, K):
        t_in = t_offsets + k  # (128,)
        valid = (t_in >= 0) & (t_in < Tpad) & mask_t[None, :]  # broadcasting over h

        # Load Bx_pad[b, h, t_in]
        in_ptrs = Bx_pad_ptr + b * stride_Bx_b + h_offsets[:, None] * stride_Bx_h + t_in[None, :] * stride_Bx_t
        x_vals = tl.load(in_ptrs, mask=valid, other=0.0)  # (64, 128)

        # Load w[h, h, k]
        w_ptrs = w_ptr + h_offsets[:, None] * stride_w_o + h_offsets[None, :] * stride_w_i + k * stride_w_k
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)  # (64,)

        # Accumulate (grouped: input channel = output channel)
        acc += w_vals[:, None] * x_vals

    # Add bias
    b_vals = tl.load(bias_ptr + h_offsets, mask=mask_h, other=0.0)  # (64,)
    acc += b_vals[:, None]

    # Store conv_out[b, h, t]
    out_ptrs = out_ptr + b * stride_out_b + h_offsets[:, None] * stride_out_h + t_offsets[None, :] * stride_out_t
    tl.store(out_ptrs, acc, mask=mask)
