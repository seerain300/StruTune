@triton.jit
def in_proj_kernel_B(
    x_ptr,               # *float32, x (B, L, H)
    w_ptr,               # *float32, in_proj_weight (3H, H, L)
    bias_ptr,            # *float32, in_proj_bias (3H)
    out_ptr,             # *float32, BCx (B, 3H, L)
    B, L, H,             # sizes
    stride_x_b, stride_x_l, stride_x_h,     # strides for x (B, L, H)
    stride_w_o, stride_w_i, stride_w_l,     # strides for w (3H, H, L)
    stride_out_b, stride_out_o, stride_out_l   # strides for out (B, 3H, L)
):

    o_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    o_offsets = o_block * 64 + tl.arange(0, 64)   # [64] channels in {0..3H-1}
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128] positions along sequence

    mask_o = o_offsets < (3 * H)
    mask_l = l_offsets < L
    mask = mask_o[:, None] & mask_l[None, :]

    acc = tl.zeros((64, 128), dtype=tl.float32)

    # For each output channel o in [0, 3H)
    for o in range(0, 3 * H):
        # Compute corresponding input channel i = o % H
        i = o % H
        # Load x[b, l, i]
        x_ptrs = x_ptr + b * stride_x_b + l_offsets[None, :] * stride_x_l + i * stride_x_h
        x_vals = tl.load(x_ptrs, mask=mask_l[None, :], other=0.0)  # (1, 128), broadcast along o

    bias_val = tl.load(bias_ptr + o_offsets, mask=mask_o, other=0.0)  # (64,)
    acc += bias_val[:, None]

    # Store to out[b, o, l]
    out_ptrs = out_ptr + b * stride_out_b + o_offsets[:, None] * stride_out_o + l_offsets[None, :] * stride_out_l
    tl.store(out_ptrs, acc, mask=mask)

    stride_out_b, stride_out_h, stride_out_l  # strides for out (B, H, L)

    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

            bx_vals = tl.load(bx_ptrs, mask=mask & (l_src[None, :] < L), other=0.0)  # (64,128)

            w_val = tl.load(w_ptrs)  # scalar

    # Add bias
    bias_vals = tl.load(bias_ptr + h_offsets, mask=mask_h, other=0.0)  # (64,)
    acc += bias_vals[:, None]

    # Store to out[b, h, l]
    out_ptrs = out_ptr + b * stride_out_b + h_offsets[:, None] * stride_out_h + l_offsets[None, :] * stride_out_l
    tl.store(out_ptrs, acc, mask=mask)

):

    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    acc = tl.zeros((64, 128), dtype=tl.float32)

    # For each output channel h
    for ho in range(0, H):
        # For each input channel h in y (same index), sum over L dimension
        y_ptrs = y_ptr + b * stride_y_b + l_offsets[None, :] * stride_y_l + ho * stride_y_h
        y_vals = tl.load(y_ptrs, mask=mask, other=0.0)  # (64,128)

    # Add bias
    b_vals = tl.load(b_ptr + h_offsets, mask=mask_h, other=0.0)  # (64,)
    acc += b_vals[:, None]

    # Store output
    out_ptrs = out_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + h_offsets[:, None] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask)

        w_ptrs = w_ptr + o * stride_w_o + i * stride_w_i + l_offsets * stride_w_l

        # Accumulate: acc += w_vals[:, None] * x_vals[None, :]
        acc += w_vals[:, None] * x_vals[None, :]

    # Add bias for channel o
    bias_val = tl.load(bias_ptr + o_offsets, mask=mask_o, other=0.0)  # (64,)
    acc += bias_val[:, None]

    # Store to out[b, o, l]
    out_ptrs = out_ptr + b * stride_out_b + o_offsets[:, None] * stride_out_o + l_offsets[None, :] * stride_out_l
    tl.store(out_ptrs, acc, mask=mask)


# 2) Chunk BCx (B, 3H, L) into three parts: B, C, X_proj, each (B, L, H)
@triton.jit
def chunk3_kernel(
    inp_ptr,              # *float32, input (B, 3H, L)
    outB_ptr,             # *float32, output B (B, L, H)
    outC_ptr,             # *float32, output C (B, L, H)
    outX_ptr,             # *float32, output X_proj (B, L, H)
    B, L, H,              # sizes
    stride_inp_b, stride_inp_o, stride_inp_l,   # strides for inp (B, 3H, L)
    stride_out_b, stride_out_l, stride_out_h    # strides for outputs (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64] channels in {0..H-1}
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128] positions

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    # Copy x_proj from inp[b, l, 0..H-1]
    for ho in range(0, H):
        in_ptrs_x = inp_ptr + b * stride_inp_b + ho * stride_inp_o + l_offsets[None, :] * stride_inp_l
        vals_x = tl.load(in_ptrs_x, mask=mask, other=0.0)  # (64,128)
        out_ptrs_x = outX_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_x, vals_x, mask=mask)

    # Copy B from inp[b, l, H..2H-1]
    for ho in range(0, H):
        in_ptrs_B = inp_ptr + b * stride_inp_b + (H + ho) * stride_inp_o + l_offsets[None, :] * stride_inp_l
        vals_B = tl.load(in_ptrs_B, mask=mask, other=0.0)  # (64,128)
        out_ptrs_B = outB_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_B, vals_B, mask=mask)

    # Copy C from inp[b, l, 2H..3H-1]
    for ho in range(0, H):
        in_ptrs_C = inp_ptr + b * stride_inp_b + (2 * H + ho) * stride_inp_o + l_offsets[None, :] * stride_inp_l
        vals_C = tl.load(in_ptrs_C, mask=mask, other=0.0)  # (64,128)
        out_ptrs_C = outC_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + ho * stride_out_h
        tl.store(out_ptrs_C, vals_C, mask=mask)


# 3) Element-wise gating: out = a * b (vectorized), a,b are (B, L, H)
@triton.jit
def gate_mul_kernel(
    a_ptr, b_ptr, out_ptr,   # pointers
    B, L, H,                 # sizes
    stride_a_b, stride_a_l, stride_a_h,    # strides for a (B, L, H)
    stride_b_b, stride_b_l, stride_b_h,    # strides for b (B, L, H)
    stride_out_b, stride_out_l, stride_out_h   # strides for out (B, L, H)
):
    # Grid: (H_tiles, L_tiles, B)
    h_block = tl.program_id(0)
    l_block = tl.program_id(1)
    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets = l_block * 128 + tl.arange(0, 128) # [128]

    mask_h = h_offsets < H
    mask_l = l_offsets < L
    mask = mask_h[:, None] & mask_l[None, :]

    a_ptrs = a_ptr + b * stride_a_b + l_offsets[None, :] * stride_a_l + h_offsets[:, None] * stride_a_h
    b_ptrs = b_ptr + b * stride_b_b + l_offsets[None, :] * stride_b_l + h_offsets[:, None] * stride_b_h
    out_ptrs = out_ptr + b * stride_out_b + l_offsets[None, :] * stride_out_l + h_offsets[:, None] * stride_out_h

    a_vals = tl.load(a_ptrs, mask=mask, other=0.0)
    b_vals = tl.load(b_ptrs, mask=mask, other=0.0)
    out_vals = a_vals * b_vals
    tl.store(out_ptrs, out_vals, mask=mask)


# 4) Grouped causal 1D convolution:

    b = tl.program_id(2)

    h_offsets = h_block * 64 + tl.arange(0, 64)   # [64]
    l_offsets
