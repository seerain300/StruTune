import torch
import triton
import triton.language as tl

# 1) Triple linear projection: given x[B, S, H], compute three outputs (B,S,H)
#    out[b, s, h] = sum_k x[b, s, k] * W[h, k] + bias[h]
#    W0, W1, W2 each are (H, H), biases b0, b1, b2 are (H,)
@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,         # *f32, (B, S, H)
    W_ptr, b_ptr,  # *f32, (H,H), (H,)
    out_ptr,       # *f32, (B,S,H)
    B, S, H,
    x_strideB, x_strideS, x_strideH,
    W_strideH, W_strideK,
    out_strideB, out_strideS, out_strideH,
    BLOCK_S: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    # accumulator for output vector (BLOCK_S,)
    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # iterate over K dimension in chunks of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # load W[h, k] vector
        w_vec = tl.load(W_ptr + h * W_strideH + k_offsets * W_strideK, mask=mask_k, other=0.0)
        # broadcast w_vec to [BLOCK_S, BLOCK_K]
        w_mat = tl.broadcast_to(w_vec[None, :], [BLOCK_S, BLOCK_K])

        # load x[b, s, k] matrix for all s in tile and k chunk
        x_ptrs = x_ptr + b * x_strideB + s_offsets[:, None] * x_strideS + k_offsets[None, :] * x_strideH
        x_mask = (mask_s[:, None] & mask_k[None, :])
        x_mat = tl.load(x_ptrs, mask=x_mask, other=0.0)

        # accumulate outer product sum_k x * w
        # acc += sum over K of x_mat[:, k] * w_mat[:, k]
        # Triton doesn't have a direct sum over axis, so we reduce manually:
        for kk in range(BLOCK_K):
            k_valid = (k_start + kk) < H
            # skip when out of range
            if k_valid:
                x_vec = x_mat[:, kk]
                acc += x_vec * w_mat[:, kk]

    # add bias[h]
    bias_h = tl.load(b_ptr + h)
    acc += bias_h

    # store results to out[b, s, h] for all s in tile
    out_ptrs = out_ptr + b * out_strideB + s_offsets * out_strideS + h * out_strideH
    tl.store(out_ptrs, acc, mask=mask_s)

    A_ptr, B_ptr, C_ptr,  # *f32, shapes (B,S,H)
    B, S, H,
    A_strideB, A_strideS, A_strideH,
    B_strideB, B_strideS, B_strideH,
    C_strideB, C_strideS, C_strideH,
    BLOCK_S: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask = s_offsets < S
    a_ptrs = A_ptr + b * A_strideB + s_offsets * A_strideS + h * A_strideH
    b_ptrs = B_ptr + b * B_strideB + s_offsets * B_strideS + h * B_strideH
    c_ptrs = C_ptr + b * C_strideB + s_offsets * C_strideS + h * C_strideH
    a = tl.load(a_ptrs, mask=mask, other=0.0)
    b_mat = tl.load(b_ptrs, mask=mask, other=0.0)
    c = a * b_mat
    tl.store(c_ptrs, c, mask=mask)

    convW_ptr,    # *f32, (H, H, 4), groups=H (depthwise)
    convB_ptr,    # *f32, (H,)
    out_ptr,      # *f32, (B, H, S)
    B, S, H,
    Bx_strideB, Bx_strideS, Bx_strideH,
    convW_strideC, convW_strideG, convW_strideK,  # convW is (C=H, G=H, K=4) but since groups=C, G==C
    out_strideB, out_strideC, out_strideS,
    BLOCK_S: tl.constexpr,
):

    c = tl.program_id(1)
    tile_s = tl.program_id(2)
    t_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_t = t_offsets < S

    # initialize accumulator
    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # causal kernel_size=4: accumulate t+k-1 indices
    for k in range(4):

        valid = (t_in >= 0) & (t_in < S) & mask_t

        bx_ptrs = Bx_ptr + b * Bx_strideB + t_in * Bx_strideS + c * Bx_strideH

        w_val = tl.load(convW_ptr + c * convW_strideC + c * convW_strideG + k * convW_strideK)
        acc += bx_vals * w_val

    # add bias[c]
    bias_c = tl.load(convB_ptr + c)
    acc += bias_c

    # store conv_out[b, c, t]
    out_ptrs = out_ptr + b * out_strideB + c * out_strideC + t_offsets * out_strideS
    tl.store(out_ptrs, acc, mask=mask_t)

#    Here out_proj_weight is (H, H), out_proj_bias is (H).
#    Output has shape (B, S, H)

    out_ptr,       # *f32, (B,S,H)
    B, S, H,
    y_strideB, y_strideS, y_strideH,
    W_strideH, W_strideK,  # Wout is (H,H), i.e., (C,H)
    out_strideB, out_strideS, out_strideH,

):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H
        # load weights Wout[h, k] as vector
        w_vec = tl.load(Wout_ptr + h * W_strideH + k_offsets * W_strideK, mask=mask_k, other=0.0)
        w_mat = tl.broadcast_to(w_vec[None, :], [BLOCK_S, BLOCK_K])

        # load y[b, s, k] matrix
        y_ptrs = y_ptr + b * y_strideB + s_offsets[:, None] * y_strideS + k_offsets[None, :] * y_strideH
        y_mask = (mask_s[:, None] & mask_k[None, :])

    bias_h = tl.load(bout_ptr + h)
    acc += bias_h

    out_ptrs = out_ptr + b * out_strideB + s_offsets * out_strideS + h * out_strideH

        t_in = t_offsets + k  # input position (explicitly padded to length S+3)
        # Bx_padded[b, c, t_in] is always valid for t_in in [0, S+3)
        bx_ptrs = Bx_ptr + b * Bx_strideB + t_in * Bx_strideS + c * Bx_strideH
        bx_vals = tl.load(bx_ptrs, mask=mask_t, other=0.0)
        w_val = tl.load(convW_ptr + c * convW_strideC + c * convW_strideG + k * convW_strideK)
        acc += bx_vals * w_val

    bias_c = tl.load(convB_ptr + c)
    acc += bias_c

    out_ptrs = out_ptr + b * out_strideB + c * out_strideC + t_offsets * out_strideS
    tl.store(out_ptrs, acc, mask=mask_t)

# 4) Final linear projection over y (B,S,H): y @ out_proj_weight^T + out_proj_bias
@triton.jit
def final_linear_bsh_kernel(
    y_ptr,         # *f32, (B,S,H)
    Wout_ptr, bout_ptr,  # *f32, (H,H), (H,)
    out_ptr,       # *f32, (B,S,H)
    B, S, H,
    y_strideB, y_strideS, y_strideH,
    W_strideH, W_strideK,  # Wout is (H,H), i.e., (C,H)
    out_strideB, out_strideS, out_strideH,
    BLOCK_S: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H
        # load weights Wout[h, k] as vector
        w_vec = tl.load(Wout_ptr + h * W_strideH + k_offsets * W_strideK, mask=mask_k, other=0.0)
        w_mat = tl.broadcast_to(w_vec[None, :], [BLOCK_S, BLOCK_K])

        # load y[b, s, k] matrix
        y_ptrs = y_ptr + b * y_strideB + s_offsets[:, None] * y_strideS + k_offsets[None, :] * y_strideH
        y_mask = (mask_s[:, None] & mask_k[None, :])
        y_mat = tl.load(y_ptrs, mask=y_mask, other=0.0)

        for kk in range(BLOCK_K):
            k_valid = (k_start + kk) < H
            if k_valid:
                y_vec = y_mat[:, kk]
                acc += y_vec * w_mat[:, kk]

    bias_h = tl.load(bout_ptr + h)
    acc += bias_h

    out_ptrs = out_ptr + b * out_strideB + s_offsets * out_strideS + h * out_strideH
    tl.store(out_ptrs, acc, mask=mask_s)
