import torch
import triton
import triton.language as tl

# 1) Triple linear projection: x[B, S, H] @ W[h, k]^T + bias[h] -> out[B, S, H]
@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,         # *f32, (B, S, H)
    W_ptr, b_ptr,  # *f32, (H, H), (H,)
    out_ptr,       # *f32, (B, S, H)
    B: tl.int32, S: tl.int32, H: tl.int32,
    x_s0, x_s1, x_s2,
    W_s0, W_s1,
    out_s0, out_s1, out_s2,
    BLOCK_S: tl.constexpr, BLOCK_K: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # Loop over hidden dimension K in chunks
    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # x[b, s, k] for all s in tile, k in chunk
        x_ptrs = x_ptr + b * x_s0 + s_offsets[:, None] * x_s1 + k_offsets[None, :] * x_s2
        x_vals = tl.load(x_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)
        # W[h, k]
        w_ptrs = W_ptr + h * W_s0 + k_offsets * W_s1
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)  # shape [BLOCK_K]
        # Accumulate outer product along K chunk
        # acc += sum_k (x_vals[:, k] * W[h, k])
        # Do this by multiplying each column and reduce
        for kk in range(BLOCK_K):
            k_valid = k_start + kk < H
            if k_valid:
                w_k = w_vals[kk]
                acc += x_vals[:, kk] * w_k

    # Add bias[h]
    b_val = tl.load(b_ptr + h)
    acc += b_val

    # Store to out[b, s, h]
    out_ptrs = out_ptr + b * out_s0 + s_offsets * out_s1 + h * out_s2
    tl.store(out_ptrs, acc, mask=mask_s)

# 2) Element-wise multiplication: out = A * B for tensors of shape (B, S, H)
@triton.jit
def elemwise_mul_bsh_kernel(
    A_ptr, B_ptr, Out_ptr,
    B: tl.int32, S: tl.int32, H: tl.int32,
    A_s0, A_s1, A_s2,
    B_s0, B_s1, B_s2,
    O_s0, O_s1, O_s2,
    BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    a_ptrs = A_ptr + b * A_s0 + s_offsets * A_s1 + h * A_s2
    b_ptrs = B_ptr + b * B_s0 + s_offsets * B_s1 + h * B_s2
    out_ptrs = Out_ptr + b * O_s0 + s_offsets * O_s1 + h * O_s2

    a = tl.load(a_ptrs, mask=mask_s, other=0.0)
    b_vals = tl.load(b_ptrs, mask=mask_s, other=0.0)
    out = a * b_vals
    tl.store(out_ptrs, out, mask=mask_s)

# 3) Grouped causal conv1d (kernel_size=4, groups=H):
#    Input is Bx_padded[B, 3H, S+3] (left pad of 3). We compute conv_out[B, H, S].
#    For each channel c in [0..H-1], conv_out[b, c, t] = sum_{k=0..3} Bx_padded[b, c, t+k-1] * conv_weight[c, c, k] + conv_bias[c]
@triton.jit
def grouped_causal_conv1d_kernel(
    Input_ptr,     # *f32, (B, 3H, S+3)
    convW_ptr, convB_ptr,   # *f32, (H, H, 4), (H,)
    out_ptr,       # *f32, (B, H, S)
    B: tl.int32, S: tl.int32, H: tl.int32,
    inp_s0, inp_s1, inp_s2,
    w_s0, w_s1, w_s2,
    out_s0, out_s1, out_s2,
    BLOCK_S: tl.constexpr
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # We hardcode kernel_size=4; no padding on right (causal only left pad)
    for k in range(4):
        # input index along S is s + k - 1 (left causal pad)
        s_in = s_offsets + (k - 1)
        # valid only if 0 <= s_in < S+3, but since S_in is s_offsets + (k-1), we ensure s_offsets start from t=0.
        # We need s_in >= 0 and < S+3. For k=0, s_in = s_offsets - 1; since s_offsets>=0, s_in>=-1, but s_in must be >=0, so we add mask.
        # Here we rely on the fact that left padding is handled by F.pad, so s_in will be in bounds when k>0 for t=0,1,2; for k=0 t=0 is -1 -> corresponds to zero in padded input. We'll mask accordingly.
        # Build mask: if s_in < 0, input value is 0. Else s_in < S+3.
        mask_in = (s_in >= 0) & (s_in < (S + 3)) & mask_s

        # Input address: Input[b, c, s_in]
        inp_ptrs = Input_ptr + b * inp_s0 + c * inp_s1 + s_in * inp_s2
        inp_vals = tl.load(inp_ptrs, mask=mask_in, other=0.0)

        # conv weight at (c, c, k)
        w_ptrs = convW_ptr + c * w_s0 + c * w_s1 + k * w_s2
        w_k = tl.load(w_ptrs)

        acc += inp_vals * w_k

    # Add conv bias[c]
    b_c = tl.load(convB_ptr + c)
    acc += b_c

    # Store conv_out[b, c, s]
    out_ptrs = out_ptr + b * out_s0 + c * out_s1 + s_offsets * out_s2
    tl.store(out_ptrs, acc, mask=mask_s)

# 4) Final linear projection: y[B, S, H] @ out_proj_weight^T + out_proj_bias
@triton.jit
def linear_bsh_kernel(
    y_ptr,         # *f32, (B, S, H)
    W_ptr, b_ptr,  # *f32, (H, H), (H,)
    out_ptr,       # *f32, (B, S, H)
    B: tl.int32, S: tl.int32, H: tl.int32,
    y_s0, y_s1, y_s2,
    W_s0, W_s1,
    out_s0, out_s1, out_s2,
    BLOCK_S: tl.constexpr, BLOCK_K: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros([BLOCK_S], dtype=tl.float32)

    # Loop over K=H in chunks
    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # y[b, s, k] for all s in tile, k in chunk
        y_ptrs = y_ptr + b * y_s0 + s_offsets[:, None] * y_s1 + k_offsets[None, :] * y_s2
        y_vals = tl.load(y_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)
        # W[h, k]
        w_ptrs = W_ptr + h * W_s0 + k_offsets * W_s1
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)  # shape [BLOCK_K]

        # Accumulate outer product along K chunk
        for kk in range(BLOCK_K):
            k_valid = k_start + kk < H
            if k_valid:
                w_k = w_vals[kk]
                acc += y_vals[:, kk] * w_k

    # Add bias[h]
    b_val = tl.load(b_ptr + h)
    acc += b_val

    # Store to out[b, s, h]
    out_ptrs = out_ptr + b * out_s0 + s_offsets * out_s1 + h * out_s2
    tl.store(out_ptrs, acc, mask=mask_s)
