#    Output shape: (B, S, H).
@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,            # *f32, (B, S, 1)
    W0_ptr, b0_ptr,   # *f32, (H, 1), (H,)
    W1_ptr, b1_ptr,   # *f32, (H, 1), (H,)
    W2_ptr, b2_ptr,   # *f32, (H, 1), (H,)

    W0_stride_h, W0_stride_k,
    W1_stride_h, W1_stride_k,
    W2_stride_h, W2_stride_k,

):

    h = tl.program_id(1)
    tile_s = tl.program_id(2)

    mask_s = s_offsets < S
