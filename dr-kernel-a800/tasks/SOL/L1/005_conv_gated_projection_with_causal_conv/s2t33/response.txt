# 2) Element-wise gating: Bx = B_out * X_out
@triton.jit
def elemwise_mul_bsh_kernel(
    in_ptr1, in_ptr2, out_ptr,
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    in_stride_b, in_stride_s, in_stride_h,
    out_stride_b, out_stride_s, out_stride_h,
    BLOCK_S: tl.constexpr = 128
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_s = tl.program_id(2)

    s_offsets = pid_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask = s_offsets < S

    in1_ptrs = in_ptr1 + pid_b * in_stride_b + pid_h * in_stride_h + s_offsets * in_stride_s
    in2_ptrs = in_ptr2 + pid_b * in_stride_b + pid_h * in_stride_h + s_offsets * in_stride_s

    v1 = tl.load(in1_ptrs, mask=mask, other=0.0)
    v2 = tl.load(in2_ptrs, mask=mask, other=0.0)

    out_ptrs = out_ptr + pid_b * out_stride_b + pid_h * out_stride_h + s_offsets * out_stride_s
    tl.store(out_ptrs, v1 * v2, mask=mask)


# 3) Grouped causal 1D convolution with kernel_size=4, groups=H
@triton.jit
def grouped_causal_conv1d_kernel(
    x_ptr,        # *f32, (B, S, H) input after gating
    convW_ptr, convB_ptr,   # *f32, convW (H, H, 4), convB (H,)
    out_ptr,      # *f32, (B, H, S) output
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    x_stride_b, x_stride_s, x_stride_h,
    convW_stride_group, convW_stride_c, convW_stride_k,  # (H, H, 4)
    out_stride_b, out_stride_h, out_stride_s,
    BLOCK_S: tl.constexpr = 128
):
    # Grid: (B, H, tiles of S)
    b = tl.program_id(0)
    c = tl.program_id(1)  # group index equals output channel
    tile_s = tl.program_id(2)

    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # Fixed kernel_size=4 (causal). We emulate left padding of 3.
    for k in range(4):
        t_offsets = s_offsets + (k - 1)  # causal shift
        valid = (t_offsets >= 0) & (t_offsets < S) & mask_s
        x_ptrs = x_ptr + b * x_stride_b + t_offsets * x_stride_s + c * x_stride_h
        x_vals = tl.load(x_ptrs, mask=valid, other=0.0)

        w_ptrs = convW_ptr + c * convW_stride_group + c * convW_stride_c + k * convW_stride_k
        w_val = tl.load(w_ptrs)  # scalar per (group,c,k)
        acc += x_vals * w_val

    # Add bias
    b_val = tl.load(convB_ptr + c)
    acc += b_val

    out_ptrs = out_ptr + b * out_stride_b + c * out_stride_h + s_offsets * out_stride_s
    tl.store(out_ptrs, acc, mask=mask_s)


# 4) Final linear projection: y (B, S, H) -> out (B, S, H) using out_proj_weight (H, H), bias (H)
@triton.jit
def linear_bsh_kernel(
    y_ptr,        # *f32, (B, S, H)
    outW_ptr, outB_ptr,  # *f32, out_proj_weight (H, H), out_proj_bias (H,)
    out_ptr,      # *f32, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    y_stride_b, y_stride_s, y_stride_h,
    outW_stride_h, outW_stride_k,
    out_stride_b, out_stride_s, out_stride_h,
    BLOCK_S: tl.constexpr = 128, BLOCK_K: tl.constexpr = 128
):
    # Grid: (B, H, tiles of S)
    b = tl.program_id(0)
    h_out = tl.program_id(1)  # output hidden index
    tile_s = tl.program_id(2)

    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # For each output hidden h_out, compute dot over input hidden h_in
    for k0 in range(0, H, BLOCK_K):
        h_in = k0 + tl.arange(0, BLOCK_K)
        mask_k = h_in < H

        y_ptrs = y_ptr + b * y_stride_b + h_out * y_stride_h + s_offsets[:, None] * y_stride_s + h_in[None, :] * y_stride_h
        y_vals = tl.load(y_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)

        outW_ptrs = outW_ptr + h_out * outW_stride_h + h_in * outW_stride_k
        outW_vals = tl.load(outW_ptrs, mask=mask_k, other=0.0)

        acc += tl.sum(y_vals * outW_vals[None, :], axis=1)

    # Add bias
    b_val = tl.load(outB_ptr + h_out)
    acc += b_val

    out_ptrs = out_ptr + b * out_stride_b + h_out * out_stride_h + s_offsets * out_stride_s
    tl.store(out_ptrs, acc, mask=mask_s)
