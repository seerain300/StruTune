@triton.jit
def in_proj_kernel(

    W_ptr,            # *const float, in_proj_weight: (I, H), I=3*H
    Out_ptr,          # *float, output BCx: (B, S, I)
    B: tl.int32,      # batch size
    S: tl.int32,      # sequence length
    H: tl.int32,      # hidden size
    I: tl.int32,      # I = 3*H
    # strides
    x_b_stride: tl.int32, x_s_stride: tl.int32, x_h_stride: tl.int32,
    w_i_stride: tl.int32, w_h_stride: tl.int32,
    out_b_stride: tl.int32, out_s_stride: tl.int32, out_i_stride: tl.int32,
    BLOCK_I: tl.constexpr,  # tile size along I
    BLOCK_H: tl.constexpr,  # tile size along H reduction
):
    pid = tl.program_id(axis=0)  # each program handles one (b, s)
    b = pid // S
    s = pid % S

    # Base pointers for this (b, s)
    x_base = X_ptr + b * x_b_stride + s * x_s_stride
    out_base = Out_ptr + b * out_b_stride + s * out_s_stride

    # Loop over I in tiles
    for i_start in range(0, I, BLOCK_I):
        i_offsets = i_start + tl.arange(0, BLOCK_I)
        i_mask = i_offsets < I

        # Accumulator for output vector of size BLOCK_I
        acc = tl.zeros((BLOCK_I,), dtype=tl.float32)

        # Reduction over H in tiles
        for h_start in range(0, H, BLOCK_H):
            h_offsets = h_start + tl.arange(0, BLOCK_H)
            h_mask = h_offsets < H

            # Load x[b, s, h_offsets] as a vector (BLOCK_H,)
            x_ptrs = x_base + h_offsets * x_h_stride
            x_vals = tl.load(x_ptrs, mask=h_mask, other=0.0).to(tl.float32)  # (BLOCK_H,)

            # Load w[i_offsets, h_offsets] as a matrix (BLOCK_I, BLOCK_H)
            w_ptrs = W_ptr + i_offsets[:, None] * w_i_stride + h_offsets[None, :] * w_h_stride

    w_i_stride: tl.int32, w_h_stride: tl.int32,
    out_b_stride: tl.int32, out_s_stride: tl.int32, out_i_stride: tl.int32,
    BLOCK_I: tl.constexpr,  # tile size along I
    BLOCK_H: tl.constexpr,  # tile size along H reduction
):
    pid = tl.program_id(axis=0)  # each program handles one (b, s)
    b = pid // S
    s = pid % S

    # Base pointers for this (b, s)
    x_base = X_ptr + b * x_b_stride + s * x_s_stride
    out_base = Out_ptr + b * out_b_stride + s * out_s_stride

    # Loop over I in tiles
    for i_start in range(0, I, BLOCK_I):
        i_offsets = i_start + tl.arange(0, BLOCK_I)
        i_mask = i_offsets < I

        # Accumulator for output vector of size BLOCK_I
        acc = tl.zeros((BLOCK_I,), dtype=tl.float32)

        # Reduction over H in tiles
        for h_start in range(0, H, BLOCK_H):
            h_offsets = h_start + tl.arange(0, BLOCK_H)
            h_mask = h_offsets < H

            # Load x[b, s, h_offsets] as a vector (BLOCK_H,)
            x_ptrs = x_base + h_offsets * x_h_stride
            x_vals = tl.load(x_ptrs, mask=h_mask, other=0.0).to(tl.float32)  # (BLOCK_H,)

            # Load w[i_offsets, h_offsets] as a matrix (BLOCK_I, BLOCK_H)
            w_ptrs = W_ptr + i_offsets[:, None] * w_i_stride + h_offsets[None, :] * w_h_stride
            w_mask = (i_mask[:, None]) & (h_mask[None, :])
            w_vals = tl.load(w_ptrs, mask=w_mask, other=0.0).to(tl.float32)  # (BLOCK_I, BLOCK_H)

            # acc[i] += sum_h w[i, h] * x[h]
            acc += tl.sum(w_vals * x_vals[None, :], axis=1)

        # Store results to Out[b, s, i_offsets]
        out_ptrs = out_base + i_offsets * out_i_stride
        tl.store(out_ptrs, acc, mask=i_mask)


@triton.jit
def grouped_causal_conv1d_kernel(
    Bx_ptr,           # *const float, input for conv: (B, H, S) — Bx derived from BCx chunks
    W_ptr,            # *const float, conv_weight: (H, 1, 4)
    Bias_ptr,         # *const float, conv_bias: (H,)
    Out_ptr,          # *float, output conv: (B, H, S)
    B: tl.int32,      # batch size
    S: tl.int32,      # sequence length
    H: tl.int32,      # hidden size (groups)
    K: tl.constexpr,  # kernel_size, here 4
    BLOCK_T: tl.constexpr,  # tile size along S (output length)
):
    # Each program handles one (b, g) channel
    pid = tl.program_id(axis=0)
    b = pid // H
    g = pid % H

    # Base pointers
    # Bx_ptr layout is (B, H, S); for each b, we stride over S with step H
    Bx_base = Bx_ptr + b * (H * S) + g * S
    Out_base = Out_ptr + b * (H * S) + g * S  # Out is (B, H, S)

    # Preload bias
    bias = tl.load(Bias_ptr + g).to(tl.float32)

    # Conv: grouped per g; use conv_weight[g, 0, k]
    for t_start in range(0, S, BLOCK_T):
        t_offsets = t_start + tl.arange(0, BLOCK_T)
        t_mask = t_offsets < S

        acc = tl.zeros((BLOCK_T,), dtype=tl.float32)

        # k loop: K=4
        for k in range(0, K):
            pos = t_offsets + k - 1  # causal: output at t depends on input at t-k+1
            valid = (pos >= 0) & (pos < S) & t_mask

            # Address for Bx[b, g, pos] = x[b, pos, g] via transposed view -> Bx_ptr layout (B, H, S)
            bx_ptrs = Bx_ptr + b * (H * S) + g * S + pos
            bx_vals = tl.load(bx_ptrs, mask=valid, other=0.0).to(tl.float32)

            # conv_weight[g, 0, k]
            w_val = tl.load(W_ptr + g * (1 * K) + 0 * K + k).to(tl.float32)
            acc += bx_vals * w_val

        acc += bias
        out_ptrs = Out_base + t_offsets
        tl.store(out_ptrs, acc, mask=t_mask)
