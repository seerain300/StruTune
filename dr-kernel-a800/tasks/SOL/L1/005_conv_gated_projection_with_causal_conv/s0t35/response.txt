import torch
import triton
import triton.language as tl

# Triton kernel 1: in_proj
# Input x: (B, S, H), in_proj_weight: (M, H), in_proj_bias: (M,)
# Output BCx: (B, S, M), where M=3*H
@triton.jit
def in_proj_kernel(
    x_ptr,           # *float32, shape (B, S, H)
    weight_ptr,      # *float32, shape (M, H)
    bias_ptr,        # *float32, shape (M,)
    out_ptr,         # *float32, shape (B, S, M)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr, M: tl.constexpr,
    BLOCK_M: tl.constexpr
):
    # Grid: (B, S, ceil_div(M, BLOCK_M))
    b = tl.program_id(0)
    s = tl.program_id(1)
    m_block = tl.program_id(2)

    m_offsets = m_block * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = m_offsets < M

    # Accumulator for this (b, s, m_tile)
    acc = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Loop over H dimension (compile-time constant)
    for h in range(0, H):
        # Load x[b, s, h] as a scalar
        x_val = tl.load(x_ptr + b * (S * H) + s * H + h, mask=True, other=0.0)  # scalar float32
        # Load weight[m, h] vector for current h across m tile
        w_ptr = weight_ptr + m_offsets * H + h
        w_val = tl.load(w_ptr, mask=mask_m, other=0.0)  # [BLOCK_M] float32
        # Fused multiply-add
        acc += x_val * w_val  # broadcast x_val across the tile

    # Add bias
    b_val = tl.load(bias_ptr + m_offsets, mask=mask_m, other=0.0)
    acc += b_val

    # Store to out[b, s, m]
    out_ptr_block = out_ptr + b * (S * M) + s * M + m_offsets
    tl.store(out_ptr_block, acc, mask=mask_m)


# Triton kernel 2: elementwise gating y = A * B
# Inputs A: (B, S, H), B: (B, S, H), output Y: (B, S, H)
@triton.jit
def gate_mul_kernel(
    A_ptr, B_ptr, Out_ptr,
    Bsz: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    grid = (Bsz, S, triton.cdiv(H, BLOCK_H))
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_block = tl.program_id(2)
    h_offsets = h_block * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H

    A_row_ptr = A_ptr + b * (S * H) + s * H
    B_row_ptr = B_ptr + b * (S * H) + s * H
    Out_row_ptr = Out_ptr + b * (S * H) + s * H

    A_vals = tl.load(A_row_ptr + h_offsets, mask=mask_h, other=0.0)
    B_vals = tl.load(B_row_ptr + h_offsets, mask=mask_h, other=0.0)
    Out_vals = A_vals * B_vals
    tl.store(Out_row_ptr + h_offsets, Out_vals, mask=mask_h)


# Triton kernel 3: out_proj (matmul-like) computes output[b, s, h] = sum_{h2} y[b, s, h2] * out_w[h2, h] + out_bias[h]
# Inputs y: (B, S, H), out_proj_weight: (H, H), out_proj_bias: (H,)
# Output out: (B, S, H)
@triton.jit
def out_proj_kernel(
    y_ptr,           # *float32, shape (B, S, H)
    out_w_ptr,       # *float32, shape (H, H)
    out_b_ptr,       # *float32, shape (H,)
    out_ptr,         # *float32, shape (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    grid = (B, S, triton.cdiv(H, BLOCK_H))
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_block = tl.program_id(2)

    h_offsets = h_block * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H

    # Accumulator for output across h2
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # Loop over h2 (compile-time constant H)
    for h2 in range(0, H):
        # Load y[b, s, h2]
        y_val = tl.load(y_ptr + b * (S * H) + s * H + h2, mask=True, other=0.0)  # scalar
        # Load out_w[h2, h_offsets]
        out_w_row = out_w_ptr + h2 * H + h_offsets  # vector over h_offsets
        w_vals = tl.load(out_w_row, mask=mask_h, other=0.0)
        acc += y_val * w_vals

    # Add bias
    bias_vals = tl.load(out_b_ptr + h_offsets, mask=mask_h, other=0.0)
    acc += bias_vals

    # Store output[b, s, h]
    out_row_ptr = out_ptr + b * (S * H) + s * H
    tl.store(out_row_ptr + h_offsets, acc, mask=mask_h)

                conv_weight: torch.Tensor, conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor, out_proj_bias: torch.Tensor):

        in_proj_weight: (3H, H), float32
        in_proj_bias: (3H,), float32
        conv_weight: (H, 1, 4), float32
        conv_bias: (H,), float32
        out_proj_weight: (H, H), float32
        out_proj_bias: (H,), float32
        Returns: (B, S, H), float32
        """

        device = x.device
        assert x.dtype == torch.float32, "x must be float32"
        assert in_proj_weight.dtype == torch.float32, "in_proj_weight must be float32"
        assert in_proj_bias.dtype == torch.float32, "in_proj_bias must be float32"
        assert conv_weight.dtype == torch.float32, "conv_weight must be float32"
        assert conv_bias.dtype == torch.float32, "conv_bias must be float32"
        assert out_proj_weight.dtype == torch.float32, "out_proj_weight must be float32"
        assert out_proj_bias.dtype == torch.float32, "out_proj_bias must be float32"

        B, S, H = x.shape
        M = 3 * H  # since in_proj_weight is (3H, H)

        # 1) Triton in_proj: BCx = x @ in_proj_weight^T + bias, output: (B, S, M)
        BCx = torch.empty((B, S, M), device=device, dtype=torch.float32)

        BLOCK_M = 128  # tile along M
        grid_in = (B, S, triton.cdiv(M, BLOCK_M))
        in_proj_kernel[grid_in](
            x.contiguous(), in_proj_weight.contiguous(), in_proj_bias.contiguous(),
            BCx,
            B=B, S=S, H=H, M=M, BLOCK_M=BLOCK_M,
            num_warps=4, num_stages=2
        )

        # 2) Split BCx along last dim into B, C, x_proj, each (B, S, H)
        # PyTorch indexing to get correct slices
        B_t = BCx[:, :, :H]
        C_t = BCx[:, :, H:2*H]
        x_proj = BCx[:, :, 2*H:]

        # 3) Triton gating: Bx = B_t * x_proj
        Bx = torch.empty((B, S, H), device=device, dtype=torch.float32)
        grid_gate = (B, S, triton.cdiv(H, 64))
        gate_mul_kernel[grid_gate](
            B_t.contiguous(), x_proj.contiguous(), Bx,
            B=B, S=S, H=H, BLOCK_H=64,
            num_warps=2, num_stages=2
        )

        # 4) Pad for causal conv: left-pad by K-1 = 3
        # PyTorch F.pad: (pad_left, pad_right) along the last dimension (sequence)
        Bx_padded = torch.nn.functional.pad(Bx, (3, 0))  # becomes (B, S+3, H)

        # 5) Conv1d with groups=H and kernel_size=4. This must match original exactly.
        # conv_weight: (H, 1, 4), conv_bias: (H,)
        conv_out = torch.nn.functional.conv1d(
            Bx_padded, conv_weight, conv_bias, stride=1, padding=0, dilation=1, groups=H
        )  # shape: (B, H, S)

        # 6) Output gating: y = C_t * conv_out[:, :, :S]
        conv_out_S = conv_out[:, :, :S]  # (B, H, S)
        y = C_t * conv_out_S  # elementwise multiply; y shape (B, H, S)

        # 7) Transpose back to (B, S, H) for final projection
        y_T = y.transpose(-1, -2).contiguous()  # (B, S, H), float32

        # 8) Triton out_proj: y_T -> output (B, S, H)
        output = torch.empty((B, S, H), device=device, dtype=torch.float32)

        BLOCK_H = 64
        grid_out = (B, S, triton.cdiv(H, BLOCK_H))
        out_proj_kernel[grid_out](
            y_T.contiguous(), out_proj_weight.contiguous(), out_proj_bias.contiguous(),
            output,
            B=B, S=S, H=H, BLOCK_H=BLOCK_H,
            num_warps=2, num_stages=2
        )

        return output

# Inputs:
#   x_ptr: (B, S, H)

#   out_ptr: (B, S, M), float32

    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr, M: tl.constexpr,
