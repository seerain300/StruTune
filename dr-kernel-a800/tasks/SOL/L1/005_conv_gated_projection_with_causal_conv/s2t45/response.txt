import torch
import triton
import triton.language as tl

# 1) Triple linear projection: x[B, S, H] -> three outputs (B,S,H): B, C, X
#    Each uses weight (H,H) and bias (H): out[b, s, h] = sum_k x[b, s, k] * weight[h, k] + bias[h]
@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,         # *f32, (B, S, H)
    W_ptr,         # *f32, (H, H) for one of the three groups
    b_ptr,         # *f32, (H,)
    out_ptr,       # *f32, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    x_stride0, x_stride1, x_stride2,
    out_stride0, out_stride1, out_stride2,
    W_stride0, W_stride1,
    BLOCK_K: tl.constexpr
):
    b_idx = tl.program_id(0)  # batch index
    h_idx = tl.program_id(1)  # hidden index
    tile_s = tl.program_id(2)  # tile index along sequence

    s_offsets = tile_s * 128 + tl.arange(0, 128)
    mask_s = s_offsets < S

    # Accumulator for 128 positions in the sequence
    acc = tl.zeros((128,), dtype=tl.float32)

    # Loop over K dimension (hidden input features) in chunks
    for k0 in range(0, H, BLOCK_K):
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < H

        # Load x[b, s, k] for all s in this tile and k in chunk
        x_ptrs = x_ptr + b_idx * x_stride0 + s_offsets[:, None] * x_stride1 + k_range[None, :] * x_stride2
        x_vals = tl.load(x_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)

        # Load weight row W[h, k]
        w_ptrs = W_ptr + h_idx * W_stride0 + k_range * W_stride1
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)  # shape (BLOCK_K,)

        # Accumulate: sum over K chunk
        # x_vals is (128, BLOCK_K), w_vals is (BLOCK_K,)
        # We need to dot-reduce along K: acc += sum_k x_vals[:, k] * w_vals[k]
        # Do it by iterating over BLOCK_K:
        for kk in range(BLOCK_K):
            k_valid = (k0 + kk) < H
            # Pick the kk-th column: x[:, kk] * w[kk]
            # Mask handling: if k_valid is False, contribution is zero
            contrib = tl.where(k_valid, x_vals[:, kk] * w_vals[kk], 0.0)
            acc += contrib

    # Add bias
    bias_val = tl.load(b_ptr + h_idx)
    acc += bias_val

    # Store results to out[b, s, h] for this tile
    out_ptrs = out_ptr + b_idx * out_stride0 + s_offsets * out_stride1 + h_idx * out_stride2
    tl.store(out_ptrs, acc, mask=mask_s)


# 2) Element-wise gating: out = B * X, tensors of shape (B, S, H)
@triton.jit
def elemwise_mul_bsh_kernel(
    B_ptr, X_ptr, out_ptr,
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    B_stride0, B_stride1, B_stride2,
    X_stride0, X_stride1, X_stride2,
    out_stride0, out_stride1, out_stride2,
    BLOCK_S: tl.constexpr
):
    b_idx = tl.program_id(0)
    h_idx = tl.program_id(1)
    tile_s = tl.program_id(2)
    s_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    B_ptrs = B_ptr + b_idx * B_stride0 + s_offsets * B_stride1 + h_idx * B_stride2
    X_ptrs = X_ptr + b_idx * X_stride0 + s_offsets * X_stride1 + h_idx * X_stride2
    out_ptrs = out_ptr + b_idx * out_stride0 + s_offsets * out_stride2 + h_idx * out_stride1  # note strides

    B_vals = tl.load(B_ptrs, mask=mask_s, other=0.0)
    X_vals = tl.load(X_ptrs, mask=mask_s, other=0.0)
    tl.store(out_ptrs, B_vals * X_vals, mask=mask_s)


# 3) Grouped causal 1D convolution:
#    Input Bx: (B, S, H), convW: (H, H, 4), convB: (H,)
#    Output conv_out: (B, H, S)
#    For each (b, c, t): conv_out[b, c, t] = sum_{k=0..3} Bx[b, c, t+k-1] * convW[c, c, k] + convB[c]
#    Note left padding of 3 for kernel_size=4 (causal).
@triton.jit
def grouped_causal_conv1d_bsh_to_bhs_kernel(
    Bx_ptr,         # *f32, (B, S, H)
    convW_ptr,      # *f32, (H, H, 4)
    convB_ptr,      # *f32, (H,)
    out_ptr,        # *f32, (B, H, S)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    Bx_stride0, Bx_stride1, Bx_stride2,
    convW_stride0, convW_stride1, convW_stride2,
    out_stride0, out_stride1, out_stride2,
    BLOCK_S: tl.constexpr
):
    b_idx = tl.program_id(0)   # batch
    c_idx = tl.program_id(1)   # channel
    tile_s = tl.program_id(2)  # tile along sequence

    t_offsets = tile_s * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_t = t_offsets < S

    # Accumulator for conv outputs over the tile
    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # Compute contributions for k in [0..3]
    for k in range(4):
        s_idx = t_offsets + k - 1  # left padding by 3 for kernel_size=4
        valid = s_idx >= 0
        # Load Bx[b, c, s_idx]
        Bx_ptrs = Bx_ptr + b_idx * Bx_stride0 + s_idx * Bx_stride1 + c_idx * Bx_stride2
        Bx_vals = tl.load(Bx_ptrs, mask=mask_t & valid, other=0.0)

        # Load conv weight convW[c, c, k]
        convW_val = tl.load(convW_ptr + c_idx * convW_stride0 + c_idx * convW_stride1 + k * convW_stride2)

        acc += Bx_vals * convW_val

    # Add bias
    bias_val = tl.load(convB_ptr + c_idx)
    acc += bias_val

    # Store to conv_out[b, c, t]
    out_ptrs = out_ptr + b_idx * out_stride0 + c_idx * out_stride1 + t_offsets * out_stride2
    tl.store(out_ptrs, acc, mask=mask_t)


# 4) Final linear projection: y[B,H,S] -> out[B,S,H]
#    Implements out[b, s, h_out] = sum_{h_in=0..H-1} y[b, h_out, s] * out_proj_weight[h_out, h_in] + out_proj_bias[h_out]
@triton.jit
def final_linear_bsh_to_bsh_kernel(
    y_ptr,          # *f32, (B, H, S)
    out_proj_w_ptr, # *f32, (H, H)
    out_proj_b_ptr, # *f32, (H,)
    out_ptr,        # *f32, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    y_stride0, y_stride1, y_stride2,
    out_proj_w_stride0, out_proj_w_stride1,
    out_stride0, out_stride1, out_stride2,
    BLOCK_H_IN: tl.constexpr
):
    b_idx = tl.program_id(0)
    s_idx = tl.program_id(1)
    h_out = tl.program_id(2)

    # Accumulate over H_in in chunks
    acc = tl.zeros((), dtype=tl.float32)
    for h_in0 in range(0, H, BLOCK_H_IN):
        h_in = h_in0 + tl.arange(0, BLOCK_H_IN)
        mask_h_in = h_in < H

        # Load y[b, h_out, s] (scalar per h_out) and vector y[b, h_in, s]
        y_scalar = tl.load(y_ptr + b_idx * y_stride0 + h_out * y_stride1 + s_idx * y_stride2)
        y_vec_ptrs = y_ptr + b_idx * y_stride0 + h_in * y_stride1 + s_idx * y_stride2
        y_vec = tl.load(y_vec_ptrs, mask=mask_h_in, other=0.0)

        # Load out_proj_weight[h_out, h_in]
        w_ptrs = out_proj_w_ptr + h_out * out_proj_w_stride0 + h_in * out_proj_w_stride1
        w_vec = tl.load(w_ptrs, mask=mask_h_in, other=0.0)

        # Accumulate: y_scalar * sum(h_in) w_vec[h_in] * y_vec[h_in]
        # Since y_scalar is scalar and w_vec*y_vec is vector, we need to reduce over h_in:
        # sum over valid h_in
        # Note: This is a simple reduction loop; Triton supports Python range with constexpr sizes.
        for kk in range(BLOCK_H_IN):
            hi_valid = (h_in0 + kk) < H
            contrib = tl.where(hi_valid, y_scalar * (w_vec[kk] * y_vec[kk]), 0.0)
            acc += contrib

    # Add bias
    bias_val = tl.load(out_proj_b_ptr + h_out)
    acc += bias_val

    # Store to out[b, s, h_out]
    out_ptr_store = out_ptr + b_idx * out_stride0 + s_idx * out_stride2 + h_out * out_stride1
    tl.store(out_ptr_store, acc)


class ModelNew(torch.nn.Module):

        super().__init__()

        4) Output gating y = C * conv_out
        5) Final linear projection to output (B, S, H)
        """

        device = x.device

  4) FinalLinearBShToBShKernel: implement y (B,H,S) -> output (B,S,H) via out_proj_weight (H,H) and out_proj_bias (H). This is straightforward matmul-like reduction over H.

Performance considerations
- These kernels are simple and correctness-focused. They tile along sequence length S with BLOCK_S=128 and reduce over hidden dimension H in chunks. While not as optimized as fused kernels, they ensure correctness. Once validated, further optimization (tl.dot, shared memory tiling, unrolling) can be added.

import triton
import triton.language as tl

# 1) Triple linear projection: x[B, S, H] -> three outputs (B,S,H): B, C, X
#    Each uses weight (H,H) and bias (H): out[b, s, h] = sum_k x[b, s, k] * weight[h, k] + bias[h]
@triton.jit
def triple_linear_bsh_kernel(
    x_ptr,         # *f32, (B, S, H)
    W_ptr,         # *f32, (H, H) for one of the three groups
    b_ptr,         # *f32, (H,)
    out_ptr,       # *f32, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    x_stride0, x_stride1, x_stride2,
    out_stride0, out_stride1, out_stride2,
    W_stride0, W_stride1,
    BLOCK_K: tl.constexpr
):
    b_idx = tl.program_id(0)  # batch index
    h_idx = tl.program_id(1)  # hidden index
    tile_s = tl.program_id(2)  # tile index along sequence

    s_offsets = tile_s * 128 + tl.arange(0, 128)
    mask_s = s_offsets < S

    acc = tl.zeros((128,), dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < H

        x_ptrs = x_ptr + b_idx * x_stride0 + s_offsets[:, None] * x_stride1 + k_range[None, :] * x_stride2
        x_vals = tl.load(x_ptrs, mask=mask_s[:, None] & mask_k[None, :], other=0.0)  # (128, BLOCK_K)

        w_ptrs = W_ptr + h_idx * W_stride0 + k_range * W_stride1
        w_vals = tl.load(w_ptrs, mask=mask_k, other=0.0)  # (BLOCK_K,)

        # Reduce over K-chunk
        for kk in range(BLOCK_K):
            k_valid = (k0 + kk) < H
            contrib = tl.where(k_valid, x_vals[:, kk] * w_vals[kk], 0.0)
            acc += contrib

    bias_val = tl.load(b_ptr + h_idx)
    acc += bias_val

    out_ptrs = out_ptr + b_idx * out_stride0 + s_offsets * out_stride1 + h_idx * out_stride2
    tl.store(out_ptrs, acc, mask=mask_s)
