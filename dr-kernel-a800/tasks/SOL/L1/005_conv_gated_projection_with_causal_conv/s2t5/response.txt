    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    for k in range(0, 4):

    conv_out_ptrs = conv_out_ptr + b * conv_out_stride_b + c * conv_out_stride_c + s_offsets * conv_out_stride_s

@triton.jit
def final_linear_gemv_bsh_kernel(
    y_ptr,            # *T, (B, S, H)
    wy_ptr,           # *T, (H, H)
    bb_ptr,           # *T, (H,)
    out_ptr,          # *T, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    y_stride_b, y_stride_s, y_stride_h,
    wy_stride_w, wy_stride_h,
    out_stride_b, out_stride_s, out_stride_h,
    K_BLOCK: tl.constexpr,
):
    # Grid: (B, S, H) — each program computes one element (b, s, h)
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_out = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    # GEMV over K (H) in tiles of K_BLOCK
    for k in range(0, H, K_BLOCK):
        k_offsets = k + tl.arange(0, K_BLOCK)
        mask_k = k_offsets < H
        # y[b, s, k_offsets]
        y_ptrs = y_ptr + b * y_stride_b + s * y_stride_s + k_offsets * y_stride_h
        y_vals = tl.load(y_ptrs, mask=mask_k, other=0.0)  # (K_BLOCK,)
        # wy[h_out, k_offsets]
        wy_ptrs = wy_ptr + h_out * wy_stride_w + k_offsets * wy_stride_h
        wy_vals = tl.load(wy_ptrs, mask=mask_k, other=0.0)
        # accumulate dot product
        for i in range(0, K_BLOCK):
            # safe multiply-accumulate
            acc += y_vals[i] * wy_vals[i]

    # add bias
    b_bias = tl.load(bb_ptr + h_out * wy_stride_h)  # bias for h_out
    out_val = acc + b_bias

    # store to out[b, s, h_out]
    out_ptr_hs = out_ptr + b * out_stride_b + s * out_stride_s + h_out * out_stride_h
    tl.store(out_ptr_hs, out_val)

    b2_val = tl.load(b2_ptr + h_out * b2_stride)
    acc += b2_val
    X_ptrs = X_out_ptr + b * X_stride_b + s_offsets * X_stride_s + h_out * X_stride_h
    tl.store(X_ptrs, acc, mask=mask_s)


# Kernel 2: Grouped causal 1D convolution, depthwise with kernel_size=4
# Input Bx is (B, H, S) pre-split; conv_weight is (H, H, 4), groups=H
@triton.jit
def grouped_causal_conv1d_kernel(
    Bx_ptr,          # *T, (B, H, S)
    conv_w_ptr,      # *T, (H, H, 4)
    conv_b_ptr,      # *T, (H,)
    conv_out_ptr,    # *T, (B, H, S)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    Bx_stride_b, Bx_stride_c, Bx_stride_s,
    conv_w_stride_c, conv_w_stride_k,
    conv_out_stride_b, conv_out_stride_c, conv_out_stride_s,
    BLOCK_S: tl.constexpr,
):
    # Grid: (B, H, ceil_div(S, BLOCK_S))
    b = tl.program_id(0)
    c = tl.program_id(1)
    s_block = tl.program_id(2)

    s_offsets = s_block * BLOCK_S + tl.arange(0, BLOCK_S)
    mask_s = s_offsets < S

    acc = tl.zeros((BLOCK_S,), dtype=tl.float32)

    # Loop over kernel taps k=0..3 (kernel_size=4)
    for k in range(0, 4):
        in_s = s_offsets + (k - 1)  # causal: shift by (k-1)
        # Mask for valid input indices
        mask_in = (in_s >= 0) & (in_s < S) & mask_s
        # Load Bx[b, c, in_s]
        Bx_ptrs = Bx_ptr + b * Bx_stride_b + c * Bx_stride_c + in_s * Bx_stride_s
        Bx_vals = tl.load(Bx_ptrs, mask=mask_in, other=0.0)  # (BLOCK_S,)
        # Load conv_weight[c, c, k] as scalar
        conv_w_val = tl.load(conv_w_ptr + c * conv_w_stride_c + c * conv_w_stride_k + k * conv_w_stride_k)
        # Accumulate
        acc += Bx_vals * conv_w_val

    # Add bias
    conv_b_val = tl.load(conv_b_ptr + c * conv_b_stride)  # conv_b_stride should be 1 (assumed contiguous)
    acc += conv_b_val

    # Store conv_out[b, c, s_offsets]
    conv_out_ptrs = conv_out_ptr + b * conv_out_stride_b + c * conv_out_stride_c + s_offsets * conv_out_stride_s
    tl.store(conv_out_ptrs, acc, mask=mask_s)


# Kernel 3: Final linear projection: output[b, s, h] = sum_h' y[b, s, h'] * out_proj_weight[h', h] + bias[h]
@triton.jit
def final_linear_gemv_bsh_kernel(
    y_ptr,            # *T, (B, S, H)
    wy_ptr,           # *T, (H, H)
    bb_ptr,           # *T, (H,)
    out_ptr,          # *T, (B, S, H)
    B: tl.constexpr, S: tl.constexpr, H: tl.constexpr,
    y_stride_b, y_stride_s, y_stride_h,
    wy_stride_w, wy_stride_h,
    out_stride_b, out_stride_s, out_stride_h,
    K_BLOCK: tl.constexpr,
):
    # Grid: (B, S, H) — each program computes one element (b, s, h)
    b = tl.program_id(0)
    s = tl.program_id(1)
    h_out = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    # GEMV over K (H) in tiles of K_BLOCK
    for k in range(0, H, K_BLOCK):
        k_offsets = k + tl.arange(0, K_BLOCK)
        mask_k = k_offsets < H
        # y[b, s, k_offsets]
        y_ptrs = y_ptr + b * y_stride_b + s * y_stride_s + k_offsets * y_stride_h
        y_vals = tl.load(y_ptrs, mask=mask_k, other=0.0)  # (K_BLOCK,)
        # wy[h_out, k_offsets]
        wy_ptrs = wy_ptr + h_out * wy_stride_w + k_offsets * wy_stride_h
        wy_vals = tl.load(wy_ptrs, mask=mask_k, other=0.0)
        # accumulate dot product
        for i in range(0, K_BLOCK):
            acc += y_vals[i] * wy_vals[i]

    # add bias
    b_bias = tl.load(bb_ptr + h_out * wy_stride_h)  # bias for h_out
    out_val = acc + b_bias

    # store to out[b, s, h_out]
    out_ptr_hs = out_ptr + b * out_stride_b + s * out_stride_s + h_out * out_stride_h
    tl.store(out_ptr_hs, out_val)


# Elementwise multiply kernels (kept in Triton to avoid torch ops)
@triton.jit
def elementwise_mul_bsx_kernel(B_ptr, X_ptr, out_ptr, B, S, H, stride_b, stride_s, stride_h):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)
    B_val = tl.load(B_ptr + b * stride_b + s * stride_s + h * stride_h)
    X_val = tl.load(X_ptr + b * stride_b + s * stride_s + h * stride_h)
    out_val = B_val * X_val
    tl.store(out_ptr + b * stride_b + s * stride_s + h * stride_h, out_val)

@triton.jit
def elementwise_mul_bsh_kernel(C_ptr, conv_ptr, out_ptr, B, S, H, stride_b, stride_s, stride_h):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)
    C_val = tl.load(C_ptr + b * stride_b + s * stride_s + h * stride_h)
    conv_val = tl.load(conv_ptr + b * stride_b + s * stride_s + h * stride_h)
    out_val = C_val * conv_val
    tl.store(out_ptr + b * stride_b + s * stride_s + h * stride_h, out_val)
