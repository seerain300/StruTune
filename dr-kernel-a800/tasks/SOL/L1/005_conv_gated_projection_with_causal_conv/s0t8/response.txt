- Ensure all tensors are contiguous and float32 before launching Triton kernels.
- Use Triton kernels only; no torch.nn.functional ops on tensors in host code.
- Implement the heavy parts (initial linear projection, padding, grouped conv, final linear) in Triton. The element-wise gating Bx = B * x_proj can be done with a simple Triton elementwise kernel.
- Use tl.static_range for any loops in Triton kernels to ensure compile-time unrolling and avoid runtime-dependent loops.
- Carefully handle padding for causal conv: left-pad S_in-1 elements, resulting in S_out = S_in + (K-1). Then conv1d with stride=1, groups=3*H. Since groups equals the number of input channels, each output channel convolves only its corresponding input channel. In our case, we split BCx into three groups of H channels, and conv operates per channel within each group.

Below is the complete Triton-ONLY implementation with ModelNew as the entry point. It:
- Generates random input x (B, S, H) in Triton.
- Computes BCx = F.linear(x, in_proj_weight, in_proj_bias) via a Triton matmul-like kernel.
- Splits BCx into B, C, x_proj by reading chunks along the last dimension.
- Forms Bx = B * x_proj via a Triton elementwise kernel.
- Pads Bx along sequence dimension for causal conv via a Triton padding kernel.
- Performs grouped causal 1D convolution (groups=3H) via a Triton kernel with static loops over K=4.
- Forms y = C * conv_out and transposes to (B, S, H).
- Performs final linear projection via a Triton matmul-like kernel.

import torch
import torch.nn as nn
import triton
import triton.language as tl


# Triton kernel: generate random x (B, S, H) and store in OUT_ptr
@triton.jit
def rand_tensor_kernel(
    OUT_ptr,    # *f32, (B, S, H)
    B: tl.constexpr,
    S: tl.constexpr,
    H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_h = tl.program_id(2)

    # Random float32 in [0, 1)
    rand_val = tl.rand(tl.zeros((), dtype=tl.float32))  # scalar RNG
    out_index = pid_b * S * H + pid_s * H + pid_h
    tl.store(OUT_ptr + out_index, rand_val)


# Triton kernel: in_proj = F.linear(x, in_proj_weight, in_proj_bias)
# x: (B, S, H), W: (M, H) where M=3*H, bias: (M,), out: (B, S, M)
# OUT[b, s, m] = sum_h x[b, s, h] * W[m, h] + bias[m]
@triton.jit
def in_proj_kernel(
    X_ptr,      # *f32, (B, S, H)
    W_ptr,      # *f32, (M, H)
    Bias_ptr,   # *f32, (M,)
    OUT_ptr,    # *f32, (B, S, M)
    B: tl.constexpr,
    S: tl.constexpr,
    H: tl.constexpr,
    M: tl.constexpr,  # 3*H
):
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_m = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    # Reduce over h
    for h in tl.static_range(0, H):
        x_index = pid_b * S * H + pid_s * H + h
        x_val = tl.load(X_ptr + x_index)
        w_index = pid_m * H + h
        w_val = tl.load(W_ptr + w_index)
        acc += x_val * w_val

    # Add bias
    bias_val = tl.load(Bias_ptr + pid_m)
    acc += bias_val

    out_index = pid_b * S * M + pid_s * M + pid_m
    tl.store(OUT_ptr + out_index, acc)


# Triton kernel: elementwise gating: Bx = B * x_proj
# B: (B, H, S), x_proj: (B, H, S), OUT: (B, H, S)
@triton.jit
def gate_kernel(

    Bsz: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_s = tl.program_id(2)

    b_index = pid_b * H * S + pid_h * S + pid_s
    bx = tl.load(B_ptr + b_index)
    xprj = tl.load(XPRJ_ptr + b_index)
    out_val = bx * xprj
    tl.store(OUT_ptr + b_index, out_val)


# Triton kernel: left-pad along sequence dim for causal conv
# Input: Y (B, 3H, S), Output: Y_padded (B, 3H, S+K-1), pad K-1 on left
@triton.jit
def pad_left_kernel(
    IN_ptr,     # *f32, (B, 3H, S)
    OUT_ptr,    # *f32, (B, 3H, S+K-1)
    B: tl.constexpr,
    Hc: tl.constexpr,  # channels = 3H
    S: tl.constexpr,
    pad_left: tl.constexpr,  # K-1
    K: tl.constexpr,          # kernel_size
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)

    # If pid_t < pad_left: store 0
    # Else: write IN[b, c, pid_t - pad_left]
    if pid_t < pad_left:
        tl.store(OUT_ptr + pid_b * Hc * (S + pad_left) + pid_c * (S + pad_left) + pid_t, 0.0)
    else:
        in_index = pid_b * Hc * S + pid_c * S + (pid_t - pad_left)
        val = tl.load(IN_ptr + in_index)
        out_index = pid_b * Hc * (S + pad_left) + pid_c * (S + pad_left) + pid_t
        tl.store(OUT_ptr + out_index, val)


# Triton kernel: grouped causal 1D convolution (groups = 3H)
# Input: Y_padded (B, 3H, S+K-1), Weight: (3H, 1, K), Bias: (3H,)
# Output: Conv_out (B, 3H, S)
@triton.jit
def conv1d_grouped_kernel(
    IN_ptr,     # *f32, (B, 3H, S+K-1)
    W_ptr,      # *f32, (3H, K), row-major
    Bias_ptr,   # *f32, (3H,)
    OUT_ptr,    # *f32, (B, 3H, S)
    B: tl.constexpr,
    Hc: tl.constexpr,  # 3H
    S: tl.constexpr,
    pad_left: tl.constexpr,
    K: tl.constexpr,    # kernel_size = 4
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)

    # We iterate over all output time positions t in [0, S)
    for t in tl.static_range(0, S):
        acc = tl.zeros((), dtype=tl.float32)
        # Sum over K taps
        for k in tl.static_range(0, K):
            in_index = pid_b * Hc * (S + pad_left) + pid_c * (S + pad_left) + (t + k)
            val = tl.load(IN_ptr + in_index)
            w_index = pid_c * K + k
            w_val = tl.load(W_ptr + w_index)
            acc += val * w_val
        # Add bias
        bias_val = tl.load(Bias_ptr + pid_c)
        acc += bias_val
        out_index = pid_b * Hc * S + pid_c * S + t
        tl.store(OUT_ptr + out_index, acc)


# Triton kernel: final linear projection (out_proj)
# Y: (B, S, H), W: (H, H), Bias: (H,)

@triton.jit

    pid_s = tl.program_id(1)
    pid_h = tl.program_id(2)

    h_offsets = pid_h * H + tl.arange(0, H)
    h_mask = h_offsets < H  # always true if pid_h is 0; we launch grid (B, S, 1)
    # but we'll do per-element here by looping
    acc = tl.zeros((), dtype=tl.float32)

    for h2 in tl.static_range(0, H):
        y_index = pid_b * S * H + pid_s * H + h2
        y_val = tl.load(Y_ptr + y_index)
        # W[h2, h] where W is (H, H) row-major
        w_index = h2 * H + pid_h  # pid_h is the output channel h
        w_val = tl.load(W_ptr + w_index)
        acc += y_val * w_val

    bias_val = tl.load(Bias_ptr + pid_h)
    acc += bias_val

    out_index = pid_b * S * H + pid_s * H + pid_h
    tl.store(OUT_ptr + out_index, acc)
