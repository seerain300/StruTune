import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    encoder_ptr,      # *f32, [B, Stext, H]
    hidden_ptr,       # *f32, [B, Simg, H]
    out_ptr,          # *f32, [B, T, H], T = Stext + Simg
    B, Stext, Simg, H,
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_s, stride_h_h,
    stride_o_b, stride_o_t, stride_o_h,
    BLOCK_H: tl.constexpr,
):
    # 2D grid over (B, T)
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Determine source: if t < Stext, take from encoder; else take from hidden at s = t - Stext
    h = tl.arange(0, BLOCK_H)
    if t < Stext:
        src = encoder_ptr + b * stride_e_b + t * stride_e_t + h * stride_e_h
        dst = out_ptr + b * stride_o_b + t * stride_o_t + h * stride_o_h
    else:
        s = t - Stext
        src = hidden_ptr + b * stride_h_b + s * stride_h_s + h * stride_h_h
        dst = out_ptr + b * stride_o_b + t * stride_o_t + h * stride_o_h

    # Mask to avoid OOB when H is not multiple of BLOCK_H (not needed here since H is actual size)
    # But keep it safe:
    mask = h < H
    vals = tl.load(src, mask=mask, other=0.0)
    tl.store(dst, vals, mask=mask)


@triton.jit
def matvec_row_kernel(
    in_ptr,           # *f32, concatenated [B, T, H]
    weightT_ptr,      # *f32, [H, H] (transpose of process_weight)
    out_ptr,          # *f32, [B, T, H]
    B, T, H,
    stride_i_b, stride_i_t, stride_i_h,
    stride_w_k, stride_w_n,   # weight_T strides: dim-0 is k, dim-1 is n
    stride_o_b, stride_o_t, stride_o_h,
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Output vector accumulator for this (b, t)
    out_row = tl.zeros([H], dtype=tl.float32)

    # Loop over input dimension k
    for k in range(0, H):
        # Load input vector element for this (b, t, k)
        in_ptr_k = in_ptr + b * stride_i_b + t * stride_i_t + k * stride_i_h
        x = tl.load(in_ptr_k)  # scalar load

        # Load weight_T[k, :] vector of length H
        w_ptrs = weightT_ptr + k * stride_w_k + tl.arange(0, H) * stride_w_n
        w = tl.load(w_ptrs)

        # Accumulate dot product for this k across H
        out_row += x * w

    # Store the resulting vector to out[b, t, :]
    out_dst = out_ptr + b * stride_o_b + t * stride_o_t + tl.arange(0, H) * stride_o_h
    tl.store(out_dst, out_row)

        # Ensure on same device and dtype, float32
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "All tensors must be on CUDA device."

        H = hidden_states.shape[2]

            num_stages=1,
        )

            processed.stride(0), processed.stride(1), processed.stride(2),
            num_warps=1,
            num_stages=1,
        )
