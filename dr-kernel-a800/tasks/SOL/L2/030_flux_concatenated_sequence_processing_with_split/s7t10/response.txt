import torch
import triton
import triton.language as tl


@triton.jit
def cat_rows_kernel(
    e_ptr,                # *float, pointer to encoder_hidden_states: [B, T, H]
    i_ptr,                # *float, pointer to hidden_states: [B, I, H]
    out_ptr,              # *float, pointer to output X_cat: [B, T+I, H]
    B, T, I, H,           # int32 sizes
    stride_eb, stride_et, stride_eh,  # strides for encoder_hidden_states
    stride_ib, stride_ih, stride_ih2, # strides for hidden_states (note: two ih entries; we only need ih)
    stride_ob, stride_om, stride_oh,  # strides for output
):
    # program ids
    b = tl.program_id(0)
    p = tl.program_id(1)  # row index in [0, T+I)

    # compute which input tensor to use
    use_encoder = p < T

    # pointers for current row
    if use_encoder:
        row_ptr = e_ptr + b * stride_eb + p * stride_et
    else:
        row_ptr = i_ptr + b * stride_ib + (p - T) * stride_ih

    # output pointer for this row
    out_row_ptr = out_ptr + b * stride_ob + p * stride_om

    # copy H elements
    # We assume H is reasonably small and copy via loop. This keeps code simple and correct.
    # Triton's vectorized load/store expects contiguous or structured offsets; this scalar loop is safe.
    for j in range(0, H):
        val = tl.load(row_ptr + j * stride_eh if use_encoder else row_ptr + j * stride_ih2)
        tl.store(out_row_ptr + j * stride_oh, val)

    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        # Ensure CUDA tensors

        H = hidden_states.shape[2]
