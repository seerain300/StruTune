import torch
import triton
import triton.language as tl


@triton.jit
def cat_rows_kernel(
    e_ptr,         # pointer to encoder_hidden_states (shape: [B, T, H])
    i_ptr,         # pointer to hidden_states (shape: [B, I, H])
    out_ptr,       # pointer to output X_cat (shape: [B, M, H], M = T + I)
    B: tl.constexpr,
    T: tl.constexpr,
    I: tl.constexpr,
    H: tl.constexpr,
    stride_eb, stride_et, stride_eh,
    stride_ib, stride_ih, stride_ih2,  # stride_ih, stride_ih2 (unused but kept for signature)
    stride_ob, stride_om, stride_oh,
    BLOCK_H: tl.constexpr,
):
    # program ids: one per (b, p)
    b = tl.program_id(0)
    p = tl.program_id(1)

    # Compute row index within concatenated sequence
    M = T + I

    # Determine source tensor and row index
    is_img = p >= T
    src_row = p - T if is_img else p

    # Compute offsets for loading
    e_off = b * stride_eb + src_row * stride_et
    i_off = b * stride_ib + src_row * stride_ih  # stride_ih unused but keep signature
    o_off = b * stride_ob + p * stride_om

    # Vector of H positions
    h = tl.arange(0, BLOCK_H)
    mask = h < H

    # Load from encoder or image depending on p
    # Note: This branching is resolved at compile time for each program since T/I are constexpr
    vals = tl.load(e_ptr + e_off + h * stride_eh, mask=mask, other=0.0) if not is_img else tl.load(i_ptr + i_off + h * stride_ih, mask=mask, other=0.0)

    # Store to output
    tl.store(out_ptr + o_off + h * stride_oh, vals, mask=mask)

        - Performs linear projection using torch.matmul (no torch ops on the concatenated result).

        """

        B = hidden_states.shape[0]
        T = encoder_hidden_states.shape[1]
        I = hidden_states.shape[1]
        H = hidden_states.shape[2]
        assert encoder_hidden_states.shape[0] == B and encoder_hidden_states.shape[2] == H
        assert process_weight.shape == (H, H), "process_weight must be [H, H]"

        # Ensure contiguous tensors for predictable strides
        e = encoder_hidden_states.contiguous()
        i = hidden_states.contiguous()
        w = process_weight.contiguous()

        # Allocate output concatenated tensor X_cat: [B, M, H], M = T + I
        M = T + I

        grid = (B, M)
        cat_rows_kernel[grid](
