import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    encoder_ptr,      # *f32, [B, Stext, H]
    hidden_ptr,       # *f32, [B, Simg, H]
    out_ptr,          # *f32, [B, T, H], T = Stext + Simg
    B, Stext, Simg, H,
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_s, stride_h_h,
    stride_o_b, stride_o_t, stride_o_h,
):
    # 2D grid over (B, T)
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Determine source: encoder if t < Stext, else hidden at s = t - Stext
    if t < Stext:
        src_t = t
        src_ptr = encoder_ptr + b * stride_e_b + src_t * stride_e_t
    else:
        s = t - Stext
        src_ptr = hidden_ptr + b * stride_h_b + s * stride_h_s

    dst_ptr = out_ptr + b * stride_o_b + t * stride_o_t

    # Copy row-wise across H
    # We loop over H; since H is a runtime value, use a simple loop.
    # For safety, mask not needed; bounds are guaranteed by grid.
    for i in range(0, H):
        val = tl.load(src_ptr + i * stride_e_h)
        tl.store(dst_ptr + i * stride_o_h, val)


@triton.jit
def matvec_row_kernel(
    in_ptr,           # *f32, concatenated [B, T, H_in]
    weightT_ptr,      # *f32, process_weight.T [H_out, H_in]
    out_ptr,          # *f32, [B, T, H_out]
    B, T, H_in, H_out,
    stride_i_b, stride_i_t, stride_i_h,
    stride_w_k, stride_w_n,  # weight_T strides: k (dim-0) and n (dim-1)
    stride_o_b, stride_o_t, stride_o_h,
):
    # 2D grid over (B, T)
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Output vector for this (b, t)
    out_base = out_ptr + b * stride_o_b + t * stride_o_t

    # Compute dot product for each output channel n = 0..H_out-1
    # We loop over H_in to accumulate scalar results; this is robust and correct.
    for n in range(0, H_out):
        acc = tl.zeros([1], dtype=tl.float32)
        # Accumulate dot: in[b, t, k] * weightT[k, n]
        for k in range(0, H_in):
            in_val = tl.load(in_ptr + b * stride_i_b + t * stride_i_t + k * stride_i_h)
            w_val = tl.load(weightT_ptr + k * stride_w_k + n * stride_w_n)
            acc += in_val * w_val
        # Store scalar result
        tl.store(out_base + n * stride_o_h, acc)

    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        """
        Triton-optimized version of the original run() function.
        - Concatenates encoder_hidden_states and hidden_states along the sequence dimension.
        - Applies linear projection using process_weight.T via Triton kernel.
        - Splits back into processed_encoder_hidden_states and processed_hidden.
        """
        # Ensure CUDA and dtype
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "Tensors must be on CUDA device."
        assert hidden_states.dtype == torch.float32 and encoder_hidden_states.dtype == torch.float32 and process_weight.dtype == torch.float32, "Use float32 tensors."

        # Shapes
        B = hidden_states.shape[0]
        Stext = encoder_hidden_states.shape[1]
        Simg = hidden_states.shape[1]
        H_in = hidden_states.shape[2]
        H_out = process_weight.shape[0]  # process_weight is [H_out, H_in]

        # Make inputs contiguous
        encoder = encoder_hidden_states.contiguous()
        hidden = hidden_states.contiguous()
        weight_T = process_weight.t().contiguous()  # [H_out, H_in]

        # 1) Triton concatenation: concatenated [B, T, H_in], T = Stext + Simg
        T = Stext + Simg
        concatenated = torch.empty((B, T, H_in), device=hidden.device, dtype=hidden.dtype)

        # Launch concat kernel: grid (B, T)
        grid_concat = (B, T)
        concat_kernel[grid_concat](
            encoder, hidden, concatenated,
            B, Stext, Simg, H_in,
            encoder.stride(0), encoder.stride(1), encoder.stride(2),
            hidden.stride(0), hidden.stride(1), hidden.stride(2),
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
        )

        # 2) Triton matvec per (b, t): out [B, T, H_out]
        out = torch.empty((B, T, H_out), device=hidden.device, dtype=hidden.dtype)

        # Launch matvec kernel: grid (B, T)
        grid_matvec = (B, T)
        matvec_row_kernel[grid_matvec](
            concatenated, weight_T, out,
            B, T, H_in, H_out,
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            weight_T.stride(0), weight_T.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
        )

        # 3) Split back along sequence dimension
        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

        return processed_encoder, processed_hidden

def matvec_row_kernel(
    in_ptr,           # *f32, concatenated [B, T, H_in]
    weightT_ptr,      # *f32, process_weight.T [H_in, H_out] (i.e., original [H_out, H_in] transposed)
    out_ptr,          # *f32, [B, T, H_out]
    B, T, H_in, H_out,
    stride_i_b, stride_i_t, stride_i_h,
    stride_w_i, stride_w_o,  # weight_T strides: i (dim-0) and o (dim-1)

):
