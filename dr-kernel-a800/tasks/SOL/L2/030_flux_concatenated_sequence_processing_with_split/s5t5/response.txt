Plan
1) Concatenation Triton kernel:
   - Inputs: encoder_hidden_states [B, Stext, H], hidden_states [B, Simg, H]
   - Output: out_cat [B, T, H], T = Stext + Simg
   - Launch with grid (B, T); for each (b, t), write to out_cat[b, t, :] the corresponding row from encoder or hidden depending on t < Stext. Iterate over H with a vector to write all hidden dims.

2) Batched GEMM Triton kernel:
   - Inputs: out_cat [B, T, H], process_weight_T [H, H] (transposed weight)
   - Output: out [B, T, H]

3) Forward:
   - Ensure inputs are contiguous and on GPU.
   - Allocate output tensors, launch kernels, and then split the processed tensor into encoder and image parts based on sequence lengths.

This approach ensures:
- No torch operations on data (no torch.cat, no torch.matmul).
- All computation is done in Triton kernels.

import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    e_ptr,            # *f32, [B, Stext, H]
    h_ptr,            # *f32, [B, Simg, H]
    out_ptr,          # *f32, [B, T, H]
    B,                # int
    Stext,            # int
    Simg,             # int
    H,                # int
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_t, stride_h_h,
    stride_out_b, stride_out_t, stride_out_h,
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    # bounds check for grid
    if pid_b >= B or pid_t >= (Stext + Simg):
        return

    # Decide source: encoder or hidden
    is_encoder = pid_t < Stext

    # Base pointers
    if is_encoder:
        base_src = e_ptr + pid_b * stride_e_b + pid_t * stride_e_t
    else:
        base_src = h_ptr + pid_b * stride_h_b + (pid_t - Stext) * stride_h_t

    # Write entire hidden dimension vector
    h_vec = tl.arange(0, H)
    vals = tl.load(base_src + h_vec * stride_e_h, mask=h_vec < H, other=0.0)
    tl.store(out_ptr + pid_b * stride_out_b + pid_t * stride_out_t + h_vec * stride_out_h, vals)


@triton.jit
def batched_gemm_kernel_loopk(
    a_ptr,            # *f32, [B, T, H] (concatenated)
    wT_ptr,           # *f32, [H, H] (transpose of process_weight)
    out_ptr,          # *f32, [B, T, H]
    B, T, H,
    stride_a_b, stride_a_t, stride_a_h,
    stride_w_b, stride_w_t, stride_w_h,   # wT is [H, H], strides for its dims
    stride_out_b, stride_out_t, stride_out_h,
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    if pid_b >= B or pid_t >= T:
        return

    # Compute out[b, t, :] = a[b, t, :] @ wT  (no bias)
    # We'll loop over k=0..H-1 and accumulate into a vector acc[h]
    h_vec = tl.arange(0, H)
    acc = tl.zeros((H,), dtype=tl.float32)

    # Loop over K (hidden dimension)
    for k in range(0, H):
        a_val = tl.load(a_ptr + pid_b * stride_a_b + pid_t * stride_a_t + k * stride_a_h)
        # Load wT[k, h] vector for all h
        w_vec = tl.load(wT_ptr + k * stride_w_b + h_vec * stride_w_h)
        acc += a_val * w_vec

    # Store result
    tl.store(out_ptr + pid_b * stride_out_b + pid_t * stride_out_t + h_vec * stride_out_h, acc)


class ModelNew(torch.nn.Module):
    def forward(
        self,
        hidden_states: torch.Tensor,
        encoder_hidden_states: torch.Tensor,
        process_weight: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-only implementation:
        - Concatenates encoder_hidden_states and hidden_states along the sequence dim (Triton).
        - Applies linear projection via batched GEMM with process_weight.T (Triton).
        - Splits results back into encoder and image parts (no torch ops on data).
        """
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "Inputs must be on CUDA device."
        assert hidden_states.dtype == torch.float32 and encoder_hidden_states.dtype == torch.float32 and process_weight.dtype == torch.float32, "Use float32 tensors."

        # Ensure contiguity
        e = encoder_hidden_states.contiguous()
        h = hidden_states.contiguous()
        W = process_weight.contiguous()  # [H, H]
        B, Stext, H = e.shape
        B_h, Simg, H2 = h.shape
        assert B == B_h and H == H2, "Batch and hidden_dim must match."
        T = Stext + Simg

        # Allocate concatenated tensor
        out_cat = torch.empty((B, T, H), device=e.device, dtype=torch.float32)

        # Launch concat kernel: grid (B, T)
        grid_concat = (B, T)
        concat_kernel[grid_concat](
            e, h, out_cat,
            B, Stext, Simg, H,
            e.stride(0), e.stride(1), e.stride(2),
            h.stride(0), h.stride(1), h.stride(2),
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            num_warps=4,
            num_stages=2,
        )

        # Transpose weight for A @ W_T
        W_T = W  # we'll treat as [H, H] without transpose; process_weight is already [H, H]
        # Create W_T explicitly to be safe: W_T = process_weight.t().contiguous()
        W_T = process_weight.t().contiguous()

        # Allocate output
        out = torch.empty((B, T, H), device=e.device, dtype=torch.float32)

        # Launch GEMM kernel: one program per (b, t), loop over H
        grid_gemm = (B, T)
        batched_gemm_kernel_loopk[grid_gemm](
            out_cat, W_T, out,
            B, T, H,
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            W_T.stride(0), W_T.stride(1), W_T.stride(2),  # W_T is [H, H]
            out.stride(0), out.stride(1), out.stride(2),
            num_warps=4,
            num_stages=2,
        )

        # Split along sequence dimension
        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

        return processed_encoder, processed_hidden
