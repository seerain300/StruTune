import torch
import triton
import triton.language as tl


@triton.jit
def concat_rows_to_A_kernel(
    enc_ptr,       # *ptr to [B, T, H]
    img_ptr,       # *ptr to [B, I, H]
    A_ptr,         # *ptr to [M, H], M = B*(T+I)
    B, T, I, H,    # dimensions
    stride_b_e, stride_t_e, stride_h_e,  # enc strides
    stride_b_i, stride_i_i, stride_h_i,  # img strides
    stride_m_a, stride_h_a,               # A strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    # Each program handles one output row m
    pid_m = tl.program_id(axis=0)
    m = pid_m

    # Compute batch and sequence index within the concatenated stream
    total_seq = T + I
    b_val = m // total_seq
    seq_val = m % total_seq

    # Determine source tensor: if seq_val < T -> encoder, else -> image
    mask_encoder = seq_val < T

    if mask_encoder:
        src_row_e = seq_val
        # Load row from encoder_hidden_states[b, seq, :]
        ptr_e = enc_ptr + b_val * stride_b_e + src_row_e * stride_t_e + tl.arange(0, BLOCK_N) * stride_h_e
        vals_e = tl.load(ptr_e, mask=tl.arange(0, BLOCK_N) < H, other=0.0)
        # Store into A[m, :]
        A_row_ptr = A_ptr + m * stride_m_a + tl.arange(0, BLOCK_N) * stride_h_a
        tl.store(A_row_ptr, vals_e, mask=tl.arange(0, BLOCK_N) < H)
    else:
        src_row_i = seq_val - T
        # Load row from hidden_states[b, seq - T, :]
        ptr_i = img_ptr + b_val * stride_b_i + src_row_i * stride_i_i + tl.arange(0, BLOCK_N) * stride_h_i
        vals_i = tl.load(ptr_i, mask=tl.arange(0, BLOCK_N) < H, other=0.0)
        # Store into A[m, :]
        A_row_ptr = A_ptr + m * stride_m_a + tl.arange(0, BLOCK_N) * stride_h_a
        tl.store(A_row_ptr, vals_i, mask=tl.arange(0, BLOCK_N) < H)


@triton.jit
def batched_matmul_kernel(
    A_ptr,     # *ptr to [M, H], M = B*(T+I)
    B_ptr,     # *ptr to [H, H] (process_weight.T)
    C_ptr,     # *ptr to [M, H]
    M, H,      # sizes
    stride_m_a, stride_n_a,          # A strides: [M, H]
    stride_m_b, stride_n_b,          # B strides: [H, H]
    stride_m_c, stride_n_c,          # C strides: [M, H]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,           # e.g., 64, 128, 256
    BLOCK_K: tl.constexpr,
):
    # 2D grid over output tiles
    pid_m = tl.program_id(axis=0)  # tile over M
    pid_n = tl.program_id(axis=1)  # tile over N=H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K = H
    for k in range(0, H, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_n_a
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < H), other=0.0)

        # Load B tile: [BLOCK_K, BLOCK_N], B is [H, H]
        b_ptrs = B_ptr + offs_k[:, None] * stride_m_b + offs_n[None, :] * stride_n_b
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < H) & (offs_n[None, :] < H), other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store C
    c_ptrs = C_ptr + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < H))


@triton.jit
def copy_rows_split_kernel(
    C_ptr,        # *ptr to [M, H], M = B*(T+I)
    out_enc_ptr,  # *ptr to [B, T, H]
    out_hid_ptr,  # *ptr to [B, I, H]
    B, T, I, H,   # dims
    stride_m_c, stride_h_c,            # C strides: [M, H]
    stride_b_e, stride_t_e, stride_h_e,  # enc strides
    stride_b_h, stride_i_h, stride_h_h,  # hid strides
    BLOCK_N: tl.constexpr,
):
    # One program per output row in each stream
    pid = tl.program_id(axis=0)
    m = pid  # m indexes C row
    # Determine if this row belongs to encoder or hidden stream
    total_seq = T + I
    b_val = m // total_seq
    seq_val = m % total_seq

    # Load C[m, :] tile
    offs_n = tl.arange(0, BLOCK_N)
    c_ptrs = C_ptr + m * stride_m_c + offs_n * stride_h_c
    vals = tl.load(c_ptrs, mask=offs_n < H, other=0.0)

    # Write to encoder output if seq_val < T
    if seq_val < T:
        out_e_ptrs = out_enc_ptr + b_val * stride_b_e + seq_val * stride_t_e + offs_n * stride_h_e
        tl.store(out_e_ptrs, vals, mask=offs_n < H)
    else:
        src_rel = seq_val - T
        out_h_ptrs = out_hid_ptr + b_val * stride_b_h + src_rel * stride_i_h + offs_n * stride_h_h
        tl.store(out_h_ptrs, vals, mask=offs_n < H)

def run(
    hidden_states: torch.Tensor,
    encoder_hidden_states: torch.Tensor,
    process_weight: torch.Tensor,
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Flux concatenated sequence processing pattern.
    
    1. Concatenate encoder_hidden_states and hidden_states along sequence dimension
    2. Apply linear projection to the concatenated sequence
    3. Split back into separate encoder and image streams
    
    Args:
        hidden_states: Image latent sequence [batch, img_seq_len, hidden_dim]
        encoder_hidden_states: Text conditioning sequence [batch, text_seq_len, hidden_dim]
        process_weight: Linear projection weight [hidden_dim, hidden_dim]
        
    Returns:
        Tuple of (processed_encoder_hidden_states, processed_hidden_states)
    """
    text_seq_len = encoder_hidden_states.shape[1]
    img_seq_len = hidden_states.shape[1]
    
    # Step 1: Concatenate sequences along sequence dimension
    # Shape: [batch, text_seq_len + img_seq_len, hidden_dim]
    concatenated = torch.cat([encoder_hidden_states, hidden_states], dim=1)
    
    # Step 2: Apply linear projection (no bias)
    # processed = concatenated @ process_weight.T
    processed = torch.matmul(concatenated, process_weight.t())
    
    # Step 3: Split back into separate streams
    processed_encoder = processed[:, :text_seq_len, :]
    processed_hidden = processed[:, text_seq_len:, :]
    
    return processed_encoder, processed_hidden
