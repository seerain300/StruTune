What the original PyTorch code does
- It takes:

  - encoder_hidden_states: [B, T, K]

- Splits back:
  processed_encoder = processed[:, :T, :]

- ModelNew.forward must not use any torch operations for computation (no torch.cat, no torch slicing/reshape for concatenation/splitting, no torch.matmul). All computation must be done by Triton kernels.
- The heavy part is the GEMM. Concatenation and splitting must be implemented in Triton, too.

- A frequent pitfall with Triton is launching the GEMM kernel with an incorrect grid or misconfiguring tiling. A robust approach is:
  - Concatenate into a single [B, T+P, K] tensor using Triton (data movement, simple).
  - Perform GEMM in Triton with a 3D grid over (batch, m_tiles, n_tiles), reducing over K in chunks.
  - Split the result into the two outputs via Triton kernels (data movement only).

Correctness-first strategy
- Implement a Triton kernel to concatenate [B, T, K] and [B, P, K] into [B, T+P, K] without torch.cat.

  - For each (b, l), if l < T, copy from encoder[b, l, :], else from hidden[b, l-T, :]. Use masks for K tail.
- Implement the GEMM in Triton using a 3D grid:
  - A is the concatenated tensor viewed as [M, K] where M = B*(T+P). We’ll pass A as contiguous [M, K] to the kernel.
  - W is process_weight.T [K, K] as a contiguous [K, K].
  - Output C_flat [M, K], which we will later reshape to [B, T+P, K].

- Keep dtype float32. Triton kernels should be robust to arbitrary B, T, P, K, including non-multiples, using masks for boundary conditions.

import torch
import triton
import triton.language as tl


@triton.jit
def _concat_encoder_hidden_kernel(
    encoder_ptr,  # *float, [B, T, K]
    hidden_ptr,   # *float, [B, P, K]
    out_ptr,      # *float, [B, T+P, K]
    B, T, P, K,
    BLOCK_K: tl.constexpr,
):
    # program ids
    b = tl.program_id(0)
    l = tl.program_id(1)
    k_block = tl.program_id(2)

    # compute k offsets and mask
    k_offsets = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
    k_mask = k_offsets < K

    # source tensors for this l
    # if l < T: src = encoder[b, l, :]
    # else:      src = hidden[b, l - T, :]
    is_encoder = l < T
    src_b = b
    src_l = l if is_encoder else (l - T)

    # compute base pointers
    # encoder: out[b, l, k] = encoder[b, l, k]
    # hidden:  out[b, l, k] = hidden[b, l - T, k]
    # Use strides, assuming contiguous tensors
    encoder_base = src_b * (T * K) + src_l * K
    hidden_base = src_b * (P * K) + (src_l) * K  # src_l is l - T when hidden
    out_base = b * ((T + P) * K) + l * K

    # load
    enc_vals = tl.load(encoder_ptr + encoder_base + k_offsets, mask=k_mask, other=0.0)
    hid_vals = tl.load(hidden_ptr + hidden_base + k_offsets, mask=k_mask, other=0.0)

    # select based on is_encoder
    vals = tl.where(is_encoder, enc_vals, hid_vals)

    # store
    tl.store(out_ptr + out_base + k_offsets, vals, mask=k_mask)


@triton.jit
def _gemm_matmul_kernel(
    A_ptr,  # *float, [M, K] where M = B*(T+P)
    W_ptr,  # *float, [K, K]
    C_ptr,  # *float, [M, K]
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3D grid: (b, m_tile, n_tile). Triton doesn't allow host-side loop over b;
    # we pass b via program_id(0) and compute offsets accordingly.
    b = tl.program_id(0)
    m_tile = tl.program_id(1)
    n_tile = tl.program_id(2)

    m_offsets = m_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offsets = n_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offsets = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K
    for k in range(0, K, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        # A tile: A is [M, K], contiguous row-major => A[i, j] offset = i*K + j
        # For each i in m_offsets, load a vector of size BLOCK_K from column k_offsets
        A_ptrs = A_ptr + m_offsets[:, None] * K + k_offsets[None, :]
        A_mask = (m_offsets[:, None] < M) & (k_mask[None, :])
        A_tile = tl.load(A_ptrs, mask=A_mask, other=0.0)

        # W tile: W is [K, K], contiguous row-major => W[i, j] offset = i*K + j
        W_ptrs = W_ptr + k_offsets[:, None] * K + n_offsets[None, :]
        W_mask = (k_mask[:, None]) & (n_offsets[None, :] < N)
        W_tile = tl.load(W_ptrs, mask=W_mask, other=0.0)

        # Accumulate
        acc += tl.dot(A_tile, W_tile)

    # Store C: C is [M, N], contiguous row-major => C[i, j] offset = i*N + j
    C_ptrs = C_ptr + m_offsets[:, None] * N + n_offsets[None, :]
    M_mask = m_offsets[:, None] < M
    N_mask = n_offsets[None, :] < N
    store_mask = M_mask & N_mask
    tl.store(C_ptrs, acc, mask=store_mask)


@triton.jit
def _split_encoder_kernel(
    C_ptr,   # *float, [B, T+P, K]
    out_ptr, # *float, [B, T, K]
    B, T, P, K,
    BLOCK_K: tl.constexpr,
):
    # grid: (B, T, ceil_div(K, BLOCK_K))
    b = tl.program_id(0)
    t = tl.program_id(1)
    k_block = tl.program_id(2)

    k_offsets = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
    k_mask = k_offsets < K

    C_base = (b * (T + P) + t) * K
    out_base = (b * T + t) * K

    vals = tl.load(C_ptr + C_base + k_offsets, mask=k_mask, other=0.0)
    tl.store(out_ptr + out_base + k_offsets, vals, mask=k_mask)


@triton.jit
def _split_hidden_kernel(
    C_ptr,   # *float, [B, T+P, K]
    out_ptr, # *float, [B, P, K]
    B, T, P, K,
    BLOCK_K: tl.constexpr,
):
    # grid: (B, P, ceil_div(K, BLOCK_K))
    b = tl.program_id(0)
    p = tl.program_id(1)
    k_block = tl.program_id(2)

    k_offsets = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
    k_mask = k_offsets < K

    C_base = (b * (T + P) + (p + T)) * K
    out_base = (b * P + p) * K

    vals = tl.load(C_ptr + C_base + k_offsets, mask=k_mask, other=0.0)
    tl.store(out_ptr + out_base + k_offsets, vals, mask=k_mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        # Shapes
        B = hidden_states.shape[0]
        P = hidden_states.shape[1]  # img_seq_len
        T = encoder_hidden_states.shape[1]  # text_seq_len
        K = hidden_states.shape[2]  # hidden_dim

        device = hidden_states.device
        dtype = hidden_states.dtype

        # Ensure tensors are contiguous
        encoder = encoder_hidden_states.contiguous()
        hidden = hidden_states.contiguous()
        weight_t = process_weight.transpose(0, 1).contiguous()  # [K, K]

        # Step 1: Triton concatenation -> Acat [B, T+P, K]
        total_L = T + P
        Acat = torch.empty((B, total_L, K), device=device, dtype=dtype)

        BLOCK_K = 64
        grid_concat = (B, total_L, triton.cdiv(K, BLOCK_K))
        _concat_encoder_hidden_kernel[grid_concat](
            encoder, hidden, Acat,
            B, T, P, K,
            BLOCK_K=BLOCK_K, num_warps=4, num_stages=2
        )

        # Step 2: GEMM via Triton: Acat [B, T+P, K] flattened to [M, K], W [K, K]
        M = B * total_L
        C_flat = torch.empty((M, K), device=device, dtype=dtype)

        # Create A_2d [M, K] contiguous from Acat
        A_2d = Acat.reshape(M, K).contiguous()

        # GEMM kernel config
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32

        # Triton grid over (batch, m_tiles, n_tiles). We launch per (b, tile_m, tile_n).
        grid_gemm = (B, triton.cdiv(M, BLOCK_M), triton.cdiv(K, BLOCK_N))
        _gemm_matmul_kernel[grid_gemm](
            A_2d, weight_t, C_flat,
            M, K, K,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=3
        )

        # Step 3: Triton splitting into encoder and hidden streams
        processed_encoder = torch.empty((B, T, K), device=device, dtype=dtype)
        processed_hidden = torch.empty((B, P, K), device=device, dtype=dtype)

        BLOCK_K_s = 64
        grid_e = (B, T, triton.cdiv(K, BLOCK_K_s))
        _split_encoder_kernel[grid_e](
            C_flat, processed_encoder,
            B, T, P, K,
            BLOCK_K=BLOCK_K_s, num_warps=4, num_stages=2
        )

        grid_h = (B, P, triton.cdiv(K, BLOCK_K_s))
        _split_hidden_kernel[grid_h](
            C_flat, processed_hidden,
            B, T, P, K,
            BLOCK_K=BLOCK_K_s, num_warps=4, num_stages=2
        )

        return processed_encoder, processed_hidden

- The concatenation kernel uses a simple 3D grid over (batch, sequence position, K-tile) and masks, making it robust for arbitrary T, P, K. The GEMM uses a 3D grid over (batch, M-tiles, N-tiles), reducing over K with masks for boundaries, which is standard Triton matmul practice.

Performance and tuning
- The chosen block sizes (64/64/32) are reasonable defaults. For performance-critical cases, autotuning block sizes based on K can further improve throughput. Once correctness is validated across diverse inputs (as per evaluator’s workloads), performance tuning can be added without compromising correctness.
