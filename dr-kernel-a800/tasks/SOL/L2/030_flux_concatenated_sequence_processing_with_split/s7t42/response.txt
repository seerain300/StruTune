```python
import torch
import triton
import triton.language as tl


@triton.jit
def cat_rows_kernel(
    e_ptr,        # *f32, encoder_hidden_states
    i_ptr,        # *f32, hidden_states
    out_ptr,      # *f32, output concatenated [B, T+I, H]
    T, I, H,      # ints
    stride_e_b, stride_e_t, stride_e_h,   # strides for e
    stride_i_b, stride_i_i, stride_i_h,   # strides for i
    stride_out_b, stride_out_p, stride_out_h,  # strides for out
):
    # 2D grid: (batch, row index)
    b = tl.program_id(0)
    p = tl.program_id(1)

    # Compute source index for i (image) rows
    src_i = p - T  # if p >= T, src_i in [0, I); else negative (unused)

    # Masks: if p < T, take from e[b, p, :]; else take from i[b, src_i, :]
    use_e = p < T

    # Prepare h offsets (columns)
    offs_h = tl.arange(0, H)

    # Compute base pointers
    e_row_ptr = e_ptr + b * stride_e_b + p * stride_e_t
    i_row_ptr = i_ptr + b * stride_i_b + src_i * stride_i_i
    out_row_ptr = out_ptr + b * stride_out_b + p * stride_out_p

    # Load and store with mask
    # If use_e, load from e; else load from i; either way we store to out
    # Triton allows masked loads/stores per dimension.
    # We use a combined mask that is always true in H dimension (since offs_h < H), and a scalar condition for source.

    # Load from e if use_e; else from i if not use_e (but i_row_ptr may be invalid when use_e).
    # To avoid invalid pointers, we only load when use_e; otherwise we use "other" value.
    # Note: Triton doesn't support dynamic control flow here, so we load both and select via tl.where.
    # However, Triton requires pointers to be valid; thus we do two loads: one guarded by use_e and one by not use_e,
    # and then combine. Since Triton expects valid pointers, we'll set the non-selected load to zeros.

    if use_e:
        vals = tl.load(e_row_ptr + offs_h * stride_e_h, mask=offs_h < H, other=0.0)
    else:
        vals = tl.load(i_row_ptr + offs_h * stride_i_h, mask=offs_h < H, other=0.0)

    # Store to out regardless
    tl.store(out_row_ptr + offs_h * stride_out_h, vals, mask=offs_h < H)

        B = hidden_states.shape[0]

        I = hidden_states.shape[1]
        H = hidden_states.shape[2]
        assert encoder_hidden_states.shape == (B, T, H), "encoder_hidden_states must be [B, T, H]"
        assert hidden_states.shape == (B, I, H), "hidden_states must be [B, I, H]"
        assert process_weight.shape == (H, H), "process_weight must be [H, H]"
```

```python
import torch
import triton
import triton.language as tl


@triton.jit
def cat_rows_kernel(
    e_ptr,        # *f32, encoder_hidden_states [B, T, H]
    i_ptr,        # *f32, hidden_states [B, I, H]
    out_ptr,      # *f32, output concatenated [B, T+I, H]
    T, I, H,      # ints
    stride_e_b, stride_e_t, stride_e_h,
    stride_i_b, stride_i_i, stride_i_h,
    stride_out_b, stride_out_p, stride_out_h,
):

    b = tl.program_id(0)
    p = tl.program_id(1)

    # If p < T, take row from e; else take row from i at index p - T
    src_i = p - T  # valid when p >= T
    offs_h = tl.arange(0, H)

    e_row_ptr = e_ptr + b * stride_e_b + p * stride_e_t
    i_row_ptr = i_ptr + b * stride_i_b + src_i * stride_i_i
    out_row_ptr = out_ptr + b * stride_out_b + p * stride_out_p

    if p < T:
        vals = tl.load(e_row_ptr + offs_h * stride_e_h, mask=offs_h < H, other=0.0)
    else:
        vals = tl.load(i_row_ptr + offs_h * stride_i_h, mask=offs_h < H, other=0.0)

    tl.store(out_row_ptr + offs_h * stride_out_h, vals, mask=offs_h < H)


@triton.jit
def batched_matmul_rows_kernel(
    x_ptr,        # *f32, X_cat [B, M, H]
    w_ptr,        # *f32, process_weight [H, H]
    y_ptr,        # *f32, output [B, M, H]
    B, M, H,      # ints
    stride_x_b, stride_x_m, stride_x_h,
    stride_w_h, stride_w_k,  # process_weight has shape [H, H] here; original expects [H, H], so stride_w_k = stride_w_h
    stride_y_b, stride_y_m, stride_y_h,
    BLOCK_K: tl.constexpr,
):
    # 2D grid: (batch, row index)
    b = tl.program_id(0)
    m = tl.program_id(1)

    # We compute y[b, m, :] = x[b, m, :] @ w, where w is [H, H].
    # Accumulate in float32.
    acc = tl.zeros((H,), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load x[b, m, k_tile] as vector of length BLOCK_K
        x_row_ptr = x_ptr + b * stride_x_b + m * stride_x_m
        x_vals = tl.load(x_row_ptr + offs_k * stride_x_h, mask=mask_k, other=0.0)  # shape [BLOCK_K], float32

        # Load w[k_tile, :] as matrix [BLOCK_K, H]
        w_base_ptr = w_ptr + offs_k * stride_w_h  # each column is w[offs_k, :]
        w_mat = tl.load(w_base_ptr + tl.arange(0, H) * stride_w_k, mask=mask_k[:, None], other=0.0)  # [BLOCK_K, H]

        # Accumulate: acc += sum over k in tile of x_vals[k] * w_mat[k, :]
        # This is equivalent to acc += tl.sum(x_vals[:, None] * w_mat, axis=0)
        # Triton supports elementwise mul and tl.sum; we can implement as:
        for kk in range(BLOCK_K):
            # If kk beyond H, x_vals[kk] is 0 due to masked load
            acc += x_vals[kk] * w_mat[kk, :]

    # Store result to y[b, m, :]
    y_row_ptr = y_ptr + b * stride_y_b + m * stride_y_m
    tl.store(y_row_ptr + tl.arange(0, H) * stride_y_h, acc, mask=tl.arange(0, H) < H)
```

```python
        processed_encoder = processed[:, :T, :]
        processed_hidden = processed[:, T:, :]

        return processed_encoder, processed_hidden
```

Notes
- This implementation uses Triton for both the concatenation and the linear projection. The concatenation kernel writes each row of the concatenated matrix without using torch.cat. The matmul kernel computes one output row per program using a simple loop over the hidden dimension in tiles, accumulating in float32 and storing back in the original dtype. This avoids complex 2D tiling and reduces the chance of runtime errors.
- The forward function does not use any torch operations (no torch.cat, no torch.matmul, no elementwise tensor math). It only allocates tensors, launches Triton kernels, and slices the result on the host to return the two streams. This satisfies the evaluation constraint that ModelNew.forward must invoke Triton kernels and avoid torch operations.
