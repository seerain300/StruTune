        # Load X tile: [BLOCK_M, BLOCK_K]
        for m0 in range(0, BLOCK_M):
            m = m0
            x_ptrs = X_ptr + m * H + k_ids
            x_vals = tl.load(x_ptrs)  # [BLOCK_K]
            x_tile[m0, :] = x_vals

        # Load W tile: [BLOCK_K, BLOCK_N]
        for n0 in range(0, BLOCK_N):
            n = n0
            w_ptrs = W_ptr + k_ids[:, None] * H + n  # [BLOCK_K, 1] -> broadcast load
            w_vals = tl.load(w_ptrs)                # [BLOCK_K, 1]
            w_tile[:, n0] = tl.load(W_ptr + k_ids[:, None] * H + n)  # correct 2D load across N

        # Accumulate
        acc += tl.dot(x_tile, w_tile)

    # Store result to Y
    for m0 in range(0, BLOCK_M):
        m = m0
        for n0 in range(0, BLOCK_N):
            n = n0
            y_ptrs = Y_ptr + m * H + n
            y_val = acc[m, n]
            tl.store(y_ptrs, y_val)


@triton.jit
def fill_zeros_kernel(
    out_ptr,          # *T, [B, M, H]
    B: tl.constexpr,
    M: tl.constexpr,
    H: tl.constexpr,
):
    b = tl.program_id(0)
    m = tl.program_id(1)
    h = tl.program_id(2)
    out_ptr[b * M * H + m * H + h] = 0  # zero fill


@triton.jit
def copy_rows_slice_kernel(
    src_ptr,          # *T, [B, M, H]
    dst_ptr,          # *T, [B, N, H] (N=T or I)
    B: tl.constexpr,
    M: tl.constexpr,  # total rows
    N: tl.constexpr,  # rows to copy (T or I)
    H: tl.constexpr,
):
    b = tl.program_id(0)
    row = tl.program_id(1)
    col = tl.program_id(2)  # hidden dim
    # src row index for first half: row
    # dst row index: row
    src_row_ptr = src_ptr + b * M * H + row * H + col
    dst_row_ptr = dst_ptr + b * N * H + row * H + col
    val = tl.load(src_row_ptr)
    tl.store(dst_row_ptr, val)
