import torch
import triton
import triton.language as tl


@triton.jit
def batched_linear_row_kernel(
    in_ptr,           # *f32, concatenated [B, T, H_in]
    weightT_ptr,      # *f32, [H_in, H_out] (process_weight.T)
    out_ptr,          # *f32, [B, T, H_out] (initialized to zeros)
    B: tl.constexpr,
    T: tl.constexpr,
    H_in: tl.constexpr,   # input hidden dim (concatenated row length)
    H_out: tl.constexpr,  # output hidden dim (process_weight shape[1])
    stride_i_b: tl.constexpr,
    stride_i_t: tl.constexpr,
    stride_i_h: tl.constexpr,  # input hidden stride
    stride_w_k: tl.constexpr,  # weight_T dim-0 stride
    stride_w_n: tl.constexpr,  # weight_T dim-1 stride
    stride_o_b: tl.constexpr,
    stride_o_t: tl.constexpr,
    stride_o_h: tl.constexpr,  # output hidden stride
    BLOCK_K: tl.constexpr,     # chunk size over H_in
    BLOCK_H: tl.constexpr,     # output tile size
):
    # Each program handles one (b, t) and accumulates the whole output row
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Vector of output offsets for this tile
    n_offsets = tl.arange(0, BLOCK_H)
    # Accumulator for output row slice
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # Loop over input hidden dimension in chunks
    k0 = 0
    while k0 < H_in:
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H_in

        # Load input chunk: concat[b, t, k_offsets]
        in_ptrs = in_ptr + b * stride_i_b + t * stride_i_t + k_offsets * stride_i_h
        in_chunk = tl.load(in_ptrs, mask=mask_k, other=0.0)  # shape [BLOCK_K]

        # Load weight_T tile: [BLOCK_K, BLOCK_H] for current k chunk and n offsets
        w_ptrs = weightT_ptr + k_offsets[:, None] * stride_w_k + n_offsets[None, :] * stride_w_n
        mask_n = n_offsets < H_out
        tile_mask = mask_k[:, None] & mask_n[None, :]
        w_tile = tl.load(w_ptrs, mask=tile_mask, other=0.0)  # shape [BLOCK_K, BLOCK_H]

        # Accumulate: acc += sum_k(in_chunk[k] * w_tile[k, :])
        # This is a vector outer-product accumulation over BLOCK_K.
        # Implement elementwise multiply and reduction over axis=0
        # acc is [BLOCK_H], in_chunk[:, None] is [BLOCK_K, 1], w_tile is [BLOCK_K, BLOCK_H]
        # We need to broadcast in_chunk and reduce over k: sum_k(w_tile[k, :] * in_chunk[k])
        # However, Triton doesn't have a direct outer product reduce; do it explicitly:
        acc += tl.sum(w_tile * in_chunk[:, None], axis=0)

        k0 += BLOCK_K

    # Store accumulated result into out[b, t, 0:BLOCK_H]
    o_ptrs = out_ptr + b * stride_o_b + t * stride_o_t + n_offsets * stride_o_h
    # Mask for n_offsets < H_out
    tl.store(o_ptrs, acc, mask=n_offsets < H_out)

    stride_o_h: tl.constexpr,  # output hidden stride
