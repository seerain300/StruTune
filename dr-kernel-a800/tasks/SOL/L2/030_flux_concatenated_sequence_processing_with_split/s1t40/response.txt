@triton.jit
def _matmul_kernel_3d(
    A_ptr, W_ptr, C_ptr,
    M, K, N,
    A_stride_m, A_stride_k,
    W_stride_k, W_stride_n,
    C_stride_m, C_stride_n,

    tn = tl.program_id(1)  # tile index over N dimension (cols)
    tk = tl.program_id(2)  # tile index over K reduction dimension

    m_start = tm * BLOCK_M
    n_start = tn * BLOCK_N
    k_start = tk * BLOCK_K

    m_offsets = m_start + tl.arange(0, BLOCK_M)
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    k_offsets = k_start + tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # Current K offsets
        k_offsets = k + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        W_tile = tl.load(W_ptrs, mask=W_mask, other=0.0)

        # Accumulate
        acc += tl.dot(A_tile, W_tile)

    C_mask = (m_offsets[:, None] < M) & (n_offsets[None, :] < N)
    tl.store(C_ptrs, acc, mask=C_mask)

    b = tl.program_id(0)
    l_tile = tl.program_id(1)
    k_tile = tl.program_id(2)

    L = T + P

    # Vector of K indices for this tile
    k_offsets = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
    k_mask = k_offsets < K

    # Determine source: if l < T -> encoder, else -> hidden at index (l - T)
    # We'll compute both enc and hid pointers, but only load/store one based on condition.
    # To avoid scalar condition inside vectorized operations, we use masked loads/stores.
    # First, prepare enc pointers for all k (we will mask where condition is false)
    enc_ptrs = encoder_ptr + b * encoder_stride_b + l_tile * encoder_stride_t + k_offsets * encoder_stride_k
    # Hid pointers (l_tile - T can be negative; we guard with mask)
    hid_ptrs = hidden_ptr + b * hidden_stride_b + (l_tile - T) * hidden_stride_p + k_offsets * hidden_stride_k

    # We need to select between enc and hid based on l_tile < T.
    # Triton doesn't support scalar if on vector data, so we perform two masked loads and then two masked stores.
    # First load enc if l_tile < T, else load zeros for enc (won't be stored).
    load_mask_enc = k_mask & (l_tile < L)  # l_tile < L is always true, but we need to ensure we use enc only when l_tile < T
    # Correction: we need to distinguish T from P. So we replace L with T: l_tile < T decides source
    # Compute final pointers: since we can't branch, we rely on masks to only store when source is valid.
    # For enc:
    # If l_tile < T: valid source; else invalid source mask will zero out store.
    # For hid:
    # If l_tile >= T: valid source; else invalid source mask will zero out store.
    # We'll do this via two separate stores, each masked by (l_tile < T) and its complement.
    # However, Triton's scalar condition on masked store is fine: we pass scalar condition to select.
    is_encoder = l_tile < T

    # We need to perform masked load for enc and hid based on is_encoder, but Triton doesn't support mixing
    # pointers by scalar; therefore, we do two loads, both using k_mask, then store with is_encoder mask.
    # Better: perform a scalar-branch using tl.where is not available, so we do two masked loads into two tensors
    # and then store with masks. In practice, Triton supports scalar condition in masked load/store.
    # Load encoder values only when is_encoder
    A_vals = tl.load(enc_ptrs, mask=k_mask & is_encoder, other=0.0)
    # Load hidden values only when not is_encoder (i.e., l_tile >= T)
    B_vals = tl.load(hid_ptrs, mask=k_mask & (~is_encoder), other=0.0)

    # Compute out pointers for current (b, l_tile, k_offsets)
    out_ptrs = out_ptr + b * out_stride_b + l_tile * out_stride_l + k_offsets * out_stride_k

    # Store the selected values; only one of A_vals or B_vals is non-zero depending on mask
    tl.store(out_ptrs, A_vals, mask=k_mask & is_encoder)
    tl.store(out_ptrs, B_vals, mask=k_mask & (~is_encoder))

# Triton GEMM: A [M, K] x W [K, K] -> C [M, K]
# We will launch this with 3D grid over (M tiles, N tiles, K tiles)
@triton.jit
def _matmul_kernel_3d(
    A_ptr, W_ptr, C_ptr,
    M, K, N,
    A_stride_m, A_stride_k,
    W_stride_k, W_stride_n,
    C_stride_m, C_stride_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    tm = tl.program_id(0)  # tile index over M (rows)
    tn = tl.program_id(1)  # tile index over N (cols)
    tk = tl.program_id(2)  # tile index over K (reduction)

    m_start = tm * BLOCK_M
    n_start = tn * BLOCK_N
    k_start = tk * BLOCK_K

    m_offsets = m_start + tl.arange(0, BLOCK_M)
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    k_offsets = k_start + tl.arange(0, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, K, BLOCK_K):
        k_curr = kk + tl.arange(0, BLOCK_K)
        k_mask = k_curr < K

        # A tile: [BLOCK_M, BLOCK_K]
        A_ptrs = A_ptr + m_offsets[:, None] * A_stride_m + k_curr[None, :] * A_stride_k
        A_mask = (m_offsets[:, None] < M) & (k_mask[None, :])
        A_tile = tl.load(A_ptrs, mask=A_mask, other=0.0)

        # W tile: [BLOCK_K, BLOCK_N]
        W_ptrs = W_ptr + k_curr[:, None] * W_stride_k + n_offsets[None, :] * W_stride_n
        W_mask = (k_mask[:, None]) & (n_offsets[None, :] < N)
        W_tile = tl.load(W_ptrs, mask=W_mask, other=0.0)

        acc += tl.dot(A_tile, W_tile)

    # Store result to C
    C_ptrs = C_ptr + m_offsets[:, None] * C_stride_m + n_offsets[None, :] * C_stride_n
    C_mask = (m_offsets[:, None] < M) & (n_offsets[None, :] < N)
    tl.store(C_ptrs, acc, mask=C_mask)

# Triton kernel: split C [B, T+P, K] into processed_encoder [B, T, K]
@triton.jit
def _split_encoder_kernel(
    C_ptr, out_ptr,
    B, T, K,
    C_stride_b, C_stride_t, C_stride_k,
    out_stride_b, out_stride_t, out_stride_k,
    BLOCK_T: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    t_tile = tl.program_id(1)
    k_tile = tl.program_id(2)

    t_offsets = t_tile * BLOCK_T + tl.arange(0, BLOCK_T)
    k_offsets = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
    t_mask = t_offsets < T
    k_mask = k_offsets < K

    C_ptrs = C_ptr + b * C_stride_b + t_offsets[:, None] * C_stride_t + k_offsets[None, :] * C_stride_k
    C_mask = t_mask[:, None] & k_mask[None, :]
    vals = tl.load(C_ptrs, mask=C_mask, other=0.0)

    out_ptrs = out_ptr + b * out_stride_b + t_offsets[:, None] * out_stride_t + k_offsets[None, :] * out_stride_k
    tl.store(out_ptrs, vals, mask=C_mask)
