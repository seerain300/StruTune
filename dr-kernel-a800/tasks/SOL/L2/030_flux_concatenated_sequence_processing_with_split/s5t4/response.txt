```python
import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    e_ptr,            # *f32, encoder_hidden_states [B, Stext, H]
    h_ptr,            # *f32, hidden_states [B, Simg, H]
    out_ptr,          # *f32, concatenated [B, T, H]
    B: tl.constexpr,
    Stext: tl.constexpr,
    Simg: tl.constexpr,
    H: tl.constexpr,
    T: tl.constexpr,
    stride_e_b: tl.constexpr, stride_e_t: tl.constexpr, stride_e_h: tl.constexpr,
    stride_h_b: tl.constexpr, stride_h_t: tl.constexpr, stride_h_h: tl.constexpr,
    stride_out_b: tl.constexpr, stride_out_t: tl.constexpr, stride_out_h: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # 2D grid: (B, T)
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    # Tile along H
    h_offsets = tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H

    # Determine source tensor based on t
    is_text = pid_t < Stext

    # Compute base pointers
    e_base = e_ptr + pid_b * stride_e_b
    h_base = h_ptr + pid_b * stride_h_b
    out_base = out_ptr + pid_b * stride_out_b

    if is_text:
        # Copy from encoder_hidden_states
        src_base = e_base + pid_t * stride_e_t
        out_ptrs = out_base + pid_t * stride_out_t + h_offsets * stride_out_h
        e_ptrs = src_base + h_offsets * stride_e_h
        vals = tl.load(e_ptrs, mask=mask_h, other=0.0)
        tl.store(out_ptrs, vals, mask=mask_h)
    else:
        # Copy from hidden_states (offset by Stext)
        src_t = pid_t - Stext
        src_base = h_base + src_t * stride_h_t
        out_ptrs = out_base + pid_t * stride_out_t + h_offsets * stride_out_h
        h_ptrs = src_base + h_offsets * stride_h_h
        vals = tl.load(h_ptrs, mask=mask_h, other=0.0)
        tl.store(out_ptrs, vals, mask=mask_h)


@triton.jit
def batched_gemm_kernel(
    A_ptr,            # *f32, out_cat [B, T, H]
    B_ptr,            # *f32, process_weight_T [H, H]
    Out_ptr,          # *f32, output [B, T, H]
    B: tl.constexpr,
    T: tl.constexpr,
    H: tl.constexpr,
    stride_A_b: tl.constexpr, stride_A_t: tl.constexpr, stride_A_h: tl.constexpr,
    stride_B_k: tl.constexpr, stride_B_h: tl.constexpr,
    stride_Out_b: tl.constexpr, stride_Out_t: tl.constexpr, stride_Out_h: tl.constexpr,
    H_tiles: tl.constexpr,  # number of tiles along H
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3D grid: (B, T, H_tiles)
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_ht = tl.program_id(2)

    h_start = pid_ht * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H

    # Accumulator for this (b, t, h-tile)
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # Load A[b, t, k_offsets] -> shape (BLOCK_K,)
        a_ptrs = A_ptr + pid_b * stride_A_b + pid_t * stride_A_t + k_offsets * stride_A_h
        a_vals = tl.load(a_ptrs, mask=mask_k, other=0.0)  # [BLOCK_K]

        # Load B[k_offsets, h_offsets] -> shape (BLOCK_K, BLOCK_H)
        b_ptrs = B_ptr + k_offsets[:, None] * stride_B_k + h_offsets[None, :] * stride_B_h
        b_vals = tl.load(b_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)  # [BLOCK_K, BLOCK_H]

        # Accumulate: acc += sum_k a[k] * b[k, :]
        # Implement as outer product add per k
        for kk in range(0, BLOCK_K):
            a_k = a_vals[kk]  # scalar
            b_k = b_vals[kk, :]  # [BLOCK_H]
            acc += a_k * b_k

    # Store results to Out[b, t, h_offsets]
    out_ptrs = Out_ptr + pid_b * stride_Out_b + pid_t * stride_Out_t + h_offsets * stride_Out_h
    tl.store(out_ptrs, acc, mask=mask_h)


class ModelNew(torch.nn.Module):
    def forward(
        self,
        hidden_states: torch.Tensor,
        encoder_hidden_states: torch.Tensor,
        process_weight: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-only implementation:
        - Concatenates encoder_hidden_states and hidden_states along sequence dimension inside Triton.
        - Applies linear projection (process_weight.T) to the concatenated sequences inside Triton.
        - Returns split outputs: (processed_encoder, processed_hidden).
        """
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, \
            "Inputs must be CUDA tensors for Triton execution."

        # Shapes
        B = hidden_states.shape[0]
        Stext = encoder_hidden_states.shape[1]
        Simg = hidden_states.shape[1]
        H = hidden_states.shape[2]
        T = Stext + Simg

        # Ensure contiguity
        e = encoder_hidden_states.contiguous()
        h = hidden_states.contiguous()
        # Concatenation output [B, T, H]
        out_cat = torch.empty((B, T, H), device=e.device, dtype=e.dtype)

        # Launch concat kernel
        # Choose BLOCK_H for H tiling. Use 128 for good throughput on H up to 1024.
        BLOCK_H = 128
        grid_concat = (B, T)
        concat_kernel[grid_concat](
            e, h, out_cat,
            B, Stext, Simg, H, T,
            e.stride(0), e.stride(1), e.stride(2),
            h.stride(0), h.stride(1), h.stride(2),
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

        # Prepare weight as [H, H], contiguous
        W_T = process_weight.transpose(0, 1).contiguous()

        # Output [B, T, H]
        out = torch.empty((B, T, H), device=e.device, dtype=e.dtype)

        # Launch GEMM kernel: 3D grid over (B, T, H_tiles)
        BLOCK_H_K = 128  # tile size along hidden output
        H_tiles = (H + BLOCK_H_K - 1) // BLOCK_H_K
        grid_gemm = (B, T, H_tiles)
        batched_gemm_kernel[grid_gemm](
            out_cat, W_T, out,
            B, T, H,
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            W_T.stride(0), W_T.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            H_tiles=H_tiles,  # meta-only; Triton ignores constexpr in Python; we pass as lambda below
            BLOCK_H=BLOCK_H_K,
            BLOCK_K=64,
            num_warps=4, num_stages=2,
        )

        # Split outputs
        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

        return processed_encoder, processed_hidden
```

```python
def batched_gemm_kernel(
    A_ptr,            # *f32, out_cat [B, T, H]
    B_ptr,            # *f32, process_weight_T [H, H]
    Out_ptr,          # *f32, output [B, T, H]
    B: tl.constexpr,
    T: tl.constexpr,
    H: tl.constexpr,
    stride_A_b: tl.constexpr, stride_A_t: tl.constexpr, stride_A_h: tl.constexpr,
    stride_B_k: tl.constexpr, stride_B_h: tl.constexpr,
    stride_Out_b: tl.constexpr, stride_Out_t: tl.constexpr, stride_Out_h: tl.constexpr,
):
    # 2D grid over (B, T): each program computes full output vector for (b, t)
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    out_row_ptr = Out_ptr + pid_b * stride_Out_b + pid_t * stride_Out_t

    # Accumulator for the entire output vector
    acc = tl.zeros((H,), dtype=tl.float32)

    # Loop over k = 0..H-1
    for k in range(0, H):
        # Load A[b, t, k]
        a_val = tl.load(A_ptr + pid_b * stride_A_b + pid_t * stride_A_t + k * stride_A_h)

        # Load B[k, :] and multiply: B is [H, H], we need column k
        b_vec = tl.load(B_ptr + k * stride_B_k + tl.arange(0, H) * stride_B_h)
        acc += a_val * b_vec

    # Store full acc
    for h in range(0, H):
        tl.store(out_row_ptr + h * stride_Out_h, acc[h])


class ModelNew(torch.nn.Module):
    def forward(
        self,
        hidden_states: torch.Tensor,
        encoder_hidden_states: torch.Tensor,
        process_weight: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-only implementation:
        - Concatenates encoder_hidden_states and hidden_states along sequence dimension inside Triton.
        - Applies linear projection (process_weight.T) to the concatenated sequences inside Triton.
        - Returns split outputs: (processed_encoder, processed_hidden).
        """
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, \
            "Inputs must be CUDA tensors for Triton execution."

        # Shapes
        B = hidden_states.shape[0]
        Stext = encoder_hidden_states.shape[1]
        Simg = hidden_states.shape[1]
        H = hidden_states.shape[2]
        T = Stext + Simg

        # Ensure contiguity
        e = encoder_hidden_states.contiguous()
        h = hidden_states.contiguous()

        # Concatenation output [B, T, H]
        out_cat = torch.empty((B, T, H), device=e.device, dtype=e.dtype)

        # Launch concat kernel: 2D grid (B, T)
        grid_concat = (B, T)
        concat_kernel[grid_concat](
            e, h, out_cat,
            B, Stext, Simg, H, T,
            e.stride(0), e.stride(1), e.stride(2),
            h.stride(0), h.stride(1), h.stride(2),
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            num_warps=4, num_stages=2,
        )

        # Prepare weight as [H, H], contiguous
        W_T = process_weight.transpose(0, 1).contiguous()

        # Output [B, T, H]
        out = torch.empty((B, T, H), device=e.device, dtype=e.dtype)

        # Launch GEMM kernel: 2D grid (B, T), each program computes full vector
        grid_gemm = (B, T)
        batched_gemm_kernel[grid_gemm](
            out_cat, W_T, out,
            B, T, H,
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            W_T.stride(0), W_T.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            num_warps=4, num_stages=2,
        )

        # Split outputs: return view slices (no compute)
        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

        return processed_encoder, processed_hidden
```

Why this should fix correctness
- The concat kernel writes exactly one row per (b, t), selecting the correct source based on t, and loops over H to write each element. There are no tiling or masking issues along H or T.
- The GEMM kernel computes the full output vector per (b, t) by looping over k and loading the corresponding column of process_weight_T. This ensures every element of out is computed, and there is no missing tile along H. Masks aren’t needed because we loop over all H.

Performance notes
- This per-(b, t) kernel is simple and correct. For larger H and T, it may be slower than a fully tiled GEMM, but correctness is the priority here. Once confirmed, we can improve performance by:
  - Tiling H into blocks and launching a 3D grid over (B, T, H_tiles) with proper masks. This increases parallelism but requires careful handling of boundaries and meta-parameters.
  - Using vectorized loads (e.g., loading b_vec as a 128-element chunk and using tl.dot with scalar a_val) to reduce loop overhead. Triton’s tl.dot requires 2D tiles, so we’d implement a small unrolled dot across K in chunks. This is a next step after correctness.
