import triton
import triton.language as tl


@triton.jit
def concat_by_row_kernel(
    e_ptr,        # *f32, [B, Stext, H]
    h_ptr,        # *f32, [B, Simg, H]
    out_ptr,      # *f32, [B, T, H]
    B, Stext, Simg, H,
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_t, stride_h_h,
    stride_out_b, stride_out_t, stride_out_h,
):
    # 2D grid: (B, T)
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    b = pid_b
    t = pid_t

    # Determine source: encoder if t < Stext else hidden at t - Stext
    is_encoder = t < Stext

    # Base pointers for the row
    if is_encoder:
        e_row_ptr = e_ptr + b * stride_e_b + t * stride_e_t
    else:
        h_row_ptr = h_ptr + b * stride_h_b + (t - Stext) * stride_h_t

    # We need to write to out[b, t, :]
    out_row_ptr = out_ptr + b * stride_out_b + t * stride_out_t

    # Copy the entire hidden vector of length H
    h_vec = tl.arange(0, H)
    # Load from source and store to output
    if is_encoder:
        vals = tl.load(e_row_ptr + h_vec * stride_e_h, mask=h_vec < H, other=0.0)
    else:
        vals = tl.load(h_row_ptr + h_vec * stride_h_h, mask=h_vec < H, other=0.0)
    tl.store(out_row_ptr + h_vec * stride_out_h, vals, mask=h_vec < H)

def matvec_row_kernel(
    in_ptr,       # *f32, [B, T, H]
    wT_ptr,       # *f32, [H, H] (transposed process_weight)
    out_ptr,      # *f32, [B, T, H]

):
    # 2D grid: (B, T)
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    b = pid_b
    t = pid_t

    in_row_ptr = in_ptr + b * stride_in_b + t * stride_in_t
    out_row_ptr = out_ptr + b * stride_out_b + t * stride_out_t

    # Compute out[b, t, :] = sum_k in[b, t, k] * wT[k, :]
    # Loop over k and write all h_out
    for k in range(0, H):
        a_k = tl.load(in_row_ptr + k * stride_in_h)

        h_vec = tl.arange(0, H)

            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),

        )

        # Note: Triton will read process_weight as [H, H] with strides (H, 1)

        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

    wT_ptr,       # *f32, [H, H] (transposed process_weight: rows = input hidden, cols = output hidden)
    out_ptr,      # *f32, [B, T, H] (result)
    B, T, H,

    stride_w_b, stride_w_k, stride_w_h,  # wT is [H, H], k is row index, h is col index

):
