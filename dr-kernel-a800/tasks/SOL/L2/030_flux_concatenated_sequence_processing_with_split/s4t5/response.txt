                               stride_b_i=img.stride(0), stride_i_i=img.stride(1), stride_h_i=img.stride(2),
                               stride_m_a=A.stride(0), stride_h_a=A.stride(1))

    # Now, perform matmul in PyTorch for robustness: C = A @ process_weight.T
    # Ensure process_weight is [H, H]
    weight_T = process_weight.t()
    C = torch.matmul(A, weight_T)  # [M_total, H]

    # Split C into processed_encoder and processed_hidden using Triton copy kernels.
    # processed_encoder: rows [0 : B*T, :]
    # processed_hidden: rows [B*T : M_total, :]
    # We'll implement per-batch copy kernels.

    @triton.jit
    def copy_rows_to_B_T_kernel(
        C_ptr,               # [M_total, H]
        out_ptr,             # [B, T, H]
        B, T, I, H,          # dims
        start_row,            # starting row in C
        stride_c_row, stride_c_col,  # C strides
        out_stride_b, out_stride_t, out_stride_h,  # out strides
        num_warps: tl.constexpr,
    ):
        # Grid over columns and row tiles
        pid_n = tl.program_id(0)
        pid_m = tl.program_id(1)
        offs_n = pid_n * 128 + tl.arange(0, 128)  # column tile
        offs_m = pid_m * 1 + tl.arange(0, 1)      # per-row tile
        mask_n = offs_n < H
        mask_m = offs_m < 1

        # Map C rows to batch and seq in encoder output
        m = start_row + offs_m  # since BLOCK_M=1
        M_total = B * (T + I)
        # We'll implement per-row program: offs_m is a scalar. So, we'll set grid=(T,) and loop over columns internally. To keep it vectorized, we can use grid=(T, H) by passing H as constexpr. However, Triton expects constexpr for tiling. We'll set BLOCK_N=128 and handle H via loops. Triton supports loops. We'll implement while loops.

        # For simplicity, set grid=(T,) and iterate columns in kernel. But Triton kernels are launched with fixed grid. We'll implement grid=(T, 1), and use masks for columns.

        # Since Triton doesn't support arbitrary loops well across entire tensor, we'll implement per-row program: grid=(T,), and inside, we iterate columns in chunks. Triton allows while loops.
        # For clarity, we'll set grid=(T,) and in kernel, iterate over columns.

        # Compute b and seq for each m (scalar)
        # But m is scalar per program; we can't use it to index out. Therefore, we'll implement grid=(T, 1) and use pid_m=0, pid_n iterating columns via loop. Triton allows loops. We'll implement column loop using while with n counter.

        # Implement column loop:
        n = 0
        while n < H:
            offs_n_chunk = n + tl.arange(0, 128)
            mask_n_chunk = offs_n_chunk < H
            # Load C[start_row + m, offs_n_chunk]
            c_ptrs = C_ptr + (start_row + m) * stride_c_row + offs_n_chunk * stride_c_col
            values = tl.load(c_ptrs, mask=mask_n_chunk, other=0.0)
            # Store to out[b, seq, :]
            b = m // T
            seq = m % T
            out_ptrs = out_ptr + b * out_stride_b + seq * out_stride_t + offs_n_chunk * out_stride_h
            tl.store(out_ptrs, values, mask=mask_n_chunk)
            n += 128
