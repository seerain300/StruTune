import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    enc_ptr,            # *const T, [B, T, H]
    hid_ptr,            # *const T, [B, I, H]
    out_ptr,            # *T,       [B, L, H]
    B, T, I, H,         # sizes
    enc_stride_b, enc_stride_t, enc_stride_k,
    hid_stride_b, hid_stride_i, hid_stride_k,
    out_stride_b, out_stride_l, out_stride_k,
    BLOCK_M: tl.constexpr,  # tile along sequence (L) dimension
    BLOCK_N: tl.constexpr,  # tile along feature (H) dimension
):
    # program ids: (batch, tiles over L, tiles over H)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    L = T + I

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # rows in [0, L)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # cols in [0, H)

    mask_m = offs_m < L
    mask_n = offs_n < H

    # Create 2D pointer grids for load/store
    # Load from encoder for rows where offs_m < T, store at out[b, offs_m, offs_n]
    # Load from hidden for rows where offs_m >= T, store at out[b, offs_m, offs_n]
    # We'll build masks for each case and perform masked load/store.

    # 1) encoder part: m indices valid if mask_m & (offs_m < T)
    m_mask_enc = mask_m & (offs_m < T)
    # For these rows, compute source m = offs_m[m_mask_enc]
    # Triton allows elementwise boolean operations; we can compute pointer grids directly.

    # Build pointer grids
    # Note: Triton supports broadcasting of scalar pid_b with vector offsets.
    # We need to construct enc pointers for [BLOCK_M, BLOCK_N] but only for enc rows.
    # Let's first compute the 2D grid of addresses for enc: enc_ptr + pid_b*enc_stride_b + m*enc_stride_t + n*enc_stride_k
    # We'll create a 2D meshgrid: offs_m[:, None], offs_n[None, :]
    m2d = offs_m[:, None]  # shape [BLOCK_M, 1], but Triton will broadcast when used in pointer arithmetic
    n2d = offs_n[None, :]  # shape [1, BLOCK_N]

    # Build enc pointer: base for pid_b and m, then add n offsets
    enc_ptrs = enc_ptr + pid_b * enc_stride_b + m2d * enc_stride_t + n2d * enc_stride_k
    # Mask for enc pointers: m_mask_enc[:, None] & mask_n[None, :]
    enc_mask = m_mask_enc[:, None] & mask_n[None, :]

    # Build out pointers for encoder rows: out_ptr + pid_b*out_stride_b + m2d*out_stride_l + n2d*out_stride_k
    out_ptrs_enc = out_ptr + pid_b * out_stride_b + m2d * out_stride_l + n2d * out_stride_k

    # Perform masked load/store
    # First, load from enc; for m not in encoder range, set to zero (safe since we mask)
    # Triton tl.load requires pointer and mask; we can't branch on scalar; mask covers it.
    enc_vals = tl.load(enc_ptrs, mask=enc_mask, other=0.0)
    tl.store(out_ptrs_enc, enc_vals, mask=enc_mask)

    # 2) hidden part: m indices valid if mask_m & (offs_m >= T)
    m_mask_hid = mask_m & (offs_m >= T)
    # For these rows, source hidden row index is m - T
    # We need to compute pointers for hid[b, m-T, n]
    hid_m = offs_m - T  # vector of length BLOCK_M; negative where m < T, but we mask out those
    hid_ptrs = hid_ptr + pid_b * hid_stride_b + hid_m[:, None] * hid_stride_i + n2d * hid_stride_k
    # Mask for hidden pointers: m_mask_hid[:, None] & mask_n[None, :]
    hid_mask = m_mask_hid[:, None] & mask_n[None, :]

    # Build out pointers for hidden rows: out_ptr + pid_b*out_stride_b + m2d*out_stride_l + n2d*out_stride_k
    # We can reuse out_ptrs_enc; we just need to select the appropriate mask for store. Since enc and hid masks are disjoint (mask_m is split),
    # we can store to out_ptrs_enc for both masks because out_ptrs_enc is correct for both; but we must ensure we don't write twice.
    # Better: construct out_ptrs_hid and out_ptrs_enc separately and use respective masks.

    # However, we need to avoid writing twice. A simple way: after writing enc part, do nothing for enc rows; for hid rows, write to out[b, m, n].
    # We'll compute out_ptrs_hid and store only where hid_mask is true.
    out_ptrs_hid = out_ptr + pid_b * out_stride_b + m2d * out_stride_l + n2d * out_stride_k
    tl.store(out_ptrs_hid, tl.load(hid_ptrs, mask=hid_mask, other=0.0), mask=hid_mask)


@triton.jit
def batched_matmul_kernel(
    A_ptr,       # *const fp32 or fp16, [B, L, H]
    WT_ptr,      # *const fp32 or fp16, [H, H]
    C_ptr,       # *fp32 or fp16,       [B, L, H]
    B, L, H,     # sizes
    A_stride_b, A_stride_m, A_stride_k,
    WT_stride_p, WT_stride_q,  # WT is [P=H, Q=H] -> (stride_p, stride_q)
    C_stride_b, C_stride_m, C_stride_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3D grid: (batch, tiles over M=L, tiles over N=H)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # rows in [0, L)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # cols in [0, H)
    mask_m = offs_m < L
    mask_n = offs_n < H

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (H) in blocks
    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # A[b, m, k] -> [BLOCK_M, BLOCK_K]
        A_ptrs = A_ptr + pid_b * A_stride_b + (offs_m[:, None] * A_stride_m) + (offs_k[None, :] * A_stride_k)
        A_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(A_ptrs, mask=A_mask, other=0.0).to(tl.float32)

        # WT[k, n] -> [BLOCK_K, BLOCK_N]
        WT_ptrs = WT_ptr + offs_k[:, None] * WT_stride_p + offs_n[None, :] * WT_stride_q
        WT_mask = mask_k[:, None] & mask_n[None, :]
        wt = tl.load(WT_ptrs, mask=WT_mask, other=0.0).to(tl.float32)

        # acc += a @ wt
        acc += tl.dot(a, wt)

    # Store back to C[b, m, n] with masks
    C_ptrs = C_ptr + pid_b * C_stride_b + (offs_m[:, None] * C_stride_m) + (offs_n[None, :] * C_stride_n)
    C_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(C_ptrs, acc, mask=C_mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        """
        Triton-optimized forward:
        1) Concatenate encoder_hidden_states and hidden_states along sequence dimension into [B, L, H].
        2) Apply linear projection using a Triton batched GEMM: C = A_cat @ process_weight.T.
        3) Split back into [B, T, H] and [B, I, H].
        """
        # Input checks and shapes
        B = hidden_states.shape[0]
        T = encoder_hidden_states.shape[1]
        I = hidden_states.shape[1]
        H = hidden_states.shape[2]
        assert encoder_hidden_states.shape[2] == H
        assert process_weight.shape[0] == H and process_weight.shape[1] == H, "process_weight must be [H, H]"

        # Make inputs contiguous and prepare dtypes for Triton
        # We'll do computation in fp32 for stability, and cast back to original dtype if needed.
        dtype_in = hidden_states.dtype
        # Concatenation output
        L = T + I
        out_cat = torch.empty((B, L, H), device=hidden_states.device, dtype=dtype_in)

        # Prepare pointers and strides for concat
        # Ensure inputs are contiguous (they likely are, but enforce)
        enc = encoder_hidden_states.contiguous()
        hid = hidden_states.contiguous()

        # Strides
        enc_stride_b, enc_stride_t, enc_stride_k = enc.stride()
        hid_stride_b, hid_stride_i, hid_stride_k = hid.stride()
        out_stride_b, out_stride_l, out_stride_k = out_cat.stride()

        # Launch concat kernel with a 3D grid covering (B, tiles over L, tiles over H)
        BLOCK_M = 64  # tile along L
        BLOCK_N = 64  # tile along H
        grid = (B, triton.cdiv(L, BLOCK_M), triton.cdiv(H, BLOCK_N))
        concat_kernel[grid](
            enc, hid, out_cat,
            B, T, I, H,
            enc_stride_b, enc_stride_t, enc_stride_k,
            hid_stride_b, hid_stride_i, hid_stride_k,
            out_stride_b, out_stride_l, out_stride_k,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        )

        # GEMM: out_cat [B, L, H] @ process_weight.T [H, H] -> C [B, L, H]
        # Convert inputs to fp32 for accumulation; we can store in original dtype
        WT = process_weight.t().contiguous()  # [H, H]
        A = out_cat  # [B, L, H], keep in original dtype, we'll cast to fp32 in kernel via .to(tl.float32)

        # Strides
        A_stride_b, A_stride_m, A_stride_k = A.stride()
        WT_stride_p, WT_stride_q = WT.stride()
        # Output C will be [B, L, H], same dtype as A
        C = torch.empty((B, L, H), device=A.device, dtype=A.dtype)
        C_stride_b, C_stride_m, C_stride_n = C.stride()

        # Kernel launch parameters
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        grid_mm = (B, triton.cdiv(L, BLOCK_M), triton.cdiv(H, BLOCK_N))
        batched_matmul_kernel[grid_mm](
            A, WT, C,
            B, L, H,
            A_stride_b, A_stride_m, A_stride_k,
            WT_stride_p, WT_stride_q,
            C_stride_b, C_stride_m, C_stride_n,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=3,
        )

        # Split back: C has shape [B, L, H], split at T
        processed_encoder = C[:, :T, :]
        processed_hidden = C[:, T:, :]
        return processed_encoder, processed_hidden

    out_ptrs_hid = out_ptr + pid_b * out_stride_b + offs_m[:, None] * out_stride_l + offs_n[None, :] * out_stride_k
    hid_vals = tl.load(hid_ptrs, mask=mask_hid, other=0.0)
    tl.store(out_ptrs_hid, hid_vals, mask=mask_hid)


@triton.jit
def batched_matmul_kernel(
    A_ptr,       # *const, [B, L, H] (original dtype)
    WT_ptr,      # *const, [H, H] (original dtype)
    C_ptr,       # *fp32,  [B, L, H] (fp32 output)
    B, L, H,     # sizes
    A_stride_b, A_stride_m, A_stride_k,
    WT_stride_p, WT_stride_q,  # WT is [P=H, Q=H]
    C_stride_b, C_stride_m, C_stride_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3D grid: (batch, tiles over M=L, tiles over N=H)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # rows in [0, L)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # cols in [0, H)
    mask_m = offs_m < L
    mask_n = offs_n < H

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load A[b, m, k] -> [BLOCK_M, BLOCK_K]
        A_ptrs = A_ptr + pid_b * A_stride_b + (offs_m[:, None] * A_stride_m) + (offs_k[None, :] * A_stride_k)
        A_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(A_ptrs, mask=A_mask, other=0.0).to(tl.float32)  # cast to fp32 for accumulation

        # Load WT[k, n] -> [BLOCK_K, BLOCK_N]
        WT_ptrs = WT_ptr + offs_k[:, None] * WT_stride_p + offs_n[None, :] * WT_stride_q
        WT_mask = mask_k[:, None] & mask_n[None, :]
        wt = tl.load(WT_ptrs, mask=WT_mask, other=0.0).to(tl.float32)  # cast to fp32 for accumulation

        # Accumulate
        acc += tl.dot(a, wt)

    # Store back to C[b, m, n] with masks
    C_ptrs = C_ptr + pid_b * C_stride_b + (offs_m[:, None] * C_stride_m) + (offs_n[None, :] * C_stride_n)
    C_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(C_ptrs, acc, mask=C_mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        """
        Triton-optimized forward:
        1) Concatenate encoder_hidden_states and hidden_states along sequence dimension into [B, L, H].
        2) Apply linear projection using a Triton batched GEMM: C = A_cat @ process_weight.T.
        3) Split back into [B, T, H] and [B, I, H].
        """
        B = hidden_states.shape[0]
        T = encoder_hidden_states.shape[1]
        I = hidden_states.shape[1]
        H = hidden_states.shape[2]
        assert encoder_hidden_states.shape[2] == H
        assert process_weight.shape[0] == H and process_weight.shape[1] == H, "process_weight must be [H, H]"

        # Make inputs contiguous
        enc = encoder_hidden_states.contiguous()
        hid = hidden_states.contiguous()
        # Concatenation output
        L = T + I
        out_cat = torch.empty((B, L, H), device=hidden_states.device, dtype=hidden_states.dtype)

        # Strides for concat
        enc_stride_b, enc_stride_t, enc_stride_k = enc.stride()
        hid_stride_b, hid_stride_i, hid_stride_k = hid.stride()
        out_stride_b, out_stride_l, out_stride_k = out_cat.stride()

        # Launch concat kernel
        BLOCK_M = 64  # tile along L
        BLOCK_N = 64  # tile along H
        grid = (B, triton.cdiv(L, BLOCK_M), triton.cdiv(H, BLOCK_N))
        concat_kernel[grid](
            enc, hid, out_cat,
            B, T, I, H,
            enc_stride_b, enc_stride_t, enc_stride_k,
            hid_stride_b, hid_stride_i, hid_stride_k,
            out_stride_b, out_stride_l, out_stride_k,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        )

        # GEMM: out_cat [B, L, H] @ process_weight.T [H, H] -> C [B, L, H] in fp32
        WT = process_weight.t().contiguous()  # [H, H]
        A = out_cat  # [B, L, H], original dtype

        # Strides
        A_stride_b, A_stride_m, A_stride_k = A.stride()
        WT_stride_p, WT_stride_q = WT.stride()
        C = torch.empty((B, L, H), device=A.device, dtype=torch.float32)  # store fp32 for stability
        C_stride_b, C_stride_m, C_stride_n = C.stride()

        # Kernel launch params
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        grid_mm = (B, triton.cdiv(L, BLOCK_M), triton.cdiv(H, BLOCK_N))
        batched_matmul_kernel[grid_mm](
            A, WT, C,
            B, L, H,
            A_stride_b, A_stride_m, A_stride_k,
            WT_stride_p, WT_stride_q,
            C_stride_b, C_stride_m, C_stride_n,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=3,
        )

        # Split back: C has shape [B, L, H], split at T
        processed_encoder = C[:, :T, :]
        processed_hidden = C[:, T:, :]
        return processed_encoder, processed_hidden
