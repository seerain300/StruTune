What the original PyTorch code computes

Why previous submissions failed

Key insight to simplify and make robust
- The split after the linear projection is fixed by the concatenation order. We can perform the linear projection on the two inputs separately and then sum them. This avoids any need to handle the concatenation in Triton and removes a large source of error. Mathematically:
  - Concatenated = [E | H], where E has shape [B, T, H], H has shape [B, I, H].
  - processed = Concatenated @ W^T = E @ W^T + H @ W^T.

    - processed_encoder = (E @ W^T)[:, :T, :]
    - processed_hidden = (H @ W^T)[:, I:, :]  (Note: after matmul, rows correspond to original positions; splitting at L = T + I gives the correct streams without explicit concat.)
- This formulation keeps the heavy numerical work as two batched matmuls on [B, M, H] where M is T and I, both of which are small compared to H in typical workloads and often multiples of tile sizes, making Triton GEMM more robust.

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr,        # *A, shape [B, M, K]
    WT_ptr,       # *W^T, shape [K, N]
    C_ptr,        # *C, shape [B, M, N]
    B, M, N, K,   # sizes
    stride_A_b, stride_A_m, stride_A_k,
    stride_W_k, stride_W_n,
    stride_C_b, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs: each program computes a tile [BLOCK_M, BLOCK_N] for a given (b, m_block, n_block)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute the offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # rows in M
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # cols in N

    # Masks for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K in tiles
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A[b, offs_m, offs_k] with broadcasting (BLOCK_M, BLOCK_K)
        A_ptrs = A_ptr + pid_b * stride_A_b + offs_m[:, None] * stride_A_m + offs_k[None, :] * stride_A_k
        # Load with masks; other=0.0 ensures OOB elements are zeros
        A_tile = tl.load(A_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        A_tile = A_tile.to(tl.float32)

        # Pointers for W_T[offs_k, offs_n] -> shape (BLOCK_K, BLOCK_N)
        WT_ptrs = WT_ptr + offs_k[:, None] * stride_W_k + offs_n[None, :] * stride_W_n
        WT_tile = tl.load(WT_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        WT_tile = WT_tile.to(tl.float32)

        # acc += A_tile @ WT_tile  (tl.dot accumulates in fp32)
        acc += tl.dot(A_tile, WT_tile)

    # Store results: C[b, offs_m, offs_n]
    C_ptrs = C_ptr + pid_b * stride_C_b + offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n
    # Store with mask on m and n
    tl.store(C_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(
        self,
        hidden_states: torch.Tensor,         # [B, I, H]
        encoder_hidden_states: torch.Tensor, # [B, T, H]
        process_weight: torch.Tensor,        # [H, H]
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-optimized forward:
        - Compute processed_encoder = encoder_hidden_states @ process_weight.T  -> [B, T, H]
        - Compute processed_hidden   = hidden_states        @ process_weight.T  -> [B, I, H]
        - Return the two split tensors directly (no concatenation).
        """
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "Inputs must be on CUDA"
        # Ensure contiguous for predictable strides
        A1 = encoder_hidden_states.contiguous()  # [B, T, H]
        A2 = hidden_states.contiguous()          # [B, I, H]
        WT = process_weight.t().contiguous()     # [H, H]

        B1, T, H1 = A1.shape
        B2, I, H2 = A2.shape
        assert H1 == H2 == WT.shape[0], "Hidden dim mismatch or weight shape"
        assert WT.shape[1] == H1, "Weight output dim mismatch"

        # Output 1: [B, T, H]
        C1 = torch.empty((B1, T, H1), device=A1.device, dtype=torch.float32)  # accumulate/store fp32
        # Output 2: [B, I, H]
        C2 = torch.empty((B2, I, H2), device=A2.device, dtype=torch.float32)

        # Grids
        # Choose tiles that work well for typical dims; Triton will handle masks.
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32

        grid1 = (B1, triton.cdiv(T, BLOCK_M), triton.cdiv(H1, BLOCK_N))
        grid2 = (B2, triton.cdiv(I, BLOCK_M), triton.cdiv(H2, BLOCK_N))

        # Launch kernel for first matmul
        batched_matmul_kernel[grid1](
            A1, WT, C1,
            B1, T, H1, H1,
            A1.stride(0), A1.stride(1), A1.stride(2),
            WT.stride(0), WT.stride(1),
            C1.stride(0), C1.stride(1), C1.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=3,
        )

        # Launch kernel for second matmul
        batched_matmul_kernel[grid2](
            A2, WT, C2,
            B2, I, H2, H2,
            A2.stride(0), A2.stride(1), A2.stride(2),
            WT.stride(0), WT.stride(1),
            C2.stride(0), C2.stride(1), C2.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=3,
        )

        # Cast outputs back to original dtype of hidden states (typically same as process_weight)
        # Note: original run used float32 by default; if you need to match a specific dtype, adjust here.
        processed_encoder = C1
        processed_hidden = C2
        return processed_encoder, processed_hidden

import torch
import torch.nn as nn
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def add_kernel(
    x_ptr,  # Pointer to first input
    y_ptr,  # Pointer to second input
    out_ptr,  # Pointer to output
    n_elements,  # Total number of elements in input/output
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles a contiguous block of data of size BLOCK_SIZE
    block_start = tl.program_id(0) * BLOCK_SIZE
    # Create a range of offsets [0..BLOCK_SIZE-1]
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    # Mask to ensure we don't go out of bounds
    mask = offsets < n_elements
    # Load input values
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    # Perform the elementwise addition
    out = x + y
    # Store the result
    tl.store(out_ptr + offsets, out, mask=mask)

def triton_add(x: torch.Tensor, y: torch.Tensor):
    """
    This function wraps the Triton kernel call. It:
      1. Ensures the inputs are contiguous on GPU.
      2. Calculates the grid (blocks) needed.
      3. Launches the Triton kernel.
    """
    assert x.is_cuda and y.is_cuda, "Tensors must be on CUDA."
    x = x.contiguous()
    y = y.contiguous()

    # Prepare output tensor
    out = torch.empty_like(x)

    # Number of elements in the tensor
    n_elements = x.numel()
    BLOCK_SIZE = 128  # Tunable parameter for block size

    # Determine the number of blocks needed
    grid = lambda meta: ((n_elements + meta["BLOCK_SIZE"] - 1) // meta["BLOCK_SIZE"],)

    # Launch the Triton kernel
    add_kernel[grid](x, y, out, n_elements, BLOCK_SIZE=BLOCK_SIZE)
    return out

class ModelNew(nn.Module):
    def forward(self, a, b):
        # Instead of "return a + b", call our Triton-based addition
        return triton_add(a, b)

  - Inputs: encoder_hidden_states [B, T, H], hidden_states [B, I, H], process_weight [H, H] (no bias).
  - Concatenate along sequence: concatenated = [B, L, H], L = T + I.
  - Linear projection: processed = concatenated @ process_weight.T → [B, L, H].
  - Split: processed_encoder = processed[:, :T, :], processed_hidden = processed[:, T:, :].

- Evaluator constraints:
  - All numerical computation must be performed by Triton kernels. No torch.cat or torch.matmul for the heavy work. Host code can allocate tensors, compute shapes/strides, ensure contiguity, and launch Triton kernels, and return results.

- Why previous submissions had failures:
  - Some runs showed “INCORRECT_NUMERICAL,” likely due to precision differences (fp32 accumulation vs fp16 matmul) or subtle masking/indexing mismatches.
  - Others had “RUNTIME_ERROR,” indicating kernel-level issues such as out-of-bounds access or improper grid sizing. A 3D grid with per-batch program IDs can be tricky for certain shapes and requires careful masking.

- Optimization and correctness strategy:
  - Avoid Triton concatenation. Use the linearity of the projection:
    - processed = (E | H) @ W^T = E @ W^T + H @ W^T.
    - Therefore, compute two batched matmuls:
      - C1 = encoder_hidden_states @ W^T → [B, T, H]
      - C2 = hidden_states @ W^T → [B, I, H]
    - Return (C1, C2). These are exactly the two split streams without ever forming the concatenated tensor.
  - Implement a robust Triton batched GEMM kernel:
    - Grid over (batch, M-tiles, N-tiles), iterate reduction K in tiles (BLOCK_K) with masks.
    - Load tiles from A [B, M, K] and W_T [K, N], cast to fp32, accumulate in fp32, store back to output. Triton will cast appropriately on store to the output pointer’s dtype.
    - Ensure inputs are contiguous and strides are passed correctly.
  - This removes concatenation and minimizes the risk of runtime errors and numerical mismatches, while satisfying the “TRITON-ONLY” requirement.

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr,        # *A, shape [B, M, K]
    WT_ptr,       # *W^T, shape [K, N]
    C_ptr,        # *C, shape [B, M, N]
    B, M, N, K,   # sizes
    stride_A_b, stride_A_m, stride_A_k,
    stride_W_k, stride_W_n,
    stride_C_b, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Each program computes a [BLOCK_M, BLOCK_N] tile for a specific (b, m_block, n_block)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # rows in M
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # cols in N

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[b, offs_m, offs_k] -> (BLOCK_M, BLOCK_K)
        A_ptrs = A_ptr + pid_b * stride_A_b + offs_m[:, None] * stride_A_m + offs_k[None, :] * stride_A_k
        A_tile = tl.load(A_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        A_tile = A_tile.to(tl.float32)

        # Load W_T[offs_k, offs_n] -> (BLOCK_K, BLOCK_N)
        WT_ptrs = WT_ptr + offs_k[:, None] * stride_W_k + offs_n[None, :] * stride_W_n
        WT_tile = tl.load(WT_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        WT_tile = WT_tile.to(tl.float32)

        # Accumulate
        acc += tl.dot(A_tile, WT_tile)

    # Store C[b, offs_m, offs_n]
    C_ptrs = C_ptr + pid_b * stride_C_b + offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n
    tl.store(C_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
