    - Use fp32 accumulation unconditionally to avoid precision pitfalls (even if inputs are fp16/bf16), which is typically what frameworks do for matmul kernels.
    - Use reasonable block sizes along M and N (sequence and hidden dims) to achieve parallelism and coalesced loads; avoid overly large blocks that increase register pressure without clear benefit.
  - Additionally, ensure correctness across all shapes:
    - Use masks for M/N boundaries.
    - Keep tensors contiguous before launching Triton (A and WT), and use strides explicitly in the kernel to avoid any unexpected stride-related issues.
    - Avoid any torch.cat and torch.matmul in the forward; everything is computed by Triton kernels.

```python
import torch
import triton
import triton.language as tl


# Kernel: compute out = A @ WT, where
# A: [B, M, D] (M can be T or I depending on which stream),
# WT: [D, D] (process_weight transposed),
# out: [B, M, D]
# Each program handles a (b, m_block, n_block) tile.
@triton.jit
def _matmul_blocked_reduce8_kernel(
    A_ptr, WT_ptr, Out_ptr,
    B, M, D,
    A_s0, A_s1, A_s2,
    WT_s0, WT_s1,
    Out_s0, Out_s1, Out_s2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    b_pid = tl.program_id(0)
    m_block = tl.program_id(1)
    n_block = tl.program_id(2)

    # Offsets for this tile
    m_offsets = m_block * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offsets = n_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in small chunks to reduce numeric deviation
    for k in range(0, D, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)

        # Load A tile: shape [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + b_pid * A_s0 + m_offsets[:, None] * A_s1 + k_offsets[None, :] * A_s2
        a_mask = (m_offsets[:, None] < M) & (k_offsets[None, :] < D)
        A_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)
        A_tile = A_tile.to(tl.float32)  # accumulate in fp32

        # Load WT tile: shape [BLOCK_K, BLOCK_N]
        wt_ptrs = WT_ptr + k_offsets[:, None] * WT_s0 + n_offsets[None, :] * WT_s1
        wt_mask = (k_offsets[:, None] < D) & (n_offsets[None, :] < D)
        WT_tile = tl.load(wt_ptrs, mask=wt_mask, other=0.0)
        WT_tile = WT_tile.to(tl.float32)

        # Fused multiply-add into acc
        acc += tl.dot(A_tile, WT_tile)

    # Write back results
    out_ptrs = Out_ptr + b_pid * Out_s0 + m_offsets[:, None] * Out_s1 + n_offsets[None, :] * Out_s2
    out_mask = (m_offsets[:, None] < M) & (n_offsets[None, :] < D)
    # Store as fp32 (Out tensor should be fp32 for correctness)
    tl.store(out_ptrs, acc, mask=out_mask)


def _run_triton_blocked(B, M, D, A: torch.Tensor, WT: torch.Tensor, out: torch.Tensor):
    """
    Launch the blocked GEMM kernel. Assumes A is [B, M, D], WT is [D, D], out is [B, M, D].
    All tensors must be contiguous. We accumulate in fp32; out is created as fp32.
    """
    # Ensure contiguity
    A = A.contiguous()
    WT = WT.contiguous()
    # We store output as fp32 for numerical fidelity
    if out.dtype != torch.float32:
        out = out.to(torch.float32)

    # Choose block sizes; small BLOCK_K improves numeric fidelity, moderate BLOCK_M/BLOCK_N for throughput
    # Tuning: if M or D is large, use 64; else 32. BLOCK_N 64 or 128 works; 64 is safer for varying D.
    BLOCK_M = 64 if M >= 64 else 32
    BLOCK_N = 64 if D >= 64 else 32
    BLOCK_K = 8  # small chunk to reduce numeric deviation

    grid = (B, triton.cdiv(M, BLOCK_M), triton.cdiv(D, BLOCK_N))
    _matmul_blocked_reduce8_kernel[grid](
        A, WT, out,
        B, M, D,
        A.stride(0), A.stride(1), A.stride(2),
        WT.stride(0), WT.stride(1),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=2,
    )
    return out


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
                encoder_hidden_states: torch.Tensor,
                process_weight: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-optimized version of the original run:
        1) Compute processed_encoder = encoder_hidden_states @ process_weight.T
        2) Compute processed_hidden = hidden_states @ process_weight.T
        All computation is done by Triton kernels; no torch.cat or torch.matmul is used.
        """
        # Ensure process_weight is transposed [D, D] and contiguous
        WT = process_weight.t().contiguous()  # [D, D]

        # Compute encoder stream: [B, T, D] @ [D, D] -> [B, T, D]
        B, T, D = encoder_hidden_states.shape
        processed_encoder = torch.empty((B, T, D), device=encoder_hidden_states.device, dtype=torch.float32)
        _run_triton_blocked(B, T, D, encoder_hidden_states, WT, processed_encoder)

        # Compute hidden stream: [B, I, D] @ [D, D] -> [B, I, D]
        B2, I, D2 = hidden_states.shape
        assert B2 == B and D2 == D, "Batch and hidden_dim must match across inputs."
        processed_hidden = torch.empty((B, I, D), device=hidden_states.device, dtype=torch.float32)
        _run_triton_blocked(B, I, D, hidden_states, WT, processed_hidden)

        return processed_encoder, processed_hidden
```
