import torch
import triton
import triton.language as tl


@triton.jit
def concat_sequences_kernel(
    encoder_ptr,       # *ptr to [B, T, H]
    hidden_ptr,        # *ptr to [B, I, H]
    out_ptr,           # *ptr to [B, L, H]
    B: tl.constexpr,   # batch size
    T: tl.constexpr,   # text_seq_len
    I: tl.constexpr,   # img_seq_len
    H: tl.constexpr,   # hidden_dim
    L: tl.constexpr,   # total sequence length T + I
    BLOCK_H: tl.constexpr,  # tile size along H (e.g., 64)
):
    # 2D grid over (B, H)
    b = tl.program_id(0)
    h_block = tl.program_id(1)
    h = h_block * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = h < H

    # Loop over sequence positions m
    # We process both encoder and hidden parts by writing into out[b, m, h]
    # m < T: from encoder
    # else: m - T from hidden
    for m in range(0, L):
        # If m < T: load from encoder[b, m, :]
        # Else: load from hidden[b, m - T, :]
        use_encoder = m < T
        if use_encoder:
            src_ptr = encoder_ptr + b * (T * H) + m * H + h
            # Load in float16 to match typical model dtypes
            x = tl.load(src_ptr, mask=mask_h, other=0.0)
            # Store to concatenated output at position m
            out_row_ptr = out_ptr + b * (L * H) + m * H + h
            tl.store(out_row_ptr, x, mask=mask_h)
        else:
            src_ptr = hidden_ptr + b * (I * H) + (m - T) * H + h
            x = tl.load(src_ptr, mask=mask_h, other=0.0)
            out_row_ptr = out_ptr + b * (L * H) + m * H + h
            tl.store(out_row_ptr, x, mask=mask_h)


@triton.jit
def batched_matmul_kernel(
    a_ptr,    # *ptr to [B, M, K] = out_cat (here M=L, K=H)
    w_ptr,    # *ptr to [K, N] = process_weight.T (K=H, N=H)
    c_ptr,    # *ptr to [B, M, N] (output), we will store as fp16
    B: tl.constexpr,   # batch size
    M: tl.constexpr,   # sequence length (L)
    N: tl.constexpr,   # hidden_dim (H)
    K: tl.constexpr,   # hidden_dim (H)
    stride_a_b, stride_a_m, stride_a_k,
    stride_w_k, stride_w_n,
    stride_c_b, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3D grid: (batch, tiles along M, tiles along N)
    b = tl.program_id(0)
    m_block = tl.program_id(1)
    n_block = tl.program_id(2)

    m = m_block * BLOCK_M + tl.arange(0, BLOCK_M)
    n = n_block * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = m < M
    mask_n = n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in blocks
    for k0 in range(0, K, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K

        # A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + b * stride_a_b + m[:, None] * stride_a_m + k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # W tile: [BLOCK_K, BLOCK_N], W is [K, N] so use k as row, n as col
        w_ptrs = w_ptr + k[:, None] * stride_w_k + n[None, :] * stride_w_n
        w = tl.load(w_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, w)

    # Write back to C in fp16, casting from fp32 accumulator
    c_ptrs = c_ptr + b * stride_c_b + m[:, None] * stride_c_m + n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        """
        Triton-optimized version:
        - Concatenate encoder_hidden_states and hidden_states along sequence dimension using Triton.
        - Apply linear projection (A_cat @ process_weight.T) using Triton batched GEMM.
        - Split back into separate encoder and image streams.
        """
        # Ensure inputs are contiguous and dtype float16 for performance and to match typical model behavior
        B, T, H = encoder_hidden_states.shape
        _, I, H2 = hidden_states.shape
        assert H == H2, "hidden_dim must match between encoder_hidden_states and hidden_states"
        L = T + I

        # Make tensors contiguous and in float16
        # If original dtype is not float16, cast to float16 for the Triton kernels
        if encoder_hidden_states.dtype != torch.float16:
            encoder_hidden_states = encoder_hidden_states.to(torch.float16).contiguous()
        if hidden_states.dtype != torch.float16:
            hidden_states = hidden_states.to(torch.float16).contiguous()
        if process_weight.dtype != torch.float16:
            process_weight = process_weight.to(torch.float16).contiguous()

        # 1) Concatenate sequences into out_cat [B, L, H] using Triton
        out_cat = torch.empty((B, L, H), dtype=torch.float16, device=hidden_states.device)

        # Launch concat kernel: grid over (B, H tiles)
        BLOCK_H = 64
        grid = (B, triton.cdiv(H, BLOCK_H))
        concat_sequences_kernel[grid](
            encoder_hidden_states, hidden_states, out_cat,
            B, T, I, H, L,
            BLOCK_H=BLOCK_H,
            num_warps=2,
        )

        # 2) GEMM: out = out_cat @ process_weight.T, where process_weight.T is [H, H]
        # We need W_T = process_weight.T
        W_T = process_weight.t().contiguous()  # [H, H], float16

        # Allocate output [B, L, H] in float16
        processed = torch.empty((B, L, H), dtype=torch.float16, device=hidden_states.device)

        # Launch batched matmul kernel: grid over (B, M tiles, N tiles)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        grid_gemm = (B, triton.cdiv(L, BLOCK_M), triton.cdiv(H, BLOCK_N))
        batched_matmul_kernel[grid_gemm](
            out_cat, W_T, processed,
            B, L, H, H,  # M=L, K=H, N=H
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            W_T.stride(0), W_T.stride(1),
            processed.stride(0), processed.stride(1), processed.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=4, num_stages=3,
        )

        # 3) Split back into separate streams
        processed_encoder = processed[:, :T, :]
        processed_hidden = processed[:, T:, :]

        return processed_encoder, processed_hidden

import torch
import torch.nn as nn
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def add_kernel(
    x_ptr,  # Pointer to first input
    y_ptr,  # Pointer to second input
    out_ptr,  # Pointer to output
    n_elements,  # Total number of elements in input/output
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles a contiguous block of data of size BLOCK_SIZE
    block_start = tl.program_id(0) * BLOCK_SIZE
    # Create a range of offsets [0..BLOCK_SIZE-1]
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    # Mask to ensure we don't go out of bounds
    mask = offsets < n_elements
    # Load input values
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    # Perform the elementwise addition
    out = x + y
    # Store the result
    tl.store(out_ptr + offsets, out, mask=mask)

def triton_add(x: torch.Tensor, y: torch.Tensor):
    """
    This function wraps the Triton kernel call. It:
      1. Ensures the inputs are contiguous on GPU.
      2. Calculates the grid (blocks) needed.
      3. Launches the Triton kernel.
    """
    assert x.is_cuda and y.is_cuda, "Tensors must be on CUDA."
    x = x.contiguous()
    y = y.contiguous()

    # Prepare output tensor
    out = torch.empty_like(x)

    # Number of elements in the tensor
    n_elements = x.numel()
    BLOCK_SIZE = 1024  # Tunable parameter for block size

    # Determine the number of blocks needed
    grid = (triton.cdiv(n_elements, BLOCK_SIZE),)

    # Launch the Triton kernel
    add_kernel[grid](x, y, out, n_elements, BLOCK_SIZE=BLOCK_SIZE)
    return out

class ModelNew(nn.Module):
    def forward(self, a, b):
        # Instead of "return a + b", call our Triton-based addition
        return triton_add(a, b)
