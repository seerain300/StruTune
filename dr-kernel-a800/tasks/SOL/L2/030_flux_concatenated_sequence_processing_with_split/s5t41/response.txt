import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    encoder_ptr,        # *f32, [B, Stext, H]
    hidden_ptr,         # *f32, [B, Simg, H]
    out_ptr,            # *f32, [B, T, H], T = Stext + Simg
    B, Stext, Simg, H,  # runtime integers
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_s, stride_h_h,
    stride_o_b, stride_o_t, stride_o_h,
):

    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    if t < Stext:
        src_t = t
        src = encoder_ptr + b * stride_e_b + src_t * stride_e_t
        dst = out_ptr + b * stride_o_b + t * stride_o_t
    else:
        src_t = t - Stext
        src = hidden_ptr + b * stride_h_b + src_t * stride_h_s
        dst = out_ptr + b * stride_o_b + t * stride_o_t

    for i in range(0, H):
        val = tl.load(src + i * stride_h_h)
        tl.store(dst + i * stride_o_h, val)


@triton.jit
def batched_gemv_kernel(
    in_ptr,          # *f32, concatenated [B, T, H]
    weight_ptr,      # *f32, process_weight [H, H]
    out_ptr,         # *f32, processed [B, T, H]
    B, T, H,         # runtime integers
    stride_in_b, stride_in_t, stride_in_h,
    stride_w_h, stride_w_k,   # weight strides: dim-0 = h (rows), dim-1 = k (cols)
    stride_out_b, stride_out_t, stride_out_h,
    BLOCK_K: tl.constexpr,    # chunk size for input dimension
):
    # Grid is (B, T); each program computes the full output vector for one (b, t)
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    in_row_ptr = in_ptr + b * stride_in_b + t * stride_in_t  # points to concatenated[b, t, :]
    out_row_ptr = out_ptr + b * stride_out_b + t * stride_out_t

    # Accumulator for the output row
    acc = tl.zeros([H], dtype=tl.float32)

    # Loop over input dimension H in chunks of BLOCK_K
    k0 = 0
    while k0 < H:
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # Load input chunk in_row[k_offsets]
        in_chunk = tl.load(in_row_ptr + k_offsets * stride_in_h, mask=mask_k, other=0.0)  # shape [BLOCK_K]

        # Load weight tile [BLOCK_K, H] where columns correspond to output positions (k_offsets, all H)
        # We need weight[k, out_h] for k in k_offsets and out_h in 0..H-1
        # We'll build pointers for each column (out_h) across k_offsets rows.
        # acc will be updated as acc += sum_k in_chunk[k] * weight[k, out_h]
        # We compute this by iterating over k in the chunk and updating acc.
        for kk in range(BLOCK_K):
            k = k_offsets[kk]
            # If kk >= BLOCK_K or k >= H (due to mask), skip; but our while ensures k<H.
            # Load weight[k, :] vector of length H
            w_row_ptrs = weight_ptr + k * stride_w_h + tl.arange(0, H) * stride_w_k
            w_row = tl.load(w_row_ptrs)  # [H]
            # Multiply and accumulate
            acc += in_chunk[kk] * w_row

        k0 += BLOCK_K

    # Store the accumulated output row
    tl.store(out_row_ptr + tl.arange(0, H) * stride_out_h, acc)
