```python
import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    encoder_ptr,       # *f32, [B, Stext, H_in]
    hidden_ptr,        # *f32, [B, Simg, H_in]
    out_ptr,           # *f32, [B, T, H_in], T = Stext + Simg
    B: tl.constexpr,
    Stext: tl.constexpr,
    Simg: tl.constexpr,
    H_in: tl.constexpr,
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_s, stride_h_h,      # hidden_s = t - Stext
    stride_o_b, stride_o_t, stride_o_h,
):
    # Grid: (B, T)
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Choose source based on token position
    if t < Stext:
        src_row = encoder_ptr + b * stride_e_b + t * stride_e_t
        dst_row = out_ptr + b * stride_o_b + t * stride_o_t
        # Copy H_in elements with bounds check
        for i in range(0, H_in):
            # Triton supports scalar loads/stores; index i is a Python-int constexpr in for-loop
            val = tl.load(src_row + i * stride_e_h)
            tl.store(dst_row + i * stride_o_h, val)
    else:
        s = t - Stext
        src_row = hidden_ptr + b * stride_h_b + s * stride_h_s
        dst_row = out_ptr + b * stride_o_b + t * stride_o_t
        for i in range(0, H_in):
            val = tl.load(src_row + i * stride_h_h)
            tl.store(dst_row + i * stride_o_h, val)


@triton.jit
def batched_gemv_kernel(
    in_ptr,            # *f32, concatenated [B, T, H_in]
    weightT_ptr,       # *f32, process_weight.T [H_in, H_out]
    out_ptr,           # *f32, processed [B, T, H_out]
    B, T, H_in, H_out,
    stride_in_b, stride_in_t, stride_in_h,
    stride_w_k, stride_w_n,   # strides for weight_T: dim-0=k, dim-1=n
    stride_out_b, stride_out_t, stride_out_h,
):
    # Grid: (B, T)
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Compute out[b, t, :]
    # Initialize accumulator vector
    acc = [0.0] * H_out

    # Loop over input dimension and accumulate
    for k in range(0, H_in):
        # Load input scalar: in[b, t, k]
        in_val = tl.load(in_ptr + b * stride_in_b + t * stride_in_t + k * stride_in_h)
        # Load weight row slice: weight_T[k, :]
        row_ptr = weightT_ptr + k * stride_w_k
        # Accumulate into acc
        for n in range(0, H_out):
            w_val = tl.load(row_ptr + n * stride_w_n)
            acc[n] += in_val * w_val

    # Store results
    dst_row = out_ptr + b * stride_out_b + t * stride_out_t
    for n in range(0, H_out):
        tl.store(dst_row + n * stride_out_h, acc[n])


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        # Shapes
        B = hidden_states.shape[0]
        Stext = encoder_hidden_states.shape[1]
        Simg = hidden_states.shape[1]
        T = Stext + Simg
        H_in = hidden_states.shape[2]  # hidden_dim
        # Weight is [H_in, H_out]; here H_out == H_in because process_weight is [hidden_dim, hidden_dim]
        H_out = process_weight.shape[1]

        # Ensure tensors are on the same device and dtype float32
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "Tensors must be on CUDA"
        assert hidden_states.dtype == torch.float32 and encoder_hidden_states.dtype == torch.float32 and process_weight.dtype == torch.float32, "Use float32 tensors"
        hidden_states = hidden_states.contiguous()
        encoder_hidden_states = encoder_hidden_states.contiguous()
        process_weight_T = process_weight.t().contiguous()  # weight transposed [H_in, H_out]

        # Allocate concatenated tensor
        concatenated = torch.empty((B, T, H_in), dtype=torch.float32, device=hidden_states.device)

        # Launch concatenation kernel: one program per (b, t)
        grid_concat = (B, T)
        concat_kernel[grid_concat](
            encoder_hidden_states, hidden_states, concatenated,
            B, Stext, Simg, H_in,
            encoder_hidden_states.stride(0), encoder_hidden_states.stride(1), encoder_hidden_states.stride(2),
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
        )

        # Allocate processed tensor
        processed = torch.empty((B, T, H_out), dtype=torch.float32, device=hidden_states.device)

        # Launch GEMV kernel: one program per (b, t)
        grid_gemv = (B, T)
        batched_gemv_kernel[grid_gemv](
            concatenated, process_weight_T, processed,
            B, T, H_in, H_out,
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            process_weight_T.stride(0), process_weight_T.stride(1),
            processed.stride(0), processed.stride(1), processed.stride(2),
        )

        # Split back into separate streams
        processed_encoder = processed[:, :Stext, :]
        processed_hidden = processed[:, Stext:, :]

        return processed_encoder, processed_hidden
```

- Forward does not use any torch matmul on data; it only allocates, launches Triton kernels, and slices outputs.
