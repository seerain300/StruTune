      - Store acc into C at rows m = b*(T+I) + s.
  - Accumulate in fp32, and write results as fp32. Inputs and outputs are contiguous.

- copy_rows_encoder_kernel and copy_rows_hidden_kernel (per-batch):
  - These kernels copy selected rows from C into the respective output tensors processed_encoder and processed_hidden per batch. We use a 2D grid over (batch, tiles of columns H) and iterate over rows within the kernel using a fixed loop bound, which Triton supports. We pass T and I as tl.int32 to the kernels so loops are static at compile time (grid dimension 0 is batch, which we can loop across rows up to B*T or B*I).
  - This avoids torch slicing and ensures Triton-only execution.

Forward method:
- Ensure inputs are contiguous and in fp32 (the original example uses fp32; we can assume the same).
- Allocate C [M, H] as fp32.
- Launch matmul_concat2D_kernel with an appropriate grid.
- Launch per-batch copy kernels to produce processed_encoder [B, T, H] and processed_hidden [B, I, H].
- Return the two outputs.

Final code

import torch
import triton
import triton.language as tl


@triton.jit
def matmul_concat2D_kernel(
    enc_ptr,        # *fp32, [B, T, H]
    img_ptr,        # *fp32, [B, I, H]
    weight_t_ptr,   # *fp32, [H, H]  (process_weight.T)
    C_ptr,          # *fp32, [M, H], M = B*(T+I)
    B: tl.int32, T: tl.int32, I: tl.int32, H: tl.int32,
    stride_b_e, stride_t_e, stride_h_e,
    stride_b_i, stride_i_i, stride_h_i,
    stride_b_c, stride_h_c,
    stride_k_w, stride_n_w,
    # meta-parameters
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Grid: (B, cdiv(T+I, BLOCK_M), cdiv(H, BLOCK_N))
    pid_b = tl.program_id(0)
    pid_s = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_b

    # Tile of sequence indices s in [0, T+I)
    s = pid_s * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    valid_s = s < (T + I)

    # Output columns n
    n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    valid_n = n < H

    # Map sequence index s to source: if s < T -> enc[b, s, :], else -> img[b, s - T, :]
    use_img = s >= T  # [BLOCK_M], boolean-like tensor for masking

    # Accumulator for this tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over H in tiles
    for k0 in range(0, H, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)  # [BLOCK_K]
        valid_k = k < H

        # Build pointer grids for A_tile = A[m, k], where m = b*(T+I) + s
        # enc pointers: b*stride_b_e + s*stride_t_e + k*stride_h_e
        # img pointers: b*stride_b_i + (s - T)*stride_i_i + k*stride_h_i
        enc_ptrs = enc_ptr + b * stride_b_e + s[:, None] * stride_t_e + k[None, :] * stride_h_e  # [BM, BK]
        img_ptrs = img_ptr + b * stride_b_i + (s[:, None] - T) * stride_i_i + k[None, :] * stride_h_i  # [BM, BK]

        # Masks: valid rows, valid k, and source selection
        enc_mask = valid_s[:, None] & valid_k[None, :] & (~use_img[:, None])
        img_mask = valid_s[:, None] & valid_k[None, :] & use_img[:, None]

        # Load A_tile using masked loads
        A_tile = tl.zeros((BLOCK_M, BLOCK_K), dtype=tl.float32)
        A_tile += tl.load(enc_ptrs, mask=enc_mask, other=0.0)
        A_tile += tl.load(img_ptrs, mask=img_mask, other=0.0)

        # Load weight tile: W[k, n]
        W_ptrs = weight_t_ptr + k[:, None] * stride_k_w + n[None, :] * stride_n_w  # [BK, BN]
        W_mask = (k[:, None] < H) & (valid_n[None, :])
        W_tile = tl.load(W_ptrs, mask=W_mask, other=0.0)

        # Accumulate
        acc += tl.dot(A_tile, W_tile)

    # Store results into C[m, n], where m = b*(T+I) + s
    m = b * (T + I) + s  # [BLOCK_M]
    C_ptrs = C_ptr + m[:, None] * stride_b_c + n[None, :] * stride_h_c
    C_mask = valid_s[:, None] & valid_n[None, :]
    tl.store(C_ptrs, acc, mask=C_mask)
