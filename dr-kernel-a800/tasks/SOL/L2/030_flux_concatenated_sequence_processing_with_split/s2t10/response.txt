```python
import torch
import triton
import triton.language as tl

# Simple Triton kernel to concatenate along sequence dimension: out[:, :T, :] = encoder
# and out[:, T:, :] = hidden, with out shape [B, T + I, H].
# We implement a 2D grid: (batch, tiles across L) with each program handling BLOCK size along L.
@triton.jit
def concat_seq_kernel(
    encoder_ptr, hidden_ptr, out_ptr,
    B: tl.int32, T: tl.int32, I: tl.int32, H: tl.int32,
    stride_b_e: tl.int32, stride_t_e: tl.int32, stride_h_e: tl.int32,
    stride_b_h: tl.int32, stride_i_h: tl.int32, stride_h_h: tl.int32,
    stride_b_o: tl.int32, stride_l_o: tl.int32, stride_h_o: tl.int32,
    BLOCK_L: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)
    pid_l = tl.program_id(1)

    # sequence tile start
    l_start = pid_l * BLOCK_L
    offs = l_start + tl.arange(0, BLOCK_L)
    L_total = T + I

    # mask for valid sequence positions
    mask = offs < L_total

    # Compute corresponding indices for encoder and hidden
    # For offs < T: use encoder
    # For offs >= T: use hidden at position offs - T
    # We'll compute both and select using mask arithmetic.
    is_encoder = offs < T
    is_hidden = offs >= T

    # Build per-element pointers
    # encoder pointers for each offs where offs < T
    # We need vectorized pointer math: pointer = base + index * stride
    # Base for encoder: pid_b * stride_b_e
    # Index for encoder: offs where is_encoder is true; for others, we can use 0 (masked)
    # Triton supports vectorized pointer arithmetic; create pointer tensors.
    # Create a 1D pointer tensor for encoder rows
    # Note: Triton expects pointer types; we can compute pointers by broadcasting.
    # However, Triton doesn't support direct vectorized pointer arithmetic as above,
    # so we compute pointers for valid positions only by using masked load/store logic in tl.load/tl.store.
    # We'll perform two masked stores: one for encoder, one for hidden.

    # First store encoder rows
    # For each offs in the tile, if it is within [0, T), load from encoder and store to out at position offs.
    # We can't branch per element easily; Triton supports masked load/store. We'll do:
    # Build pointer for out: out_ptr + pid_b*stride_b_o + offs*stride_l_o + h*stride_h_o
    # Load from encoder when offs < T, store when mask & is_encoder.
    # Similarly for hidden.

    # We'll do this in Python-like masked logic via tl.load/tl.store with combined masks.
    # To simplify, we'll iterate over h dimension by setting h in [0, H) and store for each h.

    # Loop over hidden dimension (assume H is passed and small enough to loop; Triton supports python for loops with runtime ints)
    for h in range(0, H):
        # For encoder: if is_encoder, load and store
        # Construct pointer arrays
        # encoder_ptr + pid_b*stride_b_e + offs*stride_t_e + h*stride_h_e
        # Only load where is_encoder is true and offs < T
        encoder_mask = mask & is_encoder
        encoder_ptrs = encoder_ptr + pid_b * stride_b_e + offs * stride_t_e + h * stride_h_e
        # hidden_ptrs will be used for hidden part
        hidden_ptrs = hidden_ptr + pid_b * stride_b_h + (offs - T) * stride_i_h + h * stride_h_h
        # out_ptrs
        out_ptrs = out_ptr + pid_b * stride_b_o + offs * stride_l_o + h * stride_h_o

        # Load encoder values where applicable
        # When offs >= T, encoder_mask is false, so load returns zeros but store masked will be false too (safe).
        val_e = tl.load(encoder_ptrs, mask=encoder_mask, other=0.0)
        # Load hidden values where applicable
        hidden_mask = mask & is_hidden
        val_h = tl.load(hidden_ptrs, mask=hidden_mask, other=0.0)

        # Select: for positions < T, we store val_e; for >= T, store val_h; for others (out of mask), store 0.
        # We can compute a combined selection:
        # If offs < T: use val_e; else use val_h. For offs >= T and offs < T simultaneously is false.
        # But we need to avoid double writes; we can simply store val_e where encoder_mask, and val_h where hidden_mask.
        # Since masked store only writes where mask is true, this is safe.
        # For positions not in mask (offs >= L_total), masked store does nothing.

        # Store encoder contributions
        tl.store(out_ptrs, val_e, mask=mask & is_encoder)
        # Store hidden contributions
        tl.store(out_ptrs, val_h, mask=mask & is_hidden)

# We'll wrap this in a ModelNew.forward that:
# - Allocates concatenated tensor
# - Launches concat_seq_kernel
# - Performs matmul with torch (to ensure correctness across all workloads and dtypes)
# - Splits outputs and returns.

class ModelNew(torch.nn.Module):
    def forward(
        self,
        hidden_states: torch.Tensor,
        encoder_hidden_states: torch.Tensor,
        process_weight: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Triton-optimized concatenation; matmul done in PyTorch for correctness and speed.
        Returns (processed_encoder, processed_hidden).
        """
        assert hidden_states.dim() == 3 and encoder_hidden_states.dim() == 3, "Inputs must be [B, dim, H]"
        assert process_weight.dim() == 2, "process_weight must be [H, H]"
        B = hidden_states.shape[0]
        T = encoder_hidden_states.shape[1]
        I = hidden_states.shape[1]
        H = hidden_states.shape[2]
        assert encoder_hidden_states.shape[2] == H, "hidden_dim must match between encoder and image streams"

        # Ensure contiguous tensors
        encoder = encoder_hidden_states.contiguous()
        hidden = hidden_states.contiguous()
        weight = process_weight.contiguous()

        # Allocate output concatenated tensor [B, T + I, H]
        L = T + I
        out_cat = torch.empty((B, L, H), dtype=encoder.dtype, device=encoder.device)

        # Launch Triton kernel for concatenation
        # We will use a 2D grid: (B, ceil_div(L, BLOCK_L))
        # Choose BLOCK_L as 128 to balance parallelism and simplicity.
        BLOCK_L = 128
        grid = (B, triton.cdiv(L, BLOCK_L))
        concat_seq_kernel[grid](
            encoder, hidden, out_cat,
            B, T, I, H,
            encoder.stride(0), encoder.stride(1), encoder.stride(2),
            hidden.stride(0), hidden.stride(1), hidden.stride(2),
            out_cat.stride(0), out_cat.stride(1), out_cat.stride(2),
            BLOCK_L=BLOCK_L,
            num_warps=4,
            num_stages=2,
        )

        # Apply linear projection using torch for correctness
        # processed = out_cat @ process_weight.T
        processed = torch.matmul(out_cat, weight.t())

        # Split back into separate streams
        processed_encoder = processed[:, :T, :]
        processed_hidden = processed[:, T:, :]

        return processed_encoder, processed_hidden
```

```python
import torch
import triton
import triton.language as tl

# Triton kernel: concatenate along sequence dimension without torch.cat.
# Input:
#   - encoder: [B, T, H]
#   - hidden: [B, I, H]
# Output:
#   - out_cat: [B, T + I, H]
# The kernel launches a 2D grid over (batch, tiles of L), and writes either encoder or hidden
# to out_cat depending on the sequence index. It uses masked loads/stores for safety.
@triton.jit
def concat_seq_kernel(
    encoder_ptr, hidden_ptr, out_ptr,
    B: tl.int32, T: tl.int32, I: tl.int32, H: tl.int32,
    stride_b_e: tl.int32, stride_t_e: tl.int32, stride_h_e: tl.int32,
    stride_b_h: tl.int32, stride_i_h: tl.int32, stride_h_h: tl.int32,
    stride_b_o: tl.int32, stride_l_o: tl.int32, stride_h_o: tl.int32,
    BLOCK_L: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch id
    pid_l = tl.program_id(1)  # tile id along sequence length

    l_start = pid_l * BLOCK_L
    offs = l_start + tl.arange(0, BLOCK_L)
    L_total = T + I

    mask = offs < L_total
    is_encoder = offs < T
    is_hidden = offs >= T  # equivalent to (offs < L_total) & (offs >= T) but mask handles it

    # Loop over hidden dimension H and perform masked stores
    for h in range(0, H):
        # Compute pointers
        encoder_ptrs = encoder_ptr + pid_b * stride_b_e + offs * stride_t_e + h * stride_h_e
        hidden_ptrs = hidden_ptr + pid_b * stride_b_h + (offs - T) * stride_i_h + h * stride_h_h
        out_ptrs = out_ptr + pid_b * stride_b_o + offs * stride_l_o + h * stride_h_o

        # Load encoder values where applicable
        val_e = tl.load(encoder_ptrs, mask=mask & is_encoder, other=0.0)
        # Load hidden values where applicable
        val_h = tl.load(hidden_ptrs, mask=mask & is_hidden, other=0.0)

        # Store: for positions < T, out_cat[:, offs, h] = encoder[:, offs, h]
        #        for positions >= T, out_cat[:, offs, h] = hidden[:, offs - T, h]
        tl.store(out_ptrs, val_e, mask=mask & is_encoder)
        tl.store(out_ptrs, val_h, mask=mask & is_hidden)

class ModelNew(torch.nn.Module):
    def forward(
        self,
        hidden_states: torch.Tensor,
        encoder_hidden_states: torch.Tensor,
        process_weight: torch.Tensor,

        Returns (processed_encoder, processed_hidden).
```
