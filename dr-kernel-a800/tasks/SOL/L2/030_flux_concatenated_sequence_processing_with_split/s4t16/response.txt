    BLOCK_K: tl.constexpr,               # reduction tile, e.g., 64
):

    b = m // (T + I)
    seq = m % (T + I)

    acc_vec = tl.zeros((BLOCK_N,), dtype=tl.float32)

import triton.language as tl


@triton.jit
def matmul_concat_row_kernel(
    enc_ptr,        # *ptr to [B, T, H]
    img_ptr,        # *ptr to [B, I, H]
    weight_T_ptr,   # *ptr to [H, H]
    out_ptr,        # *ptr to [M_total, H], M_total = B*(T+I)
    B, T, I, H,                 # dims
    stride_b_e, stride_t_e, stride_h_e,  # enc strides
    stride_b_i, stride_i_i, stride_h_i,  # img strides
    stride_m_out, stride_h_out,          # out strides
    stride_m_w, stride_h_w,              # weight strides
    BLOCK_N: tl.constexpr,               # tile over hidden dim, e.g., 128
    BLOCK_K: tl.constexpr,               # reduction tile, e.g., 64
):

    m = tl.program_id(axis=0)
    M_total = B * (T + I)
    # Compute batch and sequence index for this row
    b = m // (T + I)
    seq = m % (T + I)

    # Determine source tensor and row index
    if seq < T:
        src_ptr = enc_ptr + b * stride_b_e + seq * stride_t_e
    else:
        src_ptr = img_ptr + b * stride_b_i + (seq - T) * stride_i_i

    # Accumulator vector for H outputs
    acc_vec = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over hidden_dim in tiles of BLOCK_N
    for n_start in range(0, H, BLOCK_N):
        offs_n = n_start + tl.arange(0, BLOCK_N)
        mask_n = offs_n < H

        a_vec = tl.load(a_ptrs, mask=mask_n, other=0.0)

        # Accumulate dot with weight_T[:, n] for each n in this tile
        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # Load weight_T tile [BLOCK_K, BLOCK_N]
            w_ptrs = weight_T_ptr + offs_k[:, None] * stride_m_w + offs_n[None, :] * stride_h_w
            w_tile = tl.load(w_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            # We implement reduction by summing w_tile * a_vec expanded along k.

            acc_vec += tl.sum(w_tile * a_vec[None, :], axis=0)

@triton.jit
def copy_rows_kernel(
    src_ptr,      # *ptr to [M_total, H]
    dst_ptr,      # *ptr to [B, S, H], where S is T or I
    B, T, I, H, S,   # dims (S is T for encoder, I for hidden)
    start_row,             # starting row in src for dst batch
    stride_m_src, stride_h_src,
    stride_b_dst, stride_s_dst, stride_h_dst,
    BLOCK_N: tl.constexpr,
):
    # Each program handles one row copy: dst batch b, row s in [0, S), column tile n
    pid_row = tl.program_id(axis=0)
    b = pid_row // S
    s = pid_row % S
    src_row = start_row + s

    for n_start in range(0, H, BLOCK_N):
        offs_n = n_start + tl.arange(0, BLOCK_N)
        mask_n = offs_n < H

        src_ptrs = src_ptr + src_row * stride_m_src + offs_n * stride_h_src
        vals = tl.load(src_ptrs, mask=mask_n, other=0.0)

        dst_ptrs = dst_ptr + b * stride_b_dst + s * stride_s_dst + offs_n * stride_h_dst
        tl.store(dst_ptrs, vals, mask=mask_n)

import torch
import triton
import triton.language as tl


@triton.jit
def concat_rows_to_A_kernel(
    enc_ptr,        # *ptr to [B, T, H]
    img_ptr,        # *ptr to [B, I, H]
    A_ptr,          # *ptr to [M_total, H], M_total = B*(T+I)
    B, T, I, H,                 # dims
    stride_b_e, stride_t_e, stride_h_e,  # enc strides
    stride_b_i, stride_i_i, stride_h_i,  # img strides
    stride_m_a, stride_h_a,              # A strides
    BLOCK_H: tl.constexpr,               # tile size for H (e.g., 128)
):
    # Each program handles one output row m in [0, M_total)
    m = tl.program_id(axis=0)
    M_total = B * (T + I)
    b = m // (T + I)
    seq = m % (T + I)

    # Determine source tensor and row index
    if seq < T:
        src_ptr = enc_ptr + b * stride_b_e + seq * stride_t_e
    else:
        seq_img = seq - T
        src_ptr = img_ptr + b * stride_b_i + seq_img * stride_i_i

    # Copy the entire row into A[m, :]
    for n_start in range(0, H, BLOCK_H):
        offs_n = n_start + tl.arange(0, BLOCK_H)
        mask_n = offs_n < H
        vals = tl.load(src_ptr + offs_n * stride_h_e if seq < T else src_ptr + offs_n * stride_h_i, mask=mask_n, other=0.0)
        tl.store(A_ptr + m * stride_m_a + offs_n * stride_h_a, vals, mask=mask_n)

def matmul_concat_row_kernel(
    A_ptr,          # *ptr to [M_total, H]
    weight_T_ptr,   # *ptr to [H, H]
    out_ptr,        # *ptr to [M_total, H]
    B, T, I, H,                 # dims
    stride_m_a, stride_h_a,     # A strides
    stride_m_w, stride_h_w,     # weight_T strides
    stride_m_out, stride_h_out, # out strides
    BLOCK_N: tl.constexpr,      # tile over hidden dim, e.g., 128
    BLOCK_K: tl.constexpr,      # reduction tile, e.g., 64
):
    # Each program handles one output row m in [0, M_total)
    m = tl.program_id(axis=0)
    M_total = B * (T + I)

    # Compute batch and sequence index for this row (same as concat mapping)
    b = m // (T + I)
    seq = m % (T + I)

    # Determine source tensor and row index (for reading A[m, :])
    if seq < T:
        src_ptr = A_ptr + m * stride_m_a
        # Load a_vec = A[m, :]
        a_vec = tl.load(src_ptr + tl.arange(0, H) * stride_h_a)  # H is runtime; Triton allows vectorization here with BLOCK_H=H
        # Note: Triton vector length must be constexpr; instead, iterate over H in tiles:
        # We'll reconstruct by re-reading A per tile. Simpler: re-load segment per BLOCK_H.
        # To avoid complexity, we can compute a_vec by loading the entire row in one shot using BLOCK_H=H.
        # However, Triton requires BLOCK_H constexpr. So we'll compute in tiles and keep a_vec as running acc.
        # Let's redefine: compute per-segment and accumulate into out[m, :].
    else:
        # This branch is not used for computing a_vec; we don't need it since A already contains concatenated rows.
        pass

    # We'll instead load a_vec by reading from A_ptr at row m: concat_rows_to_A ensures row correctness.
    # However, Triton doesn't support dynamic vector length beyond constexpr. So we implement a_vec as:
    # We'll compute the matmul by iterating over hidden tiles. To do that, we need to read A[m, :] in tiles.
    # Since A is [M_total, H], we can load segment of A[m, :] for each n tile and multiply with weight_T.

    # Accumulator vector for H outputs
    acc_vec = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over hidden_dim in tiles of BLOCK_N
    for n_start in range(0, H, BLOCK_N):
        offs_n = n_start + tl.arange(0, BLOCK_N)
        mask_n = offs_n < H

        # Load a_vec segment from A[m, :] for this n tile
        a_ptrs = A_ptr + m * stride_m_a + offs_n * stride_h_a
        a_vec = tl.load(a_ptrs, mask=mask_n, other=0.0)

        # Accumulate dot with weight_T[:, n] for each n in this tile
        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # Load weight_T tile [BLOCK_K, BLOCK_N]
            w_ptrs = weight_T_ptr + offs_k[:, None] * stride_m_w + offs_n[None, :] * stride_h_w
            w_tile = tl.load(w_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            # Multiply and reduce along k: acc_vec += sum_k w_tile[k, :] * a_vec[k]
            acc_vec += tl.sum(w_tile * a_vec[None, :], axis=0)

    # Store the resulting acc_vec to out[m, :]
    out_ptrs = out_ptr + m * stride_m_out + offs_n * stride_h_out
    tl.store(out_ptrs, acc_vec, mask=mask_n)
