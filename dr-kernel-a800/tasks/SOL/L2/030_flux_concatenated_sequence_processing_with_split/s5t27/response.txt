import torch
import triton
import triton.language as tl


@triton.jit
def concat_kernel(
    encoder_ptr,      # *f32, [B, Stext, H]
    hidden_ptr,       # *f32, [B, Simg, H]
    out_ptr,          # *f32, [B, T, H], T = Stext + Simg
    B: tl.constexpr,
    Stext: tl.constexpr,
    Simg: tl.constexpr,
    H: tl.constexpr,
    stride_e_b: tl.constexpr,
    stride_e_t: tl.constexpr,
    stride_e_h: tl.constexpr,
    stride_h_b: tl.constexpr,
    stride_h_s: tl.constexpr,  # s = t - Stext
    stride_h_h: tl.constexpr,
    stride_o_b: tl.constexpr,
    stride_o_t: tl.constexpr,
    stride_o_h: tl.constexpr,
    BLOCK_H: tl.constexpr,     # set to H (or a large multiple) but we use h loop up to H
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Determine source: if t < Stext, take from encoder; else take from hidden at s = t - Stext
    h = tl.arange(0, H)  # vectorize over hidden dimension
    if t < Stext:
        src_ptr = encoder_ptr + b * stride_e_b + t * stride_e_t + h * stride_e_h
    else:
        s = t - Stext
        src_ptr = hidden_ptr + b * stride_h_b + s * stride_h_s + h * stride_h_h

    dst_ptr = out_ptr + b * stride_o_b + t * stride_o_t + h * stride_o_h

    vals = tl.load(src_ptr)  # load entire hidden vector
    tl.store(dst_ptr, vals)  # store into concatenated tensor at position t


@triton.jit
def gemv_row_kernel(
    concat_ptr,       # *f32, [B, T, H_in]
    weightT_ptr,      # *f32, [H_in, H_out] == [H_in, H], here H_in == H_out == H
    out_ptr,          # *f32, [B, T, H_out]
    B: tl.constexpr,
    T: tl.constexpr,
    H_in: tl.constexpr,   # input hidden dimension
    H_out: tl.constexpr,  # output hidden dimension (same as H_in here)
    stride_c_b: tl.constexpr,
    stride_c_t: tl.constexpr,
    stride_c_h: tl.constexpr,
    stride_w_k: tl.constexpr,  # weight_T dim-0 (input H_in)
    stride_w_n: tl.constexpr,  # weight_T dim-1 (output H_out)
    stride_o_b: tl.constexpr,
    stride_o_t: tl.constexpr,
    stride_o_h: tl.constexpr,
    BLOCK_K: tl.constexpr,     # chunk size for input dimension
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    if (b >= B) or (t >= T):
        return

    # Initialize accumulator for this (b, t)
    acc = tl.zeros([H_out], dtype=tl.float32)

    k0 = 0
    while k0 < H_in:
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H_in

        # Load input chunk: concat[b, t, k_offsets] -> vector of length BLOCK_K
        in_ptrs = concat_ptr + b * stride_c_b + t * stride_c_t + k_offsets * stride_c_h
        in_chunk = tl.load(in_ptrs, mask=mask_k, other=0.0)  # [BLOCK_K]

        # Load weight tile: weight_T[k_offsets, 0:H_out] -> [BLOCK_K, H_out]
        # Since H_out == H_in == H, we can loop over n in [0, H_out) for simplicity
        # We accumulate acc[n] += sum_k(in_chunk[k] * weightT[k, n]) for n in 0..H_out-1
        for n in range(0, H_out):
            w_ptrs = weightT_ptr + k_offsets * stride_w_k + n * stride_w_n
            w_chunk = tl.load(w_ptrs, mask=mask_k, other=0.0)  # [BLOCK_K]
            acc[n] += tl.sum(in_chunk * w_chunk, axis=0)

        k0 += BLOCK_K

    # Store the result for this (b, t) row across all H_out
    out_ptrs = out_ptr + b * stride_o_b + t * stride_o_t + tl.arange(0, H_out) * stride_o_h
    tl.store(out_ptrs, acc)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor):
        """
        Triton-optimized version:
        - Concatenate along sequence dimension via Triton
        - Linear projection via Triton per (batch, token)
        - Split back via torch slicing (no compute)
        Returns (processed_encoder, processed_hidden)
        """
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "Inputs must be CUDA tensors"
        assert hidden_states.dtype == torch.float32 and encoder_hidden_states.dtype == torch.float32 and process_weight.dtype == torch.float32, "Use float32"

        # Shapes
        B = hidden_states.shape[0]
        Stext = encoder_hidden_states.shape[1]
        Simg = hidden_states.shape[1]
        H = hidden_states.shape[2]
        T = Stext + Simg

        # Ensure contiguity
        enc = encoder_hidden_states.contiguous()
        hid = hidden_states.contiguous()
        W_T = process_weight.t().contiguous()  # [H, H]
        device = enc.device

        # Allocate concatenated output
        concatenated = torch.empty((B, T, H), dtype=torch.float32, device=device)

        # Launch concatenation kernel
        grid_concat = (B, T)
        concat_kernel[grid_concat](
            enc, hid, concatenated,
            B, Stext, Simg, H,
            enc.stride(0), enc.stride(1), enc.stride(2),
            hid.stride(0), Simg, hid.stride(2),
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            BLOCK_H=H,
            num_warps=4,
            num_stages=2,
        )

        # Allocate processed output
        processed = torch.empty((B, T, H), dtype=torch.float32, device=device)

        # Launch GEMV per (b, t)
        grid_gemv = (B, T)
        gemv_row_kernel[grid_gemv](
            concatenated, W_T, processed,
            B, T, H, H,  # H_in == H_out == H
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            W_T.stride(0), W_T.stride(1),
            processed.stride(0), processed.stride(1), processed.stride(2),
            BLOCK_K=64,
            num_warps=4,
            num_stages=2,
        )

        # Split back
        processed_encoder = processed[:, :Stext, :]
        processed_hidden = processed[:, Stext:, :]
        return processed_encoder, processed_hidden
