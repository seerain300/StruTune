import torch
import triton
import triton.language as tl


# Kernel 1: Concatenate along the sequence dimension using Triton.
# Input:
#   encoder_ptr: [B, Stext, H]
#   hidden_ptr: [B, Simg, H]
# Output:
#   out_ptr: [B, T, H], where T = Stext + Simg
@triton.jit
def concat_seq_kernel(
    encoder_ptr, hidden_ptr, out_ptr,
    B, Stext, Simg, H,
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_t, stride_h_h,
    stride_o_b, stride_o_t, stride_o_h,
    # runtime loop counters
    T,  # T = Stext + Simg
):
    b = tl.program_id(0)
    t = tl.program_id(1)

    # Safety guard (grid may be larger than B or T in some setups)
    if (b >= B) or (t >= T):
        return

    # For t < Stext: write from encoder
    if t < Stext:
        src_t = t
        e = encoder_ptr + b * stride_e_b + src_t * stride_e_t
        o = out_ptr + b * stride_o_b + t * stride_o_t
        # Copy across H dimension
        # H is runtime; do a simple loop
        for i in range(0, H):
            val = tl.load(e + i * stride_e_h)
            tl.store(o + i * stride_o_h, val)
    else:
        # For t >= Stext: write from hidden, shifted by (t - Stext)
        src_t = t - Stext
        h = hidden_ptr + b * stride_h_b + src_t * stride_h_t
        o = out_ptr + b * stride_o_b + t * stride_o_t
        for i in range(0, H):
            val = tl.load(h + i * stride_h_h)
            tl.store(o + i * stride_o_h, val)


# Kernel 2: Batched GEMV using 3D tiling across H_out (BLOCK_H).
# For each (b, t), compute out[b, t, :] = weight_T @ concat[b, t, :].
# weight_T: [H_in, H_out]
# concat:   [B, T, H_in]
# out:      [B, T, H_out]
@triton.jit
def batched_gemv_kernel(
    concat_ptr, weightT_ptr, out_ptr,
    B, T, H_in, H_out,
    stride_c_b, stride_c_t, stride_c_h,
    stride_w_k, stride_w_h,
    stride_o_b, stride_o_t, stride_o_h,
    BLOCK_H: tl.constexpr,  # tile size along output H_out
    BLOCK_K: tl.constexpr,  # tile size along input H_in
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    h_block = tl.program_id(2)

    # Safety guard
    if (b >= B) or (t >= T):
        return

    # Compute the output vector indices for this block
    h_start = h_block * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H_out

    # Initialize accumulator for the BLOCK_H outputs
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # Loop over input dimension in chunks of BLOCK_K
    k0 = 0
    while k0 < H_in:
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H_in

        # Load input vector chunk: concat[b, t, k_offsets]
        # Pointer arithmetic: base for (b, t) then add k_offsets * stride_c_h
        in_vec_ptrs = concat_ptr + b * stride_c_b + t * stride_c_t + k_offsets * stride_c_h
        in_chunk = tl.load(in_vec_ptrs, mask=mask_k, other=0.0)  # shape [BLOCK_K]

        # Load weight matrix block: weightT[k_offsets, h_offsets]
        # Pointer arithmetic: weightT[k, h] with strides (stride_w_k, stride_w_h)
        w_ptrs = weightT_ptr + k_offsets[:, None] * stride_w_k + h_offsets[None, :] * stride_w_h
        w_mask = mask_k[:, None] & mask_h[None, :]
        w_block = tl.load(w_ptrs, mask=w_mask, other=0.0)  # shape [BLOCK_K, BLOCK_H]

        # Accumulate: sum over k of in_chunk[k] * w_block[k, :]
        # Broadcasting: w_block [BK, BH], in_chunk [BK] -> sum over axis=0
        acc += tl.sum(w_block * in_chunk[:, None], axis=0)

        k0 += BLOCK_K

    # Store the accumulated outputs back to out[b, t, h_offsets]
    out_ptrs = out_ptr + b * stride_o_b + t * stride_o_t + h_offsets * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_h)

        - Concatenates encoder_hidden_states and hidden_states along the sequence dimension in Triton.
        - Applies linear projection with process_weight.T using a 3D-tiled Triton GEMV kernel.
        - Splits the result back into encoder and hidden streams.

        Assumes all inputs are CUDA tensors of dtype float32 and contiguous.
        """
        # Ensure device and dtype

        assert hidden_states.dtype == torch.float32 and encoder_hidden_states.dtype == torch.float32 and process_weight.dtype == torch.float32, "Use float32 tensors"
        hidden_states = hidden_states.contiguous()

        B = hidden_states.size(0)
        Stext = encoder_hidden_states.size(1)
        Simg = hidden_states.size(1)
        H = hidden_states.size(2)
        H_in = H  # hidden dimension from both streams
        T = Stext + Simg

        # Prepare concatenated tensor [B, T, H] in Triton
        concatenated = torch.empty((B, T, H), device=hidden_states.device, dtype=hidden_states.dtype)

        grid_concat = (B, T)

            B, Stext, Simg, H,
            encoder_hidden_states.stride(0), encoder_hidden_states.stride(1), encoder_hidden_states.stride(2),
            hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2),
            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),
            T,
            num_warps=1,  # simple copy; small compute
        )

        # Prepare output tensor [B, T, H_in] (same as H)
        out = torch.empty((B, T, H_in), device=hidden_states.device, dtype=hidden_states.dtype)

        # Weight is [H, H]; we need process_weight.T [H_in, H_in]
        weight_T = process_weight.t().contiguous()

        # Launch batched GEMV kernel with tiling. Choose block sizes based on H_in/H_out.
        # We will use BLOCK_H=128 and BLOCK_K=64 by default; these work well for moderate H.
        BLOCK_H = 128
        BLOCK_K = 64
        grid_gemv = (B, T, triton.cdiv(H_in, BLOCK_H))

            B, T, H_in, H_in,  # H_out == H_in

            weight_T.stride(0), weight_T.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            BLOCK_H=BLOCK_H, BLOCK_K=BLOCK_K,
            num_warps=4,  # balance occupancy
        )

        # Split back into encoder and hidden streams
        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

        Stext = encoder_hidden_states.size(1)
        Simg = hidden_states.size(1)

        H_in = H

            concatenated.stride(0), concatenated.stride(1), concatenated.stride(2),

            num_warps=1,  # simple copy op
        )

        out = torch.empty((B, T, H_in), device=hidden_states.device, dtype=hidden_states.dtype)
        weight_T = process_weight.t().contiguous()  # [H, H]

            weight_T.stride(0), weight_T.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            BLOCK_K=BLOCK_K, BLOCK_N=BLOCK_N,
            num_warps=8,
            num_stages=2,
        )

        # 4) Split back into encoder and hidden streams
        processed_encoder = out[:, :Stext, :]
        processed_hidden = out[:, Stext:, :]

        return processed_encoder, processed_hidden
