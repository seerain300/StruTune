import triton
import triton.language as tl


@triton.jit
def matvec_row_kernel(
    input_ptr,         # *f32, in [B, T, H], where T = text_seq_len + img_seq_len
    weightT_ptr,       # *f32, process_weight.T [H, H]
    output_ptr,        # *f32, out [B, T, H]
    B: tl.constexpr,
    T: tl.constexpr,
    H: tl.constexpr,
    stride_in_b: tl.constexpr,
    stride_in_t: tl.constexpr,
    stride_in_h: tl.constexpr,
    stride_w_h: tl.constexpr,       # stride along rows of weightT (dim 0)
    stride_w_k: tl.constexpr,       # stride along cols of weightT (dim 1), typically 1
    stride_out_b: tl.constexpr,
    stride_out_t: tl.constexpr,
    stride_out_h: tl.constexpr,
):
    # Each program handles one (b, t) pair and writes the entire vector h=0..H-1
    b = tl.program_id(0)
    t = tl.program_id(1)

    # Bounds: if b >= B or t >= T, exit (grid ensures b < B, t < T, but keep defensive)
    if (b >= B) or (t >= T):
        return

    # Initialize output vector for this (b, t)
    # We'll store one element at a time to avoid potential Triton vectorization issues.
    for h in range(0, H):
        acc = 0.0
        # Compute the dot: sum_k input[b, t, k] * weightT[k, h]
        for k in range(0, H):
            # Compute linear indices based on strides
            in_offset = b * stride_in_b + t * stride_in_t + k * stride_in_h
            w_offset = k * stride_w_h + h * stride_w_k
            a_val = tl.load(input_ptr + in_offset)
            b_val = tl.load(weightT_ptr + w_offset)
            acc += a_val * b_val
        # Store the result
        out_offset = b * stride_out_b + t * stride_out_t + h * stride_out_h
        tl.store(output_ptr + out_offset, acc)
