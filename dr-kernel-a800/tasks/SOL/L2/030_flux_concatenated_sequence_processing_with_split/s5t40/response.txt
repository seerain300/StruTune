import torch
import triton
import triton.language as tl


@triton.jit
def batched_linear_kernel(
    encoder_ptr,           # *f32, [B, Stext, H]
    hidden_ptr,            # *f32, [B, Simg, H]
    weightT_ptr,           # *f32, [H, H] (process_weight transposed)
    out_ptr,               # *f32, [B, T, H] output
    B, T, Stext, Simg, H,  # sizes
    stride_e_b, stride_e_t, stride_e_h,
    stride_h_b, stride_h_s, stride_h_h,  # hidden: s = t - Stext
    stride_w_k, stride_w_n,                # weight_T strides: dim-0=k, dim-1=n
    stride_o_b, stride_o_t, stride_o_h,
    BLOCK_N: tl.constexpr,  # tile size along output H
    BLOCK_K: tl.constexpr,  # tile size along input H
):
    # Grid: (B, T, ceil_div(H, BLOCK_N))
    b = tl.program_id(0)
    t = tl.program_id(1)
    n_block = tl.program_id(2)
    if (b >= B) or (t >= T):
        return

    # Output tile indices
    n_start = n_block * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < H

    # Accumulator for this (b, t) over the BLOCK_N outputs
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)

    # Loop over input dimension in chunks of BLOCK_K
    for k0 in range(0, H, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < H

        # Determine source tensor: if t < Stext, read from encoder at t; else read from hidden at s = t - Stext
        if t < Stext:
            src_t = t
            in_ptrs = encoder_ptr + b * stride_e_b + src_t * stride_e_t + k_offsets * stride_e_h
        else:
            s = t - Stext
            in_ptrs = hidden_ptr + b * stride_h_b + s * stride_h_s + k_offsets * stride_h_h

        # Load input chunk (vector over k)
        in_chunk = tl.load(in_ptrs, mask=mask_k, other=0.0)  # [BLOCK_K]

        # Load weight tile: shape [BLOCK_K, BLOCK_N]
        w_ptrs = weightT_ptr + k_offsets[:, None] * stride_w_k + n_offsets[None, :] * stride_w_n
        w_tile = tl.load(w_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # [BLOCK_K, BLOCK_N]

        # Accumulate: for each k in chunk, acc[n] += in_chunk[k] * w_tile[k, n]
        # Vectorized across both dimensions
        # Expand in_chunk to [1, BLOCK_K], w_tile to [BLOCK_K, BLOCK_N], then do dot -> [BLOCK_N]
        contrib = tl.dot(in_chunk[:, None], w_tile)  # [BLOCK_N]
        acc += contrib

    # Store results
    out_ptrs = out_ptr + b * stride_o_b + t * stride_o_t + n_offsets * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_n)

        w_tile = tl.load(w_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # [BLOCK_K, BLOCK_N]

        contrib = tl.dot(in_chunk[:, None], w_tile)  # [BLOCK_N]

    out_ptrs = out_ptr + b * stride_out_b + t * stride_out_t + n_offsets * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_n)
