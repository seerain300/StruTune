import torch
import triton
import triton.language as tl


# 1) Triton: pad last dimension (constant 0) - 1D flattened
@triton.jit
def pad_last_dim_kernel(
    inp_ptr,        # *float32, input flattened
    out_ptr,        # *float32, output flattened
    n_in,           # int, number of valid elements in input
    out_len,        # int, total number of elements in output (after padding)
    pad,            # int, pad size added at the end
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask_in = offsets < n_in
    vals = tl.load(inp_ptr + offsets, mask=mask_in, other=0.0)
    tl.store(out_ptr + offsets, vals, mask=mask_in)
    # write zeros for pad region
    for i in range(BLOCK):
        idx = start + i
        if idx >= n_in and idx < out_len:
            tl.store(out_ptr + idx, 0.0)


# 2) Triton: 1D inclusive cumsum along flattened tensor
@triton.jit
def cumsum_1d_kernel(
    inp_ptr,        # *float32
    out_ptr,        # *float32
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    vals = tl.load(inp_ptr + offsets, mask=mask, other=0.0)
    # inclusive scan within the block
    acc = 0.0
    out_vals = tl.zeros([BLOCK], dtype=tl.float32)
    for i in range(BLOCK):
        idx = start + i
        # guard outside n_elements
        if idx < n_elements:
            acc += vals[i]
            out_vals[i] = acc
    tl.store(out_ptr + offsets, out_vals, mask=mask)


# 3) Triton: create lower-triangular mask with diagonal=-1 for [chunk, chunk]
@triton.jit
def tril_mask_kernel(
    out_ptr,           # *bool, shape [chunk, chunk]
    chunk: tl.constexpr,
):
    # We'll fill a chunk x chunk boolean matrix
    # out_ptr is a 1D contiguous buffer of length chunk*chunk
    for i in range(chunk):
        for j in range(chunk):
            # lower triangular if j <= i - 1 (diagonal = -1)
            is_lower = (j <= (i - 1))  # Python ints; j==i also allowed since i-1>=j for j<=i-1
            idx = i * chunk + j
            # Store as int8 1 for True, 0 for False
            val = 1 if is_lower else 0
            tl.store(out_ptr + idx, val)


# 4) Triton: elementwise exp over 1D array
@triton.jit
def exp_kernel(
    in_ptr,      # *float32
    out_ptr,     # *float32
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    vals = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    out_vals = tl.exp(vals)
    tl.store(out_ptr + offsets, out_vals, mask=mask)


# 5) Triton: dense_reduce_G over state_size (s) for each (b, nc, i, j, h)
# Computes G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s]
@triton.jit
def dense_reduce_G_kernel(
    C_ptr,            # *float32, [B, N, Chunk, H, S]
    B_ptr,            # *float32, [B, N, Chunk, H, S]
    G_ptr,            # *float32, [B, N, Chunk, Chunk, H]
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    S_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    i = tl.program_id(axis=2)   # chunk index
    j = tl.program_id(axis=3)   # chunk index
    h = tl.program_id(axis=4)   # head index

    acc = tl.zeros([1], dtype=tl.float32)

    for s0 in range(0, S, S_TILE):
        s_off = s0 + tl.arange(0, S_TILE)
        mask_s = s_off < S

        # C offsets: (((b*N + nc)*Chunk + i)*H + h)*S + s_off
        C_offsets = (((b * N + nc) * Chunk + i) * H + h) * S + s_off
        C_vec = tl.load(C_ptr + C_offsets, mask=mask_s, other=0.0)  # (S_TILE,)

        # B offsets: (((b*N + nc)*Chunk + j)*H + h)*S + s_off
        B_offsets = (((b * N + nc) * Chunk + j) * H + h) * S + s_off
        B_vec = tl.load(B_ptr + B_offsets, mask=mask_s, other=0.0)  # (S_TILE,)

        acc += tl.sum(C_vec * B_vec, axis=0)

    # Store to G[b, nc, i, j, h] at linear index
    G_idx = (b * N + nc) * (Chunk * Chunk * H) + (i * Chunk + j) * H + h
    tl.store(G_ptr + G_idx, acc)


# 6) Triton: dense_reduce_S over chunk_size (t) and head_dim (d) for each (b, nc, h, s)
# Computes S[b, nc, h, s] = sum_{t,d} B_decay[b, nc, t, h, s] * hidden[b, nc, t, h, d]
@triton.jit
def dense_reduce_S_kernel(
    B_decay_ptr,  # *float32, [B, N, Chunk, H, S]
    hidden_ptr,   # *float32, [B, N, Chunk, H, D]
    S_out_ptr,    # *float32, [B, N, H, S]
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    T_TILE: tl.constexpr,
    D_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    h = tl.program_id(axis=2)
    s = tl.program_id(axis=3)

    acc = tl.zeros([1], dtype=tl.float32)

    # reduce over t in tiles
    for t0 in range(0, Chunk, T_TILE):
        t_off = t0 + tl.arange(0, T_TILE)
        mask_t = t_off < Chunk

        # B_decay offsets: (((b*N + nc)*Chunk + t)*H + h)*S + s
        B_offsets = (((b * N + nc) * Chunk + t_off) * H + h) * S + s
        B_vec = tl.load(B_decay_ptr + B_offsets, mask=mask_t, mask=mask_t, other=0.0)  # (T_TILE,)

        # reduce over d in tiles
        for d0 in range(0, D, D_TILE):
            d_off = d0 + tl.arange(0, D_TILE)
            mask_d = d_off < D

            hidden_offsets = (((b * N + nc) * Chunk + t_off[:, None]) * H + h) * D + d_off[None, :]
            mask = mask_t[:, None] & mask_d[None, :]
            hidden_mat = tl.load(hidden_ptr + hidden_offsets, mask=mask, other=0.0)  # (T_TILE, D_TILE)

            col_sums = tl.sum(hidden_mat, axis=1)  # sum over D_TILE for each t -> (T_TILE,)
            acc += tl.sum(B_vec * col_sums, axis=0)

    # Store S[b, nc, h, s]
    S_idx = (b * N + nc) * (H * S) + (h * S + s)
    tl.store(S_out_ptr + S_idx, acc)
