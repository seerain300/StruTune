To comply with the “no torch numerical ops” requirement, we implement three Triton kernels:
- pad_lastdim_kernel: Pads the last dimension of a 3D tensor [B, N1, N3] to N3_pad by constant 0 into a 3D output [B, N1, N3_pad].
- cumsum_last_dim_kernel: Computes cumulative sum along the last dimension for each (b, n1) row of a 4D tensor [B, N1, N2, N3]. This replaces torch.cumsum used on A_chunked_perm.
- segment_sum_lower_tri_exp_kernel: Computes the lower-triangular (diagonal = -1) segment sum per chunk and then applies exp. This replaces torch.tril + cumsum + exp used on A_perm.
- add_inplace_kernel: Performs y += D * hidden_states_padded elementwise. This replaces torch’s elementwise add. It is actually launched in forward.

@triton.jit
def pad_lastdim_kernel(x_ptr, y_ptr,
                        B: tl.int32, N1: tl.int32, N3: tl.int32, N3_pad: tl.int32,
                        in_stride_b: tl.int32, in_stride_n1: tl.int32, in_stride_n3: tl.int32,
                        out_stride_b: tl.int32, out_stride_n1: tl.int32, out_stride_n3: tl.int32):
    """
    Pad the last dimension of x (shape [B, N1, N3]) to N3_pad by constant 0
    into y (shape [B, N1, N3_pad]).
    Each program handles one (b, n1) row and iterates over j from 0 to N3_pad-1.
    If j < N3, copy x[b, n1, j]; else write 0.
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)

    for j in range(0, N3_pad):
        in_offset = b * in_stride_b + n1 * in_stride_n1 + j * in_stride_n3
        out_offset = b * out_stride_b + n1 * out_stride_n1 + j * out_stride_n3
        is_valid = j < N3
        x_val = tl.load(x_ptr + in_offset, mask=is_valid, other=0.0)
        tl.store(y_ptr + out_offset, x_val)


@triton.jit

                            B: tl.int32, N1: tl.int32, N2: tl.int32, N3: tl.int32,

    Compute cumulative sum along the last dimension (N3) for each (b, n1, n2) row of a 4D tensor.
    x_ptr and y_ptr point to contiguous tensors shaped [B, N1, N2, N3].
    Each program handles one (b, n1, n2) row and scans across N3.
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)

    # Initialize running sum
    running = 0.0

    for j in range(0, N3):
        in_offset = b * stride_b + n1 * stride_n1 + n2 * stride_n2 + j * stride_n3
        # Load element; store as float32 to y_ptr
        x_val = tl.load(x_ptr + in_offset)  # x_ptr points to float32
        running += x_val
        out_offset = b * y_ptr.stride(0) + n1 * y_ptr.stride(1) + n2 * y_ptr.stride(2) + j * y_ptr.stride(3)
        tl.store(y_ptr + out_offset, running)

@triton.jit
def add_inplace_kernel(x_ptr, y_ptr, D: tl.float32, B: tl.int32, N1: tl.int32, N3: tl.int32,
                        stride_b: tl.int32, stride_n1: tl.int32, stride_n3: tl.int32):
    """
    Elementwise add y = y + D * x for tensors shaped [B, N1, N3].
    Launch grid: (B, N1, ceil_div(N3, BLOCK)).
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    block = tl.program_id(2)
    offs = block * 1024 + tl.arange(0, 1024)
    mask = offs < N3
    base = b * stride_b + n1 * stride_n1
    x_ptrs = x_ptr + base + offs * stride_n3
    y_ptrs = y_ptr + base + offs * stride_n3
    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
    y_vals = tl.load(y_ptrs, mask=mask, other=0.0)
    y_vals = y_vals + D * x_vals
    tl.store(y_ptrs, y_vals, mask=mask)
