    j = tl.program_id(3)
    h = tl.program_id(4)

        B_val = tl.load(B_addr)

    G_addr = G_ptr + b * g_stride_b + nc * g_stride_nc + i * g_stride_i + j * g_stride_j + h * g_stride_h

@triton.jit
def inter_chunk_propagate(Decay_ptr, States_ptr, NewStates_ptr,
                           Bsz, NC, N, H, Sstate,
                           d_stride_b, d_stride_h, d_stride_i, d_stride_j,
                           s_stride_b, s_stride_nc, s_stride_t, s_stride_h, s_stride_d, s_stride_s,
                           ns_stride_b, ns_stride_nc, ns_stride_t, ns_stride_h, ns_stride_d, ns_stride_s):
    # Grid over (b, i, h, d, s)
    b = tl.program_id(0)
    i = tl.program_id(1)
    h = tl.program_id(2)
    d = tl.program_id(3)
    s = tl.program_id(4)

    acc = tl.zeros((), dtype=tl.float32)
    for j in range(0, NC):
        D_addr = Decay_ptr + b * d_stride_b + h * d_stride_h + i * d_stride_i + j * d_stride_j
        decay_val = tl.load(D_addr)

        St_val = tl.load(S_addr)

@triton.jit
def pad_last_dim_1D(In_ptr, Out_ptr,
                    Bsz, InS, OutS, H, D,
                    in_stride_b, in_stride_s, in_stride_h, in_stride_d,
                    out_stride_b, out_stride_s, out_stride_h, out_stride_d):
    # Grid over (b, s_in, h, d)
    b = tl.program_id(0)
    s_in = tl.program_id(1)
    h = tl.program_id(2)
    d = tl.program_id(3)

    val = tl.load(In_ptr + b * in_stride_b + s_in * in_stride_s + h * in_stride_h + d * in_stride_d)
    tl.store(Out_ptr + b * out_stride_b + s_in * out_stride_s + h * out_stride_h + d * out_stride_d, val)
    for pad in range(InS, OutS):
        tl.store(Out_ptr + b * out_stride_b + pad * out_stride_s + h * out_stride_h + d * out_stride_d, 0.0)
