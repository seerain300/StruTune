High-level plan:
- Triton kernels:
  - cumsum_exp_diff: computes A_cumsum along chunk_size per (b, h) using a Triton scan, then writes exp(A_cumsum[:, :, :, -1:] - A_cumsum).
  - segment_sum_lower_tri_scan: computes inclusive cumsum along rows for each (b, nc, h), masked to lower-triangular (j <= i), and exponentiates; replaces segment_sum.
  - contraction_CxB: computes G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s], replacing einsum 'bcihs,bcjhs->bcijh'.

- Forward:

import torch

import triton.language as tl


@triton.jit
def cumsum_exp_diff(A_ptr, Out_ptr,
                    Bsz, NC, H, N,
                    a_stride_b, a_stride_nc, a_stride_t, a_stride_h,
                    out_stride_b, out_stride_nc, out_stride_t, out_stride_h):
    # Grid over (b, nc, h)
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)

    # Hillis–Steele inclusive scan along t in [0, N)
    acc = tl.zeros((), dtype=tl.float32)
    for t in range(0, N):
        val = tl.load(A_ptr + b * a_stride_b + nc * a_stride_nc + t * a_stride_t + h * a_stride_h)
        acc = acc + val
        tl.store(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h, acc)

    # Now compute exp( last - current ) for each t
    last = tl.load(Out_ptr + b * out_stride_b + nc * out_stride_nc + (N - 1) * out_stride_t + h * out_stride_h)
    for t in range(0, N):
        curr = tl.load(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h)
        diff = last - curr
        tl.store(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h, tl.exp(diff))


@triton.jit
def segment_sum_lower_tri_scan(A_ptr, L_ptr,
                                Bsz, NC, H, N,
                                a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                                l_stride_b, l_stride_nc, l_stride_i, l_stride_j, l_stride_h):
    # Grid over (b, nc, h, i)
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)

    # Inclusive scan along j in [0, N), but only accumulate when j <= i
    acc = tl.zeros((), dtype=tl.float32)
    for j in range(0, N):
        include = j <= i
        val = tl.load(A_ptr + b * a_stride_b + nc * a_stride_nc + j * a_stride_i + h * a_stride_h)
        # if include: val else 0
        val = tl.where(include, val, 0.0)
        acc = acc + val
        # exp(acc) stored in L at (i, j)
        tl.store(L_ptr + b * l_stride_b + nc * l_stride_nc + i * l_stride_i + j * l_stride_j + h * l_stride_h, tl.exp(acc))

    nc = tl.program_id(1)
    i = tl.program_id(2)

@triton.jit
def pad_last_dim_1D(X_ptr, Y_ptr,
                    length_in, pad, length_out,
                    x_stride, y_stride):
    # This kernel pads the last dimension (1D vector) of X into Y by inserting zeros.
    # It expects X and Y to be 1D contiguous tensors of lengths length_in and length_out, with length_out = length_in + pad.
    # We launch grid over output indices.

        tl.store(Y_ptr + idx * y_stride, val)
    else:

        tl.store(Y_ptr + idx * y_stride, 0.0)
