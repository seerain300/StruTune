            hidden_chunked.stride(0), hidden_chunked.stride(1), hidden_chunked.stride(2), hidden_chunked.stride(3), hidden_chunked.stride(4),
            Y_diag.stride(0), Y_diag.stride(1), Y_diag.stride(2), Y_diag.stride(3), Y_diag.stride(4)
        )

        # 7) Reshape to [B, seq_len, H*D] and cast to bfloat16
        output = Y_diag.reshape(batch_size, seq_len, num_heads * head_dim).to(torch.bfloat16)

        # Return output (no final_state)
        return output

@triton.jit
def pad_last_dim_3d(input_ptr, output_ptr, B, I, pad, BLOCK_I: tl.constexpr):
    # input: [B, I], output: [B, I+pad]
    b = tl.program_id(0)
    for i in range(0, I + pad):
        if i < pad:
            tl.store(output_ptr + b * (I + pad) + i, 0.0)
        else:
            j = i - pad
            val = tl.load(input_ptr + b * I + j)
            tl.store(output_ptr + b * (I + pad) + i, val)


@triton.jit
def cumsum_last_dim_4d(x_ptr, y_ptr, B, Nc, I, J, BLOCK_I: tl.constexpr):
    # x_ptr: [B, Nc, I, J], y_ptr: [B, Nc, I, J], cumsum along last dim J
    # Launch grid over (B, Nc, I). Each program does a loop over J.
    b = tl.program_id(0)
    nc = tl.program_id(1)
    i = tl.program_id(2)
    s = 0.0
    for j in range(0, J):
        v = tl.load(x_ptr + b * (Nc * I * J) + nc * (I * J) + i * J + j)
        s += v
        tl.store(y_ptr + b * (Nc * I * J) + nc * (I * J) + i * J + j, s)
