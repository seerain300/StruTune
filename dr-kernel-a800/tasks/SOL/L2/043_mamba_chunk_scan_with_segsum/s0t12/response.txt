                         in_stride_b: tl.int32, in_stride_n1: tl.int32, in_stride_n2: tl.int32,
                         out_stride_b: tl.int32, out_stride_n1: tl.int32, out_stride_n2: tl.int32,
                         BLOCK: tl.constexpr):

    Pad last dimension (N2) of a [B, N1, N2] tensor to N_out, appending zeros.
    out[b, n1, j] = in[b, n1, j] if j < N_in, else 0.
    Grid: (B, N1, ceil_div(N_out, BLOCK))

    n1 = tl.program_id(1)

  - einsum-like kernels for contractions using broadcasting:
    - G_contract_kernel: compute G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s] over state_size, replacing torch.einsum('bcihs,bcjhs->bcijh').
    - states_contract_kernel: compute states[b, nc, h, d, s] = sum_t B[b, nc, t, h, s] * hidden_states[b, nc, t, h, d], replacing torch.einsum('bcths,bcthd->bchds').
    - Y_diag_einsum_kernel: compute Y_diag[b, nc, i, h, d] = sum_j M[b, nc, i, j, h] * hidden_states[b, nc, j, h, d], replacing torch.einsum('bcihs,bcjhd->bcihd').
    - decay_einsum_kernel: compute new_states[b, i, h, d, s] = sum_j decay[b, i, j] * states_with_init[b, j, h, d, s], replacing torch.einsum('bhij,bjhds->bihds').
    - C_times_states_einsum_kernel: compute Y_off[b, nc, t, h, d] = sum_s C[b, nc, t, h, s] * states[b, nc, h, d, s], replacing torch.einsum('bcths,bchds->bcthd').
    - final_add_kernel: compute y = Y_diag + Y_off elementwise, replacing torch addition.
  - add_d_residual_kernel: y += D * hidden_padded elementwise, replacing torch addition.

- Avoid any torch numerical ops in forward; only host-side shape/stride metadata and tensor allocations are used.

@triton.jit
def A_transpose_copy_kernel(A_ptr, A_T_ptr,
                            B: tl.int32, M: tl.int32, N: tl.int32,
                            A_stride_b: tl.int32, A_stride_m: tl.int32, A_stride_n: tl.int32,
                            A_T_stride_b: tl.int32, A_T_stride_m: tl.int32, A_T_stride_n: tl.int32,
                            BLOCK: tl.constexpr):
    """
    Copy A [B, M, N] to A_T [B, N, M] without using torch.transpose.
    Grid: (B, N, ceil_div(M, BLOCK))
    """
    b = tl.program_id(0)
    n = tl.program_id(1)  # destination m index
    chunk = tl.program_id(2)

    m = chunk * BLOCK + tl.arange(0, BLOCK)
    mask = m < M

    in_offset = b * A_stride_b + m * A_stride_m + n * A_stride_n
    out_offset = b * A_T_stride_b + n * A_T_stride_m + m * A_T_stride_n

    vals = tl.load(A_ptr + in_offset, mask=mask, other=0.0)
    tl.store(A_T_ptr + out_offset, vals, mask=mask)

def pad_last_dim_kernel(in_ptr, out_ptr,
                         B: tl.int32, N1: tl.int32, N2: tl.int32, N_in: tl.int32, N_out: tl.int32,
                         in_stride_b: tl.int32, in_stride_n1: tl.int32, in_stride_n2: tl.int32,

    Pad last dimension (N2) of a [B, N1, N2] tensor to N_out, appending zeros.
    out[b, n1, j] = in[b, n1, j] if j < N_in, else 0.
    Grid: (B, N1, ceil_div(N_out, BLOCK))

    tl.store(out_ptr + out_offset, in_vals, mask=mask_out)

def chunk_copy_kernel(in_ptr, out_ptr,
                      B: tl.int32, N1: tl.int32, N2: tl.int32, N3: tl.int32,
                      in_stride_b: tl.int32, in_stride_n1: tl.int32, in_stride_n2: tl.int32, in_stride_n3: tl.int32,
                      out_stride_b: tl.int32, out_stride_n1: tl.int32, out_stride_n2: tl.int32, out_stride_n3: tl.int32,
                      BLOCK: tl.constexpr):
    """
    Copy blocks from in_ptr [B, N1, N2, N3] into out_ptr [B, N1, N2, N3] with block size BLOCK.
    For each (b, n1, n2), copy N3 elements starting at n1*BLOCK to out[b, n1, n2, :].
    Grid: (B, N1, N2)
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)

    base_in = b * in_stride_b + n1 * in_stride_n1 + n2 * in_stride_n2
    base_out = b * out_stride_b + n1 * out_stride_n1 + n2 * out_stride_n2

    offs = tl.arange(0, BLOCK)
    mask = offs < N3
    in_offsets = base_in + (n1 * N3 + offs) * in_stride_n3
    out_offsets = base_out + offs * out_stride_n3
    vals = tl.load(in_ptr + in_offsets, mask=mask, other=0.0)
    tl.store(out_ptr + out_offsets, vals, mask=mask)


@triton.jit
def cumsum_last_dim_kernel(x_ptr, y_ptr,
                            B: tl.int32, N1: tl.int32, N2: tl.int32, N3: tl.int32,
                            x_stride_b: tl.int32, x_stride_n1: tl.int32, x_stride_n2: tl.int32, x_stride_n3: tl.int32,
                            y_stride_b: tl.int32, y_stride_n1: tl.int32, y_stride_n2: tl.int32, y_stride_n3: tl.int32,
                            BLOCK: tl.constexpr):
    """
    Compute cumulative sum along the last dimension (N3) for each row (b, n1, n2).
    x_ptr and y_ptr point to tensors shaped [B, N1, N2, N3].
    Grid: (B, N1, N2)
    Each program processes one row across N3, sequentially.
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)

    run_sum = 0.0
    for j in range(0, N3):
        x_offset = b * x_stride_b + n1 * x_stride_n1 + n2 * x_stride_n2 + j * x_stride_n3
        val = tl.load(x_ptr + x_offset)
        run_sum = run_sum + val
        y_offset = b * y_stride_b + n1 * y_stride_n1 + n2 * y_stride_n2 + j * y_stride_n3
        tl.store(y_ptr + y_offset, run_sum)
