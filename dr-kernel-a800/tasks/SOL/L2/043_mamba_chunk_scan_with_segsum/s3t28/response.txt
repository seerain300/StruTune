# Triton kernel: Compute intra-chunk outputs for y_diag with grid over d.
@triton.jit
def compute_y_diag_kernel_full(B_ptr, C_ptr, L_ptr, hidden_ptr, y_ptr,
                               B, N, T, NH, S, HD,
                               B_stride_b, B_stride_n, B_stride_t, B_stride_nh, B_stride_s,

    n = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    nh = tl.program_id(axis=3)
    d = tl.program_id(axis=4)

    # Accumulate y[b, n, i, nh, d] = sum_j M[b, n, i, j, nh] * hidden[b, n, j, nh, d], where
    # M[b, n, i, j, nh] = sum_s C[b, n, i, nh, s] * B[b, n, j, nh, s] * L[b, nh, n, i, j]
    acc = tl.zeros((1,), dtype=tl.float32)

    j = 0

        G_ij = tl.zeros((1,), dtype=tl.float32)
        s = 0
        while s < S:

            C_val = tl.load(C_ptr + b * C_stride_b + n * C_stride_n + i * C_stride_t + nh * C_stride_nh + s * C_stride_s)
            G_ij += B_val * C_val
            s += 1
        L_val = tl.load(L_ptr + b * L_stride_b + nh * L_stride_nh + n * L_stride_n + i * L_stride_t + j * L_stride_j)
        M_ij = G_ij * L_val
        # Now contribute to y for this d: multiply hidden[b, n, j, nh, d]
        hidden_val = tl.load(hidden_ptr + b * hidden_stride_b + n * hidden_stride_n + j * hidden_stride_t + nh * hidden_stride_nh + d * hidden_stride_d)
        acc += M_ij * hidden_val
        j += 1

    # Store y[b, n, i, nh, d]
    y_addr = y_ptr + b * y_stride_b + n * y_stride_n + i * y_stride_t + nh * y_stride_nh + d * y_stride_d
    tl.store(y_addr, acc)
