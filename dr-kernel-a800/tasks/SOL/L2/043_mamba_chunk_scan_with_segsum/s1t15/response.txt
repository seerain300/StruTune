# Triton kernel: per-(b, nc, h) inclusive scan along vector of length N (chunk_size=256), then exponentiate.
# Used for segment_sum(A_perm) where we mask upper-tri and compute cumsum along rows.
@triton.jit
def inclusive_scan_exp(A_ptr, Out_ptr,
                       Bsz, NC, H, N,
                       a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                       out_stride_b, out_stride_nc, out_stride_i, out_stride_h, out_stride_t):
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)  # row index within chunk_size

    row_ptr = A_ptr + b * a_stride_b + nc * a_stride_nc + i * a_stride_i + h * a_stride_h
    row = tl.load(row_ptr + tl.arange(0, N), mask=tl.arange(0, N) < N, other=0.0)
    # Lower-triangular mask: j > i => zero
    j_offsets = tl.arange(0, N)
    lower_mask = j_offsets <= i
    row = tl.where(lower_mask, row, 0.0)

    acc = row
    offset = 1
    while offset < N:
        shifted = acc[j_offsets - offset]
        shifted = tl.where(j_offsets >= offset, shifted, 0.0)
        acc = acc + shifted
        offset *= 2

    exp_row = tl.exp(acc)
    out_row_ptr = Out_ptr + b * out_stride_b + nc * out_stride_nc + i * out_stride_i + h * out_stride_h
    tl.store(out_row_ptr + tl.arange(0, N) * out_stride_t, exp_row)


# Triton kernel: cumsum_exp_diff across chunk_size N per (b, h). Produces exp(A_cumsum[:, :, :, -1:] - A_cumsum)
@triton.jit
def cumsum_exp_diff(A_ptr, Out_ptr,
                    Bsz, H, N,
                    a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                    out_stride_b, out_stride_nc, out_stride_i, out_stride_h, out_stride_t):
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)

    # Load vector A for this (b, nc, h) across N
    row_ptr = A_ptr + b * a_stride_b + nc * a_stride_nc + h * a_stride_h
    row = tl.load(row_ptr + tl.arange(0, N), mask=tl.arange(0, N) < N, other=0.0)

    # Inclusive scan
    acc = row
    offset = 1
    while offset < N:
        shifted = acc[tl.arange(0, N) - offset]
        shifted = tl.where(tl.arange(0, N) >= offset, shifted, 0.0)
        acc = acc + shifted
        offset *= 2

    # diff = last - current, then exp(diff)
    last = acc[N - 1]
    for t in range(0, N):
        curr = acc[t]
        diff = last - curr
        tl.store(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_i + h * out_stride_h, tl.exp(diff))

# Computes G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s]
@triton.jit
def contraction_CxB(C_ptr, B_ptr, G_ptr,

    b = tl.program_id(0)
    nc = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)

        b_val = tl.load(B_ptr + b * b_stride_b + nc * b_stride_nc + j * b_stride_j + h * b_stride_h + s * b_stride_s)

# grid over (b, h, i) where i iterates over N implicitly; we store per-t diff via loop using out vector.
@triton.jit
def cumsum_exp_diff(A_ptr, Out_ptr,
                    Bsz, H, N,
                    a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                    out_stride_b, out_stride_nc, out_stride_i, out_stride_h, out_stride_t):
    b = tl.program_id(0)
    h = tl.program_id(1)
    nc = tl.program_id(2)  # dummy, we ignore nc here; A_ptr expects (B, S, H)

    # Load vector A for this (b, h) across N
    row_ptr = A_ptr + b * a_stride_b + h * a_stride_h
    row = tl.load(row_ptr + tl.arange(0, N), mask=tl.arange(0, N) < N, other=0.0)

    # Inclusive scan
    acc = row
    offset = 1
    while offset < N:

        shifted = tl.where(tl.arange(0, N) >= offset, shifted, 0.0)
        acc = acc + shifted
        offset *= 2

    # Now Out_ptr[b, nc, i, h] holds exp(acc[i] - acc[0]) per i, computed by loop
    # Store per-t diff as exp(diff) for each i
    # We cannot directly index nc, so we assume out is allocated for a single nc=0 and return per-t exp(diff)
    # This kernel is used to produce per-t decay factor for each position t within the chunk.
    last = acc[N - 1]  # scalar last element of cumsum
    base = Out_ptr + b * out_stride_b  # skip nc since we store per-t

        curr = acc[t]
        diff = last - curr
        tl.store(base + t * out_stride_i + h * out_stride_h, tl.exp(diff))

    acc = tl.zeros((), dtype=tl.float32)
    for s in range(0, S):
        c_val = tl.load(C_ptr + b * c_stride_b + nc * c_stride_nc + i * c_stride_i + h * c_stride_h + s * c_stride_s)
        b_val = tl.load(B_ptr + b * b_stride_b + nc * b_stride_nc + j * b_stride_j + h * b_stride_h + s * b_stride_s)
        acc += c_val * b_val

    g_ptrs = G_ptr + b * g_stride_b + nc * g_stride_nc + i * g_stride_i + j * g_stride_j + h * g_stride_h
    tl.store(g_ptrs, acc)


# Triton kernel: diagonal_output
# Y[b, nc, i, h, d] = sum_j M[b, nc, i, j, h] * hidden[b, nc, j, h, d]
@triton.jit
def diagonal_output(M_ptr, hidden_ptr, Y_ptr,
                    Bsz, NC, H, N, D,
                    m_stride_b, m_stride_nc, m_stride_i, m_stride_j, m_stride_h,
                    h_stride_b, h_stride_nc, h_stride_j, h_stride_h, h_stride_d,
                    y_stride_b, y_stride_nc, y_stride_i, y_stride_h, y_stride_d):
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)
    d = tl.program_id(4)

    acc = tl.zeros((), dtype=tl.float32)
    for j in range(0, N):
        m_val = tl.load(M_ptr + b * m_stride_b + nc * m_stride_nc + i * m_stride_i + j * m_stride_j + h * m_stride_h)
        h_val = tl.load(hidden_ptr + b * h_stride_b + nc * h_stride_nc + j * h_stride_j + h * h_stride_h + d * h_stride_d)
        acc += m_val * h_val

    y_ptrs = Y_ptr + b * y_stride_b + nc * y_stride_nc + i * y_stride_i + h * y_stride_h + d * y_stride_d
    tl.store(y_ptrs, acc)


# Triton kernel: propagate_decay
# new_states[b, i, h, d, s] = sum_j decay_chunk[b, h, i, j] * states_with_init[b, j, h, d, s]
@triton.jit
def propagate_decay(decay_ptr, init_ptr, states_ptr, out_ptr,
                    Bsz, NC, H, N, D, S,
                    d_stride_b, d_stride_h, d_stride_i, d_stride_j, d_stride_k,
                    i_stride_b, i_stride_nc, i_stride_t, i_stride_h, i_stride_d, i_stride_s,
                    o_stride_b, o_stride_nc, o_stride_t, o_stride_h, o_stride_d, o_stride_s):
    b = tl.program_id(0)
    h = tl.program_id(1)
    i = tl.program_id(2)  # target chunk index
    d = tl.program_id(3)
    s = tl.program_id(4)

    acc = tl.zeros((), dtype=tl.float32)
    for j in range(0, N + 1):  # include initial chunk j=0
        decay_val = tl.load(decay_ptr + b * d_stride_b + h * d_stride_h + i * d_stride_i + j * d_stride_j)
        if j == 0:
            # initial state contribution
            init_val = tl.load(init_ptr + b * i_stride_b + i * i_stride_t + h * i_stride_h + d * i_stride_d + s * i_stride_s)
            acc += decay_val * init_val
        else:
            prev_val = tl.load(states_ptr + b * i_stride_b + (j - 1) * i_stride_t + h * i_stride_h + d * i_stride_d + s * i_stride_s)
            acc += decay_val * prev_val

    tl.store(out_ptr + b * o_stride_b + i * o_stride_t + h * o_stride_h + d * o_stride_d + s * o_stride_s, acc)


# Triton kernel: pad last dimension (F.pad equivalent for [B, S, H, D] -> [B, S+pad, H, D])
@triton.jit
def pad_last_dim(src_ptr, dst_ptr,
                 B, S, H, D, pad,
                 src_stride_b, src_stride_s, src_stride_h, src_stride_d,
                 dst_stride_b, dst_stride_s, dst_stride_h, dst_stride_d):
    b = tl.program_id(0)
    s = tl.program_id(1)
    h = tl.program_id(2)
    d = tl.program_id(3)
    if s < S:
        val = tl.load(src_ptr + b * src_stride_b + s * src_stride_s + h * src_stride_h + d * src_stride_d)
        tl.store(dst_ptr + b * dst_stride_b + s * dst_stride_s + h * dst_stride_h + d * dst_stride_d, val)
