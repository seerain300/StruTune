# Triton kernel: compute L = exp(segment_sum(A_perm)), where A_perm is [B, H, Nc, I], segment_sum applies cumsum along I,
# and L[i,j] = exp(sum_{t=0..i} A_perm[b,h,nc,t] for j <= i; otherwise 0. This implements lower-triangular mask (diagonal=-1).
# We will launch this kernel over grid (B, H, Nc, I) and write L to a 5D tensor [B, H, Nc, I, I].
@triton.jit
def segment_sum_lower_tri_exp(in_ptr, out_ptr,
                              B, H, Nc, I: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_nc = tl.program_id(2)
    i = tl.program_id(3)  # we will also use j = tl.program_id(4)
    j = tl.program_id(4)  # second index for output [i,j]
    # guard
    if (pid_b >= 0 and pid_b < B) and (pid_h >= 0 and pid_h < H) and (pid_nc >= 0 and pid_nc < Nc) and (i >= 0 and i < I) and (j >= 0 and j < I):
        # compute inclusive cumsum up to i
        s = 0.0
        for t in range(I):
            ptr = in_ptr + pid_b * (H * Nc * I) + pid_h * (Nc * I) + pid_nc * I + t
            val = tl.load(ptr)
            s += val
            # store s for (i,j) only if j <= i-1 (lower-triangular mask, diagonal=-1)
            if j <= (i - 1):
                out_offset = pid_b * (H * Nc * I * I) + pid_h * (Nc * I * I) + pid_nc * (I * I) + i * I + j
                tl.store(out_ptr + out_offset, tl.exp(s))
            # else store 0
            else:
                out_offset = pid_b * (H * Nc * I * I) + pid_h * (Nc * I * I) + pid_nc * (I * I) + i * I + j
                tl.store(out_ptr + out_offset, 0.0)
