@triton.jit

                        d_stride_b: tl.int32, d_stride_n1: tl.int32, d_stride_n3: tl.int32,
                        MAX_BLOCK: tl.constexpr):
    """
    Compute y += D * hidden for the last dimension N3 in blocks of MAX_BLOCK.
    y shape [B, N1, N3], D shape [1, 1, N3] (broadcasted over batch and n1).
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)

    base_y = b * y_stride_b + n1 * y_stride_n1

    # Load D values for this block (broadcast across b, n1)
    d_base = 0 * d_stride_b + 0 * d_stride_n1  # D is [1,1, N3], so index (0,0, :)
