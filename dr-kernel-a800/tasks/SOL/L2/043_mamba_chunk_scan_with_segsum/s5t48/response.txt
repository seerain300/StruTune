import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def pad_last_dim_kernel(out_ptr, in_ptr, B, S, D_in, D_out, PAD_SIZE, K: tl.constexpr):
    # in_ptr: [B, S, D_in]
    # out_ptr: [B, S, D_out], D_out = D_in + PAD_SIZE
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    if b >= B:
        return
    d = tl.arange(0, K)
    d_out = d + PAD_SIZE
    in_idx = ((b * S + s) * D_in) + d
    out_idx = ((b * S + s) * D_out) + d_out
    vals = tl.load(in_ptr + in_idx)
    tl.store(out_ptr + out_idx, vals)


@triton.jit
def hidden_to_chunks_kernel(out_ptr, in_ptr, B, S_padded, D_in, NC, K, H, D, PAD_SIZE):
    # in_ptr: [B, S_padded, D_in + PAD_SIZE]

    b = pid // (NC * K * H * D)

    nc = rem // (K * H * D)
    rem2 = rem % (K * H * D)

    h = (rem2 % (H * D)) // D
    d = rem2 % D
    if b >= B or nc >= NC or t >= K or h >= H or d >= D:
        return
    # Map t to the chunk index and original d (padded)

    t_in = t
    d_in = d + PAD_SIZE

    val = tl.load(in_ptr + src)
    tl.store(out_ptr + dst, val)


@triton.jit
def write_final_state_kernel(final_ptr, B, H, D):

    pid = tl.program_id(0)
    # We launch grid=(B * H * D,)

    rem = pid % (H * D)
    h = rem // D
    d = rem % D
    if b >= B or h >= H or d >= D:
        return
    zero = 0.0
    tl.store(final_ptr + (b * H + h) * D + d, zero)


class ModelNew(nn.Module):

        dtype_in = hidden_states.dtype

        D_out = D + PAD_SIZE
        out_pad = torch.empty((B_size, S, D_out), dtype=torch.float32, device=device)

        grid_pad = (B_size * S,)
        pad_last_dim_kernel[grid_pad](out_pad, hidden_states_f, B_size, S, D, D_out, PAD_SIZE)

        # 2) Reshape into chunks [B, NC, 256, H, D]
        S_padded = S + PAD_SIZE
        NC = (S_padded + 255) // 256  # number of chunks
        K = 256
        out_chunks = torch.empty((B_size, NC, K, H, D), dtype=torch.float32, device=device)

            out_pad,
