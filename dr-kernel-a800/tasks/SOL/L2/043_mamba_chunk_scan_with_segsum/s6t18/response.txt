import torch
import torch.nn as nn
import triton
import triton.language as tl


# 1) Pad sequence along last dim: out[b, pos] = in[b, pos] if pos < L else 0
@triton.jit
def pad_seq_kernel(
    in_ptr,            # *float32, input [B, L], contiguous
    out_ptr,           # *float32, output [B, L_out], contiguous
    B: tl.constexpr,   # batch size
    L: tl.constexpr,   # original seq_len
    L_out: tl.constexpr,  # padded seq_len
    pad_right: tl.constexpr  # number of zeros to append on the right
):
    b = tl.program_id(0)
    pos = tl.program_id(1)
    if pos < L:
        val = tl.load(in_ptr + b * L + pos)
        tl.store(out_ptr + b * L_out + pos, val)
    else:
        tl.store(out_ptr + b * L_out + pos, 0.0)


# 2) Lower-triangular mask: out[i, j] = 1 if i >= j - diagonal, else 0
@triton.jit
def lower_tri_mask_kernel(
    out_ptr,           # *float32, output [I, I] contiguous
    I: tl.constexpr,   # chunk_size
    diagonal: tl.constexpr  # -1
):
    rows = tl.arange(0, I)
    cols = tl.arange(0, I)
    row_idx = rows[:, None]  # [I, 1]
    col_idx = cols[None, :]  # [1, I]
    cond = (row_idx >= (col_idx + diagonal))  # diagonal=-1 => i >= j
    out_val = tl.where(cond, 1.0, 0.0)
    offsets = row_idx * I + col_idx
    tl.store(out_ptr + offsets, out_val)


# 3) Inclusive per-row cumsum over a 2D buffer [I, I]
@triton.jit
def per_row_cumsum_kernel(
    in_ptr,            # *float32, input [I, I], contiguous
    out_ptr,           # *float32, output [I, I], contiguous
    I: tl.constexpr
):
    r = tl.program_id(0)  # row index
    cols = tl.arange(0, I)
    in_row_ptr = in_ptr + r * I + cols
    out_row_ptr = out_ptr + r * I + cols
    acc = 0.0
    for j in range(0, I):
        val = tl.load(in_row_ptr + j)
        acc = acc + val
        tl.store(out_row_ptr + j, acc)


# 4a) Elementwise exp per row of a matrix: out[i, j] = exp * in[i, j]
@triton.jit
def elementwise_exp_rows_kernel(
    in_ptr,            # *float32, input [I, I], contiguous
    out_ptr,           # *float32, output [I, I], contiguous
    I: tl.constexpr,
    exp_value  # scalar float32
):
    r = tl.program_id(0)  # row index
    cols = tl.arange(0, I)
    row_in_ptr = in_ptr + r * I + cols
    row_out_ptr = out_ptr + r * I + cols
    for j in range(0, I):
        val = tl.load(row_in_ptr + j) * exp_value
        tl.store(row_out_ptr + j, val)


# 5) Contraction einsum('bcihs,bcjhs->bcijh') implemented over B and C per chunk:
#    We pass B_flat and C_flat with layout [B*N*I*H*S], and output G_flat [B*N*I*I*H]
@triton.jit
def contraction_kernel(
    B_flat_ptr,        # *float32, flattened B per chunk
    C_flat_ptr,        # *float32, flattened C per chunk
    G_flat_ptr,        # *float32, flattened G output
    N: tl.constexpr,   # number of chunks
    I: tl.constexpr,   # chunk_size
    H: tl.constexpr,   # num_heads (assumed 1 in original)
    S: tl.constexpr,   # state_size
    diagonal: tl.constexpr  # -1
):
    # Grid: (B, N, I, I, H)
    b = tl.program_id(0)
    nc = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)
    # Offsets for B and C over flattened dimension
    # B layout: [B*N*I*H*S], index = ((b*N + nc)*I + i)*H*S + h*S + s
    # C layout: [B*N*I*H*S], index = ((b*N + nc)*I + j)*H*S + h*S + s
    # G layout: [B*N*I*I*H], index = ((b*N)*I*I + nc*I*I + i*I + j)*H + h
    # Loop over s in [0, S)
    g_idx = ((b * N + nc) * I * I + i * I + j) * H + h
    sum_val = 0.0
    for s in range(0, S):
        b_idx = ((b * N + nc) * I * H + i * H + s) * S + h * S + s
        c_idx = ((b * N + nc) * I * H + j * H + s) * S + h * S + s
        b_val = tl.load(B_flat_ptr + b_idx)
        c_val = tl.load(C_flat_ptr + c_idx)
        sum_val = sum_val + b_val * c_val
    tl.store(G_flat_ptr + g_idx, sum_val)


# 6) Diagonal output term Y_diag = einsum('bcijh,bcjhd->bcihd') implemented per (b, nc, h, d):
#    Grid spans (B, N, H, D). For each (b, nc, h, d), loop over i in [0, I) and accumulate j.
@triton.jit
def y_diag_triton_kernel(
    M_flat_ptr,        # *float32, flattened M with layout [B*N*I*I*H]
    V_flat_ptr,        # *float32, flattened V with layout [B*N*I*H*D]
    Y_flat_ptr,        # *float32, flattened Y with layout [B*N*I*H*D]
    B: tl.constexpr,   # batch size
    N: tl.constexpr,   # number of chunks
    I: tl.constexpr,   # chunk_size
    H: tl.constexpr,   # num_heads
    D: tl.constexpr,   # head_dim
    diagonal: tl.constexpr  # -1 (unused, but kept for signature)
):
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)
    d = tl.program_id(3)
    # M index: ((b*N + nc)*I*I + i*I + j)*H + h
    # V index: ((b*N + nc)*I*H*D + i*(H*D) + j*(H*D) + h*D + d)
    # Y index: same as V
    # Loop over i then j to compute sum_j M[i, j, h] * V[j, h, d]
    for i in range(0, I):
        acc = 0.0
        for j in range(0, I):
            m_idx = ((b * N + nc) * I * I + i * I + j) * H + h
            v_idx = ((b * N + nc) * I * H * D + i * (H * D) + j * (H * D) + h * D + d)
            M_val = tl.load(M_flat_ptr + m_idx)
            V_val = tl.load(V_flat_ptr + v_idx)
            acc = acc + M_val * V_val
        Y_idx = v_idx
        tl.store(Y_flat_ptr + Y_idx, acc)


# 7) Cumsum along rows for a matrix [I, I] per (b, h, nc)
@triton.jit
def cumsum_rows_kernel(
    in_ptr,            # *float32, input [I, I], contiguous
    out_ptr,           # *float32, output [I, I], contiguous
    I: tl.constexpr
):
    r = tl.program_id(0)  # row index
    cols = tl.arange(0, I)
    in_row_ptr = in_ptr + r * I + cols
    out_row_ptr = out_ptr + r * I + cols
    acc = 0.0
    for j in range(0, I):
        val = tl.load(in_row_ptr + j)
        acc = acc + val
        tl.store(out_row_ptr + j, acc)


# 8) Elementwise exp per row of a matrix: out[i, j] = exp * in[i, j]
#    Same as 4a, used for inter-chunk decay
@triton.jit
def elementwise_exp_rows_kernel(
    in_ptr,            # *float32, input [I, I], contiguous
    out_ptr,           # *float32, output [I, I], contiguous
    I: tl.constexpr,
    exp_value  # scalar float32
):
    r = tl.program_id(0)  # row index
    cols = tl.arange(0, I)
    row_in_ptr = in_ptr + r * I + cols
    row_out_ptr = out_ptr + r * I + cols
    for j in range(0, I):
        val = tl.load(row_in_ptr + j) * exp_value
        tl.store(row_out_ptr + j, val)


# 9) D residual: elementwise multiplication of padded hidden by D per (b, l, h, d)
@triton.jit
def d_residual_kernel(
    Hid_ptr,           # *float32, padded hidden [B, L_out, H, D], contiguous
    D_ptr,             # *float32, D tensor [B, 1, 1, D], contiguous
    Out_ptr,           # *float32, output [B, L_out, H, D], contiguous
    B: tl.constexpr,   # batch size
    L_out: tl.constexpr,  # padded seq_len
    H: tl.constexpr,   # num_heads
    D_dim: tl.constexpr  # head_dim
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    h = tl.program_id(2)
    d = tl.program_id(3)
    Hid_val = tl.load(Hid_ptr + b * (L_out * H * D_dim) + l * (H * D_dim) + h * D_dim + d)
    D_val = tl.load(D_ptr + b * D_dim + d)  # D has shape [B, 1, 1, D]
    Out_val = Hid_val * D_val
    tl.store(Out_ptr + b * (L_out * H * D_dim) + l * (H * D_dim) + h * D_dim + d, Out_val)


class ModelNew(nn.Module):
    def __init__(self):

        # No parameters; everything is computed in kernels

        batch_size, seq_len, num_heads, head_dim = hidden_states.shape
        state_size = 256
        n_groups = 1

        B_expanded = B.expand(batch_size, seq_len, num_heads, state_size)

        D_flat = D.contiguous().view(batch_size, 1, 1, head_dim)

        grid_D = (batch_size, seq_len_padded, num_heads, head_dim)
