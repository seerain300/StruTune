def pad_last_dim_1D(From_ptr, To_ptr,
                    B, S, D, S_padded,
                    from_stride_b, from_stride_s, from_stride_d,
                    to_stride_b, to_stride_sp, to_stride_d):
    # Each program copies one element from (b, s, d) to (b, s_padded, d) where s_padded = s + pad.
    # pad = S_padded - S
    pad = S_padded - S
    pid = tl.program_id(0)  # linear id over B*S*D
    # compute b, s, d from pid
    # Note: we assume S <= S_padded; for s >= S, we set d = 0
    b = pid // (S * D)
    tmp = pid % (S * D)
    s = tmp // D
    d = tmp % D
    s_src = s
    s_dst = s + pad
    # bounds check for source s < S
    in_bounds = s < S
    val = tl.load(From_ptr + b * from_stride_b + s_src * from_stride_s + d * from_stride_d, mask=in_bounds, other=0.0)
    tl.store(To_ptr + b * to_stride_b + s_dst * to_stride_sp + d * to_stride_d, val, mask=in_bounds)

    i = tl.program_id(0)

    # G[i, j] = sum_s C[i, s] * B[j, s]
    # Since N and S_state are small (256), we can loop over them.
    # Initialize accumulator vector of length N

    # Loop over j
    for j in range(0, N):
        acc = tl.zeros((), dtype=tl.float32)
        # Loop over s
        for s in range(0, S_state):
            c_val = tl.load(C_ptr + i * C_stride_i + s * C_stride_s)
            b_val = tl.load(B_ptr + j * B_stride_j + s * B_stride_s)
            acc += c_val * b_val
        G_vec[j] = acc
    # Store G_vec to G[i, :]
    # We assume G_ptr layout is [Bsz, N] with G_stride_i and G_stride_j
    # However, our grid uses only Bsz rows, and we compute vectors for each. If Bsz>1, we need to map to which b,
    # but the original reshape leads to per-batch chunks; here, inputs to this kernel are specific to each b
    # in the calling function. We'll store per i across the batch dimension handled by the caller.
    # For simplicity, store as (i, 0) in G_ptr; caller manages B dimension. If Bsz > 1, we need an extra dimension.

    # But Triton kernel doesn't receive Bsz; caller will pass a pointer to (Bsz, N) and we store per i across rows.
    # So we compute base = i * G_stride_i, then store G_vec[j] at base + j * G_stride_j.
    base = i * G_stride_i

        device = hidden_states.device

        # Build mask of shape [S_padded, S_padded]
        diag_mask = torch.tril(torch.ones((S_padded, S_padded), device=device, dtype=torch.bool), diagonal=-1)

        # We’ll use torch.einsum for correctness here.

        # we’ll compute G using torch.einsum.

        return output, final_state
