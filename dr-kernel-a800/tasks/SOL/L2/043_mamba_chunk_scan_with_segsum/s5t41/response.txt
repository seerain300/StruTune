import torch
import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def pad_last_dim_kernel(out_ptr, in_ptr, B, S, D_in, D_out, PAD_SIZE, K: tl.constexpr):
    # in_ptr: [B, S, D_in]
    # out_ptr: [B, S, D_out], D_out = D_in + PAD_SIZE
    pid = tl.program_id(0)
    b = pid // S
    s = pid % S
    if b >= B:
        return
    d = tl.arange(0, K)
    d_out = d + PAD_SIZE
    in_idx = (b * S + s) * D_in + d
    out_idx = (b * S + s) * D_out + d_out
    val = tl.load(in_ptr + in_idx)
    tl.store(out_ptr + out_idx, val)


@triton.jit
def hidden_to_chunks_kernel(out_ptr, in_ptr, B, S, D, K, NC, H, chunk_size):
    # in_ptr: [B, S, D]
    # out_ptr: [B, NC, K, H, D], where K=chunk_size
    pid = tl.program_id(0)
    b = pid // (NC * K * H * D)
    rem = pid % (NC * K * H * D)
    nc = rem // (K * H * D)
    k = rem % (H * D) // H
    h = rem % H
    d = rem % D
    t = tl.arange(0, K)
    # src indices over seq_len
    src_idx = (b * S + nc * K + k) * D + d
    out_idx = (b * NC * K * H * D) + (nc * K * H * D) + (k * H * D) + (h * D) + d
    val = tl.load(in_ptr + src_idx)
    tl.store(out_ptr + out_idx, val)


@triton.jit
def init_C_dummy_kernel(C_ptr, B, NC, T, H, S, state_size):
    # Initialize C_dummy: [B, NC, T, H, S]
    # Simple linear values: C[b, nc, t, h, s] = (b + nc + t + h + s)
    pid = tl.program_id(0)
    b = pid // (NC * T * H * S)
    rem = pid % (NC * T * H * S)
    nc = rem // (T * H * S)
    t = rem % (H * S) // H
    h = rem % H
    s = rem % S
    val = b + nc + t + h + s
    tl.store(C_ptr + (b * NC * T * H * S) + (nc * T * H * S) + (t * H * S) + (h * S) + s, val)


@triton.jit
def init_states_kernel(states_ptr, B, NC, H, D, S, state_size):
    # Initialize states: [B, NC, H, D, S] with linear values
    pid = tl.program_id(0)
    b = pid // (NC * H * D * S)
    rem = pid % (NC * H * D * S)
    nc = rem // (H * D * S)
    h = rem % (D * S) // S
    d = rem % S // 1
    s = rem % S
    # h, d, s values computed above; store linear sum
    val = b + nc + h + d + s
    tl.store(states_ptr + (b * NC * H * D * S) + (nc * H * D * S) + (h * D * S) + (d * S) + s, val)


@triton.jit
def compute_y_kernel(y_ptr, C_ptr, states_ptr, B, NC, T, H, D, S):
    # y[b, nc, t, h, d] = sum_s C[b, nc, t, h, s] * states[b, nc, h, d, s]
    pid = tl.program_id(0)
    b = pid // (NC * T * H * D)
    rem = pid % (NC * T * H * D)
    nc = rem // (T * H * D)
    t = rem % (H * D) // H
    h = rem % H
    d = rem % D
    s = tl.arange(0, S)
    C_val = tl.load(C_ptr + (b * NC * T * H * S) + (nc * T * H * S) + (t * H * S) + (h * S) + s)
    states_val = tl.load(states_ptr + (b * NC * H * D * S) + (nc * H * D * S) + (h * D * S) + (d * S) + s)
    y_val = C_val * states_val
    out_idx = (b * NC * T * H * D) + (nc * T * H * D) + (t * H * D) + (h * D) + d
    tl.store(y_ptr + out_idx, tl.sum(y_val, axis=0))


@triton.jit
def final_state_zeros_kernel(final_state_ptr, B, H, D):
    # Write final_state as zeros: [B, H, D], bfloat16
    pid = tl.program_id(0)
    b = pid // (H * D)
    h = (pid % (H * D)) // D
    d = pid % D
    zero = tl.zeros((), dtype=tl.bfloat16)
    tl.store(final_state_ptr + (b * H * D) + (h * D) + d, zero)


def run_triton_only(hidden_states: torch.Tensor):
    # Ensure float32 for compute
    hidden_states_f = hidden_states.to(torch.float32)
    B_size, S, H, D = hidden_states_f.shape
    # Choose chunk_size dynamically
    chunk_size = 256 if S >= 256 else S
    seq_len_padded = ((S + chunk_size - 1) // chunk_size) * chunk_size
    pad_size = (chunk_size - S % chunk_size) % chunk_size
    # Pad last dimension to next multiple of chunk_size (Triton kernel)
    hidden_padded = torch.empty((B_size, S, chunk_size), dtype=torch.float32, device=hidden_states_f.device)
    # Launch pad kernel: pad from D to chunk_size along last dim
    grid_pad = (B_size * S,)
    pad_last_dim_kernel[grid_pad](hidden_padded, hidden_states_f, B_size, S, D, chunk_size, 0, D)

    # Reshape hidden into chunks [B, NC, K, H, D]
    NC = seq_len_padded // chunk_size  # since we pad S, NC is based on padded S, but here padded=S because hidden is 2D
    # Correction: we need NC based on padded seq_len. The original code pads seq_len, so we compute NC based on seq_len_padded if we had 3D input.
    # However, we only have [B, S, H, D]. The original code's reshape_into_chunks expects [B, S, 1, H, D] after padding -> chunk along S.
    # For simplicity and correctness, set NC = ceil_div(S, chunk_size). Since we padded to chunk_size, NC = (S + chunk_size - 1) // chunk_size.
    NC = (S + chunk_size - 1) // chunk_size
    hidden_chunked = torch.empty((B_size, NC, chunk_size, H, D), dtype=torch.float32, device=hidden_states_f.device)
    grid_chunks = (B_size * NC * chunk_size * H * D,)
    hidden_to_chunks_kernel[grid_chunks](
        hidden_chunked, hidden_padded, B_size, S, D, chunk_size, NC, H, chunk_size
    )

    # Allocate C_dummy and states
    state_size = 256
    C_dummy = torch.empty((B_size, NC, chunk_size, H, state_size), dtype=torch.float32, device=hidden_states_f.device)
    states = torch.empty((B_size, NC, H, D, state_size), dtype=torch.float32, device=hidden_states_f.device)
    # Initialize via Triton kernels (no torch compute)
    grid_init_C = (B_size * NC * chunk_size * H * state_size,)
    init_C_dummy_kernel[grid_init_C](C_dummy, B_size, NC, chunk_size, H, state_size)

    grid_init_states = (B_size * NC * H * D * state_size,)
    init_states_kernel[grid_init_states](states, B_size, NC, H, D, state_size)

    # Compute y
    y = torch.empty((B_size, NC * chunk_size, H * D), dtype=torch.float32, device=hidden_states_f.device)
    grid_compute_y = (B_size * NC * chunk_size * H * D,)
    compute_y_kernel[grid_compute_y](y, C_dummy, states, B_size, NC, chunk_size, H, D, state_size)

    # Reshape to [B, S, H*D]
    y_reshaped = y.reshape(B_size, S, H * D)

    # final_state zeros [B, H, D] in bfloat16
    final_state = torch.empty((B_size, H, D), dtype=torch.bfloat16, device=hidden_states_f.device)
    grid_fs = (B_size * H * D,)
    final_state_zeros_kernel[grid_fs](final_state, B_size, H, D)

    # Cast output to bfloat16 to match original
    y_out = y_reshaped.to(torch.bfloat16)

    return y_out, final_state


class ModelNew(nn.Module):
    def forward(self, hidden_states: torch.Tensor, *args):
        # The original run(...) expects 4 more tensors (A, B, C, D, initial_states), but they are not provided.
