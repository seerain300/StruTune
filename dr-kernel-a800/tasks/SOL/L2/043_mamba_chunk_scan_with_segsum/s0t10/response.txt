    """
    Compute cumulative sum along the last dimension (N3) for each row (b, n1, n2).
    x_ptr and y_ptr point to contiguous tensors shaped [B, N1, N2, N3].
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)

    # Running sum along N3

    # Iterate over N3

        val = tl.load(x_ptr + idx)

        tl.store(y_ptr + idx, run_sum)
