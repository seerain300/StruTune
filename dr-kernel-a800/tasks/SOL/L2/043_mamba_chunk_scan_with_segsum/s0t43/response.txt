import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def pad_last_dim_kernel(x_ptr, y_ptr,
                         B: tl.int32, N1: tl.int32, N2: tl.int32, N3: tl.int32,
                         N3_out: tl.int32):
    """
    Pad last dimension of a 4D tensor [B, N1, N2, N3] to N3_out.
    y[b, n1, n2, i] = x[b, n1, n2, i] for i in [0..min(N3-1, N3_out-1)], zeros elsewhere.
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)

    # Each program processes one row across N3_out
    # Compute row offset in flattened memory (contiguous [B, N1, N2, N3] layout)
    # Memory offset for x and y rows:
    # x_row_offset = ((b * N1 + n1) * N2 + n2) * N3
    # y_row_offset = ((b * N1 + n1) * N2 + n2) * N3_out
    # Initialize offsets
    # We pass y_ptr and x_ptr directly; compute offsets from strides
    # For contiguous tensors, strides: [N1*N2*N3, N2*N3, N3, 1]
    # But here we assume contiguous and use simple index arithmetic:
    # base_x = ((b * N1 + n1) * N2 + n2) * N3
    # base_y = ((b * N1 + n1) * N2 + n2) * N3_out
    base_x = ((b * N1 + n1) * N2 + n2) * N3
    base_y = ((b * N1 + n1) * N2 + n2) * N3_out

    # Write zeros to y row up to N3_out
    # We will loop and write zeros for i >= N3
    for i in range(N3_out):
        # If i < N3: copy from x; else: write 0
        if i < N3:
            val = tl.load(x_ptr + base_x + i)
        else:
            val = 0.0
        tl.store(y_ptr + base_y + i, val)


@triton.jit
def cumsum_last_dim_kernel(x_ptr, y_ptr,
                            B: tl.int32, N1: tl.int32, N2: tl.int32, N3: tl.int32):
    """
    Cumulative sum along last dimension for each row (b, n1, n2).
    x_ptr and y_ptr point to contiguous tensors shaped [B, N1, N2, N3].
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)
    base_x = ((b * N1 + n1) * N2 + n2) * N3
    base_y = ((b * N1 + n1) * N2 + n2) * N3

    running = 0.0
    for i in range(0, N3):
        val = tl.load(x_ptr + base_x + i)
        running += val
        tl.store(y_ptr + base_y + i, running)


@triton.jit
def segment_sum_lower_tri_exp_kernel(x_ptr, y_ptr,
                                     B: tl.int32, N1: tl.int32, N2: tl.int32, N3: tl.int32):
    """
    For each row (b, n1, n2), compute segment sum with lower-triangular mask (diagonal = -1)
    along last dimension: for i in [0..N3-1], sum over j in [0..i-1] of x[b,n1,n2,j],
    then apply exp. y[b,n1,n2,i] = exp(sum_{j=0..i-1} x[b,n1,n2,j]).
    """
    b = tl.program_id(0)
    n1 = tl.program_id(1)
    n2 = tl.program_id(2)
    base = ((b * N1 + n1) * N2 + n2) * N3

    running = 0.0
    for i in range(0, N3):
        # compute segment sum over j in [0..i-1]
        seg_sum = 0.0
        for j in range(0, i):
            val = tl.load(x_ptr + base + j)
            seg_sum += val
        out = tl.exp(seg_sum)
        tl.store(y_ptr + base + i, out)


@triton.jit
def add_inplace_kernel(x_ptr, y_ptr, val: tl.float32, N: tl.int32, BLOCK: tl.constexpr):
    """
    Elementwise y += val for a 1D contiguous vector of length N.
    Launch as 1D grid over N.
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    y = y + val
    tl.store(y_ptr + offsets, y, mask=mask)
