import torch
import triton
import triton.language as tl


# Triton: pad last dimension (constant 0) - flattened 1D
@triton.jit
def pad_last_dim_kernel(
    inp_ptr,         # *float32, input flattened (n_in)
    out_ptr,         # *float32, output flattened (n_out)
    n_in,            # int32, number of valid elements in input
    out_len,         # int32, total number of elements in output (n_in + pad)
    pad,             # int32, pad size
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask_in = offsets < n_in
    # out indices map to input by subtracting pad
    in_idx = offsets - pad
    tl.store(out_ptr + offsets, tl.load(inp_ptr + in_idx, mask=mask_in, other=0.0))


# Triton: inclusive cumsum along 1D with lower-triangular mask (diagonal=-1)
# For segment_sum use: per-chunk vector along chunk_size; here we implement a simple 1D cumsum.
@triton.jit
def cumsum_1d_kernel(
    in_ptr,          # *float32, 1D input
    out_ptr,         # *float32, 1D output
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    # load a block
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    # inclusive cumsum within block
    acc = tl.zeros([BLOCK], dtype=tl.float32)
    for i in range(0, BLOCK):
        acc[i] = tl.where(i == 0, x[i], acc[i - 1] + x[i])
        tl.store(out_ptr + offsets + i, acc[i], mask=mask[i])


# Triton: elementwise exp over 1D array
@triton.jit
def exp_kernel_1d(
    in_ptr,          # *float32
    out_ptr,         # *float32
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    y = tl.exp(x)
    tl.store(out_ptr + offsets, y, mask=mask)


# Triton: dense reduction G = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s]
# Inputs:
#   C_ptr: [B, N, Chunk, H, S]
#   B_ptr: [B, N, Chunk, H, S]
# Outputs:
#   G_ptr: [B, N, Chunk, Chunk, H]
# We specialize H=1 as per original code.
@triton.jit
def dense_reduce_G_kernel(
    C_ptr, B_ptr, G_ptr,
    Bsz: tl.constexpr,     # not used directly, but can be passed
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,       # 1
    S: tl.constexpr,
    T_TILE: tl.constexpr,  # tile over i (Chunk)
    J_TILE: tl.constexpr,  # tile over j (Chunk)
    S_TILE: tl.constexpr,  # tile over s (S)
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    i = tl.program_id(axis=2)  # i in [0, Chunk)
    j = tl.program_id(axis=3)  # j in [0, Chunk)
    h = tl.program_id(axis=4)  # 0

    acc = tl.zeros([1], dtype=tl.float32)

    # tile over s dimension
    for s0 in range(0, S, S_TILE):
        s_off = s0 + tl.arange(0, S_TILE)
        mask_s = s_off < S

        # load C[i, s_off] and B[j, s_off], then compute dot product
        # C offsets: (((b*N + nc)*Chunk + i)*H + h)*S + s_off
        C_offsets = (((b * N + nc) * Chunk + i) * H + h) * S + s_off
        B_offsets = (((b * N + nc) * Chunk + j) * H + h) * S + s_off
        C_vec = tl.load(C_ptr + C_offsets, mask=mask_s, other=0.0)  # (S_TILE,)
        B_vec = tl.load(B_ptr + B_offsets, mask=mask_s, other=0.0)  # (S_TILE,)

        acc += tl.sum(C_vec * B_vec, axis=0)

    # store acc to G[b, nc, i, j, h] with H=1
    g_idx = ((b * N + nc) * Chunk * Chunk + i * Chunk + j) * H + h  # with H=1
    tl.store(G_ptr + g_idx, acc)


# Triton: dense reduction S = sum_{t,d} B_decay[b, nc, t, h, s] * hidden[b, nc, t, h, d]
# Inputs:
#   B_decay_ptr: [B, N, Chunk, H, S] with H=1
#   hidden_ptr:  [B, N, Chunk, H, D] with H=1

@triton.jit
def dense_reduce_S_kernel(
    B_decay_ptr, hidden_ptr, S_ptr,
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,   # 1
    S: tl.constexpr,
    D: tl.constexpr,
    T_TILE: tl.constexpr,
    D_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    h = tl.program_id(axis=2)   # 0
    s = tl.program_id(axis=3)

    acc = tl.zeros([1], dtype=tl.float32)

    # tile over t (chunk_size) and d (head_dim)
    for t0 in range(0, Chunk, T_TILE):
        t_off = t0 + tl.arange(0, T_TILE)
        mask_t = t_off < Chunk

        # for each t in the tile, sum over d
        for d0 in range(0, D, D_TILE):
            d_off = d0 + tl.arange(0, D_TILE)
            mask_d = d_off < D

            # hidden offsets: (((b*N + nc)*Chunk + t_off[:, None])*H + h)*D + d_off[None, :]
            hidden_offsets = (((b * N + nc) * Chunk + t_off[:, None]) * H + h) * D + d_off[None, :]
            mask = mask_t[:, None] & mask_d[None, :]
            hidden_mat = tl.load(hidden_ptr + hidden_offsets, mask=mask, other=0.0)  # (T_TILE, D_TILE)

            col_sums = tl.sum(hidden_mat, axis=1)  # sum over D_TILE -> (T_TILE,)
            B_vec = tl.load(B_decay_ptr + (((b * N + nc) * Chunk + t_off) * H + h) * S + s, mask=mask_t, other=0.0)  # (T_TILE,)

            acc += tl.sum(B_vec * col_sums, axis=0)

    s_idx = ((b * N + nc) * H + h) * S + s  # H=1 -> (b*N + nc)*S + s
    tl.store(S_ptr + s_idx, acc)


def _launch_pad_last_dim(inp: torch.Tensor, pad: int) -> torch.Tensor:
    # inp: [B, S, H, D] as 1D contiguous
    n_in = inp.numel()
    n_out = n_in + pad
    out = torch.empty(n_out, dtype=torch.float32, device=inp.device)
    BLOCK = 1024
    grid = (triton.cdiv(n_out, BLOCK),)
    pad_last_dim_kernel[grid](inp, out, n_in, n_out, pad, BLOCK=BLOCK, num_warps=4)
    return out


def _launch_cumsum_1d(x: torch.Tensor) -> torch.Tensor:
    # x: 1D contiguous float32
    n = x.numel()
    y = torch.empty_like(x)
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    cumsum_1d_kernel[grid](x, y, n, BLOCK=BLOCK, num_warps=4)
    return y


def _launch_exp_1d(x: torch.Tensor) -> torch.Tensor:
    # x: 1D contiguous float32
    n = x.numel()
    y = torch.empty_like(x)
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    exp_kernel_1d[grid](x, y, n, BLOCK=BLOCK, num_warps=4)
    return y


def _launch_dense_reduce_G(C: torch.Tensor, B: torch.Tensor, G: torch.Tensor):
    # Shapes: C [B, N, Chunk, 1, S], B [B, N, Chunk, 1, S], G [B, N, Chunk, Chunk, 1]

    assert H == 1, "G reduction expects H=1"

    grid = (B, N, Chunk, Chunk, 1)
    T_TILE = 64   # tile over i (Chunk)
    J_TILE = 64   # tile over j (Chunk)
    S_TILE = 64   # tile over s (S)
    dense_reduce_G_kernel[grid](
        C, B, G,
        Bsz, N, Chunk, H, S,
        T_TILE=T_TILE, J_TILE=J_TILE, S_TILE=S_TILE,
        num_warps=4
    )


def _launch_dense_reduce_S(B_decay: torch.Tensor, hidden: torch.Tensor, S_out: torch.Tensor):

def pad_last_dim_kernel(
    inp_ptr,         # *float32, input flattened (n_in)
