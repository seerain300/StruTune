# Triton kernel: segment_sum_lower_tri_scan
# Computes for each (b, nc, h): row[j] = sum_{k=0..j} A[b, nc, j, h], for j in [0..N-1].
# Then exponentiates row and stores into L_out[b, nc, i, j, h] where i is program_id(3).
# Note: We compute row per i (i.e., for each output row index), but actually we need lower-triangular structure over (i,j).
# Here we store row and later use it in forward to form lower-triangular matrices. The original code applies a lower-tri mask
# before cumsum. This kernel mirrors that: it computes sum along rows but forward applies the mask. For now, we compute row sums.
@triton.jit
def segment_sum_lower_tri_scan(A_perm_ptr, L_out_ptr,
                                Bsz, NC, H, N,
                                a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                                l_stride_b, l_stride_nc, l_stride_i, l_stride_j, l_stride_h):
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)
    i = tl.program_id(3)  # output row index i in [0..N-1]

    row = tl.zeros((N,), dtype=tl.float32)
    # Inclusive scan across j in [0..N-1]
    for j in range(0, N):
        val = tl.load(A_perm_ptr + b * a_stride_b + nc * a_stride_nc + j * a_stride_i + h * a_stride_h)
        row = row + (j <= i) * val  # apply lower-triangular mask by zeroing contributions for j > i; init row with zeros
    # Exponentiate row and store
    exp_row = tl.exp(row)
    l_ptrs = L_out_ptr + b * l_stride_b + nc * l_stride_nc + i * l_stride_i
    for j in range(0, N):
        tl.store(l_ptrs + j * l_stride_j + h * l_stride_h, exp_row[j])

    nc = tl.program_id(1)
    h = tl.program_id(2)

        val = tl.load(A_ptr + b * a_stride_b + nc * a_stride_nc + t * a_stride_t + h * a_stride_h)

        # Store A_cumsum

    # For each t, diff = last - A_cumsum[t]

        curr = tl.load(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h)
