    scale: tl.constexpr  # scalar float32
):

    nc = tl.program_id(1)
    cols = tl.arange(0, I)
    base = (b * N + nc) * I
    ptr_in = in_ptr + base + cols
    ptr_out = out_ptr + base + cols
    val = tl.load(ptr_in)
    val = val * tl.exp(scale)
    tl.store(ptr_out, val)  # Vector store, Triton broadcasts scalar
