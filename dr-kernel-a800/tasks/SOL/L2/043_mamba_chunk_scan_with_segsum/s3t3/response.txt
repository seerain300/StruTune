# 2) Triton kernel: inclusive cumsum along last axis for 4D tensor [B, N, T, H].
#    Each program handles one row (b, n) and scans along T, writing to out_ptr.
@triton.jit
def cumsum_last_axis_4d_kernel(in_ptr, out_ptr,
                               B, N, T, H,
                               in_stride_b, in_stride_n, in_stride_t, in_stride_h,
                               out_stride_b, out_stride_n, out_stride_t, out_stride_h,
                               BLOCK_T: tl.constexpr):
    pid = tl.program_id(axis=0)
    b = pid // N
    n = pid % N
    running = 0.0
    for t in range(0, T):
        val = tl.load(in_ptr + b * in_stride_b + n * in_stride_n + t * in_stride_t + 0 * in_stride_h)
        running += val
        tl.store(out_ptr + b * out_stride_b + n * out_stride_n + t * out_stride_t + 0 * out_stride_h, running)

# 3) Triton kernel: inclusive cumsum along axis -2 for a logically expanded tensor
#    with shape [B, N, T, H, S]. This axis is H (num_heads). We operate by passing
#    sizes and strides, and compute addresses for each (b, n, t, s) across H.
@triton.jit
def cumsum_axis_minus_two_5d_kernel(in_ptr, out_ptr,
                                    B, N, T, H, S,
                                    in_stride_b, in_stride_n, in_stride_t, in_stride_h, in_stride_s,
                                    out_stride_b, out_stride_n, out_stride_t, out_stride_h, out_stride_s,
                                    BLOCK_H: tl.constexpr):
    # Grid is over all (b, n, t, s) rows: axis0 = B*N*T*S
    pid = tl.program_id(axis=0)
    total = B * N * T * S
    if pid >= total:
        return
    b = pid // (N * T * S)
    rem = pid % (N * T * S)
    n = rem // (T * S)
    t = (rem // S) % T
    s = rem % S

    running = 0.0
    for h in range(0, H):
        in_addr = in_ptr + b * in_stride_b + n * in_stride_n + t * in_stride_t + h * in_stride_h + s * in_stride_s
        val = tl.load(in_addr)
        running += val
        out_addr = out_ptr + b * out_stride_b + n * out_stride_n + t * out_stride_t + h * out_stride_h + s * out_stride_s
        tl.store(out_addr, running)

# 4) Triton kernel: apply tril mask (diagonal = -1) to a 5D tensor [B, N, T, H, S].
#    We zero elements where s < t. The indexing uses strides and sizes.
@triton.jit

                        B, N, T, H, S,
                        in_stride_b, in_stride_n, in_stride_t, in_stride_h, in_stride_s,
                        out_stride_b, out_stride_n, out_stride_t, out_stride_h, out_stride_s,
                        BLOCK_S: tl.constexpr):

    pid = tl.program_id(axis=0)
    total = B * N * T * S
    if pid >= total:
        return
    b = pid // (N * T * S)
    rem = pid % (N * T * S)
    n = rem // (T * S)
    t = (rem // S) % T
    s = rem % S

    in_addr = in_ptr + b * in_stride_b + n * in_stride_n + t * in_stride_t + s * in_stride_s
    out_addr = out_ptr + b * out_stride_b + n * out_stride_n + t * out_stride_t + s * out_stride_s

    if s >= t:
        val = tl.load(in_addr)
        tl.store(out_addr, val)
    else:
        tl.store(out_addr, 0.0)
