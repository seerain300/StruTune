import torch
import triton
import triton.language as tl


# 1) Triton cumsum along a vector (used for A_cumsum, segment_sum scans)
@triton.jit
def cumsum_exp_diff(A_ptr, Out_ptr,
                     Bsz, NC, H, N,
                     a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                     out_stride_b, out_stride_nc, out_stride_t, out_stride_h):
    # One program per (b, nc, h), scan across N entries in A[..., i, h]
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    # Hillis–Steele inclusive scan along N
    for t in range(0, N):
        v = tl.load(A_ptr + b * a_stride_b + nc * a_stride_nc + t * a_stride_t + h * a_stride_h)
        acc = acc + v
        tl.store(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h, acc)

    # After scan, compute exp(A_cumsum[:, :, :, -1:] - A_cumsum)
    last = tl.load(Out_ptr + b * out_stride_b + nc * out_stride_nc + (N - 1) * out_stride_t + h * out_stride_h)
    for t in range(0, N):
        curr = tl.load(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h)
        diff = last - curr
        tl.store(Out_ptr + b * out_stride_b + nc * out_stride_nc + t * out_stride_t + h * out_stride_h, tl.exp(diff))


# 2) Triton segment_sum for lower-triangular inclusive scan (used to compute L)
@triton.jit
def segment_sum_lower_tri_scan(A_ptr, L_ptr,
                                Bsz, NC, H, N,
                                a_stride_b, a_stride_nc, a_stride_i, a_stride_h,
                                l_stride_b, l_stride_nc, l_stride_i, l_stride_j, l_stride_h):
    # Grid over (b, nc, h); within each program, we process row i and apply lower-triangular mask: j<=i
    b = tl.program_id(0)
    nc = tl.program_id(1)
    h = tl.program_id(2)

    for i in range(0, N):
        acc = tl.zeros((), dtype=tl.float32)
        # j loop: lower-triangular only (j <= i)
        for j in range(0, N):
            if j <= i:
                v = tl.load(A_ptr + b * a_stride_b + nc * a_stride_nc + i * a_stride_i + h * a_stride_h)
                acc = acc + v
        exp_row = tl.exp(acc)
        # Store exp_row to L[b, nc, i, :, h]; j dimension is implicit in pointer arithmetic
        l_row_ptr = L_ptr + b * l_stride_b + nc * l_stride_nc + i * l_stride_i + h * l_stride_h
        # We write the same exp_row across j positions. Triton requires pointer grid; here j is index in store.
        # To create a vector, we can expand acc by broadcasting: write acc to all j positions.
        # Since Triton does not support broadcasting directly in store from scalar, we recompute per j.
        # Compute L as exp of segment_sum on masked row (lower-triangular inclusive). We need per-j values based on i,j.
        # Instead, precompute per j: sum_{r<=j} A[i, r] when j<=i, else 0. We re-implement with two-level loop.
        # (This is the correct approach: we need the cumulative at each j.)
        # Re-implement proper lower-triangular scan per j:
        for jj in range(0, N):
            partial = tl.zeros((), dtype=tl.float32)
            for rr in range(0, jj + 1):  # inclusive up to jj
                if i == rr:  # Access A at (i, h) for each jj step. Note: we need per-j prefix; here we need A[i, h] per jj.
                    # There's no direct dependency on jj for A[i, h] (it's constant for this row). The original logic
                    # uses per-row A_cumsum. However, the original code masks upper-tri entries to -inf then computes
                    # cumsum; since we exponentiate after, zeros are fine. The earlier version computed exp of cumsum
                    # only for lower-tri entries. The correct logic per j is to add A[i] only if j<=i, else 0.
                    # To implement that, we can store exp(0)=1 for upper-tri (which matches original post-fill).
                    if rr <= i:
                        a_val = tl.load(A_ptr + b * a_stride_b + nc * a_stride_nc + i * a_stride_i + h * a_stride_h)
                        partial = partial + a_val
            # Write exp(partial) to L[b, nc, i, jj, h]
            tl.store(L_ptr + b * l_stride_b + nc * l_stride_nc + i * l_stride_i + jj * l_stride_j + h * l_stride_h,
                     tl.exp(partial))


# 3) Triton contraction: G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s]
@triton.jit
def contraction_CxB(C_ptr, B_ptr, G_ptr,
                    Bsz, NC, H, N, S,
                    c_stride_b, c_stride_nc, c_stride_i, c_stride_h, c_stride_s,
                    b_stride_b, b_stride_nc, b_stride_j, b_stride_h, b_stride_s,
                    g_stride_b, g_stride_nc, g_stride_i, g_stride_j, g_stride_h):
    b = tl.program_id(0)
    nc = tl.program_id(1)
    i = tl.program_id(2)
    j = tl.program_id(3)
    h = tl.program_id(4)

    acc = tl.zeros((), dtype=tl.float32)
    for s in range(0, S):
        c_val = tl.load(C_ptr + b * c_stride_b + nc * c_stride_nc + i * c_stride_i + h * c_stride_h + s * c_stride_s)
        b_val = tl.load(B_ptr + b * b_stride_b + nc * b_stride_nc + j * b_stride_j + h * b_stride_h + s * b_stride_s)
        acc += c_val * b_val

    g_ptrs = G_ptr + b * g_stride_b + nc * g_stride_nc + i * g_stride_i + j * g_stride_j + h * g_stride_h
    tl.store(g_ptrs, acc)

@triton.jit
def propagate_decay(decay_ptr, init_ptr, states_ptr,
                    Bsz, NC, H, N, S,
                    d_stride_bh, d_stride_ij, d_stride_k,  # decay strides for [B,H,NC+1,NC+1]

                    states_stride_b, states_stride_i, states_stride_h, states_stride_d, states_stride_s):
    # Each program computes a single (b, i, h, d, s) output as sum_j decay[b,h,i,j] * init_or_prev[b,j,h,d,s]
    # Here we implement the loop over j manually using Triton's loop; grid can be (B, NC, H, D, S).
    b = tl.program_id(0)
    i = tl.program_id(1)
    h = tl.program_id(2)
    d = tl.program_id(3)
    s = tl.program_id(4)

    acc = tl.zeros((), dtype=tl.float32)
    for j in range(0, N + 1):  # decay has NC+1 along k dimension

    b = tl.program_id(0)

    h = tl.program_id(2)
    d = tl.program_id(3)

    in_ptrs = in_ptr + b * in_stride_b + idx_in * in_stride_s + h * in_stride_h + d * in_stride_d
    out_ptrs = out_ptr + b * out_stride_b + s_out * out_stride_s + h * out_stride_h + d * out_stride_d

    if valid_in:
        val = tl.load(in_ptrs)
    else:
        val = 0.0
    tl.store(out_ptrs, val)
