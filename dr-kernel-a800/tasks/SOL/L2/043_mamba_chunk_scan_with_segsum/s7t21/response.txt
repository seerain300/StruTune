import torch
import triton
import triton.language as tl


# 1) Triton kernel: pad last dimension (constant 0) - flattened 1D
@triton.jit
def pad_last_dim_kernel(
    inp_ptr,    # *float32, input flattened
    out_ptr,    # *float32, output flattened
    n_in: tl.constexpr,   # number of valid elements (flattened)
    out_len: tl.constexpr,  # total number of elements (flattened) = n_in + pad
    pad: tl.constexpr,    # pad size added
):
    i = tl.program_id(axis=0)
    if i < n_in:
        val = tl.load(inp_ptr + i)
        tl.store(out_ptr + i, val)
    else:
        tl.store(out_ptr + i, 0.0)


# 2) Inclusive cumsum along 1D using block tiling (2D grid over blocks)
@triton.jit
def cumsum_1d_kernel(
    in_ptr,     # *float32, input flattened
    out_ptr,    # *float32, output flattened
    n_elements: tl.constexpr,  # total elements to process
    BLOCK: tl.constexpr,       # block size
):
    # axis 0: which block; axis 1: intra-block lane
    pid_block = tl.program_id(axis=0)
    lane = tl.program_id(axis=1)
    start = pid_block * BLOCK
    # running sum for this block
    running = tl.zeros([1], dtype=tl.float32)
    # iterate over the block with compile-time bounds
    for i in range(0, BLOCK):
        idx = start + i
        if idx < n_elements:
            x = tl.load(in_ptr + idx)
            running += x
            tl.store(out_ptr + idx, running)
        else:
            # out-of-range elements get 0
            tl.store(out_ptr + idx, 0.0)


# 3) Dense reduction G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s]
# Grid: (B, N, Chunk, Chunk, H) with N=num_chunks, Chunk=chunk_size, H=num_heads.
# We reduce over S=state_size in tiles (S_TILE=64).
@triton.jit
def dense_reduce_G_kernel(
    C_ptr,      # *float32, [B, N, Chunk, H, S]
    B_ptr,      # *float32, [B, N, Chunk, H, S]
    G_ptr,      # *float32, [B, N, Chunk, Chunk, H]
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    S_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    j = tl.program_id(axis=3)
    h = tl.program_id(axis=4)

    # initialize accumulator
    acc = tl.zeros([1], dtype=tl.float32)

    # loop over state_size in tiles
    for s0 in range(0, S, S_TILE):
        # vectorized offsets for s
        s_off = s0 + tl.arange(0, S_TILE)
        mask = s_off < S

        # compute base offsets for C and B at fixed (b, nc, i, h) and (b, nc, j, h)
        # C offset: (((b*N + nc)*Chunk + i)*H + h)*S + s
        # B offset: (((b*N + nc)*Chunk + j)*H + h)*S + s
        C_offsets = (((b * N + nc) * Chunk + i) * H + h) * S + s_off
        B_offsets = (((b * N + nc) * Chunk + j) * H + h) * S + s_off

        # load C[:, s] and B[:, s] vectors
        C_vec = tl.load(C_ptr + C_offsets, mask=mask, other=0.0)
        B_vec = tl.load(B_ptr + B_offsets, mask=mask, other=0.0)

        # multiply and reduce across tile
        acc += tl.sum(C_vec * B_vec, axis=0)

    # store result G[b, nc, i, j, h] = acc
    g_idx = (((b * N + nc) * Chunk + i) * Chunk + j) * H + h
    tl.store(G_ptr + g_idx, acc)


# 4) Dense reduction S[b, nc, h, s] = sum_{t,d} B_decay[b, nc, t, h, s] * hidden[b, nc, t, h, d]
# Grid: (B, N, H, S) with N=num_chunks, H=num_heads, S=state_size.
# We reduce over T=chunk_size and D=head_dim in tiles (T_TILE=128, D_TILE=64).
@triton.jit
def dense_reduce_S_kernel(
    B_decay_ptr,  # *float32, [B, N, Chunk, H, S]
    hidden_ptr,   # *float32, [B, N, Chunk, H, D]
    S_ptr,        # *float32, [B, N, H, S]
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,
    S: tl.constexpr,
    D: tl.constexpr,
    T_TILE: tl.constexpr,
    D_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    h = tl.program_id(axis=2)
    s = tl.program_id(axis=3)

    acc = tl.zeros([1], dtype=tl.float32)

    # reduce over chunk_size in tiles
    for t0 in range(0, Chunk, T_TILE):
        t_off = t0 + tl.arange(0, T_TILE)
        mask_t = t_off < Chunk
        # base offset for B_decay: (((b*N + nc)*Chunk + t)*H + h)*S + s
        B_offsets = (((b * N + nc) * Chunk + t_off) * H + h) * S + s
        # load vector over t for fixed (b, nc, s, h)
        B_vec = tl.load(B_decay_ptr + B_offsets, mask=mask_t, other=0.0)

        # reduce over head_dim in tiles
        for d0 in range(0, D, D_TILE):
            d_off = d0 + tl.arange(0, D_TILE)
            mask_d = d_off < D
            # hidden offsets: (((b*N + nc)*Chunk + t_off[:, None])*H + h)*D + d_off[None, :]
            hidden_offsets = (((b * N + nc) * Chunk + t_off[:, None]) * H + h) * D + d_off[None, :]
            # broadcast mask: (BLOCK_T, BLOCK_D)
            mask = mask_t[:, None] & mask_d[None, :]
            hidden_mat = tl.load(hidden_ptr + hidden_offsets, mask=mask, other=0.0)

            # accumulate: sum over d then over t
            # First sum over D_TILE
            # acc += sum_d (B_vec[t] * sum_d hidden_mat[t, d])
            # We need to multiply B_vec[t] with the column-wise sums of hidden_mat
            # Compute column sums over D_TILE for each t
            col_sums = tl.sum(hidden_mat, axis=1)  # shape (BLOCK_T,)
            # Outer multiply with B_vec
            acc += tl.sum(B_vec * col_sums, axis=0)

    # store S[b, nc, h, s] = acc
    s_idx = ((b * N + nc) * H + h) * S + s
    tl.store(S_ptr + s_idx, acc)


# 5) Elementwise exp over 1D array
@triton.jit
def exp_kernel(
    in_ptr,      # *float32
    out_ptr,     # *float32
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    for i in range(0, BLOCK):
        idx = start + i
        if idx < n_elements:
            x = tl.load(in_ptr + idx)
            y = tl.exp(x)
            tl.store(out_ptr + idx, y)

import triton
import triton.language as tl


# Triton: pad last dimension (constant 0) - flattened 1D
@triton.jit
def pad_last_dim_kernel(
    inp_ptr,    # *float32, input flattened
    out_ptr,    # *float32, output flattened
    n_in: tl.constexpr,   # number of valid elements
    out_len: tl.constexpr,  # total number of elements
    pad: tl.constexpr,    # pad size added
):
    i = tl.program_id(axis=0)
    if i < n_in:
        val = tl.load(inp_ptr + i)
        tl.store(out_ptr + i, val)
    else:
        tl.store(out_ptr + i, 0.0)


# Triton: inclusive cumsum along 1D using block tiling (2D grid over blocks)
@triton.jit
def cumsum_1d_kernel(
    in_ptr,     # *float32, input flattened
    out_ptr,    # *float32, output flattened
    n_elements: tl.constexpr,  # total elements to process
    BLOCK: tl.constexpr,       # block size
):
    pid_block = tl.program_id(axis=0)
    lane = tl.program_id(axis=1)
    start = pid_block * BLOCK
    running = tl.zeros([1], dtype=tl.float32)
    for i in range(0, BLOCK):
        idx = start + i
        if idx < n_elements:
            x = tl.load(in_ptr + idx)
            running += x
            tl.store(out_ptr + idx, running)
        else:
            tl.store(out_ptr + idx, 0.0)


# Triton: compute G[b, nc, i, j, h] = sum_s C[b, nc, i, h, s] * B[b, nc, j, h, s]
# Specialized for H=1
@triton.jit
def dense_reduce_G_kernel(
    C_ptr,      # *float32, [B, N, Chunk, 1, S]
    B_ptr,      # *float32, [B, N, Chunk, 1, S]
    G_ptr,      # *float32, [B, N, Chunk, Chunk, 1]
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,  # must be 1
    S: tl.constexpr,
    S_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    j = tl.program_id(axis=3)
    h = tl.program_id(axis=4)  # h should be 0

    acc = tl.zeros([1], dtype=tl.float32)

    for s0 in range(0, S, S_TILE):
        s_off = s0 + tl.arange(0, S_TILE)
        mask = s_off < S

        # offsets for C: (((b*N + nc)*Chunk + i)*H + h)*S + s
        C_offsets = (((b * N + nc) * Chunk + i) * H + h) * S + s_off
        # offsets for B: (((b*N + nc)*Chunk + j)*H + h)*S + s
        B_offsets = (((b * N + nc) * Chunk + j) * H + h) * S + s_off

        C_vec = tl.load(C_ptr + C_offsets, mask=mask, other=0.0)
        B_vec = tl.load(B_ptr + B_offsets, mask=mask, other=0.0)

        acc += tl.sum(C_vec * B_vec, axis=0)

    g_idx = (((b * N + nc) * Chunk + i) * Chunk + j) * H + h  # with H=1, this is (((b*N+nc)*Chunk + i)*Chunk + j)
    tl.store(G_ptr + g_idx, acc)


# Triton: compute S[b, nc, h, s] = sum_{t,d} B_decay[b, nc, t, h, s] * hidden[b, nc, t, h, d], specialized for H=1
@triton.jit
def dense_reduce_S_kernel(
    B_decay_ptr,  # *float32, [B, N, Chunk, 1, S]
    hidden_ptr,   # *float32, [B, N, Chunk, 1, D]
    S_ptr,        # *float32, [B, N, 1, S]  # we store S[b, nc, 0, s]
    Bsz: tl.constexpr,
    N: tl.constexpr,
    Chunk: tl.constexpr,
    H: tl.constexpr,  # must be 1
    S: tl.constexpr,
    D: tl.constexpr,
    T_TILE: tl.constexpr,
    D_TILE: tl.constexpr,
):
    b = tl.program_id(axis=0)
    nc = tl.program_id(axis=1)
    h = tl.program_id(axis=2)  # 0
    s = tl.program_id(axis=3)

    acc = tl.zeros([1], dtype=tl.float32)

    # reduce over chunk_size in tiles
    for t0 in range(0, Chunk, T_TILE):
        t_off = t0 + tl.arange(0, T_TILE)
        mask_t = t_off < Chunk

        # B_decay offsets: (((b*N + nc)*Chunk + t)*H + h)*S + s, with H=1
        B_offsets = (((b * N + nc) * Chunk + t_off) * H + h) * S + s
        B_vec = tl.load(B_decay_ptr + B_offsets, mask=mask_t, other=0.0)  # shape (T_TILE,)

        # reduce over head_dim in tiles
        for d0 in range(0, D, D_TILE):
            d_off = d0 + tl.arange(0, D_TILE)
            mask_d = d_off < D

            # hidden offsets: (((b*N + nc)*Chunk + t_off[:, None])*H + h)*D + d_off[None, :]
            hidden_offsets = (((b * N + nc) * Chunk + t_off[:, None]) * H + h) * D + d_off[None, :]
            mask = mask_t[:, None] & mask_d[None, :]
            hidden_mat = tl.load(hidden_ptr + hidden_offsets, mask=mask, other=0.0)

            col_sums = tl.sum(hidden_mat, axis=1)  # sum over D_TILE for each t -> shape (T_TILE,)
            acc += tl.sum(B_vec * col_sums, axis=0)

    s_idx = ((b * N + nc) * H + h) * S + s  # with H=1, this is (b*N + nc)*S + s
    tl.store(S_ptr + s_idx, acc)
