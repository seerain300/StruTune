import torch
import triton
import triton.language as tl


# Triton kernels

@triton.jit
def reduce_sum_squares_kernel(x_ptr, var_ptr,
                               H, eps,
                               BLOCK_H: tl.constexpr):
    """
    Compute sum(x^2) over H for each (b, s) and write to var[b*S + s].
    x_ptr: [B*S*H], contiguous
    var_ptr: [B*S]
    """
    pid = tl.program_id(axis=0)
    base = pid * H
    sum_acc = 0.0
    for h0 in range(0, H, BLOCK_H):
        offs = h0 + tl.arange(0, BLOCK_H)
        idx = base + offs
        mask = offs < H
        x = tl.load(x_ptr + idx, mask=mask, other=0.0)
        sum_acc += tl.sum(x * x, axis=0)
    mean = sum_acc / H
    inv_std = 1.0 / tl.sqrt(mean + eps)
    tl.store(var_ptr + pid, inv_std)


@triton.jit
def tanh_kernel(x_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise tanh for 1D tensor of length N.
    x_ptr: input
    out_ptr: output
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    # tanh via exp: tanh(z) = (e^{2z} - 1) / (e^{2z} + 1)
    z = 2.0 * x
    e2z = tl.exp(z)
    y = (e2z - 1.0) / (e2z + 1.0)
    tl.store(out_ptr + offsets, y, mask=mask)


@triton.jit
def matvec_kernel(A_ptr, W_ptr, Out_ptr,
                  M, N, K,
                  stride_a0, stride_a1,
                  stride_w0, stride_w1,
                  BLOCK_N: tl.constexpr):
    """
    GEMV: Out[M, K] = A[M, N] @ W[N, K]
    A: [M, N], W: [N, K], Out: [M, K]
    Grid: (M, K) to parallelize over rows and output columns.
    """
    pid_m = tl.program_id(axis=0)  # row index m
    pid_k = tl.program_id(axis=1)  # output column k
    acc = tl.zeros((), dtype=tl.float32)
    # Loop over N in tiles
    for n0 in range(0, N, BLOCK_N):
        offs_n = n0 + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        a = tl.load(A_ptr + pid_m * stride_a0 + offs_n * stride_a1, mask=mask_n, other=0.0)  # [BLOCK_N]
        w = tl.load(W_ptr + offs_n * stride_w0 + pid_k * stride_w1, mask=mask_n, other=0.0)  # [BLOCK_N]
        acc += tl.sum(a * w, axis=0)
    tl.store(Out_ptr + pid_m * K + pid_k, acc)


@triton.jit
def bmm_kernel(A_ptr, B_ptr, Out_ptr,
               M, N, K,
               stride_a0, stride_a1,
               stride_b0, stride_b1,
               stride_o0, stride_o1,
               BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    GEMM: Out[M, K] = A[M, N] @ B[N, K]
    Launch grid over (M, K) with internal loops over N in tiles.
    A: [M, N], B: [N, K], Out: [M, K]
    """
    pid_m = tl.program_id(axis=0)  # row index m
    pid_k = tl.program_id(axis=1)  # output column k
    acc = tl.zeros((), dtype=tl.float32)
    for n0 in range(0, N, BLOCK_N):
        offs_n = n0 + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        # Load A row slice: shape [BLOCK_N]
        a = tl.load(A_ptr + pid_m * stride_a0 + offs_n * stride_a1, mask=mask_n, other=0.0)
        # Load B column slice: shape [BLOCK_N]
        b = tl.load(B_ptr + offs_n * stride_b0 + pid_k * stride_b1, mask=mask_n, other=0.0)
        acc += tl.sum(a * b, axis=0)
    tl.store(Out_ptr + pid_m * stride_o0 + pid_k * stride_o1, acc)

import torch
import triton
import triton.language as tl


# Triton kernels

@triton.jit
def reduce_sum_squares_kernel(x_ptr, var_ptr,
                               H, eps,
                               BLOCK_H: tl.constexpr):
    """
    Compute sum(x^2) over H for each (b, s) and write to var[b*S + s].
    x_ptr: [B*S*H], contiguous
    var_ptr: [B*S]
    """
    pid = tl.program_id(axis=0)
    base = pid * H
    sum_acc = 0.0
    for h0 in range(0, H, BLOCK_H):
        offs = h0 + tl.arange(0, BLOCK_H)
        mask = offs < H
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0)
        sum_acc += tl.sum(x * x, axis=0)
    mean = sum_acc / H
    inv_std = 1.0 / tl.sqrt(mean + eps)
    tl.store(var_ptr + pid, inv_std)


@triton.jit
def tanh_kernel(x_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    """
    Elementwise tanh for 1D tensor of length N.
    x_ptr: input
    out_ptr: output
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    # tanh via exp: tanh(z) = (e^{2z} - 1) / (e^{2z} + 1)
    z = 2.0 * x
    e2z = tl.exp(z)
    y = (e2z - 1.0) / (e2z + 1.0)
    tl.store(out_ptr + offsets, y, mask=mask)


@triton.jit
def matvec_kernel(A_ptr, W_ptr, Out_ptr,
                  M, N, K,
                  stride_a0, stride_a1,
                  stride_w0, stride_w1,
                  BLOCK_N: tl.constexpr):
    """
    GEMV: Out[M, K] = A[M, N] @ W[N, K]
    A: [M, N], W: [N, K], Out: [M, K]
    Grid: (M, K) to parallelize over rows and output columns.
    """
    pid_m = tl.program_id(axis=0)  # row index m
    pid_k = tl.program_id(axis=1)  # output column k
    acc = tl.zeros((), dtype=tl.float32)
    # Loop over N in tiles
    for n0 in range(0, N, BLOCK_N):
        offs_n = n0 + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        a = tl.load(A_ptr + pid_m * stride_a0 + offs_n * stride_a1, mask=mask_n, other=0.0)  # [BLOCK_N]
        w = tl.load(W_ptr + offs_n * stride_w0 + pid_k * stride_w1, mask=mask_n, other=0.0)  # [BLOCK_N]
        acc += tl.sum(a * w, axis=0)
    tl.store(Out_ptr + pid_m * K + pid_k, acc)
