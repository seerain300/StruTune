import torch
import triton
import triton.language as tl


# Kernel 1: per-element rstd and normalized vector (1D, length N)
# Inputs: x_ptr (float16), out_rstd_ptr (float32), out_norm_ptr (float16), N, eps
@triton.jit
def rstd_and_norm_1d_kernel(x_ptr, out_rstd_ptr, out_norm_ptr, N: tl.constexpr, eps: tl.float32):
    idx = tl.program_id(axis=0)
    if idx >= N:
        return
    x = tl.load(x_ptr + idx)
    # Accumulate sum of squares in float32
    x32 = x.to(tl.float32)
    sum_sq = 0.0
    for k in range(0, N):
        sum_sq += x32[k] * x32[k]
    mean = sum_sq / N
    rstd = 1.0 / tl.sqrt(mean + eps)
    norm = x * rstd
    tl.store(out_rstd_ptr + idx, rstd)  # float32
    tl.store(out_norm_ptr + idx, norm)  # same dtype as x (float16 if input is fp16)


# Kernel 2: elementwise tanh over 1D vector
@triton.jit
def tanh_1d_kernel(in_ptr, out_ptr, N: tl.constexpr):
    idx = tl.program_id(axis=0)
    if idx >= N:
        return
    x = tl.load(in_ptr + idx)
    y = tl.tanh(x)
    tl.store(out_ptr + idx, y)


# Kernel 3: F.linear-like for 1D x (length N) and W (shape [K, N]) -> out[K]
# out[i] = sum_j x[j] * W[i, j]
@triton.jit
def linear_dot_kernel(x_ptr, W_ptr, out_ptr, N: tl.constexpr, K: tl.constexpr):
    i = tl.program_id(axis=0)
    if i >= K:
        return
    acc = 0.0
    for j in range(0, N):
        xj = tl.load(x_ptr + j)
        Wik = tl.load(W_ptr + i * N + j)
        acc += xj * Wik
    tl.store(out_ptr + i, acc)


# Kernel 4: batched 3x3 matmul for A[n, s, 3, H] @ B[n, s, 3, 3] -> C[n, s, 3, 3]
# Grid is (N, S).
@triton.jit
def bmm_3x_kernel(A_ptr, B_ptr, C_ptr, N: tl.constexpr, S: tl.constexpr, H: tl.constexpr):
    n = tl.program_id(axis=0)
    s = tl.program_id(axis=1)
    if (n >= N) or (s >= S):
        return
    # C[i, j] = sum_k A[i, k] * B[k, j]
    for i in range(0, 3):
        for j in range(0, 3):
            acc = 0.0
            for k in range(0, 3):
                Aijk = tl.load(A_ptr + n * S * 3 * H + s * 3 * H + i * H + k)
                Bkij = tl.load(B_ptr + k * 3 + j)
                acc += Aijk * Bkij
            tl.store(C_ptr + n * S * 3 * 3 + s * 3 * 3 + i * 3 + j, acc)
