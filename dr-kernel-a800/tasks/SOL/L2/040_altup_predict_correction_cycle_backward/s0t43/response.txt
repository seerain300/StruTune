# Triton kernel: per-row variance + rsqrt for 2D tensor [N, H]
# Computes rstd[i] = rsqrt(mean_j(x[i, j]^2) + eps), writes to out[N]
@triton.jit
def var_rstd_row_kernel(x_ptr, out_ptr, N, H, eps, BLOCK_H: tl.constexpr):
    row = tl.program_id(0)  # 0..N-1
    if row >= N:
        return
    sumsq = tl.zeros((), dtype=tl.float32)
    for h_start in range(0, H, BLOCK_H):
        cols = h_start + tl.arange(0, BLOCK_H)
        mask = cols < H
        x = tl.load(x_ptr + row * H + cols, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / H
    rstd = tl.rsqrt(mean + eps)
    tl.store(out_ptr + row, rstd)


# Triton kernel: batched matmul C[b, m, n] = A[b, m, k] @ B[b, n, k]
# A shape: [S, H, K], B shape: [B, K, N], C shape: [B, S, N]
@triton.jit
def bmm_triton_kernel(A_ptr, B_ptr, C_ptr, S, H, K, B, N,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    b = tl.program_id(0)  # batch over B
    m = tl.program_id(1)  # tile over S
    n = tl.program_id(2)  # tile over N

    offs_m = m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, K, BLOCK_K):
        k = k_start + offs_k
        mask_k = k < K

        # A[b, m, k] -> [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + b * (S * H * K) + (offs_m[:, None] * K + k[None, :])
        a_mask = (offs_m[:, None] < S) & (mask_k[None, :])
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # B[b, n, k] -> [BLOCK_K, BLOCK_N]
        # Note: B is (B, K, N) contiguous; we index over N then K
        b_ptrs = B_ptr + b * (K * N) + (offs_n[None, :] * K + k[:, None])
        b_mask = (offs_n[None, :] < N) & (mask_k[:, None])
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Write C[b, m, n] = acc
    c_ptrs = C_ptr + b * (S * N) + (offs_m[:, None] * N + offs_n[None, :])
    c_mask = (offs_m[:, None] < S) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


# Triton kernel: reduce sum over a vector (for demonstration)
@triton.jit
def reduce_sum_vec_kernel(vec_ptr, out_ptr, L, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < L
    v = tl.load(vec_ptr + offs, mask=mask, other=0.0)
    s = tl.sum(v, axis=0)
    tl.store(out_ptr + pid, s)


def launch_var_rstd_row(x_2d: torch.Tensor, out_1d: torch.Tensor, eps: float):
    N, H = x_2d.shape
    # Grid: one program per row
    grid = (N,)
    # Choose block size for H (e.g., 128)
    BLOCK_H = 128
    var_rstd_row_kernel[grid](
        x_2d, out_1d, N, H, eps, BLOCK_H=BLOCK_H,
        num_warps=4, num_stages=2
    )
    return out_1d


def launch_bmm_triton(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor, S: int, H: int, K: int, B: int, N: int):
    # Grid over (batch, S tiles, N tiles)
    BLOCK_M = 64  # tile over S
    BLOCK_N = 64  # tile over N
    BLOCK_K = 64  # tile over K
    grid = (B, (S + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    bmm_triton_kernel[grid](
        A, B, C, S, H, K, B, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=2
    )
    return C


def launch_reduce_sum_vec(vec: torch.Tensor, out_sum: torch.Tensor):
    L = vec.numel()
    BLOCK = 256
    grid = (triton.cdiv(L, BLOCK),)
    # reduce_sum_vec_kernel expects float32 inputs; ensure vec is float32
    reduce_sum_vec_kernel[grid](
        vec, out_sum, L, BLOCK=BLOCK,
        num_warps=4, num_stages=2
    )
    return out_sum
