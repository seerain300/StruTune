# Triton kernel: Fused normalize + linear (3 outputs) + tanh for one input vector
# Inputs:
#   x_ptr: *f32, input vector [hidden_size]
#   norm_w_ptr: *f32, norm_weight [hidden_size]
#   route_w_ptr: *f32, router_weight [3, hidden_size]
# Outputs:
#   out_ptr: *f32, length 4 per row: routed[0], routed[1], routed[2], tanh(routed[0])
@triton.jit
def normalize_linear_tanh_kernel(
    x_ptr,          # *f32, input vector [hidden_size]
    norm_w_ptr,     # *f32, norm_weight [hidden_size]
    route_w_ptr,    # *f32, router_weight [3, hidden_size]
    out_ptr,        # *f32, output [4] per row (indexing: 0=routed[0], 1=routed[1], 2=routed[2], 3=tanh(routed[0]))
    eps,            # f32
    hidden_size,    # runtime int
):
    pid = tl.program_id(0)
    # Sum over vector to compute mean of squares
    offs = tl.arange(0, hidden_size)
    x = tl.load(x_ptr + offs)
    x_sq = x * x
    sum_sq = tl.sum(x_sq, axis=0)
    mean = sum_sq / hidden_size
    rstd = tl.rsqrt(mean + eps)
    normalized = x * rstd
    normed = normalized * tl.load(norm_w_ptr + offs)

    # Compute routed for k=0,1,2
    # routed[k] = sum_j normed[j] * route_w[k, j]
    # We'll write routed[0..2] and tanh(routed[0]) to out_ptr
    for k in range(3):
        route_w_row = tl.load(route_w_ptr + k * hidden_size + offs)
        routed_k = tl.sum(normed * route_w_row, axis=0)
        tl.store(out_ptr + pid * 4 + k, routed_k)
    # tanh(routed[0])
    routed0 = tl.load(out_ptr + pid * 4 + 0)
    tl.store(out_ptr + pid * 4 + 3, tl.math.tanh(routed0))

#   K: runtime int
#   N: runtime int
@triton.jit
def gemv_t_sum_kernel(
    A_ptr,          # *f32, A[M, K]
    W_ptr,          # *f32, W[K, N]
    Y_ptr,          # *f32, Y[M, N]
    m,              # row index
    K,              # runtime int
    N,              # runtime int
):

    acc = tl.zeros([N], dtype=tl.float32)

        a_k = tl.load(A_ptr + pid * K + k)
        w_k = tl.load(W_ptr + k * N + tl.arange(0, N))
