import torch.nn as nn
import triton
import triton.language as tl


# Triton kernel: batched matmul
# C[b, m, n] = sum_k A[b, m, k] * B[b, n, k]
# A is a 1D buffer of length S*H*A; we view it as [S, H, A] in Python when launching.
# B is a 2D buffer of length B*A*A; we view it as [B, A, A] in Python when launching.
# C is a 3D buffer of length S*H*B; we view it as [S, H, B] in Python when launching.
@triton.jit
def bmm_triton_kernel(A_ptr, B_ptr, C_ptr,
                       S, H, A, B,  # shapes: A is [S,H,A], B is [B,A,A], C is [S,H,B]
                       BLOCK_M: tl.constexpr,  # tile size over S (m dimension)
                       BLOCK_N: tl.constexpr,  # tile size over B (n dimension)
                       BLOCK_K: tl.constexpr):  # tile size over A (k dimension)
    # Program id over (b, m-tile, n-tile)
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute indices for this tile
    m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # along S
    n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # along B (seq_len)
    k = tl.arange(0, BLOCK_K)                     # along A (altup_num_inputs, fixed 3)

    # Masks for boundaries
    mask_m = m < S
    mask_n = n < B
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    # We set num_warps=4 to keep good occupancy on common GPUs.
    for k0 in range(0, A, BLOCK_K):
        # A_ptr index: (((b*S + m)*H + k) * A) + b_offset * (A*A)
        # For fixed b, b*S + m = pid_b*S + m
        # So A_idx = (((pid_b*S + m)*H + k) * A) + b_offset * (A*A)
        # We need b_offset for B. Since b is the same for A and B (batched),
        # we can use b=pid_b directly.
        # Build pointer for A: A_ptr + (((pid_b*S + m)*H + (k0 + kk))*A)
        # Note: m is vector (BLOCK_M), k is vector (BLOCK_K).
        # We'll broadcast to (BLOCK_M, BLOCK_K).
        # We'll compute A_idx with broadcasting: ((pid_b*S + m)[:,None]*H + (k0 + k)[None,:]) * A
        # Compute A_idx for the current tile
        # A_idx = (((pid_b*S + m)[:, None]) * H + (k0 + k)[None, :]) * A
        # Then A_ptr + A_idx gives a [BLOCK_M, BLOCK_K] matrix of pointers.
        # Similarly for B: B_ptr + (pid_b*(A*A) + n[None, :]*(A) + (k0 + k)[:, None])
        # Compute A_idx: ((pid_b*S + m)[:, None] * H + (k0 + k)[None, :]) * A
        # Compute B_idx: (pid_b * (A*A) + n[None, :] * A + (k0 + k)[:, None])
        # Accumulate: acc += sum_k A[m, k] * B[n, k] over k
        # We'll do it via explicit indexing:
        # Create broadcasted indices
        m_i = pid_b * S + m[:, None]          # [BLOCK_M, 1]
        k_idx = (k0 + k)[None, :]             # [1, BLOCK_K]
        A_idx = (m_i * H + k_idx) * A        # [BLOCK_M, BLOCK_K]
        # Load A values: shape [BLOCK_M, BLOCK_K]
        A_vals = tl.load(A_ptr + A_idx, mask=mask_m[:, None], other=0.0)

        # Load B values: shape [BLOCK_N, BLOCK_K]
        # n is [BLOCK_N], (k0 + k) is [BLOCK_K]
        n_i = n[None, :]                      # [1, BLOCK_N]
        B_idx = pid_b * (A * A) + n_i * A + (k0 + k)[:, None]  # [BLOCK_N, BLOCK_K]
        B_vals = tl.load(B_ptr + B_idx, mask=mask_n[None, :], other=0.0)

        # Outer product accumulation: acc += A_vals @ B_vals.T
        # A_vals: [BM, BK], B_vals: [BN, BK] -> we need [BM, BK] * [BK, BN]
        # So we compute A_vals[:, :, None] * B_vals[None, :, :] over BK and sum
        # More simply, since BK is small (3), we can unroll:
        # For each kk in [0, BLOCK_K):
        #   acc += A_vals[:, kk] * B_vals[:, kk]
        # But Triton expects vectorized. Instead, we can do a reduction by multiplying elementwise and summing.
        # Use broadcasted multiply and reduce over K axis:
        # Create a [BM, BN, BK] by repeating along BN:
        # Broadcast B_vals along BM: [BM, BN, BK] via expanding
        # Triton doesn't support tl.dot with 2D in this context; we do explicit loop for K.
        # Implement loop over kk in BLOCK_K:
        # Note: Triton loops must have static trip count; BLOCK_K is constexpr.
        for kk in range(0, BLOCK_K):
            kval = k0 + kk
            # mask for k: valid_k = (kval < A)
            valid_k = kval < A
            # Load single column from A: shape [BM]
            A_col = tl.load(A_ptr + (((pid_b * S + m) * H + kval) * A), mask=mask_m, other=0.0)  # [BM]
            # Load single column from B: shape [BN]
            B_col = tl.load(B_ptr + (pid_b * (A * A) + n * A + kval), mask=mask_n, other=0.0)   # [BN]
            # Outer product: [BM, BN]
            acc += A_col[:, None] * B_col[None, :]

    # Store results to C: index = ((b*S + m) * H + n) * B
    C_idx = ((pid_b * S + m)[:, None] * H + n[None, :]) * B  # [BM, BN]
    C_mask = mask_m[:, None] & mask_n[None, :]
    # Store acc with mask
    tl.store(C_ptr + C_idx, acc, mask=C_mask)
