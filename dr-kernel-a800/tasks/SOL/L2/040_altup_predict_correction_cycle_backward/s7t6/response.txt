@triton.jit
def normalize_rstd_kernel(x_ptr, rstd_ptr, H: tl.constexpr, eps: tl.float32):
    pid = tl.program_id(0)
    # Each program handles one row
    # Load x
    offs = tl.arange(0, H)
    x = tl.load(x_ptr + pid * H + offs)
    x2 = x * x
    mean = tl.sum(x2, axis=0) / H
    rstd = tl.rsqrt(mean + eps)
    tl.store(rstd_ptr + pid, rstd)
