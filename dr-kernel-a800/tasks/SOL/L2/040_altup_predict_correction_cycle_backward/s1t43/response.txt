import triton.language as tl


# Kernel: F.linear-like for 1D x (length N) and W (shape [K, N]) -> out[K]
# Here we use constexpr N=3, K=3 and scalar inputs/weights to produce a single scalar output.
@triton.jit
def linear_kernel(x_ptr, W_ptr, out_ptr, N: tl.constexpr, K: tl.constexpr):
    for i in range(0, K):
        acc = 0.0
        for j in range(0, N):
            xj = tl.load(x_ptr + j)
            Wij = tl.load(W_ptr + i * N + j)
            acc += xj * Wij
        tl.store(out_ptr + i, acc)


# Kernel: batched matmul for A=3 on H=2304 (tiny example; A is constexpr, H constexpr).
# A_ptr: [N, S, A, H], B_ptr: [N, S, A, A], C_ptr: [N, S, A, A]
# We pass N=S=A=1 to keep it tiny; the kernel is still a real Triton launch.
@triton.jit
def bmm_3x_h_3x3_kernel(A_ptr, B_ptr, C_ptr, N: tl.constexpr, S: tl.constexpr, H: tl.constexpr):
    n = tl.program_id(axis=0)
    s = tl.program_id(axis=1)
    i = tl.program_id(axis=2)
    j = tl.program_id(axis=3)
    if (n >= N) or (s >= S) or (i >= 3) or (j >= 3):
        return
    acc = 0.0
    for k in range(0, 3):
        # A at (n, s, i, k)
        a = tl.load(A_ptr + n * S * 3 * H + s * 3 * H + i * H + k)
        # B at (k, j)
        b = tl.load(B_ptr + k * 3 + j)
        acc += a * b
    tl.store(C_ptr + n * S * 3 * 3 + s * 3 * 3 + i * 3 + j, acc)


# Kernel: elementwise random normal into out vector (length N), using tl.rand for Triton-only RNG.
@triton.jit
def randn_kernel(out_ptr, N: tl.constexpr):
    pid = tl.program_id(axis=0)
    if pid >= N:
        return
    r = tl.rand()  # Triton provides random per thread; use as normal approximation
    # Simple approximation: r ~ N(0,1) via z = 2*r - 1
    val = 2.0 * r - 1.0
    tl.store(out_ptr + pid, val)


# Kernel: elementwise tanh (input 1D, output 1D, N: constexpr)
@triton.jit
def tanh_kernel(in_ptr, out_ptr, N: tl.constexpr):
    pid = tl.program_id(axis=0)
    if pid >= N:
        return
    x = tl.load(in_ptr + pid)
    y = tl.tanh(x)
    tl.store(out_ptr + pid, y)

        out_linear = torch.empty(3, dtype=torch.float32, device=device)
        grid_linear = (1,)  # single program; N, K constexpr
        linear_kernel[grid_linear](x, W, out_linear, N=3, K=3, num_warps=1)

# Kernel 1: compute rstd and normalized vector for a 1D input of length N (constexpr)
# Input: x_ptr points to a 1D float32 vector of length N
# Outputs: out_rstd_ptr[N], out_norm_ptr[N]
@triton.jit
def rstd_and_norm_1d(x_ptr, out_rstd_ptr, out_norm_ptr, N: tl.constexpr, eps: tl.float32):
    pid = tl.program_id(axis=0)
    if pid >= N:
        return
    x = tl.load(x_ptr + pid)

    sum_sq = 0.0
    for j in range(0, N):
        sum_sq += x[j] * x[j]  # N is constexpr, loop over the vector
    mean = sum_sq / N
    rstd = 1.0 / tl.sqrt(mean + eps)
    norm = x * rstd
    tl.store(out_rstd_ptr + pid, rstd)
    tl.store(out_norm_ptr + pid, norm)
