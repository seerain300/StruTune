# Inputs:
#   x_ptr: *f32, vector [hidden_size]
#   norm_w_ptr: *f32, vector [hidden_size]

@triton.jit
def normalize_linear_tanh_kernel(
    x_ptr,                 # *f32, length = hidden_size
    norm_w_ptr,            # *f32, length = hidden_size
    route_w_ptr,           # *f32, length = 3 * hidden_size (we'll index as [k, j])
    out_ptr,               # *f32, length = 4
    hidden_size: tl.constexpr,
    eps: tl.constexpr,
):

    sum_x2 = 0.0

        x_j = tl.load(x_ptr + j)
        sum_x2 += x_j * x_j

        norm_w_j = tl.load(norm_w_ptr + j)
        x_scaled_j = x_j * rstd * norm_w_j
