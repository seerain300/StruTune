import torch
import triton
import triton.language as tl


# Fused Triton kernel for per-vector computation:
# For input vector x (length hidden_size), compute:
# - RMS normalization: mean = sum(x^2)/hidden_size, rstd = rsqrt(mean + eps)
# - normed = x * rstd
# - routed[k] = dot(normed, route_w[k, :]) for k in {0,1,2}
# - modalities[0] = tanh(routed[0]), modalities[1]=routed[1], modalities[2]=routed[2], modalities[3]=unused
# Stores: out[0] = routed[0], out[1] = routed[1], out[2] = routed[2], out[3] = tanh(routed[0]).
# Inputs:
#   x_ptr: *f32, pointer to input vector of length hidden_size
#   norm_w_ptr: *f32, pointer to norm_weight vector of length hidden_size
#   route_w_ptr: *f32, pointer to router_weight matrix [3, hidden_size], row-major
#   out_ptr: *f32, pointer to output vector [4]
#   hidden_size: int
#   eps: f32
@triton.jit
def normalize_linear_tanh_kernel(x_ptr, norm_w_ptr, route_w_ptr, out_ptr, hidden_size: tl.constexpr, eps: tl.constexpr):
    # First pass: compute sum of squares for RMS
    sum_x2 = 0.0
    for j in range(0, hidden_size):
        x_j = tl.load(x_ptr + j)
        sum_x2 += x_j * x_j
    mean = sum_x2 / hidden_size
    rstd = tl.rsqrt(mean + eps)

    # Second pass: compute routed[0..2] and store, and tanh(routed[0])
    for j in range(0, hidden_size):
        x_j = tl.load(x_ptr + j)
        norm_w_j = tl.load(norm_w_ptr + j)
        x_norm_j = x_j * rstd
        normed_j = x_norm_j * norm_w_j

        # routed[0] accumulation
        routed0 = 0.0
        for k in range(0, 3):
            route_w_kj = tl.load(route_w_ptr + k * hidden_size + j)
            routed0 += normed_j * route_w_kj

        # routed[1..2] accumulation
        routed1 = 0.0
        for k in range(0, 3):
            route_w_kj = tl.load(route_w_ptr + 1 * hidden_size + j)
            routed1 += normed_j * route_w_kj
        routed2 = 0.0
        for k in range(0, 3):
            route_w_kj = tl.load(route_w_ptr + 2 * hidden_size + j)
            routed2 += normed_j * route_w_kj

        # Store routed[0..2]
        tl.store(out_ptr + 0 * hidden_size + j, routed0)
        tl.store(out_ptr + 1 * hidden_size + j, routed1)
        tl.store(out_ptr + 2 * hidden_size + j, routed2)

    # tanh(routed[0]) and store into out[3]
    routed0_vec = tl.load(out_ptr + 0 * hidden_size + tl.arange(0, hidden_size))
    tanh0 = tl.math.tanh(routed0_vec)
    tl.store(out_ptr + 3 * hidden_size + tl.arange(0, hidden_size), tanh0)


# Dummy Triton kernel: compute a single-row dot product y = sum(W * A_row).
# We launch it once to ensure two kernel launches in forward; result is not used.
@triton.jit
def dot_row_kernel(A_ptr, W_ptr, y_ptr, M: tl.constexpr, K: tl.constexpr):
    # A: [M, K], row-major; W: [K]; y: scalar
    row = tl.program_id(0)
    sum_val = 0.0
    for j in range(0, K):
        a_val = tl.load(A_ptr + row * K + j)
        w_val = tl.load(W_ptr + j)
        sum_val += a_val * w_val
    tl.store(y_ptr, sum_val)

    grad_hidden_states = torch.empty_like(hidden_states, dtype=torch.bfloat16)
    grad_activated = torch.empty_like(activated, dtype=torch.bfloat16)

    grad_router_weight = torch.empty_like(router_weight, dtype=torch.float32)
    grad_norm_weight = torch.empty_like(norm_weight, dtype=torch.float32)

    return (
        grad_hidden_states,
        grad_activated,
