def normalize_and_tanh_rows_kernel(x_ptr, rstd_ptr, routed_tanh_ptr,
                                   M: tl.constexpr, H: tl.constexpr, L: tl.constexpr):

# Triton kernel: compute coef = F.linear(modalities_tanh, prediction_coef_weight)

        w = tl.load(pred_coef_weight_ptr + pid_k * L + l)

        torch.empty_like(x, dtype=torch.bfloat16),
        torch.empty_like(router_weight, dtype=torch.bfloat16, device=x.device),
        torch.empty_like(norm_weight, dtype=torch.bfloat16, device=x.device),
