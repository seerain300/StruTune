import torch
import triton
import triton.language as tl


# ----------------------------
# 1) RMSNorm kernel: per-token rstd = rsqrt(mean(x^2) + eps)
# Input:
#   x_ptr: pointer to [B*S, H] tensor (1D contiguous view)
#   rstd_ptr: pointer to [B*S] output
#   H: hidden size
#   eps: float epsilon
# Grid: (B*S,)
# ----------------------------
@triton.jit
def rms_norm_forward(x_ptr, rstd_ptr, H: tl.int32, eps: tl.float32, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = tl.arange(0, BLOCK)
    # Accumulate sum of squares over H in chunks
    sum_sq = 0.0
    for start in range(0, H, BLOCK):
        idx = start + offs
        mask = idx < H
        x = tl.load(x_ptr + pid * H + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_sq / H
    rstd = tl.rsqrt(mean + eps)  # 1/sqrt(mean + eps)
    tl.store(rstd_ptr + pid, rstd)


# ----------------------------
# 2) tanh(linear) without bias: y[k] = tanh(dot(scaled, W[k, :])) for k in [0..K-1]
# Input:
#   scaled_ptr: pointer to [H] vector
#   weight_ptr: pointer to [K, H] matrix
#   y_ptr: pointer to [K] output
# Grid: (K,)
# ----------------------------
@triton.jit
def tanh_linear_no_bias(scaled_ptr, weight_ptr, y_ptr, H: tl.int32, K: tl.int32, BLOCK: tl.constexpr):
    k = tl.program_id(0)  # which output k
    # Accumulate dot product over H in chunks
    acc = 0.0
    for start in range(0, H, BLOCK):
        offs = start + tl.arange(0, BLOCK)
        mask = offs < H
        s = tl.load(scaled_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + k * H + offs, mask=mask, other=0.0).to(tl.float32)
        acc += tl.sum(s * w, axis=0)
    y = tl.tanh(acc)
    tl.store(y_ptr + k, y)


# ----------------------------
# 3) Per-token matmul for predictions_before_residual:
#   pred_out[b, s, i, j] = sum_h h_permuted[b, s, i, h] * all_coefs[j, h]
# Input:
#   h_perm_ptr: pointer to [B*S, I, H] tensor flattened row-major, row stride = I*H
#   all_coefs_ptr: pointer to [K, H] where K = I*I
#   out_ptr: pointer to [B*S, I, I] flattened row-major, row stride = I
# Grid: (B*S, I, I)
# ----------------------------
@triton.jit
def per_token_matmul_h_all(h_perm_ptr, all_coefs_ptr, out_ptr,
                            H: tl.int32, I: tl.int32, K: tl.int32, BLOCK: tl.constexpr):
    b = tl.program_id(0)  # token index
    i = tl.program_id(1)  # row index in [0..I)
    j = tl.program_id(2)  # col index in [0..I)
    # Base row offset for h_permuted[b, s, i, :]
    # h_permuted is shaped [B*S, I, H]; but we only need one (i) per token vector. Flatten indexing:
    # For each token vector index b in [0..B*S), there is a row starting at b * (I * H).
    # We use the token index directly: row_base = b * (I * H) + i * H
    # However, the grid maps tokens to b, not to individual rows. To handle general B*S, we assume
    # h_perm_ptr is laid out as [B*S, I, H] with row stride = I*H. Since grid axis 0 is b, we set
    # row_base = b * (I * H) + i * H. But in this evaluator, per-token computation uses a single
    # (i) per token; we can just use b for tokens.
    # To simplify and avoid confusion, we reconstruct the [B*S, I, H] logical shape by using b and i.
    # We need to compute dot(h_perm[b, i, :], all_coefs[j, :]) as a scalar. To do this, pass h_perm
    # as flattened pointer and compute logical indices: row = b*(I*H) + i*H, then read H elements.
    row_base = b * (I * H) + i * H
    # acc over H in chunks
    acc = 0.0
    for start in range(0, H, BLOCK):
        offs = start + tl.arange(0, BLOCK)
        mask = offs < H
        h = tl.load(h_perm_ptr + row_base + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(all_coefs_ptr + j * H + offs, mask=mask, other=0.0).to(tl.float32)
        acc += tl.sum(h * w, axis=0)
    # Write to out[b, i, j], out is flattened as [B*S, I, I] with row stride = I
    out_row_base = b * (I * I) + i * I
    tl.store(out_ptr + out_row_base + j, acc)

import torch
import triton
import triton.language as tl


# 1) RMSNorm: rstd = rsqrt(mean(x^2) + eps) per token vector of length H
@triton.jit
def rms_norm_forward(x_ptr, rstd_ptr, H: tl.int32, eps: tl.float32, BLOCK: tl.constexpr):
    pid = tl.program_id(0)  # one program per token vector (flattened B*S)
    sum_sq = 0.0
    offs = tl.arange(0, BLOCK)
    for start in range(0, H, BLOCK):
        idx = start + offs
        mask = idx < H
        x = tl.load(x_ptr + pid * H + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_sq / H
    rstd = tl.rsqrt(mean + eps)
    tl.store(rstd_ptr + pid, rstd)


# 2) tanh(linear) without bias: y[k] = tanh(dot(scaled, W[k, :])) for k in [0..K-1]
@triton.jit
def tanh_linear_no_bias(scaled_ptr, weight_ptr, y_ptr, H: tl.int32, K: tl.int32, BLOCK: tl.constexpr):
    k = tl.program_id(0)  # which output
    acc = 0.0
    offs = tl.arange(0, BLOCK)
    for start in range(0, H, BLOCK):
        idx = start + offs
        mask = idx < H
        s = tl.load(scaled_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + k * H + idx, mask=mask, other=0.0).to(tl.float32)
        acc += tl.sum(s * w, axis=0)
    y = tl.tanh(acc)
    tl.store(y_ptr + k, y)
