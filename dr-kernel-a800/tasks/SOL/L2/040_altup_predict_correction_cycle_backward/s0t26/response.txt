# We view A as [S, H, A], B as [B, A, A], C as [S, H, B]
@triton.jit
def bmm_triton_kernel_1d(A_ptr, B_ptr, C_ptr,
                          S, H, B,
                          stride_A_S, stride_A_H, stride_A_K,
                          stride_B_N, stride_B_K, stride_B_K2,  # second K dim stride (A)
                          stride_C_S, stride_C_H, stride_C_N,
                          A_size,  # A (K dimension)
                          BLOCK_K: tl.constexpr):
    b = tl.program_id(0)  # batch S
    m = tl.program_id(1)  # row H
    n = tl.program_id(2)  # col B

    # Accumulator
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over K in chunks
    for k0 in range(0, A_size, BLOCK_K):
        # Indices for A[b, m, k]
        a_off = b * stride_A_S + m * stride_A_H + (k0 + tl.arange(0, BLOCK_K)) * stride_A_K
        # Mask for A
        mask_a = (k0 + tl.arange(0, BLOCK_K)) < A_size

        # Indices for B[b, n, k]
        # B is [B, A, A], we index by n (rows) and (k0 + arange) (cols)
        b_off = n * stride_B_N + (k0 + tl.arange(0, BLOCK_K)) * stride_B_K
        # Mask for B
        mask_b = (k0 + tl.arange(0, BLOCK_K)) < A_size

        # Load A and B chunks (promote to fp32)
        a_chunk = tl.load(A_ptr + a_off, mask=mask_a, other=0.0)
        b_chunk = tl.load(B_ptr + b_off, mask=mask_b, other=0.0)

        # Cast to fp32 for accumulation
        a_chunk = a_chunk.to(tl.float32)
        b_chunk = b_chunk.to(tl.float32)

        # Accumulate dot product over BLOCK_K
        # a_chunk is [BLOCK_K], b_chunk is [BLOCK_K] -> sum
        acc += tl.sum(a_chunk * b_chunk, axis=0)
