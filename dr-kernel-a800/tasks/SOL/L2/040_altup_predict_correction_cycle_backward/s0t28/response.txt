# Triton kernel: per-row variance + rsqrt for 2D tensor [N, H]
# Computes rstd[i] = rsqrt(mean_j(x[i, j]^2) + eps), written to out[N]
@triton.jit
def var_rstd_row_kernel(x_ptr, out_ptr, N, H, eps, BLOCK_H: tl.constexpr):
    row = tl.program_id(0)  # 0..N-1
    if row >= N:
        return
    sumsq = tl.zeros((), dtype=tl.float32)
    # Loop over columns in chunks of BLOCK_H
    for off in range(0, H, BLOCK_H):
        cols = off + tl.arange(0, BLOCK_H)
        mask = cols < H
        # Load row elements; for masked lanes, load 0
        x = tl.load(x_ptr + row * H + cols, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / H
    rstd = tl.math.rsqrt(mean + eps)
    tl.store(out_ptr + row, rstd)


# Triton kernel: GEMV-like for routed = tanh(a @ W), where a is [A], W is [A, OUT]
# We implement routed_out[L] = tanh(sum_k a[k] * W[k, L]) for each L in 0..OUT-1.
# Inputs:
#   a_ptr: 1D float32, length A
#   W_ptr: 2D float32, shape [A, OUT]
#   routed_ptr: 1D float32, length OUT (output)
@triton.jit
def linear_gemv_tanh_kernel(a_ptr, W_ptr, routed_ptr, A, OUT, BLOCK_K: tl.constexpr):
    out_idx = tl.program_id(0)  # 0..OUT-1
    if out_idx >= OUT:
        return
    acc = tl.zeros((), dtype=tl.float32)
    # Dot-product accumulation over K=A in chunks
    for off in range(0, A, BLOCK_K):
        k = off + tl.arange(0, BLOCK_K)
        mask_k = k < A
        a = tl.load(a_ptr + k, mask=mask_k, other=0.0)
        # Load W[k, out_idx] vector; mask invalid k
        w = tl.load(W_ptr + k * OUT + out_idx, mask=mask_k, other=0.0)
        acc += tl.sum(a * w, axis=0)
    acc = tl.math.tanh(acc)
    tl.store(routed_ptr + out_idx, acc)


# Triton kernel: batched matmul
# C[b, m, n] = sum_k A[b, m, k] * B[b, n, k]
# A is a 1D buffer of length S*H*A; we view it as [S, H, A].
# B is a 2D buffer of length B*A*A; we view it as [B, A, A].
# C is a 3D buffer of length S*H*B; we view it as [S, H, B].
@triton.jit
def bmm_triton_kernel(A_ptr, B_ptr, C_ptr,
                      S: tl.constexpr, H: tl.constexpr, A: tl.constexpr, B: tl.constexpr,
                      STRIDE_A_B, STRIDE_A_M, STRIDE_A_K,
                      STRIDE_B_N, STRIDE_B_K, STRIDE_B_A,
                      STRIDE_C_B, STRIDE_C_M, STRIDE_C_N,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    b = tl.program_id(0)  # batch index
    m = tl.program_id(1)  # row in H
    n = tl.program_id(2)  # col in B

    # Accumulator
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over K=A in chunks
    for k0 in range(0, A, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < A

        # Load A[b, m, k] as a vector of length BLOCK_K
        a_vec = tl.load(
            A_ptr + b * STRIDE_A_B + m * STRIDE_A_M + k * STRIDE_A_K,
            mask=mask_k,
            other=0.0
        )
        a_vec = a_vec.to(tl.float32)

        # Load B[n, k, A] as a vector of length BLOCK_K (note: we only need B[n, k] across k)
        b_vec = tl.load(
            B_ptr + n * STRIDE_B_N + k * STRIDE_B_K,
            mask=mask_k,
            other=0.0
        )
        b_vec = b_vec.to(tl.float32)

        # Accumulate dot product for this chunk
        acc += tl.sum(a_vec * b_vec, axis=0)

    # Store result to C[b, m, n]
    tl.store(C_ptr + b * STRIDE_C_B + m * STRIDE_C_M + n * STRIDE_C_N, acc)


# Triton kernel: simple reduction over a 1D vector of length L
@triton.jit
def reduce_sum_vec_kernel(x_ptr, out_ptr, L: tl.constexpr, BLOCK: tl.constexpr):
    idx = tl.program_id(0)
    if idx != 0:
        return
    total = tl.zeros((), dtype=tl.float32)
    for off in range(0, L, BLOCK):
        v = tl.load(x_ptr + off + tl.arange(0, BLOCK), mask=(off + tl.arange(0, BLOCK) < L), other=0.0)
        total += tl.sum(v, axis=0)
    tl.store(out_ptr, total)
