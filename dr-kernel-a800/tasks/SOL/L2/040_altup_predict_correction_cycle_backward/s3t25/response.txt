# Triton kernel: reduce sum of squares across H for each (b, s)
@triton.jit
def sum_squares_reduce_kernel(x_ptr, out_ptr, H: tl.constexpr, BLOCK_H: tl.constexpr):
    """
    Grid: (B*S,)
    For each pid = b*s:
    Accumulate sum(x[b, s, :])^2 into out[pid].
    We loop across H in tiles and use atomic_add to aggregate.
    """
    pid = tl.program_id(axis=0)
    total = 0.0
    for h0 in range(0, H, BLOCK_H):
        offs = h0 + tl.arange(0, BLOCK_H)
        mask = offs < H
        # x[b, s, :] is contiguous with stride H in the (b, s) row
        x = tl.load(x_ptr + pid * H + offs, mask=mask, other=0.0)
        sq = x * x
        total += tl.sum(sq, axis=0)
    # Accumulate into out[pid]
    tl.atomic_add(out_ptr + pid, total)


# Triton kernel: elementwise rsqrt for N elements
@triton.jit
def rsqrt_kernel(inp_ptr, out_ptr, N, eps, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N
    x = tl.load(inp_ptr + offsets, mask=mask, other=0.0)
    inv_std = 1.0 / tl.sqrt(x + eps)
    tl.store(out_ptr + offsets, inv_std, mask=mask)


# Triton kernel: elementwise tanh via exp

    y = (e2z - 1.0) / (e2z + 1.0)
    tl.store(out_ptr + offsets, y, mask=mask)

                  M, N, K,
                  stride_a_m, stride_a_n,
                  stride_w_n, stride_w_k,
                  BLOCK_N: tl.constexpr):

    Compute Out[m, k] = sum_n A[m, n] * W[n, k] for m in [0, M), k fixed by program_id(axis=1).
    We launch with grid=(M, K). For each (m, k), we reduce over N.

        n_idx = n0 + tl.arange(0, BLOCK_N)
        mask_n = n_idx < N
        a = tl.load(A_ptr + pid_m * stride_a_m + n_idx * stride_a_n,
                    mask=mask_n, other=0.0)
        w = tl.load(W_ptr + n_idx * stride_w_n + pid_k * stride_w_k,
                    mask=mask_n, other=0.0)
        acc += tl.sum(a * w, axis=0)
    out_index = pid_m * K + pid_k
    tl.store(Out_ptr + out_index, acc)

    For each pid = b*s:

@triton.jit
def matvec_kernel(A_ptr, W_ptr, Out_ptr,
                  M, N, K,
                  stride_a_m, stride_a_n,

# Grid: (B, S, tiles over H). Axis2 is split into tiles of size BLOCK_H. We compute partial[H_tile, K] and atomic_add into out.
@triton.jit
def assemble_predictions_kernel(hidden_ptr, all_coefs_ptr, out_ptr,
                                 H, B, S, N, K,
                                 stride_h_b, stride_h_s, stride_h_h, stride_h_n,
                                 stride_ac_b, stride_ac_s, stride_ac_n, stride_ac_k,

    Program grid: (B, S, ceil(H / BLOCK_H))
    Each program computes predictions for a tile of H over a fixed (b, s), accumulating into out[b, s, :, :]
    using atomic_add. This avoids torch.bmm.

            valid_h = hh < H
            # Load hidden[b, s, hh, :N]
            hidden_row_ptrs = hidden_ptr + b * stride_h_b + s * stride_h_s + hh * stride_h_h + tl.arange(0, N) * stride_h_n
            mask_h = valid_h
            hidden_vals = tl.load(hidden_row_ptrs, mask=mask_h, other=0.0)  # shape [N], vector load
            # Load all_coefs[b, s, :, k]
            ac_ptrs = all_coefs_ptr + b * stride_ac_b + s * stride_ac_s + tl.arange(0, N) * stride_ac_n + k * stride_ac_k
            ac_vals = tl.load(ac_ptrs, mask=(tl.arange(0, N) < N), other=0.0)  # [N]
            # Multiply and reduce over N
            partial[h] = tl.sum(hidden_vals * ac_vals, axis=0)
        # Atomic add partial to out[b, s, h:h+BLOCK_H, k]
        # out_ptr index: b*stride_out_b + s*stride_out_s + hh*stride_out_h + k*stride_out_k
        for h in range(0, BLOCK_H):
            hh = h0 + h
            valid_h = hh < H
            out_ptrs = out_ptr + b * stride_out_b + s * stride_out_s + hh * stride_out_h + k * stride_out_k
            tl.atomic_add(out_ptrs, partial[h], mask=valid_h)


def _launch_sum_squares(hidden: torch.Tensor):
    """
    hidden: [H, B, S], float32 contiguous
    Returns var[b*s] of shape [B*S] in float32.
    """
    B, S = hidden.shape[1], hidden.shape[2]
    H = hidden.shape[0]
    var = torch.zeros(B * S, device=hidden.device, dtype=torch.float32)
    grid = (B * S,)
    BLOCK_H = 128
    sum_squares_reduce_kernel[grid](hidden, var, H, BLOCK_H, num_warps=4)
    return var


def _launch_rsqrt(var: torch.Tensor, eps: float):
    """
    var: [B*S] float32
    Returns rstd[b*s] of shape [B*S] in float32.
    """
    B_S = var.numel()
    rstd = torch.empty(B_S, device=var.device, dtype=torch.float32)
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(B_S, BLOCK_SIZE),)
    rsqrt_kernel[grid](var, rstd, B_S, eps, BLOCK_SIZE, num_warps=4)
    return rstd


def _launch_tanh(x: torch.Tensor):
    """
    x: 1D tensor float32
    Returns tanh(x) in float32.
    """
    N = x.numel()
    out = torch.empty(N, device=x.device, dtype=torch.float32)
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(N, BLOCK_SIZE),)
    tanh_kernel[grid](x, out, N, BLOCK_SIZE, num_warps=4)
    return out


def _launch_gemm_vec(A: torch.Tensor, W: torch.Tensor):
    """
    A: [M, N], float32 contiguous
    W: [N, K], float32 contiguous
    Returns Out[M, K] in float32.
    """
    M, N = A.shape
    K = W.shape[1]
    Out = torch.empty(M * K, device=A.device, dtype=torch.float32)
    stride_a_m = A.stride(0)
    stride_a_n = A.stride(1)
    stride_w_n = W.stride(0)
    stride_w_k = W.stride(1)
    grid = (M, K)
    BLOCK_N = 128
    matvec_kernel[grid](A, W, Out, M, N, K,
                        stride_a_m, stride_a_n,
                        stride_w_n, stride_w_k,
                        BLOCK_N, num_warps=4)
    return Out.view(M, K)


def _assemble_predictions(hidden: torch.Tensor, all_coefs: torch.Tensor, B: int, S: int, H: int, K: int):
    """
    hidden: [H, B, S, N], float32 contiguous
    all_coefs: [B, S, N, K], float32 contiguous
    Returns predictions: [H, B, S, K], float32
    We avoid torch.bmm and use Triton kernel tiling over H and K with atomic_add.
    """
    # Prepare output
    predictions = torch.zeros(H, B * S * K, device=hidden.device, dtype=torch.float32)
    # We will write out in [H, B, S, K] layout: out[b*s*k, h] where b*s*k is linear index
    # Compute grid and strides
    # Create a temporary 3D out [H, B, S*K] and write per k; but we'll use atomic_add into a [H, B, S, K] tensor.
    out = torch.zeros(H, B, S, K, device=hidden.device, dtype=torch.float32)
    # Strides for out
    stride_out_b = out.stride(0)  # 0
    stride_out_s = out.stride(1)  # K
    stride_out_h = out.stride(2)  # B*K*S -> actually 1 for out.stride(2)=K, but for [H,H,S,K] we need H stride:
    # Note: out has shape (H, B, S, K), so strides are:
    stride_out_h = out.stride(0)  # H stride for first dim
    stride_out_k = out.stride(3)  # 1
    # Strides for hidden [H, B, S, N]
    stride_h_b = hidden.stride(0)  # H
    stride_h_s = hidden.stride(1)  # S*N
    stride_h_h = hidden.stride(2)  # N
    stride_h_n = hidden.stride(3)  # 1
    # Strides for all_coefs [B, S, N, K]
    stride_ac_b = all_coefs.stride(0)  # S*N*K
    stride_ac_s = all_coefs.stride(1)  # N*K
    stride_ac_n = all_coefs.stride(2)  # K
    stride_ac_k = all_coefs.stride(3)  # 1
    # Launch Triton kernel with grid over (B, S, tiles over H)
    tiles_H = triton.cdiv(H, 128)
    grid = (B, S, tiles_H)
    BLOCK_H = 128
    BLOCK_K = K  # since K=9, fine
    assemble_predictions_kernel[grid](
        hidden, all_coefs, out
