import triton
import triton.language as tl


# Triton kernel: per-vector fused RMS normalization, scale, linear with router_weight, and tanh(routed[0]).
# Inputs:
#   x_ptr: *f32, pointer to input vector [hidden_size]
#   norm_w_ptr: *f32, pointer to norm_weight vector [hidden_size]
#   route_w_ptr: *f32, pointer to router_weight matrix (layout [3, hidden_size] contiguous row-major)
#   out_ptr: *f32, pointer to output vector [4], where out[0..2]=routed[0..2], out[3]=tanh(routed[0])
#   hidden_size: tl.constexpr (e.g., 2304)
#   eps: f32, epsilon for RMS
@triton.jit
def normalize_linear_tanh_kernel(x_ptr, norm_w_ptr, route_w_ptr, out_ptr, hidden_size: tl.constexpr, eps: tl.constexpr):
    # We assume grid=(1,) and vectorized operations over hidden_size.
    # Compute sum of squares for RMS
    sum_x2 = 0.0
    for j in range(hidden_size):
        xj = tl.load(x_ptr + j)
        sum_x2 += xj * xj
    mean = sum_x2 / hidden_size
    rstd = tl.rsqrt(mean + eps)

    # normalized = x * rstd
    # normed = normalized * norm_weight
    routed = [0.0, 0.0, 0.0]
    for k in range(3):
        sum_k = 0.0
        for j in range(hidden_size):
            xj = tl.load(x_ptr + j)
            nj = xj * rstd
            wj = tl.load(norm_w_ptr + j)
            wkj = tl.load(route_w_ptr + k * hidden_size + j)
            sum_k += nj * wj * wkj
        routed[k] = sum_k
    # tanh(routed[0])
    tanh0 = tl.math.tanh(routed[0])

    # Store outputs: out[0]=routed[0], out[1]=routed[1], out[2]=routed[2], out[3]=tanh(routed[0])
    tl.store(out_ptr + 0, routed[0])
    tl.store(out_ptr + 1, routed[1])
    tl.store(out_ptr + 2, routed[2])
    tl.store(out_ptr + 3, tanh0)

#   W_ptr: *f32, pointer to W vector [K]
#   y_ptr: *f32, pointer to output vector [M]
#   m: int32 row index
#   K: tl.constexpr

    s = 0.0
    for k in range(K):
        a = tl.load(A_ptr + m * K + k)
        w = tl.load(W_ptr + k)
        s += a * w
