# Triton kernel: per-row variance + rsqrt for 2D tensor [N, H]
# Computes rstd[i] = rsqrt(mean_j(x[i, j]^2) + eps), written to out[N]
@triton.jit
def var_rstd_row_kernel(x_ptr, out_ptr, N, H, eps, BLOCK_H: tl.constexpr):
    row = tl.program_id(0)  # 0..N-1
    if row >= N:
        return
    sumsq = tl.zeros((), dtype=tl.float32)
    # loop over columns in chunks
    for col_start in range(0, H, BLOCK_H):
        cols = col_start + tl.arange(0, BLOCK_H)
        mask = cols < H
        vals = tl.load(x_ptr + row * H + cols, mask=mask, other=0.0)  # [BLOCK_H], float32
        sumsq += tl.sum(vals * vals, axis=0)
    mean = sumsq / H
    rstd = tl.rsqrt(mean + eps)
    tl.store(out_ptr + row, rstd)


# Triton kernel: 2D GEMV with tanh, C[b, h] = tanh(sum_k A[b, h, k] * W[k, A])
# A: [B, H, K] flattened as [B, H*K], W: [K, A], C: [B, A]
@triton.jit
def small_gemv_triton_2d(A_ptr, W_ptr, C_ptr, B, H, A, K: tl.constexpr, BLOCK_K: tl.constexpr):
    b = tl.program_id(0)  # 0..B-1
    h = tl.program_id(1)  # 0..H-1
    acc = tl.zeros((), dtype=tl.float32)
    # loop over K in tiles
    for k_start in range(0, K, BLOCK_K):
        ks = k_start + tl.arange(0, BLOCK_K)  # [BLOCK_K]
        mask = ks < K
        # A[b, h, ks] flattened index: b*H*K + h*K + ks
        a_vals = tl.load(A_ptr + b * H * K + h * K + ks, mask=mask, other=0.0)  # [BLOCK_K]
        w_vals = tl.load(W_ptr + ks * A, mask=mask, other=0.0)  # [BLOCK_K]
        acc += tl.sum(a_vals * w_vals, axis=0)
    # apply tanh
    out = tl.math.tanh(acc)
    tl.store(C_ptr + b * A + h, out)

    b = tl.program_id(0)  # 0..B-1
    s = tl.program_id(1)  # 0..S-1
    n = tl.program_id(2)  # 0..H-1  wait, B=3, but we want n in 0..2? No: we compute C[b, s, n] for n in 0..2 across s,h.
    # Fix: we launch grid (B, S, H), and inside we compute C[b, s, n] for n fixed by n = program_id(2) % 3?
    # Better: launch grid (B, S, 3), compute one n per program. To cover n=0..2, we set grid (B, S, 3) and branch.
    # Since grid is (B, S, H), we use n = tl.program_id(2), and we launch total programs B*S*H but only handle n in 0..2?
    # Simpler: define grid as (B, S, 3). Then b=pid0, s=pid1, n=pid2 (0..2). We implement n via tl.program_id(2).
    # Therefore, use grid (B, S, 3) for this kernel.
    # The code below assumes grid is (B, S, 3); adjust launch accordingly in forward.
    # Compute scalar acc
    acc = tl.zeros((), dtype=tl.float32)
    for k_start in range(0, H, BLOCK_K):
        ks = k_start + tl.arange(0, BLOCK_K)  # [BLOCK_K], over hidden dimension
        mask_k = ks < H
        # A[b, s, ks]: index = b*S*H + s*H + ks
        a_vals = tl.load(A_ptr + b * S * H + s * H + ks, mask=mask_k, other=0.0)  # [BLOCK_K]
        # B[b, n, ks]: index = b*B*H + n*H + ks (with B=3) but we can generalize B by passing B and n

        b_vals = tl.load(B_ptr + n * H + ks, mask=mask_k, other=0.0)  # [BLOCK_K]

    # Store to C: C is flattened [B, S, 3] -> index = b*(S*3) + s*3 + n
