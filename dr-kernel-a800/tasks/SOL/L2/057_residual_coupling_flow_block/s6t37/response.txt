import math
import torch
import torch.nn.functional as F

import triton
import triton.language as tl


@triton.jit
def conv1d_k5_p2(
    x_ptr,         # *float, [B, C_in, T_in]
    w_ptr,         # *float, [C_out, C_in, 5]
    b_ptr,         # *float, [C_out]
    y_ptr,         # *float, [B, C_out, T_out]
    B, C_in, T_in, C_out, T_out,
    BLOCK_T: tl.constexpr,
):
    # program ids
    b_id = tl.program_id(0)
    co = tl.program_id(1)

    # time offsets for this program
    t_offsets = tl.program_id(2) * BLOCK_T + tl.arange(0, BLOCK_T)
    t_mask = t_offsets < T_out

    # accumulate result for these offsets
    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # loop over input channels and kernel taps
    for ci in range(0, C_in):
        for k in range(0, 5):
            # valid conv with padding=2: index = t_offsets + 2 - k
            t_idx = t_offsets + 2 - k
            in_bounds = (t_idx >= 0) & (t_idx < T_in) & t_mask

            # compute x offset: ((b * C_in + ci) * T_in + t_idx)
            x_offsets = (b_id * C_in + ci) * T_in + t_idx
            # load x, masked for out-of-bounds
            x_val = tl.load(x_ptr + x_offsets, mask=in_bounds, other=0.0)

            # load weight scalar w[co, ci, k]
            w_off = co * (C_in * 5) + ci * 5 + k
            w_val = tl.load(w_ptr + w_off)

            acc += x_val * w_val

    # add bias
    b_val = tl.load(b_ptr + co)
    acc = acc + b_val

    # store to y
    y_offsets = (b_id * C_out + co) * T_out + t_offsets
    tl.store(y_ptr + y_offsets, acc, mask=t_mask)


@triton.jit
def add_bias(
    y_ptr,     # *float, [B, C_out, T]
    b_ptr,     # *float, [C_out]
    B, C_out, T,
):
    b_id = tl.program_id(0)
    co = tl.program_id(1)
    t_offsets = tl.program_id(2) * 128 + tl.arange(0, 128)
    mask = t_offsets < T

    y_offsets = (b_id * C_out + co) * T + t_offsets
    val = tl.load(y_ptr + y_offsets, mask=mask, other=0.0)
    b_val = tl.load(b_ptr + co)
    val = val + b_val
    tl.store(y_ptr + y_offsets, val, mask=mask)


@triton.jit
def relu_inplace(
    y_ptr,     # *float, [B, C_out, T]
    B, C_out, T,
):
    b_id = tl.program_id(0)
    co = tl.program_id(1)
    t_offsets = tl.program_id(2) * 128 + tl.arange(0, 128)
    mask = t_offsets < T

    y_offsets = (b_id * C_out + co) * T + t_offsets
    val = tl.load(y_ptr + y_offsets, mask=mask, other=0.0)
    val = tl.maximum(val, 0.0)
    tl.store(y_ptr + y_offsets, val, mask=mask)


@triton.jit
def mul_mask_broadcast(
    y_ptr,         # *float, [B, C_out, T]
    mask_ptr,      # *float, [B, 1, T] (we load as [B, T] ignoring channel dim=1)
    B, C_out, T,
):
    # Note: mask has shape [B, 1, T]; we index as [B, T] by ignoring channel dim=1
    b_id = tl.program_id(0)
    co = tl.program_id(1)
    t_offsets = tl.program_id(2) * 128 + tl.arange(0, 128)
    mask_t = t_offsets < T

    y_offsets = (b_id * C_out + co) * T + t_offsets
    y_val = tl.load(y_ptr + y_offsets, mask=mask_t, other=0.0)

    # mask_ptr[b_id, 0, t_offsets]
    mask_offsets = b_id * T + t_offsets
    mask_val = tl.load(mask_ptr + mask_offsets, mask=mask_t, other=1.0)

    y_val = y_val * mask_val
    tl.store(y_ptr + y_offsets, y_val, mask=mask_t)


@triton.jit
def add_broadcast_to_channels(
    y_src_ptr,  # *float, [B, C_sub, T_sub] (e.g., [B, 96, T-3])
    y_dst_ptr,  # *float, [B, 192, T_sub] (e.g., [B, 192, T-3])
    B, C_sub, T_sub,
):
    # copy y_src (C_sub channels) to y_dst (2*C_sub channels) and replicate across channels
    b_id = tl.program_id(0)
    sub_c = tl.program_id(1)  # 0..C_sub-1
    dst_c = sub_c * 2          # map 96 -> 192
    t_offsets = tl.program_id(2) * 128 + tl.arange(0, 128)
    mask = t_offsets < T_sub

    src_offsets = (b_id * C_sub + sub_c) * T_sub + t_offsets
    src_val = tl.load(y_src_ptr + src_offsets, mask=mask, other=0.0)

    dst_offsets = (b_id * 192 + dst_c) * T_sub + t_offsets
    tl.store(y_dst_ptr + dst_offsets, src_val, mask=mask)
