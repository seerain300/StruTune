@triton.jit
def conv0_kernel(x_ptr, w_ptr, y_ptr,
                 B, Ci, Co, T, T_out,
                 x_stride_b, x_stride_c, x_stride_t,
                 w_stride_ci, w_stride_co, w_stride_k,

    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    co = pid0 % Co
    b = pid0 // Co
    t0 = pid1 * BLOCK_T
    t_offsets = t0 + tl.arange(0, BLOCK_T)
    t_mask = t_offsets < T_out

    # accumulator for this output channel and time block
    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # loop over input channels and kernel taps
    # Ci is 96, which is fine as a runtime loop bound here (simple kernel)
    for ci in range(0, Ci):
        for k in range(0, 5):
            t_in = t_offsets + k - 2  # padding=2

            mask = (t_in >= 0) & (t_in < T) & t_mask
