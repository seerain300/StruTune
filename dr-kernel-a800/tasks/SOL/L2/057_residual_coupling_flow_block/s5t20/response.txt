import math
import torch
import torch.nn as nn
import triton
import triton.language as tl


def _ceil_div(a, b):
    return (a + b - 1) // b


@triton.jit
def conv1d_triton_relu(
    x_ptr,            # *float32, input [B, Cin, T_in]
    w_flat_ptr,       # *float32, weights flattened to [Cout, Cin*K]
    b_ptr,            # *float32, bias [Cout]
    y_ptr,            # *float32, output [B, Cout, T_out]
    B: tl.constexpr,
    Cin: tl.constexpr,
    T_in: tl.constexpr,
    Cout: tl.constexpr,
    K: tl.constexpr,  # kernel size
    PADDING: tl.constexpr,
    x_stride_b: tl.constexpr, x_stride_c: tl.constexpr, x_stride_t: tl.constexpr,
    w_stride_co: tl.constexpr, w_stride_ci: tl.constexpr,  # weights strides
    y_stride_b: tl.constexpr, y_stride_c: tl.constexpr, y_stride_t: tl.constexpr,
    BLOCK_CO: tl.constexpr, BLOCK_POS: tl.constexpr, BLOCK_CIN: tl.constexpr,
):
    # Grid: (B, ceil(Cout/BLOCK_CO), ceil(T_out/BLOCK_POS))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_pos = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    pos_offsets = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)

    co_mask = co_offsets < Cout
    pos_mask = pos_offsets < (T_in - 2 * PADDING)  # T_out == T_in - 2*PADDING for K=5, PADDING=2

    # Accumulator for output y[b, co, pos]
    acc = tl.zeros((BLOCK_CO, BLOCK_POS), dtype=tl.float32)

    # Loop over input channels and kernel positions
    for ci in range(0, Cin, BLOCK_CIN):
        ci_offsets = ci + tl.arange(0, BLOCK_CIN)
        ci_mask = ci_offsets < Cin

        # For each kernel element k in 0..K-1
        for k in range(0, K):
            t_in = pos_offsets + PADDING + k  # input time index for this output pos
            # Build 2D masks for combined [ci, t_in]
            mask2d = ci_mask[:, None] & pos_mask[None, :]  # (BLOCK_CIN, BLOCK_POS)

            # Load x[b, ci, t_in]: broadcast b and (ci, t_in)
            # We use masked load to handle boundary; but since pos_mask is for output pos, we need to ensure
            # t_in is valid for the current pos. We rely on pos_mask being true for positions that map to valid t_in.
            # However, due to t_in = pos_offsets + padding + k, and pos_offsets < T_out, t_in in [PADDING, PADDING+K-1+T_out)
            # For valid pos, t_in should be within [0, T_in-1]. To be safe, recompute mask based on T_in:
            # If pos_mask ensures pos < T_out, and PADDING+K <= T_in, then t_in < T_in. For K=5, PADDING=2, T_out=T_in-4.
            # Still, Triton requires mask per load. We mask by checking t_in < T_in.
            mask2d = mask2d & (t_in[None, :] < T_in)

            # Compute pointers for x load: x[b, ci, t_in]
            x_ptrs = x_ptr + pid_b * x_stride_b + ci_offsets[:, None] * x_stride_c + t_in[None, :] * x_stride_t
            x_vals = tl.load(x_ptrs, mask=mask2d, other=0.0)  # (BLOCK_CIN, BLOCK_POS)

            # Load weights for all co for this (ci, k): w[co, ci*K + k]
            w_ptrs = w_flat_ptr + co_offsets[:, None] * w_stride_co + (ci_offsets[None, :] * K + k) * w_stride_ci
            w_mask = co_mask[:, None] & (ci_offsets[None, :] < Cin)
            w_vals = tl.load(w_ptrs, mask=w_mask, other=0.0)  # (BLOCK_CO, BLOCK_CIN)

            # Accumulate: sum over ci
            # acc[co, pos] += sum_ci w[co, ci*K + k] * x[b, ci, t_in]
            # We need to broadcast w_vals over BLOCK_POS and x_vals over BLOCK_CO, and reduce over ci dimension.
            # Triton doesn't have reduce along arbitrary axis directly, so we do it manually:
            # For each c in BLOCK_CIN, multiply w_vals[:, c] with x_vals[c, :] and add.
            for c in range(0, BLOCK_CIN):
                # If c >= Cin, mask ensures w_vals[:, c] and x_vals[c, :] are zeros, so fine.
                acc += w_vals[:, c][:, None] * x_vals[c, :][None, :]

    # Add bias
    b_vals = tl.load(b_ptr + co_offsets, mask=co_mask, other=0.0)  # (BLOCK_CO,)
    acc += b_vals[:, None]

    # Apply ReLU
    acc = tl.maximum(acc, 0.0)

    # Store to y[b, co, pos]
    y_ptrs = y_ptr + pid_b * y_stride_b + co_offsets[:, None] * y_stride_c + pos_offsets[None, :] * y_stride_t
    y_mask = co_mask[:, None] & pos_mask[None, :]
    tl.store(y_ptrs, acc, mask=y_mask)


@triton.jit
def apply_mask_to_h_triton(
    h_ptr,            # *float32, [B, C1, T] h from conv+ReLU
    mask_ptr,         # *float32, [B, 1, T] x_mask
    out_ptr,          # *float32, [B, C1, T] masked h
    B: tl.constexpr, C1: tl.constexpr, T: tl.constexpr,
    h_stride_b: tl.constexpr, h_stride_c: tl.constexpr, h_stride_t: tl.constexpr,
    mask_stride_b: tl.constexpr, mask_stride_c: tl.constexpr, mask_stride_t: tl.constexpr,
    out_stride_b: tl.constexpr, out_stride_c: tl.constexpr, out_stride_t: tl.constexpr,
    BLOCK_C: tl.constexpr, BLOCK_T: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)

    c_offsets = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    c_mask = c_offsets < C1
    t_mask = t_offsets < T

    # Load h[b, c, t]
    h_ptrs = h_ptr + pid_b * h_stride_b + c_offsets[:, None] * h_stride_c + t_offsets[None, :] * h_stride_t
    h_vals = tl.load(h_ptrs, mask=c_mask[:, None] & t_mask[None, :], other=0.0)

    # Load mask[b, 1, t] (channel dimension is size 1)
    mask_ptrs = mask_ptr + pid_b * mask_stride_b + 0 * mask_stride_c + t_offsets[None, :] * mask_stride_t
    mask_vals = tl.load(mask_ptrs, mask=t_mask[None, :], other=1.0)  # assume mask is 1s as per provided get_inputs

    # Multiply elementwise
    out_vals = h_vals * mask_vals

    # Store
    out_ptrs = out_ptr + pid_b * out_stride_b + c_offsets[:, None] * out_stride_c + t_offsets[None, :] * out_stride_t
    tl.store(out_ptrs, out_vals, mask=c_mask[:, None] & t_mask[None, :])


@triton.jit
def add_h_to_x1_triton(
    x1_ptr,           # *float32, [B, C1, T] original x1
    h_masked_ptr,     # *float32, [B, C1, T] masked h
    out_ptr,          # *float32, [B, C1, T] updated x1 (x1 + h_masked or x1 - h_masked)
    B: tl.constexpr, C1: tl.constexpr, T: tl.constexpr,
    x1_stride_b: tl.constexpr, x1_stride_c: tl.constexpr, x1_stride_t: tl.constexpr,
    h_stride_b: tl.constexpr, h_stride_c: tl.constexpr, h_stride_t: tl.constexpr,
    out_stride_b: tl.constexpr, out_stride_c: tl.constexpr, out_stride_t: tl.constexpr,
    ADD: tl.constexpr,   # bool-like constexpr: True=add, False=subtract
    BLOCK_C: tl.constexpr, BLOCK_T: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)

    c_offsets = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    c_mask = c_offsets < C1
    t_mask = t_offsets < T

    x1_ptrs = x1_ptr + pid_b * x1_stride_b + c_offsets[:, None] * x1_stride_c + t_offsets[None, :] * x1_stride_t
    h_ptrs = h_masked_ptr + pid_b * h_stride_b + c_offsets[:, None] * h_stride_c + t_offsets[None, :] * h_stride_t

    x1_vals = tl.load(x1_ptrs, mask=c_mask[:, None] & t_mask[None, :], other=0.0)
    h_vals = tl.load(h_ptrs, mask=c_mask[:, None] & t_mask[None, :], other=0.0)

    if ADD:
        out_vals = x1_vals + h_vals
    else:
        out_vals = x1_vals - h_vals

    out_ptrs = out_ptr + pid_b * out_stride_b + c_offsets[:, None] * out_stride_c + t_offsets[None, :] * out_stride_t
    tl.store(out_ptrs, out_vals, mask=c_mask[:, None] & t_mask[None, :])


@triton.jit
def concat_and_add_v1_triton(
    x0_ptr, x1_ptr, x1_upd_ptr, out_ptr,
    B: tl.constexpr, C0: tl.constexpr, C1: tl.constexpr, T: tl.constexpr,
    x0_stride_b: tl.constexpr, x0_stride_c: tl.constexpr, x0_stride_t: tl.constexpr,
    x1_stride_b: tl.constexpr, x1_stride_c: tl.constexpr, x1_stride_t: tl.constexpr,
    x1u_stride_b: tl.constexpr, x1u_stride_c: tl.constexpr, x1u_stride_t: tl.constexpr,
    out_stride_b: tl.constexpr, out_stride_c: tl.constexpr, out_stride_t: tl.constexpr,
    BLOCK_C0: tl.constexpr, BLOCK_C1: tl.constexpr, BLOCK_T: tl.constexpr,
):
    # This kernel writes out[b, c, t] where:
    # c in [0, C0): out = x0[b, c, t]
    # c in [C0, C0+C1): out = x1_upd[b, c - C0, t]
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)

    c_offsets = pid_c * (BLOCK_C0 + BLOCK_C1) + tl.arange(0, BLOCK_C0 + BLOCK_C1)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    c_mask = c_offsets < (C0 + C1)
    t_mask = t_offsets < T

    # For first half (x0)
    c0_mask = c_offsets < C0
    # Compute per-element source c for x0
    c0_sel = c_offsets  # only valid where c0_mask
    x0_ptrs = x0_ptr + pid_b * x0_stride_b + c0_sel[:, None] * x0_stride_c + t_offsets[None, :] * x0_stride_t
    # Broadcast mask: for c0 elements where c0_mask is True, load; else treat as 0
    x0_mask = c0_mask[:, None] & t_mask[None, :]
    x0_vals = tl.load(x0_ptrs, mask=x0_mask, other=0.0)

    # For second half (x1_upd)
    c1_sel = c_offsets - C0  # valid where c_offsets >= C0
    c1_mask = c_offsets >= C0
    x1u_ptrs = x1u_ptr + pid_b * x1u_stride_b + c1_sel[:, None] * x1u_stride_c + t_offsets[None, :] * x1u_stride_t
    x1u_mask = c1_mask[:, None] & t_mask[None, :]
    x1u_vals = tl.load(x1u_ptrs, mask=x1u_mask, other=0.0)

    # Combine: out = x0 where c < C0, out = x1_upd otherwise
    out_vals = tl.where(c0_mask[:, None], x0_vals, x1u_vals)

    out_ptrs = out_ptr + pid_b * out_stride_b + c_offsets[:, None] * out_stride_c + t_offsets[None, :] * out_stride_t
    out_mask = c_mask[:, None] & t_mask[None, :]
    tl.store(out_ptrs, out_vals, mask=out_mask)

def run(
    x: torch.Tensor,
    x_mask: torch.Tensor,
    reverse: bool,
    transform_0_conv0_weight: torch.Tensor,
    transform_0_conv0_bias: torch.Tensor,
    transform_0_conv1_weight: torch.Tensor,
    transform_0_conv1_bias: torch.Tensor,
    transform_0_conv2_weight: torch.Tensor,
    transform_0_conv2_bias: torch.Tensor,
    transform_1_conv0_weight: torch.Tensor,
    transform_1_conv0_bias: torch.Tensor,
    transform_1_conv1_weight: torch.Tensor,
    transform_1_conv1_bias: torch.Tensor,
    transform_1_conv2_weight: torch.Tensor,
    transform_1_conv2_bias: torch.Tensor,
    transform_2_conv0_weight: torch.Tensor,
    transform_2_conv0_bias: torch.Tensor,
    transform_2_conv1_weight: torch.Tensor,
    transform_2_conv1_bias: torch.Tensor,
    transform_2_conv2_weight: torch.Tensor,
    transform_2_conv2_bias: torch.Tensor,
    transform_3_conv0_weight: torch.Tensor,
    transform_3_conv0_bias: torch.Tensor,
    transform_3_conv1_weight: torch.Tensor,
    transform_3_conv1_bias: torch.Tensor,
    transform_3_conv2_weight: torch.Tensor,
    transform_3_conv2_bias: torch.Tensor,
):
    """
    Triton-optimized residual coupling flow block.
    Forward: x1 = x1 + transform(x0) for each layer
    Reverse: x1 = x1 - transform(x0) for each layer (in reverse order)
    """
    # Ensure tensors are on CUDA and contiguous
    device = x.device
    if not x.is_cuda:
        raise RuntimeError("ModelNew expects CUDA tensors. Move inputs to CUDA.")
    for t in [
        transform_0_conv0_weight, transform_0_conv0_bias,
        transform_0_conv1_weight, transform_0_conv1_bias,
        transform_0_conv2_weight, transform_0_conv2_bias,
        transform_1_conv0_weight, transform_1_conv0_bias,
        transform_1_conv1_weight, transform_1_conv1_bias,
        transform_1_conv2_weight, transform_1_conv2_bias,
        transform_2_conv0_weight, transform_2_conv0_bias,
        transform_2_conv1_weight, transform_2_conv1_bias,
        transform_2_conv2_weight, transform_2_conv2_bias,
        transform_3_conv0_weight, transform_3_conv0_bias,
        transform_3_conv1_weight, transform_3_conv1_bias,
        transform_3_conv2_weight, transform_3_conv2_bias,
        x_mask,
    ]:
        if not t.is_cuda:
            t = t.to(device)

    B, C, T = x.shape
    half_channels = C // 2

    # Launch parameters tuned for typical sizes (C=384, T up to ~2447)
    BLOCK_CO = 192  # matches hidden_channels=192 exactly; for general, cdiv(Cout, 128) but here it divides
    BLOCK_CIN = 32  # input channels tile
    BLOCK_POS = 128 # time tile
    # For mask and add_h_to_x1
    BLOCK_C = 64
    BLOCK_T = 128

    transforms = [
        (transform_0_conv0_weight, transform_0_conv0_bias,
         transform_0_conv1_weight, transform_0_conv1_bias,
         transform_0_conv2_weight, transform_0_conv2_bias),
        (transform_1_conv0_weight, transform_1_conv0_bias,
         transform_1_conv1_weight, transform_1_conv1_bias,
         transform_1_conv2_weight, transform_1_conv2_bias),
        (transform_2_conv0_weight, transform_2_conv0_bias,
         transform_2_conv1_weight, transform_2_conv1_bias,
         transform_2_conv2_weight, transform_2_conv2_bias),
        (transform_3_conv0_weight, transform_3_conv0_bias,
         transform_3_conv1_weight, transform_3_conv1_bias,
         transform_3_conv2_weight, transform_3_conv2_bias),
    ]

    # Helper to run one transform: returns updated x with concatenated output
    def apply_one_transform(x0, conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b, ADD=True):
        # x0: [B, C0, T], compute h = conv1d(x0) -> ReLU -> conv1d -> ReLU -> conv1d
        B0, C0, T0 = x0.shape
        C1 = conv0_w.shape[0]  # out_channels for conv0; after conv1, still C1 (192); after conv2, half_channels (96)
        # conv1 output channels: conv1_w.shape[0] = 192; conv2 output channels: half_channels = 96
        # We need to know Cout for each conv step. Use shapes directly.
        # conv0 -> C1=192
        h0 = torch.empty((B0, C1, T0), device=device, dtype=torch.float32)
        # Launch conv1d + ReLU (three times)
        # conv0
        grid0 = (B0, _ceil_div(C1, BLOCK_CO), _ceil_div(T0 - 2 * 2, BLOCK_POS))
        conv1d_triton_relu[grid0](
            x0, conv0_w.view(C1, -1), conv0_b, h0,
            B0, x0.shape[1], T0, C1, 5, 2,
            x0.stride(0), x0.stride(1), x0.stride(2),
            conv0_w.stride(0), conv0_w.stride(1),  # weights are [Cout, Cin*K], so strides reflect that
            h0.stride(0), h0.stride(1), h0.stride(2),
            BLOCK_CO=BLOCK_CO, BLOCK_POS=BLOCK_POS, BLOCK_CIN=BLOCK_CIN,
            num_warps=4, num_stages=2
        )
        # ReLU conv0
        # conv1
        Cmid = conv1_w.shape[0]  # 192
        h1 = torch.empty((B0, Cmid, T0), device=device, dtype=torch.float32)
        grid1 = (B0, _ceil_div(Cmid, BLOCK_CO), _ceil_div(T0 - 2 * 2, BLOCK_POS))
        conv1d_triton_relu[grid1](
            h0, conv1_w.view(Cmid, -1), conv1_b, h1,
            B0, C1, T0, Cmid, 5, 2,
            h0.stride(0), h0.stride(1), h0.stride(2),
            conv1_w.stride(0), conv1_w.stride(1),
            h1.stride(0), h1.stride(1), h1.stride(2),
            BLOCK_CO=BLOCK_CO, BLOCK_POS=BLOCK_POS, BLOCK_CIN=BLOCK_CIN,
            num_warps=4, num_stages=2
        )
        # ReLU conv1
        # conv2
        Cout2 = conv2_w.shape[0]  # 96
        h2 = torch.empty((B0, Cout2, T0), device=device, dtype=torch.float32)
        grid2 = (B0, _ceil_div(Cout2, BLOCK_CO), _ceil_div(T0 - 2 * 2, BLOCK_POS))
        conv1d_triton_relu[grid2](
            h1, conv2_w.view(Cout2, -1), conv2_b, h2,
            B0, Cmid, T0, Cout2, 5, 2,
            h1.stride(0), h1.stride(1), h1.stride(2),
            conv2_w.stride(0), conv2_w.stride(1),
            h2.stride(0), h2.stride(1), h2.stride(2),
            BLOCK_CO=BLOCK_CO, BLOCK_POS=BLOCK_POS, BLOCK_CIN=BLOCK_CIN,
            num_warps=4, num_stages=2
        )
