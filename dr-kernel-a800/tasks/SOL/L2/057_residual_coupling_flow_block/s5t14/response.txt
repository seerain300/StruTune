import math
import torch
import triton
import triton.language as tl


@triton.jit
def conv1d_triton_layer(
    x_ptr,          # *f32, [B, Cin, T]
    w_ptr,          # *f32, [Cout, Cin*K]
    b_ptr,          # *f32, [Cout]
    out_ptr,        # *f32, [B, Cout, T]
    B, Cin, T, Cout,
    PAD, K,         # K is kernel size, PAD = (K-1)//2
    x_stride0, x_stride1, x_stride2,
    w_stride0, w_stride1,
    out_stride0, out_stride1, out_stride2,
    BLOCK_CO: tl.constexpr, BLOCK_POS: tl.constexpr
):
    # Grid: (B, ceil(Cout/BLOCK_CO), ceil(T/BLOCK_POS))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_pos = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    pos_offsets = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)

    co_mask = co_offsets < Cout
    pos_mask = pos_offsets < T

    # Accumulator for output [BLOCK_CO, BLOCK_POS]
    acc = tl.zeros((BLOCK_CO, BLOCK_POS), dtype=tl.float32)

    # Loop over input channels Cin and kernel taps K
    for cin in range(0, Cin):
        for k in range(0, K):
            pos_in = pos_offsets - PAD + k  # vector [BLOCK_POS]
            in_bounds = (pos_in >= 0) & (pos_in < T)

            # Load x[b, cin, pos_in] vector for current pos_offsets
            x_ptrs = x_ptr + pid_b * x_stride0 + cin * x_stride1 + pos_in * x_stride2
            x_vals = tl.load(x_ptrs, mask=pos_mask & in_bounds, other=0.0)  # [BLOCK_POS]

            # Load weights w[co, cin*K + k] for co_offsets
            w_ptrs = w_ptr + co_offsets * w_stride0 + (cin * K + k) * w_stride1
            w_vals = tl.load(w_ptrs, mask=co_mask, other=0.0)  # [BLOCK_CO]

            # Outer product and accumulate
            # x_vals[:, None] -> [BLOCK_POS, 1], w_vals[None, :] -> [1, BLOCK_CO]
            acc += x_vals[:, None] * w_vals[None, :]

    # Add bias
    b_vals = tl.load(b_ptr + co_offsets, mask=co_mask, other=0.0)  # [BLOCK_CO]
    acc += b_vals[:, None]

    # Apply ReLU
    acc = tl.maximum(acc, 0.0)

    # Store output
    out_base = pid_b * out_stride0
    out_ptrs = out_ptr + out_base + co_offsets[:, None] * out_stride1 + pos_offsets[None, :] * out_stride2
    tl.store(out_ptrs, acc, mask=co_mask[:, None] & pos_mask[None, :])


@triton.jit
def apply_mask_to_h_triton(h_ptr, mask_ptr, out_ptr,
                           B, Cout, T,
                           h_stride0, h_stride1, h_stride2,
                           mask_stride0, mask_stride1, mask_stride2,
                           out_stride0, out_stride1, out_stride2,
                           BLOCK_CO: tl.constexpr, BLOCK_POS: tl.constexpr):
    # Grid: (B, ceil(Cout/BLOCK_CO), ceil(T/BLOCK_POS))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_pos = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    pos_offsets = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)

    co_mask = co_offsets < Cout
    pos_mask = pos_offsets < T

    h_ptrs = h_ptr + pid_b * h_stride0 + co_offsets[:, None] * h_stride1 + pos_offsets[None, :] * h_stride2
    h_vals = tl.load(h_ptrs, mask=co_mask[:, None] & pos_mask[None, :], other=0.0)

    # mask has shape [B, 1, T] -> load scalar per pos
    mask_ptrs = mask_ptr + pid_b * mask_stride0 + 0 * mask_stride1 + pos_offsets * mask_stride2
    mask_vals = tl.load(mask_ptrs, mask=pos_mask, other=1.0)  # [BLOCK_POS], broadcast across co

    h_vals = h_vals * mask_vals[None, :]  # broadcast along channels

    out_base = pid_b * out_stride0
    out_ptrs = out_ptr + out_base + co_offsets[:, None] * out_stride1 + pos_offsets[None, :] * out_stride2
    tl.store(out_ptrs, h_vals, mask=co_mask[:, None] & pos_mask[None, :])


@triton.jit
def add_h_to_x1_triton(x1_ptr, h_ptr, out_ptr,
                       B, half_c, T,
                       x1_stride0, x1_stride1, x1_stride2,
                       h_stride0, h_stride1, h_stride2,
                       out_stride0, out_stride1, out_stride2,
                       ADD: tl.constexpr,  # True => add, False => subtract
                       BLOCK_CO: tl.constexpr, BLOCK_POS: tl.constexpr):
    # Grid: (B, ceil(half_c/BLOCK_CO), ceil(T/BLOCK_POS))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_pos = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    pos_offsets = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)

    co_mask = co_offsets < half_c
    pos_mask = pos_offsets < T

    x1_ptrs = x1_ptr + pid_b * x1_stride0 + co_offsets[:, None] * x1_stride1 + pos_offsets[None, :] * x1_stride2
    x1_vals = tl.load(x1_ptrs, mask=co_mask[:, None] & pos_mask[None, :], other=0.0)

    h_ptrs = h_ptr + pid_b * h_stride0 + co_offsets[:, None] * h_stride1 + pos_offsets[None, :] * h_stride2
    h_vals = tl.load(h_ptrs, mask=co_mask[:, None] & pos_mask[None, :], other=0.0)

    if ADD:
        res = x1_vals + h_vals
    else:
        res = x1_vals - h_vals

    out_base = pid_b * out_stride0
    out_ptrs = out_ptr + out_base + co_offsets[:, None] * out_stride1 + pos_offsets[None, :] * out_stride2
    tl.store(out_ptrs, res, mask=co_mask[:, None] & pos_mask[None, :])

                          out_stride0, out_stride1, out_stride2,
                          BLOCK_CO: tl.constexpr, BLOCK_POS: tl.constexpr):
    # Grid: (B, ceil(half_c/BLOCK_CO), ceil(T/BLOCK_POS))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_pos = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    pos_offsets = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)

    co_mask = co_offsets < half_c
    pos_mask = pos_offsets < T

    # Copy x0 into out for first half channels [0, half_c)
    base0 = pid_b * x0_stride0
    x0_ptrs = x0_ptr + base0 + co_offsets[:, None] * x0_stride1 + pos_offsets[None, :] * x0_stride2
    x0_vals = tl.load(x0_ptrs, mask=co_mask[:, None] & pos_mask[None, :], other=0.0)

    out_base = pid_b * out_stride0
    out_ptrs = out_ptr + out_base + co_offsets[:, None] * out_stride1 + pos_offsets[None, :] * out_stride2
    tl.store(out_ptrs, x0_vals, mask=co_mask[:, None] & pos_mask[None, :])


@triton.jit
def concat_add_v2_triton(x0_ptr, x1_ptr, out_ptr,
                          B, half_c, T,
                          x0_stride0, x0_stride1, x0_stride2,
                          x1_stride0, x1_stride1, x1_stride2,
