@triton.jit
def conv1d_triton_fused_relu(
    x_ptr,         # *const float, input [B, Cin, T]
    w_ptr,         # *const float, weights [Cout, Cin, K]
    b_ptr,         # *const float, bias [Cout]
    y_ptr,         # *float, output [B, Cout, T_out]
    B: tl.int32, Cin: tl.int32, Cout: tl.int32, T: tl.int32, K: tl.int32,
    stride_b: tl.int32, stride_cin: tl.int32, stride_t: tl.int32,
    w_stride_co: tl.int32, w_stride_ci: tl.int32, w_stride_k: tl.int32,
    pad: tl.int32,
    T_out: tl.int32,
    BLOCK_POS: tl.constexpr,  # tile over output positions
    BLOCK_CO: tl.constexpr,   # tile over output channels
):
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_pos = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)  # [BLOCK_CO]
    pos_offsets = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)  # [BLOCK_POS]

    co_mask = co_offsets < Cout
    pos_mask = pos_offsets < T_out

    acc = tl.zeros((BLOCK_CO, BLOCK_POS), dtype=tl.float32)

    # For each conv layer: Cin = Cin of the input, W shape [Cout, Cin, K]
    # We iterate ci in [0, Cin) and k in [0, K), accumulate into acc
    for ci in range(0, Cin):
        for kk in range(0, K):
            input_pos = pos_offsets - pad + kk  # [BLOCK_POS]
            in_bounds = (input_pos >= 0) & (input_pos < T) & pos_mask
            x_vec = tl.load(
                x_ptr + pid_b * stride_b + ci * stride_cin + input_pos * stride_t,
                mask=in_bounds,
                other=0.0
            ).to(tl.float32)  # [BLOCK_POS]

            # Load weight vector for (co, ci, kk) and broadcast over pos
            w_vec = tl.load(
                w_ptr + co_offsets * w_stride_co + ci * w_stride_ci + kk * w_stride_k,
                mask=co_mask,
                other=0.0
            ).to(tl.float32)  # [BLOCK_CO]

            # Outer product accumulation
            acc += w_vec[:, None] * x_vec[None, :]

    # Add bias
    b_vec = tl.load(b_ptr + co_offsets, mask=co_mask, other=0.0).to(tl.float32)  # [BLOCK_CO]
    acc += b_vec[:, None]

    # Apply ReLU
    acc = tl.maximum(acc, 0.0)

    # Store output
    out_ptrs = y_ptr + pid_b * (Cout * T_out) + co_offsets[:, None] * T_out + pos_offsets[None, :]
    out_mask = co_mask[:, None] & pos_mask[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


@triton.jit
def concat_and_add_triton(
    x0_ptr,      # [B, C0, T]
    x1_ptr,      # [B, C1, T]
    h_ptr,       # [B, C1, T] (to be added/subtracted to x1)
    out_ptr,     # [B, C0 + C1, T]
    B: tl.int32, C0: tl.int32, C1: tl.int32, T: tl.int32,
    stride_x0_b: tl.int32, stride_x0_c: tl.int32, stride_x0_t: tl.int32,
    stride_x1_b: tl.int32, stride_x1_c: tl.int32, stride_x1_t: tl.int32,
    stride_h_b: tl.int32, stride_h_c: tl.int32, stride_h_t: tl.int32,
    stride_out_b: tl.int32, stride_out_c: tl.int32, stride_out_t: tl.int32,
    ADD: tl.constexpr,
    BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)

    c_offsets = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
    t_offsets = pid_t * BLOCK_C + tl.arange(0, BLOCK_C)

    c_mask = c_offsets < (C0 + C1)
    t_mask = t_offsets < T

    # Write x0 to out[:, :C0, :]
    x0_base = x0_ptr + pid_b * stride_x0_b
    out_base = out_ptr + pid_b * stride_out_b
    for i in range(0, BLOCK_C):
        c = c_offsets[i]
        if c < C0:
            val = tl.load(
                x0_base + c * stride_x0_c + t_offsets[i] * stride_x0_t,
                mask=t_mask[i],
                other=0.0
            ).to(tl.float32)
            tl.store(
                out_base + c * stride_out_c + t_offsets[i] * stride_out_t,
                val,
                mask=t_mask[i]
            )

    # Write x1 to out[:, C0:, :]
    x1_base = x1_ptr + pid_b * stride_x1_b
    for i in range(0, BLOCK_C):
        c = c_offsets[i]
        if (c >= C0) & (c < (C0 + C1)):
            c_rel = c - C0
            val = tl.load(
                x1_base + c_rel * stride_x1_c + t_offsets[i] * stride_x1_t,
                mask=t_mask[i],
                other=0.0
            ).to(tl.float32)
            if ADD:
                h_val = tl.load(
                    h_ptr + pid_b * stride_h_b + c_rel * stride_h_c + t_offsets[i] * stride_h_t,
                    mask=t_mask[i],
                    other=0.0
                ).to(tl.float32)
                val = val + h_val
            else:
                h_val = tl.load(
                    h_ptr + pid_b * stride_h_b + c_rel * stride_h_c + t_offsets[i] * stride_h_t,
                    mask=t_mask[i],
                    other=0.0
                ).to(tl.float32)
                val = val - h_val
            tl.store(
                out_base + c * stride_out_c + t_offsets[i] * stride_out_t,
                val,
                mask=t_mask[i]
            )


@triton.jit
def add_mask_to_h_triton(

    pid_t = tl.program_id(2)

    c_offsets = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
    t_offsets = pid_t * BLOCK_C + tl.arange(0, BLOCK_C)

    c_mask = c_offsets < C
    t_mask = t_offsets < T

    for i in range(0, BLOCK_C):
        c = c_offsets[i]
        if c < C:
            h_vals = tl.load(
                h_ptr + pid_b * stride_h_b + c * stride_h_c + t_offsets * stride_h_t,
                mask=t_mask,
                other=0.0
            ).to(tl.float32)

                mask=t_mask,
                other=0.0
            ).to(tl.float32)
            h_vals = h_vals * m_vals
            tl.store(
                out_ptr + pid_b * stride_out_b + c * stride_out_c + t_offsets * stride_out_t,
                h_vals,
                mask=t_mask
            )

    conv0_w, conv0_b,  # [C1, C0, K], [C1]

    device: torch.device

    B = x0.shape[0]
    C0 = x0.shape[1]
    T = x0.shape[2]
    K = 5
    C1 = conv0_w.shape[0]
    C2 = conv1_w.shape[0]
    C3 = conv2_w.shape[0]
    pad = (K - 1) // 2
    T_out = T  # stride=1, dilation=1, no padding effect on length when K odd and symmetric

        conv0_w.stride(0), conv0_w.stride(1), conv0_w.stride(2),
        pad,
        T_out,

    conv1d_triton_fused_relu[(B, triton.cdiv(C2, 64), triton.cdiv(T_out, 64))](

        conv1_b,
        h1,
        B, C1, C2, T_out, K,
        h0.stride(0), h0.stride(1), h0.stride(2),
        conv1_w.stride(0), conv1_w.stride(1), conv1_w.stride(2),

        conv2_w.stride(0), conv2_w.stride(1), conv2_w.stride(2),
        pad,

        BLOCK_POS=64, BLOCK_CO=64
    )

    return h2

                def copy_first_half_kernel(out_ptr, x0_ptr, B: tl.int32, half_channels: tl.int32, T: tl.int32,
                                            stride_out_b: tl.int32, stride_out_c: tl.int32, stride_out_t: tl.int32,
                                            stride_x0_b: tl.int32, stride_x0_c: tl.int32, stride_x0_t: tl.int32,
                                            BLOCK_C: tl.constexpr, BLOCK_T: tl.constexpr):
                    pid_b = tl.program_id(0)
                    pid_c = tl.program_id(1)
                    pid_t = tl.program_id(2)

                    c_offsets = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
                    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

                    c_mask = c_offsets < half_channels
                    t_mask = t_offsets < T

                    # Compute source addresses: x0[b, c, t]
                    src_addrs = x0_ptr + pid_b * stride_x0_b + c_offsets[:, None] * stride_x0_c + t_offsets[None, :] * stride_x0_t
                    mask = c_mask[:, None] & t_mask[None, :]
                    vals = tl.load(src_addrs, mask=mask, other=0.0)

                    # Compute destination addresses: out[b, c, t] for c < half_channels
                    dest_addrs = out_ptr + pid_b * stride_out_b + c_offsets[:, None] * stride_out_c + t_offsets[None, :] * stride_out_t
                    tl.store(dest_addrs, vals, mask=mask)

                # Launch copy first half
                copy_first_half_kernel[(B, triton.cdiv(half_channels, 64), triton.cdiv(T, 64))](
                    out, x0,
                    B, half_channels, T,
                    out.stride(0), out.stride(1), out.stride(2),
                    x0.stride(0), x0.stride(1), x0.stride(2),
                    BLOCK_C=64, BLOCK_T=64
                )

                # 2) Write second half: out[:, half_channels:, :] = x1_out
                # Define copy kernel for second half: c in [half_channels, 2*half_channels)
                @triton.jit
                def copy_second_half_kernel(out_ptr, x1_ptr, B: tl.int32, half_channels: tl.int32, T: tl.int32,
                                             stride_out_b: tl.int32, stride_out_c: tl.int32, stride_out_t: tl.int32,
                                             stride_x1_b: tl.int32, stride_x1_c: tl.int32, stride_x1_t: tl.int32,
                                             BLOCK_C: tl.constexpr, BLOCK_T: tl.constexpr):
                    pid_b = tl.program_id(0)
                    pid_c = tl.program_id(1)
                    pid_t = tl.program_id(2)

                    c_offsets = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
                    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

                    c_mask = c_offsets < half_channels  # second half channels are [half_channels, 192)
                    t_mask = t_offsets < T

                    # src c index in second half: c_src = c_offsets + half_channels
                    src_addrs = x1_ptr + pid_b * stride_x1_b + (c_offsets[:, None] + half_channels) * stride_x1_c + t_offsets[None, :] * stride_x1_t
                    mask = c_mask[:, None] & t_mask[None, :]
                    vals = tl.load(src_addrs, mask=mask, other=0.0)

                    dest_addrs = out_ptr + pid_b * stride_out_b + (half_channels + c_offsets[:, None]) * stride_out_c + t_offsets[None, :] * stride_out_t
                    tl.store(dest_addrs, vals, mask=mask)

                # Launch copy second half
                copy_second_half_kernel[(B, triton.cdiv(half_channels, 64), triton.cdiv(T, 64))](
                    out, x1_out,
                    B, half_channels, T,
                    out.stride(0), out.stride(1), out.stride(2),
                    x1_out.stride(0), x1_out.stride(1), x1_out.stride(2),
                    BLOCK_C=64, BLOCK_T=64
                )

                # Now out has first half = x0 and second half = x1_out. Assign x_out = out
                x_out = out

        else:
            # Reverse: apply transforms in reverse order and subtract h
            # Similar structure, just subtract h when updating x1
            # Initialize x_out as x
            x_out = x
            for t in range(3, -1, -1):
                conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b = (
                    transform_3 if t == 3 else
                    (transform_2 if t == 2 else
                     (transform_1 if t == 1 else transform_0))
                )

                x0 = x_out[:, :half_channels, :]
                x1 = x_out[:, half_channels:, :]

                h2 = apply_transform_triton_single(x0, conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b, device)

                # mask multiply
                masked_h2 = torch.empty_like(h2)
                add_mask_to_h_triton[(B, triton.cdiv(half_channels, 64), triton.cdiv(T, 64))](
                    h2,
                    x_mask,
                    masked_h2,
                    B, half_channels, T,
                    h2.stride(0), h2.stride(1), h2.stride(2),
                    x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                    masked_h2.stride(0), masked_h2.stride(1), masked_h2.stride(2),
                    BLOCK_C=64
                )

                # Update x1: x1 = x1 - masked_h2
                x1_out = torch.empty_like(x1)
                add_h_to_x1_triton[(B, triton.cdiv(half_channels, 64), triton.cdiv(T, 64))](
                    x1,
                    masked_h2,
                    x1_out,
                    B, half_channels, T,
                    x1.stride(0), x1.stride(1), x1.stride(2),
                    masked_h2.stride(0), masked_h2.stride(1), masked_h2.stride(2),
                    x1_out.stride(0), x1_out.stride(1), x1_out.stride(2),
                    ADD=False,
                    BLOCK_C=64
                )

                # Concatenate back: out = [x0, x1_out]
                out = torch.empty((B, C, T), dtype=torch.float32, device=device)
                # Copy first half
                copy_first_half_kernel[(B, triton.cdiv(half_channels, 64), triton.cdiv(T, 64))](
                    out, x0,
                    B, half_channels, T,
                    out.stride(0), out.stride(1), out.stride(2),
                    x0.stride(0), x0.stride(1), x0.stride(2),
                    BLOCK_C=64, BLOCK_T=64
                )
                # Copy second half
                copy_second_half_kernel[(B, triton.cdiv(half_channels, 64), triton.cdiv(T, 64))](
                    out, x1_out,
                    B, half_channels, T,
                    out.stride(0), out.stride(1), out.stride(2),
                    x1_out.stride(0), x1_out.stride(1), x1_out.stride(2),
                    BLOCK_C=64, BLOCK_T=64
                )
                x_out = out

        return x_out
