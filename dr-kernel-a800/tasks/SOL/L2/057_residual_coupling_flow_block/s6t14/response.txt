@triton.jit
def conv1d_k5_p2(x_ptr, w_ptr, b_ptr, y_ptr,
                 B, C_in, C_out, T_in, T_out,
                 x_stride_b, x_stride_c, x_stride_t,
                 w_stride_co, w_stride_ci, w_stride_k,
                 y_stride_b, y_stride_c, y_stride_t,
                 BLOCK_T: tl.constexpr):
    # program ids: pid0 over (B * C_out), pid1 over time tiles
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    b = pid0 // C_out
    co = pid0 % C_out

    # time offsets for this tile
    t_offsets = pid1 * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    # accumulator
    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # loop over input channels and kernel taps
    for ci in range(0, C_in):
        for k in range(0, 5):
            src_t = t_offsets - 2 + k  # padding=2, valid conv
            src_mask = (src_t >= 0) & (src_t < T_in) & mask_t
            x_ptrs = x_ptr + b * x_stride_b + ci * x_stride_c + src_t * x_stride_t
            # load with mask, use 0.0 for out-of-range
            x_vals = tl.load(x_ptrs, mask=src_mask, other=0.0)
            w_vals = tl.load(w_ptr + co * w_stride_co + ci * w_stride_ci + k * w_stride_k)
            acc += x_vals * w_vals

    # add bias
    bias = tl.load(b_ptr + co)
    acc = acc + bias

    # store results
    y_ptrs = y_ptr + b * y_stride_b + co * y_stride_c + t_offsets * y_stride_t
    tl.store(y_ptrs, acc, mask=mask_t)

    h_ptrs = h_ptr + b * h_stride_b + c * h_stride_c + t_offsets * h_stride_t
