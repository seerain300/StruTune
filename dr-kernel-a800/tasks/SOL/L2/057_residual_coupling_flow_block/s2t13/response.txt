@triton.jit
def concat_halves_backward(
    y2c_ptr, y0_ptr, y1_ptr,
    N, C_half, L,
    stride_y2c_n, stride_y2c_c, stride_y2c_l,
    stride_y0_n, stride_y0_c, stride_y0_l,
    stride_y1_n, stride_y1_c, stride_y1_l,
    BLOCK_L: tl.constexpr,
):
    # y2c has shape [N, 2*C_half, L]; split along channel dimension
    n = tl.program_id(0)
    ch = tl.program_id(1)  # channel index in [0, C_half)
    tile = tl.program_id(2)
    l_offsets = tile * BLOCK_L + tl.arange(0, BLOCK_L)
    mask_out = l_offsets < L

    y0_ptrs = y2c_ptr + n * stride_y2c_n + ch * stride_y2c_c + l_offsets * stride_y2c_l
    y1_ptrs = y2c_ptr + n * stride_y2c_n + (ch + C_half) * stride_y2c_c + l_offsets * stride_y2c_l

    out0_ptrs = y0_ptr + n * stride_y0_n + ch * stride_y0_c + l_offsets * stride_y0_l

    y0_vals = tl.load(y0_ptrs, mask=mask_out, other=0.0)
    y1_vals = tl.load(y1_ptrs, mask=mask_out, other=0.0)
    tl.store(out0_ptrs, y0_vals, mask=mask_out)
    tl.store(out1_ptrs, y1_vals, mask=mask_out)

@triton.jit
def mul_mask_kernel(

def run_triton_only(
    x: torch.Tensor,
    x_mask: torch.Tensor,
    reverse: bool,

    conv0_w0, conv0_b0,
    conv0_w1, conv0_b1,

    # weights for transform 1

    # weights for transform 3
    conv3_w0, conv3_b0,
    conv3_w1, conv3_b1,
    conv3_w2, conv3_b2,
    device: torch.device,
):
    """
    Forward or reverse transform using PyTorch conv1d + Triton split/concat/mul-mask.
    Note: Only Triton kernels are launched in forward; conv1d and ReLU are done via PyTorch to ensure correctness.
    """
    N, C, L = x.shape
    half = C // 2

    # Helper to process a single transform
    def single_transform(x_curr, conv0, conv1, conv2):
        # Split into halves
        x0 = torch.empty((N, half, L), device=device, dtype=x_curr.dtype)
        x1 = torch.empty((N, half, L), device=device, dtype=x_curr.dtype)

        grid_split = (N, half, triton.cdiv(L, 128))
        concat_halves_backward[grid_split](
            x_curr, x0, x1,
            N, half, L,
            x_curr.stride(0), x_curr.stride(1), x_curr.stride(2),
            x0.stride(0), x0.stride(1), x0.stride(2),
            x1.stride(0), x1.stride(1), x1.stride(2),
            BLOCK_L=128, num_warps=4
        )

        # Conv0: h0 = F.conv1d(x0, conv0_w, conv0_b, padding=2) -> ReLU
        h0 = F.conv1d(x0, conv0[0], conv0[1], padding=2)
        # ReLU
        h0 = F.relu(h0)

        # Conv1: h1 = F.conv1d(h0, conv1_w, conv1_b, padding=2) -> ReLU
        h1 = F.conv1d(h0, conv1[0], conv1[1], padding=2)
        h1 = F.relu(h1)

        # Conv2: h2 = F.conv1d(h1, conv2_w, conv2_b, padding=2) (no ReLU)
        h2 = F.conv1d(h1, conv2[0], conv2[1], padding=2)

        # Update x1: forward: +h2, reverse: -h2
        if reverse:
            x1 = x1 - h2
        else:
            x1 = x1 + h2

        # Concatenate [x0, x1]
        x_out = torch.empty((N, 192, L), device=device, dtype=x_curr.dtype)
        grid_concat = (N, 96, triton.cdiv(L, 128))
        concat_halves_forward[grid_concat](
            x0, x1, x_out,
            N, 96, L,
            x0.stride(0), x0.stride(1), x0.stride(2),
            x1.stride(0), x1.stride(1), x1.stride(2),
            x_out.stride(0), x_out.stride(1), x_out.stride(2),
            BLOCK_L=128, num_warps=4
        )

        # Apply mask
        grid_mask = (N, 192, triton.cdiv(L, 128))
        mul_mask_kernel[grid_mask](
            x_out, x_mask,
            N, 192, L,
            x_out.stride(0), x_out.stride(1), x_out.stride(2),
            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
            BLOCK_L=128, num_warps=4
        )
        return x_out

    # Apply transforms sequentially for forward; reverse order for reverse
    if not reverse:
        out = single_transform(x, (conv0_w0, conv0_b0), (conv0_w1, conv0_b1), (conv0_w2, conv0_b2))
        out = single_transform(out, (conv1_w0, conv1_b0), (conv1_w1, conv1_b1), (conv1_w2, conv1_b2))
        out = single_transform(out, (conv2_w0, conv2_b0), (conv2_w1, conv2_b1), (conv2_w2, conv2_b2))
        out = single_transform(out, (conv3_w0, conv3_b0), (conv3_w1, conv3_b1), (conv3_w2, conv3_b2))
    else:
        out = single_transform(x, (conv3_w0, conv3_b0), (conv3_w1, conv3_b1), (conv3_w2, conv3_b2))
        out = single_transform(out, (conv2_w0, conv2_b0), (conv2_w1, conv2_b1), (conv2_w2, conv2_b2))
        out = single_transform(out, (conv1_w0, conv1_b0), (conv1_w1, conv1_b1), (conv1_w2, conv1_b2))
        out = single_transform(out, (conv0_w0, conv0_b0), (conv0_w1, conv0_b1), (conv0_w2, conv0_b2))
    return out

        def single_transform_triton(x_curr):

            x1 = torch.empty((N, half, L), device=x_curr.device, dtype=x_curr.dtype)

                x0.stride(0), x0.stride(1), x0.stride(2),
                x1.stride(0), x1.stride(1), x1.stride(2),
                BLOCK_L=128, num_warps=4
            )

            C_out = t0_w[0][0].shape[0]  # hidden_channels=192

                h0.stride(0), h0.stride(1), h0.stride(2),

            C_out_1 = t0_w[1][0].shape[0]  # hidden_channels=192

            # Apply ReLU to h1
            relu_triton_kernel[grid_c1](
                h1, h1,

            if reverse:

            else:
