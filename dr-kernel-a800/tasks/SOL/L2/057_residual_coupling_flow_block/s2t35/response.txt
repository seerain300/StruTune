  - Split into x0 = x[:, :96, :], x1 = x[:, 96:, :].

    - Conv1: h0 -> h1, then ReLU
    - Conv2: h1 -> h2 (no ReLU)

- Our previous Triton Conv1d implementation likely mismatched PyTorch’s output for several reasons:

  - Accumulation precision: Using different accumulation strategy can slightly change results. However, PyTorch’s default is float32; we’ll also accumulate in float32.

- Fixes:

After correctness is achieved, we can revisit blocked vectorization for speed.

  - Load weights as scalar per (co, ci, k) and accumulate FMA in float32. Store in the original dtype of output.
  - ReLU kernel uses y = max(y, 0).

import math
import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def conv1d_forward_kernel_serial(
    x_ptr, w_ptr, b_ptr, y_ptr,
    N, C_in, C_out, L_in, L_out, K,
    stride_x_n, stride_x_c, stride_x_l,
    stride_w_co, stride_w_ci, stride_w_k,
    stride_y_n, stride_y_c, stride_y_l,
):
    # Each program handles one (n, co) pair and iterates over all output positions
    n = tl.program_id(0)
    co = tl.program_id(1)

    # Bias for this output channel
    b_val = tl.load(b_ptr + co)

    # Loop over output time positions
    for lo in range(0, L_out):
        acc = tl.full((), b_val, tl.float32)

        P = K // 2  # padding

        # Accumulate over input channels and kernel taps
        for ci in range(0, C_in):
            for k in range(0, K):
                li = lo + P - k  # input index with padding
                in_bounds = (li >= 0) & (li < L_in)
                # x[n, ci, li]
                x_ptrs = x_ptr + n * stride_x_n + ci * stride_x_c + li * stride_x_l
                x_val = tl.load(x_ptrs, mask=in_bounds, other=0.0)
                # weight[co, ci, k]
                w_ptrs = w_ptr + co * stride_w_co + ci * stride_w_ci + k * stride_w_k
                w_val = tl.load(w_ptrs)
                acc += x_val * w_val

        # Store y[n, co, lo]
        y_ptrs = y_ptr + n * stride_y_n + co * stride_y_c + lo * stride_y_l
        # Cast acc to output dtype by loading a dummy to infer? Safer: store as float32, then cast on host if needed.
        tl.store(y_ptrs, acc)


@triton.jit
def relu_kernel(
    x_ptr, y_ptr,
    N, C, L,
    stride_x_n, stride_x_c, stride_x_l,
    stride_y_n, stride_y_c, stride_y_l,
):
    # Elementwise ReLU: y = max(x, 0)
    n = tl.program_id(0)
    c = tl.program_id(1)
    # iterate over time
    for l in range(0, L):
        x_ptrs = x_ptr + n * stride_x_n + c * stride_x_c + l * stride_x_l
        y_ptrs = y_ptr + n * stride_y_n + c * stride_y_c + l * stride_y_l
        x_val = tl.load(x_ptrs)
        y_val = tl.maximum(x_val, 0.0)
        tl.store(y_ptrs, y_val)

def split_halves_forward(

        in1_ptrs = x_full_ptr + n * stride_x_full_n + (ch + C_half) * stride_x_full_c + l * stride_x_full_l

        val0 = tl.load(in0_ptrs)
        val1 = tl.load(in1_ptrs)
        tl.store(out0_ptrs, val0)
        tl.store(out1_ptrs, val1)

                mask_val = tl.load(mask_ptrs)
                y_val = y_val * mask_val
                tl.store(y_ptrs, y_val)

import math
import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def conv1d_forward_kernel(
    x_ptr, w_ptr, b_ptr, y_ptr,
    N, C_in, C_out, L_in, L_out, K,
    stride_x_n, stride_x_c, stride_x_l,
    stride_w_co, stride_w_ci, stride_w_k,
    stride_y_n, stride_y_c, stride_y_l,
    BLOCK_L: tl.constexpr,
):
    # Grid: (N, C_out, tiles along L_out)
    n = tl.program_id(0)
    co = tl.program_id(1)
    tile = tl.program_id(2)

    # Output time offsets for this tile
    l_out_offsets = tile * BLOCK_L + tl.arange(0, BLOCK_L)
    mask_out = l_out_offsets < L_out

    # Bias for this output channel
    b_val = tl.load(b_ptr + co)

    # Accumulator for this tile
    acc = tl.full((BLOCK_L,), b_val, tl.float32)

    P = K // 2  # padding

    # Loop over input channels and kernel taps
    for ci in range(0, C_in):
        for k in range(0, K):
            li = l_out_offsets + P - k  # input indices with padding
            mask_in = (li >= 0) & (li < L_in)

            # Load x[n, ci, li] (masked)
            x_ptrs = x_ptr + n * stride_x_n + ci * stride_x_c + li * stride_x_l
            x_vals = tl.load(x_ptrs, mask=mask_in, other=0.0)

            # Load weight w[co, ci, k] (scalar)
            w_ptrs = w_ptr + co * stride_w_co + ci * stride_w_ci + k * stride_w_k
            w_val = tl.load(w_ptrs)

            # FMA accumulate in fp32
            acc += x_vals * w_val

    # Store results to y[n, co, l_out_offsets]
    y_ptrs = y_ptr + n * stride_y_n + co * stride_y_c + l_out_offsets * stride_y_l
    tl.store(y_ptrs, acc, mask=mask_out)


@triton.jit
def relu_kernel(
    x_ptr, y_ptr,
    N, C, L,
    stride_x_n, stride_x_c, stride_x_l,
    stride_y_n, stride_y_c, stride_y_l,
):
    # Elementwise ReLU: y = max(x, 0)
    n = tl.program_id(0)
    c = tl.program_id(1)
    for l in range(0, L):
        x_ptrs = x_ptr + n * stride_x_n + c * stride_x_c + l * stride_x_l
        y_ptrs = y_ptr + n * stride_y_n + c * stride_y_c + l * stride_y_l
        x_val = tl.load(x_ptrs)
        y_val = tl.maximum(x_val, 0.0)
        tl.store(y_ptrs, y_val)


@triton.jit
def split_halves_forward(
    x_full_ptr, x0_ptr, x1_ptr,
    N, C_half, L,
    stride_x_full_n, stride_x_full_c, stride_x_full_l,
    stride_x0_n, stride_x0_c, stride_x0_l,
    stride_x1_n, stride_x1_c, stride_x1_l,
):
    # Read x_full[:, :C_half, :] into x0 and x_full[:, C_half:, :] into x1
    n = tl.program_id(0)
    ch = tl.program_id(1)
    for l in range(0, L):
        in0_ptrs = x_full_ptr + n * stride_x_full_n + ch * stride_x_full_c + l * stride_x_full_l
        in1_ptrs = x_full_ptr + n * stride_x_full_n + (ch + C_half) * stride_x_full_c + l * stride_x_full_l
        out0_ptrs = x0_ptr + n * stride_x0_n + ch * stride_x0_c + l * stride_x0_l
        out1_ptrs = x1_ptr + n * stride_x1_n + ch * stride_x1_c + l * stride_x1_l
        val0 = tl.load(in0_ptrs)
        val1 = tl.load(in1_ptrs)
        tl.store(out0_ptrs, val0)
        tl.store(out1_ptrs, val1)


@triton.jit
def concat_halves_forward(
    x0_ptr, x1_ptr, x_full_ptr,
    N, C_half, L,
    stride_x0_n, stride_x0_c, stride_x0_l,
    stride_x1_n, stride_x1_c, stride_x1_l,
    stride_x_full_n, stride_x_full_c, stride_x_full_l,
):
    # Write x0 into first half of x_full, x1 into second half
    n = tl.program_id(0)
    ch = tl.program_id(1)
    for l in range(0, L):
        in0_ptrs = x0_ptr + n * stride_x0_n + ch * stride_x0_c + l * stride_x0_l
        in1_ptrs = x1_ptr + n * stride_x1_n + ch * stride_x1_c + l * stride_x1_l
        out0_ptrs = x_full_ptr + n * stride_x_full_n + ch * stride_x_full_c + l * stride_x_full_l
        out1_ptrs = x_full_ptr + n * stride_x_full_n + (ch + C_half) * stride_x_full_c + l * stride_x_full_l
        val0 = tl.load(in0_ptrs)
        val1 = tl.load(in1_ptrs)
        tl.store(out0_ptrs, val0)
        tl.store(out1_ptrs, val1)


@triton.jit
def mul_mask_kernel(
    y_ptr, mask_ptr, N, C, L,
    stride_y_n, stride_y_c, stride_y_l,
    stride_mask_n, stride_mask_c, stride_mask_l,
):
    # y *= mask, mask has shape [N, 1, L]
    for n in range(0, N):
        for c in range(0, C):
            for l in range(0, L):
                y_ptrs = y_ptr + n * stride_y_n + c * stride_y_c + l * stride_y_l
                mask_ptrs = mask_ptr + n * stride_mask_n + 0 * stride_mask_c + l * stride_mask_l
                y_val = tl.load(y_ptrs)
                mask_val = tl.load(mask_ptrs)
                y_val = y_val * mask_val
                tl.store(y_ptrs, y_val)
