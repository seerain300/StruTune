import math
import torch
import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def conv1d_triton(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N, IC, OC, T, K, pad,
    x_sN, x_sC, x_sT,
    w_sO, w_sI, w_sK,
    out_sN, out_sC, out_sT,
    BLOCK_T: tl.constexpr,
):
    """
    Triton implementation of Conv1d (cross-correlation) for stride=1, padding=pad, dilation=1.
    x: [N, IC, T] (float32)
    w: [OC, IC, K] (float32)
    b: [OC] (float32)
    out: [N, OC, T] (float32)
    Grid: (N, OC, ceil_div(T, BLOCK_T))
    Accumulate in float32.
    """
    pid_n = tl.program_id(0)
    pid_oc = tl.program_id(1)
    pid_block = tl.program_id(2)

    t_out_start = pid_block * BLOCK_T
    t_out_idx = t_out_start + tl.arange(0, BLOCK_T)
    valid_t = t_out_idx < T

    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # Loop over input channels and kernel taps
    for ic in range(0, IC):
        for k in range(0, K):
            t_in = t_out_idx + k - pad  # valid when 0 <= t_in < T
            valid_t_in = valid_t & (t_in >= 0) & (t_in < T)
            x_offsets = pid_n * x_sN + ic * x_sC + t_in * x_sT
            x_vals = tl.load(x_ptr + x_offsets, mask=valid_t_in, other=0.0)
            w_val = tl.load(w_ptr + pid_oc * w_sO + ic * w_sI + k * w_sK)
            acc += x_vals * w_val

    # Add bias
    b_val = tl.load(b_ptr + pid_oc)
    acc += b_val

    # Store to out[n, oc, t_out_idx]
    out_offsets = pid_n * out_sN + pid_oc * out_sC + t_out_idx * out_sT
    tl.store(out_ptr + out_offsets, acc, mask=valid_t)


@triton.jit
def mask_apply_triton(
    y_ptr, mask_ptr, out_ptr,
    N, OC, T,
    mask_sN, mask_sC, mask_sT,
    y_sN, y_sC, y_sT,
    out_sN, out_sC, out_sT,
    BLOCK: tl.constexpr,
):
    """
    Multiply y [N, OC, T] by mask [N, 1, T], broadcasting mask across channels, and store to out.
    Grid: (N, OC, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_oc = tl.program_id(1)
    pid_block = tl.program_id(2)

    t_start = pid_block * BLOCK
    t_idx = t_start + tl.arange(0, BLOCK)
    valid_t = t_idx < T

    # Load mask for this batch and time slice; mask is [N,1,T], so we use channel=0
    mask_offsets = pid_n * mask_sN + 0 * mask_sC + t_idx * mask_sT
    mask_vals = tl.load(mask_ptr + mask_offsets, mask=valid_t, other=0.0).to(tl.float32)

    y_offsets = pid_n * y_sN + pid_oc * y_sC + t_idx * y_sT
    y_vals = tl.load(y_ptr + y_offsets, mask=valid_t, other=0.0).to(tl.float32)

    out_vals = y_vals * mask_vals

    out_offsets = pid_n * out_sN + pid_oc * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid_t)


@triton.jit
def relu_triton(
    inp_ptr, out_ptr,
    N, OC, T,
    inp_sN, inp_sC, inp_sT,
    out_sN, out_sC, out_sT,
    BLOCK: tl.constexpr,
):
    """
    Elementwise ReLU on inp [N, OC, T], store to out.
    Grid: (N, OC, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_oc = tl.program_id(1)
    pid_block = tl.program_id(2)

    t_start = pid_block * BLOCK
    t_idx = t_start + tl.arange(0, BLOCK)
    valid_t = t_idx < T

    inp_offsets = pid_n * inp_sN + pid_oc * inp_sC + t_idx * inp_sT
    inp_vals = tl.load(inp_ptr + inp_offsets, mask=valid_t, other=0.0).to(tl.float32)
    out_vals = tl.maximum(inp_vals, 0.0)

    out_offsets = pid_n * out_sN + pid_oc * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid_t)


@triton.jit
def add_h_triton(
    x1_ptr, h_ptr, out_ptr,
    N, C, T,
    x1_sN, x1_sC, x1_sT,
    h_sN, h_sC, h_sT,
    out_sN, out_sC, out_sT,
    reverse: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Elementwise add or subtract h to x1: out = x1 + h if reverse==False else out = x1 - h.
    x1_ptr/h_ptr/out_ptr point to tensors of shape [N, C, T].
    Grid: (N, C, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_block = tl.program_id(2)

    t_start = pid_block * BLOCK
    t_idx = t_start + tl.arange(0, BLOCK)
    valid_t = t_idx < T

    x1_offsets = pid_n * x1_sN + pid_c * x1_sC + t_idx * x1_sT
    h_offsets = pid_n * h_sN + pid_c * h_sC + t_idx * h_sT

    x1_vals = tl.load(x1_ptr + x1_offsets, mask=valid_t, other=0.0).to(tl.float32)
    h_vals = tl.load(h_ptr + h_offsets, mask=valid_t, other=0.0).to(tl.float32)

    if reverse:
        out_vals = x1_vals - h_vals
    else:
        out_vals = x1_vals + h_vals

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid_t)

            N, OC1, T,
            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),

        y1_relu = torch.empty_like(y1_masked)

            N, OC1, T,

            y1_relu.stride(0), y1_relu.stride(1), y1_relu.stride(2),
            BLOCK=128, num_warps=4, num_stages=2
        )

        # conv2: IC=192, OC=96, K=5, padding=2
        OC2 = 96
        IC2 = 192
        K2 = 5
        pad2 = K2 // 2
        h = torch.empty((N, OC2, T), dtype=x.dtype, device=x.device)  # coupling tensor

            h.stride(0), h.stride(1), h.stride(2),
            BLOCK_T=128, num_warps=4, num_stages=2
        )

        # apply mask for coupling (broadcast over channels)
        h_masked = torch.empty_like(h)
        mask_apply_triton[(N, OC2, triton.cdiv(T, 128))](
            h, x_mask, h_masked,
            N, OC2, T,
            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
            h.stride(0), h.stride(1), h.stride(2),
            h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
            BLOCK=128, num_warps=4, num_stages=2
        )

        # Update x1: x1 = x1 + h_masked (forward) or x1 = x1 - h_masked (reverse)
        out_x1 = torch.empty_like(x1)
        add_h_triton[(N, x1.shape[1], triton.cdiv(T, 128))](
            x1, h_masked, out_x1,
            N, x1.shape[1], T,
            x1.stride(0), x1.stride(1), x1.stride(2),
            h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
            out_x1.stride(0), out_x1.stride(1), out_x1.stride(2),
            reverse=reverse,
            BLOCK=128, num_warps=4, num_stages=2
        )

        # Reconstruct x: cat([x0, out_x1], dim=1)
        new_x = torch.empty((N, C, T), dtype=x.dtype, device=x.device)
        # write x0 into first half
        new_x[:, :half_channels, :] = x0
        # write updated x1 into second half
        new_x[:, half_channels:, :] = out_x1

        # Apply mask to the entire tensor after each transform (original code does this)
        new_x = new_x * x_mask

        # Update current x for next iteration
        cur_x = new_x.clone()

    return cur_x

Notes on correctness and performance:
- Conv1d: Implemented as cross-correlation with stride=1, padding=K//2, dilation=1. Each program computes a vector of output positions (BLOCK_T) for a given (batch, output-channel) pair. We loop over input channels and kernel taps, load x and w, accumulate in float32, add bias, and store. This matches F.conv1d behavior for the given weights and padding.
- Masking: The mask x_mask is [N, 1, T]. The Triton mask_apply_triton kernel loads the mask slice and multiplies the entire tensor y [N, OC, T] by it, broadcasting across channels. This matches the original’s “x = x * x_mask” behavior after each transform.
- ReLU: Implemented in Triton as elementwise max(0). We apply ReLU after conv0 and conv1 exactly like the original.
- Affine coupling: Implemented in Triton add_h_triton kernel, adding or subtracting h from x1 depending on reverse. Then we reconstruct x via concatenation using tensor slicing (which is metadata/memcpy on GPU, not torch.cat in the host).
- Dtypes: All kernels operate in float32. Inputs from get_inputs are float32 by default. The forward returns the final tensor.

This implementation ensures no torch.conv1d or torch.cat in the forward, uses Triton for all computational kernels, and should fix previous runtime errors by applying the mask across the entire output and launching kernels correctly.

import math
import torch
import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def conv1d_triton_2d(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N, IC, OC, T, K, pad,
    x_sN, x_sC, x_sT,
    w_sO, w_sI, w_sK,
    out_sN, out_sC, out_sT,
    BLOCK_T: tl.constexpr,
):
    """
    Triton implementation of Conv1d (cross-correlation) for stride=1, padding=pad, dilation=1.
    2D grid over (N, OC), loop over T in chunks of BLOCK_T.
    x: [N, IC, T] (float32)
    w: [OC, IC, K] (float32)
    b: [OC] (float32)
    out: [N, OC, T] (float32)
    """
    pid_n = tl.program_id(0)
    pid_oc = tl.program_id(1)

    # Iterate over output time positions in chunks
    for t_start in range(0, T, BLOCK_T):
        t_out_idx = t_start + tl.arange(0, BLOCK_T)
        valid_t = t_out_idx < T

        # Accumulator for this oc and chunk
        acc = tl.zeros([BLOCK_T], dtype=tl.float32)

        # Loop over input channels and kernel taps
        for ic in range(0, IC):
            for k in range(0, K):
                t_in = t_out_idx + k - pad  # vector
                valid_t_in = valid_t & (t_in >= 0) & (t_in < T)

                # Load x[n, ic, t_in]
                x_offsets = pid_n * x_sN + ic * x_sC + t_in * x_sT
                x_vals = tl.load(x_ptr + x_offsets, mask=valid_t_in, other=0.0)

                # Load w[oc, ic, k] as scalar
                w_val = tl.load(w_ptr + pid_oc * w_sO + ic * w_sI + k * w_sK)

                # Accumulate
                acc += x_vals * w_val

        # Add bias
        b_val = tl.load(b_ptr + pid_oc)
        acc += b_val

        # Store to out[n, oc, t_out_idx]
        out_offsets = pid_n * out_sN + pid_oc * out_sC + t_out_idx * out_sT
        tl.store(out_ptr + out_offsets, acc, mask=valid_t)

@triton.jit
def add_h_triton(
    x1_ptr, h_ptr, out_ptr,
    N, C, T,
    x1_sN, x1_sC, x1_sT,
    h_sN, h_sC, h_sT,
    out_sN, out_sC, out_sT,
    reverse: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Elementwise add or subtract h to x1: out = x1 + h if reverse==False else out = x1 - h.
    2D grid: (N, C), iterate over T in chunks of BLOCK.
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)

    for t_start in range(0, T, BLOCK):
        t_idx = t_start + tl.arange(0, BLOCK)
        valid_t = t_idx < T

        x1_offsets = pid_n * x1_sN + pid_c * x1_sC + t_idx * x1_sT
        h_offsets = pid_n * h_sN + pid_c * h_sC + t_idx * h_sT

        x1_vals = tl.load(x1_ptr + x1_offsets, mask=valid_t, other=0.0).to(tl.float32)
        h_vals = tl.load(h_ptr + h_offsets, mask=valid_t, other=0.0).to(tl.float32)

        if reverse:
            out_vals = x1_vals - h_vals
        else:
            out_vals = x1_vals + h_vals

        out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
        tl.store(out_ptr + out_offsets, out_vals, mask=valid_t)


@torch.no_grad()
def run(
    x: torch.Tensor,
    x_mask: torch.Tensor,
    reverse: bool,
    transform_0_conv0_weight, transform_0_conv0_bias,
    transform_0_conv1_weight, transform_0_conv1_bias,
    transform_0_conv2_weight, transform_0_conv2_bias,
    transform_1_conv0_weight, transform_1_conv0_bias,
    transform_1_conv1_weight, transform_1_conv1_bias,
    transform_1_conv2_weight, transform_1_conv2_bias,
    transform_2_conv0_weight, transform_2_conv0_bias,
    transform_2_conv1_weight, transform_2_conv1_bias,
    transform_2_conv2_weight, transform_2_conv2_bias,
    transform_3_conv0_weight, transform_3_conv0_bias,
    transform_3_conv1_weight, transform_3_conv1_bias,
    transform_3_conv2_weight, transform_3_conv2_bias,
):
    """
    Triton-only implementation of the original forward.
    - All conv1d operations are computed by Triton.
    - Mask application, ReLU, and affine coupling are computed by Triton.
    - No torch.conv1d or torch.cat in the forward.
    """
    N, C, T = x.shape
    assert C == 192, "This implementation assumes C=192 for half_channels=96."
    half_channels = C // 2

    # Transform list to process. If reverse=True, process in reversed order.
    transforms = [
        (transform_0_conv0_weight, transform_0_conv0_bias,
         transform_0_conv1_weight, transform_0_conv1_bias,
         transform_0_conv2_weight, transform_0_conv2_bias),
        (transform_1_conv0_weight, transform_1_conv0_bias,
         transform_1_conv1_weight, transform_1_conv1_bias,
         transform_1_conv2_weight, transform_1_conv2_bias),
        (transform_2_conv0_weight, transform_2_conv0_bias,
         transform_2_conv1_weight, transform_2_conv1_bias,
         transform_2_conv2_weight, transform_2_conv2_bias),
        (transform_3_conv0_weight, transform_3_conv0_bias,
         transform_3_conv1_weight, transform_3_conv1_bias,
         transform_3_conv2_weight, transform_3_conv2_bias),
    ]
    if not reverse:
        transform_list = transforms
    else:
        transform_list = list(reversed(transforms))

    # Maintain current x during iterations; final x is returned
    cur_x = x.clone()

    # Common BLOCK sizes for loops
    BLOCK_T = 128
    BLOCK = 128

    for w0, b0, w1, b1, w2, b2 in transform_list:
        # Split current x into halves
        x0 = cur_x[:, :half_channels, :].contiguous()  # [N, 96, T]
        x1 = cur_x[:, half_channels:, :].contiguous()  # [N, 96, T]

        # conv0: IC=96, OC=192, K=5, padding=2
        OC0 = 192
        IC0 = 96
        K0 = 5
        pad0 = K0 // 2
        y0 = torch.empty((N, OC0, T), dtype=x.dtype, device=x.device)
        grid_conv0 = (N, OC0)
        conv1d_triton_2d[grid_conv0](
            x0, w0, b0, y0,
            N, IC0, OC0, T, K0, pad0,
            x0.stride(0), x0.stride(1), x0.stride(2),
            w0.stride(0), w0.stride(1), w0.stride(2),
            y0.stride(0), y0.stride(1), y0.stride(2),
            BLOCK_T=BLOCK_T, num_warps=4, num_stages=2
        )

        # apply mask and ReLU to y0 (mask is [N,1,T], broadcast along channels)
        y0_masked = torch.empty_like(y0)
        grid_mask0 = (N, OC0)
        mask_apply_triton[grid_mask0](
            y0, x_mask, y0_masked,
            N, OC0, T,
            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
            y0.stride(0), y0.stride(1), y0.stride(2),
            y0_masked.stride(0), y0_masked.stride(1), y0_masked.stride(2),
            BLOCK=BLOCK, num_warps=4, num_stages=2
        )
        y0_relu = torch.empty_like(y0_masked)
        grid_relu0 = (N, OC0)
        relu_triton[grid_relu0](
            y0_masked, y0_relu,
            N, OC0, T,
            y0_masked.stride(0), y0_masked.stride(1), y0_masked.stride(2),
            y0_relu.stride(0), y0_relu.stride(1), y0_relu.stride(2),
            BLOCK=BLOCK, num_warps=4, num_stages=2
        )

        # conv1: IC=192, OC=192, K=5, padding=2
        OC1 = 192
        IC1 = 192
        K1 = 5
        pad1 = K1 // 2
        y1 = torch.empty((N, OC1, T), dtype=x.dtype, device=x.device)
        grid_conv1 = (N, OC1)
        conv1d_triton_2d[grid_conv1](
            y0_relu, w1, b1, y1,
            N, IC1, OC1, T, K1, pad1,
            y0_relu.stride(0), y0_relu.stride(1), y0_relu.stride(2),
            w1.stride(0), w1.stride(1), w1.stride(2),
            y1.stride(0), y1.stride(1), y1.stride(2),
            BLOCK_T=BLOCK_T
