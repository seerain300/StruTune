    """
    N, C, T = x.shape
    assert C == 192, "This implementation assumes channels=192."
    half_channels = C // 2

    # Process transforms sequentially (forward) or reversed (backward)
    transforms = [
        (transform_0_conv0_weight, transform_0_conv0_bias,
         transform_0_conv1_weight, transform_0_conv1_bias,
         transform_0_conv2_weight, transform_0_conv2_bias),
        (transform_1_conv0_weight, transform_1_conv0_bias,
         transform_1_conv1_weight, transform_1_conv1_bias,
         transform_1_conv2_weight, transform_1_conv2_bias),
        (transform_2_conv0_weight, transform_2_conv0_bias,
         transform_2_conv1_weight, transform_2_conv1_bias,
         transform_2_conv2_weight, transform_2_conv2_bias),
        (transform_3_conv0_weight, transform_3_conv0_bias,
         transform_3_conv1_weight, transform_3_conv1_bias,
         transform_3_conv2_weight, transform_3_conv2_bias),
    ]

    for w0, b0, w1, b1, w2, b2 in (transforms if not reverse else reversed(transforms)):
        # Split x into x0 and x1
        x0 = x[:, :half_channels, :].contiguous()  # [N, 96, T]
        x1 = x[:, half_channels:, :].contiguous()  # [N, 96, T]

        # conv0: out_channels=192, in_channels=96, K=5, padding=2
        h0 = F.conv1d(x0, w0, b0, padding=2)  # [N, 192, T]
        # mask and ReLU via Triton

        multiply_mask_triton[(N, 192, triton.cdiv(T, 256))](h0, x_mask, h0_masked, N, 192, T,

        h0_relu = torch.empty_like(h0_masked)
        relu_triton[(N, 192, triton.cdiv(T, 256))](h0_masked, h0_relu, N, 192, T,
                                                   h0_masked.stride(0), h0_masked.stride(1), h0_masked.stride(2),

                                                   BLOCK=256, num_warps=4)

        # conv1
        h = F.conv1d(h0_relu, w1, b1, padding=2)  # [N, 192, T]
        # mask and ReLU via Triton
        h_masked = torch.empty_like(h)
        multiply_mask_triton[(N, 192, triton.cdiv(T, 256))](h, x_mask, h_masked, N, 192, T,
                                                             h.stride(0), h.stride(1), h.stride(2),
                                                             x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                             h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
                                                             BLOCK=256, num_warps=4)
        h_relu = torch.empty_like(h_masked)
        relu_triton[(N, 192, triton.cdiv(T, 256))](h_masked, h_relu, N, 192, T,
                                                   h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
                                                   h_relu.stride(0), h_relu.stride(1), h_relu.stride(2),
                                                   BLOCK=256, num_warps=4)

        # conv2: out_channels=96, in_channels=192, K=5, padding=2
        y = F.conv1d(h_relu, w2, b2, padding=2)  # [N, 96, T]

        # Apply mask to y
        y_masked = torch.empty_like(y)
        multiply_mask_triton[(N, 96, triton.cdiv(T, 256))](y, x_mask, y_masked, N, 96, T,
                                                            y.stride(0), y.stride(1), y.stride(2),
                                                            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                            y_masked.stride(0), y_masked.stride(1), y_masked.stride(2),
                                                            BLOCK=256, num_warps=4)

        # Update x1 via coupling
        out_x1 = torch.empty_like(x1)
        is_add = 1 if not reverse else 0
        add_h_triton[(N, 96, triton.cdiv(T, 256))](x1, y_masked, out_x1, N, 96, T,
                                                   x1.stride(0), x1.stride(1), x1.stride(2),
                                                   y_masked.stride(0), y_masked.stride(1), y_masked.stride(2),
                                                   out_x1.stride(0), out_x1.stride(1), out_x1.stride(2),
                                                   is_add=is_add, BLOCK=256, num_warps=4)

        # Concatenate [x0, out_x1] -> new x
        x = torch.cat([x0, out_x1], dim=1)  # [N, 192, T]

        # Finally, scale the entire output by x_mask (broadcast along channels)
        x_scaled = torch.empty_like(x)
        final_mask_scale_triton[(N, 192, triton.cdiv(T, 256))](x, x_mask, x_scaled, N, 192, T,
                                                               x.stride(0), x.stride(1), x.stride(2),
                                                               x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                               x_scaled.stride(0), x_scaled.stride(1), x_scaled.stride(2),
                                                               BLOCK=256, num_warps=4)
        x = x_scaled

    return x
