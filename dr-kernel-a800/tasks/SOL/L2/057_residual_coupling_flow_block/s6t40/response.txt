import math
import torch
import triton
import triton.language as tl

# Triton kernels: conv1d with K=5, padding=2 (valid conv), output length = T_in - 1
@triton.jit
def conv1d_k5_p2(
    x_ptr,           # *f32, input: [B, Cin, T_in]
    w_ptr,           # *f32, weight: [Cout, Cin, 5]
    bias_ptr,        # *f32, bias: [Cout] or None (we assume always provided)
    y_ptr,           # *f32, output: [B, Cout, T_out], T_out = T_in - 1
    B: tl.constexpr,
    Cin: tl.constexpr,
    Cout: tl.constexpr,
    T_in: tl.constexpr,
    T_out: tl.constexpr,
    stride_x_b, stride_x_c, stride_x_t,
    stride_w_co, stride_w_ci, stride_w_k,
    stride_y_b, stride_y_c, stride_y_t,
    BLOCK_T: tl.constexpr,
):
    # Program ids
    pid_b = tl.program_id(0)  # batch index
    pid_cout = tl.program_id(1)  # output channel index
    pid_tile = tl.program_id(2)  # tile along time

    # Compute time offsets for this tile
    t_offsets = pid_tile * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    # Initialize accumulator for this (b, cout, tile)
    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # Accumulate over input channels and kernel taps
    for ci in range(0, Cin):
        for k in range(0, 5):
            # Input index with padding: t_in = t - 2 + k
            t_in = t_offsets - 2 + k
            valid = (t_in >= 0) & (t_in < T_in) & mask_t

            # Load x[b, ci, t_in] with mask; other=0
            x_offset = pid_b * stride_x_b + ci * stride_x_c + t_in * stride_x_t
            x_val = tl.load(x_ptr + x_offset, mask=valid, other=0.0)

            # Load weight w[cout, ci, k]
            w_offset = cout * stride_w_co + ci * stride_w_ci + k * stride_w_k
            w_val = tl.load(w_ptr + w_offset)

            # FMA
            acc += x_val * w_val

    # Add bias for this cout
    bias_val = tl.load(bias_ptr + cout)
    acc += bias_val

    # Store to y[b, cout, t_offsets]
    y_offset = pid_b * stride_y_b + cout * stride_y_c + t_offsets * stride_y_t
    tl.store(y_ptr + y_offset, acc, mask=mask_t)


@triton.jit
def add_bias(y_ptr, bias_ptr, B, Cout, T, stride_y_b, stride_y_c, stride_y_t):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)
    t = pid_t
    y_offset = pid_b * stride_y_b + pid_c * stride_y_c + t * stride_y_t
    val = tl.load(y_ptr + y_offset)
    bias_val = tl.load(bias_ptr + pid_c)
    tl.store(y_ptr + y_offset, val + bias_val)


@triton.jit
def relu_kernel(y_ptr, B, C, T, stride_y_b, stride_y_c, stride_y_t):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)
    t = pid_t
    y_offset = pid_b * stride_y_b + pid_c * stride_y_c + t * stride_y_t
    val = tl.load(y_ptr + y_offset)
    val = tl.maximum(val, 0.0)
    tl.store(y_ptr + y_offset, val)


@triton.jit
def mul_mask(y_ptr, mask_ptr, B, C, T, stride_y_b, stride_y_c, stride_y_t, stride_m_b, stride_m_c, stride_m_t):
    # mask shape: [B, 1, T]; broadcast along C
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)
    t = pid_t
    y_offset = pid_b * stride_y_b + pid_c * stride_y_c + t * stride_y_t
    val = tl.load(y_ptr + y_offset)

    # Load mask for channel 0 (broadcast works; mask only has C=1)
    m_offset = pid_b * stride_m_b + 0 * stride_m_c + t * stride_m_t
    m_val = tl.load(mask_ptr + m_offset)

    val = val * m_val
    tl.store(y_ptr + y_offset, val)


@triton.jit
def add_or_sub(y_ptr, add_ptr, other_ptr, B, C, T, stride_y_b, stride_y_c, stride_y_t):
    # y_ptr points to the second half x1
    # add_ptr points to h2 (transform output), other_ptr may be None (we pass a dummy tensor); but here we only add/sub
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)
    t = pid_t
    y_offset = pid_b * stride_y_b + pid_c * stride_y_c + t * stride_y_t
    val = tl.load(y_ptr + y_offset)
    add_val = tl.load(add_ptr + y_offset)  # h2 value at this (b, c, t)
    # sign is + for forward, - for reverse (not used here)
    val = val + add_val
    tl.store(y_ptr + y_offset, val)


@triton.jit
def copy_to(y_dst_ptr, src_ptr, B, C_dst, C_src, T, stride_dst_b, stride_dst_c, stride_dst_t,
            stride_src_b, stride_src_c, stride_src_t):
    # Copy src (shape [B, C_src, T]) to y_dst (shape [B, C_dst, T]), only for c in [0, C_src)
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)
    t = pid_t
    src_offset = pid_b * stride_src_b + pid_c * stride_src_c + t * stride_src_t
    val = tl.load(src_ptr + src_offset)
    dst_offset = pid_b * stride_dst_b + pid_c * stride_dst_c + t * stride_dst_t
    tl.store(y_dst_ptr + dst_offset, val)


# Forward path: run 4 transforms sequentially, apply ReLU and mask after each conv+bias, then couple and concatenate halves
def ModelNew(
    x: torch.Tensor,
    x_mask: torch.Tensor,
    reverse: bool,
    transform_0_conv0_weight: torch.Tensor,
    transform_0_conv0_bias: torch.Tensor,
    transform_0_conv1_weight: torch.Tensor,
    transform_0_conv1_bias: torch.Tensor,
    transform_0_conv2_weight: torch.Tensor,
    transform_0_conv2_bias: torch.Tensor,
    transform_1_conv0_weight: torch.Tensor,
    transform_1_conv0_bias: torch.Tensor,
    transform_1_conv1_weight: torch.Tensor,
    transform_1_conv1_bias: torch.Tensor,
    transform_1_conv2_weight: torch.Tensor,
    transform_1_conv2_bias: torch.Tensor,
    transform_2_conv0_weight: torch.Tensor,
    transform_2_conv0_bias: torch.Tensor,
    transform_2_conv1_weight: torch.Tensor,
    transform_2_conv1_bias: torch.Tensor,
    transform_2_conv2_weight: torch.Tensor,
    transform_2_conv2_bias: torch.Tensor,
    transform_3_conv0_weight: torch.Tensor,
    transform_3_conv0_bias: torch.Tensor,
    transform_3_conv1_weight: torch.Tensor,
    transform_3_conv1_bias: torch.Tensor,
    transform_3_conv2_weight: torch.Tensor,
    transform_3_conv2_bias: torch.Tensor,
):
    """
    Triton-only forward. Applies 4 transforms sequentially to x0 (first 96 channels) and couples to x1 (second 96 channels).
    No torch ops in forward; all math is done by Triton kernels.
    """
    device = x.device
    assert x.is_cuda, "ModelNew requires CUDA tensors"
    B, C, T = x.shape
    assert C == 192, "Channel size must be 192"
    half_channels = 192 // 2  # 96

    # We'll run 4 transforms
    # Allocate final output y: [B, 192, T_final] where T_final = T - 12 (each conv reduces time by 1, 3 convs -> -3; but actually each conv reduces time by 1 total in this pipeline, so T_final = T - 4*3 = T - 12)
    T_final = T - 12
    y = torch.empty((B, 192, T_final), dtype=x.dtype, device=device)

    # Helper to run one transform: conv0 -> conv1 -> conv2, with ReLU and mask, coupling, and concatenate halves into y
    def run_one_transform(x0, x1, conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b):
        # Output tensors for each conv
        # conv0: [B, 192, T-1]
        y0 = torch.empty((B, 192, T - 1), dtype=x.dtype, device=device)
        # conv1: [B, 192, T-2]
        y1 = torch.empty((B, 192, T - 2), dtype=x.dtype, device=device)
        # conv2: [B, 96, T-3]
        y2 = torch.empty((B, 96, T - 3), dtype=x.dtype, device=device)

        # Launch conv0
        grid0 = (B, 192, triton.cdiv(T - 1, 128))
        conv1d_k5_p2[grid0](
            x0, conv0_w, conv0_b, y0,
            B, 96, 192, T, T - 1,
            x0.stride(0), x0.stride(1), x0.stride(2),
            conv0_w.stride(0), conv0_w.stride(1), conv0_w.stride(2),
            y0.stride(0), y0.stride(1), y0.stride(2),
            BLOCK_T=128
        )

        # Bias and ReLU conv0 output
        grid_bias0 = (B, 192, T - 1)
        add_bias[grid_bias0](y0, conv0_b, B, 192, T - 1, y0.stride(0), y0.stride(1), y0.stride(2))
        grid_relu0 = (B, 192, T - 1)
        relu_kernel[grid_relu0](y0, B, 192, T - 1, y0.stride(0), y0.stride(1), y0.stride(2))

        # Apply mask: multiply by x_mask (B,1,T), broadcast across C
        grid_mask0 = (B, 192, T - 1)
        mul_mask[grid_mask0](y0, x_mask, B, 192, T - 1, y0.stride(0), y0.stride(1), y0.stride(2),
                             x_mask.stride(0), x_mask.stride(1), x_mask.stride(2))

        # conv1: input is y0 (B,192,T-1)
        grid1 = (B, 192, triton.cdiv(T - 2, 128))
        conv1d_k5_p2[grid1](
            y0, conv1_w, conv1_b, y1,
            B, 192, 192, T - 1, T - 2,
            y0.stride(0), y0.stride(1), y0.stride(2),
            conv1_w.stride(0), conv1_w.stride(1), conv1_w.stride(2),
            y1.stride(0), y1.stride(1), y1.stride(2),
            BLOCK_T=128
        )

        # Bias and ReLU conv1 output
        grid_bias1 = (B, 192, T - 2)
        add_bias[grid_bias1](y1, conv1_b, B, 192, T - 2, y1.stride(0), y1.stride(1), y1.stride(2))
        grid_relu1 = (B, 192, T - 2)
        relu_kernel[grid_relu1](y1, B, 192, T - 2, y1.stride(0), y1.stride(1), y1.stride(2))

        # Apply mask conv1 output
        grid_mask1 = (B, 192, T - 2)
        mul_mask[grid_mask1](y1, x_mask, B, 192, T - 2, y1.stride(0), y1.stride(1), y1.stride(2),
                             x_mask.stride(0), x_mask.stride(1), x_mask.stride(2))

        # conv2: input is y1 (B,192,T-2)
        grid2 = (B, 96, triton.cdiv(T - 3, 128))
        conv1d_k5_p2[grid2](
            y1, conv2_w, conv2_b, y2,
            B, 192, 96, T - 2, T - 3,
            y1.stride(0), y1.stride(1), y1.stride(2),
            conv2_w.stride(0), conv2_w.stride(1), conv2_w.stride(2),
            y2.stride(0), y2.stride(1), y2.stride(2),
            BLOCK_T=128
        )

        # Bias and ReLU conv2 output
        grid_bias2 = (B, 96, T - 3)
        add_bias[grid_bias2](y2, conv2_b, B, 96, T - 3, y2.stride(0), y2.stride(1), y2.stride(2))
        grid_relu2 = (B, 96, T - 3)
        relu_kernel[grid_relu2](y2, B, 96, T - 3, y2.stride(0), y2.stride(1), y2.stride(2))

        # Apply mask conv2 output
        grid_mask2 = (B, 96, T - 3)
        mul_mask[grid_mask2](y2, x_mask, B, 96, T - 3, y2.stride(0), y2.stride(1), y2.stride(2),
                             x_mask.stride(0), x_mask.stride(1), x_mask.stride(2))

        # Coupling: add h2 to x1
        # x1 has shape [B, 96, T-3]
        # y2 has shape [B, 96, T-3]
        grid_add = (B, 96, T - 3)
        add_or_sub[grid_add](x1, y2, None, B, 96, T - 3, x1.stride(0), x1.stride(1), x1.stride(2))

        # Concatenate into y[:, :96, :], y[:, 96:, :]
        # Write x0 (first half channels) into y[:, :96, :time0]
        time0 = T_final  # T_final = T - 12
        grid_copy_x0 = (B, 96, time0)
        copy_to[y](
            y, x0, B, 96, 96, time0,
            y.stride(0), y.stride(1), y.stride(2),
            x0.stride(0), x0.stride(1), x0.stride(2),
            grid=grid_copy_x0
        )

        # Write x1 (second half channels) into y[:, 96:, :]
        grid_copy_x1 = (B, 96, time0)
        copy_to[y](
            y, x1, B, 96, 96, time0,
            y.stride(0), y.stride(1), y.stride(2),
            x1.stride(0), x1.stride(1), x1.stride(2),
            grid=grid_copy_x1
        )

    # Initialize x0 and x1 from input x
    x0 = x[:, :half_channels, :]
    x1 = x[:, half_channels:, :]

    # Run 4 transforms
    run_one_transform(x0, x1, transform_0_conv0_weight, transform_0_conv0_bias,
                      transform_0_conv1_weight, transform_0_conv1_bias,
                      transform_0_conv2_weight, transform_0_conv2_bias)

    run_one_transform(x0, x1, transform_1_conv0_weight, transform_1_conv0_bias,
                      transform_1_conv1_weight, transform_1_conv1_bias,
                      transform_1_conv2_weight, transform_1_conv2_bias)

    run_one_transform(x0, x1, transform_2_conv0_weight, transform_2_conv0_bias,
                      transform_2_conv1_weight, transform_2_conv1_bias,
                      transform_2_conv2_weight, transform_2_conv2_bias)

    run_one_transform(x0, x1, transform_3_conv0_weight, transform_3_conv0_bias,
                      transform_3_conv1_weight, transform_3_conv1_bias,
                      transform_3_conv2_weight, transform_3_conv2_bias)

    return y
