2) add_bias: Triton elementwise kernel to add bias per channel across the tensor. Launch after conv.

3) relu_kernel: Triton elementwise kernel for ReLU. Launch after bias add.

4) mul_mask: Triton elementwise kernel to multiply by x_mask (broadcasted across channels). We will pass mask as [B, 1, T] and load per channel (broadcast inside kernel). Launch after ReLU.

5) add_or_sub: Triton elementwise kernel to add or subtract a tensor h to/from a selected half of x. We will implement add_or_sub for the last half of x (channels 96..191), reading shifted x1_half at its output time length. Launch after mask multiply.

```python
# Conv1d with K=5, padding=2 (valid conv), output time = input time - 1
@triton.jit
def conv1d_k5_p2(
    x_ptr,          # *const T, x: [B, Cin, T_in]
    w_ptr,          # *const T, w: [Cout, Cin, K]
    bias_ptr,       # *const T, bias: [Cout]
    y_ptr,          # *T, y: [B, Cout, T_out]
    B: tl.constexpr, Cin: tl.constexpr, T_in: tl.constexpr,
    Cout: tl.constexpr, K: tl.constexpr, T_out: tl.constexpr,
    x_stride_b, x_stride_c, x_stride_t,
    w_stride_co, w_stride_ci, w_stride_k,
    y_stride_b, y_stride_c, y_stride_t,
    BLOCK_T: tl.constexpr,
):
    pid_b = tl.program_id(axis=0)
    pid_co = tl.program_id(axis=1)
    pid_t_blk = tl.program_id(axis=2)

    t_offsets = pid_t_blk * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    # Accumulator
    acc = tl.zeros((BLOCK_T,), dtype=tl.float32)

    # Loop over input channels and kernel taps
    for ci in range(0, Cin):
        for k in range(0, K):
            t_src = t_offsets - 2 + k  # padding=2 -> start at t - 2
            mask_src = (t_src >= 0) & (t_src < T_in) & mask_t
            # x[b, ci, t_src]
            x_off = pid_b * x_stride_b + ci * x_stride_c + t_src * x_stride_t
            x_val = tl.load(x_ptr + x_off, mask=mask_src, other=0.0)
            x_val = x_val.to(tl.float32)
            # w[co, ci, k]
            w_off = pid_co * w_stride_co + ci * w_stride_ci + k * w_stride_k
            w_val = tl.load(w_ptr + w_off)
            w_val = w_val.to(tl.float32)
            acc += x_val * w_val

    # add bias
    b = tl.load(bias_ptr + pid_co).to(tl.float32)
    acc += b

    # store y[b, co, t_offsets]
    y_off = pid_b * y_stride_b + pid_co * y_stride_c + t_offsets * y_stride_t
    tl.store(y_ptr + y_off, acc, mask=mask_t)

                                    transform_2_conv2_weight, transform_2_conv2_bias, T)

        # transform_3
        h2_3, y_out = run_transform(x[:, :96, :], transform_3_conv0_weight, transform_3_conv0_bias,
                                    transform_3_conv1_weight, transform_3_conv1_bias,
                                    transform_3_conv2_weight, transform_3_conv2_bias, T)

        # Return final output (after all transforms). We add the last h2 to the second half (if forward), but since forward uses y_out updated above, return y_out.
        return y_out
```
