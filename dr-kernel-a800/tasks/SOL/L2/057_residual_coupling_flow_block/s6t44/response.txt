import math
import torch
import triton
import triton.language as tl


@triton.jit
def conv1d_k5_p2(x_ptr, w_ptr, y_ptr,
                 B, C_IN, C_OUT, T_IN, T_OUT,
                 BLOCK_CO: tl.constexpr, BLOCK_T: tl.constexpr):
    # Grid: (B, ceil_div(C_OUT, BLOCK_CO), ceil_div(T_OUT, BLOCK_T))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_t = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    co_mask = co_offsets < C_OUT
    t_mask = t_offsets < T_OUT

    # Initialize accumulator for the output vector at these (b, co, t)
    acc = tl.zeros([BLOCK_CO, BLOCK_T], dtype=tl.float32)

    # Loop over input channels and kernel taps (K=5, padding=2 => valid conv)
    for ci in range(C_IN):
        # For each kernel tap k, compute input time indices
        # t_in = t_out - 2 + k
        # We loop k=0..4
        for k in range(5):
            t_in = t_offsets - 2 + k  # vector of input time indices
            # Valid mask for loads: 0 <= t_in < T_IN
            valid_t = (t_in >= 0) & (t_in < T_IN)
            # Compute base offsets for x and w
            # x offset: ((b * C_IN + ci) * T_IN + t_in)
            # w offset: ((co * C_IN + ci) * K + k)
            # Note: x is [B, C_IN, T_IN], contiguous in time
            # We can flatten x storage as [B, C_IN, T_IN] with strides: B stride = C_IN*T_IN, C stride = T_IN, T stride = 1
            # For pointer arithmetic, we use linear index: ((b * C_IN + ci) * T_IN + t_in)
            # w is [C_OUT, C_IN, K]
            # So w offset: co* (C_IN*K) + ci*K + k
            # Here ci varies per outer loop; k varies per inner loop
            # Load input vector x[b, ci, t_in] for all t_offsets
            # Using linear index: ((b * C_IN + ci) * T_IN + t_in)
            x_idx = (pid_b * C_IN + ci) * T_IN + t_in
            # W vector for these co_offsets and fixed (ci, k)
            w_idx = co_offsets * (C_IN * 5) + ci * 5 + k
            # Load weights as vector over co
            w_vec = tl.load(w_ptr + w_idx, mask=co_mask, other=0.0)  # [BLOCK_CO]
            # Load x as vector over t_offsets
            x_vec = tl.load(x_ptr + x_idx, mask=valid_t, other=0.0)  # [BLOCK_T]
            # Outer product accumulate: acc[co, t] += w[co] * x[t]
            acc += w_vec[:, None] * x_vec[None, :]

    # Store the accumulated result to y
    # y is [B, C_OUT, T_OUT], linear index: ((b * C_OUT + co) * T_OUT + t)
    y_idx = (pid_b * C_OUT + co_offsets) * T_OUT + t_offsets
    # Store only within bounds
    tl.store(y_ptr + y_idx, acc, mask=co_mask[:, None] & t_mask[None, :])


@triton.jit
def add_bias(y_ptr, b_ptr, B, C_OUT, T_OUT, BLOCK_CO: tl.constexpr, BLOCK_T: tl.constexpr):
    # Grid: (B, ceil_div(C_OUT, BLOCK_CO), ceil_div(T_OUT, BLOCK_T))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_t = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    co_mask = co_offsets < C_OUT
    t_mask = t_offsets < T_OUT

    y_idx = (pid_b * C_OUT + co_offsets) * T_OUT + t_offsets
    y = tl.load(y_ptr + y_idx, mask=co_mask[:, None] & t_mask[None, :], other=0.0)
    b_vec = tl.load(b_ptr + co_offsets, mask=co_mask, other=0.0)  # [BLOCK_CO]
    y = y + b_vec[:, None]
    tl.store(y_ptr + y_idx, y, mask=co_mask[:, None] & t_mask[None, :])


@triton.jit
def relu(y_ptr, B, C_OUT, T_OUT, BLOCK_CO: tl.constexpr, BLOCK_T: tl.constexpr):
    # Grid: (B, ceil_div(C_OUT, BLOCK_CO), ceil_div(T_OUT, BLOCK_T))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_t = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    co_mask = co_offsets < C_OUT
    t_mask = t_offsets < T_OUT

    y_idx = (pid_b * C_OUT + co_offsets) * T_OUT + t_offsets
    y = tl.load(y_ptr + y_idx, mask=co_mask[:, None] & t_mask[None, :], other=0.0)
    y = tl.maximum(y, 0.0)
    tl.store(y_ptr + y_idx, y, mask=co_mask[:, None] & t_mask[None, :])


@triton.jit
def mul_mask(y_ptr, mask_ptr, B, C_OUT, T_OUT, BLOCK_CO: tl.constexpr, BLOCK_T: tl.constexpr):
    # Grid: (B, ceil_div(C_OUT, BLOCK_CO), ceil_div(T_OUT, BLOCK_T))
    pid_b = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_t = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    co_mask = co_offsets < C_OUT
    t_mask = t_offsets < T_OUT

    y_idx = (pid_b * C_OUT + co_offsets) * T_OUT + t_offsets
    mask_vec = tl.load(mask_ptr, mask=t_mask, other=1.0)  # mask_ptr is [B, 1, T], single tensor
    y = tl.load(y_ptr + y_idx, mask=co_mask[:, None] & t_mask[None, :], other=0.0)
    # Broadcast mask across channels: shape [BLOCK_CO, BLOCK_T] is fine since mask_vec is [BLOCK_T]
    # Here we multiply elementwise by mask across time
    # Note: we only have one mask tensor of shape [B, 1, T]; we load it as a vector and broadcast
    y = y * mask_vec[None, :]
    tl.store(y_ptr + y_idx, y, mask=co_mask[:, None] & t_mask[None, :])


@triton.jit
def add_second_half(y_ptr, h2_ptr, B, C_HALF, T_H2, T_FINAL, BLOCK_CO: tl.constexpr, BLOCK_T: tl.constexpr):

    pid_t = tl.program_id(2)

    co_offsets = pid_co * BLOCK_CO + tl.arange(0, BLOCK_CO)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)

    co_mask = co_offsets < C_HALF
    t_mask = t_offsets < T_FINAL

    h2_val = tl.load(h2_ptr + h2_idx, mask=co_mask[:, None] & t_mask[None, :], other=0.0)
    y_val = y_val + h2_val
    tl.store(y_ptr + y_base, y_val, mask=co_mask[:, None] & t_mask[None, :])

    return (x + y - 1) // y
