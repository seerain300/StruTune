import math
import torch
import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def mask_apply_triton(x_ptr, mask_ptr, out_ptr, N, C, T,
                       x_sN, x_sC, x_sT,
                       m_sN, m_sC, m_sT,
                       out_sN, out_sC, out_sT,
                       BLOCK: tl.constexpr):
    """
    Elementwise: out[n, c, t] = x[n, c, t] * mask[n, 0, t]
    x: [N, C, T], mask: [N, 1, T], out: [N, C, T]
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    # Load mask[n, 0, t_idx] (broadcast along channels)
    m_offsets = pid_n * m_sN + 0 * m_sC + t_idx * m_sT
    mask_vals = tl.load(mask_ptr + m_offsets, mask=valid, other=1.0)

    # Load x[n, c, t_idx]
    x_offsets = pid_n * x_sN + pid_c * x_sC + t_idx * x_sT
    x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)

    # Multiply
    out_vals = x_vals * mask_vals

    # Store to out[n, c, t_idx]
    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid)


@triton.jit
def relu_triton(x_ptr, out_ptr, N, C, T,
                x_sN, x_sC, x_sT,
                out_sN, out_sC, out_sT,
                BLOCK: tl.constexpr):
    """
    Elementwise ReLU: out[n, c, t] = max(x[n, c, t], 0)
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    x_offsets = pid_n * x_sN + pid_c * x_sC + t_idx * x_sT
    x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)

    # ReLU
    x_vals = tl.maximum(x_vals, 0.0)

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, x_vals, mask=valid)


@triton.jit
def add_h_triton(x1_ptr, h_ptr, out_ptr, N, halfC, T,
                 x1_sN, x1_sC, x1_sT,
                 h_sN, h_sC, h_sT,
                 out_sN, out_sC, out_sT,
                 is_add: tl.constexpr,  # 1 for add, 0 for subtract
                 BLOCK: tl.constexpr):
    """
    Elementwise coupling on the second half channels:
    out[n, c, t] = x1[n, c, t] + h[n, c, t] if is_add else out[n, c, t] = x1[n, c, t] - h[n, c, t]
    x1 and out are [N, halfC, T]; h is [N, halfC, T] (result of conv2 after mask).
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    x_offsets = pid_n * x1_sN + pid_c * x1_sC + t_idx * x1_sT
    h_offsets = pid_n * h_sN + pid_c * h_sC + t_idx * h_sT
    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT

    x1_vals = tl.load(x1_ptr + x_offsets, mask=valid, other=0.0)
    h_vals = tl.load(h_ptr + h_offsets, mask=valid, other=0.0)

    if is_add:
        out_vals = x1_vals + h_vals
    else:
        out_vals = x1_vals - h_vals

    tl.store(out_ptr + out_offsets, out_vals, mask=valid)


@triton.jit
def final_mask_scale_triton(x_ptr, mask_ptr, out_ptr, N, C, T,
                            x_sN, x_sC, x_sT,
                            m_sN, m_sC, m_sT,
                            out_sN, out_sC, out_sT,
                            BLOCK: tl.constexpr):
    """
    Scale entire tensor by mask along time: out[n, c, t] = x[n, c, t] * mask[n, 0, t]
    x: [N, C, T], mask: [N, 1, T], out: [N, C, T]
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    m_offsets = pid_n * m_sN + 0 * m_sC + t_idx * m_sT
    scale = tl.load(mask_ptr + m_offsets, mask=valid, other=1.0)

    x_offsets = pid_n * x_sN + pid_c * x_sC + t_idx * x_sT
    x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)

    out_vals = x_vals * scale

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid)

         transform_3_conv1_weight, transform_3_conv1_bias,

                                                            h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),

                                                                   is_add=1 if not reverse else 0,
                                                                   BLOCK=256, num_warps=4)

            # concatenate halves
            x = torch.cat([x0, out_x1], dim=1)  # [N, 192, T]

            # Apply mask to the full output (broadcast along channels)
            x = x.clone()  # ensure we don't modify input x

                                                                   x_scaled.stride(0), x_scaled.stride(1), x_scaled.stride(2),
                                                                   BLOCK=256, num_warps=4)
            x = x_scaled
    else:
        for w0, b0, w1, b1, w2, b2 in reversed(transforms):
            x0 = x[:, :half_channels, :].contiguous()
            x1 = x[:, half_channels:, :].contiguous()

            # conv0
            y0 = torch.nn.functional.conv1d(x0, w0, b0, padding=2)
            y0_masked = torch.empty_like(y0)
            mask_apply_triton[(N, 192, triton.cdiv(T, 256))](y0, x_mask, y0_masked, N, 192, T,

                                                            BLOCK=256, num_warps=4)
            # ReLU
            y0_relu = torch.empty_like(y0_masked)
            relu_triton[(N, 192, triton.cdiv(T, 256))](y0_masked, y0_relu, N, 192, T,
                                                       y0_masked.stride(0), y0_masked.stride(1), y0_masked.stride(2),
                                                       y0_relu.stride(0), y0_relu.stride(1), y0_relu.stride(2),
                                                       BLOCK=256, num_warps=4)

            # conv1
            y = torch.nn.functional.conv1d(y0_relu, w1, b1, padding=2)
            y_masked = torch.empty_like(y)
            mask_apply_triton[(N, 192, triton.cdiv(T, 256))](y, x_mask, y_masked, N, 192, T,
                                                            y.stride(0), y.stride(1), y.stride(2),
                                                            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                            y_masked.stride(0), y_masked.stride(1), y_masked.stride(2),
                                                            BLOCK=256, num_warps=4)
            # ReLU
            y_relu = torch.empty_like(y_masked)
            relu_triton[(N, 192, triton.cdiv(T, 256))](y_masked, y_relu, N, 192, T,
                                                       y_masked.stride(0), y_masked.stride(1), y_masked.stride(2),
                                                       y_relu.stride(0), y_relu.stride(1), y_relu.stride(2),
                                                       BLOCK=256, num_warps=4)

            # conv2
            h = torch.nn.functional.conv1d(y_relu, w2, b2, padding=2)
            h_masked = torch.empty_like(h)
            mask_apply_triton[(N, 96, triton.cdiv(T, 256))](h, x_mask, h_masked, N, 96, T,
                                                            h.stride(0), h.stride(1), h.stride(2),
                                                            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                            h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
                                                            BLOCK=256, num_warps=4)

            # coupling: x1 = x1 - h
            out_x1 = torch.empty_like(x1)
            add_h_triton[(N, half_channels, triton.cdiv(T, 256))](x1, h_masked, out_x1, N, half_channels, T,
                                                                   x1.stride(0), x1.stride(1), x1.stride(2),
                                                                   h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
                                                                   out_x1.stride(0), out_x1.stride(1), out_x1.stride(2),
                                                                   is_add=0,  # subtract in reverse
                                                                   BLOCK=256, num_warps=4)

            # concatenate
            x = torch.cat([x0, out_x1], dim=1)

            # apply mask to the full output
            x_scaled = torch.empty_like(x)
            final_mask_scale_triton[(N, 192, triton.cdiv(T, 256))](x, x_mask, x_scaled, N, 192, T,
                                                                   x.stride(0), x.stride(1), x.stride(2),
                                                                   x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                                   x_scaled.stride(0), x_scaled.stride(1), x_scaled.stride(2),
                                                                   BLOCK=256, num_warps=4)
            x = x_scaled

    return x
