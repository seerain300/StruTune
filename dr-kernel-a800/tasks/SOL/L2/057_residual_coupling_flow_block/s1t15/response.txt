import math
import torch
import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def conv1d_triton_safe(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N, IC, OC, T, K, pad,
    x_sN, x_sC, x_sT,
    w_sO, w_sI, w_sK,
    out_sN, out_sC, out_sT,
):
    """
    Safe Triton conv1d (cross-correlation) for stride=1, padding=pad, dilation=1.
    One program instance computes the entire output vector for a single (n, oc).
    x: [N, IC, T] (float32)
    w: [OC, IC, K] (float32)
    b: [OC] (float32)
    out: [N, OC, T] (float32)
    Grid: (N, OC)
    Accumulate in float32 and store in out (same dtype as input).
    """
    pid_n = tl.program_id(0)
    pid_oc = tl.program_id(1)

    # accumulator for output vector length T
    acc = tl.zeros([T], dtype=tl.float32)

    # Loop over input channels and kernel taps
    for ic in range(0, IC):
        for k in range(0, K):
            t_in = tl.arange(0, T) + k - pad
            valid = (t_in >= 0) & (t_in < T)
            # Load x[pid_n, ic, t_in]
            x_offsets = pid_n * x_sN + ic * x_sC + t_in * x_sT
            x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)
            # Load w[pid_oc, ic, k]
            w_val = tl.load(w_ptr + pid_oc * w_sO + ic * w_sI + k * w_sK)
            acc += x_vals * w_val

    # Add bias
    b_val = tl.load(b_ptr + pid_oc)
    acc += b_val

    # Store to out[pid_n, pid_oc, :]
    out_offsets = pid_n * out_sN + pid_oc * out_sC + tl.arange(0, T) * out_sT
    tl.store(out_ptr + out_offsets, acc)


@triton.jit
def apply_mask_triton(x_ptr, mask_ptr, out_ptr, N, C, T,
                       mask_sN, mask_sC, mask_sT,
                       x_sN, x_sC, x_sT,
                       out_sN, out_sC, out_sT):
    """
    Multiply tensor x (N, C, T) by mask (N, 1, T), write to out.
    Broadcast mask over channels.
    Grid: (N, C, 1) to keep it simple; process one element at a time.
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    # read scalar mask for time t=0 (since mask[:, 0, :] is used)
    m = tl.load(mask_ptr + pid_n * mask_sN + 0 * mask_sC + 0 * mask_sT)
    # read x element
    x_val = tl.load(x_ptr + pid_n * x_sN + pid_c * x_sC + 0 * x_sT)
    # multiply
    y = x_val * m
    # store
    tl.store(out_ptr + pid_n * out_sN + pid_c * out_sC + 0 * out_sT, y)


@triton.jit
def relu_triton(x_ptr, out_ptr, N, C, T,
                x_sN, x_sC, x_sT,
                out_sN, out_sC, out_sT):
    """
    Elementwise ReLU for tensor (N, C, T), write to out.
    Grid: (N, C, 1) similar to mask.
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    x_val = tl.load(x_ptr + pid_n * x_sN + pid_c * x_sC + 0 * x_sT)
    y = tl.maximum(x_val, 0.0)
    tl.store(out_ptr + pid_n * out_sN + pid_c * out_sC + 0 * out_sT, y)


@triton.jit
def add_h_triton(x1_ptr, h_ptr, out_ptr, N, C, T,
                 x1_sN, x1_sC, x1_sT,
                 h_sN, h_sC, h_sT,
                 out_sN, out_sC, out_sT,
                 add_flag: tl.constexpr):
    """
    Elementwise add/subtract h to x1. If add_flag=True, out = x1 + h; else out = x1 - h.
    Grid: (N, C, 1).
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    x_val = tl.load(x1_ptr + pid_n * x1_sN + pid_c * x1_sC + 0 * x1_sT)
    h_val = tl.load(h_ptr + pid_n * h_sN + pid_c * h_sC + 0 * h_sT)
    if add_flag:
        y = x_val + h_val
    else:
        y = x_val - h_val
    tl.store(out_ptr + pid_n * out_sN + pid_c * out_sC + 0 * out_sT, y)

                       out_sN, out_sC, out_sT):

    launch enough programs to cover all elements via grid.
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t = tl.program_id(2)
    t = pid_t * 1  # since BLOCK_T=1, we just use t=pid_t; but we pass t=0 in launcher
    m = tl.load(mask_ptr + pid_n * mask_sN + 0 * mask_sC + 0 * mask_sT)
    x_val = tl.load(x_ptr + pid_n * x_sN + pid_c * x_sC + t * x_sT)
    y = x_val * m
    tl.store(out_ptr + pid_n * out_sN + pid_c * out_sC + t * out_sT, y)

    transform_1_conv0_weight, transform_1_conv0_bias,
    transform_1_conv1_weight, transform_1_conv1_bias,
    transform_1_conv2_weight, transform_1_conv2_bias,
    transform_2_conv0_weight, transform_2_conv0_bias,
    transform_2_conv1_weight, transform_2_conv1_bias,

    transform_3_conv2_weight, transform_3_conv2_bias,
):

    Triton-enhanced implementation that:
    - Uses torch.conv1d for convs (PyTorch/cuDNN is robust and fast).
    - Uses Triton for mask application, ReLU, and affine coupling.
    - Applies mask to the entire x after each transform, matching original behavior.
    """
    N, C, T = x.shape
    assert C == 192, "This implementation assumes C=192 for half_channels=96."
    half_channels = C // 2
    assert x_mask.shape == (N, 1, T), "x_mask must be [N, 1, T]"

    transforms = [
        (transform_0_conv0_weight, transform_0_conv0_bias,
         transform_0_conv1_weight, transform_0_conv1_bias,
         transform_0_conv2_weight, transform_0_conv2_bias),
        (transform_1_conv0_weight, transform_1_conv0_bias,
         transform_1_conv1_weight, transform_1_conv1_bias,
         transform_1_conv2_weight, transform_1_conv2_bias),
        (transform_2_conv0_weight, transform_2_conv0_bias,
         transform_2_conv1_weight, transform_2_conv1_bias,
         transform_2_conv2_weight, transform_2_conv2_bias),
        (transform_3_conv0_weight, transform_3_conv0_bias,
         transform_3_conv1_weight, transform_3_conv1_bias,
         transform_3_conv2_weight, transform_3_conv2_bias),
    ]

    for w0, b0, w1, b1, w2, b2 in (transforms if not reverse else reversed(transforms)):
        # Split current x into halves
        x0 = x[:, :half_channels, :].contiguous()  # [N, 96, T]
        x1 = x[:, half_channels:, :].contiguous()  # [N, 96, T]

        apply_mask_triton[(N, 192, 1)](
            y0, x_mask, y0_masked,
            N, 192, T,
            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
            y0.stride(0), y0.stride(1), y0.stride(2),
            y0_masked.stride(0), y0_masked.stride(1), y0_masked.stride(2),
        )
        y0_relu = torch.empty_like(y0_masked)
        relu_triton[(N, 192, 1)](
            y0_masked, y0_relu,
            N, 192, T,
            y0_masked.stride(0), y0_masked.stride(1), y0_masked.stride(2),
            y0_relu.stride(0), y0_relu.stride(1), y0_relu.stride(2),
        )

        # conv1: [N, 192, T]
        y1 = F.conv1d(y0_relu, w1, b1, padding=(w1.shape[2] // 2))

        # mask + ReLU on y1
        y1_masked = torch.empty_like(y1)
        apply_mask_triton[(N, 192, 1)](
            y1, x_mask, y1_masked,
            N, 192, T,
            x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
            y1.stride(0), y1.stride(1), y1.stride(2),
            y1_masked.stride(0), y1_masked.stride(1), y1_masked.stride(2),
        )
        y1_relu = torch.empty_like(y1_masked)
        relu_triton[(N, 192, 1)](
            y1_masked, y1_relu,
            N, 192, T,
            y1_masked.stride(0), y1_masked.stride(1), y1_masked.stride(2),
            y1_relu.stride(0), y1_relu.stride(1), y1_relu.stride(2),
        )

        # conv2: [N, 96, T]
        y2 = F.conv1d(y1_relu, w2, b2, padding=(w2.shape[2] // 2))

            x1, y2, x1_out,
            N, 96, T,

            y2.stride(0), y2.stride(1), y2.stride(2),

        )
