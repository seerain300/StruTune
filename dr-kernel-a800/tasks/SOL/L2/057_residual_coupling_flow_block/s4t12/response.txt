        PAD: tl.constexpr,            # padding (2)

        t_offsets = t_start + tl.arange(0, BLOCK_T)      # [BLOCK_T]

        t_mask = t_offsets < T_out
        mask = co_mask[:, None] & t_mask[None, :]

    import triton

    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: Conv1d (K=5, padding=2) + ReLU
# Inputs:
#   x_ptr: [N, C_in, T_in] contiguous
#   w_ptr: [C_out, C_in, K] contiguous
#   b_ptr: [C_out] contiguous
# Output:
#   y_ptr: [N, C_out, T_out] contiguous, T_out = T_in - 1
if TRITON_AVAILABLE:
    @triton.jit
    def conv1d_relu_fixed_kernel(
        x_ptr, w_ptr, b_ptr, y_ptr,
        N: tl.int32, C_in: tl.int32, T_in: tl.int32, C_out: tl.int32, T_out: tl.int32,
        K: tl.constexpr, PAD: tl.constexpr,
        BLOCK_CO: tl.constexpr, BLOCK_T: tl.constexpr
    ):
        pid_n = tl.program_id(0)      # batch index
        pid_co = tl.program_id(1)     # block along output channels
        pid_t = tl.program_id(2)      # block along output time

        co_start = pid_co * BLOCK_CO
        t_start = pid_t * BLOCK_T

        co_offsets = co_start + tl.arange(0, BLOCK_CO)   # [BLOCK_CO]
        t_offsets = t_start + tl.arange(0, BLOCK_T)      # [BLOCK_T]

        co_mask = co_offsets < C_out
        t_mask = t_offsets < T_out
        mask = co_mask[:, None] & t_mask[None, :]

        # accumulator [BLOCK_CO, BLOCK_T]
        acc = tl.zeros((BLOCK_CO, BLOCK_T), dtype=tl.float32)

        # loop over input channels and kernel taps
        for ci in range(0, C_in):
            for k in range(0, K):
                t_in = t_offsets + (k - PAD)               # [BLOCK_T]
                in_bounds = (t_in >= 0) & (t_in < T_in) & t_mask  # [BLOCK_T]

                # load x[n, ci, t_in] (vector along t)
                x_offs = ((pid_n * C_in + ci) * T_in) + t_in  # [BLOCK_T]
                x_vals = tl.load(x_ptr + x_offs, mask=in_bounds, other=0.0).to(tl.float32)

                # load weights w[co, ci, k] (vector along co)
                w_offs = co_offsets * (C_in * K) + ci * K + k  # [BLOCK_CO]
                w_vals = tl.load(w_ptr + w_offs, mask=co_mask, other=0.0).to(tl.float32)

                # outer product accumulate: [BLOCK_CO, 1] * [1, BLOCK_T]
                acc += w_vals[:, None] * x_vals[None, :]

        # add bias and ReLU
        b_vals = tl.load(b_ptr + co_offsets, mask=co_mask, other=0.0).to(tl.float32)
        acc = acc + b_vals[:, None]
        acc = tl.maximum(acc, 0.0)

        # store to y
        y_offs = ((pid_n * C_out + co_offsets[:, None]) * T_out) + t_offsets[None, :]
        tl.store(y_ptr + y_offs, acc, mask=mask)

# Triton kernel: concatenate two halves of channels into a single tensor y of size (N, 2*C_half, T_out)
# y[:, :C_half, :] = a, y[:, C_half:, :] = b
if TRITON_AVAILABLE:
    @triton.jit
    def concat_halves_triton(
        a_ptr, b_ptr, y_ptr,
        N: tl.int32, C_half: tl.int32, T: tl.int32, C_total: tl.int32,
        BLOCK_C: tl.constexpr, BLOCK_T: tl.constexpr
    ):
        pid_n = tl.program_id(0)
        pid_c = tl.program_id(1)
        pid_t = tl.program_id(2)

        c_start = pid_c * BLOCK_C
        t_start = pid_t * BLOCK_T

        c_offsets = c_start + tl.arange(0, BLOCK_C)
        t_offsets = t_start + tl.arange(0, BLOCK_T)

        total_mask = c_offsets < C_total
        t_mask = t_offsets < T
        mask = total_mask[:, None] & t_mask[None, :]

        # Write first half: a into y[:, :C_half, :]
        a_mask = (c_offsets < C_half)[:, None] & t_mask[None, :]
        a_offs = ((pid_n * C_half + c_offsets[:, None]) * T) + t_offsets[None, :]
        vals_a = tl.load(a_ptr + a_offs, mask=a_mask, other=0.0)
        y_offs_a = ((pid_n * C_total + c_offsets[:, None]) * T) + t_offsets[None, :]
        tl.store(y_ptr + y_offs_a, vals_a, mask=mask)

        # Write second half: b into y[:, C_half:, :]
        b_mask = (c_offsets >= C_half)[:, None] & t_mask[None, :]
        c_eff = c_offsets - C_half  # effective channel index in b
        b_mask_eff = (c_eff >= 0) & (c_eff < C_half) & t_mask
        b_offs = ((pid_n * C_half + c_eff[:, None]) * T) + t_offsets[None, :]
        vals_b = tl.load(b_ptr + b_offs, mask=b_mask_eff[:, None] & t_mask[None, :], other=0.0)
        y_offs_b = ((pid_n * C_total + c_offsets[:, None]) * T) + t_offsets[None, :]
        tl.store(y_ptr + y_offs_b, vals_b, mask=b_mask & t_mask[None, :])

# Triton kernel: ReLU elementwise on a 1D contiguous tensor
if TRITON_AVAILABLE:
    @triton.jit
    def relu_triton_kernel(
        in_ptr, out_ptr, numel: tl.int32, BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < numel
        vals = tl.load(in_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        vals = tl.maximum(vals, 0.0)
        tl.store(out_ptr + offs, vals, mask=mask)

# Triton kernel: elementwise add on a 1D contiguous tensor out = a + b
if TRITON_AVAILABLE:
    @triton.jit
    def add_triton_kernel(
        a_ptr, b_ptr, out_ptr, numel: tl.int32, BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < numel
        a = tl.load(a_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        out = a + b
        tl.store(out_ptr + offs, out, mask=mask)

# Triton kernel: elementwise multiply (mask) on a 1D contiguous tensor out = x * mask
# x: [N*C*T], mask: [N*C*T], broadcast over channels. In practice, mask is x_mask flattened.
if TRITON_AVAILABLE:
    @triton.jit
    def mask_mul_triton_kernel(
        x_ptr, mask_ptr, out_ptr, numel: tl.int32, BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < numel
        x = tl.load(x_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0).to(tl.float32)  # mask is 0/1
        out = x * m
        tl.store(out_ptr + offs, out, mask=mask)
