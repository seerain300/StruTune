@triton.jit
def conv1d_relu_triton(
    x_ptr,          # *float32, input [B, Cin, T_in]
    w_ptr,          # *float32, weight [Cout, Cin*K]
    bias_ptr,       # *float32, bias [Cout]
    out_ptr,        # *float32, output [B, Cout, T_out]
    B: tl.constexpr, Cin: tl.constexpr, T_in: tl.constexpr,
    Cout: tl.constexpr, K: tl.constexpr,
    x_stride_b, x_stride_c, x_stride_t,
    w_stride_c, w_stride_cin_k,
    out_stride_b, out_stride_c, out_stride_t,
    BLOCK_CO: tl.constexpr, BLOCK_POS: tl.constexpr,
    PAD: tl.constexpr,
):
    # program ids: (batch, channel-block, pos-block)
    b = tl.program_id(0)
    co_block = tl.program_id(1)
    pos_block = tl.program_id(2)

    co_offsets = co_block * BLOCK_CO + tl.arange(0, BLOCK_CO)
    t_offsets = pos_block * BLOCK_POS + tl.arange(0, BLOCK_POS)

    # Masks for bounds
    co_mask = co_offsets < Cout
    pos_mask = t_offsets < T_in  # T_out == T_in with padding symmetric

    # Accumulator for output across Cin and K
    acc = tl.zeros((BLOCK_CO, BLOCK_POS), dtype=tl.float32)

    # Iterate over input channels and kernel positions
    for ci in range(Cin):
        # For each kernel tap k in 0..K-1
        for k in range(K):
            t_src = t_offsets - PAD + k  # source time index
            # Valid if t_src in [0, T_in-1]
            valid_t = (t_src >= 0) & (t_src < T_in)

            # Load x[b, ci, t_src] with mask (masked as 0 for invalid)
            x_idx = b * x_stride_b + ci * x_stride_c + t_src * x_stride_t
            x_val = tl.load(x_ptr + x_idx, mask=valid_t, other=0.0)  # [BLOCK_POS]

            # Load w[co, ci, k] for all co in this block
            w_idx = co_offsets * w_stride_c + ci * w_stride_cin_k + k
            w_val = tl.load(w_ptr + w_idx, mask=co_mask, other=0.0)  # [BLOCK_CO]

            # Outer product accumulate
            acc += w_val[:, None] * x_val[None, :]

    # Add bias
    bias_val = tl.load(bias_ptr + co_offsets, mask=co_mask, other=0.0)  # [BLOCK_CO]
    acc += bias_val[:, None]

    # ReLU
    acc = tl.maximum(acc, 0.0)

    # Store to out[b, co, t_offsets] with masks
    out_idx = b * out_stride_b + co_offsets[:, None] * out_stride_c + t_offsets[None, :] * out_stride_t
    store_mask = co_mask[:, None] & pos_mask[None, :]
    tl.store(out_ptr + out_idx, acc, mask=store_mask)

        BLOCK_CO=block_co, BLOCK_POS=block_pos, PAD=pad,

def _mul_mask_to_h_triton(h, mask):

def _add_h_to_x1_triton(x1, h, add=True):
