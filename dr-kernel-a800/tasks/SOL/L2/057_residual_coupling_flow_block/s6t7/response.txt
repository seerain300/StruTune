@triton.jit
def conv1d_k5_p2(x_ptr, w_ptr, bias_ptr, y_ptr,
                 B: tl.constexpr, C_in: tl.constexpr, C_out: tl.constexpr,
                 T_in: tl.constexpr, T_out: tl.constexpr,
                 x_bs, x_cs, x_ts,
                 w_os, w_cs, w_ks,
                 y_bs, y_cs, y_ts,
                 BLOCK_T: tl.constexpr):
    """
    Conv1d with kernel_size=5, padding=2 (valid conv). Output length T_out = T_in - 1.
    Grid: (B, C_out, ceil(T_out / BLOCK_T))
    For each program: b, co, tile of time t_offsets.
    """
    b = tl.program_id(0)
    co = tl.program_id(1)
    pid_t = tl.program_id(2)
    t_offsets = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)
    t_mask = t_offsets < T_out

    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # Loop over input channels and kernel taps
    for ci in range(0, C_in):
        for k in range(0, 5):  # kernel_size = 5
            src_time = t_offsets - 2 + k  # padding=2
            # valid indices
            valid = (src_time >= 0) & (src_time < T_in)
            # load x[b, ci, src_time] with mask
            x_off = b * x_bs + ci * x_cs + src_time * x_ts
            # masked load
            x_val = tl.load(x_ptr + x_off, mask=valid & t_mask, other=0.0)
            # load w[co, ci, k]
            w_off = co * w_os + ci * w_cs + k * w_ks
            w_val = tl.load(w_ptr + w_off)
            acc += x_val * w_val

    # add bias
    b_val = tl.load(bias_ptr + co)
    acc = acc + b_val

    # store
    y_off = b * y_bs + co * y_cs + t_offsets * y_ts
    tl.store(y_ptr + y_off, acc, mask=t_mask)


@triton.jit
def add_bias(y_ptr, B: tl.constexpr, C: tl.constexpr, T: tl.constexpr,

@triton.jit

@triton.jit
def add_or_sub(y_ptr, addend_ptr, B: tl.constexpr, C: tl.constexpr, T: tl.constexpr,
               op_add: tl.constexpr, y_bs, y_cs, y_ts):
    """

    """

    b = pid_bc // C
    c = pid_bc % C
    t_offsets = tl.arange(0, 128)
    t_mask = t_offsets < T
    y_off = b * y_bs + c * y_cs + t_offsets * y_ts
    y_val = tl.load(y_ptr + y_off, mask=t_mask, other=0.0)
    add_off = b * y_bs + c * y_cs + t_offsets * y_ts
    add_val = tl.load(addend_ptr + add_off, mask=t_mask, other=0.0)
    if op_add:
        y_val = y_val + add_val
    else:
        y_val = y_val - add_val
    tl.store(y_ptr + y_off, y_val, mask=t_mask)


@triton.jit
def copy_to(dst_ptr, src_ptr, B: tl.constexpr, C: tl.constexpr, T: tl.constexpr,
            dst_bs, dst_cs, dst_ts, src_bs, src_cs, src_ts):

    b = tl.program_id(0)

    pid_t = tl.program_id(2)

    src_off = b * src_bs + c * src_cs + t_offsets * src_ts
    dst_off = b * dst_bs + c * dst_cs + t_offsets * dst_ts
    val = tl.load(src_ptr + src_off, mask=t_mask, other=0.0)
