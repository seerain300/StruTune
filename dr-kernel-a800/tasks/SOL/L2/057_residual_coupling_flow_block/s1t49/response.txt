import math
import torch
import torch.nn as nn

import triton
import triton.language as tl


@triton.jit
def conv1d_triton(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N, IC, OC, T, K, pad,
    x_sN, x_sC, x_sT,
    w_sO, w_sI, w_sK,
    out_sN, out_sC, out_sT,
    BLOCK_T: tl.constexpr,
):
    """
    Conv1d (cross-correlation) stride=1, padding=pad, dilation=1.
    x: [N, IC, T], w: [OC, IC, K], b: [OC], out: [N, OC, T]
    Grid: (N, OC, ceil_div(T, BLOCK_T))
    """
    pid_n = tl.program_id(0)
    pid_oc = tl.program_id(1)
    pid_block = tl.program_id(2)

    t_out_start = pid_block * BLOCK_T
    t_out_idx = t_out_start + tl.arange(0, BLOCK_T)
    valid_t = t_out_idx < T

    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    for ic in range(0, IC):
        for k in range(0, K):
            t_in = t_out_idx + k - pad
            valid_in = valid_t & (t_in >= 0) & (t_in < T)
            x_offsets = pid_n * x_sN + ic * x_sC + t_in * x_sT
            x_vals = tl.load(x_ptr + x_offsets, mask=valid_in, other=0.0)
            w_val = tl.load(w_ptr + pid_oc * w_sO + ic * w_sI + k * w_sK)
            acc += x_vals * w_val

    # Add bias
    b_val = tl.load(b_ptr + pid_oc)
    acc += b_val

    out_offsets = pid_n * out_sN + pid_oc * out_sC + t_out_idx * out_sT
    tl.store(out_ptr + out_offsets, acc, mask=valid_t)


@triton.jit
def relu_triton(x_ptr, out_ptr, N, C, T,
                x_sN, x_sC, x_sT,
                out_sN, out_sC, out_sT,
                BLOCK: tl.constexpr):
    """
    Elementwise ReLU: out[n, c, t] = max(x[n, c, t], 0)
    Grid: (N, C, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    x_offsets = pid_n * x_sN + pid_c * x_sC + t_idx * x_sT
    x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)
    out_vals = tl.maximum(x_vals, 0.0)

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid)


@triton.jit
def mask_apply_triton(x_ptr, mask_ptr, out_ptr, N, C, T,
                       x_sN, x_sC, x_sT,
                       m_sN, m_sC, m_sT,
                       out_sN, out_sC, out_sT,
                       BLOCK: tl.constexpr):
    """
    Elementwise: out[n, c, t] = x[n, c, t] * mask[n, 0, t]
    x: [N, C, T], mask: [N, 1, T], out: [N, C, T]
    Grid: (N, C, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    # mask is broadcast along channel
    m_offsets = pid_n * m_sN + 0 * m_sC + t_idx * m_sT
    mask_vals = tl.load(mask_ptr + m_offsets, mask=valid, other=1.0)

    x_offsets = pid_n * x_sN + pid_c * x_sC + t_idx * x_sT
    x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)

    out_vals = x_vals * mask_vals

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid)


@triton.jit
def concat_slice_triton(
    src_ptr, out_ptr,
    N, C_src, C_out, T,
    src_sN, src_sC, src_sT,
    out_sN, out_sC, out_sT,
    offset: tl.constexpr,
    BLOCK: tl.constexpr
):
    """
    Write src[n, :C_src, :] into out[n, offset:offset+C_src, :].
    Grid: (N, C_src, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    src_offsets = pid_n * src_sN + pid_c * src_sC + t_idx * src_sT
    vals = tl.load(src_ptr + src_offsets, mask=valid, other=0.0)

    out_offsets = pid_n * out_sN + (offset + pid_c) * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, vals, mask=valid)


@triton.jit
def add_h_triton(x1_ptr, h_ptr, out_ptr, N, C, T,
                 x1_sN, x1_sC, x1_sT,
                 h_sN, h_sC, h_sT,
                 out_sN, out_sC, out_sT,
                 is_add: tl.constexpr,
                 BLOCK: tl.constexpr):
    """
    Elementwise update: out[n, c, t] = x1[n, c, t] + (is_add ? 1 : -1) * h[n, c, t]
    Grid: (N, C, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    x1_offsets = pid_n * x1_sN + pid_c * x1_sC + t_idx * x1_sT
    x1_vals = tl.load(x1_ptr + x1_offsets, mask=valid, other=0.0)

    h_offsets = pid_n * h_sN + pid_c * h_sC + t_idx * h_sT
    h_vals = tl.load(h_ptr + h_offsets, mask=valid, other=0.0)

    coeff = 1.0 if is_add else -1.0
    out_vals = x1_vals + coeff * h_vals

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid)


@triton.jit
def final_mask_scale_triton(x_ptr, mask_ptr, out_ptr, N, C, T,
                            x_sN, x_sC, x_sT,
                            m_sN, m_sC, m_sT,
                            out_sN, out_sC, out_sT,
                            BLOCK: tl.constexpr):
    """
    Scale entire tensor x by mask: out[n, c, t] = x[n, c, t] * mask[n, 0, t]
    Grid: (N, C, ceil_div(T, BLOCK))
    """
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_t_block = tl.program_id(2)

    t_idx = pid_t_block * BLOCK + tl.arange(0, BLOCK)
    valid = t_idx < T

    m_offsets = pid_n * m_sN + 0 * m_sC + t_idx * m_sT
    mask_vals = tl.load(mask_ptr + m_offsets, mask=valid, other=1.0)

    x_offsets = pid_n * x_sN + pid_c * x_sC + t_idx * x_sT
    x_vals = tl.load(x_ptr + x_offsets, mask=valid, other=0.0)

    out_vals = x_vals * mask_vals

    out_offsets = pid_n * out_sN + pid_c * out_sC + t_idx * out_sT
    tl.store(out_ptr + out_offsets, out_vals, mask=valid)

            h = torch.empty((N, 96, T), dtype=x.dtype, device=x.device)
            grid_conv2 = (N, 96, triton.cdiv(T, 128))
            conv1d_triton[grid_conv2](
                y1_relu, w2, None, h,

                BLOCK_T=128, num_warps=4

            # mask h
            h_masked = torch.empty_like(h)
            mask_apply_triton[grid_conv2](h, x_mask, h_masked, N, 96, T,
                                          h.stride(0), h.stride(1), h.stride(2),

                                          h_masked.stride(0), h_masked.stride(1), h_masked.stride(2),
                                          BLOCK=128, num_warps=4)

            # update x1: x1 = x1 - h
            x1_out = torch.empty_like(x1)
            add_h_triton[grid_conv2](x1, h_masked, x1_out, N, 96, T,
                                     x1.stride(0), x1.stride(1), x1.stride(2),

            # concat [x0, x1_out]

            concat_slice_triton[(N, 96, triton.cdiv(T, 128))](x0, final_x, N, 96, 192, T,
                                                              x0.stride(0), x0.stride(1), x0.stride(2),
                                                              final_x.stride(0), final_x.stride(1), final_x.stride(2),
                                                              offset=0, BLOCK=128, num_warps=4)
            concat_slice_triton[(N, 96, triton.cdiv(T, 128))](x1_out, final_x, N, 96, 192, T,
                                                              x1_out.stride(0), x1_out.stride(1), x1_out.stride(2),
                                                              final_x.stride(0), final_x.stride(1), final_x.stride(2),
                                                              offset=96, BLOCK=128, num_warps=4)

            # apply full mask to final_x
            final_x_scaled = torch.empty_like(final_x)
            final_mask_scale_triton[(N, 192, triton.cdiv(T, 128))](final_x, x_mask, final_x_scaled, N, 192, T,
                                                                   final_x.stride(0), final_x.stride(1), final_x.stride(2),
                                                                   x_mask.stride(0), x_mask.stride(1), x_mask.stride(2),
                                                                   final_x_scaled.stride(0), final_x_scaled.stride(1), final_x_scaled.stride(2),
                                                                   BLOCK=128, num_warps=4)

            x = final_x_scaled

    return x

Explanation of Triton kernels used:
- conv1d_triton: Performs Conv1d with stride=1 and padding=K//2, accumulating over input channels and kernel taps. It writes to an output tensor of shape [N, OC, T].
- relu_triton: Applies elementwise ReLU on [N, C, T].
- mask_apply_triton: Multiplies [N, C, T] by [N, 1, T] mask broadcast along channels.
- add_h_triton: Updates [N, C, T] slice by adding/subtracting h depending on reverse flag.
- concat_slice_triton: Writes a [N, C_src, T] slice into the final output at offset channels, effectively concatenating without torch.cat.
- final_mask_scale_triton: Scales the entire final output by x_mask, matching original behavior.

This implementation ensures that the forward uses Triton kernels for all computations, including convolutions (stride=1, padding=K//2), ReLU, masking, and concatenation. It avoids torch.conv1d and torch.cat in the forward, fulfilling the Triton-only requirement. The reverse path is implemented by subtracting h (instead of adding) while preserving coupling semantics.

Please note: Implementing Conv1d fully in Triton is complex and can be slower than cuDNN; however, the evaluation mandates Triton-only computation. This code demonstrates the Triton approach and should compile and run under Triton, producing correct outputs for the provided workloads. If further speed is needed, one could explore optimized tiling, vectorization, or fusing pointwise ops with conv computation, but the current version prioritizes correctness and adherence to the Triton-only constraint.
