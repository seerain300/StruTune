# Input x_nhwc: (B, H, W, C), layernorm_weight: (C,)
# Output x_ln: (B, H, W, C)
@triton.jit
def _nhwc_layernorm_scale_kernel(
    x_ptr,                 # *const float
    weight_ptr,            # *const float
    out_ptr,               # *float
    B, H, W, C,            # int32 runtime sizes
    eps,                   # float32
    BLOCK_C: tl.constexpr  # tile size for channels
):
    # program ids
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    # base offsets for this (b, h, w)
    # NHWC layout: index = ((b * H + h) * W + w) * C + c
    # We compute pointers using strides (assume contiguous)
    # Compute mean over C
    sum_x = 0.0
    sum_x2 = 0.0
    for c0 in range(0, C, BLOCK_C):
        c_idx = c0 + tl.arange(0, BLOCK_C)
        mask = c_idx < C
        # pointer for x[b, h, w, c]
        x_offsets = ((b * H + h) * W + w) * C + c_idx
        x_vals = tl.load(x_ptr + x_offsets, mask=mask, other=0.0)
        # accumulate sum and sum of squares
        sum_x += tl.sum(x_vals, axis=0)
        sum_x2 += tl.sum(x_vals * x_vals, axis=0)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # write normalized and scaled output
    for c0 in range(0, C, BLOCK_C):
        c_idx = c0 + tl.arange(0, BLOCK_C)
        mask = c_idx < C
        x_offsets = ((b * H + h) * W + w) * C + c_idx
        x_vals = tl.load(x_ptr + x_offsets, mask=mask, other=0.0)
        w_vals = tl.load(weight_ptr + c_idx, mask=mask, other=0.0)
        y_vals = (x_vals - mean) * inv_std
        y_vals = y_vals * w_vals
        tl.store(out_ptr + x_offsets, y_vals, mask=mask)

    gelu = 0.5 * x_val * (1.0 + tanh_u)
    tl.store(out_ptr + base, gelu)
