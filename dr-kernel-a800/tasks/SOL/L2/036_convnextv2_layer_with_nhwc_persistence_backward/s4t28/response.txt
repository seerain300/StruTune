Below is ModelNew where all computation is performed via Triton kernels. The entry point forward launches the necessary Triton kernels in the correct order, avoiding any torch computation in the host code.

1) Depthwise Conv2d (groups=C)
- One Triton program per output element y[b, c, oh, ow]
- Loop over 7x7 kernel taps
- Compute input indices with padding=3 and masked loads
- Multiply with per-channel weight and accumulate
- Store to output with explicit strides

2) Permute NCHW -> NHWC
- One Triton program per output element in NHWC layout
- Compute source index in NCHW and copy value

3) LayerNorm across C per (b, h, w)
- Triton kernel to compute per-(b, h, w) sum and sum of squares across C
- Triton kernel to normalize and scale by layernorm_weight, storing back in NCHW

4) Linear projection (GEMV) x_ln @ W.T to 4C
- Triton kernel to compute x_expanded[b, c_out] by looping over input channels C and accumulating dot products with W[c_out, c_in]

5) GELU (tanh approximation) elementwise
- Triton kernel to apply GELU on x_expanded

6) GRN
- Triton reduction kernel to compute per-(b, c_out) spatial norm over H and W
- Triton normalization kernel to compute norm_features = global_feat / (gf_mean + eps), and update x_gelu

import torch
import triton
import triton.language as tl


@triton.jit
def depthwise_conv2d_groupsC_per_output_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    # One program per output element (b, c, oh, ow)
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    acc = 0.0  # float32 accumulation

    # Loop over 7x7 kernel
    for kh in range(7):
        for kw in range(7):
            ih = oh + kh - pad_h
            iw = ow + kw - pad_w
            # Masked load for valid input coordinates
            in_bounds = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
            x_offset = b * stride_xB + c * stride_xC + ih * stride_xH + iw * stride_xW
            # Safe load with mask; if out of bounds, value is 0
            x_val = tl.load(x_ptr + x_offset, mask=in_bounds, other=0.0)

            # Load per-channel weight
            w_offset = c * stride_wC + kh * stride_wKH + kw * stride_wKW
            w_val = tl.load(w_ptr + w_offset)

            acc += x_val * w_val

    # Store result
    y_offset = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_offset, acc)


@triton.jit
def permute_nchw_to_nhwc_kernel(
    x_nchw_ptr, x_nhwc_ptr,
    B, C, H, W,

    stride_yB, stride_yC, stride_yH, stride_yW,
):
    # One program per output element in NHWC layout
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)
    c = tl.program_id(3)

    # Map to NCHW input index
    x_offset = b * stride_xB + c * stride_xC + h * stride_xH + w * stride_xW

    val = tl.load(x_nchw_ptr + x_offset)
    tl.store(x_nhwc_ptr + nhwc_offset, val)


@triton.jit
def layernorm_reduce_sums_sumsq_kernel(
    x_ptr, sum_ptr, sumsq_ptr,
    B, H, W, C,
    stride_xB, stride_xC, stride_xH, stride_xW,
):
    # One program per (b, h, w), reduce over C
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    acc_sum = 0.0
    acc_sumsq = 0.0

    for c in range(C):
        x_offset = b * stride_xB + c * stride_xC + h * stride_xH + w * stride_xW
        val = tl.load(x_ptr + x_offset)
        acc_sum += val
        acc_sumsq += val * val

    tl.store(sum_ptr + b * H * W + h * W + w, acc_sum)
    tl.store(sumsq_ptr + b * H * W + h * W + w, acc_sumsq)


@triton.jit
def layernorm_normalize_scale_kernel(
    x_ptr, sum_ptr, sumsq_ptr, weight_ptr, y_ptr,
    B, H, W, C, eps,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    # One program per (b, h, w), normalize across C and scale by weight
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    sum_val = tl.load(sum_ptr + b * H * W + h * W + w)
    sumsq_val = tl.load(sumsq_ptr + b * H * W + h * W + w)
    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    std = tl.sqrt(var + eps)

    for c in range(C):
        x_offset = b * stride_xB + c * stride_xC + h * stride_xH + w * stride_xW
        val = tl.load(x_ptr + x_offset)
        norm = (val - mean) / std
        weight = tl.load(weight_ptr + c)
        y_offset = b * stride_yB + c * stride_yC + h * stride_yH + w * stride_yW
        tl.store(y_ptr + y_offset, norm * weight)

@triton.jit
def gelu_tanh_kernel(
    x_ptr, y_ptr,
    B, C,
    stride_xB, stride_xC,
    stride_yB, stride_yC,
):
    # One program per element (b, c)
    b = tl.program_id(0)
    c = tl.program_id(1)
    x_offset = b * stride_xB + c * stride_xC
    x_val = tl.load(x_ptr + x_offset)
    # GELU tanh approximation: 0.5 * x * (1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    sqrt_2_over_pi = 0.7978845608028654
    x3 = x_val * x_val * x_val
    inner = sqrt_2_over_pi * (x_val + 0.044715 * x3)
    tanh_inner = tl.tanh(inner)
    y_val = 0.5 * x_val * (1.0 + tanh_inner)
    y_offset = b * stride_yB + c * stride_yC
    tl.store(y_ptr + y_offset, y_val)


@triton.jit
def grn_reduce_global_l2_kernel(
    x_ptr, norm_ptr,
    B, C, H, W,
    stride_xB, stride_xC, stride_xH, stride_xW,
):
    # One program per (b, c): compute L2 over H and W
    b = tl.program_id(0)
    c = tl.program_id(1)
    acc = 0.0

            val = tl.load(x_ptr + x_offset)
            acc += val * val
    norm = tl.sqrt(acc)
    tl.store(norm_ptr + b * C + c, norm)
