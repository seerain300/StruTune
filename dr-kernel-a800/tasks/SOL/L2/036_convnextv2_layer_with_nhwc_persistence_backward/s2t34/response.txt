Final diagnosis and plan

- The evaluator keeps reporting 0/14 workloads correct with “RUNTIME_ERROR.” This indicates Triton kernel launch issues or illegal memory access, not just decoy kernels.
- A common source of Triton runtime errors in this context:
  - Incorrect indexing into tensors with dynamic strides. Triton kernels must assume contiguous memory unless strides are explicitly passed and used. If we assume a contiguous layout but the tensor is not contiguous (or has non-standard strides), kernel writes/reads can access invalid memory.
  - Passing runtime values to tl.constexpr (kernel meta-parameters) or using Python range with dynamic bounds inside kernels can cause compilation/runtime problems.
- Strategy to fix:
  - Ensure all tensors passed to Triton are contiguous and float32 on CUDA. Make them contiguous explicitly before launching.
  - Pass strides explicitly to Triton kernels so we can correctly index into NCHW and NHWC layouts without assuming contiguity. Even if tensors are contiguous, using strides makes the kernels robust if someone passes non-contiguous.
  - Use only Triton for the two computationally meaningful parts:
    - NHWC LayerNorm-like scaling: For each (b,h,w), reduce across channels C to compute mean and variance; then write x_ln = (x - mean) / sqrt(var + eps) * layernorm_weight[c].
    - GELU (tanh approximation) on NCHW: Elementwise on x_expanded (B,C,H,W).
  - Invoke Triton kernels unconditionally from ModelNew.forward; do not perform any torch elementwise ops in forward.

- Performance considerations:
  - The NHWC kernel uses a 3D grid (B, H, W). Each program handles a single (b,h,w) and reduces over channels C in chunks (BLOCK_C). This is simple and robust. For typical C=128, this reduction is light.
  - GELU kernel uses 4D grid (B, C, H, W), elementwise, efficient in Triton.

import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def _nhwc_layernorm_scale_kernel(
    x_ptr,            # *float32, input NHWC: (B, H, W, C), potentially non-contiguous
    lnw_ptr,          # *float32, layernorm weight: (C,)
    y_ptr,            # *float32, output NHWC: (B, H, W, C)
    B, H, W, C, eps,  # runtime integers
    x_sN, x_sH, x_sW, x_sC,   # strides for x
    y_sN, y_sH, y_sW, y_sC,   # strides for y
    BLOCK_C: tl.constexpr,
):
    # Program IDs: each program handles one (b, h, w)
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    # Base offset for this (b, h, w) row in NHWC layout
    # NHWC strides: offset = b*x_sN + h*x_sH + w*x_sW + c*x_sC
    base = b * x_sN + h * x_sH + w * x_sW

    # Compute sum and sum of squares across channels
    sum_val = 0.0
    sum_sq = 0.0
    # Loop over channels in chunks of BLOCK_C
    for c_start in range(0, C, BLOCK_C):
        c_offsets = c_start + tl.arange(0, BLOCK_C)
        mask = c_offsets < C
        # Load x values for this (b, h, w) across c_offsets
        x_vals = tl.load(x_ptr + base + c_offsets * x_sC, mask=mask, other=0.0)
        # Reduce to scalars
        sum_val += tl.sum(x_vals, axis=0)
        sum_sq += tl.sum(x_vals * x_vals, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: write normalized and scaled output
    for c_start in range(0, C, BLOCK_C):
        c_offsets = c_start + tl.arange(0, BLOCK_C)
        mask = c_offsets < C
        x_vals = tl.load(x_ptr + base + c_offsets * x_sC, mask=mask, other=0.0)
        lnw_vals = tl.load(lnw_ptr + c_offsets, mask=mask, other=1.0)
        y_vals = (x_vals - mean) * inv_std * lnw_vals
        tl.store(y_ptr + base + c_offsets * y_sC, y_vals, mask=mask)


@triton.jit
def _gelu_nchw_kernel(
    x_ptr,            # *float32, input NCHW: (B, C, H, W)
    y_ptr,            # *float32, output NCHW: (B, C, H, W)
    B, C, H, W,       # runtime integers
    x_sN, x_sC, x_sH, x_sW,  # strides for x
    y_sN, y_sC, y_sH, y_sW,  # strides for y
    BLOCK_W: tl.constexpr,
):
    # Grid: (B, C, H, ceil_div(W, BLOCK_W))
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    w_block = tl.program_id(3)
    w_start = w_block * BLOCK_W
    w_offsets = w_start + tl.arange(0, BLOCK_W)
    mask = w_offsets < W

    base = b * x_sN + c * x_sC + h * x_sH
    x_vals = tl.load(x_ptr + base + w_offsets * x_sW, mask=mask, other=0.0)

    # GELU tanh approximation
    sqrt_2_over_pi = 0.7978845608028654  # sqrt(2/pi)
    x_cubed = x_vals * x_vals * x_vals
    u = sqrt_2_over_pi * (x_vals + 0.044715 * x_cubed)
    e2u = tl.exp(2.0 * u)
    tanh_u = (e2u - 1.0) / (e2u + 1.0)
    gelu_vals = 0.5 * x_vals * (1.0 + tanh_u)

    tl.store(y_ptr + base + w_offsets * y_sW, gelu_vals, mask=mask)


def _triton_nhwc_scale(x_nhwc: torch.Tensor, layernorm_weight: torch.Tensor, eps: float) -> torch.Tensor:
    """
    Triton implementation of NHWC LayerNorm-like scaling:
    For each (b, h, w), compute mean and var across channels C, then
    y = (x - mean) / sqrt(var + eps) * layernorm_weight[c].
    x_nhwc: (B, H, W, C), layernorm_weight: (C,)
    Returns y with same shape and dtype.
    """
    assert x_nhwc.is_cuda and layernorm_weight.is_cuda, "Triton kernels require CUDA tensors"
    assert x_nhwc.dtype == torch.float32 and layernorm_weight.dtype == torch.float32, "Use float32 tensors"
    B, H, W, C = x_nhwc.shape
    y = torch.empty_like(x_nhwc)
    # Strides
    x_sN, x_sH, x_sW, x_sC = x_nhwc.stride()
    y_sN, y_sH, y_sW, y_sC = y.stride()
    # Launch kernel: grid = (B, H, W)
    grid = (B, H, W)
    _nhwc_layernorm_scale_kernel[grid](
        x_nhwc, layernorm_weight, y,
        B, H, W, C, eps,
        x_sN, x_sH, x_sW, x_sC,
        y_sN, y_sH, y_sW, y_sC,
        BLOCK_C=64,  # chunk size over channels
    )
    return y


def _triton_gelu_nchw(x_expanded: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of GELU (tanh approximation) on NCHW tensor.
    x_expanded: (B, C, H, W), returns same shape and dtype.
    """
    assert x_expanded.is_cuda, "Triton kernels require CUDA tensors"
    assert x_expanded.dtype == torch.float32, "Use float32 tensors"
    B, C, H, W = x_expanded.shape
    y = torch.empty_like(x_expanded)
    # Strides
    x_sN, x_sC, x_sH, x_sW = x_expanded.stride()
    y_sN, y_sC, y_sH, y_sW = y.stride()
    # Launch kernel: grid = (B, C, H, ceil_div(W, BLOCK_W))
    grid = (B, C, H, triton.cdiv(W, 32))
    _gelu_nchw_kernel[grid](
        x_expanded, y,
        B, C, H, W,
        x_sN, x_sC, x_sH, x_sW,
        y_sN, y_sC, y_sH, y_sW,
        BLOCK_W=32,  # tile along W
    )
    return y
