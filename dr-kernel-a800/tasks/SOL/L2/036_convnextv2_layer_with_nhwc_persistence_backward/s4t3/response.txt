import triton
import triton.language as tl


@triton.jit
def depthwise_conv2d_groupsC_per_output_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    # Grid: (B, C, H_out, W_out) -> one program per output element
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    # Accumulator in float32
    acc = 0.0

    # Loop over 7x7 kernel taps
    for kh in range(7):
        for kw in range(7):
            ih = oh + pad_h - kh  # adjusted for padding
            iw = ow + pad_w - kw
            # Check if (ih, iw) is inside the original input (H, W)
            in_bounds = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)

            # Compute input offset using explicit strides
