@triton.jit
def depthwise_conv2d_groupsC_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    """
    Compute depthwise conv2d with groups=C:
    y[b, c, oh, ow] = sum_{kh, kw} x[b, c, oh+kh, ow+kw] * w[c, 0, kh, kw]
    padding = pad_h = pad_w = 3
    """
    # program ids: (b, c, oh, ow)
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    # Initialize accumulator in fp32
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over 7x7 kernel
    for kh in range(7):
        for kw in range(7):
            ih = oh + kh - pad_h
            iw = ow + kw - pad_w
            # bounds check
            in_bounds = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
            x_offset = b * stride_xB + c * stride_xC + ih * stride_xH + iw * stride_xW
            # safe load; other sets default value (we'll guard multiply by in_bounds anyway)
            x_val = tl.load(x_ptr + x_offset, mask=in_bounds, other=0.0)
            # weight is scalar per (c, kh, kw)
            w_offset = c * stride_wC + kh * stride_wKH + kw * stride_wKW
            w_val = tl.load(w_ptr + w_offset)
            # accumulate
            acc += x_val * w_val

    # Store result
    y_offset = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_offset, acc)


@triton.jit
def reduce_sum_channels_kernel(
    x_ptr, out_ptr,

):

        total += x_val

import torch
import triton
import triton.language as tl


@triton.jit
def depthwise_conv2d_groupsC_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    """
    Depthwise conv2d with groups=C:
    y[b, c, oh, ow] = sum_{kh,kw} x[b, c, oh+kh, ow+kw] * w[c, 0, kh, kw]
    padding = pad_h = pad_w = 3
    """
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    acc = tl.zeros((), dtype=tl.float32)
    for kh in range(7):
        for kw in range(7):
            ih = oh + kh - pad_h
            iw = ow + kw - pad_w
            in_bounds = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
            x_offset = b * stride_xB + c * stride_xC + ih * stride_xH + iw * stride_xW
            x_val = tl.load(x_ptr + x_offset, mask=in_bounds, other=0.0)
            w_offset = c * stride_wC + kh * stride_wKH + kw * stride_wKW
            w_val = tl.load(w_ptr + w_offset)
            acc += x_val * w_val

    y_offset = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_offset, acc)


@triton.jit
def reduce_sum_channels_kernel(
    x_ptr, out_ptr,
    B, H, W, C,
    stride_xB, stride_xH, stride_xW, stride_xC,
    stride_outB, stride_outH, stride_outW,
):
    """
    Sum over channels for NHWC tensor: out[b, h, w] = sum_c x[b, h, w, c]
    """
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)
    total = tl.zeros((), dtype=tl.float32)
    for c in range(0, C):
        x_offset = b * stride_xB + h * stride_xH + w * stride_xW + c * stride_xC
        x_val = tl.load(x_ptr + x_offset)
        total += x_val
    out_offset = b * stride_outB + h * stride_outH + w * stride_outW
    tl.store(out_ptr + out_offset, total)

@triton.jit
def reduce_sum_spatial_kernel(
    x_ptr, out_ptr,
    B, C, H, W,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_outB, stride_outC,
):
    """
    Sum over spatial dims (H, W) for NCHW tensor per (b, c): out[b, c] = sum_{h,w} x[b, c, h, w]^2
    """
    b = tl.program_id(0)
    c = tl.program_id(1)
    total = tl.zeros((), dtype=tl.float32)
    for h in range(0, H):
        for w in range(0, W):
            x_offset = b * stride_xB + c * stride_xC + h * stride_xH + w * stride_xW
            x_val = tl.load(x_ptr + x_offset)
            total += x_val * x_val
    out_offset = b * stride_outB + c * stride_outC
    tl.store(out_ptr + out_offset, total)
