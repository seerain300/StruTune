import torch
import triton
import triton.language as tl


# Triton kernel: NHWC LayerNorm-like scaling
# Input: x_nhwc  [B, H, W, C], float32, contiguous
# Weight: layernorm_weight [C], float32
# Output: x_ln  [B, H, W, C], float32
@triton.jit
def _nhwc_layernorm_scale_kernel(
    x_ptr,           # *const float
    layernorm_ptr,   # *const float
    out_ptr,         # *float
    B: tl.constexpr,
    H: tl.constexpr,
    W: tl.constexpr,
    C: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    # Safety: if any program_id out of bounds, return (Triton masks usually sufficient)
    # Compute sum and sum of squares over C for this (b, h, w)
    sum_x = 0.0
    sum_x2 = 0.0
    for c0 in range(0, C, BLOCK_C):
        c_idx = c0 + tl.arange(0, BLOCK_C)
        mask = c_idx < C
        # Linear index for x[b, h, w, c]
        idx = (((b * H) + h) * W + w) * C + c_idx
        x = tl.load(x_ptr + idx, mask=mask, other=0.0)
        # Upcast to float32 for reduction
        x = x.to(tl.float32)
        sum_x += tl.sum(x, axis=0)
        sum_x2 += tl.sum(x * x, axis=0)
    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Write normalized and scaled output: out = (x - mean) * inv_std * layernorm_weight[c]
    for c0 in range(0, C, BLOCK_C):
        c_idx = c0 + tl.arange(0, BLOCK_C)
        mask = c_idx < C
        idx = (((b * H) + h) * W + w) * C + c_idx
        x = tl.load(x_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        norm = (x - mean) * inv_std
        wgt = tl.load(layernorm_ptr + c_idx, mask=mask, other=0.0).to(tl.float32)
        out = norm * wgt
        tl.store(out_ptr + idx, out, mask=mask)


# Triton kernel: GELU (tanh approximation) on NCHW tensor
# Input: x_expanded [B, C, H, W], float32, contiguous
# Output: x_gelu [B, C, H, W], float32
@triton.jit
def _gelu_tanh_kernel(
    x_ptr,         # *const float
    out_ptr,       # *float
    B: tl.constexpr,
    C: tl.constexpr,
    H: tl.constexpr,
    W: tl.constexpr,
    K: tl.constexpr,  # sqrt(2/pi)
    BLOCK: tl.constexpr,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    w = tl.program_id(3)

    # Linear indexing for NCHW
    idx = (((b * C) + c) * H + h) * W + w
    x = tl.load(x_ptr + idx).to(tl.float32)

    # GELU tanh approximation:
    # gelu(x) = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    x3 = x * x * x
    inner = K * (x + 0.044715 * x3)
    # tanh(inner) = (e^{2inner} - 1) / (e^{2inner} + 1)
    e2 = tl.exp(2.0 * inner)
    tanh_inner = (e2 - 1.0) / (e2 + 1.0)
    gelu = 0.5 * x * (1.0 + tanh_inner)

    tl.store(out_ptr + idx, gelu)
