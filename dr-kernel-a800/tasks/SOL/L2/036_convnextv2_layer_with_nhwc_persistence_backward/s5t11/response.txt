- Implement and launch Triton kernels for heavy operations:
  1) fill_rand_kernel: generate random float32 tensors (for residual, grad_output, weights, etc.).
  2) depthwise_conv2d_1x7x7_nchw_kernel: perform depthwise convolution with kernel (1,7,7), padding=3, groups=C on NCHW input. Launch this for given B,C,H,W.

  7) per_channel_sum_squares_hw_kernel: compute per-channel sum of squares across (B,H,W) for x_gelu, used to derive global_features per channel (norm). Launch this.

@triton.jit
def gelu_approx_kernel(in_ptr, out_ptr, N, BLOCK: tl.constexpr):
    # Elementwise GELU approximation: y = 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    sqrt_2_over_pi = 0.7978845604444427
    inner = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
    y = 0.5 * x * (1.0 + tl.tanh(inner))
    tl.store(out_ptr + offsets, y, mask=mask)

        # We can compute sum with torch. However, the requirement is to use Triton for all computations. To comply, we implement mean/var in Triton. Triton can compute per-channel sums across (H*W) if we pass strides and NCHW contiguous. We’ll do that.

        # Compute per-channel sum and var using Triton per_channel_sum_squares_hw_kernel on x_nchw. We need to pass x_nchw as NCHW: (B,H,W,C). Our kernel expects NHWC strides. To use it, we need to define NHWC strides for x_nchw: but x_nchw is NCHW. Therefore, we implement a dedicated Triton kernel that reads NCHW as (B,C,H,W) and computes sum over H*W for each (n,c). We will create a kernel that computes per-channel sum over H*W for NCHW.

        # Define a Triton kernel that computes per-channel sum over H*W for NCHW (B,C,H,W) -> sums_ptr of shape (B*C,)
        sums_ptr = torch.empty((self.B * self.C,), device=self.device, dtype=torch.float32)
        # Launch kernel: grid = (B*C,)
        # Strides: x_nchw is contiguous NCHW. Strides: (H*W*C, H*W, W, 1). For sum over H*W, we need to access elements per (n,c) across h,w. We can implement a simple loop in Triton per (n,c) over H*W in chunks.
        # Triton doesn't support Python loops; we implement with while-like constructs using constexpr. For simplicity, we assume H*W is small or constexpr. Since H=W=14, H*W=196. We can loop over 196. Triton kernels need loops, so we implement for range(0,196,32) chunks. We’ll do that.

        # Triton kernel per_channel_sum_hw_nchw_kernel: not defined above. We need to define it. We’ll add it here.

        # Define per-channel sum over H*W for NCHW
        @triton.jit
        def per_channel_sum_hw_nchw_kernel(
            x_ptr, sums_ptr,
            B, C, H, W,
            x_stride_n, x_stride_c, x_stride_h, x_stride_w,
        ):
            pid = tl.program_id(axis=0)
            n = pid // C
            c = pid % C
            HW = H * W
            sum_ = 0.0
            for k in range(0, HW, 32):  # iterate in chunks
                offsets = k + tl.arange(0, 32)
                mask = offsets < HW
                h = offsets // W
                w = offsets % W
                x_off = n * x_stride_n + c * x_stride_c + h * x_stride_h + w * x_stride_w
                vals = tl.load(x_ptr + x_off, mask=mask, other=0.0)
                sum_ += tl.sum(vals, axis=0)
            tl.store(sums_ptr + (n * C + c), sum_)

        # Launch per-channel sum kernel
        grid_sum = (self.B * self.C,)
        x_nchw_contig = x_nchw.contiguous()  # ensure contiguous
        per_channel_sum_hw_nchw_kernel[grid_sum](
            x_nchw_contig, sums_ptr,
            self.B, self.C, self.H, self.W,
            x_nchw_contig.stride(0), x_nchw_contig.stride(1), x_nchw_contig.stride(2), x_nchw_contig.stride(3),
            BLOCK=32
        )

        # Compute mean and var from sums_ptr
        mean = (sums_ptr / (self.H * self.W)).view(self.B, self.C, 1, 1)
        # Compute var: we need sum of squared differences. We can compute it with torch for simplicity. However, the requirement is to use Triton. To comply, we compute squared differences in Triton and reduce. We implement per_channel_var_hw_nchw_kernel similarly.

        # Define per-channel variance over H*W for NCHW
        @triton.jit
        def per_channel_var_hw_nchw_kernel(
            x_ptr, sums_ptr, vars_ptr,
            B, C, H, W,
            x_stride_n, x_stride_c, x_stride_h, x_stride_w,
        ):
            pid = tl.program_id(axis=0)
            n = pid // C
            c = pid % C
            HW = H * W
            sum_ = 0.0
            sum_sq = 0.0
            for k in range(0, HW, 32):
                offsets = k + tl.arange(0, 32)
                mask = offsets < HW
                h = offsets // W
                w = offsets % W
                x_off = n * x_stride_n + c * x_stride_c + h * x_stride_h + w * x_stride_w
                vals = tl.load(x_ptr + x_off, mask=mask, other=0.0)
                sum_ += tl.sum(vals, axis=0)
                sum_sq += tl.sum(vals * vals, axis=0)
            mean = sum_ / HW
            var = sum_sq / HW - mean * mean
            tl.store(vars_ptr + (n * C + c), var)

        # Launch per-channel var kernel
        vars_ptr = torch.empty((self.B * self.C,), device=self.device, dtype=torch.float32)
        per_channel_var_hw_nchw_kernel[grid_sum](
            x_nchw_contig, sums_ptr, vars_ptr,
            self.B, self.C, self.H, self.W,
            x_nchw_contig.stride(0), x_nchw_contig.stride(1), x_nchw_contig.stride(2), x_nchw_contig.stride(3),
            BLOCK=32
        )

        var = (vars_ptr / (self.H * self.W)).view(self.B, 1, 1, 1)  # original code returns var as per-sample variance; here we return per-channel var over H*W. Note: original forward in the sample returns var as (B,1,1,1). We align with that.
