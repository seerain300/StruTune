    x_ptr,               # *const float, input: [B, C4, H, W]
    norm_ptr,            # *float, output: [B*C4]
    B: tl.int32,
    C4: tl.int32,
    H: tl.int32,
    W: tl.int32,
    BLOCK_HW: tl.constexpr,
):
    pid_bc = tl.program_id(0)  # over B*C4
    BC = B * C4
    if pid_bc >= BC:
        return

    b = pid_bc // C4
    c4 = pid_bc % C4

    sum_sq = tl.zeros((), dtype=tl.float32)
    HW = H * W
    for tile in range(0, 1024):  # covers typical HW up to 56*56
        start = tile * BLOCK_HW
        pos = start + tl.arange(0, BLOCK_HW)
        mask = pos < HW
        h = pos // W
        w = pos % W
        base = b * (C4 * HW) + c4 * HW + h * W + w
        x = tl.load(x_ptr + base, mask=mask, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)

    norm_val = tl.sqrt(sum_sq)
    tl.store(norm_ptr + pid_bc, norm_val)
