# normalize, and scale by layernorm_weight (per-channel). Writes to out_ln (B,H,W,C).
@triton.jit
def layernorm_nhwc_kernel(
    x_nhwc_ptr,          # *const float
    ln_weight_ptr,       # *const float, [C]
    out_ptr,             # *float, [B,H,W,C]
    B, H, W, C,          # runtime sizes
    eps,                 # runtime float32
):

    hw = tl.program_id(1) # over H*W
    h = hw // W
    w = hw % W

    # Accumulate sum and sum of squares over C
    sum_x = 0.0
    sum_x2 = 0.0
    for c0 in range(0, C):
        ptr = x_nhwc_ptr + b * (H * W * C) + h * (W * C) + w * C + c0
        x = tl.load(ptr)  # scalar
        sum_x += x
        sum_x2 += x * x

    var = sum_x2 / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Normalize and scale
    for c0 in range(0, C):
        in_ptr = x_nhwc_ptr + b * (H * W * C) + h * (W * C) + w * C + c0
        out_ptr_c = out_ptr + b * (H * W * C) + h * (W * C) + w * C + c0

        tl.store(out_ptr_c, y)

def gelu_pointwise_kernel(

    out_ptr,             # *float
    B, C4, H, W,         # runtime sizes
    eps,                 # runtime float32 (unused)

    pid_tile = tl.program_id(1)

    c4 = pid_bc % C4

    start = pid_tile * BLOCK_HW
    offs = start + tl.arange(0, BLOCK_HW)
    mask = offs < total

    h = offs // W
    w = offs % W

    tanh_inner = tl.tanh(inner)
    y = 0.5 * x * (1.0 + tanh_inner)

    tl.store(out_ptr + base, y, mask=mask)

    norm_ptr,            # *float
    B, C4, H, W,         # runtime sizes
    BLOCK_HW: tl.constexpr,
):

    b = pid // C4
    c4 = pid % C4

    total = H * W
    for start in range(0, total, BLOCK_HW):

        mask = offs < total
        h = offs // W
        w = offs % W

        x = tl.load(x_ptr + base, mask=mask, other=0.0)

    tl.store(norm_ptr + pid, norm_val)

    Wo = W + 6
