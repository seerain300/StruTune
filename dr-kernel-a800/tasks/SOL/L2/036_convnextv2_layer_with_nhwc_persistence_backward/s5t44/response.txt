@triton.jit

# Triton kernel: per-channel LayerNorm mean over (H,W) for NHWC tensor, output mean: (B*C,)
# y_in: (B,H,W,C), out_mean: (B*C,)
@triton.jit
def layernorm_mean_nhwc_kernel(y_ptr, out_mean_ptr, B, H, W, C,
                                y_stride_b, y_stride_h, y_stride_w, y_stride_c,
                                BLOCK_HW: tl.constexpr):
    pid_bc = tl.program_id(axis=0)
    b = pid_bc // C
    c = pid_bc % C

    sum_val = tl.zeros((), dtype=tl.float32)

# Triton kernel: apply LayerNorm normalization and per-channel weight on NHWC: y = (x - mean) / sqrt(var + eps) * gamma + beta
# y_in: (B,H,W,C), gamma/beta: (C,), y_out: (B,H,W,C)
@triton.jit
def layernorm_apply_nhwc_kernel(y_ptr, gamma_ptr, beta_ptr, y_out_ptr, B, H, W, C,
                                y_stride_b, y_stride_h, y_stride_w, y_stride_c,
                                y_out_stride_b, y_out_stride_h, y_out_stride_w, y_out_stride_c,
                                mean_ptr, var_ptr, eps: tl.constexpr,
                                BLOCK_HW: tl.constexpr):
    pid_bc = tl.program_id(axis=0)
    b = pid_bc // C
    c = pid_bc % C

    mean = tl.load(mean_ptr + b * C + c)
    var = tl.load(var_ptr + b * C + c)
    gamma = tl.load(gamma_ptr + c)
    beta = tl.load(beta_ptr + c)

    inv_std = 1.0 / tl.sqrt(var + eps)

    for h in range(0, H):
        for w_start in range(0, W, BLOCK_HW):
            w = w_start + tl.arange(0, BLOCK_HW)
            mask = w < W

# x_ptr: (B,C,H,W) NHWC, weight_ptr: (C_in, C_out), y_ptr: (B,C,H,W) NHWC
@triton.jit
def linear_matvec_nhwc_kernel(x_ptr, weight_ptr, y_ptr,
                               B, C_in, H, W, C_out,

    # We launch with grid=(B*C_out*H*W, ), so pid_out is a single linear index.

    rem = pid_out % total_per_out

    h = rem2 // W
    w = rem2 % W

# Triton kernel: GELU (tanh approximation) elementwise on NHWC y, out: same shape
@triton.jit
def gelu_tanh_nhwc_kernel(y_ptr, out_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < total_elems
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (y + 0.044715 * y * y * y)
    tanh_inner = tl.tanh(inner)
    gelu = 0.5 * y * (1.0 + tanh_inner)
    tl.store(out_ptr + offsets, gelu, mask=mask)
