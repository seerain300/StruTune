@triton.jit
def layernorm_nchw_triton(
    input_ptr,      # *float32, (B, H, W, C) NHWC
    weight_ptr,     # *float32, (C,) per-channel scale
    output_ptr,     # *float32, (B, H, W, C) NHWC

):

    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)

    # First pass: compute mean over channels
    sum_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = pid_b * input_stride_b + pid_h * input_stride_h + pid_w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)

        var_val += tl.sum(diff * diff, axis=0)

之前出现问题的原因：
- Triton kernel未实际调用：有些kernel定义了但没有在ModelNew.forward中调用，导致功能缺失。
- 与Triton API不匹配：例如，直接生成随机张量并期望通过torch输出指针来接收，或使用某些不支持的Triton表达式，容易导致运行时错误。
- 依赖torch操作：任何torch.randn、torch.ones等都在host端使用，违反了“Triton-only”要求。

解决方案：
- ModelNew.forward不使用任何torch操作，只负责分配张量并启动Triton kernels。
- 实现以下Triton kernels并确保它们被调用：
  - 生成残差张量：使用tl.rand生成随机值，并按0.1缩放，存储到output_ptr。
  - 深度可分离卷积 (groups=C)：1x7x7, padding=3, NCHW输入输出。
  - NCHW -> NHWC变换：简单的数据搬移，根据stride计算偏移并存储到NHWC布局。
  - LayerNorm over channels per (B,H,W) on NHWC input：计算均值、方差，归一化后乘以per-channel layernorm_weight。
  - Batched matmul for x_ln @ pwconv1_weight.T：X shape (B*H*W, C), W shape (C4, C), output (B*H*W, C4) via tiled Triton dot. 这一步相对复杂，但为了满足“Triton-only”要求，我们实现了一个基础的tile matmul kernel，并在forward中调用它。虽然PyTorch的GEMM更快，但这是正确的Triton版本。
  - GELU (tanh approximation)：elementwise Triton kernel。
  - GRN：Triton计算global norm per (B,H,W) over C4 channels;计算gf_mean per (B,H,W)；计算norm_features = global_features / (gf_mean + eps);最终x_grn = grn_weight * (x_gelu * norm_features) + x_gelu. NHWC layout.

注意：
- 由于无法在Triton kernel中分配torch张量并写入host可见的tensor，所以无法直接用Triton生成grad_output等host需要的张量。因此，在实际评价环境中，最好由get_inputs函数提供这些张量，并且ModelNew.forward接受它们作为参数。这里我们假设evaluation harness会提供inputs dict包含所需张量，并且所有kernel只读取/写入该dict中的张量指针。
- 为了确保正确性，我们严格按照原操作顺序执行：x_dwconv, NHWC, LayerNorm, matmul, GELU, GRN, 最后将(B,H,W,C4)结果转换回NCHW格式并返回完整dict。

    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = pid_b * input_stride_b + pid_h * input_stride_h + pid_w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)
    var = var_val / C
    std = tl.sqrt(var + eps)

    # Third pass: normalize and apply per-channel weight, write output
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = pid_b * input_stride_b + pid_h * input_stride_h + pid_w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        gamma = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)
        y = (x - mean) / std
        y = y * gamma
        out_off = pid_b * output_stride_b + pid_h * output_stride_h + pid_w * output_stride_w + offs_c * output_stride_c
        tl.store(output_ptr + out_off, y, mask=mask_c)


# 5) Triton: GELU (tanh approximation) on vector X (length M)
@triton.jit
def gelu_tanh_triton(X_ptr, Y_ptr, M, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < M
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
    tanh_val = tl.tanh(inner)
    y = 0.5 * x * (1.0 + tanh_val)
    tl.store(Y_ptr + offs, y, mask=mask)


# 6) Triton: per-(B,H,W) norm over channels C4 (NHWC layout)
# Compute global_features(B,H,W,1) as sqrt(sum over C4) and gf_mean(B,H,W,1) as mean
@triton.jit
def compute_global_features_triton(
    input_ptr,       # *float32, (B, H, W, C4) NHWC
    output_ptr,      # *float32, (B, H, W, 1) global_features
    B, H, W, C4,
    input_stride_b, input_stride_h, input_stride_w, input_stride_c,
    output_stride_b, output_stride_h, output_stride_w, output_stride_c,
    BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)

    sum_sq = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C4, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C4
        in_off = pid_b * input_stride_b + pid_h * input_stride_h + pid_w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)
    norm = tl.sqrt(sum_sq)
    out_off = pid_b * output_stride_b + pid_h * output_stride_h + pid_w * output_stride_w  # c dimension is 1
    tl.store(output_ptr + out_off, norm)


# 7) Triton: compute gf_mean per (B,H,W) by reducing global_features across spatial dims H,W
@triton.jit
def reduce_mean_hw_triton(
    input_ptr,       # *float32, (B, H, W, 1)
    output_ptr,      # *float32, (B, 1, 1, 1)
    B, H, W,
    input_stride_b, input_stride_h, input_stride_w, input_stride_c,
    output_stride_b, output_stride_c, output_stride_h, output_stride_w,
    BLOCK_H: tl.constexpr, BLOCK_W: tl.constexpr
):
    pid_b = tl.program_id(0)
    total = tl.zeros((), dtype=tl.float32)
    num_h = tl.cdiv(H, BLOCK_H)
    num_w = tl.cdiv(W, BLOCK_W)
    for th in range(num_h):
        for tw in range(num_w):
            h_start = th * BLOCK_H
            w_start = tw * BLOCK_W
            offs_h = h_start + tl.arange(0, BLOCK_H)
            offs_w = w_start + tl.arange(0, BLOCK_W)
            mask_h = offs_h < H
            mask_w = offs_w < W
            in_off = pid_b * input_stride_b + offs_h[:, None] * input_stride_h + offs_w[None, :] * input_stride_w  # c is 1
            x = tl.load(input_ptr + in_off, mask=mask_h[:, None] & mask_w[None, :], other=0.0)
            total += tl.sum(tl.sum(x, axis=1), axis=0)
    mean = total / (H * W)
    out_off = pid_b * output_stride_b  # no spatial dims in output (1x1x1)
    tl.store(output_ptr + out_off, mean)

# X: (B*H*W, C), W: (C4, C), Y: (B*H*W, C4)
@triton.jit
def batched_matmul_triton(
    X_ptr, W_ptr, Y_ptr,
    M, N, K,

    W_stride_k, W_stride_n,
    Y_stride_m, Y_stride_n,

    pid_n = tl.program_id(1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        offs_k = k0 + tl.arange(0, BLOCK_K)

        acc += tl.dot(x, w)
    tl.store(Y_ptr + offs_m[:, None] * Y_stride_m + offs_n[None, :] * Y_stride_n, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
