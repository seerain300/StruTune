):

    c = pid_nc % C

# Triton kernel: LayerNorm on NHWC tensor per (b,h,w) across channel C.
# Inputs:
#   x_nhwc: (B,H,W,C) float32
#   mean_out: (B,H,W) float32 output
#   var_out: (B,H,W) float32 output
#   gamma: (C) float32 (layernorm_weight), applied after normalization
# We compute mean and variance across C for each (b,h,w), normalize, and store.
@triton.jit
def layernorm_nhwc_kernel(
    x_ptr, mean_ptr, var_ptr, gamma_ptr,
    B, H, W, C,
    x_stride_b, x_stride_h, x_stride_w, x_stride_c,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    b = pid // (H * W)
    hw = pid % (H * W)
    h = hw // W
    w = hw % W

    # First pass: sum and sum of squares across C
    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)
    for off in range(0, C, BLOCK_C):
        c_offsets = off + tl.arange(0, BLOCK_C)
        mask = c_offsets < C
        x_ptrs = x_ptr + b * x_stride_b + h * x_stride_h + w * x_stride_w + c_offsets * x_stride_c
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
        sum_val += tl.sum(x_vals, axis=0)
        sum_sq += tl.sum(x_vals * x_vals, axis=0)
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    # Store mean and var
    tl.store(mean_ptr + b * (H * W) + h * W + w, mean)
    tl.store(var_ptr + b * (H * W) + h * W + w, var)

    # Second pass: normalize and apply gamma
    for off in range(0, C, BLOCK_C):

        # Apply gamma per channel
        gamma = tl.load(gamma_ptr + c_offsets, mask=mask, other=1.0)
        x_norm = x_norm * gamma
        # Store back
        tl.store(x_ptrs, x_norm, mask=mask)

    y = 0.5 * x * (1.0 + tanh_inner)
