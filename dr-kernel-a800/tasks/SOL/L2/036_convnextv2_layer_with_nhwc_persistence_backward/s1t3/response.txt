@triton.jit
def conv2d_depthwise_forward_triton(
    input_ptr,        # *float32, (B, C, H, W), NCHW
    weight_ptr,       # *float32, (C, 1, 7, 7), each channel's filter
    output_ptr,       # *float32, (B, C, H_out, W_out), NCHW
    B, C, H, W,
    weight_stride_c, weight_stride_h, weight_stride_w,
    output_stride_b, output_stride_c, output_stride_h, output_stride_w,

    c = tl.program_id(1)
    h_out = tl.program_id(2)
    w_out = tl.program_id(3)

    W_out = W + 2 * PAD_W - KW + 1

@triton.jit
def permute_nchw_to_nhwc_triton(
    input_ptr,      # *float32, (B, C, H, W), NCHW
    output_ptr,     # *float32, (B, H, W, C), NHWC

@triton.jit
def layernorm_nchw_triton(
    input_ptr,      # *float32, (B, H, W, C) NHWC
    weight_ptr,     # *float32, (C,)
    output_ptr,     # *float32, (B, H, W, C) NHWC
    B, H, W, C,
    input_stride_b, input_stride_h, input_stride_w, input_stride_c,
    output_stride_b, output_stride_h, output_stride_w, output_stride_c,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr
):
    # Grid: (B, H, W)
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    # First pass: compute mean over channels
    sum_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / C

    # Second pass: compute var over channels
    var_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)
    var = var_val / C
    std = tl.sqrt(var + eps)

    # Third pass: normalize and apply per-channel weight, write output
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        gamma = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)

        y = y * gamma
        out_off = b * output_stride_b + h * output_stride_h + w * output_stride_w + offs_c * output_stride_c

# 5) Triton: Batched matmul X(M,K) @ W(K,N) -> Y(M,N)
# X: (B*H*W, C), W: (C4, C), Y: (B*H*W, C4)
@triton.jit
def batched_matmul_triton(
    X_ptr, W_ptr, Y_ptr,
    M, N, K,
    X_stride_m, X_stride_k,
    W_stride_k, W_stride_n,
    Y_stride_m, Y_stride_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        acc += tl.dot(x, w)

# 6) Triton: elementwise GELU (tanh approximation) for vector X
@triton.jit
def gelu_tanh_triton(X_ptr, Y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
    tanh_val = tl.tanh(inner)
    y = 0.5 * x * (1.0 + tanh_val)
    tl.store(Y_ptr + offs, y, mask=mask)

def reduce_norm_channels_triton(

    output_ptr,     # *float32, (B, H, W, 1)

    output_stride_b, output_stride_h, output_stride_w, output_stride_c,
    eps: tl.constexpr,
    BLOCK_C4: tl.constexpr
):

    h = tl.program_id(1)
    w = tl.program_id(2)
    sum_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C4, BLOCK_C4):
        offs_c = c0 + tl.arange(0, BLOCK_C4)
        mask_c = offs_c < C4
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        sum_val += tl.sum(x * x, axis=0)
    norm = tl.sqrt(sum_val + eps)  # scalar per (b,h,w)

    tl.store(output_ptr + out_off, norm)

def apply_grn_triton(

    h = tl.program_id(1)
    w = tl.program_id(2)
    # Load norm scalar per (b,h,w) at channel 0 (since norm is 1x1x1)
    norm_off = b * norm_stride_b + h * norm_stride_h + w * norm_stride_w + 0 * norm_stride_c
    norm_val = tl.load(norm_ptr + norm_off)  # scalar

import torch
import triton
import triton.language as tl


# 1) Triton kernel: Depthwise Conv2d (groups=C), 1x7x7, padding=3, NCHW -> NCHW
@triton.jit
def conv2d_depthwise_forward_triton(
    input_ptr,        # *float32, (B, C, H, W), NCHW
    weight_ptr,       # *float32, (C, 1, 7, 7), per-channel filters
    output_ptr,       # *float32, (B, C, H_out, W_out), NCHW
    B, C, H, W,
    weight_stride_c, weight_stride_h, weight_stride_w,
    output_stride_b, output_stride_c, output_stride_h, output_stride_w,
    KH: tl.constexpr, KW: tl.constexpr, PAD_H: tl.constexpr, PAD_W: tl.constexpr
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    h_out = tl.program_id(2)
    w_out = tl.program_id(3)

    # Output size for depthwise conv with padding 3 and 1x7x7: H_out = H + 2*PAD_H - KH + 1
    H_out = H + 2 * PAD_H - KH + 1
    W_out = W + 2 * PAD_W - KW + 1

    acc = tl.zeros((), dtype=tl.float32)
    for kh in range(KH):
        for kw in range(KW):
            h_in = h_out + PAD_H - kh
            w_in = w_out + PAD_W - kw
            in_bounds = (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)
            if in_bounds:
                input_off = b * H * W + c * H * W + h_in * W + w_in  # assuming input is contiguous (B, C, H, W)
                w_off = c * weight_stride_c + kh * weight_stride_h + kw * weight_stride_w
                w_val = tl.load(weight_ptr + w_off)
                x_val = tl.load(input_ptr + input_off)
                acc += x_val * w_val

    output_off = b * output_stride_b + c * output_stride_c + h_out * output_stride_h + w_out * output_stride_w
    tl.store(output_ptr + output_off, acc)


# 2) Triton: permute NCHW -> NHWC (B, C, H, W) -> (B, H, W, C)
@triton.jit
def permute_nchw_to_nhwc_triton(
    input_ptr,      # *float32, (B, C, H, W), NCHW
    output_ptr,     # *float32, (B, H, W, C), NHWC
    B, C, H, W,
    input_stride_b, input_stride_c, input_stride_h, input_stride_w,
    output_stride_b, output_stride_h, output_stride_w, output_stride_c,
    BLOCK_H: tl.constexpr, BLOCK_W: tl.constexpr
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    num_h = tl.cdiv(H, BLOCK_H)
    num_w = tl.cdiv(W, BLOCK_W)
    for th in range(num_h):
        for tw in range(num_w):
            h_start = th * BLOCK_H
            w_start = tw * BLOCK_W
            offs_h = h_start + tl.arange(0, BLOCK_H)
            offs_w = w_start + tl.arange(0, BLOCK_W)
            mask = (offs_h[:, None] < H) & (offs_w[None, :] < W)
            h = offs_h[:, None]
            w = offs_w[None, :]
            input_off = b * input_stride_b + c * input_stride_c + h * input_stride_h + w * input_stride_w
            x = tl.load(input_ptr + input_off, mask=mask, other=0.0)
            out_off = b * output_stride_b + h * output_stride_h + w * output_stride_w + c * output_stride_c
            tl.store(output_ptr + out_off, x, mask=mask)


# 3) Triton: LayerNorm over channels for each (N,H,W) on NHWC input/output
# Compute mean and variance across C channels, then normalize and apply per-channel layernorm_weight
@triton.jit
def layernorm_nchw_triton(
    input_ptr,      # *float32, (B, H, W, C) NHWC
    weight_ptr,     # *float32, (C,)
    output_ptr,     # *float32, (B, H, W, C) NHWC
    B, H, W, C,
    input_stride_b, input_stride_h, input_stride_w, input_stride_c,
    output_stride_b, output_stride_h, output_stride_w, output_stride_c,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr
):
    # Grid: (B, H, W)
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    # First pass: compute mean over channels
    sum_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / C

    # Second pass: compute var over channels
    var_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)
    var = var_val / C
    std = tl.sqrt(var + eps)

    # Third pass: normalize and apply per-channel weight, write output
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        gamma = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)
        y = (x - mean) / std
        y = y * gamma
        out_off = b * output_stride_b + h * output_stride_h + w * output_stride_w + offs_c * output_stride_c
        tl.store(output_ptr + out_off, y, mask=mask_c)
