import torch
import triton
import triton.language as tl


# -------- Triton kernels --------

# Kernel 1: fill tensor with random normal (float32) on device
@triton.jit
def fill_randn_kernel(OUT_ptr, SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * 1 + tl.arange(0, 1)  # single element per program
    mask = offsets < SIZE
    # Generate uniform random in [0,1) and map to N(0,1)
    u = tl.random(seed=0)  # Triton's random; using fixed seed for reproducibility
    val = tl.sqrt(-2.0 * tl.log(1.0 - u)) * tl.cos(2.0 * 3.141592653589793 * u)
    tl.store(OUT_ptr + offsets, val, mask=mask)


# Kernel 2: compute mean and variance across width W for X of shape (B, C, H, W)
@triton.jit
def compute_mean_var_w_kernel(
    X_ptr,                 # *const float32, input [B, C, H, W] contiguous
    MEAN_ptr,              # *float32, output [B, C, H, 1] contiguous
    VAR_ptr,               # *float32, output [B, C, H, 1] contiguous
    B: tl.constexpr,       # int
    C: tl.constexpr,       # int
    H: tl.constexpr,       # int
    W: tl.constexpr,       # int
    BLOCK_W: tl.constexpr  # tile size along W (e.g., 128)
):
    b = tl.program_id(axis=0)
    c = tl.program_id(axis=1)
    h = tl.program_id(axis=2)

    # Base offset for this (b, c, h) row
    base = ((b * C + c) * H + h) * W
    sum_val = 0.0
    sum_sq = 0.0

    for w_start in range(0, W, BLOCK_W):
        w_idx = w_start + tl.arange(0, BLOCK_W)
        mask = w_idx < W
        x_ptrs = X_ptr + base + w_idx
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
        # Reduce within tile
        sum_val += tl.sum(x_vals)
        sum_sq += tl.sum(x_vals * x_vals)

    N = W  # reduce across width
    mean = sum_val / N
    var = sum_sq / N - mean * mean
    # Store mean and var to outputs (shape [B, C, H, 1] contiguous)
    out_idx = b * (C * H) + c * H + h
    tl.store(MEAN_ptr + out_idx, mean)
    tl.store(VAR_ptr + out_idx, var)


# Kernel 3: elementwise GELU (tanh approximation)
@triton.jit
def elementwise_gelu_tanh_kernel(
    IN_ptr,                # *const float32, input flattened [M]
    OUT_ptr,               # *float32, output flattened [M]
    M: tl.constexpr        # int, total number of elements
):
    pid = tl.program_id(axis=0)
    offsets = pid * 1 + tl.arange(0, 1)
    mask = offsets < M
    x = tl.load(IN_ptr + offsets, mask=mask, other=0.0)
    # GELU tanh approximation
    sqrt_2_over_pi = 0.7978845608028654
    c = 0.044715
    inner = sqrt_2_over_pi * (x + c * x * x * x)
    tanh_val = tl.tanh(inner)
    y = 0.5 * x * (1.0 + tanh_val)
    tl.store(OUT_ptr + offsets, y, mask=mask)


# Kernel 4: linear projection: out[M] = sum over j of IN[M] * W[J, N], where
# IN is flattened [M] with M=B*C*H*W, W has shape [K=C, N=C4], out has shape [M]
@triton.jit
def linear_matmul_kernel(
    IN_ptr,                # *const float32, input flattened [M]
    W_ptr,                 # *const float32, weight [K, N] with K=C, N=C4
    OUT_ptr,               # *float32, output flattened [M]
    B: tl.constexpr,       # int
    C: tl.constexpr,       # int (K)
    N: tl.constexpr,       # int (C4)
    BLOCK_N: tl.constexpr  # tile size along N (e.g., 64)
):
    pid_m = tl.program_id(axis=0)
    m_start = pid_m * BLOCK_N
    m_idx = m_start + tl.arange(0, BLOCK_N)
    mask_m = m_idx < (B * C * H * W)

    # Compute per-m index mapping: m_idx corresponds to flattened B*C*H*W
    # To multiply, we need i index (input channel) and flatten h,w within that channel.
    # Simplify: we treat IN as 1D and loop over N tiles to compute partial dot products.
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)

    # Loop over K=C (inner dimension), accumulate dot products into acc
    for k in range(0, C):
        # For each m in the tile, we fetch IN[m] and multiply by W[k, n] across n tile
        for n_start in range(0, N, BLOCK_N):
            n_idx = n_start + tl.arange(0, BLOCK_N)
            mask_n = n_idx < N
            # IN[m] values
            in_ptrs = IN_ptr + m_idx
            in_vals = tl.load(in_ptrs, mask=mask_m, other=0.0)  # shape [BLOCK_N]
            # W[k, n] values
            w_ptrs = W_ptr + k * N + n_idx
            w_vals = tl.load(w_ptrs, mask=mask_n, other=0.0)    # shape [BLOCK_N]
            # Multiply and reduce over n tile for each m in the tile
            # Broadcast in_vals [BLOCK_N] with w_vals [BLOCK_N]
            prod = in_vals[:, None] * w_vals[None, :]
            # Reduce across N tile: sum over last axis
            acc += tl.sum(prod, axis=1)  # sum over BLOCK_N to produce [BLOCK_N]

    # Store results
    out_ptrs = OUT_ptr + m_idx
    tl.store(out_ptrs, acc, mask=mask_m)

        pwconv1_weight = torch.empty((C, C4), device="cuda", dtype=torch.float32)

        grid_w = (total_elems_w,)
        fill_randn_kernel[grid_w](pwconv1_weight.view(-1))

        mean = torch.empty((B, C, H, 1), device="cuda", dtype=torch.float32)
