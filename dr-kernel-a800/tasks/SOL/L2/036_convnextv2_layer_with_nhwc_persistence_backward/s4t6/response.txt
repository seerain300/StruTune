import torch
import triton
import triton.language as tl


@triton.jit
def depthwise_conv2d_groupsC_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    # One program per output element (b, c, oh, ow)
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    acc = 0.0
    # Loop over 7x7 kernel
    for kh in range(7):
        in_h = oh + kh - pad_h
        for kw in range(7):
            in_w = ow + kw - pad_w
            in_bounds = (in_h >= 0) & (in_h < H) & (in_w >= 0) & (in_w < W)
            # Load input x[b, c, in_h, in_w]
            x_off = b * stride_xB + c * stride_xC + in_h * stride_xH + in_w * stride_xW
            x_val = tl.load(x_ptr + x_off, mask=in_bounds, other=0.0)
            # Load weight w[c, 0, kh, kw]
            w_off = c * stride_wC + kh * stride_wKH + kw * stride_wKW
            w_val = tl.load(w_ptr + w_off)
            acc += x_val * w_val
    # Store output y[b, c, oh, ow]
    y_off = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_off, acc)


@triton.jit
def per_hw_channel_mean_kernel(
    x_ptr, mean_ptr,
    B, C, H, W, H_out, W_out,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_mB, stride_mH, stride_mW, stride_mC,  # mean is (B,1,1,C), we store C dimension as 1 for safety
):
    b = tl.program_id(0)
    oh = tl.program_id(1)
    ow = tl.program_id(2)

    # Accumulator for sum over channels
    acc = 0.0
    for c in range(C):
        x_off = b * stride_xB + c * stride_xC + oh * stride_xH + ow * stride_xW
        x_val = tl.load(x_ptr + x_off)
        acc += x_val
    # Compute mean
    mean_val = acc / C
    # Store to mean[b, 0, 0, 0] (we ignore C in mean storage, since mean is per (b,h,w))
    tl.store(mean_ptr + b * stride_mB, mean_val)


@triton.jit
def per_hw_channel_var_kernel(
    x_ptr, mean_ptr, var_ptr,
    B, C, H, W, H_out, W_out,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_mB, stride_mH, stride_mW, stride_mC,
    stride_vB, stride_vH, stride_vW, stride_vC,
):
    b = tl.program_id(0)
    oh = tl.program_id(1)
    ow = tl.program_id(2)

    # Load mean
    mean_val = tl.load(mean_ptr + b * stride_mB)
    acc2 = 0.0
    for c in range(C):
        x_off = b * stride_xB + c * stride_xC + oh * stride_xH + ow * stride_xW
        x_val = tl.load(x_ptr + x_off)
        diff = x_val - mean_val
        acc2 += diff * diff
    var_val = acc2 / C
    tl.store(var_ptr + b * stride_vB, var_val)


@triton.jit
def per_hw_channel_std_kernel(
    var_ptr, std_ptr,
    B, eps,
    stride_vB, stride_stB,
):
    b = tl.program_id(0)
    var_val = tl.load(var_ptr + b * stride_vB)
    std_val = tl.sqrt(var_val + eps)
    tl.store(std_ptr + b * stride_stB, std_val)


@triton.jit
def layernorm_scale_kernel(
    x_ptr, mean_ptr, std_ptr, ln_weight_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_mB, stride_stB,
    stride_lwC,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    mean_val = tl.load(mean_ptr + b * stride_mB)
    std_val = tl.load(std_ptr + b * stride_stB)
    x_off = b * stride_xB + c * stride_xC + oh * stride_xH + ow * stride_xW
    x_val = tl.load(x_ptr + x_off)
    ln_w = tl.load(ln_weight_ptr + c * stride_lwC)
    y_val = (x_val - mean_val) / std_val
    y_val = y_val * ln_w
    y_off = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_off, y_val)


@triton.jit
def per_b_hw_norm2_kernel(
    x_ptr, sum_ptr,
    B, C, H, W, H_out, W_out,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_sB,
):
    b = tl.program_id(0)
    acc = 0.0
    for oh in range(H_out):
        for ow in range(W_out):
            for c in range(C):
                x_off = b * stride_xB + c * stride_xC + oh * stride_xH + ow * stride_xW
                x_val = tl.load(x_ptr + x_off)
                acc += x_val * x_val
    tl.store(sum_ptr + b * stride_sB, acc)


@triton.jit
def per_hw_mean_kernel(
    sum_ptr, mean_ptr,
    B, H_out, W_out,
    stride_sB, stride_mB,
):
    b = tl.program_id(0)
    N = H_out * W_out
    s_val = tl.load(sum_ptr + b * stride_sB)
    mean_val = s_val / (C * N)
    tl.store(mean_ptr + b * stride_mB, mean_val)


@triton.jit
def per_hw_norm_factor_kernel(
    gf_ptr, mean_ptr, norm_ptr,
    B, H_out, W_out, C, eps,
    stride_gB, stride_gH, stride_gW,
    stride_mB,
    stride_nB, stride_nH, stride_nW, stride_nC,
):
    b = tl.program_id(0)
    # Load global_features and mean per (b,h,w). We need to compute mean over C for each (b,h,w).
    # To get mean per (b,h,w), we sum global_features[b,1,h,w] over C (C==1 here, but general approach needed).
    # However, global_features is per (b,h,w) scalar, so mean = gf / C.
    # Implement: load gf per (h,w) and compute mean.
    for oh in range(H_out):
        for ow in range(W_out):
            gf_val = tl.load(gf_ptr + b * stride_gB + oh * stride_gH + ow * stride_gW)
            mean_val = tl.load(mean_ptr + b * stride_mB)  # scalar per b
            norm_val = gf_val / (mean_val + eps)
            for c in range(C):  # norm is independent of c here; store for all c
                n_off = b * stride_nB + oh * stride_nH + ow * stride_nW + c * stride_nC
                tl.store(norm_ptr + n_off, norm_val)


@triton.jit
def gelu_tanh_kernel(
    x_ptr, gelu_ptr,
    B, C_out, H_out, W_out,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_gB, stride_gC, stride_gH, stride_gW,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)
    x_off = b * stride_xB + c * stride_xC + oh * stride_xH + ow * stride_xW
    x_val = tl.load(x_ptr + x_off)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x_val + 0.044715 * x_val * x_val * x_val)
    tanh_val = tl.tanh(inner)
    gelu_val = 0.5 * x_val * (1.0 + tanh_val)
    g_off = b * stride_gB + c * stride_gC + oh * stride_gH + ow * stride_gW
    tl.store(gelu_ptr + g_off, gelu_val)


@triton.jit
def grn_scale_kernel(

    s_off = b * stride_sB + c * stride_sC + oh * stride_sH + ow * stride_sW
    tl.store(scaled_ptr + s_off, scaled_val)


@triton.jit
def grn_output_kernel(

    gw = tl.load(grn_weight_ptr + c * stride_gC)  # shape (1,1,1,C4) -> use last dim
    y_val = scaled_val * gw + x_val
    y_off = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_off, y_val)

import torch
import triton
import triton.language as tl


@triton.jit
def depthwise_conv2d_groupsC_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    # One program per output element (b, c, oh, ow)
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    acc = 0.0
    # Loop over 7x7 kernel taps
    for kh in range(7):
        in_h = oh + kh - pad_h
        for kw in range(7):
            in_w = ow + kw - pad_w
            in_bounds = (in_h >= 0) & (in_h < H) & (in_w >= 0) & (in_w < W)
            # Load input x[b, c, in_h, in_w]
            x_off = b * stride_xB + c * stride_xC + in_h * stride_xH + in_w * stride_xW
            x_val = tl.load(x_ptr + x_off, mask=in_bounds, other=0.0)
            # Load weight w[c, 0, kh, kw]
            w_off = c * stride_wC + kh * stride_wKH + kw * stride_wKW
            w_val = tl.load(w_ptr + w_off)
            acc += x_val * w_val
    # Store output y[b, c, oh, ow]
    y_off = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_off, acc)

):

    oh = tl.program_id(1)
    ow = tl.program_id(2)

        x_val = tl.load(x_ptr + x_off)
        acc += x_val

    # Store into mean[b, 0, 0, 0] (per (b,h,w), C is ignored here)
    tl.store(mean_ptr + b * stride_mB, mean_val)

    oh = tl.program_id(1)
    ow = tl.program_id(2)

    tl.store(var_ptr + b * stride_vB, var_val)
