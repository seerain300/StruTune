@triton.jit
def reduce_sums_w_kernel(
    X_ptr,             # *const float32, input tensor (B, C, H, W) contiguous
    SUM_ptr,           # *float32, output sums per row (B*C*H,), but we use a 2D output to pack sum & sumsq
    SUMSQ_ptr,         # *float32, output sum of squares per row
    B: tl.int32,       # batch size
    C: tl.int32,       # channels
    H: tl.int32,       # height
    W: tl.int32,       # width
    BLOCK_N: tl.constexpr  # tile along rows (B*C*H)
):
    """
    For each row (b, c, h), compute sum and sum of squares across width W.
    We process rows in tiles of BLOCK_N. Each program handles a tile of rows.
    Output:
      SUM[pid, 0] = sum of x[b,c,h,:]
      SUMSQ[pid, 0] = sum of x[b,c,h,:]^2
    """
    pid = tl.program_id(axis=0)
    row_start = pid * BLOCK_N
    row_idx = row_start + tl.arange(0, BLOCK_N)
    total_rows = B * C * H
    row_mask = row_idx < total_rows

    # Map row_idx to (b, c, h)
    CH = C * H
    b = row_idx // CH
    rem = row_idx % CH
    c = rem // H
    h = rem % H

    # Base offset for this row in a contiguous (B, C, H, W) tensor:
    # offset = (((b*C + c)*H + h)*W)
    offsets = (((b * C + c) * H + h) * W)  # shape [BLOCK_N], int32

    # Accumulators
    sum_vals = tl.zeros([BLOCK_N], dtype=tl.float32)
    sumsq_vals = tl.zeros([BLOCK_N], dtype=tl.float32)

    # Loop over width W
    for w in range(0, W):
        ptrs = X_ptr + offsets + w  # add w to each row's base
        vals = tl.load(ptrs, mask=row_mask, other=0.0)
        sum_vals += vals
        sumsq_vals += vals * vals

    # Store per-row results into 2D SUM/SUMSQ tensors at column 0
    # SUM has shape (total_rows, 1), SUMSQ has shape (total_rows, 1)
    out_ptrs_sum = SUM_ptr + row_idx * 1 + 0
    out_ptrs_sumsq = SUMSQ_ptr + row_idx * 1 + 0
    tl.store(out_ptrs_sum, sum_vals, mask=row_mask)
    tl.store(out_ptrs_sumsq, sumsq_vals, mask=row_mask)


@triton.jit
def elementwise_square_kernel(
    IN_ptr,        # *const float32, input tensor (B, C, H, W) flattened
    OUT_ptr,       # *float32, output tensor same shape
    TOTAL: tl.int32,  # total number of elements = B*C*H*W
    BLOCK: tl.constexpr
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < TOTAL
    x = tl.load(IN_ptr + offsets, mask=mask, other=0.0)
    y = x * x
    tl.store(OUT_ptr + offsets, y, mask=mask)


@triton.jit
def elementwise_copy_kernel(
    SRC_ptr,       # *const float32, source tensor (C, C4) contiguous
    DST_ptr,       # *float32, destination tensor (C4, C) contiguous
    C: tl.int32,   # number of rows in src
    C4: tl.int32,  # number of cols in src / rows in dst
    BLOCK: tl.constexpr
):
    pid = tl.program_id(axis=0)
    # This kernel copies SRC[c, j] -> DST[j, c] in tiles along C4.
    for c in range(0, C, BLOCK):
        c_idx = c + tl.arange(0, BLOCK)
        c_mask = c_idx < C
        # for each j in C4, copy SRC[c_idx, j] to DST[j, c_idx]
        # We'll launch grid over C4; each program handles a tile of j for a fixed c.
        # To implement 2D tiling, use nested loop over j in tiles.
        for j in range(0, C4, BLOCK):
            j_idx = j + tl.arange(0, BLOCK)
            j_mask = j_idx < C4
            # Load SRC[c_idx, j_idx]
            src_ptrs = SRC_ptr + c_idx[:, None] * C4 + j_idx[None, :]
            vals = tl.load(src_ptrs, mask=c_mask[:, None] & j_mask[None, :], other=0.0)
            # Store to DST[j_idx, c_idx] (DST is (C4, C))
            dst_ptrs = DST_ptr + j_idx[:, None] * C + c_idx[None, :]
            tl.store(dst_ptrs, vals, mask=j_mask[:, None] & c_mask[None, :])
