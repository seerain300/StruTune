import torch
import torch.nn as nn
import triton
import triton.language as tl


# Triton kernel: fill N elements with uniform random in [0, 1).
# Used to initialize random tensors (residual, grad_output, etc.).
@triton.jit
def fill_rand_kernel(out_ptr, N, seed, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    a = 1664525
    c = 1013904223
    m = 1 << 32
    rng = offsets.to(tl.int32) + seed
    rnd = (a * rng + c) % m
    rnd = rnd / m
    tl.store(out_ptr + offsets, rnd, mask=mask)


# Triton kernel: depthwise conv2d 1x7x7, padding=3, groups=C on NCHW input, produce NCHW output.
# x: (B, C, H, W), w: (C, 1, 7, 7), y: (B, C, H_out=H, W_out=W)
@triton.jit
def depthwise_conv2d_1x7x7_nchw_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W,
    pad_h, pad_w,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    w_stride_c, w_stride_kh, w_stride_kw,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_W: tl.constexpr,
):
    pid_nc = tl.program_id(axis=0)
    n = pid_nc // C
    c = pid_nc % C

    H_out = H  # output H equals input H for 1x7x7 with padding=3
    W_out = W  # output W equals input W

    # Each program computes all output positions along W for fixed (n, c)
    for ho in range(H_out):
        # base pointer for this (n, c, ho)
        base_in = x_ptr + n * x_stride_n + c * x_stride_c + ho * x_stride_h
        base_out = y_ptr + n * y_stride_n + c * y_stride_c + ho * y_stride_h

        for wo_start in range(0, W_out, BLOCK_W):
            wo = wo_start + tl.arange(0, BLOCK_W)
            mask = wo < W_out

            # Accumulate across 7x7 kernel
            acc = tl.zeros([BLOCK_W], dtype=tl.float32)
            for kh in range(7):
                h_in = ho + pad_h - kh
                in_h_valid = (h_in >= 0) & (h_in < H)
                if not in_h_valid:
                    continue
                for kw in range(7):
                    w_in = wo + pad_w - kw
                    in_w_valid = (w_in >= 0) & (w_in < W)
                    if not in_w_valid:
                        continue
                    # load x[n, c, h_in, w_in] for vector wo
                    x_ptrs = base_in + (h_in * x_stride_h) + (w_in * x_stride_w)
                    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
                    # load weight w[c, 0, kh, kw]
                    w_ptr_scalar = w_ptr + c * w_stride_c + 0 * w_stride_kh + kh * w_stride_kw
                    w_val = tl.load(w_ptr_scalar)
                    acc += x_vals * w_val

            # store acc to y[n, c, ho, wo]
            y_ptrs = base_out + wo * y_stride_w
            tl.store(y_ptrs, acc, mask=mask)


# Triton kernel: per-channel LayerNorm over spatial dims for NHWC tensor
# Input y_in: (B, H, W, C) NHWC, gamma: (C,), eps: float
# Output y_out: (B, H, W, C)
@triton.jit
def layernorm_nhwc_kernel(
    y_in_ptr, gamma_ptr, y_out_ptr,
    B, H, W, C,
    y_in_stride_n, y_in_stride_h, y_in_stride_w, y_in_stride_c,
    y_out_stride_n, y_out_stride_h, y_out_stride_w, y_out_stride_c,
    eps,
    BLOCK_HW: tl.constexpr,
):
    pid_c = tl.program_id(axis=0)  # per channel
    c = pid_c

    # Reduce sum and sum of squares over (B, H, W)
    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)

    # Iterate over (b, h, w) in chunks
    # We use while loops to cover arbitrary B,H,W
    for b in range(0, B):
        for h in range(0, H):
            for w_start in range(0, W, BLOCK_HW):
                w = w_start + tl.arange(0, BLOCK_HW)
                mask = w < W
                base = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
                vals = tl.load(base, mask=mask, other=0.0)
                # cast to float32 for reductions
                vals_f = vals.to(tl.float32)
                sum_val += tl.sum(vals_f)
                sum_sq += tl.sum(vals_f * vals_f)

    mean = sum_val / (B * H * W)
    var = sum_sq / (B * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    gamma = tl.load(gamma_ptr + c)

    # Normalize and write back
    for b in range(0, B):
        for h in range(0, H):
            for w_start in range(0, W, BLOCK_HW):
                w = w_start + tl.arange(0, BLOCK_HW)
                mask = w < W
                base_in = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
                vals = tl.load(base_in, mask=mask, other=0.0)
                vals_f = vals.to(tl.float32)
                norm = (vals_f - mean) * inv_std
                norm = norm * gamma
                base_out = y_out_ptr + b * y_out_stride_n + h * y_out_stride_h + w * y_out_stride_w + c * y_out_stride_c
                tl.store(base_out, norm, mask=mask)


# Triton kernel: linear projection (batched matvec) for NHWC input
# y_in: (B, H, W, C) NHWC, w: (C_out, C) row-major, y_out: (B, H, W, C_out)
@triton.jit
def linear_matvec_nhwc_kernel(
    y_in_ptr, w_ptr, y_out_ptr,
    B, H, W, C_in, C_out,
    y_in_stride_n, y_in_stride_h, y_in_stride_w, y_in_stride_c,
    w_stride_co, w_stride_ci,
    y_out_stride_n, y_out_stride_h, y_out_stride_w, y_out_stride_c,
    BLOCK_CI: tl.constexpr,
):
    pid_bhw = tl.program_id(axis=0)  # over B*H*W
    pid_co = tl.program_id(axis=1)   # over C_out

    # Map pid_bhw to (b, h, w)
    b = pid_bhw // (H * W)
    rem = pid_bhw % (H * W)
    h = rem // W
    w = rem % W
    c_out = pid_co

    acc = tl.zeros((), dtype=tl.float32)
    for c_start in range(0, C_in, BLOCK_CI):
        c = c_start + tl.arange(0, BLOCK_CI)
        mask = c < C_in

        # Load y_in[b, h, w, c] vector of size BLOCK_CI
        base_in = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
        y_vals = tl.load(base_in, mask=mask, other=0.0)
        y_vals_f = y_vals.to(tl.float32)

        # Load w[c_out, c] vector of size BLOCK_CI
        w_ptrs = w_ptr + c_out * w_stride_co + c * w_stride_ci
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0)
        w_vals_f = w_vals.to(tl.float32)

        # Accumulate dot product
        acc += tl.sum(y_vals_f * w_vals_f, axis=0)

    # Store acc at y_out[b, h, w, c_out]
    base_out = y_out_ptr + b * y_out_stride_n + h * y_out_stride_h + w * y_out_stride_w + c_out * y_out_stride_c
    tl.store(base_out, acc)


# Triton kernel: GELU (tanh approximation) elementwise
@triton.jit
def gelu_tanh_kernel(inp_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(inp_ptr + offsets, mask=mask, other=0.0)
    x_f = x.to(tl.float32)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x_f + 0.044715 * x_f * x_f * x_f)
    tanh_inner = tl.tanh(inner)
    gelu = 0.5 * x_f * (1.0 + tanh_inner)
    tl.store(out_ptr + offsets, gelu, mask=mask)


# Triton kernel: per-channel global L2 norm over (B,H,W) for NHWC tensor
# y_in: (B, H, W, C) NHWC, out_norm: (C,) per-channel norm
@triton.jit
def per_channel_global_l2_nhwc_kernel(
    y_in_ptr, out_norm_ptr,
    B, H, W, C,
    y_in_stride_n, y_in_stride_h, y_in_stride_w, y_in_stride_c,
    BLOCK_HW: tl.constexpr,
):
    pid_c = tl.program_id(axis=0)
    c = pid_c

    sum_sq = tl.zeros((), dtype=tl.float32)
    for b in range(0, B):
        for h in range(0, H):
            for w_start in range(0, W, BLOCK_HW):
                w = w_start + tl.arange(0, BLOCK_HW)
                mask = w < W
                base = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
                vals = tl.load(base, mask=mask, other=0.0).to(tl.float32)
                sum_sq += tl.sum(vals * vals)

    norm = tl.sqrt(sum_sq)
    tl.store(out_norm_ptr + c, norm)


# Triton kernel: elementwise scale and add for GRN: y = x * scale + add
# x: (B, H, W, C) NHWC, scale: (C,), add: (B, H, W, C) same shape NHWC, y: (B, H, W, C)
@triton.jit
def scale_add_nhwc_kernel(
    x_ptr, scale_ptr, add_ptr, y_ptr,
    B, H, W, C,
    x_stride_n, x_stride_h, x_stride_w, x_stride_c,
    y_stride_n, y_stride_h, y_stride_w, y_stride_c,
    BLOCK_HW: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # over B*H*W
    c = tl.program_id(axis=1)    # over C

    # Map pid to (b, h, w)
    b = pid // (H * W)
    rem = pid % (H * W)
    h = rem // W
    w = rem % W

    # Load scale for channel c
    s = tl.load(scale_ptr + c)

    # Load add (add_ptr is NHWC)
    base_add = add_ptr + b * x_stride_n + h * x_stride_h + w * x_stride_w + c * x_stride_c
    add_val = tl.load(base_add)

    # Load x
    base_x = x_ptr + b * x_stride_n + h * x_stride_h + w * x_stride_w + c * x_stride_c
    x_val = tl.load(base_x)

    y_val = x_val * s + add_val
    base_y = y_ptr + b * y_stride_n + h * y_stride_h + w * y_stride_w + c * y_stride_c
    tl.store(base_y, y_val)

    norm_features = per_channel_l2 / (gf_mean + eps)          # shape (1,1,1,C4)

    # x_grn_scaled = x_gelu * norm_features (broadcast), then x_grn = grn_weight * x_grn_scaled + x_gelu
    x_grn_scaled = x_gelu * norm_features  # broadcast over (B,H,W)

    x_grn = grn_weight * x_grn_scaled + x_gelu

    # Store results in a dict with required keys
    return {

        "residual": residual,
        "x_dwconv": x_dwconv,

        "global_features": global_features,  # per-(b,h,w,c) L2; not used downstream

    B, C, H, W,
    pad_h, pad_w,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,

                for kw in range(7):
                    w_in = wo + pad_w - kw
                    in_w_valid = (w_in >= 0) and (w_in < W)
                    if not in_w_valid:
                        continue
                    x_ptrs = base_in + (h_in * x_stride_h) + (w_in * x_stride_w)
                    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
                    w_ptr_scalar = w_ptr + c * w_stride_c + 0 * w_stride_kh + kh * w_stride_kw
                    w_val = tl.load(w_ptr_scalar)
                    acc += x_vals * w_val

            y_ptrs = base_out + wo * y_stride_w
            tl.store(y_ptrs, acc, mask=mask)


# Triton kernel: per-channel LayerNorm over spatial dims for NHWC tensor
# Input y_in: (B, H, W, C) NHWC, gamma: (C,), eps: float
# Output y_out: (B, H, W, C)
@triton.jit
def layernorm_nhwc_kernel(
    y_in_ptr, gamma_ptr, y_out_ptr,
    B, H, W, C,
    y_in_stride_n, y_in_stride_h, y_in_stride_w, y_in_stride_c,
    y_out_stride_n, y_out_stride_h, y_out_stride_w, y_out_stride_c,
    eps,
    BLOCK_HW: tl.constexpr,
):
    pid_c = tl.program_id(axis=0)  # per channel
    c = pid_c

    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)

    for b in range(0, B):
        for h in range(0, H):
            for w_start in range(0, W, BLOCK_HW):
                w = w_start + tl.arange(0, BLOCK_HW)
                mask = w < W
                base = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
                vals = tl.load(base, mask=mask, other=0.0).to(tl.float32)
                sum_val += tl.sum(vals)
                sum_sq += tl.sum(vals * vals)

    mean = sum_val / (B * H * W)
    var = sum_sq / (B * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    gamma = tl.load(gamma_ptr + c)

    for b in range(0, B):
        for h in range(0, H):
            for w_start in range(0, W, BLOCK_HW):
                w = w_start + tl.arange(0, BLOCK_HW)
                mask = w < W
                base_in = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
                vals = tl.load(base_in, mask=mask, other=0.0)
                vals_f = vals.to(tl.float32)
                norm = (vals_f - mean) * inv_std
                norm = norm * gamma
                base_out = y_out_ptr + b * y_out_stride_n + h * y_out_stride_h + w * y_out_stride_w + c * y_out_stride_c
                tl.store(base_out, norm, mask=mask)


# Triton kernel: linear projection (batched matvec) for NHWC input
# y_in: (B, H, W, C) NHWC, w: (C_out, C) row-major, y_out: (B, H, W, C_out)
@triton.jit
def linear_matvec_nhwc_kernel(
    y_in_ptr, w_ptr, y_out_ptr,
    B, H, W, C_in, C_out,
    y_in_stride_n, y_in_stride_h, y_in_stride_w, y_in_stride_c,
    w_stride_co, w_stride_ci,
    y_out_stride_n, y_out_stride_h, y_out_stride_w, y_out_stride_c,
    BLOCK_CI: tl.constexpr,
):
    pid_bhw = tl.program_id(axis=0)  # over B*H*W
    pid_co = tl.program_id(axis=1)   # over C_out

    b = pid_bhw // (H * W)
    rem = pid_bhw % (H * W)
    h = rem // W
    w = rem % W
    c_out = pid_co

    acc = tl.zeros((), dtype=tl.float32)
    for c_start in range(0, C_in, BLOCK_CI):
        c = c_start + tl.arange(0, BLOCK_CI)
        mask = c < C_in

        base_in = y_in_ptr + b * y_in_stride_n + h * y_in_stride_h + w * y_in_stride_w + c * y_in_stride_c
        y_vals = tl.load(base_in, mask=mask, other=0.0).to(tl.float32)

        w_ptrs = w_ptr + c_out * w_stride_co + c * w_stride_ci
        w_vals = tl.load(w_ptrs, mask=mask, other=0.0).to(tl.float32)

        acc += tl.sum(y_vals * w_vals, axis=0)

    base_out = y_out_ptr + b * y_out_stride_n + h * y_out_stride_h + w * y_out_stride_w + c_out * y_out_stride_c
    tl.store(base_out, acc)


# Triton kernel: GELU (tanh approximation) elementwise
@triton.jit
def gelu_tanh_kernel(inp_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(inp_ptr + offsets, mask=mask, other=0.0)
    x_f = x.to(tl.float32)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x_f + 0.044715 * x_f * x_f * x_f)
    tanh_inner = tl.tanh(inner)
    gelu = 0.5 * x_f * (1.0 + tanh_inner)
    tl.store(out_ptr + offsets, gelu, mask=mask)


# Triton kernel: per-channel global L2 norm over (B,H,W) for NHWC tensor
# y_in: (B, H, W, C) NHWC, out_norm: (C,) per-channel norm
@triton.jit
def per_channel_global_l2_nhwc_kernel(
    y_in_ptr, out_norm_ptr,
    B, H, W, C,
    y_in_stride_n, y_in_stride_h, y_in_stride_w, y_in_stride_c,
    BLOCK_HW: tl.constexpr,
):
    pid_c = tl.program_id(axis=0)
    c = pid_c

    sum_sq = tl.zeros((), dtype=tl.float32)
    for b in range(0, B):
        for h in range(0, H):

                vals = tl.load(base, mask=mask, other=0.0).to(tl.float32)
                sum_sq += tl.sum(vals * vals)

    norm = tl.sqrt(sum_sq)
    tl.store(out_norm_ptr + c, norm)
