import torch
import torch.nn as nn
import triton
import triton.language as tl
import math


# 1) Random fill kernels (replace torch.randn, torch.ones)
@triton.jit
def fill_rand_kernel(out_ptr, N, seed, BLOCK: tl.constexpr):
    # Fill N elements with random numbers using LCG. out_ptr: float32*.
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    a = 1664525
    c = 1013904223
    m = 1 << 32
    rng = offsets.to(tl.int32) + seed
    rnd = (a * rng + c) % m
    rnd = rnd / m
    tl.store(out_ptr + offsets, rnd, mask=mask)


# 2) Depthwise Conv2d 1x7x7 on NCHW with padding=3 (groups=C), producing x_dwconv
@triton.jit
def depthwise_conv2d_1x7x7_nchw_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W,
    pad_h, pad_w,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    w_stride_c, w_stride_kh, w_stride_kw,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_C: tl.constexpr,
):
    # One program per (n, c) and iterate over output spatial positions
    pid = tl.program_id(axis=0)
    n = pid // C
    c = pid % C

    # Output spatial dims for kernel 1x7, padding 3 -> H_out = H + 2, W_out = W - 4
    H_out = H + 2 * pad_h - 1
    W_out = W + 2 * pad_w - 7

    for ho in range(0, H_out):
        for wo in range(0, W_out):
            acc = 0.0
            # Fixed kernel: kh=0, kw in 0..6
            hi = ho + pad_h  # pad_h=3
            for kw in range(0, 7):
                wi = wo + pad_w - kw  # pad_w=3
                in_bounds = (hi >= 0) & (hi < H) & (wi >= 0) & (wi < W)
                x_off = n * x_stride_n + c * x_stride_c + hi * x_stride_h + wi * x_stride_w
                w_off = c * w_stride_c + 0 * w_stride_kh + kw * w_stride_kw  # kh fixed at 0
                val = tl.load(x_ptr + x_off, mask=in_bounds, other=0.0)
                wval = tl.load(w_ptr + w_off)
                acc += val * wval
            y_off = n * y_stride_n + c * y_stride_c + ho * y_stride_h + wo * y_stride_w
            tl.store(y_ptr + y_off, acc)


# 3) LayerNorm over (H, W) per channel on NHWC input:
# Triton kernels for per-channel mean/var reductions and normalization. We implement these as Triton kernels that reduce over H*W for each (B, C).
@triton.jit
def per_channel_mean_hw_kernel(x_ptr, mean_ptr,
                                B, H, W, C,
                                x_stride_b, x_stride_h, x_stride_w, x_stride_c,
                                BLOCK: tl.constexpr):
    # Each program handles one (b, c) and computes sum over H*W
    pid = tl.program_id(axis=0)
    b = pid // C
    c = pid % C
    sum_val = 0.0
    for h in range(0, H):
        for w in range(0, W):
            x_off = b * x_stride_b + h * x_stride_h + w * x_stride_w + c * x_stride_c
            x_val = tl.load(x_ptr + x_off)
            sum_val += x_val
    mean = sum_val / (H * W)
    # Store as 1-element tensor at (b, c)
    tl.store(mean_ptr + b * C + c, mean)


@triton.jit
def per_channel_var_hw_kernel(x_ptr, mean_ptr, var_ptr,
                               B, H, W, C,
                               x_stride_b, x_stride_h, x_stride_w, x_stride_c,
                               BLOCK: tl.constexpr):
    # Each program handles one (b, c) and computes variance over H*W
    pid = tl.program_id(axis=0)
    b = pid // C
    c = pid % C
    mean = tl.load(mean_ptr + b * C + c)
    sum_sq = 0.0
    for h in range(0, H):
        for w in range(0, W):
            x_off = b * x_stride_b + h * x_stride_h + w * x_stride_w + c * x_stride_c
            x_val = tl.load(x_ptr + x_off)
            sum_sq += x_val * x_val
    var = sum_sq / (H * W) - mean * mean
    tl.store(var_ptr + b * C + c, var)


# 4) Batched matvec for linear projection: X: (B,H,W,C) -> Y: (B,H,W,C4), W: (C4,C)
# Implement as Triton kernel with one program per (b,h,w,f) and loop over C in chunks. In original, X is (B,H,W,C) and Wt is (C,C4). We can use Wt.
@triton.jit
def batched_matvec_nhwcp_kernel(
    X_ptr, Wt_ptr, Y_ptr,
    B, H, W, C, C4,
    X_stride_b, X_stride_h, X_stride_w, X_stride_c,
    Wt_stride_f, Wt_stride_c,  # Wt: (C, C4) -> strides for f and c
    Y_stride_b, Y_stride_h, Y_stride_w, Y_stride_c,
    BLOCK_C: tl.constexpr,
):
    # One program per (b, h, w, f) where f in 0..C4-1
    pid = tl.program_id(axis=0)
    N_hw = H * W
    bh = pid // C4
    f = pid % C4
    b = bh // N_hw
    hw = bh % N_hw
    h = hw // W
    w = hw % W

    acc = 0.0
    for c0 in range(0, C, BLOCK_C):
        offs = c0 + tl.arange(0, BLOCK_C)
        mask = offs < C
        X_off = b * X_stride_b + h * X_stride_h + w * X_stride_w + offs * X_stride_c
        x_vals = tl.load(X_ptr + X_off, mask=mask, other=0.0)
        W_off = offs * Wt_stride_f + f * Wt_stride_c
        w_vals = tl.load(Wt_ptr + W_off, mask=mask, other=0.0)
        acc += tl.sum(x_vals * w_vals, axis=0)
    Y_off = b * Y_stride_b + h * Y_stride_h + w * Y_stride_w + f * Y_stride_c
    tl.store(Y_ptr + Y_off, acc)


# 5) GELU (tanh approximation) elementwise on input tensor
@triton.jit
def gelu_tanh_kernel(in_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    # Constants
    sqrt_2_over_pi = 0.7978845608028654
    cdf_coeff = 0.044715
    inner = sqrt_2_over_pi * (x + cdf_coeff * x * x * x)
    tanh_inner = tl.tanh(inner)
    gelu = 0.5 * x * (1.0 + tanh_inner)
    tl.store(out_ptr + offsets, gelu, mask=mask)


# 6) Elementwise scale-add broadcast kernels (used for GRN scaling and addition)
@triton.jit
def scale_broadcast_kernel(in_ptr, scale_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    s = tl.load(scale_ptr)  # scalar scale
    y = x * s
    tl.store(out_ptr + offsets, y, mask=mask)


@triton.jit
def add_broadcast_kernel(in_ptr, add_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    a = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(add_ptr)  # scalar add
    y = a + b
    tl.store(out_ptr + offsets, y, mask=mask)
