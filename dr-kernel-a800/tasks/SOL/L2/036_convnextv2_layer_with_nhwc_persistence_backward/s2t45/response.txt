import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _nhwc_layernorm_scale_kernel(
    x_nhwc_ptr,       # *float32, input NHWC layout: (B, H, W, C)
    layernorm_ptr,    # *float32, per-channel scale: (C,)
    out_ptr,          # *float32, output: (B, H, W, C)
    B: tl.int32,
    H: tl.int32,
    W: tl.int32,
    C: tl.int32,
    eps: tl.float32,
):
    # program over (b, h)
    b = tl.program_id(0)
    h = tl.program_id(1)

    # running sum and sum of squares across channels for each (b, h)
    sum_x = 0.0
    sum_x2 = 0.0

    # iterate over W (each program handles one (b, h) and all W)
    for w in range(0, W):
        # loop over channels in chunks
        # Note: NHWC layout means for fixed (b, h, w) and varying c, address = base + c*stride_c.
        # Base for (b, h, w): b*H*W* C + h*W* C + w* C
        base = b * H * W * C + h * W * C + w * C
        # Reduction over channels to compute mean and var
        for c0 in range(0, C, 16):
            offs = c0 + tl.arange(0, 16)
            mask = offs < C
            # pointer arithmetic for NHWC layout
            ptrs = base + offs
            x_vals = tl.load(x_nhwc_ptr + ptrs, mask=mask, other=0.0)
            # reduce within this chunk to accumulate sum and sum of squares
            # sum_x += sum(x), sum_x2 += sum(x^2)
            sum_x += tl.sum(x_vals, axis=0)
            sum_x2 += tl.sum(x_vals * x_vals, axis=0)

        # compute mean and variance for this (b, h, w)
        C_f = tl.float32(C)
        mean = sum_x / C_f
        var = sum_x2 / C_f - mean * mean
        inv_std = 1.0 / tl.sqrt(var + eps)

        # now write normalized and scaled output for all channels
        for c0 in range(0, C, 16):
            offs = c0 + tl.arange(0, 16)
            mask = offs < C
            in_ptrs = base + offs
            x_vals = tl.load(x_nhwc_ptr + in_ptrs, mask=mask, other=0.0)
            scale = tl.load(layernorm_ptr + offs, mask=mask, other=1.0)  # layernorm_weight[c]
            y_vals = (x_vals - mean) * inv_std
            y_vals = y_vals * scale
            out_ptrs = b * H * W * C + h * W * C + w * C + offs
            tl.store(out_ptr + out_ptrs, y_vals, mask=mask)


@triton.jit
def _gelu_tanh_kernel(
    x_expanded_ptr,   # *float32, input NCHW layout: (B, C, H, W)
    out_ptr,          # *float32, output: (B, C, H, W)
    B: tl.int32,
    C: tl.int32,
    H: tl.int32,
    W: tl.int32,
):
    # 4D grid (B, C, H, W)
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    w = tl.program_id(3)

    # NCHW layout: address = b*C*H*W + c*H*W + h*W + w
    in_ptr = b * C * H * W + c * H * W + h * W + w
    x_val = tl.load(x_expanded_ptr + in_ptr)
    x_f = x_val.to(tl.float32)

    # constants
    sqrt_2_over_pi = 0.7978845608028654  # sqrt(2/pi)
    cdf_coeff = 0.044715

    # tanh approximation: tanh(u) = (e^{2u} - 1)/(e^{2u} + 1)
    u = sqrt_2_over_pi * (x_f + cdf_coeff * x_f * x_f * x_f)
    e2u = tl.exp(2.0 * u)
    tanh_u = (e2u - 1.0) / (e2u + 1.0)

    gelu = 0.5 * x_f * (1.0 + tanh_u)

    out_ptr = b * C * H * W + c * H * W + h * W + w
    tl.store(out_ptr + out_ptr, gelu)  # Note: pointer arithmetic for out is same as in

@triton.jit
def _gelu_tanh_kernel(
    x_expanded_ptr,   # *float32, input NCHW: (B, C, H, W)
    out_ptr,          # *float32, output: (B, C, H, W)
    B: tl.int32,
    C: tl.int32,
    H: tl.int32,
    W: tl.int32,
):

    c = tl.program_id(1)
    h = tl.program_id(2)
    w = tl.program_id(3)

    cdf_coeff = 0.044715

    tl.store(out_ptr + out_ptr, gelu)  # Correct address computation

def _identity_copy_kernel(  # trivial Triton kernel: copy input to output
    src_ptr,            # *float32
    out_ptr,            # *float32
    total_elems: tl.int32,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < total_elems
    x = tl.load(src_ptr + offsets, mask=mask, other=0.0)
    tl.store(out_ptr + offsets, x, mask=mask)

    eps = 1e-6
    drop_path_prob = 0.1

    # Weights
    dwconv_weight = torch.randn(C, 1, 7, 7, device=device) * (1.0 / 49) ** 0.5
    layernorm_weight = torch.ones(C, device=device) + torch.randn(C, device=device) * 0.01
    pwconv1_weight = torch.randn(C * 4, C, device=device) * (2.0 / C) ** 0.5
    grn_weight = torch.zeros(1, 1, 1, C * 4, device=device) + torch.randn(1, 1, 1, C * 4, device=device) * 0.01
    pwconv2_weight = torch.randn(C, C * 4, device=device) * (2.0 / C * 4) ** 0.5

    # Inputs
    residual = torch.randn(B, C, H, W, device=device) * 0.1
    grad_output = torch.randn(B, C, H, W, device=device)

    # Drop mask
    drop_mask = (torch.rand(B, 1, 1, 1, device=device) > drop_path_prob).float()

    # Forward intermediates (not used in Triton path)
    with torch.no_grad():
        x_dwconv = F.conv2d(residual, dwconv_weight, padding=3, groups=C)
        x_nhwc = x_dwconv.permute(0, 2, 3, 1)
        mean = x_nhwc.mean(-1, keepdim=True)
        var = ((x_nhwc - mean) ** 2).mean(-1, keepdim=True)
        x_normalized = (x_nhwc - mean) / torch.sqrt(var + eps)
        x_ln = x_normalized * layernorm_weight

        x_expanded = x_ln @ pwconv1_weight.t()
        sqrt_2_over_pi = 0.7978845608028654
        cdf_coeff = 0.044715
        inner = sqrt_2_over_pi * (x_expanded + cdf_coeff * x_expanded.pow(3))
        x_gelu = 0.5 * x_expanded * (1.0 + torch.tanh(inner))

        global_features = torch.norm(x_gelu, p=2, dim=(1, 2), keepdim=True)
        gf_mean = global_features.mean(dim=-1, keepdim=True)
        norm_features = global_features / (gf_mean + eps)
        x_grn_scaled = x_gelu * norm_features
        x_grn = grn_weight * x_grn_scaled + x_gelu

    return {
        "grad_output": grad_output,
        "residual": residual,
        "x_dwconv": x_dwconv,
        "x_nhwc": x_nhwc,
        "mean": mean,
