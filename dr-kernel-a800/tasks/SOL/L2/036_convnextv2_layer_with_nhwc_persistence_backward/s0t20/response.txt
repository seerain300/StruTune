    OUT_ptr,            # *float32, output tensor flattened as [B, C, H, 1]

    C: tl.constexpr,    # int
    H: tl.constexpr,    # int
    W: tl.constexpr,    # int

):

    MEAN_ptr,           # *float32, output means [B, C, H] flattened
    VAR_ptr,            # *float32, output vars [B, C, H] flattened

    C: tl.constexpr,    # int
    H: tl.constexpr,    # int
    W: tl.constexpr,    # int

):

    B: tl.constexpr,    # int
    C: tl.constexpr,    # int
    H: tl.constexpr,    # int
    W: tl.constexpr,    # int
