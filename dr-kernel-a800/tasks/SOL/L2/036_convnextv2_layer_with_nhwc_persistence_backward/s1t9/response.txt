# 3) Triton kernel: LayerNorm over channels for each (B,H,W) on NHWC input/output
@triton.jit
def layernorm_nchw_triton(
    input_ptr,      # *float32, (B, H, W, C) NHWC
    weight_ptr,     # *float32, (C,)
    output_ptr,     # *float32, (B, H, W, C)
    B, H, W, C,
    input_stride_b, input_stride_h, input_stride_w, input_stride_c,
    output_stride_b, output_stride_h, output_stride_w, output_stride_c,
    weight_stride_c,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr
):
    # Grid: (B, H, W)
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    # First pass: compute mean over channels
    sum_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        # sum_val += tl.sum(x, axis=0)
        # Triton reduction: use tl.sum over vector
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / C

    # Second pass: compute var over channels
    var_val = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)
    var = var_val / C
    std = tl.sqrt(var + eps)

    # Third pass: normalize and apply per-channel weight, write output
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        in_off = b * input_stride_b + h * input_stride_h + w * input_stride_w + offs_c * input_stride_c
        x = tl.load(input_ptr + in_off, mask=mask_c, other=0.0)
        gamma = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)
        y = (x - mean) / std
        y = y * gamma
        out_off = b * output_stride_b + h * output_stride_h + w * output_stride_w + offs_c * output_stride_c
        tl.store(output_ptr + out_off, y, mask=mask_c)


# 4) Triton kernel: GELU (tanh approximation) elementwise over vector of length N
@triton.jit
def gelu_tanh_triton(X_ptr, Y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
    tanh_val = tl.tanh(inner)
    y = 0.5 * x * (1.0 + tanh_val)
    tl.store(Y_ptr + offs, y, mask=mask)


# 5) Triton kernel: Global Response Norm (GRN) per (B,H,W)
# Compute global_features = ||x_gelu||_2 over channels C4 -> shape (B,H,W,1)
# Then norm_features = global_features / (gf_mean + eps), where gf_mean is per (B,H,W) scalar.
# Finally x_grn = grn_weight * (x_gelu * norm_features) + x_gelu
# We need to compute per (b,h,w): reduce over C4 features, then elementwise combine.
@triton.jit
def grn_per_bhw_triton(
    x_gelu_ptr,        # *float32, (B,H,W,C4) NHWC per (b,h,w), flattened as (B,H,W,C4)
    norm_features_ptr, # *float32, (B,H,W,1)
    B, H, W, C4,
    x_stride_b, x_stride_h, x_stride_w, x_stride_c,
    eps: tl.constexpr,
    BLOCK_C4: tl.constexpr
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)

    sum_sq = tl.zeros((), dtype=tl.float32)
    for c0 in range(0, C4, BLOCK_C4):
        offs_c = c0 + tl.arange(0, BLOCK_C4)
        mask_c = offs_c < C4
        in_off = b * x_stride_b + h * x_stride_h + w * x_stride_w + offs_c * x_stride_c
        x = tl.load(x_gelu_ptr + in_off, mask=mask_c, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)
    norm = tl.sqrt(sum_sq)
    mean = norm  # scalar per (b,h,w)
    inv = 1.0 / (mean + eps)
    # Store norm_features as (B,H,W,1)
    out_off = b * x_stride_b + h * x_stride_h + w * x_stride_w
    tl.store(norm_features_ptr + out_off, inv)
