        base_bc = (b * C + c) * (H * W)
        idx = base_bc + h * W + w
        vals = tl.load(x_ptr + idx, mask=mask, other=0.0)
        acc += tl.sum(vals, axis=0)
    tl.store(sum_ptr + c, acc)


@triton.jit
def per_channel_mean_kernel(sum_ptr, mean_ptr, B, C, H, W, eps, BLOCK: tl.constexpr):
    # mean_ptr[c] = sum_ptr[c] / (B*H*W)
    c = tl.program_id(axis=0)
    denom = B * H * W
    sum_c = tl.load(sum_ptr + c)
    mean = sum_c / denom
    tl.store(mean_ptr + c, mean)


@triton.jit
def per_channel_var_kernel(sum_ptr, mean_ptr, var_ptr, B, C, H, W, eps, BLOCK: tl.constexpr):
    # var_ptr[c] = sum_ptr[c] / (B*H*W) - mean_ptr[c]^2
    c = tl.program_id(axis=0)
    denom = B * H * W
    sum_c = tl.load(sum_ptr + c)
    mean_c = tl.load(mean_ptr + c)
    var = sum_c / denom - mean_c * mean_c
    tl.store(var_ptr + c, var)


@triton.jit
def per_channel_normalize_nhwcn_kernel(x_ptr, mean_ptr, var_ptr, y_ptr,
                                       B, C, H, W,
                                       x_stride_b, x_stride_h, x_stride_w, x_stride_c,
                                       y_stride_b, y_stride_h, y_stride_w, y_stride_c,
                                       BLOCK: tl.constexpr):
    # y[b,h,w,c] = (x[b,h,w,c] - mean[c]) / sqrt(var[c] + eps)
    # Grid over (B*C, H*W)
    pid = tl.program_id(axis=0)
    c = pid % C
    pos = tl.program_id(axis=1)
    b = pos // (H * W)
    rem = pos % (H * W)
    h = rem // W
    w = rem % W
    mean_c = tl.load(mean_ptr + c)
    var_c = tl.load(var_ptr + c)
    std = tl.sqrt(var_c + eps)
    # Load x[b,h,w,c]
    x_off = b * x_stride_b + h * x_stride_h + w * x_stride_w + c * x_stride_c
    x_val = tl.load(x_ptr + x_off)
    y_val = (x_val - mean_c) / std
    # Store y[b,h,w,c] to output y_ptr, assume NHWC layout for y
    y_off = b * y_stride_b + h * y_stride_h + w * y_stride_w + c * y_stride_c
    tl.store(y_ptr + y_off, y_val)


@triton.jit
def gemv_per_channel_kernel(x_ptr, w_ptr, out_ptr,
                             B, C, H, W, F_in, F_out,
                             x_stride_b, x_stride_h, x_stride_w, x_stride_c,
                             w_stride_f_in, w_stride_f_out,
                             out_stride_b, out_stride_h, out_stride_w, out_stride_c,
                             BLOCK: tl.constexpr):
    # Compute out[b,h,w,f_out] = sum_c x[b,h,w,c] * w[f_out, c] for each (b,h,w)
    # Grid: (B*H*W, F_out)
    pid = tl.program_id(axis=0)
    f_out = tl.program_id(axis=1)
    total = B * H * W
    b = pid // (H * W)
    rem = pid % (H * W)
    h = rem // W
    w = rem % W
    acc = 0.0
    for c in range(0, C, BLOCK):
        offs_c = c + tl.arange(0, BLOCK)
        mask = offs_c < C
        x_off = b * x_stride_b + h * x_stride_h + w * x_stride_w + offs_c * x_stride_c
        x_vals = tl.load(x_ptr + x_off, mask=mask, other=0.0)
        w_off = f_out * w_stride_f_out + offs_c * w_stride_f_in
        w_vals = tl.load(w_ptr + w_off, mask=mask, other=0.0)
        acc += tl.sum(x_vals * w_vals, axis=0)
    out_off = b * out_stride_b + h * out_stride_h + w * out_stride_w + f_out * out_stride_c
    tl.store(out_ptr + out_off, acc)


@triton.jit
def gelu_approx_kernel(x_ptr, y_ptr, N, BLOCK: tl.constexpr):
    # GELU(x) = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    # y = gelu(x) elementwise over N floats
    for start in range(0, N, BLOCK):
        offs = start + tl.arange(0, BLOCK)
        mask = offs < N
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        sqrt_2_over_pi = 0.7978845608028654
        inner = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
        y = 0.5 * x * (1.0 + tl.tanh(inner))
        tl.store(y_ptr + offs, y, mask=mask)
