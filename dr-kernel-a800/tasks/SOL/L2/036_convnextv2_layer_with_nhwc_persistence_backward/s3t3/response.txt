@triton.jit
def conv2d_depthwise_kernel(

    weight_ptr,          # *f32, [C, 1, 7, 7]

    pid_h = tl.program_id(1)
    pid_wblk = tl.program_id(2)

    b = pid_bc // C
    c = pid_bc % C
    h_out = pid_h

    w_start = pid_wblk * BLOCK_W
    w_offsets = w_start + tl.arange(0, BLOCK_W)
    mask_w = w_offsets < W_out

@triton.jit
def linear_matmul_kernel(
    x_ptr,               # *f32, [B, C, H, W] flattened to [B, K] where K=C*H*W
    weight_ptr,          # *f32, [N, K] where N is output channels (e.g., 4*C)
    out_ptr,             # *f32, [B, N, H, W] flattened to [B, N, K]
    B: tl.constexpr, K: tl.constexpr, N: tl.constexpr,
    H_out: tl.constexpr, W_out: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Grid: (B, N, ceil_div(K, BLOCK_K))
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_kblk = tl.program_id(2)

    k_start = pid_kblk * BLOCK_K
    k_offsets = k_start + tl.arange(0, BLOCK_K)
    mask_k = k_offsets < K

    # Load x[b, :] vector block
    x_base = pid_b * K + k_offsets
    x_vec = tl.load(x_ptr + x_base, mask=mask_k, other=0.0)

    # Load weight[n, :] vector block
    w_base = pid_n * K + k_offsets
    w_vec = tl.load(weight_ptr + w_base, mask=mask_k, other=0.0)

    # Accumulate dot product
    acc = tl.zeros([BLOCK_K], dtype=tl.float32)
    for kk in range(BLOCK_K):
        kk_i = k_start + kk
        acc[kk] = tl.sum(x_vec[kk] * w_vec[kk])

    # Store into out[b, n, k]
    out_base = pid_b * N * K + pid_n * K + k_offsets
    tl.store(out_ptr + out_base, acc, mask=mask_k)


@triton.jit
def gelu_tanh_kernel(
    x_ptr,               # *f32, input [B, C, H, W]
    out_ptr,             # *f32, output [B, C, H, W]
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    BLOCK_W: tl.constexpr,
):
    # Implement GELU tanh approximation: y = 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    sqrt_2_over_pi = 0.7978845608028654  # sqrt(2/pi)

    pid_bc = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_wblk = tl.program_id(2)

    b = pid_bc // C
    c = pid_bc % C
    h = pid_h

    w_start = pid_wblk * BLOCK_W
    w_offsets = w_start + tl.arange(0, BLOCK_W)
    mask = w_offsets < W

    for wi in range(BLOCK_W):
        idx = b * C * H * W + c * H * W + h * W + (w_start + wi)
        x_val = tl.load(x_ptr + idx, mask=mask, other=0.0)
        x3 = x_val * x_val * x_val
        inner = sqrt_2_over_pi * (x_val + 0.044715 * x3)
        tanh_val = tl.tanh(inner)
        y = 0.5 * x_val * (1.0 + tanh_val)
        tl.store(out_ptr + idx, y, mask=mask)


@triton.jit
def norm_mean_scale_kernel(
    x_ptr,               # *f32, input [B, C, H, W]
    out_ptr,             # *f32, output [B, C, H, W]

    pid_b = tl.program_id(0)

    for c in range(C):
        for h in range(H):
            for w in range(W):
                val = tl.load(x_ptr + pid_b * C * H * W + c * H * W + h * W + w)
                sum_sq += val * val

    norm = tl.sqrt(sum_sq)
    mean_norm = 0.0  # placeholder; we need to pass mean_ptr to each (b, h, w)
    # Compute mean over H*W for this batch
    count = H * W
    for h in range(H):
        for w in range(W):
            mean_norm += tl.load(mean_ptr + pid_b * H * W + h * W + w)
    mean_norm = mean_norm / (H * W)

    inv = 1.0 / (mean_norm + eps)

@triton.jit
def conv_transpose2d_groups_kernel(
    x_ptr,               # *f32, [B, C, H, W] input
    weight_ptr,          # *f32, [C, 1, 7, 7] (groups=C)
    out_ptr,             # *f32, [B, C, H, W] output
    B: tl.constexpr, C: tl.constexpr, H: tl.constexpr, W: tl.constexpr,
    PAD_H: tl.constexpr, PAD_W: tl.constexpr,
    STRIDE_H: tl.constexpr, STRIDE_W: tl.constexpr,
    BLOCK_W: tl.constexpr,
):
    # This is a general conv-transpose2d for groups=C. We implement one output element (b,c,h,w) per program
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w = tl.program_id(3)

    # Initialize accumulator
    acc = tl.zeros((), dtype=tl.float32)

    # For each kernel (kh, kw), compute contributing input positions (h_in, w_in) such that:
    # h = h_in * STRIDE_H - PAD_H + kh
    # w = w_in * STRIDE_W - PAD_W + kw
    # We iterate over kh,kw from 0..6
    for kh in range(7):
        for kw in range(7):
            h_in = pid_h + PAD_H - kh
            w_in = pid_w + PAD_W - kw
            # Check bounds
            in_bounds = (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)
            # If in-bounds, gather input at (b, pid_c, h_in, w_in)
            if in_bounds:
                base_in = pid_b * C * H * W + pid_c * H * W + h_in * W + w_in
                val_in = tl.load(x_ptr + base_in)
                weight_idx = pid_c * 49 + kh * 7 + kw
                w_val = tl.load(weight_ptr + weight_idx)
                acc += val_in * w_val

    base_out = pid_b * C * H * W + pid_c * H * W + pid_h * W + pid_w
    tl.store(out_ptr + base_out, acc)

    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w = tl.program_id(3)

    for kh in range(7):
        for kw in range(7):
            h_in = pid_h + PAD_H - kh
            w_in = pid_w + PAD_W - kw
            in_bounds = (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)
