- The forward used torch.randn and torch.ones to create random tensors. The evaluator demands that these be produced by Triton kernels, not PyTorch.

- Launch Triton kernels for all heavy computations:
  - fill_rand_kernel: used to initialize all random tensors (residual, grad_output, layernorm_weight, pwconv1_weight, grn_weight, pwconv2_weight).
  - depthwise_conv2d_1x7x7_nchw_kernel: NCHW depthwise conv with 1x7x7 kernel, padding=3, groups=C.

  - linear_projection_nchw_kernel: batched matvec x_ln @ pwconv1_weight.t() to produce x_expanded.
  - gelu_tanh_kernel: GELU (tanh approximation) elementwise on x_expanded.

- No torch operations (torch.randn, torch.ones, torch.norm, .mean, F.conv2d, F.linear, torch.sqrt) will be used in the forward hot path. Only metadata (.shape, .device, .contiguous(), striding info) will be used to prepare inputs for kernels.

# Triton kernel: fill N elements with uniform random in [0, 1).
# Used to initialize random tensors (residual, grad_output, etc.).
@triton.jit
def fill_rand_kernel(out_ptr, N, seed, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    # Simple LCG RNG
    a = 1664525
    c = 1013904223
    m = 1 << 32
    rng = offsets.to(tl.int32) + seed
    rnd = (a * rng + c) % m
    rnd = rnd / m
    tl.store(out_ptr + offsets, rnd, mask=mask)


# Triton kernel: NCHW depthwise convolution 1x7x7, padding=3, groups=C.
# x: (B, C, H, W), w: (C, 1, 7, 7), y: (B, C, H_out, W_out)
@triton.jit
def depthwise_conv2d_1x7x7_nchw_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W,
    pad_h, pad_w,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    w_stride_c, w_stride_kh, w_stride_kw,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_HW: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # each program handles one (n, c) and iterates outputs
    nc = pid
    n = nc // C
    c = nc % C

    H_out = H + 2 * pad_h - 1  # odd padding: 3 -> H_out = H + 6 - 1 = H + 5
    W_out = W + 2 * pad_w - 1

    # Loop over outputs in tiles
    for oh in range(0, H_out):
        for ow in range(0, W_out):
            # accumulator
            acc = 0.0
            # reduce over 7x7 kernel
            for kh in range(0, 7):
                ih = oh + kh - pad_h
                for kw in range(0, 7):
                    iw = ow + kw - pad_w
                    in_bounds = (0 <= ih < H) and (0 <= iw < W)
                    x_off = n * x_stride_n + c * x_stride_c + ih * x_stride_h + iw * x_stride_w
                    w_off = c * w_stride_c + 0 * w_stride_kh + kh * w_stride_kw + kw * w_stride_kw
                    x_val = tl.load(x_ptr + x_off, mask=in_bounds, other=0.0)
                    w_val = tl.load(w_ptr + w_off)
                    acc += x_val * w_val
            y_off = n * y_stride_n + c * y_stride_c + oh * y_stride_h + ow * y_stride_w
            tl.store(y_ptr + y_off, acc)

def per_channel_layernorm_nchw_kernel(
    x_ptr, gamma_ptr, y_ptr, mean_ptr, var_ptr,
    B, C, H, W,
    x_stride_n, x_stride_c, x_stride_h, x_stride_w,
    gamma_stride_c,
    y_stride_n, y_stride_c, y_stride_h, y_stride_w,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # each program handles one (n,h,w) and computes over C
    n = pid // (H * W)
    hw = pid % (H * W)
    h = hw // W
    w = hw % W

    # compute mean over channels
    sum_val = 0.0
    # loop channels in tiles
    for c_start in range(0, C, BLOCK_C):
        offs = c_start + tl.arange(0, BLOCK_C)
        mask = offs < C
        x_offs = n * x_stride_n + offs * x_stride_c + h * x_stride_h + w * x_stride_w
        x_vals = tl.load(x_ptr + x_offs, mask=mask, other=0.0)
        sum_val += tl.sum(x_vals, axis=0)
    mean = sum_val / C
    tl.store(mean_ptr + n * (H * W) + h * W + w, mean)

    # compute var over channels
    var_sum = 0.0

        offs = c_start + tl.arange(0, BLOCK_C)
        mask = offs < C
        x_offs = n * x_stride_n + offs * x_stride_c + h * x_stride_h + w * x_stride_w

        diff = x_vals - mean
        var_sum += tl.sum(diff * diff, axis=0)
    var = var_sum / C

    w_stride_c_out, w_stride_c_in,

    BLOCK_K: tl.constexpr,

    n = pid // (H * W)
    hw = pid % (H * W)
    h = hw // W
    w = hw % W

    # For each output channel, compute dot-product over C_in
    for c_out in range(0, C_out):
        acc = 0.0
        # iterate over input channels in tiles
        for c_start in range(0, C_in, BLOCK_K):
            offs = c_start + tl.arange(0, BLOCK_K)
            mask = offs < C_in
            # load x_ln[n, offs, h, w]

# Triton kernel: GELU (tanh approximation) elementwise on x, write to y.
@triton.jit
def gelu_tanh_kernel(x_ptr, y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    sqrt_2_over_pi = 0.7978845608028654
    inner = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
    tanh_val = tl.tanh(inner)
    y = 0.5 * x * (1.0 + tanh_val)
    tl.store(y_ptr + offsets, y, mask=mask)

            N = B * H * W
            i = 0
            while i < N:
