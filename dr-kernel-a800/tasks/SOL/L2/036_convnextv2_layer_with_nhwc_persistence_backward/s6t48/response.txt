Implement this in Triton using a direct im2col-like convolution: for each output (b, c, ho, wo), loop over kh, kw, and input channel c_in, accumulate dwconv_weight[c, 0, kh, kw] * residual[b, c_in, ho+kh, wo+kw]. This matches groups=C depthwise conv.
  - Permute to NHWC: x_dwconv.permute(0, 2, 3, 1). We’ll keep NHWC as NCHW contiguous with permute; Triton kernels operate on it without any torch reduction.
  - LayerNorm over NHWC (B, H, W, C): For each (b, h, w), reduce across C to compute mean and variance, normalize, and scale by layernorm_weight (per channel). Implement two Triton kernels: reduction (sum, sum of squares) and normalization+scaling.
  - GELU (tanh approximation): Elementwise nonlinearity on x_expanded (B, 4*C, H, W). Triton kernel tiled over spatial dims.
  - Global norm reduction: Compute per-(b, c) L2 norm across (H, W) of x_gelu_out. Triton reduction kernel over tiles, producing per-(b, c) norms.
  - Drop mask scaling: Elementwise multiply by keep_prob = 1 - drop_path_prob. Triton kernel.

Plan
- Implement Triton kernels:
  1) conv2d_depthwise_groupsC_kernel: Produces x_dwconv_out (B, C, H_out, W_out) where H_out = H + 2*padding, W_out = W + 2*padding. Weight is (C, 1, 7, 7). For each output (b, c, ho, wo), accumulate over kh,kw and c_in: x_out[b,c,ho,wo] += dwconv_weight[c,0,kh,kw] * x_in[b,c_in,ho+kh,wo+kw].
  2) layernorm_reduce_nhwc_kernel: Compute per-(b,h,w) sum and sumsq across C for NHWC tensor. Output sum[B,H,W] and sumsq[B,H,W].
  3) layernorm_norm_nhwc_kernel: Compute mean = sum/C, var = sumsq/C - mean^2, inv_std = 1/sqrt(var + eps). Normalize and scale by layernorm_weight per channel, write out.
  4) gelu_pointwise_kernel: Elementwise GELU (tanh approximation) on x_expanded (B, C4, H, W), producing x_gelu_out. Tiled over HW.
  5) reduce_global_norm_kernel: Per-(b, c4) L2 norm across (H, W) of x_gelu_out. Writes per-(b, c4) norm to a 1D tensor.
  6) apply_scale_kernel: Elementwise scaling: x_scaled = x_gelu * scale[b,c4].
  7) random_normal_triton: Optional RNG helper to generate random tensors if needed; not used here because inputs are provided by the environment.

- Forward:
  - Launch conv2d_depthwise_groupsC_kernel to produce x_dwconv.
  - Permute to NHWC using .permute (no torch reduction).
  - Launch layernorm_reduce_nhwc_kernel to compute sum/sumsq.
  - Launch layernorm_norm_nhwc_kernel to produce x_ln.
  - Launch gelu_pointwise_kernel to produce x_gelu.
  - Launch reduce_global_norm_kernel to produce norms (for demonstration).
  - Launch apply_scale_kernel to produce scaled output (for demonstration).

# 1) Triton depthwise conv2d with groups=C, padding=3: out[B, C, H_out, W_out]
@triton.jit
def conv2d_depthwise_groupsC_kernel(
    x_in_ptr,           # *const float, input residual: [B, C, H, W]
    w_ptr,              # *const float, weights: [C, 1, 7, 7]
    out_ptr,            # *float, output: [B, C, H_out, W_out]
    B: tl.int32,
    C: tl.int32,
    H: tl.int32,
    W: tl.int32,
    H_out: tl.int32,
    W_out: tl.int32,
    stride_b: tl.int32,
    stride_c: tl.int32,
    in_stride_h: tl.int32,
    in_stride_w: tl.int32,
    w_stride_c: tl.int32,
    w_stride_kh: tl.int32,
    w_stride_kw: tl.int32,
    out_stride_b: tl.int32,
    out_stride_c: tl.int32,
    out_stride_h: tl.int32,
    out_stride_w: tl.int32,
    K_H: tl.constexpr,  # 7
    K_W: tl.constexpr,  # 7
):
    b = tl.program_id(axis=0)
    c = tl.program_id(axis=1)
    ho = tl.program_id(axis=2)
    wo = tl.program_id(axis=3)
    if (b >= B) or (c >= C) or (ho >= H_out) or (wo >= W_out):
        return

    acc = 0.0
    for kh in range(0, K_H):
        in_h = ho + kh - 3  # padding=3
        for kw in range(0, K_W):
            in_w = wo + kw - 3
            if (0 <= in_h < H) and (0 <= in_w < W):
                base_x = b * stride_b + c * in_stride_h + in_h * in_stride_w
                x_val = tl.load(x_in_ptr + base_x)
            else:
                x_val = 0.0
            # load weight: w_ptr[c, 0, kh, kw]
            base_w = c * w_stride_c + kh * w_stride_kh + kw * w_stride_kw
            w_val = tl.load(w_ptr + base_w)
            acc += x_val * w_val
    tl.store(out_ptr + b * out_stride_b + c * out_stride_c + ho * out_stride_h + wo * out_stride_w, acc)

# 3) Triton LayerNorm normalization + scale: out_ln = ((x_nhwc - mean) / sqrt(var+eps)) * ln_weight
@triton.jit
def layernorm_norm_nhwc_kernel(
    x_nhwc_ptr,         # *const float, NHWC input: [B, H, W, C]
    sum_ptr,            # *const float, sums: [B, H, W]
    sumsq_ptr,          # *const float, sumsqs: [B, H, W]
    ln_weight_ptr,      # *const float, layernorm_weight: [C]
    out_ptr,            # *float, output NHWC: [B, H, W, C]
    B: tl.int32,
    C: tl.int32,
    H: tl.int32,
    W: tl.int32,
    stride_b: tl.int32,
    stride_h: tl.int32,
    stride_w: tl.int32,
    stride_c: tl.int32,
    eps: tl.float32,
):
    b = tl.program_id(axis=0)
    h = tl.program_id(axis=1)
    w = tl.program_id(axis=2)
    if (b >= B) or (h >= H) or (w >= W):
        return

    sum_val = tl.load(sum_ptr + b * H + w)
    sumsq_val = tl.load(sumsq_ptr + b * H + w)
    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    base = b * stride_b + h * stride_h + w * stride_w
    for c in range(0, C):
        x = tl.load(x_nhwc_ptr + base + c * stride_c)
        norm = (x - mean) * inv_std
        ln_w = tl.load(ln_weight_ptr + c)
        y = norm * ln_w
        tl.store(out_ptr + base + c * stride_c, y)


# 4) Triton GELU pointwise: y = 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3))) on x_expanded[B, C4, H, W]
@triton.jit
def gelu_pointwise_kernel(
    x_ptr,               # *const float, input (e.g., x_expanded): [B, C4, H, W]
    y_ptr,               # *float, output: [B, C4, H, W]
    B: tl.int32,
    C4: tl.int32,
    H: tl.int32,
    W: tl.int32,
    stride_b: tl.int32,
    stride_c4: tl.int32,
    stride_h: tl.int32,
    stride_w: tl.int32,
    TILE_HW: tl.constexpr,  # e.g., 1024
):
    pid_b = tl.program_id(axis=0)
    pid_c4 = tl.program_id(axis=1)
    pid_hw = tl.program_id(axis=2)
    if pid_b >= B:
        return
    hw_start = pid_hw * TILE_HW
    idx = hw_start + tl.arange(0, TILE_HW)
    mask = idx < (H * W)
    for c4 in range(0, C4):
        base = pid_b * stride_b + c4 * stride_c4 + idx
        x = tl.load(x_ptr + base, mask=mask, other=0.0)
        # tanh approximation constants
        sqrt_2_over_pi = 0.7978845608028654
        cdf_coeff = 0.044715
        inner = sqrt_2_over_pi * (x + cdf_coeff * x * x * x)
        tanh_inner = tl.tanh(inner)
        y = 0.5 * x * (1.0 + tanh_inner)
        tl.store(y_ptr + base, y, mask=mask)


# 5) Triton global norm reduction: per-(b, c) L2 norm across (H, W) of x_gelu -> norm[B*C]
@triton.jit
def reduce_global_norm_kernel(
    x_ptr,              # *const float, input x_gelu: [B, C, H, W]
    norm_ptr,           # *float, output norms: [B*C]
    B: tl.int32,
    C: tl.int32,
    H: tl.int32,
    W: tl.int32,
    stride_b: tl.int32,
    stride_c: tl.int32,
    stride_h: tl.int32,
    stride_w: tl.int32,
    TILE_HW: tl.constexpr,  # e.g., 1024
):
    pid_b = tl.program_id(axis=0)
    for c in range(0, C):
        sum_sq = 0.0
        base_c = pid_b * stride_b + c * stride_c
        for h in range(0, H):
            for w in range(0, W):
                base = base_c + h * stride_h + w * stride_w
                x = tl.load(x_ptr + base)
                sum_sq += x * x
        norm_val = tl.sqrt(sum_sq)
        tl.store(norm_ptr + pid_b * C + c, norm_val)


# 6) Triton elementwise scaling (apply per-(b,c) scale): out = x * scale
@triton.jit
def apply_scale_kernel(
    x_ptr,              # *const float, input x: [B, C, H, W]
    scale_ptr,          # *const float, scale: [B*C]
    out_ptr,            # *float, output: [B, C, H, W]
    B: tl.int32,
    C: tl.int32,
    H: tl.int32,
    W: tl.int32,
    stride_b: tl.int32,
    stride_c: tl.int32,
    stride_h: tl.int32,
    stride_w: tl.int32,
    TILE_HW: tl.constexpr,  # e.g., 1024
):
    pid_b = tl.program_id(axis=0)
    pid_hw = tl.program_id(axis=1)
    if pid_b >= B:
        return
    hw_start = pid_hw * TILE_HW
    idx = hw_start + tl.arange(0, TILE_HW)
    mask = idx < (H * W)
    for c in range(0, C):
        scale_val = tl.load(scale_ptr + pid_b * C + c)
        base = pid_b * stride_b + c * stride_c + idx
        x = tl.load(x_ptr + base, mask=mask, other=0.0)
        y = x * scale_val
        tl.store(out_ptr + base, y, mask=mask)
