import torch.nn as nn

# Triton kernels
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:

# 1) Generate random normal tensor: out_ptr[B*C] = randn
@triton.jit
def generate_randn_kernel(out_ptr, size: tl.int32):
    pid = tl.program_id(0)
    if pid < size:
        # Triton does not have tl.randn; emulate using tl.multiple_of and simple linear indexing.
        # We assume out_ptr is a flat array of floats.
        val = tl.random((1,))  # simple uniform RNG inside Triton
        tl.store(out_ptr + pid, val)


# 2) Generate ones tensor: out_ptr[B*C] = 1.0
@triton.jit
def generate_ones_kernel(out_ptr, size: tl.int32):
    pid = tl.program_id(0)
    if pid < size:
        tl.store(out_ptr + pid, 1.0)


# 3) Depthwise conv2d with groups=C, padding=3: x_dwconv[B, C, H+6, W+6]
# We implement cross-correlation accumulation directly in Triton:
# For each (b, c, ho, wo), sum over kh,kw: residual[b, c, ho+kh, wo+kw] * dwconv_weight[c, 0, kh, kw]
@triton.jit
def conv2d_depthwise_groupsC_kernel(
    residual_ptr,        # *const float, residual: [B, C, H, W]
    dwconv_ptr,          # *const float, dwconv_weight: [C, 1, 7, 7]
    out_ptr,             # *float, output: [B, C, H+6, W+6]
    B: tl.int32, H: tl.int32, W: tl.int32, Ho: tl.int32, Wo: tl.int32, C: tl.int32,
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_ho = tl.program_id(2)
    pid_wo = tl.program_id(3)
    b = pid_b
    c = pid_c
    ho = pid_ho
    wo = pid_wo

    # For each output position, accumulate over 7x7 filter
    acc = 0.0
    # Loop over kernel size 7x7
    for kh in range(0, 7):
        for kw in range(0, 7):
            in_h = ho + kh - 3  # padding=3
            in_w = wo + kw - 3
            # Validity check
            valid_h = (in_h >= 0) & (in_h < H)
            valid_w = (in_w >= 0) & (in_w < W)
            valid = valid_h & valid_w
            # Compute residual index: (b, c, in_h, in_w)
            residual_idx = (b * C * H * W) + (c * H * W) + (in_h * W) + in_w
            # Load with mask; if invalid, use 0
            # First get weight for this (c, kh, kw)
            dw_idx = (c * (1 * 7 * 7)) + (kh * 7 + kw)
            w_val = tl.load(dwconv_ptr + dw_idx)
            r_val = tl.load(residual_ptr + residual_idx, mask=valid, other=0.0)
            acc += r_val * w_val

    out_idx = (b * C * Ho * Wo) + (c * Ho * Wo) + (ho * Wo) + wo
    tl.store(out_ptr + out_idx, acc)


# 4) Permute NCHW -> NHWC (copy kernel): x_nchw [B, C, H, W] -> x_nhwc [B, H, W, C]
@triton.jit
def permute_nchw_to_nhwc_kernel(
    x_nchw_ptr,          # *const float, input NCHW
    x_nhwc_ptr,          # *float, output NHWC
    B: tl.int32, C: tl.int32, H: tl.int32, W: tl.int32,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)
    pid_c = tl.program_id(3)
    b = pid_b
    h = pid_h
    w = pid_w
    c = pid_c

    # Input index for NCHW: ((b*C + c)*H + h)*W + w
    in_idx = ((b * C + c) * H + h) * W + w
    # Output index for NHWC: ((b*H + h)*W + w)*C + c
    out_idx = ((b * H + h) * W + w) * C + c
    val = tl.load(x_nchw_ptr + in_idx)
    tl.store(x_nhwc_ptr + out_idx, val)


# 5) LayerNorm over NHWC: per (b, h, w) mean/var across C, then normalize and scale by ln_weight
@triton.jit
def layernorm_nhwc_kernel(
    x_nhwc_ptr,          # *const float, input NHWC: [B, H, W, C]
    ln_weight_ptr,       # *const float, layernorm_weight: [C]
    out_ln_ptr,          # *float, output: [B, H, W, C]
    B: tl.int32, H: tl.int32, W: tl.int32, C: tl.int32,
    eps: tl.float32,
    BLOCK_C: tl.constexpr,  # tile size for reduction across C
):
    pid_b = tl.program_id(0)
    pid_hw = tl.program_id(1)
    h = pid_hw // W
    w = pid_hw % W

    base = pid_b * (H * W * C) + h * (W * C) + w * C

    # First pass: sum and sum of squares across C
    sum_x = 0.0
    sum_x2 = 0.0
    for c0 in range(0, C, BLOCK_C):
        offs = c0 + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_nhwc_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_x += tl.sum(x, axis=0)
        sum_x2 += tl.sum(x * x, axis=0)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and scale
    for c0 in range(0, C, BLOCK_C):
        offs = c0 + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_nhwc_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        gamma = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        y = (x - mean) * inv_std
        out = y * gamma
        tl.store(out_ln_ptr + base + offs, out, mask=mask)


# 6) GELU (tanh approximation) pointwise kernel: out_ptr[B*C4*H*W] = GELU(in_ptr)
@triton.jit
def gelu_pointwise_kernel(
    in_ptr,              # *const float, input expanded tensor [B, C4, H, W]
    out_ptr,             # *float, output [B, C4, H, W]
    B: tl.int32, C4: tl.int32, H: tl.int32, W: tl.int32,
    BLOCK_HW: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_c4 = tl.program_id(1)
    pid_hw_block = tl.program_id(2)
    b = pid_b
    c4 = pid_c4
    start = pid_hw_block * BLOCK_HW
    HW = H * W
    offs = start + tl.arange(0, BLOCK_HW)
    mask = offs < HW

    # Compute base index for this (b, c4)
    base = (b * C4 + c4) * HW
    idx = base + offs

    x = tl.load(in_ptr + idx, mask=mask, other=0.0).to(tl.float32)

    # Constants for tanh approximation
    sqrt_2_over_pi = 0.7978845608028654
    cdf_coeff = 0.044715
    inner = sqrt_2_over_pi * (x + cdf_coeff * x * x * x)
    tanh_inner = tl.tanh(inner)
    gelu = 0.5 * x * (1.0 + tanh_inner)

    tl.store(out_ptr + idx, gelu, mask=mask)


# 7) Reduce global L2 norm per (b, c4) over (H, W): out_ptr[b*C4] = sqrt(sum(x^2))
@triton.jit
def reduce_global_norm_kernel(
    x_ptr,               # *const float, input expanded tensor [B, C4, H, W]
    out_norm_ptr,        # *float, output norm [B*C4]
    B: tl.int32, C4: tl.int32, H: tl.int32, W: tl.int32,
    BLOCK_HW: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // C4
    c4 = pid % C4
    HW = H * W
    sum_val = 0.0
    for start in range(0, HW, BLOCK_HW):
        offs = start + tl.arange(0, BLOCK_HW)
        mask = offs < HW
        base = (b * C4 + c4) * HW
        vals = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(vals * vals, axis=0)
    norm = tl.sqrt(sum_val)
    tl.store(out_norm_ptr + pid, norm)


# 8) Apply scale elementwise: out_ptr[B*C4*H*W] = x_ptr[B*C4*H*W] * scale_ptr[b*C4]
@triton.jit
def apply_scale_kernel(
    x_ptr,               # *const float
    out_ptr,             # *float
    scale_ptr,           # *const float, per (b,c4) scale
    B: tl.int32, C4: tl.int32, H: tl.int32, W: tl.int32,
    BLOCK_HW: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_c4 = tl.program_id(1)
    pid_hw_block = tl.program_id(2)
    b = pid_b
    c4 = pid_c4
    HW = H * W
    start = pid_hw_block * BLOCK_HW
    offs = start + tl.arange(0, BLOCK_HW)
    mask = offs < HW
    base = (b * C4 + c4) * HW
    idx = base + offs
    x = tl.load(x_ptr + idx, mask=mask, other=0.0).to(tl.float32)
    scale = tl.load(scale_ptr + (b * C4 + c4))
    y = x * scale
    tl.store(out_ptr + idx, y, mask=mask)


# 9) Drop mask scaling: out_ptr[B*H*W] = grad_output[B*H*W] * keep_prob
@triton.jit
def drop_scale_kernel(
    grad_ptr,            # *const float
    out_ptr,             # *float
    size: tl.int32,      # B*H*W
    keep_prob: tl.float32,
):
    pid = tl.program_id(0)
    if pid < size:
        val = tl.load(grad_ptr + pid)
        tl.store(out_ptr + pid, val * keep_prob)

# 2) Generate ones tensor: out_ptr[size] = 1.0
@triton.jit
def generate_ones_kernel(out_ptr, size: tl.int32):
    pid = tl.program_id(0)
    if pid < size:
        tl.store(out_ptr + pid, 1.0)


# 3) Depthwise conv2d with groups=C, padding=3: x_dwconv[B, C, H+6, W+6]
# We implement cross-correlation directly: For each (b, c, ho, wo), sum over 7x7.
@triton.jit
def conv2d_depthwise_groupsC_kernel(
    residual_ptr,        # *const float, residual: [B, C, H, W] flattened
    dwconv_ptr,          # *const float, dwconv_weight: [C, 1, 7, 7] flattened
    out_ptr,             # *float, output: [B, C, H+6, W+6] flattened
    B: tl.int32, H: tl.int32, W: tl.int32, Ho: tl.int32, Wo: tl.int32, C: tl.int32,
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_ho = tl.program_id(2)
    pid_wo = tl.program_id(3)
    b = pid_b
    c = pid_c
    ho = pid_ho
    wo = pid_wo

    # Accumulator
    acc = 0.0
    # 7x7 kernel
    for kh in range(0, 7):
        for kw in range(0, 7):
            in_h = ho + kh - 3  # padding=3
            in_w = wo + kw - 3
            valid_h = (in_h >= 0) & (in_h < H)
            valid_w = (in_w >= 0) & (in_w < W)
            valid = valid_h & valid_w
            residual_idx = (b * C * H * W) + (c * H * W) + (in_h * W) + in_w
            dw_idx = (c * (1 * 7 * 7)) + (kh * 7 + kw)
            w_val = tl.load(dwconv_ptr + dw_idx)
            r_val = tl.load(residual_ptr + residual_idx, mask=valid, other=0.0)
            acc += r_val * w_val

    out_idx = (b * C * Ho * Wo) + (c * Ho * Wo) + (ho * Wo) + wo
    tl.store(out_ptr + out_idx, acc)


# 4) Permute NCHW -> NHWC: x_nchw [B, C, H, W] -> x_nhwc [B, H, W, C]
@triton.jit
def permute_nchw_to_nhwc_kernel(
    x_nchw_ptr,          # *const float, input NCHW
    x_nhwc_ptr,          # *float, output NHWC
    B: tl.int32, C: tl.int32, H: tl.int32, W: tl.int32,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)
    pid_c = tl.program_id(3)
    b = pid_b
    h = pid_h
    w = pid_w
    c = pid_c

    in_idx = ((b * C + c) * H + h) * W + w
    out_idx = ((b * H + h) * W + w) * C + c
    val = tl.load(x_nchw_ptr + in_idx)
    tl.store(x_nhwc_ptr + out_idx, val)


# 5) LayerNorm over NHWC: per (b, h, w), reduce across C, normalize, and scale by ln_weight
@triton.jit
def layernorm_nhwc_kernel(
    x_nhwc_ptr,          # *const float, input NHWC: [B, H, W, C]
    ln_weight_ptr,       # *const float, layernorm_weight: [C]
    out_ln_ptr,          # *float, output: [B, H, W, C]
    B: tl.int32, H: tl.int32, W: tl.int32, C: tl.int32,
    eps: tl.float32,
    BLOCK_C: tl.constexpr,  # tile for reduction across C
):
    pid_b = tl.program_id(0)
    pid_hw = tl.program_id(1)
    h = pid_hw // W
    w = pid_hw % W

    base = pid_b * (H * W * C) + h * (W * C) + w * C

    # First pass: sum and sum of squares across C
    sum_x = 0.0
    sum_x2 = 0.0
    for c0 in range(0, C, BLOCK_C):
        offs = c0 + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_nhwc_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_x += tl.sum(x, axis=0)
        sum_x2 += tl.sum(x * x, axis=0)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and scale
    for c0 in range(0, C, BLOCK_C):
        offs = c0 + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_nhwc_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        gamma = tl.load(ln_weight_ptr + offs, mask=mask, other=1.0).to(tl.float32)
        y = (x - mean) * inv_std
        out = y * gamma
        tl.store(out_ln_ptr + base + offs, out, mask=mask)
