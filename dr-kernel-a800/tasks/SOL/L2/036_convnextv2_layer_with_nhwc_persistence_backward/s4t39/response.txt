import torch
import triton
import triton.language as tl
import torch.nn as nn


def _randn_triton(shape, dtype=torch.float32, device="cuda"):
    """
    Triton kernel that fills a tensor with standard normal values via tl.rand.
    For simplicity and evaluation, we map tl.rand to uniform random; Triton lacks
    a direct randn. If the environment supports tl.rand and tl.cast, this will
    produce uniform randoms. The original code uses torch.randn; we mimic that
    here to generate random inputs.
    """
    t = torch.empty(shape, dtype=dtype, device=device)
    # Triton expects pointers, so we create a 1D buffer and write to t
    numel = t.numel()
    @triton.jit
    def randn_kernel(out_ptr, n_elements):
        # Each program writes one random element
        pid = tl.program_id(0)
        # bounds check: if pid >= n_elements, skip
        if pid < n_elements:
            # uniform in [0,1); if tl.rand is not available, Triton will not compile.
            # Replace with torch.randn in real scenarios, but here we must use Triton.
            r = tl.rand()
            # store
            tl.store(out_ptr + pid, r)
    grid = (triton.cdiv(numel, 1),)
    randn_kernel[grid](t, numel)
    return t

@triton.jit
def conv_depthwise_groupsC_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H, W, H_out, W_out,
    pad_h, pad_w,
    stride_xB, stride_xC, stride_xH, stride_xW,
    stride_wC, stride_wKH, stride_wKW,
    stride_yB, stride_yC, stride_yH, stride_yW,
):
    """
    Depthwise Conv2d with groups=C:
    y[b, c, oh, ow] = sum_{kh, kw} x[b, c, oh+kh, ow+kw] * w[c, 0, kh, kw]
    Padding: pad_h, pad_w.
    One program computes one output element (b, c, oh, ow).
    """
    b = tl.program_id(0)
    c = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    acc = tl.zeros((), dtype=tl.float32)

    for kh in range(0, 7):
        for kw in range(0, 7):
            h_in = oh + kh - pad_h
            w_in = ow + kw - pad_w
            in_bounds = (h_in >= 0) & (h_in < H) & (w_in >= 0) & (w_in < W)
            x_offset = b * stride_xB + c * stride_xC + h_in * stride_xH + w_in * stride_xW
            x_val = tl.load(x_ptr + x_offset, mask=in_bounds, other=0.0)
            w_offset = c * stride_wC + 0 * stride_wKH + kh * stride_wKW
            w_val = tl.load(w_ptr + w_offset)
            acc += x_val * w_val

    y_offset = b * stride_yB + c * stride_yC + oh * stride_yH + ow * stride_yW
    tl.store(y_ptr + y_offset, acc)


@triton.jit
def reduce_mean_channels_kernel(

    B, H, W, C,
    stride_xB, stride_xH, stride_xW, stride_xC,
    stride_outB, stride_outH, stride_outW,
):
    """
    For NHWC tensor, compute mean over channels per (b, h, w):
    out[b, h, w] = (1/C) * sum_c x[b, h, w, c]
    """
    b = tl.program_id(0)
    h = tl.program_id(1)
    w = tl.program_id(2)
    total = tl.zeros((), dtype=tl.float32)
    for c in range(0, C):
        x_offset = b * stride_xB + h * stride_xH + w * stride_xW + c * stride_xC
        x_val = tl.load(x_ptr + x_offset)
        total += x_val
    mean = total / C
    out_offset = b * stride_outB + h * stride_outH + w * stride_outW
    tl.store(out_ptr + out_offset, mean)

def permute_nchw_to_nhwc_kernel(

@triton.jit
def gelu_tanh_kernel(
    x_ptr, y_ptr,
    N,
    sqrt_2_over_pi: tl.constexpr,
):

    GELU (tanh approximation) on a flat buffer:

    """

        tanh_inner = tl.tanh(inner)
        y_val = 0.5 * x_val * (1.0 + tanh_inner)
        tl.store(y_ptr + i, y_val)

@triton.jit
def batched_gemv_kernel(x_ptr, w_ptr, out_ptr, B, C):
    """
    Compute out[i] = sum_j x[i, j] * w[j] for i in [0, B*C*H*W), j in [0, C)
    This is a batched GEMV over one row per i.
    We assume x_ptr points to a flat buffer of length (B*C*H*W)*C; for each i,
    x_row_offset = i * C. We load w_ptr as a vector of length C and accumulate.
    out_ptr is a flat buffer of length B*C*H*W.
    """
    B_hwC = B * C * H * W
    for i in range(0, B_hwC):
        acc = tl.zeros((), dtype=tl.float32)
        x_row_offset = i * C
        # load w vector
        w_vec = tl.load(w_ptr + tl.arange(0, C))  # vector load of C weights
        for j in range(0, C):
            x_val = tl.load(x_ptr + x_row_offset + j)
            acc += x_val * w_vec[j]
        tl.store(out_ptr + i, acc)

            for b in range(0, B):
                for h in range(0, H):
                    for w in range(0, W):
                        # Load x_ln_row: vector of length C from NHWC
                        x_row = tl.zeros((C,), dtype=tl.float32)
                        for c in range(0, C):
                            nhwc_offset = b * stride_xB + h * stride_xH + w * stride_xW + c * stride_xC
                            x_row[c] = tl.load(x_nhwc_ptr + nhwc_offset)
                        # Accumulator for 4*C outputs
                        acc_vec = tl.zeros((C4,), dtype=tl.float32)
                        # w_ptr is (C4, C). We need to compute dot(x_row, w[:, j]) for each j in [0, C).
                        # For each output o in [0, C4), w[o, :] is the weight vector of length C. Load it and dot.
                        for o in range(0, C4):
                            w_vec = tl.zeros((C,), dtype=tl.float32)
                            for j in range(0, C):
                                w_offset = o * stride_wO + j * stride_wC
                                w_vec[j] = tl.load(w_ptr + w_offset)
                            # dot
                            acc_vec[o] = tl.sum(x_row * w_vec)
                        # Store acc_vec to out[b, o, h, w] for all o
                        # out is (B, C4, H, W)
                        for o in range(0, C4):
                            out_offset = b * stride_outB + o * stride_outH + h * stride_outW + w * stride_outC
                            tl.store(out_ptr + out_offset, acc_vec[o])

        # Launch GEMV kernel
        grid_gmv = (1,)  # one program; it loops over B,H,W
        nhwc_gemv_kernel[grid_gmv](

            x_expanded.stride(0), x_expanded.stride(1), x_expanded.stride(2), x_expanded.stride(3),

        x_gelu = torch.empty_like(x_expanded)
        N_gelu = x_expanded.numel()
        # We need to define a Triton kernel that applies GELU elementwise. Implement using flat pointers.
        x_expanded_flat = x_expanded.reshape(-1)
        x_gelu_flat = x_gelu.reshape(-1)
        gelu_tanh_kernel[N_gelu](x_expanded_flat, x_gelu_flat, N_gelu, 0.7978845608028654)

        # Global features: norm over spatial dims for each (b, c) => sum over H_out and W_out of x_gelu[b, c, h, w]^2
        global_features = torch.empty((B, C4), dtype=torch.float32, device=device)
        # Triton kernel to reduce sum over H and W per (b, c)
        @triton.jit
        def reduce_spatial_sq_kernel(x_ptr, out_ptr,
                                     B, C4, H, W,
                                     stride_xB, stride_xC, stride_xH, stride_xW,
                                     stride_outB, stride_outC):
            total = B * C4
            for i in range(0, total):
                b = i // C4
                c4 = i % C4
                acc = tl.zeros((), dtype=tl.float32)
                for h in range(0, H):
                    for w in range(0, W):
                        offset = b * stride_xB + c4 * stride_xC + h * stride_xH + w * stride_xW
                        val = tl.load(x_ptr + offset)
                        acc += val * val
                out_offset = b * stride_outB + c4 * stride_outC
                tl.store(out_ptr + out_offset, acc)

        grid_reduce = (B, C4)
        reduce_spatial_sq_kernel[grid_reduce](
            x_gelu, global_features,
            B, C4, H_out, W_out,
            x_gelu.stride(0), x_gelu.stride(1), x_gelu.stride(2), x_gelu.stride(3),
            global_features.stride(0), global_features.stride(1),
        )

        # gf_mean = global_features.mean(dim=-1, keepdim=True) -> shape (B, 1)
        gf_mean = torch.empty((B, 1), dtype=torch.float32, device=device)
        # Triton reduction for mean per B
        @triton.jit
        def mean_1d_kernel(in_ptr, out_ptr, N):
            total = N
            sum_all = tl.zeros((), dtype=tl.float32)
            for i in range(0, total):
                sum_all += tl.load(in_ptr + i)
            mean = sum_all / N
            # out_ptr points to a single element (B,)
            tl.store(out_ptr + 0, mean)

        N_mean = B * C4
        mean_1d_kernel[N_mean](global_features.reshape(-1), gf_mean.reshape(-1), N_mean)

        # norm_features = global_features / (gf_mean + eps), broadcast over spatial
        norm_features = torch.empty((B, H_out, W_out, C4), dtype=torch.float32, device=device)
        # Implement elementwise scale: y[b,h,w,c] = global_features[b,c] / (gf_mean[b] + eps)
        # We need to broadcast gf_mean[b] to all (h,w). We'll use Triton to fill norm_features with the per-(b,c) scaled values.
        @triton.jit
        def fill_norm_features_kernel(gf_ptr, gmean_ptr, out_ptr,
                                      B, C4, H, W,
                                      stride_gfB, stride_gfC,
                                      stride_outB, stride_outH, stride_outW, stride_outC):
            total = B * C4
            for i in range(0, total):
                b = i // C4
                c4 = i % C4
                gf_val = tl.load(gf_ptr + b * stride_gfB + c4 * stride_gfC)
                gmean_val = tl.load(gmean_ptr + b)  # gf_mean[b] stored at index b
                scale = gf_val / (gmean_val + eps)
                # Fill out[b, :, :, c4] with scale
                # Loop over H and W
                for h in range(0, H):
                    for w in range(0, W):
                        out_offset = b * stride_outB + h * stride_outH + w * stride_outW + c4 * stride_outC
                        tl.store(out_ptr + out_offset, scale)

        grid_fill = (B, C4)
        fill_norm_features_kernel[grid_fill](
            global_features, gf_mean, norm_features,
            B, C4, H_out, W_out,
            global_features.stride(0), global_features.stride(1),
            norm_features.stride(0), norm_features.stride(1), norm_features.stride(2), norm_features.stride(3),
        )

        # x_grn_scaled = x_gelu * norm_features
        x_grn_scaled = torch.empty_like(x_gelu)
        scale_elementwise_kernel[B * C4 * H_out * W_out](x_gelu, norm_features.reshape(-1), x_grn_scaled.reshape(-1), B * C4 * H_out * W_out)
