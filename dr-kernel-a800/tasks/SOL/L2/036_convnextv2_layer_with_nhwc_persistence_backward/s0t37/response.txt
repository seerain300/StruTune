@triton.jit
def compute_mean_var_w_kernel(
    X_ptr,         # *const float32, input x_dwconv (B, C, H, W) but we pass it flattened (row-major)
    mean_ptr,      # *float32, output means (B*C, H)
    var_ptr,       # *float32, output vars (B*C, H)
    B: tl.int32,   # batch size
    C: tl.int32,   # channels
    H: tl.int32,   # height
    W: tl.int32,   # width
    BLOCK_W: tl.constexpr,  # chunk size for W reduction
):
    # Each program handles one (b, c, h) row and reduces across W
    pid_bc = tl.program_id(axis=0)  # ranges over B*C
    pid_h = tl.program_id(axis=1)   # ranges over H

    # decode b, c from pid_bc
    b = pid_bc // C
    c = pid_bc % C

    # base offset for this (b, c, h)
    # Flattened indexing: index = (((b*C + c)*H + h) * W + w)
    h = pid_h

    # accumulators
    s = 0.0
    s2 = 0.0

    # loop over W in chunks
    for w_start in range(0, W, BLOCK_W):
        w_idx = w_start + tl.arange(0, BLOCK_W)
        mask = w_idx < W
        # load x[b, c, h, w]
        x_ptrs = X_ptr + ((b * C + c) * H + h) * W + w_idx
        x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
        # accumulate
        s += tl.sum(x_vals, axis=0)
        s2 += tl.sum(x_vals * x_vals, axis=0)

    # compute mean and var
    w_f = tl.cast(W, tl.float32)
    mean = s / w_f
    var = s2 / w_f - mean * mean

    # store to mean_ptr[h] and var_ptr[h]
    # Output is (B*C, H), so offset = pid_bc * H + pid_h
    out_offset = pid_bc * H + pid_h
    tl.store(mean_ptr + out_offset, mean)
    tl.store(var_ptr + out_offset, var)


@triton.jit
def linear_matmul_kernel(
    A_ptr,         # *const float32, input A flattened: M = B*C*H*W
    B_ptr,         # *const float32, input B as (K, N) where K=C, N=C4
    C_ptr,         # *float32, output C flattened: M * N
    M: tl.int32,   # total number of rows in A
    N: tl.int32,   # number of columns in output (C4)
    K: tl.int32,   # inner dimension (C)
    BLOCK_M: tl.constexpr,   # tile size along M
    BLOCK_N: tl.constexpr,   # tile size along N
    BLOCK_K: tl.constexpr,   # tile size along K
):
    # 2D grid over tiles of (M, N)
    pid_m = tl.program_id(axis=0)  # tile id along M
    pid_n = tl.program_id(axis=1)  # tile id along N

    m_start = pid_m * BLOCK_M
    n_start = pid_n * BLOCK_N

    m_idx = m_start + tl.arange(0, BLOCK_M)
    n_idx = n_start + tl.arange(0, BLOCK_N)

    m_mask = m_idx < M
    n_mask = n_idx < N

    # accumulator
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_idx = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_idx < K

        # A_block: (BLOCK_M, BLOCK_K)
        A_ptrs = A_ptr + m_idx[:, None] * K + k_idx[None, :]
        A_block = tl.load(A_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # B_block: (BLOCK_K, BLOCK_N), B is (K, N)
        B_ptrs = B_ptr + k_idx[:, None] * N + n_idx[None, :]
        B_block = tl.load(B_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # acc += A_block @ B_block
        acc += tl.dot(A_block, B_block)

    # write back acc to C
    C_out_ptrs = C_ptr + m_idx[:, None] * N + n_idx[None, :]
    # outer mask for store
    outer_mask = m_mask[:, None] & n_mask[None, :]
    tl.store(C_out_ptrs, acc, mask=outer_mask)


@triton.jit
def elementwise_gelu_tanh_kernel(
    X_ptr,         # *const float32, input tensor flattened (M elements)
    Y_ptr,         # *float32, output tensor flattened (M elements)
    M: tl.int32,   # total number of elements
    BLOCK: tl.constexpr,  # chunk size
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < M
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0)

    # GELU (tanh approximation)
    # y = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    sqrt_2_over_pi = 0.7978845608028654
    c = 0.044715
    x3 = x * x * x
    inner = sqrt_2_over_pi * (x + c * x3)
    y = 0.5 * x * (1.0 + tl.tanh(inner))

    tl.store(Y_ptr + offsets, y, mask=mask)
