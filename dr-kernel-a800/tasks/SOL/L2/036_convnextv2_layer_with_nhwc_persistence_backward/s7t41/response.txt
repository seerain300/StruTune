# Triton kernels: init
# =========================
@triton.jit
def normal_fill_kernel(OUT_ptr, N, MEAN, STD, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    # Generate normal via box-muller in Triton
    u = tl.rand(offsets)  # uniform in [0,1)
    v = tl.rand(offsets)  # uniform in [0,1)
    z = tl.sqrt(-2.0 * tl.log(1.0 - u)) * tl.sign(2.0 * v - 1.0)
    val = MEAN + STD * z
    tl.store(OUT_ptr + offsets, val, mask=mask)


@triton.jit
def ones_fill_kernel(OUT_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    val = 1.0
    tl.store(OUT_ptr + offsets, val, mask=mask)

# Triton kernels: Depthwise Conv2d Forward (groups=C)
# =========================
@triton.jit
def conv2d_depthwise_forward_kernel(
    X_ptr,        # *float32, (B, C, H, W) NCHW
    W_ptr,        # *float32, (C, 1, KS, KS) depthwise weights
    Y_ptr,        # *float32, (B, C, H, W) NCHW
    B, C, H, W, KS, PADDING,        # int32
    BLOCK_HW: tl.constexpr
):
    # 1 program per (b, c)
    pid = tl.program_id(0)
    if pid >= B * C:
        return
    b = pid // C
    c = pid % C
    # Iterate over spatial positions (we use tiles for HW)
    for hw_start in range(0, H * W, BLOCK_HW):
        hw_offsets = hw_start + tl.arange(0, BLOCK_HW)
        hw_mask = hw_offsets < (H * W)
        h = hw_offsets // W
        w = hw_offsets % W
        acc = tl.zeros([BLOCK_HW], dtype=tl.float32)
        # sum over 7x7 kernel
        for ki in range(KS):
            in_h = h + (ki - PADDING)
            valid_h = (in_h >= 0) & (in_h < H)
            for kj in range(KS):
                in_w = w + (kj - PADDING)
                valid_w = (in_w >= 0) & (in_w < W)
                valid = hw_mask & valid_h & valid_w
                # pointer for X[b, c, in_h, in_w]

                x_val = tl.load(X_ptr + x_index, mask=valid, other=0.0)

                w_val = tl.load(W_ptr + w_index)
                acc += x_val * w_val
        # store Y[b, c, h, w]
        y_index = b * C * H * W + c * H * W + h * W + w
        tl.store(Y_ptr + y_index, acc, mask=hw_mask)

                    # We will compute x_expanded by flattening (B,H,W) into one dimension and looping over c_in for each out_c. This approach requires writing a different kernel,
                    # but Triton doesn't easily support 4D GEMM. To avoid torch and satisfy Triton-only, we implement a per-(b,h,w) kernel that loops over C and writes
                    # x_expanded[b, out_c, h, w]. We will do that: launch grid over (B, H, W, C4) so that each program computes one output element for fixed (b,h,w, out_c)
                    # and sums over c_in. This is acceptable and avoids torch.

        # Instead of the above complex kernel, we implement the per-(b,h,w, out_c) kernel:
        x_expanded = torch.empty((B, C4, H, W), device=self.device, dtype=torch.float32)
        @triton.jit
        def linear_proj_per_elem_kernel(
            X_ptr,        # *float32, (B, H, W, C) NHWC of x_ln_scaled
            Wt_ptr,       # *float32, (C, C4) = pwconv1_weight.T
            Y_ptr,        # *float32, (B, C4, H, W)
            B, C, H, W, C4,
            BLOCK_C: tl.constexpr
        ):
            pid_b = tl.program_id(0)
            pid_out = tl.program_id(1)
            pid_h = tl.program_id(2)
            pid_w = tl.program_id(3)
            if pid_b >= B or pid_out >= C4 or pid_h >= H or pid_w >= W:
                return
            b = pid_b
            out_c = pid_out
            h = pid_h
            w = pid_w
            # Sum over input channels c_in
            acc = 0.0
            for c_in in range(0, C, BLOCK_C):
                c_in_offsets = c_in + tl.arange(0, BLOCK_C)
                c_in_mask = c_in_offsets < C
                # For a fixed (b,h,w), read X[b,h,w,c] across c_in_offsets
                base_x = b * (H * W) * C + h * W * C + w * C
                x_vals = tl.load(X_ptr + base_x + c_in_offsets, mask=c_in_mask, other=0.0)
                # Read Wt[c_in, out_c] for each c_in in block
                wt_base = c_in_offsets * C4 + out_c  # since Wt is (C,C4), element [c_in, out_c] is c_in*C4 + out_c
                wt_vals = tl.load(Wt_ptr + wt_base, mask=c_in_mask, other=0.0)
                # Multiply and accumulate (scalar per block)
                acc += tl.sum(x_vals * wt_vals, axis=0)
            # Store Y[b, out_c, h, w]
            base_y = b * (C4 * H * W) + out_c * (H * W) + h * W + w
            tl.store(Y_ptr + base_y, acc)

        # Launch per-element GEMV: grid over (B, C4, H, W)
        linear_proj_per_elem_kernel[(B, C4, H, W)](x_ln_scaled, pwconv1_weight.t(), x_expanded, B, C, H, W, C4, BLOCK_C=64)

        # GELU forward on x_expanded
        x_expanded_flat = x_expanded.reshape(-1)
        N = x_expanded_flat.numel()
        x_gelu = torch.empty_like(x_expanded_flat)
        gelu_forward_kernel[(triton.cdiv(N, 1024),)](x_expanded_flat, x_gelu, N, BLOCK=1024)
        x_gelu = x_gelu.reshape_as(x_expanded)

        # GRN forward: global L2 norm over (B,H,W) per channel, then scale
        # Compute global_features = ||x_gelu||_2 per (b,h,w) and channels
        # We implement a Triton kernel that for each (b,h,w) reduces across C, then scales x_gelu.
        # First, allocate global_features and compute norms in Triton, then scale in Triton.

        global_features = torch.empty((B, H, W, C), device=self.device, dtype=torch.float32)
        # Kernel to compute global L2 norm per (b,h,w)
        @triton.jit
        def grn_reduce_l2_kernel(
            X_ptr,        # *float32, (B, H, W, C)
            OUT_ptr,      # *float32, (B, H, W)
            B, H, W, C,
            EPS,          # float32
            BLOCK_C: tl.constexpr
        ):
            pid = tl.program_id(0)
            if pid >= B * H * W:
                return
            b = pid // (H * W)
            h = (pid % (H * W)) // W
            w = pid % W
            sumsq = 0.0
            for c_start in range(0, C, BLOCK_C):
                c_offsets = c_start + tl.arange(0, BLOCK_C)
                c_mask = c_offsets < C
                base = b * H * W * C + h * W * C + w * C + c_offsets
                vals = tl.load(X_ptr + base, mask=c_mask, other=0.0)
                sumsq += tl.sum(vals * vals, axis=0)
            norm = tl.sqrt(sumsq + EPS)
            tl.store(OUT_ptr + pid, norm)

        BHW = B * H * W
        global_features_norm = torch.empty((BHW,), device=self.device, dtype=torch.float32)
        grn_reduce_l2_kernel[(BHW,)](x_gelu, global_features_norm, B, H, W, C, self.eps, BLOCK_C=64)
