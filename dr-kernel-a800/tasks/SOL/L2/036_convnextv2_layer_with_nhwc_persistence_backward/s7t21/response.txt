import torch
import triton
import triton.language as tl


# =========================
# Triton kernels: init random
# =========================
@triton.jit
def normal_fill_kernel(OUT_ptr, N, MEAN, STD, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    # Box-Muller transform for normal
    u = tl.rand(offsets)
    v = tl.rand(offsets)
    z = tl.sqrt(-2.0 * tl.log(1.0 - u)) * tl.sign(2.0 * v - 1.0)
    val = MEAN + STD * z
    tl.store(OUT_ptr + offsets, val, mask=mask)


# =========================
# Triton kernels: conv (depthwise) forward
# =========================
@triton.jit
def conv2d_depthwise_forward_kernel(
    X_ptr,       # input: (B, C, H, W), contiguous
    W_ptr,       # weight: (C, 1, 7, 7), contiguous
    Y_ptr,       # output: (B, C, H, W), contiguous
    B, C, H, W,  # dims
    BLOCK_HW: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_c = tl.program_id(1)
    # Each program handles one (b, c) and iterates over H*W
    for start in range(0, H * W, BLOCK_HW):
        offs = start + tl.arange(0, BLOCK_HW)
        mask = offs < (H * W)
        h_idx = offs // W
        w_idx = offs % W

        acc = tl.zeros([BLOCK_HW], dtype=tl.float32)

        # Sum over 7x7 kernel with padding=3
        for kh in range(0, 7):
            for kw in range(0, 7):
                ih = h_idx + 3 - kh
                iw = w_idx + 3 - kw
                in_bounds = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
                in_mask = mask & in_bounds
                x_off = ((pid_b * C) + pid_c) * H * W + ih * W + iw
                x_val = tl.load(X_ptr + x_off, mask=in_mask, other=0.0)

                # weight index: weight layout (C, 1, 7, 7)
                w_off = pid_c * (1 * 7 * 7) + kh * 7 + kw  # since N=1, just use kh, kw
                w_val = tl.load(W_ptr + w_off)
                acc += x_val * w_val

        # write back
        y_off = ((pid_b * C) + pid_c) * H * W + offs
        tl.store(Y_ptr + y_off, acc, mask=mask)


# =========================
# Triton kernels: permute B,C,H,W -> B,H,W,C (forward copy)
# =========================
@triton.jit
def permute_bchw_to_bhwc_forward_kernel(
    X_ptr,       # input: (B, C, H, W)
    Y_ptr,       # output: (B, H, W, C)
    B, C, H, W,  # dims
    BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)
    # Each program handles one (b, h, w) and copies across C in blocks
    for c_start in range(0, C, BLOCK_C):
        c_off = c_start + tl.arange(0, BLOCK_C)
        mask = c_off < C
        x_off = ((pid_b * C) + c_off) * H * W + pid_h * W + pid_w
        y_off = ((pid_b) * H * W * C) + pid_h * W * C + pid_w * C + c_off
        x_val = tl.load(X_ptr + x_off, mask=mask, other=0.0)
        tl.store(Y_ptr + y_off, x_val, mask=mask)


# =========================
# Triton kernels: LayerNorm stats (sum and sumsq across C)
# =========================
@triton.jit
def sum_across_channels_kernel(
    X_ptr,        # (B, H, W, C)
    OUT_ptr,      # (B, H, W)
    B, H, W, C,   # dims
    BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)
    total = tl.zeros((), dtype=tl.float32)
    for c_start in range(0, C, BLOCK_C):
        c_off = c_start + tl.arange(0, BLOCK_C)
        mask = c_off < C
        x_off = ((pid_b) * H * W * C) + pid_h * W * C + pid_w * C + c_off
        vals = tl.load(X_ptr + x_off, mask=mask, other=0.0)
        # reduce vals
        total += tl.sum(vals, axis=0)
    tl.store(OUT_ptr + ((pid_b) * H * W + pid_h * W + pid_w), total)


@triton.jit
def sumsq_across_channels_kernel(
    X_ptr,        # (B, H, W, C)
    OUT_ptr,      # (B, H, W)
    B, H, W, C,   # dims
    BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)
    total = tl.zeros((), dtype=tl.float32)
    for c_start in range(0, C, BLOCK_C):
        c_off = c_start + tl.arange(0, BLOCK_C)
        mask = c_off < C
        x_off = ((pid_b) * H * W * C) + pid_h * W * C + pid_w * C + c_off
        vals = tl.load(X_ptr + x_off, mask=mask, other=0.0)
        total += tl.sum(vals * vals, axis=0)
    tl.store(OUT_ptr + ((pid_b) * H * W + pid_h * W + pid_w), total)


@triton.jit
def layernorm_forward_kernel(
    X_ptr,       # input: (B, H, W, C)
    MEAN_ptr,    # (B, H, W)
    VAR_ptr,     # (B, H, W)
    Y_ptr,       # output: (B, H, W, C)
    B, H, W, C,  # dims
    EPS: tl.constexpr,
    BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)
    mean = tl.load(MEAN_ptr + (pid_b * H * W + pid_h * W + pid_w))
    var = tl.load(VAR_ptr + (pid_b * H * W + pid_h * W + pid_w))
    std = tl.sqrt(var + EPS)

    for c_start in range(0, C, BLOCK_C):
        c_off = c_start + tl.arange(0, BLOCK_C)
        mask = c_off < C
        x_off = ((pid_b) * H * W * C) + pid_h * W * C + pid_w * C + c_off
        x_val = tl.load(X_ptr + x_off, mask=mask, other=0.0)
        y = (x_val - mean) / std
        y_off = ((pid_b) * H * W * C) + pid_h * W * C + pid_w * C + c_off
        tl.store(Y_ptr + y_off, y, mask=mask)


# =========================
# Triton kernels: GELU forward
# =========================
@triton.jit
def gelu_forward_kernel(X_ptr, Y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0)
    sqrt_2_over_pi = 0.7978845608028654
    c = 0.044715
    inner = sqrt_2_over_pi * (x + c * x * x * x)
    t = tl.tanh(inner)
    y = 0.5 * x * (1.0 + t)
    tl.store(Y_ptr + offsets, y, mask=mask)


# =========================
# Triton kernels: GRN scaling (elementwise scale by norm_features)
# =========================
@triton.jit
def elem_scale_kernel(X_ptr, SCALE_ptr, OUT_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)

    mask = offsets < N
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0)
    scale = tl.load(SCALE_ptr)  # scalar
    y = x * scale
    tl.store(OUT_ptr + offsets, y, mask=mask)


# =========================
# Triton kernels: drop mask
# =========================
@triton.jit
def drop_mask_kernel(OUT_ptr, N, DROP_PROB, BLOCK: tl.constexpr, seed: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    # simple LCG RNG
    s = seed * offsets + 1013904223
    rnd = (s >> 32) * 1.0 / 4294967296.0
    keep = rnd > DROP_PROB
    val = tl.where(keep, 1.0, 0.0)
    tl.store(OUT_ptr + offsets, val, mask=mask)


# =========================
# Triton kernels: sqrt reduction (not directly used, but defined to avoid torch.sqrt)
# =========================
@triton.jit
def sqrt_reducer_sum_sq_kernel(ARRAY_ptr, N, OUT_ptr, BLOCK: tl.constexpr):

    mask = offsets < N
    vals = tl.load(ARRAY_ptr + offsets, mask=mask, other=0.0)
    s = tl.sum(vals * vals, axis=0)
    tl.store(OUT_ptr + pid, s)
