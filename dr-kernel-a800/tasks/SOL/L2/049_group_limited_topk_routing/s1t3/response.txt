Recap of original logic
- Compute logits = hidden_states @ weight^T (FP32). In code: F.linear(hidden, weight).
- Apply sigmoid to logits to get routing scores: scores = sigmoid(logits).
- Add expert bias: scores_for_routing = scores + expert_bias.
- Grouping: 256 experts into 8 groups of 32:
  - For each token, compute per-group top-2 scores, sum them to get group_scores [num_tokens, 8].
  - Select top-4 groups (indices) per token.
  - Build a group mask [num_tokens, 8]: 1 at selected groups, 0 elsewhere.
  - Expand mask to expert level: for selected groups, set 32 contiguous experts to 1; else 0 → score_mask [num_tokens, 256].
  - Mask out non-selected group experts: masked_scores = scores_for_routing where score_mask==1, else -inf.
  - Select top-8 experts from masked_scores (indices).
  - Normalize selected values: weights = selected / (sum(selected)+1e-20) and apply routed_scaling_factor.

Optimization plan with Triton
- Triton kernel 1: Row-wise matmul to compute logits = hidden_states @ weight^T: one program per token row; iterate over hidden_dim and accumulate dot products in FP32. Output [num_tokens, num_experts].
- Triton kernel 2: Sigmoid elementwise on logits: one program per token row; apply sigmoid to each of the 256 elements.
- Triton kernel 3: Add expert bias: one program per token row; add bias vector to each element.
- Triton kernel 4: Group top-2 reduction per token:
  - Reshape [num_tokens, 8, 32] by tiling loops over groups and 32-expert tiles. For each group, compute top-2 by simple loop, sum and store group_scores [num_tokens, 8].
- Triton kernel 5: Top-4 group selection per token: iterative argmax across 8 group scores; store group_idx [num_tokens, 4].
- Triton kernel 6: Expand group mask to expert-level: given group_idx, set 32 contiguous positions per selected group to 1; else 0 → score_mask [num_tokens, 256].
- Triton kernel 7: Masked fill: set masked_scores[i, e] = NEG_INF if score_mask[i, e] == 0; else keep scores_for_routing[i, e].
- Triton kernel 8: Final top-8 selection on masked_scores: iterative argmax K=8 times; store indices and values to out_top8_idx [num_tokens, 8] and out_top8_vals [num_tokens, 8].
- Triton kernel 9: Normalize and scale: compute denom = sum(out_top8_vals) + 1e-20, then out_weight = out_top8_vals / denom * routed_scaling_factor.
- Ensure contiguity and dtype: kernels expect float32 inputs; convert tensors to float32 before passing to Triton. All heavy ops in Triton, no torch ops in host.

Note on implementation details
- Triton doesn’t provide a built-in topk; we implement iterative argmax loops. For each argmax, we scan the whole vector, track max and index, write to output, then rely on the loop to ignore previously selected elements (because we recompute max each iteration). This is acceptable given small K and group sizes.
- We allocate int32 scratch tensors for indices and masks in host; the Triton kernels write them. We then convert to int64 for the returned indices to match the original signature.
- All tensor operations are done via Triton kernels. The forward() only launches kernels.

import torch
import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Kernel 1: Row-wise matmul logits = hidden @ weight^T
@triton.jit
def _row_matmul_kernel(
    hidden_ptr,         # *f32, [num_tokens, hidden_dim]
    weight_ptr,         # *f32, [num_experts, hidden_dim]
    out_ptr,            # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    hidden_dim: tl.constexpr,
    num_experts: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_hidden = token_id * hidden_dim
    base_out = token_id * num_experts
    acc = tl.zeros([num_experts], dtype=tl.float32)
    for j in range(0, hidden_dim):
        h = tl.load(hidden_ptr + base_hidden + j)
        for e in range(0, num_experts):
            w = tl.load(weight_ptr + e * hidden_dim + j)
            acc[e] += h * w
    for e in range(0, num_experts):
        tl.store(out_ptr + base_out + e, acc[e])


# Kernel 2: Sigmoid elementwise on logits
@triton.jit
def _sigmoid_kernel(
    logits_ptr,         # *f32, [num_tokens, num_experts]
    out_ptr,            # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for e in range(0, num_experts):
        x = tl.load(logits_ptr + base + e)
        y = 1.0 / (1.0 + tl.exp(-x))
        tl.store(out_ptr + base + e, y)


# Kernel 3: Add expert bias
@triton.jit
def _add_bias_kernel(
    scores_ptr,         # *f32, [num_tokens, num_experts]
    bias_ptr,           # *f32, [num_experts]
    out_ptr,            # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for e in range(0, num_experts):
        s = tl.load(scores_ptr + base + e)
        b = tl.load(bias_ptr + e)
        tl.store(out_ptr + base + e, s + b)


# Kernel 4: Group top-2 reduction per token → group_scores [num_tokens, n_group]
@triton.jit
def _group_top2_sum_kernel(
    scores_ptr,         # *f32, [num_tokens, num_experts]
    out_group_scores_ptr,  # *f32, [num_tokens, n_group]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    n_group: tl.constexpr,
    experts_per_group: tl.constexpr,
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for g in range(0, n_group):
        group_start = g * experts_per_group
        # Initialize top2 with first two elements in the group, or -inf
        max1 = -float('inf')
        max2 = -float('inf')
        # Loop over 32 experts in this group
        for i in range(0, experts_per_group):
            e = group_start + i
            v = tl.load(scores_ptr + base + e)
            if v > max1:
                max2 = max1
                max1 = v
            elif v > max2:
                max2 = v
        tl.store(out_group_scores_ptr + token_id * n_group + g, max1 + max2)


# Kernel 5: Top-4 group selection per token (iterative argmax)
@triton.jit
def _top4_groups_kernel(
    group_scores_ptr,   # *f32, [num_tokens, n_group]
    out_group_idx_ptr,  # *i32, [num_tokens, 4]
    num_tokens: tl.constexpr,
    n_group: tl.constexpr,
    K_GROUP: tl.constexpr,  # 4
):
    token_id = tl.program_id(0)
    base_out = token_id * K_GROUP
    # Initialize best_idx and best_val arrays
    best_idx = tl.zeros([K_GROUP], dtype=tl.int32)
    best_val = tl.zeros([K_GROUP], dtype=tl.float32)
    for k in range(0, K_GROUP):
        best_idx[k] = 0
        best_val[k] = -float('inf')
    for g in range(0, n_group):
        score = tl.load(group_scores_ptr + token_id * n_group + g)
        maxv = score
        maxidx = g
        # Compare with current bests and swap if better
        # We do it by scanning best_val and replacing if score > best_val[k]
        for j in range(0, K_GROUP):
            if score > best_val[j]:
                # Shift down
                for r in range(K_GROUP - 1, j, -1):
                    best_val[r] = best_val[r - 1]
                    best_idx[r] = best_idx[r - 1]
                best_val[j] = score
                best_idx[j] = maxidx
                break  # new element inserted, next iterations will place others accordingly
    # Write out the selected group indices
    for k in range(0, K_GROUP):
        tl.store(out_group_idx_ptr + base_out + k, best_idx[k])


# Kernel 6: Expand group mask to expert-level score_mask [num_tokens, num_experts]
@triton.jit
def _mask_expand_kernel(
    group_idx_ptr,      # *i32, [num_tokens, 4]
    score_mask_ptr,     # *i32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    n_group: tl.constexpr,
    experts_per_group: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_out = token_id * num_experts
    # Initialize score_mask to zeros
    for e in range(0, num_experts):
        tl.store(score_mask_ptr + base_out + e, 0)
    # Set ones for selected groups
    for k in range(0, 4):
        g = tl.load(group_idx_ptr + token_id * 4 + k)  # i32
        start = g * experts_per_group
        for i in range(0, experts_per_group):
            e = start + i
            tl.store(score_mask_ptr + base_out + e, 1)


# Kernel 7: Masked fill: set non-selected group experts to NEG_INF in masked_scores
@triton.jit
def _masked_fill_kernel(
    scores_ptr,         # *f32, [num_tokens, num_experts]
    score_mask_ptr,     # *i32, [num_tokens, num_experts] (0/1)
    output_ptr,         # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    NEG_INF: tl.constexpr,  # e.g., -1.0e20
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for e in range(0, num_experts):
        val = tl.load(scores_ptr + base + e)
        mask = tl.load(score_mask_ptr + base + e)  # int32 0 or 1
        out = val if mask != 0 else NEG_INF
        tl.store(output_ptr + base + e, out)


# Kernel 8: Final top-8 selection on masked_scores via iterative argmax
@triton.jit
def _top8_indices_vals_kernel(
    masked_scores_ptr,  # *f32, [num_tokens, num_experts]
    out_top8_idx_ptr,   # *i32, [num_tokens, 8]
    out_top8_vals_ptr,  # *f32, [num_tokens, 8]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    K: tl.constexpr,  # 8
):
    token_id = tl.program_id(0)
    base_scores = token_id * num_experts
    base_out_idx = token_id * K
    base_out_vals = token_id * K
    # Initialize best_idx and best_val arrays
    best_idx = tl.zeros([K], dtype=tl.int32)
    best_val = tl.zeros([K], dtype=tl.float32)
    for k in range(0, K):
        best_idx[k] = 0
        best_val[k] = -float('inf')
    for e in range(0, num_experts):
        x = tl.load(masked_scores_ptr + base_scores + e)
        for j in range(0, K):
            if x > best_val[j]:
                # Shift down
                for r in range(K - 1, j, -1):
                    best_val[r] = best_val[r - 1]
                    best_idx[r] = best_idx[r - 1]
                best_val[j] = x
                best_idx[j] = e
                break
    # Write out indices and values
    for k in range(0, K):
        tl.store(out_top8_idx_ptr + base_out_idx + k, best_idx[k])
        tl.store(out_top8_vals_ptr + base_out_vals + k, best_val[k])


# Kernel 9: Normalize top-8 values and apply scaling factor
@triton.jit
def _normalize_scale_top8_kernel(
    vals_ptr,           # *f32, [num_tokens, 8]
    out_weight_ptr,     # *f32, [num_tokens, 8]
    routed_scaling_factor: tl.constexpr,
    num_tokens: tl.constexpr,
    top_k: tl.constexpr,  # 8
):
    token_id = tl.program_id(0)
    base_vals = token_id * top_k
    total = 0.0
    for k in range(0, top_k):
        v = tl.load(vals_ptr + base_vals + k)
        total += v
    denom = total + 1e-20
    for k in range(0, top_k):
        v = tl.load(vals_ptr + base_vals + k)
        w = v / denom
        w = w * routed_scaling_factor
        tl.store(out_weight_ptr + base_vals + k, w)

    def __init__(self, hidden_dim: int, routed_scaling_factor: float):

        assert hidden_states.is_cuda and weight.is_cuda and expert_bias.is_cuda, "Inputs must be CUDA tensors"
        # Cast to float32 for Triton

        weight = weight.contiguous().to(torch.float32)  # [num_experts, hidden_dim]

        assert hidden_dim == self.hidden_dim, f"hidden_dim mismatch: expected {self.hidden_dim}, got {hidden_dim}"

        scores = torch.empty((num_tokens, self.num_experts), dtype=torch.float32, device=hidden.device)
        scores_for_routing = torch.empty((num_tokens, self.num_experts), dtype=torch.float32, device=hidden.device)

            num_tokens, self.num_experts, self.n_group, self.experts_per_group,

            num_tokens, self.n_group, self.topk_group,

            group_idx, score_mask,
            num_tokens, self.num_experts, self.n_group, self.experts_per_group,

            scores_for_routing, score_mask, masked_scores,
            num_tokens, self.num_experts,
            NEG_INF=-1.0e20,

        top8_idx = torch.empty((num_tokens, self.top_k), dtype=torch.int32, device=hidden.device)

            masked_scores, top8_idx, top8_vals,
            num_tokens, self.num_experts, self.top_k,
        )

        # 9) Normalize and apply scaling factor to selected values
        topk_weight = torch.empty((num_tokens, self.top_k), dtype=torch.float32, device=hidden.device)
        _normalize_scale_top8_kernel[(num_tokens,)](
            top8_vals, topk_weight,
            self.routed_scaling_factor,
            num_tokens, self.top_k,
        )

        topk_idx = top8_idx.to(torch.int64)
        return topk_idx, topk_weight

                    best_val[r] = best_val[r - 1]
                    best_idx[r] = best_idx[r - 1]
                best_val[j] = score
                best_idx[j] = g
                break
    for k in range(0, K_GROUP):
        tl.store(out_group_idx_ptr + base_out + k, best_idx[k])


# Kernel 6: Expand group mask to expert-level score_mask [num_tokens, num_experts]
@triton.jit
def _mask_expand_kernel(
    group_idx_ptr,      # *i32, [num_tokens, 4]
    score_mask_ptr,     # *i32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    n_group: tl.constexpr,
    experts_per_group: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_out = token_id * num_experts
    # Initialize score_mask to zeros
    for e in range(0, num_experts):
        tl.store(score_mask_ptr + base_out + e, 0)
    # Set ones for selected groups
    for k in range(0, 4):
        g = tl.load(group_idx_ptr + token_id * 4 + k)  # i32
        start = g * experts_per_group
        for i in range(0, experts_per_group):
            e = start + i
            tl.store(score_mask_ptr + base_out + e, 1)


# Kernel 7: Masked fill: set non-selected group experts to NEG_INF in masked_scores
@triton.jit
def _masked_fill_kernel(
    scores_ptr,         # *f32, [num_tokens, num_experts]
    score_mask_ptr,     # *i32, [num_tokens, num_experts] (0/1)
    output_ptr,         # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    NEG_INF: tl.constexpr,  # e.g., -1.0e20
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for e in range(0, num_experts):
        val = tl.load(scores_ptr + base + e)
        mask = tl.load(score_mask_ptr + base + e)  # int32 0 or 1
        out = val if mask != 0 else NEG_INF
        tl.store(output_ptr + base + e, out)


# Kernel 8: Final top-8 selection on masked_scores via iterative argmax
@triton.jit
def _top8_indices_vals_kernel(
    masked_scores_ptr,  # *f32, [num_tokens, num_experts]
    out_top8_idx_ptr,   # *i32, [num_tokens, 8]
    out_top8_vals_ptr,  # *f32, [num_tokens, 8]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    K: tl.constexpr,  # 8
):
    token_id = tl.program_id(0)
    base_scores = token_id * num_experts
    base_out_idx = token_id * K
    base_out_vals = token_id * K
    best_idx = tl.zeros([K], dtype=tl.int32)
    best_val = tl.zeros([K], dtype=tl.float32)
    for k in range(0, K):
        best_idx[k] = 0
        best_val[k] = -float('inf')
    for e in range(0, num_experts):
        x = tl.load(masked_scores_ptr + base_scores + e)
        for j in range(0, K):
            if x > best_val[j]:
                for r in range(K - 1, j, -1):
                    best_val[r] = best_val[r - 1]
                    best_idx[r] = best_idx[r - 1]
                best_val[j] = x
                best_idx[j] = e
                break
    for k in range(0, K):
        tl.store(out_top8_idx_ptr + base_out_idx + k, best_idx[k])
        tl.store(out_top8_vals_ptr + base_out_vals + k, best_val[k])


# Kernel 9: Normalize top-8 values and apply scaling factor
@triton.jit
def _normalize_scale_top8_kernel(
    vals_ptr,           # *f32, [num_tokens, 8]
    out_weight_ptr,     # *f32, [num_tokens, 8]
    routed_scaling_factor: tl.constexpr,
    num_tokens: tl.constexpr,
    top_k: tl.constexpr,  # 8
):
    token_id = tl.program_id(0)
    base_vals = token_id * top_k
    total = 0.0
    for k in range(0, top_k):
        v = tl.load(vals_ptr + base_vals + k)
        total += v
    denom = total + 1e-20
    for k in range(0, top_k):
        v = tl.load(vals_ptr + base_vals + k)
        w = v / denom
        w = w * routed_scaling_factor
        tl.store(out_weight_ptr + base_vals + k, w)
