- คำนวณ logits = hidden_states @ weight^T + expert_bias
- คำนวณ scores = sigmoid(logits) + expert_bias
- compute group scores: sum of top-2 within each group (8 groups, 32 experts/group)
- select top-4 groups (sorted=False)
- mask scores: set non-selected groups to -inf
- select top-8 experts from masked scores (sorted=False)
- compute final weight using selected scores and routed_scaling_factor

- forward ไม่ใช้ PyTorch ops ในการ compute จริงๆ (ยกเว้น cast ที่ host สำหรับ A/W/bias และ allocation)

import torch
import torch.nn as nn
import triton
import triton.language as tl


# Kernel 1: logits = hidden_states @ weight^T + expert_bias
# A: [M, K] row-major, W: [N, K] row-major, bias: [N], OUT: [M, N]
@triton.jit
def linear_bias_kernel(A_ptr, W_ptr, BIAS_ptr, OUT_ptr, M, N, K):
    i = tl.program_id(0)
    if i >= M:
        return
    acc = tl.zeros((N,), dtype=tl.float32)
    # Loop over K in chunks of 32
    for k0 in range(0, K, 32):
        k_ids = k0 + tl.arange(0, 32)
        # Load A[i, k0:k0+32]
        a = tl.load(A_ptr + i * K + k_ids, mask=k_ids < K, other=0.0)  # [32]
        # Load W[:, k0:k0+32] -> [N, 32]
        w_ptrs = W_ptr + tl.arange(0, N)[:, None] * K + k_ids[None, :]
        w = tl.load(w_ptrs, mask=(k_ids[None, :] < K), other=0.0)
        # acc += sum_j a[j] * w[j, :]
        acc += tl.sum(a[None, :] * w, axis=1)
    # Add bias
    bias = tl.load(BIAS_ptr + tl.arange(0, N))
    acc += bias
    # Store logits[i, :]
    tl.store(OUT_ptr + i * N + tl.arange(0, N), acc)


# Kernel 2: scores = sigmoid(logits) + expert_bias
# LOGITS: [M, N], BIAS: [N], OUT: [M, N]
@triton.jit
def sigmoid_bias_kernel(LOGITS_ptr, BIAS_ptr, OUT_ptr, M, N):
    i = tl.program_id(0)
    if i >= M:
        return
    for n0 in range(0, N, 32):
        n_ids = n0 + tl.arange(0, 32)
        logits = tl.load(LOGITS_ptr + i * N + n_ids, mask=n_ids < N, other=0.0)
        bias = tl.load(BIAS_ptr + n_ids, mask=n_ids < N, other=0.0)
        sig = 1.0 / (1.0 + tl.exp(-logits))
        out = sig + bias
        tl.store(OUT_ptr + i * N + n_ids, out, mask=n_ids < N)


# Kernel 3: compute group scores per token: sum of top-2 per group (8 groups, 32 experts/group)
# SCORES: [M, N], GROUP_SCORES: [M, 8]
@triton.jit
def compute_group_scores_kernel(SCORES_ptr, GROUP_SCORES_ptr, M, N, EXP_PER_GROUP):
    i = tl.program_id(0)
    if i >= M:
        return
    # EXP_PER_GROUP = 32, groups = 8
    for g in range(0, 8):
        base = g * EXP_PER_GROUP
        # Find top-2 within this group [base, base+32)
        # Initialize best1, best2
        best1 = tl.full((), -1.0e30, tl.float32)
        best2 = tl.full((), -1.0e30, tl.float32)
        for j in range(0, EXP_PER_GROUP):
            idx = base + j
            if idx >= N:
                continue
            score = tl.load(SCORES_ptr + i * N + idx)
            # Update best1
            if score > best1:
                best2 = best1
                best1 = score
            elif score > best2:
                best2 = score
        group_score = best1 + best2
        tl.store(GROUP_SCORES_ptr + i * 8 + g, group_score)


# Kernel 4: per token, select top-4 groups (sorted=False), store indices [M, 4] int32
# GROUP_SCORES: [M, 8], SELECTED_GROUPS: [M, 4]
@triton.jit
def select_top4_groups_kernel(GROUP_SCORES_ptr, SELECTED_GROUPS_ptr, M, NUM_GROUPS):
    i = tl.program_id(0)
    if i >= M:
        return
    max_score = tl.full((), -1.0e30, tl.float32)
    for g in range(0, NUM_GROUPS):
        score = tl.load(GROUP_SCORES_ptr + i * NUM_GROUPS + g)
        if score > max_score:
            max_score = score
    # Now find indices of top-4 groups by iterative elimination
    used = tl.zeros((NUM_GROUPS,), dtype=tl.int32)
    for t in range(0, 4):
        max_idx = 0
        for g in range(0, NUM_GROUPS):
            if used[g] == 0:
                score = tl.load(GROUP_SCORES_ptr + i * NUM_GROUPS + g)
                if score > max_idx:
                    max_idx = g
        used[max_idx] = 1
        tl.store(SELECTED_GROUPS_ptr + i * 4 + t, max_idx)

# SCORES: [M, N], SELECTED_GROUPS: [M, 4], MASKED_SCORES: [M, N]
@triton.jit
def mask_scores_with_groups_kernel(SCORES_ptr, SELECTED_GROUPS_ptr, MASKED_SCORES_ptr, M, N, NUM_GROUPS, EXP_PER_GROUP):

        return

            idx = base + j
            if idx >= N:
                continue
            orig_score = tl.load(SCORES_ptr + i * N + idx)
            new_score = orig_score  # keep, but could also directly copy SCORES -> MASKED_SCORES
            # We simply copy scores here. If we want to mask non-selected, we should either:
            # - copy selected group's scores or
            # - set all to -inf and then copy selected groups (but here we implement directly)
            tl.store(MASKED_SCORES_ptr + i * N + idx, new_score)
    # Now set non-selected groups to -inf
    neg_inf = -1.0e30
    for g in range(0, NUM_GROUPS):
        if tl.any(tl.load(SELECTED_GROUPS_ptr + i * 4 + t) != g):  # not used
            base = g * EXP_PER_GROUP
            for j in range(0, EXP_PER_GROUP):
                idx = base + j
                if idx >= N:
                    continue
                tl.store(MASKED_SCORES_ptr + i * N + idx, neg_inf)

                if score > max_val:
                    max_val = score
                    max_idx = j
        used[max_idx] = 1
        tl.store(TOPK_IDX_ptr + i * 8 + t, max_idx)
        # Accumulate numerator = routed_scaling_factor * sum(selected LOGITS)
        logit_val = tl.load(LOGITS_ptr + i * N + max_idx)
        # NUMERATOR_ptr + i is scalar (denoted by for-loop over t)
        # Triton allows scalar write when using pointer with offset, but here we create a small buffer NUMERATOR as [M] in host
        # However, here NUMERATOR_ptr is a [M] tensor. We can write via i (program_id(0)) and t.
        # We'll keep NUMERATOR_ptr as [M, 1] not valid; instead, we create NUMERATOR as [M] in host and pass pointer.
        # But in this implementation, we keep NUMERATOR_ptr as scalar pointer to [M] with one element: allocate NUMERATOR as [M]
        # and kernel reads that address to write; to keep it simple, we maintain host-side accumulation. Instead, we'll pass a [M] tensor
        # and kernel writes to element i only. For clarity, we'll implement it as host allocation and kernel writes per i.

    # Note: Since Triton kernel cannot directly write to a host tensor like this, we implement accumulation in host by calling

# Kernel 7: normalize_weight_kernel — compute topk_weight = routed_scaling_factor * numerator / (numerator + 1e-20)
# LOGITS: [M, N], TOPK_IDX: [M, 8], TOPK_WEIGHT: [M, 8]
@triton.jit
def normalize_weight_kernel(LOGITS_ptr, TOPK_IDX_ptr, TOPK_WEIGHT_ptr, M, N, R_SCALING):
    i = tl.program_id(0)
    if i >= M:
        return
    numerator = tl.full((), 0.0, tl.float32)
    for t in range(0, 8):
        idx = tl.load(TOPK_IDX_ptr + i * 8 + t)
        logit_val = tl.load(LOGITS_ptr + i * N + idx)
        numerator += logit_val
    # denom = numerator + 1e-20
    denom = numerator + 1.0e-20
    scale = R_SCALING
    for t in range(0, 8):
        idx = tl.load(TOPK_IDX_ptr + i * 8 + t)
        logit_val = tl.load(LOGITS_ptr + i * N + idx)
        weight_val = (scale * logit_val) / denom
        tl.store(TOPK_WEIGHT_ptr + i * 8 + t, weight_val)
