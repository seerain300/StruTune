import torch.nn as nn
import triton
import triton.language as tl


# Triton kernel: computes everything in one pass for each token.
# Output: topk_idx [num_tokens, 8] int32 and topk_weight [num_tokens, 8] float32
@triton.jit
def routing_kernel(
    hidden_ptr,        # *float32, [num_tokens, hidden_dim]
    weight_ptr,        # *float32, [num_experts, hidden_dim]
    bias_ptr,          # *float32, [num_experts]
    out_idx_ptr,       # *int32,   [num_tokens, 8]
    out_weight_ptr,    # *float32, [num_tokens, 8]
    num_tokens,        # int32
    hidden_dim,        # int32
    num_experts,       # int32 (assert 256)
    routed_scaling,    # float32
    BLOCK_SIZE: tl.constexpr,  # int, e.g., 128 or 256
):
    pid = tl.program_id(axis=0)  # each program handles one token
    if pid >= num_tokens:
        return

    # Vector of expert indices handled by this program (loop over groups)
    # We'll iterate groups using scalar loops since group size is small.
    # Compute logits for all 256 experts.
    scores = tl.zeros((num_experts,), dtype=tl.float32)
    for e in range(0, num_experts):
        total = tl.zeros((), dtype=tl.float32)
        for j in range(0, hidden_dim):
            # hidden_ptr is row-major: offset = pid * hidden_dim + j
            h = tl.load(hidden_ptr + pid * hidden_dim + j)
            w = tl.load(weight_ptr + e * hidden_dim + j)
            total += h * w
        scores[e] = tl.sigmoid(total)

    # Add expert bias
    for e in range(0, num_experts):
        scores[e] += tl.load(bias_ptr + e)

    # Reshape into groups [8, 32] and compute top-2 per group
    # group_scores: [8, 32]
    group_scores = tl.zeros((8, 32), dtype=tl.float32)
    # Fill group_scores with scores[e] for e in [g*32 + k]
    for g in range(0, 8):
        for kk in range(0, 32):
            e = g * 32 + kk
            group_scores[g, kk] = scores[e]

    # Compute top-2 per group and sum -> [8]
    group_top2 = tl.zeros((8,), dtype=tl.float32)
    for g in range(0, 8):
        # For each g, scan 32 entries to find top-2 values
        top1 = -float("inf")
        top2 = -float("inf")
        for kk in range(0, 32):
            v = group_scores[g, kk]
            if v > top1:
                top2 = top1
                top1 = v
            elif v > top2:
                top2 = v
        group_top2[g] = top1 + top2

    # Select top-4 groups (descending order of group_top2)
    top4_group = tl.zeros((4,), dtype=tl.int32)
    top4_score = tl.zeros((4,), dtype=tl.float32)
    # Initialize top4_group with -1
    top4_group = -tl.ones((4,), dtype=tl.int32)

    # First pass: find top-4 group indices
    for g in range(0, 8):
        include = True
        for m in range(0, 4):
            if top4_score[m] != 0.0 and top4_group[m] != -1 and top4_score[m] > group_top2[g]:
                include = False
                break
        if include:
            # Find the lowest current score slot and replace
            min_idx = 0
            min_score = top4_score[0]
            for m in range(1, 4):
                if top4_score[m] == 0.0 or (top4_score[m] < min_score and top4_group[m] != -1):
                    min_idx = m
                    min_score = top4_score[m]
            top4_group[min_idx] = g
            top4_score[min_idx] = group_top2[g]

    # Build group_mask: [8] with 1.0 for selected groups, else 0.0
    group_mask = tl.zeros((8,), dtype=tl.float32)
    for m in range(0, 4):
        group_mask[top4_group[m]] = 1.0

    # Expand group_mask to per-expert mask: if expert e belongs to group g and group_mask[g] == 1, keep; else set score to -inf
    # We'll recompute masked_scores for selecting top-8
    masked_scores = tl.zeros((num_experts,), dtype=tl.float32))
    for e in range(0, num_experts):
        e_group = e // 32
        keep = group_mask[e_group]  # float
        if keep == 1.0:
            masked_scores[e] = scores[e]
        else:
            masked_scores[e] = -float("inf")

    # Select top-8 from masked_scores (duplicates allowed)
    top8_idx = tl.zeros((8,), dtype=tl.int32)
    top8_vals = tl.zeros((8,), dtype=tl.float32))
    for m in range(0, 8):
        best_val = -float("inf")
        best_idx = -1
        for e in range(0, num_experts):
            if masked_scores[e] > best_val:
                best_val = masked_scores[e]
                best_idx = e
        # Set top8[m] = best_val and its index = best_idx, mark best_idx as excluded
        # We'll keep 8 best values; duplicates are fine
        # Just store them in order of selection loop; not strictly required but harmless.
        top8_vals[m] = best_val
        top8_idx[m] = best_idx

    # Now gather original logits for selected indices and compute normalized weights
    # selected_scores are already masked_scores[top8_idx], which are original scores or -inf
    # But we need original logits before bias, so we can't directly access them. Instead, we rely on masked_scores.
    # Normalize using masked_scores (since masked_scores equals original scores for selected, -inf otherwise).
    sum_sel = 0.0
    for m in range(0, 8):
        sum_sel += top8_vals[m]
    norm = 1e-20
    if sum_sel > 0.0:
        norm = sum_sel + 1e-20

    # Store topk_idx and topk_weight (out_weight will be scaled by routed_scaling)
    # out_idx_ptr is int32, out_weight_ptr is float32
    for m in range(0, 8):
        idx = top8_idx[m]  # int32
        # weight after scaling: masked_scores[idx] / norm * routed_scaling
        w = top8_vals[m] / norm * routed_scaling
        tl.store(out_idx_ptr + pid * 8 + m, idx)
        tl.store(out_weight_ptr + pid * 8 + m, w)

        # expert_bias: [num_experts], float32
        # routed_scaling_factor: float

        hidden_dim = hidden_states.shape[1]
        num_experts = weight.shape[0]
