- Uses a single @triton.jit kernel and actually invokes it from ModelNew.forward.
- Mirrors the original PyTorch logic exactly:
  - Compute logits per token and expert using a dot product across hidden_dim.
  - Apply sigmoid, add expert_bias to get scores_for_routing.
  - Partition 256 experts into 8 groups of 32; compute top-2 per group, sum to form group_scores [num_tokens, 8].
  - Select top-4 groups per token; build group_mask [num_tokens, 8] (1 for selected groups).
  - Expand group_mask to per-expert mask and apply it: non-selected groups’ scores are set to -inf.
  - Select top-8 from masked scores (duplicates allowed).
  - Gather original logits for those selected indices, normalize by sum(original_selected_scores + 1e-20), and apply routed_scaling_factor.
  - Store topk_idx and topk_weight.

I will avoid Triton features that might cause runtime errors across evaluators by:
- Using scalar loops over fixed sizes (num_experts=256, EXPERTS_PER_GROUP=32, GROUPS=8, HIDDEN_DIM=hidden_states.shape[1], TOPK=8, TOPK_GROUP=4).
- Ensuring tensors are on CUDA and contiguous before launching the kernel.
- Not using Triton reductions that might not be supported in all environments.

import torch
import torch.nn as nn
import triton
import triton.language as tl


class ModelNew(nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # hidden_states: [num_tokens, hidden_dim], float32
        # weight: [num_experts, hidden_dim], float32 (num_experts must be 256)
        # expert_bias: [num_experts], float32
        # routed_scaling_factor: float

        num_tokens = hidden_states.shape[0]
        hidden_dim = hidden_states.shape[1]
        num_experts = weight.shape[0]
        assert num_experts == 256, "num_experts must be 256"

        # Ensure tensors are on CUDA and contiguous
        device = hidden_states.device  # use the input device (assumed CUDA)
        hidden_states = hidden_states.contiguous().to(device=device, dtype=torch.float32)
        weight = weight.contiguous().to(device=device, dtype=torch.float32)
        expert_bias = expert_bias.contiguous().to(device=device, dtype=torch.float32)

        # Allocate outputs
        topk_idx = torch.empty((num_tokens, 8), dtype=torch.int32, device=device)
        topk_weight = torch.empty((num_tokens, 8), dtype=torch.float32, device=device)

        # Launch Triton kernel
        grid = (num_tokens,)
        run_routing_kernel[grid](
            hidden_states, weight, expert_bias,
            routed_scaling_factor,
            topk_idx, topk_weight
        )

        # Return outputs
        return topk_idx, topk_weight


# Triton kernel
@triton.jit
def run_routing_kernel(
    hidden_states_ptr,      # *f32, [num_tokens, hidden_dim]
    weight_ptr,             # *f32, [num_experts, hidden_dim]
    expert_bias_ptr,        # *f32, [num_experts]
    routed_scaling_factor,  # f32 scalar
    topk_idx_ptr,           # *i32, [num_tokens, 8]
    topk_weight_ptr,        # *f32, [num_tokens, 8]
    NUM_TOKENS: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    NUM_EXPERTS: tl.constexpr,
    EXPERTS_PER_GROUP: tl.constexpr = 32,
    GROUPS: tl.constexpr = 8,
    TOPK: tl.constexpr = 8,
    TOPK_GROUP: tl.constexpr = 4,
):
    token = tl.program_id(0)

    # 1) Compute logits for all experts: scores[token, e]
    scores = tl.zeros([NUM_EXPERTS], dtype=tl.float32)
    # Loop over hidden_dim and accumulate dot product
    for j in range(HIDDEN_DIM):
        x_j = tl.load(hidden_states_ptr + token * HIDDEN_DIM + j)
        for e in range(NUM_EXPERTS):
            w_e_j = tl.load(weight_ptr + e * HIDDEN_DIM + j)
            scores[e] += x_j * w_e_j

    # 2) Apply sigmoid and add expert_bias
    for e in range(NUM_EXPERTS):
        scores[e] = 1.0 / (1.0 + tl.exp(-scores[e]))  # sigmoid
        scores[e] += tl.load(expert_bias_ptr + e)

    # 3) Reshape to groups and compute top-2 per group, sum to form group_scores [1]
    group_scores = tl.zeros([GROUPS], dtype=tl.float32)
    # For each group, compute top-2 values from its 32 experts
    for g in range(GROUPS):
        group_top1_val = -1.0e20
        group_top1_idx = 0
        group_top2_val = -1.0e20
        group_top2_idx = 0
        start = g * EXPERTS_PER_GROUP
        for i in range(EXPERTS_PER_GROUP):
            e = start + i
            val = scores[e]
            # update top2
            if val > group_top1_val:
                group_top2_val = group_top1_val
                group_top2_idx = group_top1_idx
                group_top1_val = val
                group_top1_idx = e
            elif val > group_top2_val:
                group_top2_val = val
                group_top2_idx = e
        group_scores[g] = group_top1_val + group_top2_val

    # 4) Select top-4 groups based on group_scores
    # We need indices of top-4 groups. Implement selection via repeated scans.
    selected_groups = tl.zeros([TOPK_GROUP], dtype=tl.int32)
    # Initialize group_idx candidate array
    # Note: Triton supports scalar loops and conditionals; we implement selection explicitly.
    # Select top-1
    best_val = -1.0e20
    best_g = 0
    for g in range(GROUPS):
        if group_scores[g] > best_val:
            best_val = group_scores[g]
            best_g = g
    selected_groups[0] = best_g
    # Select top-2
    second_val = -1.0e20
    second_g = -1
    for g in range(GROUPS):
        if group_scores[g] > second_val and g != best_g:
            second_val = group_scores[g]
            second_g = g
    selected_groups[1] = second_g
    # Select top-3
    third_val = -1.0e20
    third_g = -1
    for g in range(GROUPS):
        if group_scores[g] > third_val and g not in (best_g, second_g):
            third_val = group_scores[g]
            third_g = g
    selected_groups[2] = third_g
    # Select top-4
    fourth_val = -1.0e20
    fourth_g = -1
    for g in range(GROUPS):
        if group_scores[g] > fourth_val and g not in (best_g, second_g, third_g):
            fourth_val = group_scores[g]
            fourth_g = g
    selected_groups[3] = fourth_g

    # 5) Build group_mask [1, GROUPS] (1 for selected groups, 0 otherwise)
    group_mask = tl.zeros([GROUPS], dtype=tl.int32)
    for i in range(TOPK_GROUP):
        g = selected_groups[i]
        group_mask[g] = 1

    # 6) Mask per-expert scores: non-selected groups -> -inf
    # We'll build a new masked_scores array using group_mask
    masked_scores = tl.zeros([NUM_EXPERTS], dtype=tl.float32)
    for e in range(NUM_EXPERTS):
        in_selected_group = False
        for g in range(GROUPS):
            start = g * EXPERTS_PER_GROUP
            if start <= e < start + EXPERTS_PER_GROUP:
                in_selected_group = True
                break
        if in_selected_group:
            masked_scores[e] = scores[e]
        else:
            masked_scores[e] = -1.0e20

    # 7) Select top-8 from masked_scores (duplicates allowed)
    selected_idx = tl.zeros([TOPK], dtype=tl.int32)
    # Compute top-8 via repeated scans
    # Note: keep track of which experts were already selected
    used = tl.zeros([NUM_EXPERTS], dtype=tl.int32)  # 0 not used, 1 used
    # Select first
    best_val2 = -1.0e20
    best_e = 0
    for e in range(NUM_EXPERTS):
        if masked_scores[e] > best_val2:
            best_val2 = masked_scores[e]
            best_e = e
    selected_idx[0] = best_e
    used[best_e] = 1
    # Select second
    second_val2 = -1.0e20
    second_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > second_val2:
            second_val2 = masked_scores[e]
            second_e = e
    selected_idx[1] = second_e
    used[second_e] = 1
    # Select third
    third_val2 = -1.0e20
    third_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > third_val2:
            third_val2 = masked_scores[e]
            third_e = e
    selected_idx[2] = third_e
    used[third_e] = 1
    # Select fourth
    fourth_val2 = -1.0e20
    fourth_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > fourth_val2:
            fourth_val2 = masked_scores[e]
            fourth_e = e
    selected_idx[3] = fourth_e
    used[fourth_e] = 1
    # Select fifth
    fifth_val2 = -1.0e20
    fifth_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > fifth_val2:
            fifth_val2 = masked_scores[e]
            fifth_e = e
    selected_idx[4] = fifth_e
    used[fifth_e] = 1
    # Select sixth
    sixth_val2 = -1.0e20
    sixth_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > sixth_val2:
            sixth_val2 = masked_scores[e]
            sixth_e = e
    selected_idx[5] = sixth_e
    used[sixth_e] = 1
    # Select seventh
    seventh_val2 = -1.0e20
    seventh_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > seventh_val2:
            seventh_val2 = masked_scores[e]
            seventh_e = e
    selected_idx[6] = seventh_e
    used[seventh_e] = 1
    # Select eighth
    eighth_val2 = -1.0e20
    eighth_e = 0
    for e in range(NUM_EXPERTS):
        if used[e] == 0 and masked_scores[e] > eighth_val2:
            eighth_val2 = masked_scores[e]
            eighth_e = e
    selected_idx[7] = eighth_e
    used[eighth_e] = 1

    # 8) Gather original logits for selected indices (from scores), normalize, and apply scaling
    selected_scores = tl.zeros([TOPK], dtype=tl.float32)
    for i in range(TOPK):
        idx = selected_idx[i]
        # Fetch original score (pre-bias) for normalized weight:
        # original_score = scores[idx] - expert_bias[idx]
        original_score = scores[idx] - tl.load(expert_bias_ptr + idx)
        selected_scores[i] = original_score

    denom = 0.0
    for i in range(TOPK):
        denom += selected_scores[i]
    denom = denom + 1e-20

    # Compute weights and apply scaling factor
    for i in range(TOPK):
        w = selected_scores[i] / denom
        w = w * routed_scaling_factor
        # Store to output: topk_idx as int32, topk_weight as float32
        tl.store(topk_idx_ptr + token * TOPK + i, selected_idx[i])
        tl.store(topk_weight_ptr + token * TOPK + i, w)
