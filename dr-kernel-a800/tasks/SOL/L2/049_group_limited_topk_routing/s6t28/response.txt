import torch
import triton
import triton.language as tl

# 1) Triton: matmul_logits_kernel — compute logits = hidden_states @ weight.T
@triton.jit
def matmul_logits_kernel(
    hidden_ptr,   # *f32, [M, 128]
    weight_ptr,   # *f32, [256, 128]
    logits_ptr,   # *f32, [M, 256]
    M: tl.int32,
    K: tl.int32,  # 128
    N: tl.int32,  # 256
    stride_hm: tl.int32,  # hidden_states.stride(0)
    stride_hk: tl.int32,  # hidden_states.stride(1)
    stride_wm: tl.int32,  # weight.stride(0) = N
    stride_wk: tl.int32,  # weight.stride(1) = K
    stride_lm: tl.int32,  # logits.stride(0)
    stride_ln: tl.int32,  # logits.stride(1)
):
    pid_m = tl.program_id(0)
    if pid_m >= M:
        return
    acc = tl.zeros((N,), dtype=tl.float32)
    for k in range(0, K):
        h_val = tl.load(hidden_ptr + pid_m * stride_hm + k * stride_hk)
        # loop over N (256), but we can't index weight by N directly; we need to iterate over N in chunks if needed.
        # Since N is 256, we can iterate directly:
        for n in range(0, N):
            w_val = tl.load(weight_ptr + n * stride_wm + k * stride_wk)
            acc[n] += h_val * w_val
    for n in range(0, N):
        tl.store(logits_ptr + pid_m * stride_lm + n * stride_ln, acc[n])


# 2) Triton: sigmoid_add_bias_kernel — scores = sigmoid(logits) + expert_bias
@triton.jit
def sigmoid_add_bias_kernel(
    logits_ptr,   # *f32, [M, 256]
    bias_ptr,     # *f32, [256]
    scores_ptr,   # *f32, [M, 256]
    M: tl.int32,
    N: tl.int32,
    stride_lm: tl.int32,
    stride_ln: tl.int32,
    stride_sm: tl.int32,
    stride_sn: tl.int32,
):
    pid_m = tl.program_id(0)
    if pid_m >= M:
        return
    for n in range(0, N):
        l = tl.load(logits_ptr + pid_m * stride_lm + n * stride_ln)
        b = tl.load(bias_ptr + n)
        s = 1.0 / (1.0 + tl.exp(-l)) + b
        tl.store(scores_ptr + pid_m * stride_sm + n * stride_sn, s)


# 3) Triton: group_top2_sum_kernel — per-token group_scores [M, 8] = sum(top-2 per group from scores[M, 8, 32])
@triton.jit
def group_top2_sum_kernel(
    scores_ptr,   # *f32, [M, 256]
    group_scores_ptr,  # *f32, [M, 8]
    M: tl.int32,
    group_count: tl.constexpr,  # 8
    experts_per_group: tl.constexpr,  # 32
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    for g in range(group_count):
        total = 0.0
        for ep in range(experts_per_group):
            idx = g * experts_per_group + ep
            val = tl.load(scores_ptr + pid * 256 + idx)
            total += val
        # For top-2 sum, we need to recompute with top-2 selection. To simplify, we can use the sum of top-2 of the group.
        # Since we cannot call torch.topk here, we need to implement selection manually.
        # We'll assume top-2 selection by scanning and selecting largest two. This mirrors PyTorch's topk semantics (largest=True, sorted=False).
        # We'll maintain two variables for the largest and second largest.
        max1 = -float('inf')
        max2 = -float('inf')
        for ep in range(experts_per_group):
            idx = g * experts_per_group + ep
            val = tl.load(scores_ptr + pid * 256 + idx)
            if val > max1:
                max2 = max1
                max1 = val
            elif val > max2:
                max2 = val
        total = max1 + max2
        tl.store(group_scores_ptr + pid * group_count + g, total)


# 4) Triton: topk_group_kernel (K=4) — given group_scores [M, 8], return indices of top-4 per token
@triton.jit
def topk_group_kernel(
    group_scores_ptr,  # *f32, [M, 8]
    group_idx_ptr,     # *int32, [M, 4]
    M: tl.int32,
    K: tl.constexpr,   # 4
    group_count: tl.constexpr,  # 8
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    best_vals = tl.full((K,), -float('inf'), dtype=tl.float32)
    best_idxs = tl.zeros((K,), dtype=tl.int32)
    for g in range(group_count):
        val = tl.load(group_scores_ptr + pid * group_count + g)
        for j in range(K):
            if val > best_vals[j]:
                for jj in range(K - 1, j, -1):
                    best_vals[jj] = best_vals[jj - 1]
                    best_idxs[jj] = best_idxs[jj - 1]
                best_vals[j] = val
                best_idxs[j] = g
                break
    for t in range(K):
        tl.store(group_idx_ptr + pid * K + t, best_idxs[t])


# 5) Triton: build_group_mask_kernel — given selected group_idx [M, 4], set group_mask [M, 8] to 1 at selected positions
@triton.jit
def build_group_mask_kernel(
    group_idx_ptr,     # *int32, [M, 4]
    group_mask_ptr,    # *f32, [M, 8]
    M: tl.int32,
    K: tl.constexpr,   # 4
    group_count: tl.constexpr,  # 8
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    for g in range(group_count):
        tl.store(group_mask_ptr + pid * group_count + g, 0.0)
    for t in range(K):
        g_idx = tl.load(group_idx_ptr + pid * K + t)
        tl.store(group_mask_ptr + pid * group_count + g_idx, 1.0)


# 6) Triton: expand_and_set_ninf_kernel — expand group_mask [M, 8] to [M, 256], set non-selected groups' 32 entries to -inf
@triton.jit
def expand_and_set_ninf_kernel(
    group_mask_ptr,      # *f32, [M, 8]
    masked_scores_ptr,   # *f32, [M, 256], will be mutated in-place
    M: tl.int32,
    N: tl.int32,         # 256
    stride_mm: tl.int32, # masked_scores.stride(0)
    stride_mn: tl.int32, # masked_scores.stride(1)
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    for g in range(8):
        if tl.load(group_mask_ptr + pid * 8 + g) != 1.0:
            base = g * 32
            for ep in range(32):
                idx = base + ep
                tl.store(masked_scores_ptr + pid * stride_mm + idx * stride_mn, -float('inf'))


# 7) Triton: topk_expert_kernel (K=8) — select top-8 experts from masked_scores [M, 256]
@triton.jit
def topk_expert_kernel(
    masked_scores_ptr,   # *f32, [M, 256]
    selected_expert_ptr, # *int32, [M, 8]
    M: tl.int32,
    K: tl.constexpr,     # 8
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    best_vals = tl.full((K,), -float('inf'), dtype=tl.float32)
    best_idxs = tl.zeros((K,), dtype=tl.int32)
    for n in range(256):
        val = tl.load(masked_scores_ptr + pid * 256 + n)
        for j in range(K):
            if val > best_vals[j]:
                for jj in range(K - 1, j, -1):
                    best_vals[jj] = best_vals[jj - 1]
                    best_idxs[jj] = best_idxs[jj - 1]
                best_vals[j] = val
                best_idxs[j] = n
                break
    for t in range(K):
        tl.store(selected_expert_ptr + pid * K + t, best_idxs[t])


# 8) Triton: gather_original_scores_kernel — gather original scores [M, 8] from scores [M, 256] using final_expert_idx
@triton.jit
def gather_original_scores_kernel(
    scores_ptr,          # *f32, [M, 256]
    final_expert_ptr,    # *int32, [M, 8]
    gathered_scores_ptr, # *f32, [M, 8]
    M: tl.int32,
    K: tl.constexpr,     # 8
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    for t in range(K):
        idx = tl.load(final_expert_ptr + pid * K + t)
        val = tl.load(scores_ptr + pid * 256 + idx)
        tl.store(gathered_scores_ptr + pid * K + t, val)


# 9) Triton: normalize_and_scale_kernel — normalize gathered_scores per token and apply routed_scaling_factor
@triton.jit
def normalize_and_scale_kernel(
    gathered_scores_ptr, # *f32, [M, 8]
    topk_weight_ptr,     # *f32, [M, 8]
    M: tl.int32,
    K: tl.constexpr,     # 8
    scale: tl.float32,
):
    pid = tl.program_id(0)
    if pid >= M:
        return
    sum_k = 0.0
    for t in range(K):
        val = tl.load(gathered_scores_ptr + pid * K + t)
        sum_k += val
    for t in range(K):
        val = tl.load(gathered_scores_ptr + pid * K + t)
        w = val / (sum_k + 1e-20) * scale
        tl.store(topk_weight_ptr + pid * K + t, w)

    def __init__(self):
        super().__init__()

    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # Ensure CUDA and contiguous
        assert hidden_states.is_cuda and weight.is_cuda and expert_bias.is_cuda, "All tensors must be on CUDA for Triton kernels."
        hidden_states = hidden_states.contiguous().to(torch.float32)
        weight = weight.contiguous().to(torch.float32)
        expert_bias = expert_bias.contiguous().to(torch.float32)

        M, K = hidden_states.shape  # M = num_tokens, K = 128
        N = weight.shape[0]  # 256
        device = hidden_states.device

        # Allocate outputs
        logits = torch.empty((M, N), dtype=torch.float32, device=device)
        scores = torch.empty((M, N), dtype=torch.float32, device=device)
        group_scores = torch.empty((M, 8), dtype=torch.float32, device=device)
        group_idx = torch.empty((M, 4), dtype=torch.int32, device=device)
        group_mask = torch.empty((M, 8), dtype=torch.float32, device=device)
        masked_scores = torch.empty((M, N), dtype=torch.float32, device=device)
        final_expert_idx = torch.empty((M, 8), dtype=torch.int32, device=device)
        gathered_scores = torch.empty((M, 8), dtype=torch.float32, device=device)
        topk_weight = torch.empty((M, 8), dtype=torch.float32, device=device)

        # Launch 1: matmul_logits_kernel
        grid_matmul = (M,)
        matmul_logits_kernel[grid_matmul](
            hidden_states, weight, logits,
            M, K, N,
            hidden_states.stride(0), hidden_states.stride(1),
            weight.stride(0), weight.stride(1),
            logits.stride(0), logits.stride(1),
        )

        # Launch 2: sigmoid_add_bias_kernel
        grid_sigmoid = (M,)
        sigmoid_add_bias_kernel[grid_sigmoid](
            logits, expert_bias, scores,
            M, N,
            logits.stride(0), logits.stride(1),
            scores.stride(0), scores.stride(1),
        )

        # Launch 3: group_top2_sum_kernel
        grid_groups = (M,)
        group_top2_sum_kernel[grid_groups](
            scores, group_scores,
            M, 8, 32,
        )

        # Launch 4: topk_group_kernel (K=4)
        grid_topk_groups = (M,)
        topk_group_kernel[grid_topk_groups](
            group_scores, group_idx,
            M, 4, 8,
        )

        # Launch 5: build_group_mask_kernel
        grid_mask = (M,)
        build_group_mask_kernel[grid_mask](
            group_idx, group_mask,
            M, 4, 8,
        )

        # Launch 6: expand_and_set_ninf_kernel
        grid_expand = (M,)
        expand_and_set_ninf_kernel[grid_expand](
            group_mask, masked_scores,
            M, N,
            masked_scores.stride(0), masked_scores.stride(1),
        )

        # Launch 7: topk_expert_kernel (K=8)
        grid_topk_experts = (M,)
        topk_expert_kernel[grid_topk_experts](
            masked_scores, final_expert_idx,
            M, 8,
        )

        # Launch 8: gather_original_scores_kernel
        grid_gather = (M,)
        gather_original_scores_kernel[grid_gather](
            scores, final_expert_idx, gathered_scores,
            M, 8,
        )

        # Launch 9: normalize_and_scale_kernel
        grid_norm = (M,)
        normalize_and_scale_kernel[grid_norm](
            gathered_scores, topk_weight,
            M, 8, routed_scaling_factor,
        )

        # Return: indices (int32) and normalized weights (float32)
        # Note: The original returns (topk_idx, topk_weight). We return final_expert_idx and topk_weight.
        return final_expert_idx, topk_weight
