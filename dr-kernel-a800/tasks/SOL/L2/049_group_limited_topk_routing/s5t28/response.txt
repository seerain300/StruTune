import torch
import triton
import triton.language as tl


@triton.jit
def gemv_linear_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Am, stride_Ak,
                        stride_Bn, stride_Bk,
                        stride_Cm, stride_Cn,
                        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # 2D grid: (token row, expert block)
    m = tl.program_id(0)
    n_block = tl.program_id(1)
    n_start = n_block * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < N

    # Accumulator for this block of N
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)

    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # A[m, k_offsets] -> [BLOCK_K]
        a = tl.load(A_ptr + m * stride_Am + k_offsets * stride_Ak,
                    mask=mask_k, other=0.0)  # [BLOCK_K]

        # B[n, k_offsets] -> [BLOCK_N, BLOCK_K]
        b = tl.load(B_ptr + n_offsets[:, None] * stride_Bn + k_offsets[None, :] * stride_Bk,
                    mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Accumulate: acc[n] += sum_k (A[m,k] * B[n,k])
        acc += tl.sum(b * a[None, :], axis=1)

    # Store results to C[m, n_offsets]
    tl.store(C_ptr + m * stride_Cm + n_offsets * stride_Cn, acc, mask=mask_n)


@triton.jit
def sigmoid_bias_kernel(X_ptr, Bias_ptr, Y_ptr,
                         M, N,
                         stride_Xm, stride_Xn,
                         stride_Bn,
                         stride_Ym, stride_Yn,
                         BLOCK_N: tl.constexpr):
    # One program per token row
    m = tl.program_id(0)
    # Iterate over N (experts) in blocks
    for n_start in range(0, N, BLOCK_N):
        n_offsets = n_start + tl.arange(0, BLOCK_N)
        mask = n_offsets < N
        x = tl.load(X_ptr + m * stride_Xm + n_offsets * stride_Xn, mask=mask, other=0.0)
        b = tl.load(Bias_ptr + n_offsets * stride_Bn, mask=mask, other=0.0)
        y = 1.0 / (1.0 + tl.exp(-x)) + b
        tl.store(Y_ptr + m * stride_Ym + n_offsets * stride_Yn, y, mask=mask)


@triton.jit
def top2_per_group_kernel(S_ptr, GroupScores_ptr,
                          M, G, E,
                          stride_Sm, stride_Sg, stride_Sexp,
                          stride_GSm, stride_GSe,
                          BLOCK_G: tl.constexpr, BLOCK_E: tl.constexpr):
    # One program per token row
    m = tl.program_id(0)
    # Iterate over groups
    for g_start in range(0, G, BLOCK_G):
        g_offsets = g_start + tl.arange(0, BLOCK_G)
        mask_g = g_offsets < G
        # Accumulator for per-group top-2 sum
        acc = tl.zeros([BLOCK_G], dtype=tl.float32)
        # Iterate over experts in group
        for e_start in range(0, E, BLOCK_E):
            e_offsets = e_start + tl.arange(0, BLOCK_E)
            mask_e = e_offsets < E
            # Load S[m, g, e] values
            vals = tl.load(
                S_ptr + m * stride_Sm + g_offsets[:, None] * stride_Sg + e_offsets[None, :] * stride_Sexp,
                mask=mask_g[:, None] & mask_e[None, :],
                other=-1.0e30
            )  # [BLOCK_G, BLOCK_E]
            # Compute top-2 within this block
            # Flatten for simplicity
            flat = vals.reshape(BLOCK_G * BLOCK_E)
            # We need top-2 without PyTorch; do it via reductions:
            # First max
            top1 = tl.max(flat, axis=0)
            # Remove occurrences of top1 by setting them to -inf
            flat_masked = tl.where(flat == top1, -1.0e30, flat)
            top2 = tl.max(flat_masked, axis=0)
            # Sum for this group
            acc += top1 + top2
        # Store per-group sum
        tl.store(
            GroupScores_ptr + m * stride_GSm + g_offsets * stride_GSe,
            acc,
            mask=mask_g
        )

@triton.jit
def set_group_mask_kernel(group_idx_ptr, group_mask_ptr,
                          M, G,
                          group_idx_stride, group_mask_stride0, group_mask_stride1,
                          K_CONST: tl.constexpr, G_CONST: tl.constexpr):
    # group_idx_ptr shape [M, K_CONST], group_mask_ptr shape [M, G_CONST]
    m = tl.program_id(0)
    # Load selected groups for this token
    # We assume K_CONST is the number of groups (4). We set mask to 1.0 at these indices.
    for k in range(K_CONST):
        g = tl.load(group_idx_ptr + m * group_idx_stride + k)
        # Store 1.0 at group g
        tl.store(group_mask_ptr + m * group_mask_stride0 + g * group_mask_stride1, 1.0)


@triton.jit
def expand_group_mask_kernel(group_mask_ptr, expanded_mask_ptr,
                             M, G, E,
                             group_mask_stride0, group_mask_stride1,
                             expanded_mask_stride0, expanded_mask_stride1,
                             K_CONST: tl.constexpr, E_CONST: tl.constexpr):
    # For each token row m, and for each group g, copy group_mask[g] across E experts in that group.
    m = tl.program_id(0)
    for g in range(G_CONST):
        val = tl.load(group_mask_ptr + m * group_mask_stride0 + g * group_mask_stride1)
        # val is 0.0 or 1.0
        for e in range(E_CONST):
            tl.store(expanded_mask_ptr + m * expanded_mask_stride0 + (g * E_CONST + e) * expanded_mask_stride1, val)


@triton.jit
def mask_scores_kernel(expanded_mask_ptr, scores_ptr, masked_scores_ptr,
                        M, N,
                        expanded_mask_stride0, expanded_mask_stride1,
                        scores_stride0, scores_stride1,
                        masked_scores_stride0, masked_scores_stride1,
                        BLOCK_N: tl.constexpr):
    # One program per token row
    m = tl.program_id(0)
    for n_start in range(0, N, BLOCK_N):
        n_offsets = n_start + tl.arange(0, BLOCK_N)
        mask_n = n_offsets < N
        # Load expanded_mask row [N]
        em = tl.load(expanded_mask_ptr + m * expanded_mask_stride0 + n_offsets * expanded_mask_stride1, mask=mask_n, other=0.0)
        # Load scores row [N]
        s = tl.load(scores_ptr + m * scores_stride0 + n_offsets * scores_stride1, mask=mask_n, other=0.0)
        # If em == 0, set to -inf
        s_masked = tl.where(em > 0, s, -1.0e30)
        tl.store(masked_scores_ptr + m * masked_scores_stride0 + n_offsets * masked_scores_stride1, s_masked, mask=mask_n)


@triton.jit
def topk_experts_arg_kernel(masked_scores_ptr, topk_idx_ptr,
                            M, N, K,
                            masked_scores_stride0, masked_scores_stride1,
                            topk_idx_stride0, topk_idx_stride1,
                            K_CONST: tl.constexpr):
    # Implement per-token top-K argmax on masked_scores. We support K_CONST up to 8 (matches original).
    m = tl.program_id(0)
    for k in range(K_CONST):
        best_val = -1.0e30
        best_idx = -1
        for n in range(N):
            val = tl.load(masked_scores_ptr + m * masked_scores_stride0 + n * masked_scores_stride1)
            # If this val is better than best_val, update
            if val > best_val:
                best_val = val
                best_idx = n
        tl.store(topk_idx_ptr + m * topk_idx_stride0 + k * topk_idx_stride1, best_idx)


@triton.jit
def normalize_scale_kernel(topk_idx_ptr, masked_scores_ptr, topk_weight_ptr,
                           M, N, K,
                           masked_scores_stride0, masked_scores_stride1,
                           topk_idx_stride0, topk_idx_stride1,
                           topk_weight_stride0, topk_weight_stride1,
                           scaling_factor: tl.constexpr, K_CONST: tl.constexpr):
    # Compute selected_scores per token and normalize then scale
    m = tl.program_id(0)
    denom = 0.0
    for k in range(K_CONST):
        idx = tl.load(topk_idx_ptr + m * topk_idx_stride0 + k * topk_idx_stride1)
        val = tl.load(masked_scores_ptr + m * masked_scores_stride0 + idx * masked_scores_stride1)
        denom += val
    # denom is sum of selected scores for token m
    # Store normalized and scaled
    for k in range(K_CONST):
        idx = tl.load(topk_idx_ptr + m * topk_idx_stride0 + k * topk_idx_stride1)
        val = tl.load(masked_scores_ptr + m * masked_scores_stride0 + idx * masked_scores_stride1)
        norm = val / denom
        tl.store(topk_weight_ptr + m * topk_weight_stride0 + k * topk_weight_stride1, norm * scaling_factor)
