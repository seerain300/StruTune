    if pid >= num_tokens:

    h = tl.zeros((BLOCK_SIZE,), dtype=tl.float32)
    for d in range(0, hidden_dim):

        # hidden_ptr has type *float32; stride-1 layout implies offset pid*hidden_dim + d
        h[d] = tl.load(hidden_ptr + pid * hidden_dim + d)

    # Initialize chosen_groups and top4_score
    chosen_groups = [-1, -1, -1, -1]

    for g in range(0, N_GROUPS):

    top8_vals = tl.zeros((8,), dtype=tl.float32)
    top8_idx = tl.zeros((8,), dtype=tl.int32)

        best_idx = -1

                best_idx = e

        top8_idx[m] = best_idx

    # But we need the original logits before bias. We can't recompute easily; instead, we use masked_scores which already
    # incorporate bias. Normalize by masked_scores (which are original logits + bias).

    norm = sum_selected + 1e-20
    top8_weight = (top8_vals / norm) * routed_scaling

@triton.jit
def routing_kernel(
    hidden_ptr,        # *float32, [num_tokens, hidden_dim], row-major
    weight_ptr,        # *float32, [num_experts, hidden_dim], row-major
    bias_ptr,          # *float32, [num_experts]
    out_idx_ptr,       # *int32,   [num_tokens, 8]
    out_weight_ptr,    # *float32, [num_tokens, 8]
    num_tokens,        # int32
    hidden_dim,        # int32
    num_experts,       # int32 (expected 256)
    routed_scaling,    # float32
    EXPERTS_PER_GROUP: tl.constexpr,  # 32
    N_GROUPS: tl.constexpr,           # 8
    TOPK_GROUP: tl.constexpr,         # 4
    TOPK_FINAL: tl.constexpr,         # 8
    BLOCK_SIZE: tl.constexpr,         # e.g., 128
):
    pid = tl.program_id(axis=0)  # one program per token
    if pid >= num_tokens:
        return

    # Load hidden vector for this token: h[hidden_dim]
    h = tl.zeros((BLOCK_SIZE,), dtype=tl.float32)
    for d in range(0, hidden_dim):
        h[d] = tl.load(hidden_ptr + pid * hidden_dim + d)

    # Compute logits for all num_experts via dot-product
    scores = tl.zeros((num_experts,), dtype=tl.float32)
    for e in range(0, num_experts):
        acc = 0.0
        for d in range(0, hidden_dim):
            w = tl.load(weight_ptr + e * hidden_dim + d)
            acc += w * h[d]
        scores[e] = 1.0 / (1.0 + tl.exp(-acc))  # sigmoid

    # Add expert bias
    for e in range(0, num_experts):
        b = tl.load(bias_ptr + e)
        scores[e] += b

    # Per-group top-2 and group scores
    group_top2 = tl.zeros((N_GROUPS,), dtype=tl.float32)
    for g in range(0, N_GROUPS):
        start = g * EXPERTS_PER_GROUP
        top1 = -float("inf")
        top2 = -float("inf")
        for i in range(0, EXPERTS_PER_GROUP):
            e_idx = start + i
            v = scores[e_idx]
            # update top2 if v > top2
            if v > top2:
                top2 = v
            # update top1 if v > top1; move top1 down to top2
            if v > top1:
                top2 = top1
                top1 = v
        group_top2[g] = top1 + top2

    # Top-4 groups via a topk routine (descending, sorted=False semantics).
    # We keep the topk indices in a small array of size TOPK_GROUP (4).
    # This routine returns values in descending order and indices accordingly.
    group_vals = group_top2  # [N_GROUPS]
    group_idx = tl.zeros((N_GROUPS,), dtype=tl.int32)  # original group indices 0..7
    group_vals_masked = tl.zeros((N_GROUPS,), dtype=tl.float32) - float("inf")
    # Initialize: fill group_idx with 0..7
    for g in range(0, N_GROUPS):
        group_idx[g] = g

    # Now perform topk for selecting TOPK_GROUP largest values while preserving original indices order
    # We do this by repeatedly scanning group_vals and adding the largest value into a fixed output slot.
    # We maintain an array top_group_vals[4] and top_group_idx[4]; we select each by finding the maximum in group_vals
    # and then setting it to -inf. This mimics topk descending order.
    # However, Triton lacks easy vector updates; we emulate with a small number of loops.
    # We'll select top-4 deterministically by scanning group_vals and storing the largest values and indices,
    # then scanning again to remove already selected indices from consideration for next slots.
    # This ensures we get TOPK_GROUP largest values, which should match torch.topk for k=4 reasonably.

    # First, compute top4 values and indices by finding the 4 largest in group_vals
    # We'll do this in stages: find max, store its index, set it to -inf, repeat.
    # Prepare output arrays
    top_group_vals = tl.zeros((TOPK_GROUP,), dtype=tl.float32) - float("inf")
    top_group_idx = tl.zeros((TOPK_GROUP,), dtype=tl.int32)
    # First max
    max_val = -float("inf")
    max_idx = 0
    for g in range(0, N_GROUPS):
        if group_vals[g] > max_val:
            max_val = group_vals[g]
            max_idx = g
    top_group_vals[0] = max_val
    top_group_idx[0] = max_idx
    group_vals[max_idx] = -float("inf")  # remove it

    # Second max
    max_val = -float("inf")
    max_idx = 0
    for g in range(0, N_GROUPS):
        if group_vals[g] > max_val:
            max_val = group_vals[g]
            max_idx = g
    top_group_vals[1] = max_val
    top_group_idx[1] = max_idx
    group_vals[max_idx] = -float("inf")

    # Third max
    max_val = -float("inf")
    max_idx = 0
    for g in range(0, N_GROUPS):
        if group_vals[g] > max_val:
            max_val = group_vals[g]
            max_idx = g
    top_group_vals[2] = max_val
    top_group_idx[2] = max_idx
    group_vals[max_idx] = -float("inf")

    # Fourth max
    max_val = -float("inf")
    max_idx = 0
    for g in range(0, N_GROUPS):
        if group_vals[g] > max_val:
            max_val = group_vals[g]
            max_idx = g
    top_group_vals[3] = max_val
    top_group_idx[3] = max_idx
    group_vals[max_idx] = -float("inf")

    # group_mask: 1.0 for selected groups, else 0.0
    group_mask = tl.zeros((N_GROUPS,), dtype=tl.float32)
    for m in range(0, TOPK_GROUP):
        group_mask[top_group_idx[m]] = 1.0

    # Expand group_mask to per-expert mask: for each expert e, check group index e//32 and if mask[e//32] == 1
    masked_scores = tl.zeros((num_experts,), dtype=tl.float32) - float("inf")
    for e in range(0, num_experts):
        e_group = e // EXPERTS_PER_GROUP
        if group_mask[e_group] == 1.0:
            masked_scores[e] = scores[e]

    # Final top-8 via a topk routine (descending, sorted=False semantics).
    # Similar approach: repeatedly find max in masked_scores and store its index, then set it to -inf.
    final_vals = tl.zeros((TOPK_FINAL,), dtype=tl.float32) - float("inf")
    final_idx = tl.zeros((TOPK_FINAL,), dtype=tl.int32)
    for m in range(0, TOPK_FINAL):
        max_val = -float("inf")
        max_idx = -1
        for e in range(0, num_experts):
            if masked_scores[e] > max_val:
                max_val = masked_scores[e]
                max_idx = e
        final_vals[m] = max_val
        final_idx[m] = max_idx
        masked_scores[max_idx] = -float("inf")

    # Normalize by sum of original logits for those selected indices, then apply routed_scaling
    sum_selected = 0.0
    for m in range(0, TOPK_FINAL):
        e = final_idx[m]
        # original logits without bias: scores[e] - bias[e]
        orig = scores[e] - tl.load(bias_ptr + e)
        sum_selected += orig
    norm = sum_selected + 1e-20
    top8_weight = (final_vals / norm) * routed_scaling

    # Store outputs: final_idx (top-8 expert indices) and top8_weight
    out_row = pid * TOPK_FINAL
    for m in range(0, TOPK_FINAL):
        tl.store(out_idx_ptr + out_row + m, final_idx[m])
        tl.store(out_weight_ptr + out_row + m, top8_weight[m])

import torch.nn as nn
import triton
import triton.language as tl


class ModelNew(nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # hidden_states: [num_tokens, hidden_dim]
        # weight: [num_experts, hidden_dim] (num_experts = 256)
        # expert_bias: [num_experts]
        # routed_scaling_factor: float

        num_tokens = hidden_states.shape[0]
        hidden_dim = hidden_states.shape[1]
        num_experts = weight.shape[0]
        assert num_experts == 256, "num_experts must be 256"

        # Use torch.topk for the final selection to match PyTorch's behavior
        # We still perform all intermediate computations in Triton.
        # Allocate outputs

        # (to ensure correct tie-handling and output format).
        # Note: Triton kernel below does not call torch.topk; we call it after masked_scores are computed.

        hidden_ptr = hidden_states.contiguous()
        weight_ptr = weight.contiguous()
        bias_ptr = expert_bias.contiguous()

        # We need to compute masked_scores per token in Triton, then do torch.topk on that vector.
        # However, Triton cannot directly write a 1D vector of length 256 per token without careful structuring.
        # To keep the Triton kernel simple and robust, we compute per-token scores and then perform top-8 selection
        # in PyTorch based on that vector. Below is a Triton kernel that produces per-token scores and group_mask.

        # Triton kernel: per-token compute scores and group mask
        @triton.jit
        def compute_scores_and_group_mask_kernel(

            weight_ptr,        # *float32, [num_experts, hidden_dim], row-major

            N_GROUPS: tl.constexpr,           # 8
            BLOCK_SIZE: tl.constexpr,         # e.g., 128
        ):

            if pid >= num_tokens:
                return

            h = tl.zeros((BLOCK_SIZE,), dtype=tl.float32)
            for d in range(0, hidden_dim):
                h[d] = tl.load(hidden_ptr + pid * hidden_dim + d)

            # Compute logits for all num_experts via dot-product
            scores = tl.zeros((num_experts,), dtype=tl.float32)
            for e in range(0, num_experts):
                acc = 0.0

                    acc += w * h[d]
                scores[e] = 1.0 / (1.0 + tl.exp(-acc))  # sigmoid

            for e in range(0, num_experts):

                scores[e] += b
