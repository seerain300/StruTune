import torch
import triton
import triton.language as tl


@triton.jit
def gemv_linear_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Am, stride_Ak,
                        stride_Bn, stride_Bk,
                        stride_Cm, stride_Cn,
                        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute C[m, n] = sum_{k=0}^{K-1} A[m, k] * B[n, k] for m in [0, M), n in [0, N).
    A: [M, K], B: [N, K], C: [M, N].
    Launch grid: (M, grid_n), where grid_n = ceil_div(N, BLOCK_N).
    """
    m = tl.program_id(0)
    n_block = tl.program_id(1)
    n_start = n_block * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < N

    acc = tl.zeros([BLOCK_N], dtype=tl.float32)

    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # Load A[m, k_offsets]
        a = tl.load(
            A_ptr + m * stride_Am + k_offsets * stride_Ak,
            mask=mask_k, other=0.0
        )  # [BLOCK_K]

        # Load B[n_offsets, k_offsets] -> [BLOCK_N, BLOCK_K]
        b = tl.load(
            B_ptr + n_offsets[:, None] * stride_Bn + k_offsets[None, :] * stride_Bk,
            mask=mask_n[:, None] & mask_k[None, :], other=0.0
        )

        # Accumulate: acc[n] += sum_k b[n, k] * a[k]
        acc += tl.sum(b * a[None, :], axis=1)

    # Store C[m, n_offsets]
    tl.store(C_ptr + m * stride_Cm + n_offsets * stride_Cn, acc, mask=mask_n)


@triton.jit
def sigmoid_bias_kernel(X_ptr, Bias_ptr, Y_ptr,
                         M, N,
                         stride_Xm, stride_Xn,
                         stride_Bn,
                         stride_Ym, stride_Yn,
                         BLOCK_N: tl.constexpr):
    """
    Y = sigmoid(X) + Bias, elementwise.
    X: [M, N], Bias: [N], Y: [M, N].
    Launch grid: (M, grid_n), grid_n = ceil_div(N, BLOCK_N).
    """
    m = tl.program_id(0)
    for n_start in range(0, N, BLOCK_N):
        n_offsets = n_start + tl.arange(0, BLOCK_N)
        mask = n_offsets < N
        x = tl.load(X_ptr + m * stride_Xm + n_offsets * stride_Xn, mask=mask, other=0.0)
        b = tl.load(Bias_ptr + n_offsets * stride_Bn, mask=mask, other=0.0)
        y = 1.0 / (1.0 + tl.exp(-x)) + b
        tl.store(Y_ptr + m * stride_Ym + n_offsets * stride_Yn, y, mask=mask)


@triton.jit
def top2_per_group_kernel(S_ptr, GroupScores_ptr,
                          M, G, E, K,
                          stride_Sm, stride_Sg, stride_Sexp,
                          stride_GSm, stride_GSe,
                          BLOCK_G: tl.constexpr, BLOCK_E: tl.constexpr):
    """
    Compute group_scores[m, g] = sum of top-2 scores in S[m, g, :] for g in [0, G).
    S is expected to be viewable as [M, G, E], but we rely on passing strides
    that map S[m, g, e] correctly. G=8, E=256, groups size E_per_group=32.
    Launch grid: (M, grid_g), grid_g = ceil_div(G, BLOCK_G).
    """
    m = tl.program_id(0)
    g_block = tl.program_id(1)
    g_start = g_block * BLOCK_G
    g_offsets = g_start + tl.arange(0, BLOCK_G)
    mask_g = g_offsets < G

    # For each group g, find top-2 among E_per_group=32 elements
    E_per_group = E // G  # 32
    top1 = tl.full([BLOCK_G], -1.0e30, dtype=tl.float32)
    top2 = tl.full([BLOCK_G], -1.0e30, dtype=tl.float32)

    # Iterate e over [0, E_per_group)
    for e in range(0, E_per_group):
        # Compute base offset for this group and e
        # S is conceptual [M, G, E]; to access S[m, g, e], we compute linear index using strides.
        # Here we pass the correct strides to map S[m, g, e].
        base = m * stride_Sm + g_offsets * stride_Sg + e * stride_Sexp
        # Load values (masked by mask_g)
        s = tl.load(S_ptr + base, mask=mask_g, other=-1.0e30)
        # Update top-2
        better = s > top1
        top2 = tl.where(better, top1, top2)
        top1 = tl.where(better, s, top1)
        not_better = ~better
        second_better = s > top2
        top2 = tl.where(not_better & second_better, s, top2)

    group_scores_sum = top1 + top2
    # Store [M, G]
    tl.store(GroupScores_ptr + m * stride_GSm + g_offsets * stride_GSe, group_scores_sum, mask=mask_g)


@triton.jit
def arg_topk_kernel(X_ptr, Indices_ptr,
                    M, N, K,
                    stride_Xm, stride_Xn,
                    stride_Im, stride_Ik,
                    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Arg-topk: for each row m, return indices of top-K values in X[m, :] into Indices[m, :].
    X: [M, N], Indices: [M, K], int32.
    Launch grid: (M, 1).
    """
    m = tl.program_id(0)
    # Initialize indices and values
    idxs = tl.full([K], -1, dtype=tl.int32)
    vals = tl.full([K], -1.0e30, dtype=tl.float32)

    # Loop over N in blocks
    for n_start in range(0, N, BLOCK_N):
        n_offsets = n_start + tl.arange(0, BLOCK_N)
        mask = n_offsets < N
        x = tl.load(X_ptr + m * stride_Xm + n_offsets * stride_Xn, mask=mask, other=-1.0e30)
        # For each position in this block, insert into current K slots
        for j in range(BLOCK_N):
            nj = n_start + j
            if nj >= N:
                continue
            # Compute rank among current K
            rank = 0
            for r in range(K):
                # If vals[r] < x[nj], rank++
                if vals[r] < x[nj]:
                    rank += 1
            # If rank <= K, insert
            if rank <= K:
                # Shift vals/idxs to make room at rank
                for r in range(K - 1, -1, -1):
                    if r > 0:
                        vals[r] = vals[r - 1]
                        idxs[r] = idxs[r - 1]
                    else:
                        # First insertion
                        vals[r] = x[nj]
                        idxs[r] = nj

    # Write indices
    for r in range(K):
        tl.store(Indices_ptr + m * stride_Im + r * stride_Ik, idxs[r])
