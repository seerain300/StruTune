```python
import torch
import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Kernel 1: Row-wise matmul to compute logits = hidden @ weight^T
# hidden: [M, K] = [num_tokens, hidden_dim], weight: [N, K] = [num_experts, hidden_dim], out: [M, N] = [num_tokens, num_experts]
@triton.jit
def _row_matmul_kernel(
    hidden_ptr,        # *f32, [M, K]
    weight_ptr,        # *f32, [N, K]
    out_ptr,           # *f32, [M, N]
    M, K, N,           # sizes
    stride_hm, stride_hk,   # strides for hidden: row, col
    stride_wm, stride_wk,   # strides for weight: row, col
    stride_om, stride_on,   # strides for output: row, col
    BLOCK_K: tl.constexpr,
):
    m = tl.program_id(0)  # token row index
    acc = tl.zeros((N,), dtype=tl.float32)
    # Loop over K dimension in chunks
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        # Pointers to current row chunk
        h_ptrs = hidden_ptr + m * stride_hm + offs_k * stride_hk  # [BLOCK_K]
        w_ptrs = weight_ptr + offs_k * stride_wk                  # [BLOCK_K] across K
        # Load h and w, mask out-of-range
        h = tl.load(h_ptrs, mask=offs_k < K, other=0.0)  # [BLOCK_K]
        w = tl.load(w_ptrs, mask=offs_k < K, other=0.0)  # [BLOCK_K]
        # acc += sum_k h[k] * w[k] across BLOCK_K
        acc += tl.sum(h[:, None] * w[None, :], axis=0)
    # Store result row
    out_row_ptrs = out_ptr + m * stride_om + tl.arange(0, N) * stride_on
    tl.store(out_row_ptrs, acc, mask=True)


# Kernel 2: Elementwise sigmoid on logits, out = 1 / (1 + exp(-in))
@triton.jit
def _sigmoid_kernel(
    in_ptr, out_ptr,
    M, N,
    in_stride0, in_stride1, out_stride0, out_stride1,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    # Each program handles one row m
    m = pid
    offs_n = tl.arange(0, N)
    in_row_ptrs = in_ptr + m * in_stride0 + offs_n * in_stride1
    out_row_ptrs = out_ptr + m * out_stride0 + offs_n * out_stride1
    x = tl.load(in_row_ptrs)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(out_row_ptrs, y)


# Kernel 3: Add expert bias vector to each row (broadcast across N)
@triton.jit
def _add_bias_kernel(
    in_ptr, bias_ptr, out_ptr,
    M, N,
    in_stride0, in_stride1, out_stride0, out_stride1,
    bias_stride0,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    offs_n = tl.arange(0, N)
    in_row_ptrs = in_ptr + m * in_stride0 + offs_n * in_stride1
    out_row_ptrs = out_ptr + m * out_stride0 + offs_n * out_stride1
    x = tl.load(in_row_ptrs)
    b = tl.load(bias_ptr)  # bias is length N
    y = x + b
    tl.store(out_row_ptrs, y)


# Kernel 4: For each token, compute per-group top-2 sum (sum of 2 largest per group), output [M, n_group]
@triton.jit
def _group_top2_sum_kernel(
    scores_ptr, group_scores_ptr,
    M, N, n_group, experts_per_group,
    scores_stride0, scores_stride1,
    group_scores_stride0, group_scores_stride1,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    # Iterate groups and compute sum of top-2 per group
    for g in range(0, n_group):
        base = g * experts_per_group
        # Load up to 32 values for this group
        vals = tl.zeros((experts_per_group,), dtype=tl.float32)
        for i in range(0, experts_per_group):
            ptr = scores_ptr + m * scores_stride0 + (base + i) * scores_stride1
            vals[i] = tl.load(ptr)
        # Compute top-2 by simple loop (experts_per_group is 32)
        top1 = -1.0e20
        top2 = -1.0e20
        for i in range(0, experts_per_group):
            v = vals[i]
            if v > top1:
                top2 = top1
                top1 = v
            elif v > top2:
                top2 = v
        group_scores_ptr[m * group_scores_stride0 + g * group_scores_stride1] = top1 + top2


# Kernel 5: Select top-4 groups per token based on group_scores (non-decreasing tie-break not needed since values are unique), output [M, 4]
@triton.jit
def _select_top4_groups_kernel(
    group_scores_ptr, group_idx_ptr,
    M, n_group,
    group_scores_stride0, group_scores_stride1,
    group_idx_stride0, group_idx_stride1,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    # Iterative argmax selection
    for r in range(0, 4):
        max_val = -1.0e20
        idx = -1
        # Scan all groups
        for g in range(0, n_group):
            val = group_scores_ptr[m * group_scores_stride0 + g * group_scores_stride1]
            if val > max_val:
                max_val = val
                idx = g
        # Write selected idx
        out_ptr = group_idx_ptr + m * group_idx_stride0 + r * group_idx_stride1
        tl.store(out_ptr, idx)
        # Mark as used by setting score to -inf
        group_scores_ptr[m * group_scores_stride0 + idx * group_scores_stride1] = -1.0e20


# Kernel 6: Build expert-level mask for selected groups: 1 where expert belongs to selected groups, else 0
# Inputs: group_idx [M, 4], Outputs: score_mask [M, N] int32
@triton.jit
def _build_group_mask_kernel(
    group_idx_ptr, score_mask_ptr,
    M, N, n_group, experts_per_group,
    group_idx_stride0, group_idx_stride1,
    score_mask_stride0, score_mask_stride1,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    for g in range(0, n_group):
        # Check if this group is selected (any of the 4)
        selected = 0
        for r in range(0, 4):
            idx = tl.load(group_idx_ptr + m * group_idx_stride0 + r * group_idx_stride1)
            if idx == g:
                selected = 1
                break
        # If selected, set 1 for its 32 experts; else 0
        base = g * experts_per_group
        for i in range(0, experts_per_group):
            expert_id = base + i
            ptr = score_mask_ptr + m * score_mask_stride0 + expert_id * score_mask_stride1
            tl.store(ptr, selected)


# Kernel 7: Masked fill: if score_mask[i, e] == 0, set masked_scores[i, e] = -inf; else keep scores_for_routing[i, e]
@triton.jit
def _masked_fill_kernel(
    scores_ptr, score_mask_ptr, masked_ptr,
    M, N,
    scores_stride0, scores_stride1,
    score_mask_stride0, score_mask_stride1,
    masked_stride0, masked_stride1,
    NEG_INF: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    offs = tl.arange(0, N)
    scores_row_ptrs = scores_ptr + m * scores_stride0 + offs * scores_stride1
    mask_row_ptrs = score_mask_ptr + m * score_mask_stride0 + offs * score_mask_stride1
    masked_row_ptrs = masked_ptr + m * masked_stride0 + offs * masked_stride1
    scores = tl.load(scores_row_ptrs)
    mask = tl.load(mask_row_ptrs)  # int32
    # Compute result: keep if mask==1, else -inf
    out = tl.where(mask != 0, scores, NEG_INF)
    tl.store(masked_row_ptrs, out)


# Kernel 8: Final top-8 selection from masked_scores (per token), output indices [M, 8] and values [M, 8]
@triton.jit
def _final_top8_kernel(
    masked_ptr, idx_ptr, vals_ptr,
    M, N,
    masked_stride0, masked_stride1,
    idx_stride0, idx_stride1,
    vals_stride0, vals_stride1,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    for r in range(0, 8):
        max_val = -1.0e20
        max_idx = -1
        # Scan all N experts
        for n in range(0, N):
            ptr = masked_ptr + m * masked_stride0 + n * masked_stride1
            val = tl.load(ptr)
            if val > max_val:
                max_val = val
                max_idx = n
        # Write index and value
        out_idx_ptr = idx_ptr + m * idx_stride0 + r * idx_stride1
        out_val_ptr = vals_ptr + m * vals_stride0 + r * vals_stride1
        tl.store(out_idx_ptr, max_idx)
        tl.store(out_val_ptr, max_val)


# Kernel 9: Normalize top-8 selected values: vals / sum + eps, then scale by routed_scaling_factor
@triton.jit
def _normalize_scale_kernel(
    vals_ptr, weight_ptr,
    M, K,
    vals_stride0, vals_stride1,
    routed_scale,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid
    sum_val = 0.0
    for r in range(0, 8):
        ptr = vals_ptr + m * vals_stride0 + r * vals_stride1
        val = tl.load(ptr)
        sum_val += val
    # sum_val is positive (values are non-negative scores)
    inv_sum = 1.0 / (sum_val + 1e-20)
    for r in range(0, 8):
        ptr = vals_ptr + m * vals_stride0 + r * vals_stride1
        val = tl.load(ptr)
        normalized = val * inv_sum
        scaled = normalized * routed_scale
        tl.store(ptr, scaled)


class ModelNew(nn.Module):
    def __init__(self, hidden_dim: int):
        super().__init__()
        # Constants from the original code
        self.num_experts = 256
        self.experts_per_group = 32
        self.n_group = 8
        self.topk_group = 4
        self.top_k = 8
        self.hidden_dim = hidden_dim

    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # Triton-only forward: no torch ops on tensors in host code
        if not TRITON_AVAILABLE:
            raise RuntimeError("Triton is not available.")

        # Ensure CUDA tensors and contiguity, FP32
        hidden = hidden_states.contiguous().to(torch.float32)       # [M, hidden_dim]
        weight = weight.contiguous().to(torch.float32)              # [num_experts, hidden_dim] but we will use weight^T in matmul
        bias = expert_bias.contiguous().to(torch.float32)           # [num_experts]

        M = hidden.shape[0]
        K = hidden.shape[1]
        N = self.num_experts
        assert K == self.hidden_dim, f"hidden_dim mismatch: expected {self.hidden_dim}, got {K}"
        assert weight.shape[0] == N, f"weight shape mismatch: expected num_experts={N}, got {weight.shape[0]}"
        assert weight.shape[1] == K, f"weight hidden_dim mismatch: expected {K}, got {weight.shape[1]}"

        # Allocate outputs and intermediates
        logits = torch.empty((M, N), dtype=torch.float32, device=hidden.device)           # [M, N]
        scores = torch.empty((M, N), dtype=torch.float32, device=hidden.device)           # [M, N]
        scores_for_routing = torch.empty((M, N), dtype=torch.float32, device=hidden.device)  # [M, N]
        group_scores = torch.empty((M, self.n_group), dtype=torch.float32, device=hidden.device)  # [M, 8]
        group_idx = torch.empty((M, self.topk_group), dtype=torch.int32, device=hidden.device)   # [M, 4]
        score_mask = torch.empty((M, N), dtype=torch.int32, device=hidden.device)  # [M, N], expert-level mask (0/1)
        masked_scores = torch.empty((M, N), dtype=torch.float32, device=hidden.device)  # [M, N]
        top8_idx = torch.empty((M, self.top_k), dtype=torch.int32, device=hidden.device)  # [M, 8]
        top8_vals = torch.empty((M, self.top_k), dtype=torch.float32, device=hidden.device)  # [M, 8]
        topk_weight = torch.empty((M, self.top_k), dtype=torch.float32, device=hidden.device)  # [M, 8]

        # 1) Compute logits via Triton row-wise matmul: hidden @ weight^T -> logits [M, N]
        BLOCK_K = 64
        _row_matmul_kernel[(M,)](
            hidden, weight, logits,
            M, K, N,
            hidden.stride(0), hidden.stride(1),
            weight.stride(0), weight.stride(1),
            logits.stride(0), logits.stride(1),
            BLOCK_K=BLOCK_K,
            num_warps=4,
        )

        # 2) Sigmoid activation: scores = sigmoid(logits)
        _sigmoid_kernel[(M,)](
            logits, scores,
            M, N,
            logits.stride(0), logits.stride(1), scores.stride(0), scores.stride(1),
            num_warps=4,
        )

        # 3) Add expert bias: scores_for_routing = scores + bias
        _add_bias_kernel[(M,)](
            scores, bias, scores_for_routing,
            M, N,
            scores.stride(0), scores.stride(1), scores_for_routing.stride(0), scores_for_routing.stride(1),
            bias.stride(0),
            num_warps=4,
        )

        # 4) Group top-2 sum per group: group_scores [M, 8]
        _group_top2_sum_kernel[(M,)](
            scores_for_routing, group_scores,
            M, N, self.n_group, self.experts_per_group,
            scores_for_routing.stride(0), scores_for_routing.stride(1),
            group_scores.stride(0), group_scores.stride(1),
            num_warps=1,
        )

        # 5) Select top-4 groups: group_idx [M, 4]
        _select_top4_groups_kernel[(M,)](
            group_scores, group_idx,
            M, self.n_group,
            group_scores.stride(0), group_scores.stride(1),
            group_idx.stride(0), group_idx.stride(1),
            num_warps=1,
        )

        # 6) Build expert-level mask score_mask [M, N] (int32 0/1)
        _build_group_mask_kernel[(M,)](
            group_idx, score_mask,
            M, N, self.n_group, self.experts_per_group,
            group_idx.stride(0), group_idx.stride(1),
            score_mask.stride(0), score_mask.stride(1),
            num_warps=1,
        )

        # 7) Masked fill: masked_scores[i, j] = -inf if score_mask[i, j] == 0 else scores_for_routing[i, j]
        NEG_INF = -1.0e20
        _masked_fill_kernel[(M,)](
            scores_for_routing, score_mask, masked_scores,
            M, N,
            scores_for_routing.stride(0), scores_for_routing.stride(1),
            score_mask.stride(0), score_mask.stride(1),
            masked_scores.stride(0), masked_scores.stride(1),
            NEG_INF,
            num_warps=4,
        )

        # 8) Final top-8 selection from masked_scores -> top8_idx [M, 8], top8_vals [M, 8]
        _final_top8_kernel[(M,)](
            masked_scores, top8_idx, top8_vals,
            M, N,
            masked_scores.stride(0), masked_scores.stride(1),
            top8_idx.stride(0), top8_idx.stride(1),
            top8_vals.stride(0), top8_vals.stride(1),
            num_warps=1,
        )

        # 9) Normalize selected values and apply scaling: topk_weight [M, 8]
        _normalize_scale_kernel[(M,)](
            top8_vals, routed_scaling_factor,
            M, self.top_k,
            top8_vals.stride(0), top8_vals.stride(1),
            routed_scaling_factor,
            num_warps=1,
        )

        return top8_idx, topk_weight
```
