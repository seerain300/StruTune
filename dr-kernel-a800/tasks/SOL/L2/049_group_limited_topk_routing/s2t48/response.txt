Below is the corrected code:
- A Triton kernel computes logits per token for all 256 experts via a dot-product loop (equivalent to F.linear for this setup).
- A Triton kernel applies sigmoid and adds expert bias.
- A Triton kernel computes group top-2, group scores, selects top-4 groups, builds group_mask, expands to per-expert mask, sets non-selected groups to -inf, and selects top-8 indices.
- A Triton kernel normalizes the selected logits and applies the routing scaling factor, writing topk_idx and topk_weight.

ModelNew.forward only allocates tensors, passes pointers to the Triton kernels, and returns the results. No torch operations are used in the host beyond allocations and launching the kernels.

import torch
import torch.nn as nn
import triton
import triton.language as tl


# Triton kernel: compute logits per token for all num_experts
@triton.jit
def compute_logits_kernel(
    hidden_ptr,        # *float32, [num_tokens, hidden_dim], row-major
    weight_ptr,        # *float32, [num_experts, hidden_dim], row-major
    scores_ptr,        # *float32, [num_tokens, num_experts]
    num_tokens,        # int32
    hidden_dim,        # int32
    num_experts,       # int32
    EXPERTS_PER_GROUP: tl.constexpr,  # 32
    N_GROUPS: tl.constexpr,           # 8
):
    pid = tl.program_id(axis=0)  # one program per token
    if pid >= num_tokens:
        return

    # Load hidden vector for this token
    h = tl.zeros((hidden_dim,), dtype=tl.float32)
    for d in range(0, hidden_dim):
        h[d] = tl.load(hidden_ptr + pid * hidden_dim + d)

    # Compute dot product for each expert
    out_row = pid * num_experts
    for e in range(0, num_experts):
        acc = 0.0
        for d in range(0, hidden_dim):
            w = tl.load(weight_ptr + e * hidden_dim + d)
            acc += w * h[d]
        tl.store(scores_ptr + out_row + e, acc)


# Triton kernel: apply sigmoid and add expert bias
@triton.jit
def sigmoid_add_bias_kernel(
    scores_ptr,        # *float32, [num_tokens, num_experts]
    bias_ptr,          # *float32, [num_experts]
    routed_ptr,        # *float32, [num_tokens, num_experts]
    num_tokens,        # int32
    num_experts,       # int32
):
    pid = tl.program_id(axis=0)  # one program per token
    if pid >= num_tokens:
        return

    out_row = pid * num_experts
    for e in range(0, num_experts):
        score = tl.load(scores_ptr + out_row + e)
        score = 1.0 / (1.0 + tl.exp(-score))
        b = tl.load(bias_ptr + e)
        score += b
        tl.store(routed_ptr + out_row + e, score)


# Triton kernel: group routing, top-2 per group, select top-4 groups, expand mask, select top-8, normalize
@triton.jit
def routing_group_topk_kernel(
    routed_ptr,        # *float32, [num_tokens, num_experts]
    out_idx_ptr,       # *int32,   [num_tokens, 8]
    out_weight_ptr,    # *float32, [num_tokens, 8]
    num_tokens,        # int32
    num_experts,       # int32
    routed_scaling,    # float32
    EXPERTS_PER_GROUP: tl.constexpr,  # 32
    N_GROUPS: tl.constexpr,           # 8
):
    pid = tl.program_id(axis=0)  # one program per token
    if pid >= num_tokens:
        return

    out_row = pid * num_experts
    # Read routed scores for all experts
    scores = tl.zeros((num_experts,), dtype=tl.float32)
    for e in range(0, num_experts):
        scores[e] = tl.load(routed_ptr + out_row + e)

    # Per-group top-2 and sum
    group_top2 = tl.zeros((N_GROUPS,), dtype=tl.float32)
    for g in range(0, N_GROUPS):
        start = g * EXPERTS_PER_GROUP
        top1 = -float("inf")
        top2 = -float("inf")
        for i in range(0, EXPERTS_PER_GROUP):
            e_idx = start + i
            v = scores[e_idx]
            if v > top2:
                top2 = v
            if v > top1:
                top2 = top1
                top1 = v
        group_top2[g] = top1 + top2

    # Select top-4 groups (descending order)
    top4_group = tl.zeros((4,), dtype=tl.int32)  # indices in [0..7]
    top4_score = tl.zeros((4,), dtype=tl.float32)
    for g in range(0, N_GROUPS):
        include = True
        for m in range(0, 4):
            if (top4_score[m] > 0.0) and (top4_score[m] > group_top2[g]):
                include = False
                break
        if include:
            min_idx = 0
            min_score = top4_score[0]
            for m in range(1, 4):
                if (top4_score[m] == 0.0) or ((top4_score[m] < min_score) and (top4_group[m] != -1)):
                    min_idx = m
                    min_score = top4_score[m]
            top4_group[min_idx] = g
            top4_score[min_idx] = group_top2[g]

    # Build group_mask [8] (1.0 for selected, 0.0 otherwise)
    group_mask = tl.zeros((N_GROUPS,), dtype=tl.float32)
    for m in range(0, 4):
        group_mask[top4_group[m]] = 1.0

    # Expand group_mask to per-expert mask and create masked_scores
    masked_scores = tl.zeros((num_experts,), dtype=tl.float32) - float("inf")
    for e in range(0, num_experts):
        e_group = e // EXPERTS_PER_GROUP
        keep = group_mask[e_group]  # 1.0 or 0.0
        if keep == 1.0:
            masked_scores[e] = scores[e]

    # Select top-8 from masked_scores (duplicates allowed)
    top8_vals = tl.zeros((8,), dtype=tl.float32)
    top8_idx = tl.zeros((8,), dtype=tl.int32)
    for m in range(0, 8):
        best_val = -float("inf")
        best_idx = -1
        for e in range(0, num_experts):
            if masked_scores[e] > best_val:
                best_val = masked_scores[e]
                best_idx = e
        top8_vals[m] = best_val
        top8_idx[m] = best_idx

    # Store topk_idx
    for m in range(0, 8):
        tl.store(out_idx_ptr + pid * 8 + m, top8_idx[m])

    # Normalize original logits for selected indices and apply scaling
    selected_logits = tl.zeros((8,), dtype=tl.float32)
    denom = 0.0
    for m in range(0, 8):
        e = top8_idx[m]
        selected_logits[m] = tl.load(routed_ptr + out_row + e)  # original pre-sigmoid + bias values
        denom += selected_logits[m]
    for m in range(0, 8):
        selected_logits[m] /= (denom + 1e-20)
        tl.store(out_weight_ptr + pid * 8 + m, selected_logits[m] * routed_scaling)


class ModelNew(nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # hidden_states: [num_tokens, hidden_dim], float32, device
        # weight: [num_experts, hidden_dim], float32, device (num_experts must be 256)
        # expert_bias: [num_experts], float32, device
        # routed_scaling_factor: float

        num_tokens = hidden_states.shape[0]
        hidden_dim = hidden_states.shape[1]
        num_experts = weight.shape[0]
        assert num_experts == 256, "num_experts must be 256"

        # Allocate intermediate and output tensors
        scores = torch.empty((num_tokens, num_experts), dtype=torch.float32, device=hidden_states.device)
        routed = torch.empty_like(scores)
        topk_idx = torch.empty((num_tokens, 8), dtype=torch.int32, device=hidden_states.device)
        topk_weight = torch.empty((num_tokens, 8), dtype=torch.float32, device=hidden_states.device)

        # Launch Triton kernels
        grid = (num_tokens,)

        # Kernel 1: compute logits
        compute_logits_kernel[grid](
            hidden_states,
            weight,
            scores,
            num_tokens,
            hidden_dim,
            num_experts,
            EXPERTS_PER_GROUP=32,
            N_GROUPS=8,
            num_warps=4,
        )

        # Kernel 2: sigmoid + add bias
        sigmoid_add_bias_kernel[grid](
            scores,
            expert_bias,
            routed,
            num_tokens,
            num_experts,
            num_warps=4,
        )

        # Kernel 3: group top-k, mask, top-8, normalize, write outputs
        routing_group_topk_kernel[grid](
            routed,
            topk_idx,
            topk_weight,
            num_tokens,
            num_experts,
            routed_scaling_factor,
            EXPERTS_PER_GROUP=32,
            N_GROUPS=8,
            num_warps=4,
        )

        return topk_idx, topk_weight
