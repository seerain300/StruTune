                # Initialize max and second max

                # Loop over lanes to find second

                group_scores_vec[best_idx] = -1.0e20

            for t in range(0, 4):

            # We need to build a per-expert mask and apply it

            # If group_mask[g] == 0, set scores[e] = -inf
            for e in range(0, num_experts):

                if group_mask[g] == 0:

                    scores[e] = -1.0e20

            # 6) Select top-8 from masked scores
            # We need a topk selection of 8 from 'scores'
            # Implement iterative selection (find max and its index, record, zero it)
            selected_idx = tl.zeros([8], dtype=tl.int32)
            selected_val = tl.zeros([8], dtype=tl.float32)
            for t in range(0, 8):
                best_val = -1.0e20
                best_idx = -1
                for e in range(0, num_experts):
                    if scores[e] > best_val:
                        best_val = scores[e]
                        best_idx = e
                # Record
                selected_idx[t] = best_idx
                selected_val[t] = best_val
                # Zero it out
                scores[best_idx] = -1.0e20

            # We need original logits without bias: original_scores[selected_idx]
            # Re-compute original_scores (without bias) for those selected indices
            # But original_scores can be recovered by subtracting bias from masked scores (since bias was added).
            # However, simpler: recompute original logits from hidden_states and weight for selected experts.
            # Initialize original_selected as zeros
            original_selected = tl.zeros([8], dtype=tl.float32)

                acc = 0.0
                for j in range(0, hidden_dim):

                    w_val = tl.load(weight_ptr + e * hidden_dim + j, mask=True, other=0.0)

            # Normalize by sum(original_selected + eps) and apply routed_scaling_factor
            sum_selected = tl.sum(original_selected + 1.0e-20)
            norm = (original_selected / sum_selected) * routed_scale

            # 8) Store results: topk_idx and topk_weight

                # Store weight as float32
                tl.store(out_weight_ptr + base, norm[t])

            hidden_states_c,
            weight_c,
            expert_bias_c,
            topk_idx,
            topk_weight,
            routed_scaling_factor,
            hidden_dim=hidden_dim,
            num_experts=num_experts,
            n_tokens=num_tokens,
            BLOCK_H=64  # loop step size for hidden_dim; adjust as needed
        )

        return topk_idx, topk_weight
