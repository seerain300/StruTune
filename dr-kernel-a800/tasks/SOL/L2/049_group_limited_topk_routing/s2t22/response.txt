    num_tokens,          # int

    stride_hs_d,         # int (col stride of hidden_states)
    stride_w_e,          # int (row stride of weight)
    stride_w_d,          # int (col stride of weight)
    stride_sc_t,         # int (row stride of scores)
    stride_sc_e,         # int (col stride of scores)
    BLOCK: tl.constexpr,
):

    expert = tl.program_id(1)  # each program handles one expert

    # If token >= num_tokens, return (grid ensures token in range)
    if token >= num_tokens:
        return

    acc = tl.zeros((), dtype=tl.float32)
    # Loop over hidden_dim in chunks of BLOCK
    for d in range(0, hidden_dim, BLOCK):
        offs = d + tl.arange(0, BLOCK)
        mask = offs < hidden_dim

        hs = tl.load(hidden_ptr + token * stride_hs_t + offs * stride_hs_d, mask=mask, other=0.0)

        w = tl.load(weight_ptr + expert * stride_w_e + offs * stride_w_d, mask=mask, other=0.0)

                top1 = val

@triton.jit
def mask_experts_kernel(
    scores_ptr,            # *f32, [num_tokens, num_experts]
    group_mask_ptr,        # *f32, [num_tokens, n_group] (values 0.0 or 1.0)
    masked_scores_ptr,     # *f32, [num_tokens, num_experts]
    num_tokens,            # int
    num_experts,           # int
    n_group,               # int
    stride_s_t,            # int
    stride_s_e,            # int
    stride_m_t,            # int
    stride_m_g,            # int
    BLOCK_E: tl.constexpr, # chunk of experts
):
    token = tl.program_id(0)
    if token >= num_tokens:
        return
    for e in range(0, num_experts, BLOCK_E):
        offs = e + tl.arange(0, BLOCK_E)
        mask_exp = offs < num_experts
        vals = tl.load(scores_ptr + token * stride_s_t + offs * stride_s_e, mask=mask_exp, other=0.0)
        # For each expert, determine its group index (compute once)
        for i in range(BLOCK_E):
            exp_idx = e + i
            # group = exp_idx // 32
            group = (exp_idx // 32)
            mval = tl.load(group_mask_ptr + token * stride_m_t + group * stride_m_g)
            # If mval == 0.0, set to -inf; else keep vals
            neg_inf = tl.full((), -1e30, tl.float32)
            is_masked = mval == 0.0
            vals = tl.where(is_masked, neg_inf, vals)
        tl.store(masked_scores_ptr + token * stride_s_t + offs * stride_s_e, vals, mask=mask_exp)

@triton.jit

    top8_logits_ptr,       # *f32, [num_tokens, 8]
    top8_norm_ptr,         # *f32, [num_tokens, 8]
    routed_scaling_factor, # f32
    num_tokens,            # int

        return
