- No torch operations in the host beyond tensor allocations and launching the kernel.
- Exact behavior replication:
  1) Compute logits via dot product: scores[token, e] = sum_j hidden_states[token, j] * weight[e, j].
  2) Apply sigmoid.
  3) Add expert_bias to form scores_for_routing.
  4) Reshape to groups [num_tokens, 8, 32], compute top-2 per group, sum to get group scores [num_tokens, 8].
  5) Select top-4 groups per token.
  6) Build group_mask [num_tokens, 8] (1 for selected groups, 0 otherwise), expand to per-expert mask, and apply it to scores_for_routing by setting non-selected groups to -inf.
  7) Select top-8 from masked scores (duplicates allowed).
  8) Gather original logits for those selected indices, normalize by sum(original_selected_scores + 1e-20), then apply routed_scaling_factor.
  9) Store topk_idx and topk_weight.

```python
import torch
import torch.nn as nn
import triton
import triton.language as tl


class ModelNew(nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # hidden_states: [num_tokens, hidden_dim]
        # weight: [num_experts, hidden_dim] (num_experts = 256)
        # expert_bias: [num_experts]
        # routed_scaling_factor: float

        num_tokens = hidden_states.shape[0]
        hidden_dim = hidden_states.shape[1]
        num_experts = weight.shape[0]
        assert num_experts == 256, "num_experts must be 256"
        assert weight.shape[1] == hidden_dim, "weight's second dim must match hidden_states' second dim"
        assert expert_bias.shape[0] == num_experts, "expert_bias size must match num_experts"

        # Allocate outputs
        topk_idx = torch.empty((num_tokens, 8), dtype=torch.int32, device=hidden_states.device)
        topk_weight = torch.empty((num_tokens, 8), dtype=torch.float32, device=hidden_states.device)

        # Launch Triton kernel: one program per token
        grid = (num_tokens,)
        _group_limited_topk_kernel[grid](
            hidden_ptr=hidden_states,
            weight_ptr=weight,
            bias_ptr=expert_bias,
            routed_scale=routed_scaling_factor,
            N=num_tokens,
            D=hidden_dim,
            E=num_experts,
            topk_idx_ptr=topk_idx,
            topk_weight_ptr=topk_weight,
        )

        # Return results
        return topk_idx, topk_weight


@triton.jit
def _group_limited_topk_kernel(
    hidden_ptr,          # *f32, [N, D]
    weight_ptr,          # *f32, [E, D]
    bias_ptr,            # *f32, [E]
    routed_scale,        # f32 scalar
    N,                   # int32: num_tokens
    D,                   # int32: hidden_dim
    E,                   # int32: num_experts (256)
    topk_idx_ptr,        # *i32, [N, 8]
    topk_weight_ptr,     # *f32, [N, 8]
):
    # One Triton program per token
    token = tl.program_id(0)

    # 1) Compute logits via dot product: scores[e] = sum_j hidden[token, j] * weight[e, j]
    scores = tl.zeros((E,), dtype=tl.float32)
    for j in range(0, D):
        h_j = tl.load(hidden_ptr + token * D + j)
        for e in range(0, E):
            w_ej = tl.load(weight_ptr + e * D + j)
            scores[e] += h_j * w_ej

    # 2) Apply sigmoid
    for e in range(0, E):
        scores[e] = 1.0 / (1.0 + tl.exp(-scores[e]))

    # 3) Add expert bias
    for e in range(0, E):
        b_e = tl.load(bias_ptr + e)
        scores[e] += b_e  # scores_for_routing

    # 4) Reshape into groups and compute top-2 per group, sum to get group scores
    EPG = E // 8  # 32
    group_top2_sum = tl.zeros((8,), dtype=tl.float32)
    # We will track the two best per group in two passes (simple scalar loop)
    for g in range(0, 8):
        start = g * EPG
        best1 = -1.0e30
        best2 = -1.0e30
        for e in range(start, start + EPG):
            v = scores[e]
            if v > best1:
                best2 = best1
                best1 = v
            elif v > best2:
                best2 = v
        group_top2_sum[g] = best1 + best2

    # 5) Select top-4 groups per token
    group_idx = tl.zeros((4,), dtype=tl.int32)  # indices 0..7
    group_scores = group_top2_sum  # [8]
    # Selection via scalar loop (repeated top extraction)
    for r in range(0, 4):
        best_val = -1.0e30
        best_g = -1
        for g in range(0, 8):
            if group_scores[g] > best_val:
                best_val = group_scores[g]
                best_g = g
        group_idx[r] = best_g
        # Mark selected by setting its score to -inf
        group_scores[best_g] = -1.0e30

    # 6) Build group mask [8] (1 for selected groups, 0 otherwise)
    group_mask = tl.zeros((8,), dtype=tl.int32)
    for r in range(0, 4):
        group_mask[group_idx[r]] = 1

    # Now apply mask to scores_for_routing: non-selected groups -> -inf
    masked_scores = scores  # copy to modify
    for g in range(0, 8):
        if group_mask[g] == 0:
            start = g * EPG
            for e in range(start, start + EPG):
                masked_scores[e] = -1.0e30

    # 7) Select top-8 from masked scores (duplicates allowed)
    selected_idx = tl.zeros((8,), dtype=tl.int32)
    selected_val = tl.zeros((8,), dtype=tl.float32)
    for r in range(0, 8):
        best_val = -1.0e30
        best_e = 0
        # Scan all experts to find max
        for e in range(0, E):
            if masked_scores[e] > best_val:
                best_val = masked_scores[e]
                best_e = e
        selected_idx[r] = best_e
        selected_val[r] = best_val

    # 8) Gather original logits for normalization and compute normalized weights
    original_selected = tl.zeros((8,), dtype=tl.float32)
    sum_original = 0.0
    for r in range(0, 8):
        e = selected_idx[r]
        original_selected[r] = scores[e]
        sum_original += original_selected[r]
    # Avoid division by zero
    sum_original = tl.maximum(sum_original, 1e-20)
    normalized = original_selected * routed_scale / sum_original

    # 9) Store outputs
    base_out = topk_idx_ptr + token * 8
    for r in range(0, 8):
        tl.store(base_out + r, selected_idx[r])
        tl.store(topk_weight_ptr + token * 8 + r, normalized[r])
```

- The dot product is computed explicitly to mirror F.linear behavior without relying on torch. Sigmoid and bias are applied exactly as in the original.

- Top-8 selection from masked scores is performed with a scalar loop; duplicates are allowed, matching the original topk behavior.
- Normalization uses the original logits to match PyTorch’s weighting, then applies the routed_scaling_factor.

원본 PyTorch 코드의 로직:
- hidden_states: [num_tokens, hidden_dim], dtype float32
- weight: [num_experts, hidden_dim], num_experts=256, dtype float32
- expert_bias: [num_experts], dtype float32
- routed_scaling_factor: float
- 계산 순서:
  1) scores = sigmoid(F.linear(hidden, weight))  —> logits
  2) scores_for_routing = scores + expert_bias
  3) 그룹화: 256개의 전문가를 8개 그룹으로 나눔, 각 그룹 32개. scores_for_routing를 [num_tokens, 8, 32]로 reshape하고, 각 그룹 내에서 top-2를 구해 점수를 합산하여 group_scores [num_tokens, 8] 생성.
  4) 각 토큰마다 group_scores의 상위 4개 그룹 선택.
  5) group_mask [num_tokens, 8]을 만든다. 선택된 그룹은 1, 그렇지 않으면 0.
  6) group_mask를 전문가 수준으로 확장하여 scores_for_routing을 마스킹한다: 선택되지 않은 그룹에 속하는 전문가의 점수를 -inf로 설정 (PyTorch masked_fill은 True일 때 해당 값을 사용하므로, scores_for_routing에서만 마스킹해야 한다. 고른 후 bias는 사용하지 않음).
  7) 마스킹된 점수에서 상위 8개 전문가 선택 (중복 허용).
  8) 선택된 전문가의 원래 로짓 (즉, sigmoid 전, bias 추가 전의 logits)를 수집하여 정규화: sum(original_selected_scores) + 1e-20으로 나눈 후 routed_scaling_factor 곱하기.

Triton에 맞게 구현할 때 주의사항:
- 포인터 형식: Triton에서 인자는 *f32 포인터로 전달해야 합니다. 학습본에서 “hidden_ptr = hidden” 이런 식으로 넣어도 에디터는 포인터로 인식하지만, Triton에서 호출 시 명시적으로 *f32처럼 선언되어야 합니다. 제출본에서 꼭 “hidden_ptr: *f32”와 같이 명시해야 합니다. 그리고, Triton은 고정 크기 배열을 인식하지 않으므로 host에서 넘긴 파라미터는 컴파일러가 타입 추론 못하면 오류납니다. 그러므로 E, D, N은 tl.constexpr로 넘기지 말고 int32 파라미터로 전달합니다.
- 로직 상에서 저런 중첩 루프는 숫자는 작으므로 괜찮고, 각 토큰당 256번 반복, 숨겨진 차원 D는 일반적인 경우 작은 값이라 전체 로직은 괜찮습니다. 단순히 모든 계산을 Triton에서 합니다.
- 그룹 top-2를 구하는 과정은 두 최댓값을 추적하는 방식으로 구합니다. 그 후 4개 그룹 선택은 단순히 4번 반복해서 최댓값을 찾아 뺄 방식으로 합니다.
- 그룹 마스킹은 전문가 인덱스를 그룹 구간으로 변환하여 마스킹합니다. scores_for_routing에만 마스킹을 적용해야 합니다.
- 그 후 마스킹된 점수에서 top-8을 구합니다. 중복 허용.
- 마지막으로 원래 로짓에서 selected 전문가의 점수를 수집하고 정규화합니다. 이때는 원래 computed scores[e]를 사용해야 합니다 (sigmoid + bias 후의 값은 사용하지 않습니다; 정규화는 원래 logits으로 합니다).
