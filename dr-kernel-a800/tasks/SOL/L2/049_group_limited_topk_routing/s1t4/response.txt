# Kernel 1: Row-wise matmul to compute logits = hidden @ weight^T
# Input: hidden [num_tokens, hidden_dim], weight [num_experts, hidden_dim]
# Output: logits [num_tokens, num_experts], FP32
@triton.jit
def _row_matmul_kernel(
    hidden_ptr,        # *f32, [num_tokens, hidden_dim]
    weight_ptr,        # *f32, [num_experts, hidden_dim]
    logits_ptr,        # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    hidden_dim: tl.constexpr,
    num_experts: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_out = token_id * num_experts
    # Accumulator
    acc = tl.zeros((num_experts,), dtype=tl.float32)
    # Loop over hidden_dim
    for j in range(0, hidden_dim):
        # Load hidden vector element for this token
        h = tl.load(hidden_ptr + token_id * hidden_dim + j)
        # Load corresponding weight row elements
        for e in range(0, num_experts):
            w = tl.load(weight_ptr + e * hidden_dim + j)
            acc[e] += h * w
    # Store logits
    for e in range(0, num_experts):
        tl.store(logits_ptr + base_out + e, acc[e])


# Kernel 2: Sigmoid elementwise on logits
# Input: logits [num_tokens, num_experts]
# Output: scores [num_tokens, num_experts]
@triton.jit
def _sigmoid_kernel(
    logits_ptr,        # *f32, [num_tokens, num_experts]
    scores_ptr,        # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_in = token_id * num_experts
    for e in range(0, num_experts):
        x = tl.load(logits_ptr + base_in + e)
        y = 1.0 / (1.0 + tl.exp(-x))
        tl.store(scores_ptr + base_in + e, y)


# Kernel 3: Add expert bias
# Input: scores [num_tokens, num_experts], bias [num_experts]
# Output: scores_for_routing [num_tokens, num_experts]
@triton.jit

    out_ptr,           # *f32, [num_tokens, num_experts]

    base_in = token_id * num_experts

        b = tl.load(bias_ptr + e)
        tl.store(out_ptr + base_in + e, s + b)


# Kernel 4: Group top-2 sum per token -> group_scores [num_tokens, n_group]
# We iterate over groups and compute top-2 within each group of 32 by scanning.
@triton.jit
def _group_top2_sum_kernel(
    scores_ptr,        # *f32, [num_tokens, num_experts]
    group_scores_ptr,  # *f32, [num_tokens, n_group]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    n_group: tl.constexpr,
    experts_per_group: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_gs = token_id * n_group
    for g in range(0, n_group):
        start = g * experts_per_group
        # Initialize max values
        max1 = tl.float32(-1.0e20)
        max2 = tl.float32(-1.0e20)
        for i in range(0, experts_per_group):
            e = start + i
            x = tl.load(scores_ptr + token_id * num_experts + e)
            # Update top-2
            if x > max1:
                max2 = max1
                max1 = x
            elif x > max2:
                max2 = x
        tl.store(group_scores_ptr + base_gs + g, max1 + max2)

    group_idx_ptr,     # *i32, [num_tokens, 4]
    num_tokens: tl.constexpr,
    n_group: tl.constexpr,
):

    base_out = token_id * 4
    best = tl.full((4,), -1.0e20, dtype=tl.float32)
    chosen = tl.zeros((4,), dtype=tl.int32)  # 0/1 if selected
    for g in range(0, n_group):
        gs = tl.load(group_scores_ptr + token_id * n_group + g)
        # Find best among unchosen (0) groups
        for k in range(0, 4):
            if chosen[k] == 0 and gs > best[k]:
                best[k] = gs
                chosen[k] = 1
    # Write out group indices for the 4 best
    # We need to map chosen to original g; since chosen marks which slots are used,
    # we recompute idx via comparing best with group_scores, but simpler approach:
    # Reload group scores and pick top-4; however, chosen marks already indicate which k got selected.
    # Here we just store top-4 g in descending order of best using indices 'k'.
    # Triton can't return vectors, so we store sequentially based on chosen==1.
    # To keep simple, we recompute top4 by comparing best array:
    # We know 4 positions are filled; output in descending order.

        gs = tl.load(group_scores_ptr + token_id * n_group + g)

                if count < 4:
                    top_gs[count] = gs
                    top_idx[count] = g
                    count += 1
    # Store in order (top_gs sorted descending already due to best array construction)
    for k in range(0, 4):
        # Find non-empty slot with largest top_gs[k] among filled
        # Since we populated top_gs by largest first, just store as is
        idx = top_idx[k]
        tl.store(group_idx_ptr + base_out + k, idx)


# Kernel 6: Build group_mask [num_tokens, n_group] as 1 at selected groups, else 0
@triton.jit
def _build_group_mask_kernel(

# Kernel 10: Normalize top-8 selected values and apply scaling factor
@triton.jit
def _normalize_scale_top8_kernel(
    vals_ptr,           # *f32, [num_tokens, 8]
    out_weight_ptr,     # *f32, [num_tokens, 8]

    num_tokens: tl.constexpr,
    top_k: tl.constexpr,  # 8

        w = v / denom
        w = w * routed_scaling_factor
        tl.store(out_weight_ptr + base_vals + k, w)

    token_id = tl.program_id(0)
    base_out = token_id * n_group
    for g in range(0, n_group):
        tl.store(out_mask_ptr + base_out + g, 0)
    for k in range(0, 4):
        g = tl.load(group_idx_ptr + token_id * 4 + k)
        tl.store(out_mask_ptr + base_out + g, 1)


# Kernel 7: Expand group mask to expert-level score_mask [num_tokens, num_experts]
# 1s for selected group's 32 experts; else 0
@triton.jit
def _mask_expand_kernel(
    out_mask_ptr,       # *i32, [num_tokens, n_group]
    group_idx_ptr,      # *i32, [num_tokens, 4]
    score_mask_ptr,     # *i32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    n_group: tl.constexpr,
    experts_per_group: tl.constexpr,
):
    token_id = tl.program_id(0)
    base_out = token_id * num_experts
    for e in range(0, num_experts):
        tl.store(score_mask_ptr + base_out + e, 0)
    for k in range(0, 4):
        g = tl.load(group_idx_ptr + token_id * 4 + k)
        start = g * experts_per_group
        for i in range(0, experts_per_group):
            e = start + i
            tl.store(score_mask_ptr + base_out + e, 1)


# Kernel 8: Masked fill: set non-selected group experts to NEG_INF in masked_scores
@triton.jit
def _masked_fill_kernel(
    scores_ptr,         # *f32, [num_tokens, num_experts]
    score_mask_ptr,     # *i32, [num_tokens, num_experts]
    output_ptr,         # *f32, [num_tokens, num_experts]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    NEG_INF: tl.constexpr,
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for e in range(0, num_experts):
        val = tl.load(scores_ptr + base + e)
        m = tl.load(score_mask_ptr + base + e)  # 0 or 1
        out_val = val if m != 0 else NEG_INF
        tl.store(output_ptr + base + e, out_val)


# Kernel 9: Top-8 selection from masked_scores (iterative argmax), store both indices and values
@triton.jit
def _top8_indices_vals_kernel(
    masked_scores_ptr,  # *f32, [num_tokens, num_experts]
    out_idx_ptr,        # *i32, [num_tokens, 8]
    out_vals_ptr,       # *f32, [num_tokens, 8]
    num_tokens: tl.constexpr,
    num_experts: tl.constexpr,
    top_k: tl.constexpr,  # 8
):
    token_id = tl.program_id(0)
    base = token_id * num_experts
    for k in range(0, top_k):
        max_val = tl.float32(-1.0e20)
        max_idx = 0
        for e in range(0, num_experts):
            x = tl.load(masked_scores_ptr + base + e)
            if x > max_val:
                max_val = x
                max_idx = e
        tl.store(out_idx_ptr + token_id * top_k + k, max_idx)
        tl.store(out_vals_ptr + token_id * top_k + k, max_val)


# Kernel 10: Normalize top-8 selected values and apply scaling factor
@triton.jit
def _normalize_scale_top8_kernel(
    vals_ptr,           # *f32, [num_tokens, 8]
    out_weight_ptr,     # *f32, [num_tokens, 8]
    routed_scaling_factor: tl.constexpr,
    num_tokens: tl.constexpr,
    top_k: tl.constexpr,  # 8
):
    token_id = tl.program_id(0)
    base_vals = token_id * top_k
    total = 0.0
    for k in range(0, top_k):
        v = tl.load(vals_ptr + base_vals + k)
        total += v
    denom = total + 1e-20
    for k in range(0, top_k):
        v = tl.load(vals_ptr + base_vals + k)
        w = v / denom
        w = w * routed_scaling_factor
        tl.store(out_weight_ptr + base_vals + k, w)
