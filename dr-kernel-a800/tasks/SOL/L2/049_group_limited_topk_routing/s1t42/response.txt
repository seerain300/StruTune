import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Constants
NUM_EXPERTS = 256
HIDDEN_DIM = 768  # default; should be provided by the caller or can be inferred
N_GROUP = 8
EXPERTS_PER_GROUP = NUM_EXPERTS // N_GROUP  # 32
TOPK_GROUP = 4
TOP_K = 8


@triton.jit
def _row_matmul_kernel(
    hidden_ptr,       # *f32, [M, K]
    weight_ptr,       # *f32, [N, K]
    out_ptr,          # *f32, [M, N]
    M, K, N,          # int32
    stride_hm, stride_hk,
    stride_wm, stride_wk,
    stride_om, stride_on,
    BLOCK_K: tl.constexpr,
):
    m = tl.program_id(0)  # row token
    if m >= M:
        return
    # Accumulator for row m
    acc = tl.zeros((N,), dtype=tl.float32)
    # Loop over hidden_dim in tiles of BLOCK_K
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # Load hidden row slice: [BLOCK_K]
        h = tl.load(hidden_ptr + m * stride_hm + offs_k * stride_hk, mask=mask_k, other=0.0)
        # Load weight tile: [BLOCK_K, N]
        w = tl.load(
            weight_ptr + offs_k[:, None] * stride_wk + tl.arange(0, N) * stride_wm,
            mask=mask_k[:, None],
            other=0.0,
        )
        # Dot product: [BLOCK_K] * [BLOCK_K, N] -> [N]
        acc += tl.sum(h[:, None] * w, axis=0)
    # Store result row
    tl.store(out_ptr + m * stride_om + tl.arange(0, N) * stride_on, acc)


@triton.jit
def _sigmoid_kernel(
    x_ptr, y_ptr, M, N,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
):
    m = tl.program_id(0)
    if m >= M:
        return
    for n in range(0, N):
        ptr = x_ptr + m * stride_xm + n * stride_xn
        val = tl.load(ptr)
        y = 1.0 / (1.0 + tl.exp(-val))
        out_ptr = y_ptr + m * stride_ym + n * stride_yn
        tl.store(out_ptr, y)


@triton.jit
def _bias_add_kernel(
    x_ptr, bias_ptr, y_ptr, M, N,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_b,
):
    m = tl.program_id(0)
    if m >= M:
        return
    for n in range(0, N):
        ptr = x_ptr + m * stride_xm + n * stride_xn
        val = tl.load(ptr)
        b = tl.load(bias_ptr + n * stride_b)
        y = val + b
        out_ptr = y_ptr + m * stride_ym + n * stride_yn
        tl.store(out_ptr, y)


@triton.jit
def _group_top2_sum_kernel(
    scores_ptr, group_scores_ptr, M, G, EP,
    stride_sm, stride_sn,
    stride_gm, stride_gn,
):
    m = tl.program_id(0)
    if m >= M:
        return
    for g in range(0, G):
        base = g * EP
        max1 = -1e20
        max2 = -1e20
        for e in range(0, EP):
            col = base + e
            ptr = scores_ptr + m * stride_sm + col * stride_sn
            val = tl.load(ptr)
            if val > max1:
                max2 = max1
                max1 = val
            elif val > max2:
                max2 = val
        sum_top2 = max1 + max2
        tl.store(group_scores_ptr + m * stride_gm + g * stride_gn, sum_top2)


@triton.jit
def _select_top4_groups_kernel(
    group_scores_ptr, group_idx_ptr, M, G,
    stride_gm, stride_gn,
):
    m = tl.program_id(0)
    if m >= M:
        return
    top = tl.full((4,), -1e20, dtype=tl.float32)
    idx = tl.full((4,), -1, dtype=tl.int32)
    for g in range(0, G):
        ptr = group_scores_ptr + m * stride_gm + g * stride_gn
        val = tl.load(ptr)
        for j in range(0, 4):
            if val > top[j]:
                tmp = top[j]
                top[j] = val
                for k in range(j+1, 4):
                    top[k] = tmp
                    tmp = top[k]
                top[j+1:] = top[j+1:]
                tmp_idx = idx[j]
                idx[j] = g
                for k in range(j+1, 4):
                    idx[k] = tmp_idx
                    tmp_idx = idx[k]
                idx[j+1:] = idx[j+1:]
                break
    out_ptrs = group_idx_ptr + m * 4 + tl.arange(0, 4)
    tl.store(out_ptrs, idx)


@triton.jit
def _build_group_mask_kernel(
    group_idx_ptr, score_mask_ptr, M, G, EP,
    stride_gm, stride_gn,
    stride_sm,
):
    m = tl.program_id(0)
    if m >= M:
        return
    for g in range(0, G):
        gi = tl.load(group_idx_ptr + m * stride_gm + g * stride_gn)
        base = gi * EP
        for e in range(0, EP):
            col = base + e
            ptr = score_mask_ptr + m * stride_sm + col
            tl.store(ptr, 1)
    # set remaining to 0
    for e in range(G * EP, NUM_EXPERTS):
        ptr = score_mask_ptr + m * stride_sm + e
        tl.store(ptr, 0)


@triton.jit
def _masked_fill_kernel(
    scores_ptr, score_mask_ptr, masked_ptr, M, N,
    stride_sm, stride_sn,
    stride_mm,
    NEG_INF: tl.constexpr,
):
    m = tl.program_id(0)
    if m >= M:
        return
    for n in range(0, N):
        s = tl.load(scores_ptr + m * stride_sm + n * stride_sn)
        keep = tl.load(score_mask_ptr + m * stride_mm + n)
        val = tl.where(keep != 0, s, NEG_INF)
        tl.store(masked_ptr + m * stride_mm + n, val)


@triton.jit
def _final_top8_kernel(
    masked_ptr, idx_ptr, vals_ptr, M, N,
    stride_mm, stride_mn,
    stride_im,
    stride_vm,
    NEG_INF: tl.constexpr,
):
    m = tl.program_id(0)
    if m >= M:
        return
    top_vals = tl.full((8,), NEG_INF, dtype=tl.float32)
    top_idx = tl.full((8,), -1, dtype=tl.int32)
    for n in range(0, N):
        ptr = masked_ptr + m * stride_mm + n * stride_mn
        val = tl.load(ptr)
        for j in range(0, 8):
            if val > top_vals[j]:
                top_vals[j] = val
                top_idx[j] = n
                break
    out_idx_ptrs = idx_ptr + m * 8 + tl.arange(0, 8)
    out_val_ptrs = vals_ptr + m * 8 + tl.arange(0, 8)
    tl.store(out_idx_ptrs, top_idx)
    tl.store(out_val_ptrs, top_vals)


@triton.jit
def _normalize_scale_kernel(
    vals_ptr, routed_ptr, M, K, SCALING: tl.constexpr,
    stride_vm, stride_rm,
):
    m = tl.program_id(0)
    if m >= M:
        return
    total = tl.zeros((), dtype=tl.float32)
    for k in range(0, K):
        ptr = vals_ptr + m * stride_vm + k
        val = tl.load(ptr)
        total += val
    # avoid division by zero
    total = tl.maximum(total, 1e-20)
    for k in range(0, K):
        ptr = vals_ptr + m * stride_vm + k
        val = tl.load(ptr)
        scaled = val / total * SCALING
        out_ptr = routed_ptr + m * stride_rm + k
        tl.store(out_ptr, scaled)


class ModelNew(nn.Module):
    def __init__(self, hidden_dim: int = HIDDEN_DIM, routed_scaling_factor: float = 1.0):
        super().__init__()
        self.hidden_dim = hidden_dim
        self.routed_scaling_factor = routed_scaling_factor

    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor):
        # Triton-only forward: no torch ops in host
        if not TRITON_AVAILABLE or not hidden_states.is_cuda or not weight.is_cuda or not expert_bias.is_cuda:
            raise RuntimeError("Triton is required but not available or tensors are not on CUDA.")

        # Ensure contiguous and dtype
        hidden = hidden_states.contiguous().to(torch.float32)       # [M, K]
        weight = weight.contiguous().to(torch.float32)              # [N, K]
        bias = expert_bias.contiguous().to(torch.float32)           # [N]

        M = hidden.shape[0]
        K = hidden.shape[1]
        N = weight.shape[0]
        assert N == NUM_EXPERTS, f"num_experts mismatch: expected {NUM_EXPERTS}, got {N}"
        assert K == self.hidden_dim, f"hidden_dim mismatch: expected {self.hidden_dim}, got {K}"

        # Allocate outputs and intermediates
        logits = torch.empty((M, N), dtype=torch.float32, device=hidden.device)       # [M, N]
        scores = torch.empty((M, N), dtype=torch.float32, device=hidden.device)       # [M, N]
        scores_for_routing = torch.empty((M, N), dtype=torch.float32, device=hidden.device)  # [M, N]
        group_scores = torch.empty((M, N_GROUP), dtype=torch.float32, device=hidden.device)   # [M, 8]
        group_idx = torch.empty((M, TOPK_GROUP), dtype=torch.int32, device=hidden.device)     # [M, 4]
        score_mask = torch.empty((M, N), dtype=torch.int32, device=hidden.device)             # [M, 256]
        masked_scores = torch.empty((M, N), dtype=torch.float32, device=hidden.device)        # [M, 256]
        top8_idx = torch.empty((M, TOP_K), dtype=torch.int32, device=hidden.device)           # [M, 8]
        top8_vals = torch.empty((M, TOP_K), dtype=torch.float32, device=hidden.device)        # [M, 8]
        topk_weight = torch.empty((M, TOP_K), dtype=torch.float32, device=hidden.device)      # [M, 8]

        # Launch Triton kernels
        # 1) Matmul for logits
        BLOCK_K = 64
        grid_matmul = (M,)
        _row_matmul_kernel[grid_matmul](

            M, K, N,
            hidden.stride(0), hidden.stride(1),
            weight.stride(0), weight.stride(1),
            logits.stride(0), logits.stride(1),

        # 2) Sigmoid

            logits, scores,
            logits.stride(0), logits.stride(1),

        # 3) Add bias
        _bias_add_kernel[(M,)](
            scores, bias, scores_for_routing,
            scores.stride(0), scores.stride(1),
            scores_for_routing.stride(0), scores_for_routing.stride(1),
            bias.stride(0),
            num_warps=1, num_stages=1,
        )

        # 4) Group top-2 sum: reshape implicitly by loops
        _group_top2_sum_kernel[(M,)](

            M, N_GROUP, EXPERTS_PER_GROUP,
            scores_for_routing.stride(0), scores_for_routing.stride(1),
            group_scores.stride(0), group_scores.stride(1),
            num_warps=1, num_stages=1,
        )

        # 5) Select top-4 groups
        _select_top4_groups_kernel[(M,)](
            group_scores, group_idx,
            M, N_GROUP,
            group_scores.stride(0), group_scores.stride(1),
            num_warps=1, num_stages=1,

        # 6) Build expert-level mask: set 1 for selected groups’ 32 experts, 0 otherwise
        _build_group_mask_kernel[(M,)](
            group_idx, score_mask,
            M, N_GROUP, EXPERTS_PER_GROUP,
            group_idx.stride(0), group_idx.stride(1),
            score_mask.stride(0),
            num_warps=1, num_stages=1,
        )

        # 7) Masked fill: set non-selected expert scores to -inf
        NEG_INF = -1.0e20
        _masked_fill_kernel[(M,)](
            scores_for_routing, score_mask, masked_scores,
            M, N,
            scores_for_routing.stride(0), scores_for_routing.stride(1),
            masked_scores.stride(0),
            NEG_INF,
            num_warps=1, num_stages=1,
        )

        # 8) Final top-8 selection from masked scores
        _final_top8_kernel[(M,)](
            masked_scores, top8_idx, top8_vals,
            M, N,
            masked_scores.stride(0), masked_scores.stride(1),
            top8_idx.stride(0),
            top8_vals.stride(0),
            NEG_INF,
            num_warps=1, num_stages=1,
        )

        # 9) Normalize and apply routed scaling factor
        _normalize_scale_kernel[(M,)](
            top8_vals, topk_weight,
            M, TOP_K, self.routed_scaling_factor,
            top8_vals.stride(0),
            topk_weight.stride(0),
            num_warps=1, num_stages=1,
        )

        # Return indices and normalized weights (top8_idx is not used in original, but we produce topk_weight)
        # The original returns (topk_idx, topk_weight). We return the topk_weight; indices were not used in the original.
        # To match the original signature, we return top8_idx (indices) and topk_weight (normalized routed weights).
        return top8_idx, topk_weight
