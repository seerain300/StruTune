import math

# Kernel 1: GEMM logits = hidden @ weight.T, logits[M, N], hidden[M, K], weight[N, K]
@triton.jit
def _linear_proj_kernel(
    hidden, weight, logits,
    M, K, N,
    stride_hm, stride_hk,
    stride_wn, stride_wk,
    stride_lm, stride_ln,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile along M
    pid_n = tl.program_id(1)  # tile along N
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # Iterate over K in chunks
    for k0 in range(0, K, TILE_K):
        offs_k = k0 + tl.arange(0, TILE_K)
        a_ptrs = hidden + (offs_m[:, None] * stride_hm + offs_k[None, :] * stride_hk)
        b_ptrs = weight + (offs_n[None, :] * stride_wn + offs_k[:, None] * stride_wk)
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_n[None, :] < N) & (offs_k[:, None] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)  # [TILE_M, TILE_K]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)  # [TILE_K, TILE_N]
        acc += tl.dot(a, b)

    # Store acc to logits
    logits_ptrs = logits + (offs_m[:, None] * stride_lm + offs_n[None, :] * stride_ln)
    mask_out = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(logits_ptrs, acc, mask=mask_out)

# Kernel 2: scores = sigmoid(logits) + expert_bias, scores[M, N]
@triton.jit
def _sigmoid_add_bias_kernel(
    logits, bias, scores,
    M, N,
    stride_lm, stride_ln,
    stride_bn,  # bias stride (1D)
    stride_sm, stride_sn,
):
    pid = tl.program_id(0)
    total = M * N
    idx = pid
    # guard
    if idx >= total:
        return
    row = idx // N
    col = idx % N
    # Load logits[row, col]
    ptr_l = logits + row * stride_lm + col * stride_ln
    l = tl.load(ptr_l)
    # Sigmoid
    s = 1.0 / (1.0 + tl.exp(-l))
    # Load bias[col]
    ptr_b = bias + col * stride_bn
    b = tl.load(ptr_b)
    # Store
    ptr_s = scores + row * stride_sm + col * stride_sn
    tl.store(ptr_s, s + b)

# Kernel 3: group top-2 sum per token, group_scores[M, 8]
@triton.jit
def _group_top2_sum_kernel(
    scores, group_scores,
    M, N,
    EXP_PER_GROUP: tl.constexpr,  # 32
    NUM_GROUPS: tl.constexpr,     # 8
    stride_sm, stride_sn,
    stride_gm, stride_gn,
):
    t = tl.program_id(0)
    if t >= M:
        return
    # Compute top-2 sum per group
    # For each group g in 0..NUM_GROUPS-1
    for g in range(0, NUM_GROUPS):
        start = g * EXP_PER_GROUP
        vals = scores[t, start : start + EXP_PER_GROUP]
        # Initialize top1 and top2 using first two (handle up to 32)
        # We'll load first two; if there are fewer, use -inf
        top1 = tl.full((), -float('inf'), dtype=tl.float32)
        top2 = tl.full((), -float('inf'), dtype=tl.float32)
        # Unrolled loop across 32
        for j in range(0, EXP_PER_GROUP):
            val = vals[j]
            if j == 0:
                top1 = val
            elif j == 1:
                top2 = val
            else:
                # Compare val with current top1
                if val > top1:
                    # swap: val -> top1, previous top1 -> top2
                    temp = top1
                    top1 = val
                    top2 = temp
                elif val > top2:
                    # val becomes new top2
                    top2 = val
        group_scores[t, g] = top1 + top2

# Kernel 4: select top-4 groups per token, top4_groups[M, 4] (int32 indices)
@triton.jit
def _select_top4_groups_kernel(
    group_scores, top4_groups,
    M, NUM_GROUPS: tl.constexpr,  # 8
    stride_gm, stride_gn,
    stride_tm, stride_tn,
):
    t = tl.program_id(0)
    if t >= M:
        return
    # Initialize candidates with group scores and indices
    # We’ll maintain max4 and their indices using iterative max-finding.
    # Create array of scores and indices: cand scores [NUM_GROUPS], cand idx [NUM_GROUPS]
    # Triton doesn’t support dynamic arrays; we will use repeated comparisons.
    # We need to load each group score and keep track of top4 via pairwise comparisons.
    # Implementation: bubble-like selection (works for small NUM_GROUPS=8).
    # Original order of groups is implicit; we’ll select via comparisons against current max1..max4.
    # We’ll use fixed loops over groups 0..7 and maintain 4 slots.
    # Note: Triton supports static loops; we can use Python range with NUM_GROUPS as constexpr.
    # Initialize max arrays
    max1 = tl.full((), -float('inf'), dtype=tl.float32)
    idx1 = tl.full((), -1, dtype=tl.int32)
    max2 = tl.full((), -float('inf'), dtype=tl.float32)
    idx2 = tl.full((), -1, dtype=tl.int32)
    max3 = tl.full((), -float('inf'), dtype=tl.float32)
    idx3 = tl.full((), -1, dtype=tl.int32)
    max4 = tl.full((), -float('inf'), dtype=tl.float32)
    idx4 = tl.full((), -1, dtype=tl.int32)

    # Iterate over groups and update max1..4
    for g in range(0, NUM_GROUPS):
        gs = tl.load(group_scores + t * stride_gm + g * stride_gn)
        # Compare with max1
        if gs > max1:
            # shift others down
            m2 = max1; i2 = idx1
            m3 = max2; i3 = idx2
            m4 = max3; i4 = idx3
            max1 = gs
            idx1 = tl.full((), g, dtype=tl.int32)
            # assign new m2..m4
            max2 = m2; idx2 = i2
            max3 = m3; idx3 = i3
            max4 = m4; idx4 = i4
        # Compare with max2
        elif gs > max2:
            m3 = max2; i3 = idx2
            m4 = max3; i4 = idx3
            max2 = gs
            idx2 = tl.full((), g, dtype=tl.int32)
            max3 = m3; idx3 = i3
            max4 = m4; i4
        # Compare with max3

            m4 = max3; i4 = idx3
            max3 = gs
            idx3 = tl.full((), g, dtype=tl.int32)
            max4 = m4; idx4 = i4
        # Compare with max4
        elif gs > max4:
            max4 = gs
            idx4 = tl.full((), g, dtype=tl.int32)

    # Store top4 indices
    tl.store(top4_groups + t * stride_tm + 0 * stride_tn, idx1)
    tl.store(top4_groups + t * stride_tm + 1 * stride_tn, idx2)
    tl.store(top4_groups + t * stride_tm + 2 * stride_tn, idx3)
    tl.store(top4_groups + t * stride_tm + 3 * stride_tn, idx4)

# Kernel 5: mask non-selected groups in scores → masked_scores[M, N]
@triton.jit
def _mask_nonselected_groups_kernel(
    scores, top4_groups, masked_scores,
    M, N, EXP_PER_GROUP: tl.constexpr,  # 32
    stride_sm, stride_sn,
    stride_tm, stride_tn,
    stride_mm, stride_mn,
):
    t = tl.program_id(0)
    if t >= M:
        return
    # Set all to -inf
    for j in range(0, N):
        ptr = masked_scores + t * stride_mm + j * stride_mn
        tl.store(ptr, -float('inf'))
    # Restore selected groups
    for k in range(0, 4):
        g = tl.load(top4_groups + t * stride_tm + k * stride_tn)
        g = tl.cast(g, tl.int32)
        start = g * EXP_PER_GROUP
        for jj in range(0, EXP_PER_GROUP):
            ptr = scores + t * stride_sm + (start + jj) * stride_sn
            val = tl.load(ptr)
            ptr_mask = masked_scores + t * stride_mm + (start + jj) * stride_mn
            tl.store(ptr_mask, val)

        max_val = tl.full((), -float('inf'), dtype=tl.float32)

            ptr = masked_scores + t * stride_mm + j * stride_mn
            val = tl.load(ptr)

            selected = False
            for m in range(0, s):  # since s starts at 0, this loop is empty; for safety we can keep it

                pass

            if val > max_val:

                max_idx = tl.full((), j, dtype=tl.int32)

# Kernel 7: normalize and scale to produce topk_weight [M, 8]
@triton.jit
def _normalize_and_scale_kernel(
    masked_scores, top8_indices, topk_weight,
    M, N,
    routed_scaling_factor,  # float
    stride_mm, stride_mn,
    stride_tm, stride_tn,
):
    t = tl.program_id(0)
    if t >= M:
        return
    # Gather selected scores: top8_indices gives column indices
    # We need to read masked_scores[t, idx] for s in 0..7
    denom = tl.full((), 0.0, dtype=tl.float32)
    for s in range(0, 8):
        idx = tl.load(top8_indices + t * stride_tm + s * stride_tn)
        ptr = masked_scores + t * stride_mm + idx * stride_mn
        val = tl.load(ptr)
        denom += val
    # Add small epsilon
    eps = 1e-20
    denom = denom + eps
    # Compute normalized and scaled weights
    for s in range(0, 8):
        idx = tl.load(top8_indices + t * stride_tm + s * stride_tn)
        ptr = masked_scores + t * stride_mm + idx * stride_mn
        val = tl.load(ptr)
        weight = val / denom
        weight = weight * routed_scaling_factor
        out_ptr = topk_weight + t * stride_tm + s * stride_tn
        tl.store(out_ptr, weight)
