import triton
import triton.language as tl

            out_idx_ptr,                # *int32,   [num_tokens*8]

                        max1_idx = start + i
                max2 = -float('inf')

                    if (v > max2) and (start + i != max1_idx):

        sum_vals = selected_original_logits + 1e-20
        row_sums = sum_vals.sum(dim=1)  # [num_tokens]
        denom = row_sums  # [num_tokens]
        topk_weight = (sum_vals / denom.unsqueeze(1)) * routed_scaling_factor  # [num_tokens, 8]

        return topk_idx, topk_weight
