@triton.jit
def gemv_linear_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Am, stride_Ak,
                        stride_Bn, stride_Bk,
                        stride_Cm, stride_Cn,
                        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute C[m, n] = sum_k A[m, k] * B[n, k]
    Inputs:
      A: [M, K], row-major
      B: [N, K], row-major (note: weight is [E, K] here, N=E)
      C: [M, N], row-major
    Grid:
      (M, ceil_div(N, BLOCK_N))
    """
    m = tl.program_id(0)
    n_block = tl.program_id(1)
    n_start = n_block * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < N

    acc = tl.zeros([BLOCK_N], dtype=tl.float32)
    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # Load A[m, k]
        a = tl.load(A_ptr + m * stride_Am + k_offsets * stride_Ak,
                    mask=mask_k, other=0.0)  # [BLOCK_K]

        # Load B[n, k] block for all n in this block
        b = tl.load(B_ptr + n_offsets[:, None] * stride_Bn + k_offsets[None, :] * stride_Bk,
                    mask=mask_n[:, None] & mask_k[None, :], other=0.0)  # [BLOCK_N, BLOCK_K]

        # Accumulate: acc[n] += dot(a, b[n, :])
        acc += tl.sum(b * a[None, :], axis=1)

    # Store C[m, n]
    tl.store(C_ptr + m * stride_Cm + n_offsets * stride_Cn, acc, mask=mask_n)


@triton.jit
def sigmoid_bias_kernel(X_ptr, Bias_ptr, Y_ptr,
                         M, N,
                         stride_Xm, stride_Xn,
                         stride_Bn,
                         stride_Ym, stride_Yn,
                         BLOCK_N: tl.constexpr):
    """
    Y[m, n] = sigmoid(X[m, n]) + Bias[n]
    Inputs:
      X: [M, N]
      Bias: [N]
      Y: [M, N]
    Grid:
      (M, ceil_div(N, BLOCK_N))
    """
    m = tl.program_id(0)
    for n_start in range(0, N, BLOCK_N):
        n_offsets = n_start + tl.arange(0, BLOCK_N)
        mask = n_offsets < N
        x = tl.load(X_ptr + m * stride_Xm + n_offsets * stride_Xn, mask=mask, other=0.0)
        b = tl.load(Bias_ptr + n_offsets * stride_Bn, mask=mask, other=0.0)
        y = 1.0 / (1.0 + tl.exp(-x)) + b
        tl.store(Y_ptr + m * stride_Ym + n_offsets * stride_Yn, y, mask=mask)

@triton.jit
def expand_group_mask_kernel(GroupMask_ptr, ExpandedMask_ptr,
                             M, G, E_PER_GROUP,
                             stride_Gm, stride_Gg,
                             stride_Em, stride_En,
                             BLOCK_G: tl.constexpr, BLOCK_E: tl.constexpr):
    """
    Expand GroupMask [M, G] to ExpandedMask [M, E], where for each selected group g, set E_PER_GROUP=32 entries to 1.0.
    Grid:
      (M, ceil_div(E, BLOCK_E))
    """
    m = tl.program_id(0)
    e_block = tl.program_id(1)
    e_start = e_block * BLOCK_E
    e_offsets = e_start + tl.arange(0, BLOCK_E)
    mask_e = e_offsets < E_PER_GROUP  # per-group, 32 elements

    # Loop over groups
    for g in range(0, G):
        # Read mask for this group for token m
        mask_val = tl.load(GroupMask_ptr + m * stride_Gm + g * stride_Gg)
        # If selected, write 1.0 to this group's 32 entries
        if mask_val > 0:

@triton.jit
def normalize_scale_kernel(Scores_ptr, Indices_ptr, Weights_ptr,
                           M, K, scaling,
                           stride_Sm, stride_Sk,
                           stride_Ims, stride_Iks,
                           stride_Wm, stride_Wk):
    """
    For each token m, gather selected_scores from Scores using Indices, normalize by sum, scale by 'scaling',
    and write to Weights.
    Inputs:
      Scores: [M, E], float32
      Indices: [M, K], int32
      Weights: [M, K], float32
      scaling: float32
    Grid:
      (M, 1) since we handle one token per program
    """
    m = tl.program_id(0)
    # Load selected indices for token m
    for k in range(0, K):
        idx = tl.load(Indices_ptr + m * stride_Ims + k * stride_Iks)  # int32
        score = tl.load(Scores_ptr + m * stride_Sm + idx * stride_Sk)  # float32
        # Sum of selected scores
        total = tl.zeros((), dtype=tl.float32)
        for i in range(0, K):
            idj = tl.load(Indices_ptr + m * stride_Ims + i * stride_Iks)
            sj = tl.load(Scores_ptr + m * stride_Sm + idj * stride_Sk)
            total += sj
        # Avoid division by zero
        total = tl.maximum(total, 1e-20)
        weight = (score / total) * scaling
        tl.store(Weights_ptr + m * stride_Wm + k * stride_Wk, weight)

    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):

        hidden_states: [M, K], float32, CUDA
        weight: [E, K], float32, CUDA
        expert_bias: [E], float32, CUDA
        routed_scaling_factor: float
        Returns:
        - topk_idx: [M, 8], int64
        - topk_weight: [M, 8], float32
        """
        M, K = hidden_states.shape
        E = weight.shape[0]
        G = 8
        E_PER_GROUP = E // G  # 32

        # 1) Triton GEMV: logits = hidden_states @ weight.T → [M, E]
        logits = torch.empty((M, E), dtype=torch.float32, device=hidden_states.device)
        BLOCK_N = 64  # number of experts per program

        BLOCK_K = 128  # chunk of K
        grid = (M, grid_n)
        gemv_linear_kernel[grid](
            hidden_states, weight, logits,
            M, E, K,
            hidden_states.stride(0), hidden_states.stride(1),
            weight.stride(0), weight.stride(1),
            logits.stride(0), logits.stride(1),
            BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

        # 2) Triton sigmoid + add expert bias: scores = sigmoid(logits) + expert_bias → [M, E]
        scores = torch.empty_like(logits)
        BLOCK_N2 = 128
        grid2 = (M, (E + BLOCK_N2 - 1) // BLOCK_N2)
        sigmoid_bias_kernel[grid2](
            logits, expert_bias, scores,
            M, E,

        )

        group_scores = torch.empty((M, G), dtype=torch.float32, device=hidden_states.device)

            M, G, E,
            scores.stride(0), scores.stride(1), scores.stride(2),
            group_scores.stride(0), group_scores.stride(1),
            E_PER_GROUP=32,

            group_idx, group_mask,
            M, G,
            group_idx.stride(0), group_idx.stride(1),
            group_mask.stride(0), group_mask.stride(1),

        )

        # Note: Implementing exact vector arg-topk in Triton requires elementwise index operations,
