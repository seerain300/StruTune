import torch.nn as nn
import triton
import triton.language as tl

        # hidden_states: [num_tokens, hidden_dim], float32
        # weight: [num_experts, hidden_dim], float32 (num_experts must be 256)
        # expert_bias: [num_experts], float32
        # routed_scaling_factor: float

        # Ensure contiguity and device compatibility
        hidden = hidden_states.contiguous()
        weight = weight.contiguous()
        bias = expert_bias.contiguous()

        num_tokens = hidden.shape[0]
        hidden_dim = hidden.shape[1]
        num_experts = weight.shape[0]
        assert num_experts == 256, "num_experts must be 256"

        # Output tensors
        topk_idx = torch.empty((num_tokens, 8), dtype=torch.int32, device=hidden.device)
        topk_weight = torch.empty((num_tokens, 8), dtype=torch.float32, device=hidden.device)

        # Launch Triton kernel: 2D grid over tokens and expert blocks
        grid = (num_tokens, (num_experts + 127) // 128)

        @triton.jit
        def _group_topk_routing(hidden_ptr, weight_ptr, bias_ptr, topk_idx_ptr, topk_w_ptr, num_tokens, hidden_dim, num_experts, scaling):
            # One program per token; iterate over 128-sized expert chunks
            pid_token = tl.program_id(0)
            pid_blk = tl.program_id(1)

            # Vector of 128 experts per program (loop over chunks)
            offs = pid_blk * 128 + tl.arange(0, 128)
            mask = offs < num_experts

            # Load hidden row for this token
            hidden_row_ptr = hidden_ptr + pid_token * hidden_dim

            # Compute scores for this token over the chunk of 128 experts
            scores = tl.zeros([128], dtype=tl.float32)
            for j in range(0, hidden_dim):
                h = tl.load(hidden_row_ptr + j)
                w = tl.load(weight_ptr + offs * hidden_dim + j, mask=mask, other=0.0)
                scores += h * w

            # Sigmoid and add bias
            scores = 1.0 / (1.0 + tl.exp(-scores))
            scores = scores + tl.load(bias_ptr + offs, mask=mask, other=0.0)

            # Reshape to groups [8, 32]
            group_scores = scores.view(8, 32)

            # Compute top-2 per group and sum to get group_scores [8]
            group_sums = tl.zeros([8], dtype=tl.float32)
            for g in range(0, 8):
                vals = group_scores[g, :]
                m = mask32 = (offs % 32) < hidden_dim  # simulate top-2 selection over 32 dims
                # top-2 via argmax twice; implementation detail: select two largest with care
                # First, find max index
                max_val = -float('inf')
                max_idx = 0
                for i in range(0, 32):
                    # get ith value; mask to valid
                    v = vals[i]
                    if v > max_val:
                        max_val = v
                        max_idx = i
                vals = vals - max_val  # subtract max, so it becomes 0
                second_val = -float('inf')
                for i in range(0, 32):
                    if i != max_idx and vals[i] > second_val:
                        second_val = vals[i]
                group_sums[g] = max_val + second_val

            # Select top-4 groups
            # We need per-token top-k on group_sums
            topk_indices = tl.zeros([8], dtype=tl.int32)
            # Implement top-4 via repeated argmax:
            selected = tl.zeros([8], dtype=tl.int32)
            for t in range(0, 4):
                best = -float('inf')
                chosen = 0
                for g in range(0, 8):
                    if selected[g] == 0 and group_sums[g] > best:
                        best = group_sums[g]
                        chosen = g
                selected[chosen] = 1
                topk_indices[t] = chosen

            # Build group_mask for these 8 groups
            group_mask = tl.zeros([8], dtype=tl.int32)
            group_mask[topk_indices] = 1

            # Expand group_mask to per-expert mask and apply to scores
            # For each group g in top4, set scores for its 32 experts to -inf if group_mask[g]==0
            for g in range(0, 8):
                if group_mask[g] == 0:
                    base = g * 32
                    group_experts = offs[base:base+32]
                    scores[group_experts] = -float('inf')

            # Now pick top-8 from scores (duplicates allowed)
            top8 = tl.zeros([8], dtype=tl.int32)
            top8_vals = tl.zeros([8], dtype=tl.float32)
            for t in range(0, 8):
                best = -float('inf')
                chosen = 0
                for e in range(0, 128):
                    if mask[e] and scores[e] > best:
                        best = scores[e]
                        chosen = e
                top8[t] = chosen
                top8_vals[t] = best
                # zero out chosen to avoid duplicates in remaining slots
                scores[chosen] = -float('inf')

            # Gather original logits for selected indices and normalize
            denom = 0.0
            for k in range(0, 8):
                e = top8[k]
                # recompute original logits for e
                total = 0.0
                for j in range(0, hidden_dim):
                    h = tl.load(hidden_row_ptr + j)
                    w = tl.load(weight_ptr + e * hidden_dim + j)
                    total += h * w
                orig = total
                denom += orig + 1e-20

            # Store outputs
            base = pid_token * 8
            for k in range(0, 8):
                tl.store(topk_idx_ptr + base + k, top8[k])
                # recompute original logits again is expensive; instead use top8_vals and denom from masked scores,

                orig = total
                weight_k = (orig + 1e-20) / denom
                weight_k *= scaling
                tl.store(topk_w_ptr + base + k, weight_k)

            hidden, weight, bias, topk_idx, topk_weight,
            num_tokens, hidden_dim, num_experts, routed_scaling_factor
        )
