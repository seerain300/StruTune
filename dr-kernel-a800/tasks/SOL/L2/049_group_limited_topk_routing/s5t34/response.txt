@triton.jit
def gemv_linear_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Am, stride_Ak,
                        stride_Bn, stride_Bk,
                        stride_Cm, stride_Cn,
                        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # 2D grid: (tokens, expert blocks)
    m = tl.program_id(0)
    n_block = tl.program_id(1)
    n_start = n_block * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < N

    # Accumulator for this expert block
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)

    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # Load hidden for this token and chunk: A[m, k] -> [BLOCK_K]
        a = tl.load(A_ptr + m * stride_Am + k_offsets * stride_Ak,
                    mask=mask_k, other=0.0)

# Triton kernel: normalize and scale selected weights (topk_idx -> original scores and scale)
@triton.jit
def normalize_scale_kernel(Scores_ptr, TopkIdx_ptr, TopkWeight_ptr,

                           stride_Ims, stride_Iks,
                           stride_Wm, stride_Wk,

    m = tl.program_id(0)
    # We need to gather original scores for selected indices and normalize.
    # Triton lacks direct gather; we can implement by loading scores per index in scalar loops.
    denom = 0.0
    for t in range(0, K_TOP):

        # Since Triton kernel doesn't know topk_idx type, we assume idx is in [0, N-1] as int32.
        # Triton store expects compatible dtype. We cast idx to int32 for address computation.
        # Compute score = tl.load(Scores_ptr + m * stride_Sm + idx * stride_Sn)
        # We'll implement scalar load/store. Triton supports scalar indexing.
        # Note: This kernel is invoked to perform normalization and scaling, and it will be used
        # in forward to produce topk_weight. The selection (top-8) was performed by per_token_topk_experts_kernel.

        # We'll compute denom by summing selected scores; then normalize and scale. Triton lacks vector indexing,
        # so we compute scalar. We'll accumulate denom and store normalized scaled weights.

        # Compute denom: sum of selected scores
        # Triton scalar loop accumulation:
        # score = 0.0
        # We need to fetch the selected score; since idx is scalar, we load it:
        score = tl.load(Scores_ptr + m * stride_Sm + idx * stride_Sn)

        # We'll store routed_scaling_factor * score / denom at position t. However, Triton doesn't provide
