import torch
import torch.nn as nn
import triton
import triton.language as tl


# Constants used in the Triton kernel
NUM_GROUPS = 8
EXPERTS_PER_GROUP = 32
NUM_EXPERTS = 256
TOPK_GROUP = 4
TOPK = 8


@triton.jit
def _compute_routing_one_token(
    hidden_ptr,                      # *f32, [num_tokens, hidden_dim]
    weight_ptr,                      # *f32, [num_experts, hidden_dim]
    expert_bias_ptr,                 # *f32, [num_experts]
    scores_for_routing_ptr,          # *f32, [num_tokens, num_experts]
    topk_idx_ptr,                    # *i32, [num_tokens, TOPK] flattened row-major
    selected_original_logits_ptr,    # *f32, [num_tokens, TOPK] flattened row-major
    num_tokens: tl.constexpr,        # int
    num_experts: tl.constexpr,       # int, 256
    hidden_dim: tl.constexpr,        # int
    token_id: tl.constexpr,          # int in [0, num_tokens)
):
    # Compute scores_for_routing = sigmoid(dot(hidden[token, :], weight[e, :])) + expert_bias[e]
    # Then group, mask, select top-8
    hidden_row_ptr = hidden_ptr + token_id * hidden_dim

    # 1) Compute logits and scores_for_routing for all experts
    for e in range(0, num_experts):
        acc = 0.0
        w_row_ptr = weight_ptr + e * hidden_dim
        for j in range(0, hidden_dim):
            x = tl.load(hidden_row_ptr + j)
            w = tl.load(w_row_ptr + j)
            acc += x * w
        # acc = logits for expert e at this token
        # scores_for_routing = sigmoid(acc) + expert_bias[e]
        sig = 1.0 / (1.0 + tl.exp(-acc))
        score = sig + tl.load(expert_bias_ptr + e)
        tl.store(scores_for_routing_ptr + token_id * num_experts + e, score)

    # 2) Group top-2 scores per group and sum to form group_scores
    group_scores = tl.zeros((NUM_GROUPS,), dtype=tl.float32)
    for g in range(0, NUM_GROUPS):
        start = g * EXPERTS_PER_GROUP
        group_vals = tl.zeros((EXPERTS_PER_GROUP,), dtype=tl.float32)
        # Load scores for this group
        for i in range(0, EXPERTS_PER_GROUP):
            e_idx = start + i
            group_vals[i] = tl.load(scores_for_routing_ptr + token_id * num_experts + e_idx)
        # Compute top-2 within group
        max1 = -float('inf')
        max1_idx = -1
        for i in range(0, EXPERTS_PER_GROUP):
            v = group_vals[i]
            if v > max1:
                max1 = v
                max1_idx = start + i
        max2 = -float('inf')
        for i in range(0, EXPERTS_PER_GROUP):
            v = group_vals[i]
            if (v > max2) and (start + i != max1_idx):
                max2 = v
        group_scores[g] = max1 + max2

    # 3) Select top-4 groups (unordered). We only need indices of selected groups.
    selected_group_idx = tl.zeros((TOPK_GROUP,), dtype=tl.int32)
    for k in range(0, TOPK_GROUP):
        maxv = -float('inf')
        max_idx = -1
        for g in range(0, NUM_GROUPS):
            if group_scores[g] > maxv:
                maxv = group_scores[g]
                max_idx = g
        selected_group_idx[k] = max_idx

    # 4) Mask out non-selected groups by setting scores_for_routing to -inf
    for k in range(0, TOPK_GROUP):
        g = selected_group_idx[k]
        start = g * EXPERTS_PER_GROUP
        for i in range(0, EXPERTS_PER_GROUP):
            e_idx = start + i
            # Set non-selected group elements to -inf
            # Note: We only change scores_for_routing (not the original logits we computed before).
            # But we don't have the original logits now; however, the selection is based on scores_for_routing,
            # and we can safely zero their influence by setting to -inf. This matches the intent.
            pass  # The masking is implicit via scores_for_routing values; we will enforce by recomputing selection logic from masked scores.

    # Instead of manually masking, we recompute selection from the original scores_for_routing:
    # Create a copy of scores_for_routing and set selected groups to -inf
    masked_scores = tl.zeros((num_experts,), dtype=tl.float32)
    for e in range(0, num_experts):
        keep = 0
        for k in range(0, TOPK_GROUP):
            g = selected_group_idx[k]
            start = g * EXPERTS_PER_GROUP
            if (e >= start) and (e < start + EXPERTS_PER_GROUP):
                keep = 1
                break
        v = tl.load(scores_for_routing_ptr + token_id * num_experts + e)
        masked_scores[e] = v if keep == 1 else (-float('inf'))

    # 5) Select top-8 from masked scores (duplicates allowed)
    selected_indices = tl.zeros((TOPK,), dtype=tl.int32)
    for t in range(0, TOPK):
        maxv = -float('inf')
        max_idx = -1
        for e in range(0, num_experts):
            v = masked_scores[e]
            if v > maxv:
                maxv = v
                max_idx = e
        selected_indices[t] = max_idx
        masked_scores[max_idx] = -float('inf')

    # 6) Recompute original logits for selected indices and write to output buffers
    # Also write selected_indices as topk_idx
    for t in range(0, TOPK):
        e = selected_indices[t]
        acc = 0.0
        w_row_ptr = weight_ptr + e * hidden_dim
        hidden_row_ptr = hidden_ptr + token_id * hidden_dim
        for j in range(0, hidden_dim):
            x = tl.load(hidden_row_ptr + j)
            w = tl.load(w_row_ptr + j)
            acc += x * w
        # Store original logits for this selected expert
        out_base = token_id * TOPK
        tl.store(selected_original_logits_ptr + out_base + t, acc)
        # Store index
        tl.store(topk_idx_ptr + out_base + t, e)


class ModelNew(nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):
        # hidden_states: [num_tokens, hidden_dim]
        # weight: [num_experts, hidden_dim], num_experts=256
        # expert_bias: [num_experts]
        # routed_scaling_factor: float

        num_tokens = hidden_states.shape[0]
        hidden_dim = hidden_states.shape[1]
        num_experts = weight.shape[0]
        assert num_experts == 256, "num_experts must be 256"
        assert weight.shape[1] == hidden_dim, "weight's second dimension must match hidden_dim"

        # Allocate outputs
        topk_idx = torch.empty((num_tokens, TOPK), dtype=torch.int32, device=hidden_states.device)
        selected_original_logits = torch.empty((num_tokens, TOPK), dtype=torch.float32, device=hidden_states.device)

        # Launch Triton kernel: one program per token
        grid = (num_tokens,)
        _compute_routing_one_token[grid](
            hidden_states,                      # hidden_ptr
            weight,                             # weight_ptr
            expert_bias,                        # expert_bias_ptr
            # scores_for_routing_ptr: we don't need to pass it as we compute and keep in registers (not actually done; use a temporary buffer to store the raw logits and then derive masked scores, but Triton kernel does not have direct access to other tensors; the kernel computes everything).
            # Triton kernel will write topk_idx and selected_original_logits.
            topk_idx,                           # topk_idx_ptr
            selected_original_logits,           # selected_original_logits_ptr
            num_tokens=num_tokens,
            num_experts=num_experts,
            hidden_dim=hidden_dim,
        )

        # Compute topk_weight on host: normalize original logits by sum(original_selected + 1e-20) and apply scaling factor
        original_selected = selected_original_logits  # already contains original logits per selected expert
        denom = original_selected + 1e-20            # [num_tokens, 8]
        denom = denom.sum(dim=1, keepdim=True)       # [num_tokens, 1]
        topk_weight = (original_selected / denom) * routed_scaling_factor  # [num_tokens, 8]

        # Return indices and weights
        return topk_idx, topk_weight
