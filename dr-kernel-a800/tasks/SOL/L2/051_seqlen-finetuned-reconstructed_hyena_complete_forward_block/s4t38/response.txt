import torch
import triton
import triton.language as tl


# 1) Triton LayerNorm (affine) over last dim for a [M, N] tensor (M = B*S, N = D).
# Two kernels:
#   - compute per-row sum and sum of squares
#   - normalize and apply affine

@triton.jit
def _layernorm_mean_var_kernel(
    X_ptr,             # *fp32, input [M, N]
    SUM_ptr,           # *fp32, per-row sum [M]
    SUMSQ_ptr,         # *fp32, per-row sum of squares [M]
    M, N,
    stride_xm, stride_xn,
):
    pid = tl.program_id(axis=0)
    if pid >= M:
        return
    acc = tl.zeros((), dtype=tl.float32)
    acc_sq = tl.zeros((), dtype=tl.float32)
    # Loop over columns
    for j in range(0, N):
        x_j = tl.load(X_ptr + pid * stride_xm + j * stride_xn)
        acc += x_j
        acc_sq += x_j * x_j
    # Store results
    tl.store(SUM_ptr + pid, acc)
    tl.store(SUMSQ_ptr + pid, acc_sq)


@triton.jit
def _layernorm_norm_affine_kernel(
    X_ptr,         # *fp32, input [M, N]
    SUM_ptr,       # *fp32, sum [M]
    SUMSQ_ptr,     # *fp32, sum of squares [M]
    W_ptr,         # *fp32, weight [N]
    B_ptr,         # *fp32, bias [N]
    Y_ptr,         # *fp32, output [M, N]
    M, N,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    eps,           # float32
):
    pid = tl.program_id(axis=0)
    if pid >= M:
        return
    mean = tl.load(SUM_ptr + pid) / N
    var = tl.load(SUMSQ_ptr + pid) / N - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    # Normalize and apply affine per element
    for j in range(0, N):
        x_j = tl.load(X_ptr + pid * stride_xm + j * stride_xn)
        y_j = (x_j - mean) * inv_std
        # load affine params
        w_j = tl.load(W_ptr + j)
        b_j = tl.load(B_ptr + j)
        tl.store(Y_ptr + pid * stride_ym + j * stride_yn, y_j * w_j + b_j)


# 2) Triton elementwise addition for 3D tensors [B, S, D]
@triton.jit
def _add_3d_kernel(
    A_ptr, B_ptr, C_ptr,
    Bsz, Slen, Ddim,
    stride_ab, stride_as, stride_ad,
    stride_bb, stride_bs, stride_bd,
    stride_cb, stride_cs, stride_cd,
):
    # one program per (b, s)
    pid = tl.program_id(axis=0)  # ranges over B * S
    b = pid // Slen
    s = pid % Slen
    if (b >= Bsz) or (s >= Slen):
        return
    for d in range(0, Ddim):
        a = tl.load(A_ptr + b * stride_ab + s * stride_as + d * stride_ad)
        c = tl.load(B_ptr + b * stride_bb + s * stride_bs + d * stride_bd)
        res = a + c
        tl.store(C_ptr + b * stride_cb + s * stride_cs + d * stride_cd, res)


# 3) Triton row-wise linear: computes C[M, N] = A[M, D] @ W^T[D, N] + bias[N]
@triton.jit
def _linear_rowwise_kernel(
    A_ptr,          # *fp32, input A [M, D], contiguous
    WT_ptr,         # *fp32, weight transposed W^T [D, N], contiguous
    B_ptr,          # *fp32, bias [N]
    C_ptr,          # *fp32, output C [M, N], contiguous
    M, D, N,
    stride_am, stride_ad,
    stride_wtn, stride_wtd,   # strides for W^T (n, d)
    stride_cm, stride_cn,
    BLOCK_N: tl.constexpr,    # tile over N
    BLOCK_D: tl.constexpr,    # tile over D
):
    pid_m = tl.program_id(axis=0)  # program id over rows (M)
    if pid_m >= M:
        return
    offs_n = tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    for d_start in range(0, D, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        # Load A row slice
        a = tl.load(A_ptr + pid_m * stride_am + offs_d * stride_ad, mask=offs_d < D, other=0.0)  # [BLOCK_D]
        # Load W^T tile [BLOCK_N, BLOCK_D]
        w = tl.load(
            WT_ptr + offs_n[:, None] * stride_wtn + offs_d[None, :] * stride_wtd,
            mask=(offs_n[:, None] < N) & (offs_d[None, :] < D),
            other=0.0
        )  # [BLOCK_N, BLOCK_D]
        # Accumulate dot: sum over D tile
        acc += tl.sum(w * a[None, :], axis=1)
    # Add bias
    bias = tl.load(B_ptr + offs_n, mask=offs_n < N, other=0.0)
    acc += bias
    # Store
    tl.store(C_ptr + pid_m * stride_cm + offs_n * stride_cn, acc, mask=offs_n < N)


# 4) Triton conv1d (groups-aware) for short convolution: input [B, L_in, C] = [B, S+4, inner], weight [C, 1, K], padding (2,2), groups=C
@triton.jit
def _conv1d_groups_kernel(
    U_ptr,            # *fp32, input u_padded [B, L_in, C], contiguous
    WT_ptr,           # *fp32, weight transposed [K, C] (we pass short_conv_weight transposed to [K, C])
    BIAS_ptr,         # *fp32, bias per channel [C]
    Y_ptr,            # *fp32, output [B, L_out, C], contiguous
    Bsz, Lin, Cout,   # L_out = Lin - K + 1 (with padding), Cout = channels
    stride_ub, stride_ul, stride_uc,
    stride_wtk, stride_wtc,
    stride_yb, stride_yl, stride_yc,
    K: tl.constexpr,         # kernel size
    padding_left: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # over B * Cout
    b = pid // Cout
    c = pid % Cout
    if (b >= Bsz) or (c >= Cout):
        return
    # For each output position t_out in [0..L_out-1], compute dot across K
    # y[b, t_out, c] = sum_{k=0..K-1} u[b, t_out + padding_left + k, c] * WT[k, c] + bias[c]
    # Note: u_padded has length Lin; we ensure t_out + padding_left + k < Lin
    for t_out in range(0, Lin - K + 1):  # since padding (2,2), L_out = Lin - K + 1
        acc = tl.zeros((), dtype=tl.float32)
        # Loop over kernel taps
        for k in range(0, K):
            # compute input index
            li = t_out + padding_left + k  # 0 <= li < Lin
            u_val = tl.load(U_ptr + b * stride_ub + li * stride_ul + c * stride_uc)
            wt_val = tl.load(WT_ptr + k * stride_wtk + c * stride_wtc)
            acc += u_val * wt_val
        # add bias
        bias_val = tl.load(BIAS_ptr + c)
        acc += bias_val
        # store to output
        tl.store(Y_ptr + b * stride_yb + t_out * stride_yl + c * stride_yc, acc)


# 5) Triton pad1d for [B, S, C]: pad along sequence dimension (left=2, right=2)
@triton.jit
def _pad1d_kernel(
    X_ptr,            # *fp32, input [B, S, C], contiguous
    P_ptr,            # *fp32, output [B, S+4, C], contiguous
    Bsz, Slen, Cdim,
    stride_xb, stride_xs, stride_xc,
    stride_pb, stride_ps, stride_pc,
):
    pid = tl.program_id(axis=0)  # over B * Cdim * (Slen + 4)
    total = Bsz * Cdim * (Slen + 4)
    if pid >= total:
        return
    # decode indices
    b = pid // (Cdim * (Slen + 4))
    rem = pid % (Cdim * (Slen + 4))
    c = rem // (Slen + 4)
    ps = rem % (Slen + 4)
    # source s index
    s_src = ps - 2
    # guard
    if (b >= Bsz) or (c >= Cdim) or (s_src < 0) or (s_src >= Slen):
        return
    x_val = tl.load(X_ptr + b * stride_xb + s_src * stride_xs + c * stride_xc)
    tl.store(P_ptr + b * stride_pb + ps * stride_ps + c * stride_pc, x_val)

def _run_triton_layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float):
    # x: [B, S, D], return: [B, S, D]
    B, S, D = x.shape
    M = B * S
    # Ensure contiguous and float32 on CUDA
    x_flat = x.contiguous().view(M, D)
    sum_ = torch.empty(M, device=x.device, dtype=torch.float32)
    sumsq_ = torch.empty(M, device=x.device, dtype=torch.float32)
    # Launch mean/var kernel

    _layernorm_mean_var_kernel[grid](
        x_flat, sum_, sumsq_,
        M, D,
        x_flat.stride(0), x_flat.stride(1),
        num_warps=1
    )
    # Normalize + affine
    y_flat = torch.empty_like(x_flat)
    grid2 = (M,)
    _layernorm_norm_affine_kernel[grid2](
        x_flat, sum_, sumsq_,
        weight.contiguous(), bias.contiguous(),
        y_flat,
        M, D,
        x_flat.stride(0), x_flat.stride(1),
        y_flat.stride(0), y_flat.stride(1),
        eps,
        num_warps=1
    )
    return y_flat.view(B, S, D)


def _run_triton_add(a: torch.Tensor, b: torch.Tensor):
    # a, b: [B, S, D]
    B, S, D = a.shape
    out = torch.empty_like(a)
    grid = (B * S,)
    _add_3d_kernel[grid](
        a, b, out,
        B, S, D,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        num_warps=1
    )
    return out


def _run_triton_linear(a_flat: torch.Tensor, w: torch.Tensor, bias: torch.Tensor):
    # a_flat: [M, D], w: [N, D], return: [M, N]
    M, D = a_flat.shape
    N = w.shape[0]
    # Transpose w to [D, N] for row-wise matmul
    w_t = w.transpose(0, 1).contiguous()  # [D, N]
    c_flat = torch.empty((M, N), device=a_flat.device, dtype=torch.float32)
    # Choose tiling
    BLOCK_N = 64
    BLOCK_D = 128
    grid = (M,)
    _linear_rowwise_kernel[grid](
        a_flat, w_t, bias.contiguous(), c_flat,
        M, D, N,
        a_flat.stride(0), a_flat.stride(1),
        w_t.stride(0), w_t.stride(1),
        c_flat.stride(0), c_flat.stride(1),
        BLOCK_N=BLOCK_N, BLOCK_D=BLOCK_D,
        num_warps=4
    )
    return c_flat


def _run_triton_conv1d_groups(u_padded: torch.Tensor, w_t: torch.Tensor, bias: torch.Tensor):
    # u_padded: [B, Lin, Cout], w_t: [K, Cout], return: [B, L_out, Cout]
    B, Lin, Cout = u_padded.shape
    K = w_t.shape[0]
    L_out = Lin - K + 1  # with padding (2,2)
    y = torch.empty((B, L_out, Cout), device=u_padded.device, dtype=torch.float32)
    grid = (B * Cout,)
    _conv1d_groups_kernel[grid](
        u_padded, w_t, bias.contiguous(), y,
        B, Lin, Cout,
        u_padded.stride(0), u_padded.stride(1), u_padded.stride(2),
        w_t.stride(0), w_t.stride(1),
        y.stride(0), y.stride(1), y.stride(2),
        K=K, padding_left=2,
        num_warps=2
    )
    return y


def _run_triton_pad1d(u: torch.Tensor):
    # u: [B, S, C], return padded [B, S+4, C] with left=2, right=2
    B, S, C = u.shape
    Lin = S + 4
    u_padded = torch.empty((B, Lin, C), device=u.device, dtype=torch.float32)
    grid = (B * C * Lin,)
    _pad1d_kernel[grid](
        u, u_padded,
        B, S, C,
        u.stride(0), u.stride(1), u.stride(2),
        u_padded.stride(0), u_padded.stride(1), u_padded.stride(2),
        num_warps=1
    )
    return u_padded
