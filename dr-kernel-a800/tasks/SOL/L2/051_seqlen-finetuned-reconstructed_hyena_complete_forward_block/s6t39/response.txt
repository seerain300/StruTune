import math
import triton


# Triton LayerNorm forward for 3D tensors (B, L, D): normalize over last dim, affine
@triton.jit
def layernorm_forward_kernel(
    X_ptr,        # *const float
    W_ptr,        # *const float (gamma), shape [D]
    B_ptr,        # *const float (beta), shape [D]
    Y_ptr,        # *float
    B, L, D,      # int
    eps,          # float
    stride_xb, stride_xl, stride_xd,
    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    if b >= B or l >= L:
        return

    # Compute mean across D for (b, l)
    sum_val = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        d0 += BLOCK_SIZE
    mean = sum_val / D

    # Compute variance across D for (b, l)
    var_val = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)
        d0 += BLOCK_SIZE
    var = var_val / D
    rstd = 1.0 / tl.sqrt(var + eps)

    # Normalize and apply affine
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        gamma = tl.load(W_ptr + offs * stride_w, mask=mask, other=1.0)
        beta = tl.load(B_ptr + offs * stride_b, mask=mask, other=0.0)
        y = (x - mean) * rstd
        y = y * gamma + beta
        tl.store(Y_ptr + b * stride_yb + l * stride_yl + offs * stride_yd, y, mask=mask)
        d0 += BLOCK_SIZE


# Triton fill_ones kernel: fills a 1D flattened pointer with ones (float32)
@triton.jit
def fill_ones_kernel(
    Out_ptr,      # *float
    N: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    one = 1.0
    tl.store(Out_ptr + offs, one, mask=mask)


# Triton GEMM: A[M, K] @ W[K, N] -> C[M, N], where M = B*L, K = D, N = D
@triton.jit
def linear_gemm_kernel(
    A_ptr,        # *const float, shape [M, K] flattened
    W_ptr,        # *const float, shape [K, N] flattened
    B_ptr,        # *const float, bias [N]
    C_ptr,        # *float, output [M, N] flattened
    M, K, N,
    stride_am, stride_ak,
    stride_wk, stride_wn,
    stride_cm, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    n = tl.program_id(1)
    if (m >= M) or (n >= N):
        return
    acc = 0.0
    for k0 in range(0, K, BLOCK_K):
        for i in range(0, BLOCK_N):  # iterate over N-tile
            w_vec = tl.load(W_ptr + (k0 + tl.arange(0, BLOCK_K)) * stride_wk + n * stride_wn, mask=(k0 + tl.arange(0, BLOCK_K)) < K, other=0.0)  # shape [BLOCK_K]
            # load A row segment [BLOCK_K] for this n-column
            a_vec = tl.load(A_ptr + m * stride_am + (k0 + tl.arange(0, BLOCK_K)) * stride_ak, mask=(k0 + tl.arange(0, BLOCK_K)) < K, other=0.0)  # shape [BLOCK_K]
            acc += tl.sum(a_vec * w_vec, axis=0)
    # Add bias
    b = tl.load(B_ptr + n * stride_wn, mask=n < N, other=0.0)
    acc += b
    tl.store(C_ptr + m * stride_cm + n * stride_cn, acc)


# Triton short depthwise conv with groups and padding: conv1d on Up (padded) with groups=groups and k=3
# We assume Up shape [B, groups, L_in], Wc shape [groups, 1, 3], output shape [B, groups, L_out]
@triton.jit
def conv1d_groups_exact_kernel(
    Up_ptr,       # *const float, [B, groups, L_in] flattened
    Wc_ptr,       # *const float, [groups, 1, 3] flattened
    Bo_ptr,       # *const float, [groups] bias
    Uout_ptr,     # *float, [B, groups, L_out] flattened
    B, groups, L_in, k_size,
    pad,          # padding=2
    stride_upb, stride_upg, stride_upl,
    stride_wcg, stride_wck,
    stride_uob, stride_uog, stride_uol,
):
    b = tl.program_id(0)
    g = tl.program_id(1)
    if (b >= B) or (g >= groups):
        return
    L_out = L_in + 2 * pad - k_size + 1
    # For each output position l_out
    # Note: we'll implement as a single pass over l_out = 0..L_out-1
    # Here we implement one-dimensional tile processing over L_out using a loop
    l_out = 0
    while l_out < L_out:
        acc = 0.0
        # Compute corresponding input position with padding
        inp_pos = l_out - pad
        # Iterate over kernel k=0..k_size-1
        for k in range(0, k_size):
            # Weight scalar for this group
            w_val = tl.load(Wc_ptr + g * stride_wcg + k * stride_wck)
            # Input value at group g
            valid = (inp_pos + k >= 0) & (inp_pos + k < L_in)
            val = tl.load(Up_ptr + b * stride_upb + g * stride_upg + (inp_pos + k) * stride_upl, mask=valid, other=0.0)
            acc += val * w_val
        bval = tl.load(Bo_ptr + g)
        acc += bval
        tl.store(Uout_ptr + b * stride_uob + g * stride_uog + l_out * stride_uol, acc)
        l_out += 1


# Triton exp modulation kernel: V_in[B*D*L] -> V_out[B*D*L]
# v_new = v * (exp(-t * abs(deltas)) + shift)
# deltas has shape (D,) and broadcasts over batch and sequence via t index.
@triton.jit
def exp_mod_kernel(
    V_ptr,        # *const float
    Deltas_ptr,   # *const float, shape [D]
    B, D, L,      # int
    shift,        # float
    stride_vb, stride_vd, stride_vl,
):
    pid = tl.program_id(0)
    total = B * D * L
    if pid >= total:
        return
    b = pid // (D * L)
    rem = pid % (D * L)
    d = rem // L
    l = rem % L
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    delta = tl.load(Deltas_ptr + d)
    t = l  # position along sequence
    exp_term = tl.exp(-t * tl.abs(delta))
    v_new = v * (exp_term + shift)
    tl.store(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl, v_new)


# Triton random fill kernel: fill a 1D flattened pointer with random values (float32)
@triton.jit
def randn_fill_kernel(

):
    # Note: This is a placeholder; in real scenarios, Triton does not provide RNG directly.

    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    # Use a simple constant for demonstration; you can replace with actual RNG if available.

    tl.store(Out_ptr + offs, val, mask=mask)
