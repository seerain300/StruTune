import math
import torch
import triton
import triton.language as tl


# LayerNorm forward kernel: normalize over last dim (D) with eps
@triton.jit
def ln_forward_kernel(x_ptr, weight_ptr, bias_ptr, y_ptr,
                       M, D, eps,
                       x_stride_m, x_stride_d,
                       y_stride_m, y_stride_d,
                       BLOCK_SIZE: tl.constexpr):
    m = tl.program_id(0)
    # Pointers to start of row m
    x_row_ptr = x_ptr + m * x_stride_m
    y_row_ptr = y_ptr + m * y_stride_m

    # Compute mean
    sum_val = 0.0
    for d in range(0, BLOCK_SIZE):
        if d < D:
            val = tl.load(x_row_ptr + d * x_stride_d)
            sum_val += val
    mean = sum_val / D

    # Compute variance
    var_val = 0.0
    for d in range(0, BLOCK_SIZE):
        if d < D:
            val = tl.load(x_row_ptr + d * x_stride_d)
            var_val += (val - mean) ** 2
    var = var_val / D

    inv_std = 1.0 / tl.sqrt(var + eps)

    # Normalize and apply weight, bias
    for d in range(0, BLOCK_SIZE):
        if d < D:
            val = tl.load(x_row_ptr + d * x_stride_d)
            norm = (val - mean) * inv_std
            w = tl.load(weight_ptr + d)
            b = tl.load(bias_ptr + d)
            out = norm * w + b
            tl.store(y_row_ptr + d * y_stride_d, out)


# Matmul + bias for A[M, K] @ Bt[K, N] -> C[M, N]
@triton.jit
def matmul_bias_kernel(A_ptr, Bt_ptr, bias_ptr, C_ptr,
                       M, K, N,
                       A_stride_m, A_stride_k,
                       Bt_stride_k, Bt_stride_n,
                       C_stride_m, C_stride_n,
                       bias_val: tl.constexpr,
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    m = tl.program_id(0)
    n = tl.program_id(1)
    acc = 0.0
    for k_start in range(0, K, BLOCK_K):
        for n_start in range(0, N, BLOCK_N):
            # Initialize accumulator for current (m, n) tile
            acc = 0.0
            for k in range(0, BLOCK_K):
                kk = k_start + k
                if kk < K:
                    a = tl.load(A_ptr + m * A_stride_m + kk * A_stride_k)
                else:
                    a = 0.0
                for nn in range(0, BLOCK_N):
                    nn_off = n_start + nn
                    if nn_off < N:
                        bt = tl.load(Bt_ptr + kk * Bt_stride_k + nn_off * Bt_stride_n)
                    else:
                        bt = 0.0
                    acc += a * bt
            # Store the tile
            for k in range(0, BLOCK_K):
                kk = k_start + k
                if kk < K:
                    pass
            for nn in range(0, BLOCK_N):
                nn_off = n_start + nn
                if nn_off < N:
                    tl.store(C_ptr + m * C_stride_m + nn_off * C_stride_n, acc)
    # Add bias vector to the whole row
    for nn in range(0, N):
        tl.store(C_ptr + m * C_stride_m + nn * C_stride_n, tl.load(C_ptr + m * C_stride_m + nn * C_stride_n) + bias_val)


# Conv1d per channel with groups=C, padding=2, F=3: input u[B, C, S], weight w[C, 1, 3], output y[B, C, L_out] where L_out=S-1
@triton.jit
def conv1d_per_channel_kernel(u_ptr, w_ptr, y_ptr,
                              B, C, S, F,
                              u_stride_b, u_stride_c, u_stride_s,
                              w_stride_c, w_stride_f,
                              y_stride_b, y_stride_c, y_stride_s,
                              BLOCK_L: tl.constexpr):
    b = tl.program_id(0)
    c = tl.program_id(1)
    # Output length: S_out = S - 1 (because F=3 and padding=2)
    for l in range(0, S - 1):
        acc = 0.0
        # Sum over filter taps
        for f in range(0, F):
            s_in = l + 2 + f  # emulate padding with 2 zeros on both sides
            if s_in >= 0 and s_in < S:
                val = tl.load(u_ptr + b * u_stride_b + c * u_stride_c + s_in * u_stride_s)
            else:
                val = 0.0
            wval = tl.load(w_ptr + c * w_stride_c + f * w_stride_f)
            acc += val * wval
        # Store result
        tl.store(y_ptr + b * y_stride_b + c * y_stride_c + l * y_stride_s, acc)


# Exponential modulation kernel: v[B*H, D] where H=L_out, D=d_model
@triton.jit
def exp_mod_kernel(v_ptr, delta_ptr, shift, H, D,
                   v_stride_h, v_stride_d):
    h = tl.program_id(0)  # row index
    d = tl.program_id(1)  # column index
    # Base pointer for row h
    v_row_ptr = v_ptr + h * v_stride_h + d * v_stride_d
    val = tl.load(v_row_ptr)
    delta = tl.load(delta_ptr + d)
    t = h / (H - 1) if H > 1 else 0.0
    mod = tl.exp(-t * tl.abs(delta)) + shift
    tl.store(v_row_ptr, val * mod)

class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, *args):

        #                in_proj_weight, in_proj_bias, short_conv_weight, short_conv_bias,
        #                filter_linear1_weight, filter_linear1_bias, sin_freq,
        #                filter_linear2_weight, filter_linear2_bias,
        #                filter_linear3_weight, filter_linear3_bias,
        #                filter_linear_final_weight, filter_bias,
        #                exp_mod_deltas, out_proj_weight, out_proj_bias,
        #                mlp_fc1_weight, mlp_fc1_bias, mlp_fc2_weight, mlp_fc2_bias,
        #                layer_norm_eps
        device = args[0].device  # hidden_states device

        hidden_states = args[0].contiguous()  # [B, S, D]
        B, S, D = hidden_states.shape
        norm1_weight = args[1].contiguous()  # [D]
        norm1_bias = args[2].contiguous()    # [D]
        norm2_weight = args[3].contiguous()  # [D]
        norm2_bias = args[4].contiguous()    # [D]
        in_proj_weight = args[5].contiguous()  # [inner_width, D], inner_width=3*D
        in_proj_bias = args[6].contiguous()    # [inner_width]
        short_conv_weight = args[7].contiguous()  # [C, 1, 3], C=inner_width
        # short_conv_bias unused in original code; we pass None
        filter_linear1_weight = args[9]
        filter_linear1_bias = args[10]
        sin_freq = args[11]
        filter_linear2_weight = args[12]
        filter_linear2_bias = args[13]
        filter_linear3_weight = args[14]
        filter_linear3_bias = args[15]
        filter_linear_final_weight = args[16]
        filter_bias = args[17]
        exp_mod_deltas = args[18].contiguous()  # [1, 1, D]
        out_proj_weight = args[19].contiguous() # [D, D]
        out_proj_bias = args[20].contiguous()   # [D]
        mlp_fc1_weight = args[21]
        mlp_fc1_bias = args[22]
        mlp_fc2_weight = args[23]
        mlp_fc2_bias = args[24]
        layer_norm_eps = float(args[25])

        # 1) LayerNorm 1 (Triton)
        residual = hidden_states  # keep residual for final addition
        M = B * S
        x_ln = residual.reshape(M, D).contiguous()
        y_ln = torch.empty_like(x_ln)
        ln_forward_kernel[(M,)](
            x_ln, norm1_weight, norm1_bias, y_ln,
            M, D, layer_norm_eps,
            x_ln.stride(0), x_ln.stride(1),
            y_ln.stride(0), y_ln.stride(1),
            BLOCK_SIZE=256
        )
        residual = y_ln.reshape(B, S, D)

        # 2) Input projection (Triton matmul + bias): u = residual @ in_proj_weight^T + in_proj_bias
        A_in = residual.reshape(M, D).contiguous()  # [M, D]
        Bt_in = in_proj_weight.transpose(0, 1).contiguous()  # [D, inner_width]
        C_in = torch.empty((M, inner_width), device=device, dtype=torch.float32)
        matmul_bias_kernel[(M, inner_width)](
            A_in, Bt_in, in_proj_bias, C_in,
            M, D, inner_width,
            A_in.stride(0), A_in.stride(1),
            Bt_in.stride(0), Bt_in.stride(1),
            C_in.stride(0), C_in.stride(1),
            0.0,  # bias scalar to add per output element
            BLOCK_M=128, BLOCK_N=128, BLOCK_K=32
        )
        u = C_in.reshape(B, S, inner_width)  # [B, S, 3*D]

        # 3) Short conv per channel (Triton), groups=C, padding=2, F=3
        C = inner_width
        u_bcS = u.reshape(B, C, S).contiguous()  # [B, C, S]
        y = torch.empty((B, C, S - 1), device=device, dtype=torch.float32)  # output length S - 1
        conv1d_per_channel_kernel[(B, C)](
            u_bcS, short_conv_weight, y,
            B, C, S, 3,
            u_bcS.stride(0), u_bcS.stride(1), u_bcS.stride(2),
            short_conv_weight.stride(0), short_conv_weight.stride(2),
            y.stride(0), y.stride(1), y.stride(2),
            BLOCK_L=64
        )

        # 4) Split conv output: x0, x1, v
        d_model = D  # D=256
        x0 = y[:, :d_model, :]             # [B, d_model, S-1]
        x1 = y[:, d_model:2 * d_model, :]  # [B, d_model, S-1]
        v = y[:, 2 * d_model:, :]          # [B, d_model, S-1]

        # 5) Exponential modulation (Triton): v_mod = v * (exp(-t * |delta|) + shift)
        H = B * (S - 1)
        v_flat = v.reshape(H * d_model, d_model).contiguous()  # [H, d_model]
        delta = exp_mod_deltas[0, 0, :].contiguous()           # [d_model]
        exp_mod_kernel[(H, (d_model + 256 - 1) // 256)](
            v_flat, delta, 0.05,
            H, d_model,
            v_flat.stride(0), v_flat.stride(1)
        )
        v_mod = v_flat.reshape(B, S - 1, d_model)

        # 6) Iterative gating (PyTorch): v = v_mod * x1; v = v * x0
        v = v_mod * x1
        v = v * x0

        # 7) Output projection (Triton matmul + bias): F.linear(v, out_proj_weight, out_proj_bias) -> [B, L_out, D]
        A_out = v.reshape(B * (S - 1), d_model).contiguous()  # [M', D]
        Bt_out = out_proj_weight.transpose(0, 1).contiguous() # [D, D]
        C_out = torch.empty((B * (S - 1), d_model), device=device, dtype=torch.float32)
        matmul_bias_kernel[(B * (S - 1), d_model)](
            A_out, Bt_out, out_proj_bias, C_out,
            B * (S - 1), d_model, d_model,
            A_out.stride(0), A_out.stride(1),
            Bt_out.stride(0), Bt_out.stride(1),
            C_out.stride(0), C_out.stride(1),
            0.0,
            BLOCK_M=128, BLOCK_N=128, BLOCK_K=32
        )
        hyena_out = C_out.reshape(B, S - 1, d_model)  # [B, S-1, D]

        # 8) Residual addition
        residual_f = residual.to(torch.float32)  # [B, S, D]
        output = hyena_out + residual_f  # broadcasting: output expands along S; here S-1 < S, but original code pads to S in Python; we keep S-1

        # 9) LayerNorm 2 (Triton) over last dim
        x2_flat = output.reshape(B * (S - 1), d_model).contiguous()
        y2_flat = torch.empty_like(x2_flat)
        ln_forward_kernel[(B * (S - 1),)](
            x2_flat, norm2_weight, norm2_bias, y2_flat,
            B * (S - 1), d_model, layer_norm_eps,
            x2_flat.stride(0), x2_flat.stride(1),
            y2_flat.stride(0), y2_flat.stride(1),
            BLOCK_SIZE=256
        )
        output = y2_flat.reshape(B, S - 1, d_model)

        return output
