# Triton LayerNorm (affine) over last dim for a [M, N] tensor (M = B*S, N = D).
# Kernel 1: compute mean and variance per row.
@triton.jit
def _layernorm_mean_var_kernel(
    X_ptr,          # *fp32, input [M, N], contiguous
    SUM_ptr,        # *fp32, per-row sum [M]
    SUMSQ_ptr,      # *fp32, per-row sum of squares [M]
    M, N,
    stride_xm, stride_xn,
    eps,            # float32
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # program id over rows
    # One program handles one row
    offs = tl.arange(0, BLOCK_N)
    # Accumulators
    s = tl.zeros((), dtype=tl.float32)
    ss = tl.zeros((), dtype=tl.float32)
    # Loop over columns in tiles
    for col_start in range(0, N, BLOCK_N):
        cols = col_start + offs
        mask = cols < N
        # Load row tile
        x = tl.load(X_ptr + pid * stride_xm + cols * stride_xn, mask=mask, other=0.0)
        # Reduce to scalars
        s += tl.sum(x, axis=0)
        ss += tl.sum(x * x, axis=0)
    # Write per-row sums
    tl.store(SUM_ptr + pid, s)
    tl.store(SUMSQ_ptr + pid, ss)


# Kernel 2: normalize using mean/var and apply affine (weight, bias).
@triton.jit
def _layernorm_norm_affine_kernel(
    X_ptr,          # *fp32, input [M, N]
    Weight_ptr,     # *fp32, weight [N]
    Bias_ptr,       # *fp32, bias [N]
    Y_ptr,          # *fp32, output [M, N]
    M, N,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    eps,            # float32
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # program id over rows
    offs = tl.arange(0, BLOCK_N)
    mean = tl.load(SUM_ptr + pid) / N
    var = tl.load(SUMSQ_ptr + pid) / N - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    # Write normalized + affine
    for col_start in range(0, N, BLOCK_N):
        cols = col_start + offs
        mask = cols < N
        x = tl.load(X_ptr + pid * stride_xm + cols * stride_xn, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        w = tl.load(Weight_ptr + cols, mask=mask, other=1.0)
        b = tl.load(Bias_ptr + cols, mask=mask, other=0.0)
        y = norm * w + b
        tl.store(Y_ptr + pid * stride_ym + cols * stride_yn, y, mask=mask)


# Triton row-wise linear: computes C[M, N] = A[M, D] @ W^T[D, N] + bias[N]
@triton.jit
def _linear_rowwise_kernel(
    A_ptr,          # *fp32, input A [M, D], contiguous (row-major)
    W_ptr,          # *fp32, weight W [N, D], contiguous (row-major)
    B_ptr,          # *fp32, bias [N]
    C_ptr,          # *fp32, output C [M, N], contiguous (row-major)
    M, D, N,
    stride_am, stride_ad,
    stride_wn, stride_wd,   # strides for W (n, d)
    stride_cm, stride_cn,
    BLOCK_N: tl.constexpr,  # tile over N
    BLOCK_D: tl.constexpr,  # tile over D
):
    pid = tl.program_id(axis=0)  # program id over rows (M)
    offs_n = tl.arange(0, BLOCK_N)
    # Accumulator per output column
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    # Loop over D in tiles
    for d_start in range(0, D, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        # Load A row tile: A[pid, d_start:d_start+BLOCK_D]
        a = tl.load(A_ptr + pid * stride_am + offs_d * stride_ad, mask=offs_d < D, other=0.0)  # shape [BLOCK_D]
        # For each N tile
        for n_start in range(0, N, BLOCK_N):
            offs_n_tile = n_start + offs_n
            mask_n = offs_n_tile < N
            # Load W tile: W[offs_n_tile, offs_d]
            w = tl.load(W_ptr + offs_n_tile * stride_wn + offs_d * stride_wd, mask=mask_n[:, None] & (offs_d[None, :] < D), other=0.0)  # [BLOCK_N, BLOCK_D]
            # Accumulate: acc += sum_d (A[d] * W[n,d])
            acc += tl.sum(w * a[None, :], axis=1)  # reduce over BLOCK_D
        # After finishing all D tiles for this N tile, add bias
        b = tl.load(B_ptr + offs_n_tile, mask=mask_n, other=0.0)
        acc += b
    # Store results
    tl.store(C_ptr + pid * stride_cm + offs_n * stride_cn, acc, mask=offs_n < N)

        tl.store(Y_ptr + pid * stride_cm + cols * stride_cn, y, mask=mask)

            li = o + 2 - f  # pad=2
            # Bounds check for li
            if (li >= 0) and (li < L_in):
                x_val = tl.load(X_ptr + b * stride_xb + ci * stride_xci + li * stride_xli)
            else:
                x_val = 0.0
            w_val = tl.load(W_ptr + co * stride_wco + 0 * stride_wci + f * stride_wf)
            acc += x_val * w_val
            o += 1

    # Add bias and store to Y[b, ci, o, co]. However, Y is 3D [B, C_in, L_out], and Triton grid is limited.
    # We need to write acc into Y at position (b, ci, o). Since o is dynamic, we use while and write each o.
    # We'll do that by re-looping over o and writing acc at each o.

    # We'll store acc into Y[b, ci, o] for all o. Triton allows scalar store.
    # To do that, we need to write per o. Use while loop.
    for o in range(0, L_out):
        y_ptr = Y_ptr + b * stride_yb + ci * stride_yci + o * stride_yl + co  # here we need co index in Y's co dimension
        # Note: In Triton, we cannot index with co directly here; Y_ptr is base. We need to create per-co outputs.
        # To handle this, we'll run per co as separate program_id(2). But Triton grid axes are limited. As a workaround,
        # we keep co as pid2 in a separate kernel launch, but here we have only axis 0,1. So we'll iterate co inside.
        # Since Triton doesn't allow multiple axes in while, we'll restructure: run grid=(B*C_in,) and pass C_out via tl.constexpr
        # But Triton doesn't support passing C_out into kernel dynamically that way. Given constraints, we implement
        # a simpler version: handle one co and one b,ci per program. We'll do that by launching with grid=(B*C_in,)
        # and inside we loop co and o. Triton supports while loops.

    # However, Triton kernels are 1D/2D grid; we cannot have 3D grid. So we'll use grid=(B*C_in,) and inside
    # we will iterate over co. But then we can't write per co independently. To get around, we'll use a separate
    # grid axis by defining a 2D grid manually: launch with grid=(B*C_in, 1), and pass co via tl.constexpr? Not possible.
    # Therefore, we will implement a 2D grid where axis 0 = (B*C_in), axis 1 = co. Triton allows this; we set grid=(B*C_in, C_out).
    # We'll compute co from pid1 via tl.program_id(1). Then we loop over o inside the kernel.

    # For clarity, we'll re-implement conv kernel with 2D grid:
    # pid0 = b*ci, pid1 = co
    co = pid_co
    # acc already computed above, but computed per co. We'll recompute acc per co, since we need to store per co.
    # To avoid recomputation, we can maintain acc as array? Triton doesn't support dynamic arrays. So we recompute.

    # Recompute acc per co
    acc = tl.zeros((), dtype=tl.float32)
    for f in range(0, F):
        o = 0
        while o < L_out:
            li = o + 2 - f
            if (li >= 0) and (li < L_in):
                x_val = tl.load(X_ptr + b * stride_xb + ci * stride_xci + li * stride_xli)
            else:
                x_val = 0.0
            w_val = tl.load(W_ptr + co * stride_wco + 0 * stride_wci + f * stride_wf)
            acc += x_val * w_val
            o += 1

    # Add bias
    b_bias = tl.load(BIAS_ptr + co)
    acc += b_bias

    # Store output for all o
    # We cannot write per o directly with scalar store unless we loop. Triton supports while. We'll write per o.
    o = 0
    while o < L_out:
        y_val = acc  # since same acc applies to all taps and bias; or more precisely, acc is the sum for this co at all o.
        # But we need per-o contribution. The above acc is not per o. We need to recompute contribution for each o:
        # Contribution for each o is the sum over f of x[b, ci, li] * W[co, 0, f], where li=o+2-f. Then y = contribution + b_bias.
        # So we'll recompute per o contributions.

        # Recompute contribution for this o
        contribution = tl.zeros((), dtype=tl.float32)
        for f in range(0, F):
            li = o + 2 - f
            if (li >= 0) and (li < L_in):
                x_val = tl.load(X_ptr + b * stride_xb + ci * stride_xci + li * stride_xli)
            else:
                x_val = 0.0
            w_val = tl.load(W_ptr + co * stride_wco + 0 * stride_wci + f * stride_wf)
            contribution += x_val * w_val
        y_val = contribution + b_bias
        tl.store(Y_ptr + b * stride_yb + ci * stride_yci + o * stride_yl, y_val)
        o += 1


# Triton row-wise linear: computes C[M, N] = A[M, D] @ W^T[D, N] + bias[N]
@triton.jit
def _linear_rowwise_kernel(
    A_ptr,          # *fp32, input A [M, D], contiguous (row-major)
    W_ptr,          # *fp32, weight W [N, D], contiguous (row-major)
    B_ptr,          # *fp32, bias [N]
    C_ptr,          # *fp32, output C [M, N], contiguous (row-major)
    M, D, N,
    stride_am, stride_ad,
    stride_wn, stride_wd,   # strides for W (n, d)
    stride_cm, stride_cn,
    BLOCK_N: tl.constexpr,  # tile over N
    BLOCK_D: tl.constexpr,  # tile over D
):
    pid = tl.program_id(axis=0)  # program id over rows (M)
    offs_n = tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    for d_start in range(0, D, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        a = tl.load(A_ptr + pid * stride_am + offs_d * stride_ad, mask=offs_d < D, other=0.0)  # [BLOCK_D]
