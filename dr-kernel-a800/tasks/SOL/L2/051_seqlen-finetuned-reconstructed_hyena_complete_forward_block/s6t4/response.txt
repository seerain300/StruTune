import math
import triton
import triton.language as tl


# Triton LayerNorm forward kernel: normalize across feature dim D for each (b, l)
@triton.jit
def layernorm_forward_kernel(
    X_ptr,        # *const float
    W_ptr,        # *const float (gamma), shape (D,)
    B_ptr,        # *const float (beta), shape (D,)
    Y_ptr,        # *float
    B: tl.constexpr,      # batch size
    L: tl.constexpr,      # sequence length
    D: tl.constexpr,      # feature size
    eps,                 # float
    stride_xb, stride_xl, stride_xd,
    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    if b >= B or l >= L:
        return
    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        x = tl.where(mask, x, 0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        gamma = tl.load(W_ptr + offs * stride_w, mask=mask, other=1.0)
        beta = tl.load(B_ptr + offs * stride_b, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * gamma + beta
        tl.store(Y_ptr + b * stride_yb + l * stride_yl + offs * stride_yd, y, mask=mask)
        d0 += BLOCK_SIZE


# Triton conv1d with groups=inner_width, input: U_pad of shape (B, C_in, L), weight: (C_in, 1, K)
# We implement exact K=3 and padding=2; groups=C_in means each channel convolutes independently.
@triton.jit
def conv1d_groups_exact_kernel(
    Up_ptr,  # *const float, padded input (B, C_in, L)
    W_ptr,   # *const float, weights (C_in, 1, 3)
    Bo_ptr,  # *const float, bias per channel (C_in,)
    Uout_ptr,# *float, output (B, C_in, L_out)
    B: tl.constexpr, C_in: tl.constexpr, L: tl.constexpr, L_out: tl.constexpr, K: tl.constexpr, pad: tl.constexpr,
    stride_upb, stride_upc, stride_upl,
    stride_wcr, stride_wck,  # W is (C_in, 1, K), so stride_wck corresponds to K dimension
    stride_boc,
    stride_uob, stride_uoc, stride_uol,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    l_out = tl.program_id(2)
    if b >= B or c >= C_in or l_out >= L_out:
        return
    acc = 0.0
    for k in range(K):  # K=3
        inp_pos = l_out - pad + k
        valid = (inp_pos >= 0) & (inp_pos < L)
        x_val = tl.load(Up_ptr + b * stride_upb + c * stride_upc + inp_pos * stride_upl, mask=valid, other=0.0)
        w_val = tl.load(W_ptr + c * stride_wcr + 0 * stride_wck + k * stride_wck)
        acc += x_val * w_val
    # Add bias
    bval = tl.load(Bo_ptr + c * stride_boc)
    tl.store(Uout_ptr + b * stride_uob + c * stride_uoc + l_out * stride_uol, acc)


# Triton elementwise exp modulation: V_in: (B, D, L_filter), Deltas: (B, 1, D), Shift: scalar
@triton.jit
def exp_mod_kernel(
    V_ptr,          # *const float, input tensor (B, D, L)
    Deltas_ptr,     # *const float, deltas tensor (B, 1, D) with broadcasting
    Shift,          # float
    V_out_ptr,      # *float, output tensor (B, D, L)
    B: tl.constexpr, D: tl.constexpr, L: tl.constexpr,
    stride_vb, stride_vd, stride_vl,
    stride_dlb, stride_dld, stride_dll,  # Deltas strides
    stride_ob, stride_od, stride_ol,
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    l = tl.program_id(2)
    if b >= B or d >= D or l >= L:
        return
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    # load deltas[b, 0, d]
    dval = tl.load(Deltas_ptr + b * stride_dlb + 0 * stride_dld + d * stride_dld)
    e = tl.exp(-dval * tl.abs(dval))  # note: dval here is scalar float; abs(dval) = dval (since dvals are non-negative in the provided code)
    mod = e + Shift
    v_out = v * mod
    tl.store(V_out_ptr + b * stride_ob + d * stride_od + l * stride_ol, v_out)


# Triton GEMM for linear projection: A: (M, D) = (B*L, D), W: (D, D), B: (D,), C: (B, D, L)
@triton.jit
def linear_gemm_kernel(
    A_ptr,      # *const float, input matrix A[M, D]
    W_ptr,      # *const float, weight matrix W[D, D]
    B_ptr,      # *const float, bias (D,)
    C_ptr,      # *float, output C[B, D, L]
    M: tl.constexpr, D: tl.constexpr, L: tl.constexpr,
    stride_am, stride_ad,
    stride_wd, stride_wk,
    stride_cb, stride_cd, stride_cl,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    l = tl.program_id(2)
    if b >= B or d >= D or l >= L:
        return
    # For C[b, d, l], compute dot over D: y = sum_j A[b*L + j, d] * W[d, j] + B[d]
    # We loop over D in blocks and accumulate.
    acc = 0.0
    d0 = 0
    while d0 < D:
        offs_d = d0 + tl.arange(0, BLOCK_N)
        # For each l, A rows correspond to b*L + offs_l
        # However, we compute scalar dot per (b, d) across all l entries; better restructure:
        # Since we need C per (b, d, l), we can compute A_row = A[b*L + l, :] and then dot with W[:, :].
        # But Triton kernel needs a grid; here we choose grid=(B, D, L) and compute per (b,d,l) scalar y.
        # To do so, we loop over D in blocks and accumulate:
        # For each offs_d, load A_row[b*L + l, offs_d], and W[offs_d, k], then sum outer product.
        # Implement a nested loop over k in blocks:
        k0 = 0
        while k0 < D:
            offs_k = k0 + tl.arange(0, BLOCK_K)
            # Load A_row[b*L + l, offs_d] and W[offs_d, offs_k]
            # A_row index: a_idx = (b * L + l) * stride_am + offs_d * stride_ad
            a_idx = (b * L + l) * stride_am + offs_d * stride_ad
            a_vec = tl.load(A_ptr + a_idx, mask=offs_d < D, other=0.0)  # vector of size BLOCK_N
            w_mat = tl.load(W_ptr + offs_d[:, None] * stride_wd + offs_k[None, :] * stride_wk,
                            mask=(offs_d[:, None] < D) & (offs_k[None, :] < D),
                            other=0.0)  # matrix (BLOCK_N, BLOCK_K)
            # Accumulate outer product: acc += sum_k (a_vec[:, None] * w_mat[:, k])
            # Triton supports tl.sum over axis; we can reduce over k:
            for kk in range(BLOCK_K):
                kcol = k0 + kk
                if kcol < D:
                    w_col = tl.load(W_ptr + offs_d * stride_wd + kcol * stride_wk,
                                    mask=offs_d < D, other=0.0)  # vector (BLOCK_N,)
                    acc += tl.sum(a_vec * w_col, axis=0)
            k0 += BLOCK_K
        d0 += BLOCK_N
    # Add bias
    bias = tl.load(B_ptr + d * stride_cb)
    acc += bias
    tl.store(C_ptr + b * stride_cb + d * stride_cd + l * stride_cl, acc)

# Kernel: fill a 1D tensor with ones
@triton.jit
def fill_ones_1d_kernel(X_ptr, N, stride_xn):
    pid = tl.program_id(0)

    tl.store(X_ptr + pid * stride_xn, 1.0)

# Kernel: randn fill 1D tensor using tl.rand
@triton.jit
def randn_fill_1d_kernel(X_ptr, N, stride_xn, mean, std):
    pid = tl.program_id(0)
    if pid >= N:
        return
    r = tl.rand()  # uniform in [0,1)
    val = mean + std * (r - 0.5) * 4.0  # approx N(mean, std)
    tl.store(X_ptr + pid * stride_xn, val)

import math
import triton
import triton.language as tl


# Triton LayerNorm forward kernel: normalize across feature dim D for each (b, l)
@triton.jit
def layernorm_forward_kernel(
    X_ptr,        # *const float
    W_ptr,        # *const float (gamma), shape (D,)
    B_ptr,        # *const float (beta), shape (D,)
    Y_ptr,        # *float
    B: tl.constexpr,      # batch size
    L: tl.constexpr,      # sequence length
    D: tl.constexpr,      # feature size
    eps,                 # float
    stride_xb, stride_xl, stride_xd,
    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    if b >= B or l >= L:
        return
    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        x = tl.where(mask, x, 0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        gamma = tl.load(W_ptr + offs * stride_w, mask=mask, other=1.0)
        beta = tl.load(B_ptr + offs * stride_b, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * gamma + beta
        tl.store(Y_ptr + b * stride_yb + l * stride_yl + offs * stride_yd, y, mask=mask)
        d0 += BLOCK_SIZE


# Triton conv1d with groups=inner_width, input: U_pad of shape (B, C_in, L), weight: (C_in, 1, K)
