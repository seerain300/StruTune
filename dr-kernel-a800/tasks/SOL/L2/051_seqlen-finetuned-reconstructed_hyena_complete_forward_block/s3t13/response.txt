import math
import torch
import triton
import triton.language as tl


@triton.jit
def layernorm_3d_forward_affine(
    X, Y, W, BIAS, EPS,
    B: tl.constexpr, L: tl.constexpr, D: tl.constexpr,
    BLOCK_D: tl.constexpr,
    stride_x_b, stride_x_l, stride_x_d,
    stride_y_b, stride_y_l, stride_y_d,
    stride_w, stride_bias
):
    """
    LayerNorm over last dimension for 3D tensor [B, L, D], with affine weight and bias.
    Launch as grid = (B, L). Each program handles one (b, l) row across D, looping over D in tiles.
    """
    b = tl.program_id(0)
    l = tl.program_id(1)

    # Base pointers for this (b, l) row
    x_row = X + b * stride_x_b + l * stride_x_l
    y_row = Y + b * stride_y_b + l * stride_y_l

    # Accumulate sum and sum of squares across D
    sum_x = 0.0
    sum_x2 = 0.0
    for d0 in range(0, D, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < D
        x = tl.load(x_row + d * stride_x_d, mask=mask, other=0.0)
        # reduce tile to scalar
        sum_x += tl.sum(x, axis=0)
        sum_x2 += tl.sum(x * x, axis=0)

    D_f = tl.float32(D)
    mean = sum_x / D_f
    var = sum_x2 / D_f - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: normalize and apply affine
    for d0 in range(0, D, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask = d < D
        x = tl.load(x_row + d * stride_x_d, mask=mask, other=0.0)
        w = tl.load(W + d * stride_w, mask=mask, other=1.0)
        bias = tl.load(BIAS + d * stride_bias, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * w + bias
        tl.store(y_row + d * stride_y_d, y, mask=mask)
