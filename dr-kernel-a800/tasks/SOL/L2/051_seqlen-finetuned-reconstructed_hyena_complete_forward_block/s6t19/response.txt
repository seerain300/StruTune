import math
import triton
import triton.language as tl


# Triton randn_fill kernel: fill a float32 tensor with random normal (approximation).
@triton.jit
def randn_fill_kernel(Out_ptr, size, seed, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < size
    # Approximate normal: r = N(0,1) via Ziggurat or use a fast method; here we use a simple transformation.
    # Generate uniform u in [0,1) and compute r = u * sqrt(12) - sqrt(3) which has mean 0 and variance 1.
    # Note: This is an approximation; for higher accuracy, a more sophisticated generator can be used.
    u = tl.rand(seed, offsets)
    r = (u * 2.449489742783178) - 1.7320508075688772
    tl.store(Out_ptr + offsets, r, mask=mask)


# Triton fill_ones kernel: fill a 2D tensor (B, D) with ones (float32).
@triton.jit
def fill_ones_kernel(Out_ptr, B, D, stride_yb, stride_yd, BLOCK_SIZE: tl.constexpr):
    b = tl.program_id(0)
    d = tl.program_id(1)
    if b >= B or d >= D:
        return
    idx = b * stride_yb + d * stride_yd
    tl.store(Out_ptr + idx, 1.0)


# Triton LayerNorm forward for 3D tensors (B, L, D): normalize over last dim, affine
# We implement: mean = sum / D, var = sum_sq / D - mean^2, y = ((x - mean) / sqrt(var + eps)) * gamma + beta
@triton.jit
def layernorm_forward_kernel(
    X_ptr,        # *const float
    W_ptr,        # *const float (gamma), shape [D]
    B_ptr,        # *const float (beta), shape [D]
    Y_ptr,        # *float
    B, L, D,      # int
    eps,          # float
    stride_xb, stride_xl, stride_xd,
    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    if b >= B or l >= L:
        return

    # Compute mean and variance across D for (b, l)
    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        # Reduce
        # Note: tl.sum works on vectors; we accumulate scalar sum
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Normalize and affine
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        gamma = tl.load(W_ptr + offs * stride_w, mask=mask, other=1.0)
        beta = tl.load(B_ptr + offs * stride_b, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * gamma + beta
        tl.store(Y_ptr + b * stride_yb + l * stride_yl + offs * stride_yd, y, mask=mask)
        d0 += BLOCK_SIZE


# Triton GEMM for linear: A[M, K] @ W[K, N] -> C[M, N] + bias, we implement a simple 1D GEMM.
# We use tiles (BLOCK_M, BLOCK_K, BLOCK_N) to cover M, K, N. This is a common pattern for Triton GEMM.
@triton.jit
def linear_gemm_kernel(
    A_ptr,        # *const float, shape [M, K] flattened (row-major)
    W_ptr,        # *const float, shape [K, N] flattened (row-major)
    B_ptr,        # *const float, bias [N]
    C_ptr,        # *float, output [M, N] flattened
    M, K, N,
    stride_am, stride_ak,
    stride_wk, stride_wn,
    stride_cm, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    n = tl.program_id(1)
    if m >= M or n >= N:
        return
    acc = 0.0
    # Loop over K in tiles
    k0 = 0
    while k0 < K:
        k_tile = k0 + tl.arange(0, BLOCK_K)
        n0 = 0
        while n0 < N:
            n_tile = n0 + tl.arange(0, BLOCK_N)
            # Accumulate partial matmul over a BK x BN tile
            # For each k in BK, load A row m for that k, and W k row for n_tile, then sum over k
            acc_sub = 0.0
            # Inner loop over K tile
            for kk in range(BLOCK_K):
                k_idx = k_tile[kk]
                mask_k = k_idx < K
                # A[m, k_idx] scalar
                a = tl.load(A_ptr + m * stride_am + k_idx * stride_ak, mask=mask_k, other=0.0)
                # W[k_idx, n_tile] vector
                w = tl.load(W_ptr + k_idx * stride_wk + n_tile * stride_wn, mask=(n_tile < N), other=0.0)
                acc_sub += a * w
            # Add bias and store
            b = tl.load(B_ptr + n_tile * stride_wn, mask=(n_tile < N), other=0.0)
            acc += acc_sub
            n0 += BLOCK_N
        # Store result to C[m, n]
        tl.store(C_ptr + m * stride_cm + n * stride_cn, acc)
        k0 += BLOCK_K


# Triton short depthwise conv with groups and padding=2, kernel length=3 (matches short_conv_weight shape).
# Up: [B, D, L_in], Wc: [GROUPS, Kc, 1], Bc: [GROUPS], Out v: [B, D, L_out] where L_out = L_in - 2.
@triton.jit
def conv1d_groups_exact_kernel(
    Up_ptr,       # *const float, shape [B, D, L_in]
    Wc_ptr,       # *const float, shape [GROUPS, 3, 1]
    Bc_ptr,       # *const float, shape [GROUPS]
    Out_ptr,      # *float, shape [B, D, L_out]
    B, D, L_in,   # int
    GROUPS,       # int = inner_width
    Kc,           # int = 3
    L_out,        # int = L_in - 2
    stride_upb, stride_upd, stride_upl,
    stride_wcg, stride_wck, stride_wcp,   # Wc strides
    stride_outb, stride_outd, stride_outl,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    l_out = tl.program_id(2)
    if (b >= B) or (d >= D) or (l_out >= L_out):
        return

    # Each (b, d) accumulates over GROUPS channels (here GROUPS == D, but we pass inner_width)
    acc = 0.0
    # For each group channel
    for g in range(GROUPS):
        # Accumulate over kernel taps k=0,1,2
        # Output position l_out corresponds to input position t = l_out - 2 + k
        # We must ensure t in [0, L_in-1]
        # Bias add once per group per output position
        bias = tl.load(Bc_ptr + g)
        acc += bias
        for k in range(Kc):
            t = l_out - 2 + k
            valid = (t >= 0) & (t < L_in)
            # Up[b, d, t] may be out-of-range if invalid; masked load handles this
            val = tl.load(Up_ptr + b * stride_upb + d * stride_upd + t * stride_upl, mask=valid, other=0.0)
            # Wc[g, k, 0]
            w = tl.load(Wc_ptr + g * stride_wcg + k * stride_wck + 0 * stride_wcp)
            acc += val * w
    tl.store(Out_ptr + b * stride_outb + d * stride_outd + l_out * stride_outl, acc)


# Triton exp_mod kernel: V_in[B, D, L] -> V_out[B, D, L]
# v_new = v * (exp(-t * abs(deltas)) + shift)
# deltas has shape [D], broadcasts across B and L.
@triton.jit
def exp_mod_kernel(
    V_ptr,        # *const float, shape [B, D, L]
    Deltas_ptr,   # *const float, shape [D]
    B, D, L,
    shift,
    stride_vb, stride_vd, stride_vl,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    l = tl.program_id(2)
    if (b >= B) or (d >= D) or (l >= L):
        return
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    delta = tl.load(Deltas_ptr + d)
    t = l
    exp_term = tl.exp(-t * tl.abs(delta))
    v_new = v * (exp_term + shift)
    tl.store(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl, v_new)
