import torch
import triton
import triton.language as tl


@triton.jit
def layernorm_3d_kernel(
    X_ptr, Y_ptr, W_ptr, B_ptr,
    B, S, D,
    eps,
    BLOCK_SIZE: tl.constexpr
):
    """
    LayerNorm over the last dimension D for each (b, s) row.
    Input: X[B, S, D], weight[D], bias[D]
    Output: Y[B, S, D]
    """
    b = tl.program_id(axis=0)
    s = tl.program_id(axis=1)

    # Base offset for the (b, s) row
    row_offset = b * S * D + s * D

    # Accumulate sum and sum of squares across D in fp32
    total_sum = 0.0
    total_sumsq = 0.0

    # First pass: compute mean and variance
    for start in range(0, D, BLOCK_SIZE):
        cols = start + tl.arange(0, BLOCK_SIZE)
        mask = cols < D
        x = tl.load(X_ptr + row_offset + cols, mask=mask, other=0.0)
        x = x.to(tl.float32)
        total_sum += tl.sum(x, axis=0)
        total_sumsq += tl.sum(x * x, axis=0)

    mean = total_sum / D
    var = total_sumsq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for start in range(0, D, BLOCK_SIZE):
        cols = start + tl.arange(0, BLOCK_SIZE)
        mask = cols < D
        x = tl.load(X_ptr + row_offset + cols, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(W_ptr + cols, mask=mask, other=1.0).to(tl.float32)
        bval = tl.load(B_ptr + cols, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + bval
        tl.store(Y_ptr + row_offset + cols, y, mask=mask)

LayerNorm Triton kernel の詳細
- 入力テンソル X が (B, S, D) として扱われ、各 (b, s) 行に対して最後の次元 D を正規化する。
- まず第一通過で sum(x) と sum(x^2) を計算し、mean と var (population variance) を求めます。mean = sum / D, var = E[x^2] - mean^2.
- 逆標準偏差 inv_std = 1 / sqrt(var + eps) を計算。
- 第二通過で正規化 y = (x - mean) * inv_std * weight + bias と書き込む。
- 二重ループではなく、一回のループで D を BLOCK_SIZE で分けたタイルで取り扱うことで、十分大きな D でもしっかり動きます。各タイルで sum と sum of squares を加算し、最後に mean と inv_std を得て、再び各タイルで正規化・書き込みを行う。
