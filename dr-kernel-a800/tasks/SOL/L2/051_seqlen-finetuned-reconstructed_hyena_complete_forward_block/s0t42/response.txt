def layernorm_3d_kernel(
    x_ptr,                # *const float, input: [B, S, D]
    y_ptr,                # *float, output: [B, S, D]
    weight_ptr,           # *const float, [D]
    bias_ptr,             # *const float, [D]
    B, S, D,              # int32
    EPS,                  # float32
    BLOCK_D: tl.constexpr # tile size along D
):
    # Each program handles one row (b, s)
    pid = tl.program_id(axis=0)  # 0..(B*S-1)
    s = pid % S
    b = pid // S
    base = (b * S + s) * D

    # Pass 1: compute mean and variance (unbiased=False)
    sum_val = 0.0
    sum_sq = 0.0
    off = 0
    while off < D:
        idx = off + tl.arange(0, BLOCK_D)
        mask = idx < D
        x = tl.load(x_ptr + base + idx, mask=mask, other=0.0)
        # reduce
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        off += BLOCK_D

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Pass 2: normalize and apply affine
    off = 0
    while off < D:
        idx = off + tl.arange(0, BLOCK_D)
        mask = idx < D
        x = tl.load(x_ptr + base + idx, mask=mask, other=0.0)
        w = tl.load(weight_ptr + idx, mask=mask, other=1.0)
        b = tl.load(bias_ptr + idx, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_ptr + base + idx, y, mask=mask)
        off += BLOCK_D
