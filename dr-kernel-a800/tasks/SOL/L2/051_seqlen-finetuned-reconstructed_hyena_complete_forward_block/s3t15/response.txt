```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def linear_3d_constK(
    X, W, BIAS, Y,
    B: tl.constexpr, L: tl.constexpr, D: tl.constexpr, K: tl.constexpr,
    BLOCK_D: tl.constexpr,
    stride_x_b, stride_x_l, stride_x_d,
    stride_w_o, stride_w_d,
    stride_y_b, stride_y_l, stride_y_d,
    stride_bias_o,
):
    """
    Compute Y[b, l, o] = sum_{d=0..D-1} X[b, l, d] * W[o, d] + BIAS[o]
    for grid = (B, L, K). Each program handles one output channel o for a given (b, l).
    """
    b = tl.program_id(0)
    l = tl.program_id(1)
    o = tl.program_id(2)

    # Accumulator in fp32
    acc = 0.0

    # Loop over D in tiles
    for d0 in range(0, D, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask_d = d < D
        # Load X[b, l, d]
        x_ptr = X + b * stride_x_b + l * stride_x_l + d * stride_x_d
        x = tl.load(x_ptr, mask=mask_d, other=0.0).to(tl.float32)
        # Load W[o, d]
        w_ptr = W + o * stride_w_o + d * stride_w_d
        w = tl.load(w_ptr, mask=mask_d, other=0.0).to(tl.float32)
        # Accumulate dot product
        acc += tl.sum(x * w, axis=0)

    # Add bias[o]
    bias_ptr = BIAS + o * stride_bias_o
    bias = tl.load(bias_ptr).to(tl.float32)
    acc += bias

    # Store to Y[b, l, o]
    y_ptr = Y + b * stride_y_b + l * stride_y_l + o * stride_y_d
    tl.store(y_ptr, acc)
```

    X, W, BIAS, Y,
    B: tl.constexpr, L: tl.constexpr, D: tl.constexpr, K: tl.constexpr,
    BLOCK_D: tl.constexpr,

import math
import torch
import triton
import triton.language as tl


@triton.jit
def linear_3d_constK(
    X, W, BIAS, Y,
    B: tl.constexpr, L: tl.constexpr, D: tl.constexpr, K: tl.constexpr,
    BLOCK_D: tl.constexpr,
    stride_x_b, stride_x_l, stride_x_d,
    stride_w_o, stride_w_d,
    stride_y_b, stride_y_l, stride_y_d,
    stride_bias_o,
):
    """
    Compute Y[b, l, o] = sum_{d=0..D-1} X[b, l, d] * W[o, d] + BIAS[o]
    for grid = (B, L, K). Each program handles one output channel o for a given (b, l).
    """
    b = tl.program_id(0)
    l = tl.program_id(1)
    o = tl.program_id(2)

    acc = 0.0

    # Loop over D in tiles
    for d0 in range(0, D, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        mask_d = d < D
        # Load X[b, l, d]
        x_ptr = X + b * stride_x_b + l * stride_x_l + d * stride_x_d
        x = tl.load(x_ptr, mask=mask_d, other=0.0).to(tl.float32)
        # Load W[o, d]
        w_ptr = W + o * stride_w_o + d * stride_w_d
        w = tl.load(w_ptr, mask=mask_d, other=0.0).to(tl.float32)
        # Accumulate dot product across the tile
        acc += tl.sum(x * w, axis=0)

    # Add bias[o]
    bias_ptr = BIAS + o * stride_bias_o
    bias = tl.load(bias_ptr).to(tl.float32)
    acc += bias

    # Store to Y[b, l, o]
    y_ptr = Y + b * stride_y_b + l * stride_y_l + o * stride_y_d
    tl.store(y_ptr, acc)
