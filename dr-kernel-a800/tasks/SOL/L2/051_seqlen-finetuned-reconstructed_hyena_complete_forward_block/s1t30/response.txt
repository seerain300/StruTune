import torch
import triton
import triton.language as tl


@triton.jit
def create_2d_buffer_kernel(
    out_ptr,
    N, D,
    out_stride0, out_stride1,  # for linear indexing: out_stride0 == D, out_stride1 == 1
    BLOCK: tl.constexpr
):
    # Initialize a 2D buffer (N, D) via 1D linearized pointer.
    total = N * D
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    # Linear indices: idx = i * D + d
    # out_ptr strides: row stride = D, col stride = 1 (passed as out_stride0, out_stride1)
    # Since we're linearizing, we use 1D store: out_ptr[offs]
    tl.store(out_ptr + offs, 0.0, mask=mask)


@triton.jit
def linspace_1d_kernel(
    out_ptr,  # 1D output pointer
    start, end, length,  # length: number of elements
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < length
    step = (end - start) / (length - 1) if length > 1 else 0.0
    val = start + offs * step
    tl.store(out_ptr + offs, val, mask=mask)


@triton.jit
def ones_1d_kernel(
    out_ptr,
    length,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < length
    tl.store(out_ptr + offs, 1.0, mask=mask)


@triton.jit
def gate_forward_kernel(
    v_in_ptr, gate_ptr, out_ptr,
    N, D,
    v_stride0, v_stride1,  # for elementwise linear indexing: v_stride0 == D, v_stride1 == 1
    out_stride0, out_stride1,  # same for output
    BLOCK: tl.constexpr
):
    total = N * D
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    v = tl.load(v_in_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=1.0)
    out = v * g
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def exp_mod_apply_kernel(
    v_ptr, t_ptr, deltas_ptr, out_ptr,
    N, D,
    v_stride0, v_stride1,
    out_stride0, out_stride1,
    BLOCK: tl.constexpr
):
    total = N * D
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # Map linear index to (i, d): i = offs // D, d = offs % D
    i = offs // D
    d = offs - i * D  # d = offs % D

    v = tl.load(v_ptr + offs, mask=mask, other=0.0)
    t = tl.load(t_ptr + i, mask=mask, other=0.0)
    delta = tl.load(deltas_ptr + d, mask=mask, other=0.0)
    shift = 0.05
    mod = tl.exp(-t * delta) + shift
    out = v * mod
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def add_residual_kernel(
    v_ptr, residual_ptr, out_ptr,
    N, D,
    v_stride0, v_stride1,
    out_stride0, out_stride1,
    BLOCK: tl.constexpr
):
    total = N * D
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    v = tl.load(v_ptr + offs, mask=mask, other=0.0)

    out = v + res
    tl.store(out_ptr + offs, out, mask=mask)

        grid_d = (triton.cdiv(D, BLOCK),)
        linspace_1d_kernel[grid_d](deltas, 0.0, (D - 1), D, BLOCK=BLOCK)

            N, D,

            N, D,
            out.stride(0), out.stride(1),

def gate_forward_kernel(
    v_in_ptr, gate_ptr, out_ptr,
    N, D,  # D is 256
    v_stride0, v_stride1,
    out_stride0, out_stride1,
    BLOCK: tl.constexpr
):
    total = N * D
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    g = tl.load(gate_ptr + offs, mask=mask, other=1.0)
    out = v * g
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def exp_mod_apply_kernel(
    v_ptr, t_ptr, deltas_ptr, out_ptr,
    N, D,  # D is 256
    v_stride0, v_stride1,
    out_stride0, out_stride1,
    BLOCK: tl.constexpr
):
    total = N * D
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    # Map linear index to (i, d): i = offs // D, d = offs % D
    i = offs // D
    d = offs - i * D  # equivalent to offs % D
    v = tl.load(v_ptr + offs, mask=mask, other=0.0)
    t = tl.load(t_ptr + i, mask=mask, other=0.0)           # vector t of length N
    delta = tl.load(deltas_ptr + d, mask=mask, other=0.0)  # vector deltas of length D
    shift = 0.05
    mod = tl.exp(-t * delta) + shift
    out = v * mod
    tl.store(out_ptr + offs, out, mask=mask)

    mask = offs < total
    v = tl.load(v_ptr + offs, mask=mask, other=0.0)
    res = tl.load(residual_ptr + offs, mask=mask, other=0.0)
    out = v + res
    tl.store(out_ptr + offs, out, mask=mask)
