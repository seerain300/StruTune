def layernorm_fwd_kernel(
    x_ptr,          # *f32, input pointer
    w_ptr,          # *f32, gamma (weight), length D
    b_ptr,          # *f32, beta  (bias),   length D
    out_ptr,        # *f32, output pointer
    M,              # int32, number of rows
    D,              # int32, feature dimension (last dim)
    eps,            # float32, epsilon for variance
    BLOCK_SIZE: tl.constexpr,  # tile size along D (we use D=256 here)
):

    # If row_id >= M, do nothing (guard)
    if row_id >= M:
        return

    # First pass: compute sum and sum of squares across D
    sum_val = tl.zeros((), dtype=tl.float32)
    sum_sq = tl.zeros((), dtype=tl.float32)

    # Since D is passed, we can do a single vectorized load over [0:D].

    x2 = x * x
