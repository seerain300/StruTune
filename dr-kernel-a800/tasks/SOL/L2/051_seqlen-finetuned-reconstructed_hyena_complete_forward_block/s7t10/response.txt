# Triton LayerNorm forward kernel:
# Operates on a 2D tensor [M, D] laid out linearly as [M*D].
# Each program handles one row (pid = row index). It computes mean and variance across D,
# normalizes, and applies affine (gamma, beta).
@triton.jit
def layernorm_fwd_kernel(
    x_ptr,         # *f32, input pointer to [M*D] flattened row-major
    w_ptr,         # *f32, gamma [D]
    b_ptr,         # *f32, beta  [D]
    out_ptr,       # *f32, output pointer to [M*D] flattened
    M,             # int32, number of rows
    D,             # int32, number of features per row
    eps,           # f32, epsilon for variance
    BLOCK: tl.constexpr,  # tile size over D
):
    pid = tl.program_id(axis=0)
    # Each program handles one row
    # base offset in the flattened array for this row
    base = pid * D
    # Accumulate sum and sum of squares across D
    sum_val = 0.0
    sum_sq = 0.0
    # First pass: compute mean and variance
    for offs in range(0, D, BLOCK):
        cols = offs + tl.arange(0, BLOCK)
        mask = cols < D
        x = tl.load(x_ptr + base + cols, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for offs in range(0, D, BLOCK):
        cols = offs + tl.arange(0, BLOCK)
        mask = cols < D
        x = tl.load(x_ptr + base + cols, mask=mask, other=0.0)
        gamma = tl.load(w_ptr + cols, mask=mask, other=1.0)
        beta = tl.load(b_ptr + cols, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * gamma + beta
        tl.store(out_ptr + base + cols, y, mask=mask)

        x = tl.load(x_ptr + base + cols, mask=mask, other=0.0)
        gamma = tl.load(w_ptr + cols, mask=mask, other=1.0)
        beta = tl.load(b_ptr + cols, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * gamma + beta
        tl.store(out_ptr + base + cols, y, mask=mask)


# Triton GEMV-like kernel: u = x @ W^T + bias, where x is [D], W is [K, D], bias [K], u is [K].
@triton.jit
def linear_gemv_kernel(
    x_ptr,         # *f32, input vector [D]
    w_ptr,         # *f32, weight [K, D] flattened
    b_ptr,         # *f32, bias [K]
    out_ptr,       # *f32, output [K]
    D,             # int32, features of x
    K,             # int32, output length
    BLOCK_X: tl.constexpr,  # tile over D
):
    k = tl.program_id(axis=0)  # one program per output feature
    # Accumulator
    acc = 0.0
    for d_start in range(0, D, BLOCK_X):
        d_idx = d_start + tl.arange(0, BLOCK_X)
        mask = d_idx < D
        x = tl.load(x_ptr + d_idx, mask=mask, other=0.0)  # [BLOCK_X]
        # Load corresponding row from W: w[k, d] across d tiles
        # W is [K, D] flattened; index = k*D + d
        w = tl.load(w_ptr + k * D + d_idx, mask=mask, other=0.0)  # [BLOCK_X]
        # Multiply-accumulate: sum(x * w) over BLOCK_X
        acc += tl.sum(x * w, axis=0)
    # Add bias
    bias = tl.load(b_ptr + k)
    acc = acc + bias
    tl.store(out_ptr + k, acc)


# Triton grouped depthwise conv1d with K=1: v[b, c] = sum_{i in pad} sum_{k} u[b, c_in_k + i] * w[c, 0, i]
# Here u is [B, S, D] flattened per (b, s) row, w is [C, 1, 3] where C=K, 3 is filter length.
# Output v is [B, S, C], but we only need per (b, s) row across channels.
@triton.jit
def depthwise_conv1d_groups_kernel(
    u_ptr,         # *f32, input u flattened per (b, s) row, length = S*D
    w_ptr,         # *f32, weight [C, 1, 3] flattened, length = C*3
    out_ptr,       # *f32, output v flattened per (b, s) row, length = C
    S,             # int32, sequence length
    D,             # int32, features per position
    C,             # int32, number of channels (output channels) == K (in_proj_weight rows)
    PAD: tl.constexpr,  # padding on both sides
):
    row = tl.program_id(axis=0)  # one program per (b, s) row
    # Base offset for this row in flattened u: pos runs from 0..S-1, feature runs 0..D-1
    # We loop over channels c and compute sum over padded positions.
    for c in range(0, C):
        total = 0.0
        # Loop over positions i in [0, S-1], apply padding
        for i in range(0, S):
            pos = i + PAD
            # Each channel maps to its own set of input channels (groups). Here each c maps to c.
            # Compute input feature index for this position
            base_in = pos * D  # linear base for this position across features
            # Sum over 3 filter taps: i-2, i-1, i (only valid if pos >= 2,1,0 respectively)
            # Handle boundary by load other=0.0
            if pos >= 2:
                total += tl.load(u_ptr + base_in + c)  # u[b, pos, c]
            if pos >= 1:
                total += tl.load(u_ptr + base_in + c)
            # For pos==0, no contribution from taps with negative index, handled above
        # Store result for this channel at position row in out_ptr
        tl.store(out_ptr + row * C + c, total)

    out_ptr,       # *f32, output [D_inner]
