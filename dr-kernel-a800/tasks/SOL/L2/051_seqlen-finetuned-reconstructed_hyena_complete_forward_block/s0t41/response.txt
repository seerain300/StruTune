import torch
import triton
import triton.language as tl


@triton.jit
def triton_ones_like(out_ptr, in_ptr, size: tl.constexpr, BLOCK: tl.constexpr):
    # Create a tensor of ones with same shape as input; size is the number of elements
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < size
    # Initialize output with ones
    one = tl.full([BLOCK], 1.0, tl.float32)
    tl.store(out_ptr + offsets, one, mask=mask)


@triton.jit
def layernorm_2d_kernel(
    x_ptr,                # *fp32
    weight_ptr,           # *fp32 (size D)
    bias_ptr,             # *fp32 (size D)
    out_ptr,              # *fp32
    M,                    # number of rows (B * S)
    D,                    # feature size (d_model)
    EPS,                  # epsilon
    BLOCK_D: tl.constexpr
):
    # Each program handles one row
    row = tl.program_id(axis=0)
    # Bounds check: if row >= M, return
    if row >= M:
        return

    # Accumulate sum and sum of squares over D
    sum_val = 0.0
    sum_sq = 0.0
    for d in range(0, D, BLOCK_D):
        offs = d + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr + row * D + offs, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Normalize and apply affine
    for d in range(0, D, BLOCK_D):
        offs = d + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr + row * D + offs, mask=mask, other=0.0)
        x = x.to(tl.float32)
        w = tl.load(weight_ptr + offs, mask=mask, other=1.0)
        b = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(out_ptr + row * D + offs, y, mask=mask)

    splits = uc.split(d_model, dim=1)
    x = splits[:-1]
    v = splits[-1]
    
    # Implicit Filter Generation - Positional embeddings
    t = torch.linspace(0, 1, l_filter, device=device)[None, :, None]
    bands = 2
    t_rescaled = torch.linspace(0, l_filter - 1, l_filter, device=device)[None, :, None]
    w = 2 * math.pi * t_rescaled / l_filter

    z = torch.cat([t, torch.cos(-f * w), torch.sin(-f * w)], dim=-1)

    h = F.linear(z, filter_linear1_weight, filter_linear1_bias)
    h = torch.sin(sin_freq * h)
    h = F.linear(h, filter_linear2_weight, filter_linear2_bias)
    h = torch.sin(sin_freq * h)
    h = F.linear(h, filter_linear3_weight, filter_linear3_bias)
    h = torch.sin(sin_freq * h)
    h = F.linear(h, filter_linear_final_weight, None)

    decay = torch.exp(-t * exp_mod_deltas.abs())
    h = h * (decay + exp_mod_shift)

    h = h + filter_bias.view(1, 1, d_model)
    k = h.transpose(0, 1).reshape(1, d_model, l_filter)

    return output

    weight_ptr,       # *fp32, size D
    bias_ptr,         # *fp32, size D
    out_ptr,          # *fp32, output (M, D)
    M,                # int, number of rows
    D,                # int, feature size
    EPS,              # float32, epsilon
    BLOCK_D: tl.constexpr,
):
    # One program per row
    row = tl.program_id(axis=0)
    if row >= M:
        return

    sum_val = 0.0
    sum_sq = 0.0
    # First pass: compute sum and sum of squares
    for d in range(0, D, BLOCK_D):
        offs = d + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr + row * D + offs, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: normalize and apply affine
    for d in range(0, D, BLOCK_D):
        offs = d + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr + row * D + offs, mask=mask, other=0.0)
        x = x.to(tl.float32)
        w = tl.load(weight_ptr + offs, mask=mask, other=1.0)
        b = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(out_ptr + row * D + offs, y, mask=mask)


@triton.jit
def linear_matmul_bias_kernel(
    A_ptr,            # *fp32, (M, K)
    B_ptr,            # *fp32, (N, K)
    bias_ptr,         # *fp32, (N,)
    C_ptr,            # *fp32, (M, N)
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in tl.static_range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # A: (BLOCK_M, BLOCK_K) = A[offs_m, offs_k]
        A_tile = tl.load(
            A_ptr + (offs_m[:, None] * K + offs_k[None, :]),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # B: (BLOCK_N, BLOCK_K) = B[offs_n, offs_k]
        B_tile = tl.load(
            B_ptr + (offs_n[:, None] * K + offs_k[None, :]),
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0
        )
        # Multiply-accumulate: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(A_tile, B_tile.T)

    # Add bias: (BLOCK_N,)
    bias = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0)  # (BLOCK_N,)
    # Broadcast bias to (BLOCK_M, BLOCK_N)
    acc = acc + bias[None, :]

    # Store results
    tl.store(
        C_ptr + (offs_m[:, None] * N + offs_n[None, :]),
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )


@triton.jit
def gelu_tanh_kernel(X_ptr, Y_ptr, SIZE: tl.constexpr, BLOCK: tl.constexpr):
    # GELU tanh approximation: 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    p = 0.7978845608028654  # sqrt(2/pi)
    c = 0.044715
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < SIZE
    x = tl.load(X_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    x3 = x * x * x
    inner = x + c * x3
    t = tl.tanh(p * inner)
    y = 0.5 * x * (1.0 + t)
    tl.store(Y_ptr + offs, y, mask=mask)
