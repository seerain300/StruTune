# Kernel: fill a tensor with ones (float32)
@triton.jit
def fill_ones_kernel(X_ptr, N, stride_xn, stride_xm):
    pid = tl.program_id(0)
    if pid >= N:
        return
    # This kernel is meant to be launched over a 2D grid (N, 1). We use pid as row index.
    # Each program writes a single element (row, col) across columns via constexpr loop.
    # However, for simplicity and to avoid complex indexing, we assume the caller
    # launches grid=(N, 1) and provides a tensor of shape (N, 1). We extend to arbitrary
    # shape by launching over N elements and writing [pid, 0]. For ones, we can just
    # write 1.0 to each element via another kernel that handles full shape.
    # Here we implement a generic 1D fill for 1D tensors. For higher dims, launch per-element.
    tl.store(X_ptr + pid * stride_xn + 0 * stride_xm, 1.0)


# Kernel: create random normal tensor (float32) using tl.rand, write to X_ptr
@triton.jit
def randn_fill_kernel(X_ptr, N, stride_xn, stride_xm, mean, std):
    pid = tl.program_id(0)
    if pid >= N:
        return
    r = tl.rand()  # Triton provides a random number generator
    val = mean + std * (r - 0.5) * 4.0  # approximate N(mean, std)
    tl.store(X_ptr + pid * stride_xn + 0 * stride_xm, val)


# LayerNorm forward kernel: normalize across last dimension (feature D)
# Input: X (B, L, D), Output: Y (B, L, D)
# Gamma (weight) and Beta (bias) are applied elementwise after normalization
@triton.jit
def layernorm_forward_kernel(
    X_ptr, Y_ptr,
    W_ptr, B_ptr,
    B, L, D,
    eps,
    stride_xb, stride_xl, stride_xd,
    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    if b >= B or l >= L:
        return
    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        x = tl.where(mask, x, 0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        gamma = tl.load(W_ptr + offs * stride_w, mask=mask, other=1.0)
        beta = tl.load(B_ptr + offs * stride_b, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * gamma + beta
        tl.store(Y_ptr + b * stride_yb + l * stride_yl + offs * stride_yd, y, mask=mask)
        d0 += BLOCK_SIZE

# Exponential modulation kernel: v_new[b, c, l] = v[b, c, l] * (exp(-t[b, l] * abs(deltas[c])) + shift)
@triton.jit
def exp_mod_kernel(
    V_ptr,        # input v
    Deltas_ptr,   # deltas (B, 1, D) flattened
    T_ptr,        # t (B, L_out, 1) flattened
    Vout_ptr,     # output v
    B, D, L_out,
    stride_vb, stride_vc, stride_vl,
    stride_db, stride_dc, stride_dl,  # deltas strides
    stride_tb, stride_tl,             # t strides (we ignore d since last dim=1)
    stride_vob, stride_voc, stride_vol,
    shift,  # scalar
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    l = tl.program_id(2)
    if b >= B or c >= D or l >= L_out:
        return
    v_val = tl.load(V_ptr + b * stride_vb + c * stride_vc + l * stride_vl)
    # deltas is shaped (B, 1, D); for each c, index c
    # We assume deltas layout is contiguous as (B, D) along D==last. Here we pass flattened; Triton supports multi-index via 3D launch. We access deltas[b, c] via linear indexing.
    # To avoid complex indexing, we precompute per-(B,D) and pass as a contiguous tensor.
    # For clarity, we assume that Deltas_ptr points to (B*D) layout. Compute offset: b*D + c
    delta_val = tl.load(Deltas_ptr + b * D + c)
    # t is (B, L_out), single channel; t_ptr points to b*L_out + l
    t_val = tl.load(T_ptr + b * L_out + l)
    delta_abs = tl.abs(delta_val)
    exp_val = tl.exp(-t_val * delta_abs)
    mod = exp_val + shift
    new_val = v_val * mod
    tl.store(Vout_ptr + b * stride_vob + c * stride_voc + l * stride_vol, new_val)


# Batched GEMM for output projection: C = A @ W^T + b, where
# A: (B*L_out, D) with A[b, l, d] stored linearly, W: (D, D), b: (D,)
# Output C: (B, D, L_out). We implement a reduction over D and launch over (B, D, L_out).
@triton.jit
def linear_gemm_kernel(
    A_ptr,        # (B*L, D), flattened
    W_ptr,        # (D, D)
    B_ptr,        # (D,)
    C_ptr,        # (B, D, L_out), flattened
    B, D, L_out,
    stride_ab, stride_ad,     # strides for A (A has shape (B*L, D))
    stride_wdr, stride_wdk,   # strides for W (W has shape (D, D))
    stride_cob, stride_cod, stride_col,  # strides for C
):
    b = tl.program_id(0)
    d = tl.program_id(1)
    l_out = tl.program_id(2)
    if b >= B or d >= D or l_out >= L_out:
        return
    acc = 0.0
    # We need to compute dot(A_row[b, l_out, :], W[:, d]) over the D dimension of W
    # A_row is A[b*L_out + l_out, :]; pointer offset = (b*L_out + l_out) * D + d
    # For W, we reduce over k in [0, D): W[k, d]
    for k in range(D):
        a_val = tl.load(A_ptr + (b * L_out + l_out) * D + k * stride_ad)  # assuming A is (B*L, D) contiguous: A_ptr + (b*L_out + l_out)*D + k
        w_val = tl.load(W_ptr + k * stride_wdr + d * stride_wdk)
        acc += a_val * w_val
    bval = tl.load(B_ptr + d)
    # Store to C[b, d, l_out] using provided strides
    tl.store(C_ptr + b * stride_cob + d * stride_cod + l_out * stride_col, acc + bval)
