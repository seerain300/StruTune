# Triton LayerNorm forward for 3D tensors (B, L, D): normalize over last dim, affine
@triton.jit
def layernorm_forward_kernel(
    X_ptr,        # *const float
    W_ptr,        # *const float (gamma), shape [D]
    B_ptr,        # *const float (beta), shape [D]
    Y_ptr,        # *float
    B, L, D,      # int
    eps,          # float
    stride_xb, stride_xl, stride_xd,
    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):
    b = tl.program_id(0)
    l = tl.program_id(1)
    if b >= B or l >= L:
        return

    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        x = tl.where(mask, x, 0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)
        w = tl.load(W_ptr + offs * stride_w, mask=mask, other=1.0)
        bval = tl.load(B_ptr + offs * stride_b, mask=mask, other=0.0)
        y = ((x - mean) * rstd) * w + bval
        tl.store(Y_ptr + b * stride_yb + l * stride_yl + offs * stride_yd, y, mask=mask)
        d0 += BLOCK_SIZE

# Here Up is the padded input u: Up[b, c, l] = u[b, c, l - pad] if in range, else 0. Wc is weight per channel (1xK). Bc is bias per channel.
@triton.jit
def conv1d_groups_exact_kernel(
    Up_ptr,       # *const float, padded input, shape [B, C, L_in]
    Wc_ptr,       # *const float, weight per channel, shape [C, K], here K=3
    Bc_ptr,       # *const float, bias per channel, shape [C]
    Uout_ptr,     # *float, output, shape [B, C, L_out]
    B, C, L_in, K, pad,
    stride_upb, stride_upc, stride_upl,
    stride_wc, stride_wk,
    stride_uob, stride_uoc, stride_uol,
):

    c = tl.program_id(1)
    if b >= B or c >= C:
        return

    L_out = L_in - K + 1 + 2 * pad  # depthwise conv with kernel size=3, padding=2, no stride
    l_out = 0
    while l_out < L_out:
        acc = 0.0
        k = 0
        while k < K:
            inp_l = l_out - pad + k
            valid = (inp_l >= 0) & (inp_l < L_in)
            val = tl.load(Up_ptr + b * stride_upb + c * stride_upc + inp_l * stride_upl, mask=valid, other=0.0)
            w = tl.load(Wc_ptr + c * stride_wc + k * stride_wk)
            acc += val * w
            k += 1
        bc = tl.load(Bc_ptr + c)
        acc += bc
        tl.store(Uout_ptr + b * stride_uob + c * stride_uoc + l_out * stride_uol, acc)
        l_out += 1


# Triton exp modulation kernel: V_in[B*D*L] -> V_out[B*D*L]
# v_new = v * (exp(-t * abs(deltas)) + shift)
# deltas has shape [D] and broadcasts over batch and sequence via t index.
@triton.jit
def exp_mod_kernel(
    V_ptr,        # *const float, input, shape [B, D, L] flattened
    Deltas_ptr,   # *const float, shape [D]
    B, D, L,      # int
    shift,        # float
    stride_vb, stride_vd, stride_vl,
):
    pid = tl.program_id(0)
    if pid >= B * D * L:
        return
    b = pid // (D * L)
    rem = pid % (D * L)
    d = rem // L
    l = rem % L
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    delta = tl.load(Deltas_ptr + d)
    # t is the sequence index l
    exp_term = tl.exp(-l * tl.abs(delta))
    v_new = v * (exp_term + shift)
    tl.store(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl, v_new)

    W_ptr,        # *const float (gamma), shape [D]
    B_ptr,        # *const float (beta), shape [D]
    Y_ptr,        # *float
    B, L, D,      # int
    eps,          # float

    stride_yb, stride_yl, stride_yd,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
):

    l = tl.program_id(1)
    if b >= B or l >= L:
        return

    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0
    while d0 < D:
        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + b * stride_xb + l * stride_xl + offs * stride_xd, mask=mask, other=0.0)

        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE

        offs = d0 + tl.arange(0, BLOCK_SIZE)

    B, C, L_in, K, pad,
    stride_upb, stride_upc, stride_upl,
    stride_wc, stride_wk,
    stride_uob, stride_uoc, stride_uol,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    if b >= B or c >= C:
        return

    L_out = L_in - K + 1 + 2 * pad  # depthwise conv with kernel size=3, padding=2, no stride
    l_out = 0
    while l_out < L_out:
        acc = 0.0
        k = 0
        while k < K:
            inp_l = l_out - pad + k
            valid = (inp_l >= 0) & (inp_l < L_in)
            val = tl.load(Up_ptr + b * stride_upb + c * stride_upc + inp_l * stride_upl, mask=valid, other=0.0)
            w = tl.load(Wc_ptr + c * stride_wc + k * stride_wk)
            acc += val * w
            k += 1
        bc = tl.load(Bc_ptr + c)
        acc += bc
        tl.store(Uout_ptr + b * stride_uob + c * stride_uoc + l_out * stride_uol, acc)
        l_out += 1


# Triton exp modulation kernel: V_in[B*D*L] -> V_out[B*D*L]
# v_new = v * (exp(-t * abs(deltas)) + shift)
# deltas has shape [D] and broadcasts over batch and sequence via t index.
@triton.jit
def exp_mod_kernel(
    V_ptr,        # *const float, input, shape [B, D, L] flattened
    Deltas_ptr,   # *const float, shape [D]
    B, D, L,      # int
    shift,        # float
    stride_vb, stride_vd, stride_vl,
):
    pid = tl.program_id(0)
    if pid >= B * D * L:
        return
    b = pid // (D * L)
    rem = pid % (D * L)
    d = rem // L
    l = rem % L
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    delta = tl.load(Deltas_ptr + d)
    exp_term = tl.exp(-l * tl.abs(delta))
    v_new = v * (exp_term + shift)
    tl.store(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl, v_new)


# Triton GEMM: A[M, K] @ W[K, N] -> C[M, N], where M = B*L, K = D, N = D2
# This is used for output projection: y_flat (B*L, D) @ out_proj_weight^T (D, D) + bias (D)
@triton.jit
def linear_gemm_kernel(
    A_ptr,        # *const float, shape [M, K] flattened (row-major view)
    W_ptr,        # *const float, shape [K, N] flattened
    B_ptr,        # *const float, bias [N]
    C_ptr,        # *float, output [M, N] flattened
    M, K, N,
    stride_am, stride_ak,
    stride_wk, stride_wn,
    stride_cm, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    n = tl.program_id(1)
    if m >= M or n >= N:
        return
    acc = 0.0
    for k0 in range(0, K, BLOCK_K):
        for i in range(0, N, BLOCK_N):
            a = tl.load(A_ptr + m * stride_am + (k0 + tl.arange(0, BLOCK_K)) * stride_ak, mask=(k0 + tl.arange(0, BLOCK_K)) < K, other=0.0)
            b = tl.load(W_ptr + (k0 + tl.arange(0, BLOCK_K)) * stride_wk + (i + tl.arange(0, BLOCK_N)) * stride_wn, mask=(i + tl.arange(0, BLOCK_N)) < N, other=0.0)
            # dot product across K
            acc += tl.sum(a[:, None] * b[None, :], axis=0)
    bval = tl.load(B_ptr + n * stride_wn, other=0.0)
    acc += bval
    tl.store(C_ptr + m * stride_cm + n * stride_cn, acc)


# Triton kernel to fill tensor with random normal values (float32). Not used in forward to avoid torch.randn, but present for completeness.
@triton.jit
def randn_fill_kernel(
    X_ptr,        # *float
    size,         # int total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < size
    val = tl.full([BLOCK], 0.0, tl.float32)  # placeholder; Triton does not provide tl.randn. We will not use this in forward.
    tl.store(X_ptr + offs, val, mask=mask)


# Triton kernel to fill tensor with ones (float32)
@triton.jit
def fill_ones_kernel(
    X_ptr,        # *float
    size,         # int total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < size
    ones = tl.full([BLOCK], 1.0, tl.float32)
    tl.store(X_ptr + offs, ones, mask=mask)
