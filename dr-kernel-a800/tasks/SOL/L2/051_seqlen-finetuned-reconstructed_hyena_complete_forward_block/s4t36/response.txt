import torch
import triton
import triton.language as tl


# 1) Triton LayerNorm (affine) over last dim for a [M, N] tensor (M = B*S, N = D).
# Two kernels:
#   - compute per-row sum and sum of squares
#   - normalize and apply affine

@triton.jit
def _layernorm_mean_var_kernel(
    X_ptr,             # *fp32, input [M, N]
    SUM_ptr,           # *fp32, per-row sum [M]
    SUMSQ_ptr,         # *fp32, per-row sum of squares [M]
    M, N,
    stride_xm, stride_xn,
):
    pid = tl.program_id(axis=0)
    if pid >= M:
        return
    acc = tl.zeros((), dtype=tl.float32)
    acc_sq = tl.zeros((), dtype=tl.float32)
    for j in range(0, N):
        x = tl.load(X_ptr + pid * stride_xm + j * stride_xn)
        acc += x
        acc_sq += x * x
    tl.store(SUM_ptr + pid, acc)
    tl.store(SUMSQ_ptr + pid, acc_sq)

@triton.jit
def _layernorm_norm_affine_kernel(
    X_ptr, SUM_ptr, SUMSQ_ptr,
    Y_ptr,
    WEIGHT_ptr, BIAS_ptr,
    M, N,
    eps,                     # float32
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_w,                # weight is 1D [N], contiguous
):
    pid = tl.program_id(axis=0)
    if pid >= M:
        return
    mean = tl.load(SUM_ptr + pid) / N
    var = tl.load(SUMSQ_ptr + pid) / N - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    for j in range(0, N):
        x = tl.load(X_ptr + pid * stride_xm + j * stride_xn)
        w = tl.load(WEIGHT_ptr + j * stride_w)
        b = tl.load(BIAS_ptr + j * stride_w)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(Y_ptr + pid * stride_ym + j * stride_yn, y)


# 2) Triton elementwise add: C[M, N] = A[M, N] + B[M, N]
@triton.jit
def _add_2d_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N,
    stride_am, stride_an,
    stride_bm, stride_bn,
    stride_cm, stride_cn,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    if (pid_m >= M) or (pid_n >= N):
        return
    a = tl.load(A_ptr + pid_m * stride_am + pid_n * stride_an)
    b = tl.load(B_ptr + pid_m * stride_bm + pid_n * stride_bn)
    c = a + b
    tl.store(C_ptr + pid_m * stride_cm + pid_n * stride_cn, c)


# 3) Triton linear row-wise: C[M, N] = A[M, D] @ W_t[D, N] + BIAS[N], where W_t is pre-transposed to [D, N]
@triton.jit
def _linear_rowwise_kernel(
    A_ptr,           # *fp32, A [M, D], row-major
    WT_ptr,          # *fp32, W^T [D, N], row-major
    BIAS_ptr,        # *fp32, bias [N]
    C_ptr,           # *fp32, output [M, N], row-major
    M, D, N,
    stride_am, stride_ad,  # strides for A
    stride_wt_d, stride_wt_n,  # strides for WT (d, n)
    stride_cm, stride_cn,     # strides for C
    BLOCK_N: tl.constexpr,    # tile over N
    BLOCK_D: tl.constexpr,    # tile over D
):
    pid_m = tl.program_id(axis=0)
    if pid_m >= M:
        return
    offs_n = tl.arange(0, BLOCK_N)
    for n_start in range(0, N, BLOCK_N):
        n_ids = n_start + offs_n
        mask_n = n_ids < N
        acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
        for d_start in range(0, D, BLOCK_D):
            d_ids = d_start + tl.arange(0, BLOCK_D)
            mask_d = d_ids < D
            a = tl.load(A_ptr + pid_m * stride_am + d_ids * stride_ad, mask=mask_d, other=0.0)  # [BLOCK_D]
            # Load WT[n_ids, d_ids] -> shape [BLOCK_N, BLOCK_D]
            wt = tl.load(
                WT_ptr + n_ids[:, None] * stride_wt_d + d_ids[None, :] * stride_wt_n,
                mask=mask_n[:, None] & mask_d[None, :],
                other=0.0,
            )
            acc += tl.sum(wt * a[None, :], axis=1)
        # Add bias
        b = tl.load(BIAS_ptr + n_ids, mask=mask_n, other=0.0)
        acc += b
        # Store
        tl.store(C_ptr + pid_m * stride_cm + n_ids * stride_cn, acc, mask=mask_n)


def _run_triton_layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float):
    # x: [B, S, D], weight/bias: [D]
    assert x.is_cuda and x.dtype == torch.float32 and weight.is_cuda and weight.dtype == torch.float32 and bias.is_cuda and bias.dtype == torch.float32
    B, S, D = x.shape
    x_2d = x.contiguous().view(B * S, D)
    M = B * S
    # Allocate outputs for sum and sumsq
    sum_out = torch.empty((M,), device=x.device, dtype=torch.float32)
    sumsq_out = torch.empty((M,), device=x.device, dtype=torch.float32)
    y = torch.empty_like(x_2d)

    # Kernel 1: compute mean and variance
    grid = (M,)
    _layernorm_mean_var_kernel[grid](
        x_2d, sum_out, sumsq_out,
        M, D,
        x_2d.stride(0), x_2d.stride(1),
        num_warps=4,
    )

    # Kernel 2: normalize and affine
    _layernorm_norm_affine_kernel[grid](
        x_2d, sum_out, sumsq_out,
        y,
        weight, bias,
        M, D,
        eps,
        x_2d.stride(0), x_2d.stride(1),
        y.stride(0), y.stride(1),
        weight.stride(0),
        num_warps=4,
    )
    return y.view(B, S, D)


def _run_triton_linear(a_flat: torch.Tensor, w: torch.Tensor, bias: torch.Tensor):
    # a_flat: [M, D], w: [N, D], returns [M, N]
    assert a_flat.is_cuda and a_flat.dtype == torch.float32 and w.is_cuda and w.dtype == torch.float32 and bias.is_cuda and bias.dtype == torch.float32
    M, D = a_flat.shape
    N = w.shape[0]
    a = a_flat.contiguous()
    # Pre-transpose w to [D, N] so WT[n, d] = w[d, n]
    w_t = w.transpose(0, 1).contiguous()  # [D, N]
    bias = bias.contiguous()
    out = torch.empty((M, N), device=a.device, dtype=torch.float32)
    # Launch grid over rows
    grid = (M,)
    # Choose tiles
    BLOCK_N = 64 if N >= 64 else (32 if N >= 32 else 16)
    BLOCK_D = 64 if D >= 64 else (32 if D >= 32 else 16)
    _linear_rowwise_kernel[grid](
        a, w_t, bias, out,
        M, D, N,
        a.stride(0), a.stride(1),
        w_t.stride(0), w_t.stride(1),
        out.stride(0), out.stride(1),
        BLOCK_N=BLOCK_N, BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out


def _run_triton_add(a: torch.Tensor, b: torch.Tensor):
    # a, b: [B, S, D], returns [B, S, D]
    assert a.is_cuda and a.dtype == torch.float32 and b.is_cuda and b.dtype == torch.float32
    B, S, D = a.shape
    a_flat = a.contiguous().view(B * S, D)
    b_flat = b.contiguous().view(B * S, D)
    out_flat = torch.empty_like(a_flat)
    M = B * S
    grid = (M, D)
    _add_2d_kernel[grid](
        a_flat, b_flat, out_flat,
        M, D,
        a_flat.stride(0), a_flat.stride(1),
        b_flat.stride(0), b_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1),
        num_warps=2,
    )
    return out_flat.view(B, S, D)


class ModelNew(torch.nn.Module):

    def forward(

    ):
        # Ensure CUDA float32
        hidden_states = hidden_states.contiguous().to(torch.float32)
        # 1) First LayerNorm using Triton
        layer1_out = _run_triton_layer_norm(hidden_states, norm1_weight, norm1_bias, layer_norm_eps)  # [B, S, D]
