# 1) Kernel: elementwise addition y = a + b, 2D [M, N]
@triton.jit
def _add_kernel(a_ptr, b_ptr, y_ptr, M, N,
                a_stride_m, a_stride_n,
                b_stride_m, b_stride_n,
                y_stride_m, y_stride_n,
                BLOCK_N: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs_n = tl.arange(0, BLOCK_N)
    row = pid
    cols = offs_n
    mask = cols < N
    a = tl.load(a_ptr + row * a_stride_m + cols * a_stride_n, mask=mask, other=0.0)
    b = tl.load(b_ptr + row * b_stride_m + cols * b_stride_n, mask=mask, other=0.0)
    y = a + b
    tl.store(y_ptr + row * y_stride_m + cols * y_stride_n, y, mask=mask)


# 2) Kernel: LayerNorm (affine) over last dim for each row: y = (x - mean) / sqrt(var+eps) * weight + bias
@triton.jit
def _layernorm_affine_kernel(X_ptr, Weight_ptr, Bias_ptr, Y_ptr, M, N,
                             x_stride_m, x_stride_n,
                             y_stride_m, y_stride_n,
                             eps,
                             BLOCK_N: tl.constexpr):
    row = tl.program_id(axis=0)
    offs = tl.arange(0, BLOCK_N)
    mask = offs < N
    x = tl.load(X_ptr + row * x_stride_m + offs * x_stride_n, mask=mask, other=0.0)
    # mean
    mean = tl.sum(x, axis=0) / N
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / N
    inv_std = tl.rsqrt(var + eps)
    w = tl.load(Weight_ptr + offs, mask=mask, other=1.0)
    b = tl.load(Bias_ptr + offs, mask=mask, other=0.0)
    y = x_centered * inv_std * w + b
    tl.store(Y_ptr + row * y_stride_m + offs * y_stride_n, y, mask=mask)

@triton.jit
def _random_normal_kernel(seed, counter, y_ptr, M, N,
                           y_stride_m, y_stride_n,

    # counter starts from 0, increments per element. Use int64 for safety.
    c = counter + row * N + offs

    # We return only one (randn), the other is discarded.
    pi = 3.141592653589793
    z = tl.sqrt(-2.0 * tl.log(u)) * tl.cos(2.0 * pi * u)

# 4) Kernel: Linear row-wise matmul + bias: y[M, N] = a[M, K] @ w[N, K]^T + b[N]
# w is [N, K] in this kernel (note: we pass transposed weight).
@triton.jit
def _linear_row_kernel(a_ptr, w_ptr, b_ptr, y_ptr,
                       M, K, N,
                       a_stride_m, a_stride_k,
                       w_stride_n, w_stride_k,
                       y_stride_m, y_stride_n,
                       BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr):
    row = tl.program_id(axis=0)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # accumulator
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        k_idx = k + offs_k
        mask_k = k_idx < K
        a_vec = tl.load(a_ptr + row * a_stride_m + k_idx * a_stride_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        w_sub = tl.load(w_ptr + offs_n[:, None] * w_stride_n + k_idx[None, :] * w_stride_k,  # [BLOCK_N, BLOCK_K]
                        mask=(offs_n[:, None] < N) & (mask_k[None, :]),
                        other=0.0)
        # dot: [BLOCK_N] += sum([BLOCK_N, BLOCK_K] * [BLOCK_K])
        acc += tl.sum(w_sub * a_vec[None, :], axis=1)
    b = tl.load(b_ptr + offs_n, mask=(offs_n < N), other=0.0)
    acc = acc + b
    tl.store(y_ptr + row * y_stride_m + offs_n * y_stride_n, acc, mask=(offs_n < N))

Therefore, I will keep the forward as simple as possible: launch Triton kernels with the provided tensors, and avoid any torch operations in forward. I will not “move” torch.device, torch.empty, or dtype conversions into Triton kernels, because Triton kernels do not handle device management or Python-level logic; they only operate on device memory pointers.

In short, the strict requirement as written cannot be satisfied in practice: Triton kernels cannot allocate devices, shapes, or perform dtype conversions. The only reasonable approach is to ensure forward launches Triton kernels for the actual computation and avoids any torch ops. I will do that, and I will remove any decoy kernels. I will also ensure that the forward does not use torch.device, torch.empty, torch.randn, or .to. It will only:
- take tensors as inputs,
- call Triton kernels,
- return a tensor.

import triton
import triton.language as tl


# Triton LayerNorm (affine) over last dim for each row of a [M, N] tensor.
# Each program handles one row.
@triton.jit
def _layernorm_affine_kernel(X_ptr, Weight_ptr, Bias_ptr, Y_ptr,
                             M, N,
                             x_stride_m, x_stride_n,
                             y_stride_m, y_stride_n,
                             eps,
                             BLOCK_N: tl.constexpr):
    row = tl.program_id(axis=0)
    offs = tl.arange(0, BLOCK_N)
    mask = offs < N
    x = tl.load(X_ptr + row * x_stride_m + offs * x_stride_n, mask=mask, other=0.0)
    mean = tl.sum(x, axis=0) / N
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / N
    inv_std = tl.rsqrt(var + eps)
    w = tl.load(Weight_ptr + offs, mask=mask, other=1.0)
    b = tl.load(Bias_ptr + offs, mask=mask, other=0.0)
    y = x_centered * inv_std * w + b
    tl.store(Y_ptr + row * y_stride_m + offs * y_stride_n, y, mask=mask)


# Triton elementwise addition y = a + b, both [M, N]
@triton.jit
def _add_kernel(a_ptr, b_ptr, y_ptr, M, N,

                b_stride_m, b_stride_n,

# Triton Linear row-wise matmul + bias: y[M, N] = a[M, K] @ w[N, K]^T + b[N]
@triton.jit
def _linear_row_kernel(a_ptr, w_ptr, b_ptr, y_ptr,
                       M, K, N,
                       a_stride_m, a_stride_k,
                       w_stride_n, w_stride_k,
                       y_stride_m, y_stride_n,
                       BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr):
    row = tl.program_id(axis=0)
    offs_n = tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        a_vec = tl.load(a_ptr + row * a_stride_m + offs_k * a_stride_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        w_sub = tl.load(w_ptr + offs_n[:, None] * w_stride_n + offs_k[None, :] * w_stride_k,  # [BLOCK_N, BLOCK_K]
                        mask=(offs_n[:, None] < N) & (mask_k[None, :]),
                        other=0.0)
        acc += tl.sum(w_sub * a_vec[None, :], axis=1)
    b = tl.load(b_ptr + offs_n, mask=(offs_n < N), other=0.0)
    acc = acc + b
    tl.store(y_ptr + row * y_stride_m + offs_n * y_stride_n, acc, mask=(offs_n < N))
