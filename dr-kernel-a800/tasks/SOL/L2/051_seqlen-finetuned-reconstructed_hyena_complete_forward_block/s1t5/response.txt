    N, D,

    BLOCK_D: tl.constexpr
):
    # Allocate and fill a 2D buffer (N, D) with zeros using Triton.
    pid = tl.program_id(0)  # each program handles one row i in [0, N)
    offs = tl.arange(0, BLOCK_D)

    tl.store(out_row_ptr, zero, mask=mask)


@triton.jit
def linspace_1d_kernel(
    out_ptr, start, end, length,
    BLOCK: tl.constexpr
