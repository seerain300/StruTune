        # k_offsets: -2, -1, 0 (due to padding=2 and kernel length=3)
        for k_off in range(-2, 1):
            inp_pos = l_out - pad + k_off
            if inp_pos >= 0 and inp_pos < Lp:
                if k_off == -2:
                    idx2 = 0
                elif k_off == -1:
                    idx2 = 1
                else:
                    idx2 = 2
                w_val = tl.load(Wc_ptr + b * stride_wbk + k * stride_wbk2 + idx2 * stride_wbk3)
                val = tl.load(Up_ptr + b * stride_upb + k * stride_upk + inp_pos * stride_upl)
                acc += val * w_val
        bias_idx = b * K + k
        bval = tl.load(Bo_ptr + bias_idx)
        acc += bval
        tl.store(Uout_ptr + b * stride_ob + k * stride_ok + l_out * stride_ol, acc)


# Exponential modulation kernel: V_in[B*D*L] -> V_out[B*D*L]
# v_new = v * (exp(-t * abs(deltas)) + shift)
# deltas has shape (D,) and broadcasts over batch and sequence via t index.
@triton.jit
def exp_mod_kernel(
    V_ptr,        # *const float
    Deltas_ptr,   # *const float, shape [D]
    B, D, L,      # int
    shift,        # float
    stride_vb, stride_vd, stride_vl,
):
    total = B * D * L
    pid = tl.program_id(0)
    if pid >= total:
        return
    b = pid // (D * L)
    rem = pid % (D * L)
    d = rem // L
    l = rem % L
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    delta = tl.load(Deltas_ptr + d)
    t = l  # position along sequence
    exp_term = tl.exp(-t * tl.abs(delta))
    v_new = v * (exp_term + shift)
    tl.store(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl, v_new)


# Linear GEMM: A[M, K] @ W[K, N] -> C[M, N], where M = B*L, K = D, N = D2
@triton.jit
def linear_gemm_kernel(
    A_ptr,        # *const float, shape [M, K] flattened
    W_ptr,        # *const float, shape [K, N] flattened
    B_ptr,        # *const float, bias [N]
    C_ptr,        # *float, output [M, N] flattened
    M, K, N,
    stride_am, stride_ak,
    stride_wk, stride_wn,
    stride_cm, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    n = tl.program_id(1)
    if m >= M or n >= N:
        return
    acc = 0.0
    for k0 in range(0, K, BLOCK_K):
        for kk in range(0, BLOCK_K):
            k = k0 + kk
            if k >= K:
                break
            a = tl.load(A_ptr + m * stride_am + k * stride_ak)
            w = tl.load(W_ptr + k * stride_wk + n * stride_wn)
            acc += a * w
    bval = tl.load(B_ptr + n * stride_wn)
    acc += bval
    tl.store(C_ptr + m * stride_cm + n * stride_cn, acc)


def _ceil_div(a, b):
    return (a + b - 1) // b
