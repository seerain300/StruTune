@triton.jit

    eps,

    stride_yb, stride_yl, stride_yd,
    stride_gamma, stride_beta,
    BLOCK_SIZE: tl.constexpr,
):

    l = tl.program_id(1)
    if b >= B or l >= L:
        return
    sum_val = 0.0
    sum_sq = 0.0
    d0 = 0

        offs = d0 + tl.arange(0, BLOCK_SIZE)
        mask = offs < D

        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        d0 += BLOCK_SIZE

# v_new = v * (exp(-t * abs(deltas)) + shift), deltas has shape [D], shift is scalar.
@triton.jit
def exp_mod_kernel(
    V_ptr, Deltas_ptr, B, D, L, shift, stride_vb, stride_vd, stride_vl
):
    pid = tl.program_id(0)
    total = B * D * L
    if pid >= total:
        return
    b = pid // (D * L)
    rem = pid % (D * L)
    d = rem // L
    l = rem % L
    v = tl.load(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl)
    delta = tl.load(Deltas_ptr + d)
    t = l  # position index along sequence

    v_new = v * (exp_term + shift)
    tl.store(V_ptr + b * stride_vb + d * stride_vd + l * stride_vl, v_new)
