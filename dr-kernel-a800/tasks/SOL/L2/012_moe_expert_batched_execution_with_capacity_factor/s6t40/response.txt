            # Accumulate: acc += sum_m c[m] * D[m, :]
            # We need to load D[m, :] for each m in chunk and multiply-accumulate.
            # Since D is (M, H), we load rows m by m and accumulate.
            # Triton supports loops; we use a simple while loop for correctness.
            m = m_start
            while m < M:
                d_row_ptrs = D + m * stride_D_row + tl.arange(0, BLOCK_H) * stride_D_col
                d_row = tl.load(d_row_ptrs, mask=True, other=0.0)  # vector of size BLOCK_H
                acc += c[m] * d_row
                m += 1

        # Store results to E row
        e_ptrs = E + pid * stride_E_row + tl.arange(0, BLOCK_H) * stride_E_col
        tl.store(e_ptrs, acc, mask=True)
