# Triton kernel: flatten + sort by keys (selected_experts) with stable order (ascending). This replaces torch.sort(stable=True).
@triton.jit
def sort_stable_by_keys_kernel(keys_ptr, vals_ptr, tok_ptr, out_keys_ptr, out_vals_ptr, out_tok_ptr,
                               T: tl.int32, BLOCK: tl.constexpr):
    # We implement a bitonic sort network over BLOCK lanes; we load T elements with mask and pad with max int64 to push to end.
    idx = tl.arange(0, BLOCK)
    a = tl.load(keys_ptr + idx, mask=idx < T, other=tl.max_int64)
    b = tl.load(vals_ptr + idx, mask=idx < T, other=0.0)
    c = tl.load(tok_ptr + idx, mask=idx < T, other=0)
    # Bitonic sort (ascending)
    for stage in range(2, BLOCK + 1):
        size = stage
        for stride in range(2, size + 1, 2):
            i = idx
            j = i ^ (stride // 2)
            asc = (i & size) == 0
            a_i = a[i]
            a_j = a[j]
            # Swap condition: for asc, swap if a_i > a_j; for desc, swap if a_i < a_j.
            # Stable tie-breaker for equal keys: place smaller idx before larger (i < j).
            swap = tl.where(asc, a_i > a_j, a_i < a_j)
            swap |= (a_i == a_j) & (i > j)
            ai_new = tl.where(swap, a_j, a_i)
            bi_new = tl.where(swap, b[j], b[i])
            ci_new = tl.where(swap, c[j], c[i])
            a = tl.where(i == idx, ai_new, a)
            b = tl.where(i == idx, bi_new, b)
            c = tl.where(i == idx, ci_new, c)
    # Store sorted arrays
    tl.store(out_keys_ptr + idx, a, mask=idx < T)
    tl.store(out_vals_ptr + idx, b, mask=idx < T)
    tl.store(out_tok_ptr + idx, c, mask=idx < T)
