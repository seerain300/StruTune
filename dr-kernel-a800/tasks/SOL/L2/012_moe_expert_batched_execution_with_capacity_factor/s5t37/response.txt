@triton.jit
def bincount_experts_kernel(sorted_exp, counts, E, NUM_EXPERTS: tl.constexpr, BLOCK: tl.constexpr):

    for e_idx in range(NUM_EXPERTS):
        # count occurrences of e_idx in sorted_exp
        # We use a simple loop over chunks to avoid tl.unique. Triton doesn't have tl.bincount, so we reduce manually.
        total = tl.zeros((), dtype=tl.int32)
        for off in range(0, E, BLOCK):
            offs = off + tl.arange(0, BLOCK)
            mask = offs < E
            vals = tl.load(sorted_exp + offs, mask=mask, other=0).to(tl.int32)
            eq = (vals == e_idx) & mask
            # sum booleans: convert to int32 via eq.to(tl.int32)
            total += tl.sum(eq.to(tl.int32), axis=0)
        # atomic add total into counts[e_idx]
        tl.atomic_add(counts + e_idx, total)


# Triton kernel: compute cumsum of counts into starts (cumulative per-expert starts)
@triton.jit
def cumsum_starts_kernel(counts, starts, NUM_EXPERTS: tl.constexpr):

    acc = tl.zeros((), dtype=tl.int32)

        c = tl.load(counts + e).to(tl.int32)
        acc += c
        tl.store(starts + e, acc)

            num_tokens, hidden_size, intermediate_size, num_experts_per_tok, num_experts
