import math
import torch
import triton
import triton.language as tl


# Triton kernel: compute gate_out = A @ B, where A is [1, hidden_size], B is [hidden_size, intermediate_size], output [intermediate_size].
# Inputs: A_ptr, B_ptr, C_ptr (flattened output), hidden_size, intermediate_size, and linear index m_offset for A.
@triton.jit
def bmm_gate_kernel(
    A_ptr, B_ptr, C_ptr,
    hidden_size: tl.constexpr, intermediate_size: tl.constexpr,
    m_offset: tl.constexpr,
    BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # single batch, output vector of length intermediate_size
    base_a = m_offset * hidden_size
    offs_j = tl.arange(0, BLOCK_J)
    acc = tl.zeros((BLOCK_J,), dtype=tl.float32)

    for k0 in range(0, hidden_size, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < hidden_size
        a = tl.load(A_ptr + base_a + k_idx, mask=mask_k, other=0.0)
        # B is [hidden_size, intermediate_size]
        b = tl.load(B_ptr + k_idx[:, None] * intermediate_size + offs_j[None, :], mask=(mask_k[:, None]), other=0.0)
        # acc += sum_k (a[k] * B[k, j]) over BLOCK_K
        acc += tl.sum(b * a[:, None], axis=0)

    # store acc to C_ptr[m_offset * intermediate_size + offs_j]
    c_ptrs = C_ptr + m_offset * intermediate_size + offs_j
    tl.store(c_ptrs, acc, mask=(offs_j < intermediate_size))


# Triton kernel: compute up_out = A @ B, where A is [1, hidden_size], B is [hidden_size, intermediate_size], output [intermediate_size].
@triton.jit
def bmm_up_kernel(
    A_ptr, B_ptr, C_ptr,
    hidden_size: tl.constexpr, intermediate_size: tl.constexpr,
    m_offset: tl.constexpr,
    BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    base_a = m_offset * hidden_size
    offs_j = tl.arange(0, BLOCK_J)
    acc = tl.zeros((BLOCK_J,), dtype=tl.float32)

    for k0 in range(0, hidden_size, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < hidden_size
        a = tl.load(A_ptr + base_a + k_idx, mask=mask_k, other=0.0)
        b = tl.load(B_ptr + k_idx[:, None] * intermediate_size + offs_j[None, :], mask=(mask_k[:, None]), other=0.0)
        acc += tl.sum(b * a[:, None], axis=0)

    c_ptrs = C_ptr + m_offset * intermediate_size + offs_j
    tl.store(c_ptrs, acc, mask=(offs_j < intermediate_size))


# Triton kernel: scatter-add weighted rows into result.
# Weights come from v_wt, positions v_pos, tokens v_tok. For each valid i, add weighted_out[i, :] to result[v_tok[i], :].
# We do it by launching one program per row to add into result.
@triton.jit
def scatter_add_rows_kernel(
    weights_ptr, pos_ptr, tok_ptr, a_ptr, result_ptr,
    N: tl.constexpr, hidden_size: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    i = tl.program_id(0)
    if i >= N:
        return
    w = tl.load(weights_ptr + i)
    pos = tl.load(pos_ptr + i).to(tl.int32)
    tok = tl.load(tok_ptr + i).to(tl.int32)
    # a_ptr points to the start of the row to add, length hidden_size
    offs_j = tl.arange(0, BLOCK_J)
    vals = tl.load(a_ptr + offs_j, mask=(offs_j < hidden_size), other=0.0)
    dst = result_ptr + tok * hidden_size + offs_j
    # atomic add to avoid conflicts when multiple i map to same tok
    tl.atomic_add(dst, vals, mask=(offs_j < hidden_size))

    # Note: capacity may exceed number of valid positions for some experts; we only consider valid.
    # expert_inputs shape in compute: [num_valid, hidden_size], row m_offset = e*capacity + n but only valid rows are used.
    # However, we need A per m_offset for kernels. We construct a list of A rows; since the exact order is not needed for the gated logic in our original function, we can compute using valid positions directly.

    # Compute M_total = number of valid assignments
    M_total = v_exp.numel()

    # Prepare buffers for gate_out and up_out: total rows M_total * intermediate_size
    gate_out = torch.empty((M_total, intermediate_size), dtype=torch.float32, device=hidden_states.device)
    up_out = torch.empty((M_total, intermediate_size), dtype=torch.float32, device=hidden_states.device)

    # Launch Triton BMM kernels to compute gate_out and up_out
    # One program per row m_offset in [0, M_total)
    grid = (M_total,)

    BLOCK_K = 128 if hidden_size >= 128 else 64
    BLOCK_J = 64 if intermediate_size >= 64 else 32

    # Compute gate_out
    for m in range(M_total):
        # Load the corresponding hidden_state row for this token (v_tok[m])
        h = hidden_states[v_tok[m]].to(torch.float32)  # [hidden_size]
        # Note: Triton kernel expects A_ptr to point to start of row; we pass a contiguous view
        # Here we launch kernel directly with pointers; Triton will handle pointers; we need to create a buffer A_buf for each row. In practice, we can reuse h as A by copying into a [hidden_size] buffer for that row. To avoid large buffer, we can compute per-row directly with kernel by passing h as a tensor, but Triton kernel signature expects pointers.
        # We will pre-allocate A row buffer as h for this m and use it. However, Triton doesn't support Python-side dynamic per-row buffers easily. So we will instead create a temporary contiguous tensor per row before kernel launch. Since M_total is not huge and kernel overhead is minimal, we do this here.
        # Prepare A_ptr as a contiguous tensor per row: copy h into a temporary A row buffer
        A_row = h.clone()  # bfloat16 (will be cast inside kernel)
        # Compute B_ptr for gate weights
        B_gate = expert_gate_weights[v_exp[m]].contiguous()  # [hidden_size, intermediate_size]
        # Launch kernel
        bmm_gate_kernel[grid](
            A_row, B_gate, gate_out,
            hidden_size, intermediate_size, m,
            BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        )

    # Compute up_out similarly
    for m in range(M_total):
        A_row = hidden_states[v_tok[m]].to(torch.float32)
        B_up = expert_up_weights[v_exp[m]].contiguous()  # [hidden_size, intermediate_size]
        bmm_up_kernel[grid](
            A_row, B_up, up_out,
            hidden_size, intermediate_size, m,
            BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        )

    # SiLU and elementwise multiply (in PyTorch for simplicity and correctness)
    activated = torch.nn.functional.silu(gate_out) * up_out  # [M_total, intermediate_size], float32

    # Compute expert_outputs_all = activated @ expert_down_weights per (e, n)
    # We need to map to expert_down_weights using v_exp and v_pos. However, our activated rows are not aligned to capacity grid; since v_pos is only used for capacity masking, we can compute for all valid rows. The original function sums only valid rows; here all are valid due to mask.
    expert_down = expert_down_weights[v_exp]  # [M_total, intermediate_size, hidden_size]
    expert_outputs_all = torch.bmm(activated, expert_down)  # [M_total, hidden_size], float32

    # Scatter-add into result per token (use Triton kernel for performance)
    result = torch.zeros((num_tokens, hidden_size), dtype=torch.float32, device=hidden_states.device)

    N = M_total
    # Launch scatter-add kernel: one program per row
    grid_scatter = (N,)
    BLOCK_J_hs = 128 if hidden_size >= 128 else 64
    scatter_add_rows_kernel[grid_scatter](
        v_wt_f32, v_pos, v_tok.to(torch.int32), expert_outputs_all, result,

    # Cast result back to bfloat16 to match original dtype
    result_bf16 = result.to(torch.bfloat16)
    return result_bf16

        # Run Triton-optimized path

            A_row, B_down, expert_outputs_all,
            intermediate_size, hidden_size, m,
            BLOCK_J=BLOCK_J_down, BLOCK_K=intermediate_size,
        )

    result = torch.zeros((num_tokens, hidden_size), dtype=torch.float32, device=device)
    N = M_total
    grid_scatter = (N,)
    BLOCK_J_hs = 128 if hidden_size >= 128 else 64
    scatter_add_rows_kernel[grid_scatter](
        v_wt, v_pos, v_tok.to(torch.int32), expert_outputs_all, result,
        N, hidden_size, BLOCK_J=BLOCK_J_hs,
    )

    # Cast result back to bfloat16 to match original dtype
    result_bf16 = result.to(torch.bfloat16)
    return result_bf16
