    def row_bmm(A_ptr, B_ptr, C_ptr, H: tl.int32, M: tl.int32, stride_b0: tl.int32, stride_b1: tl.int32, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):

        Compute C[m] = sum_{h=0..H-1} A[h] * B[h, m] for m in [0, M).
        A: [H] (row vector)
        B: [H, M] (matrix)
        C: [M] (output vector)

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M

# Triton kernels for sorting, counts, starts, and matmuls + SiLU.
if triton is not None:
    @triton.jit
    def odd_even_sort(
        in_ptr,           # int64*, input array length N
        out_ptr,          # int64*, output sorted array length N
        N: tl.int32,
        BLOCK: tl.constexpr,
    ):
        # Odd-even transposition sort on int64 values.
        # We perform N phases; each phase consists of alternating even and odd passes.
        # This is a simple, correct approach for small N. For large N, torch.sort may be faster.
        for phase in range(0, 64):  # enough phases for typical sizes; mask ensures safety
            idx = tl.program_id(0)
            v = tl.load(out_ptr + idx)
            # Even pass: compare (0,1), (2,3), ...
            # Odd pass:  compare (1,2), (3,4), ...
            if (phase % 2) == 0:
                neighbor = idx + 1
            else:
                neighbor = idx - 1
            if (idx % 2) == 0 and neighbor < N and neighbor > 0:
                a = tl.load(out_ptr + idx)
                b = tl.load(out_ptr + neighbor)
                minv = tl.minimum(a, b)
                maxv = tl.maximum(a, b)
                # Write back min to even index, max to neighbor
                tl.store(out_ptr + idx, minv)
                tl.store(out_ptr + neighbor, maxv)

    @triton.jit
    def counts_kernel(exp_ptr, counts_ptr, N: tl.int32, EXP: tl.int32, BLOCK: tl.constexpr):
        # Atomic add 1 into counts[exp_ptr[i]] for each i
        for i in range(0, N, BLOCK):
            idx = i + tl.arange(0, BLOCK)
            mask = idx < N
            exps = tl.load(exp_ptr + idx, mask=mask, other=0)
            # Atomic add 1 for each valid exp
            tl.atomic_add(counts_ptr + exps, 1, mask=mask)

    @triton.jit
    def cumsum_starts_kernel(counts_ptr, starts_ptr, EXP: tl.int32):
        # Compute inclusive prefix sum of counts into starts
        total = tl.zeros((), dtype=tl.int32)
        for j in range(0, EXP):
            total += tl.load(counts_ptr + j)
            tl.store(starts_ptr + j, total)

    @triton.jit
    def compute_valid_positions_kernel(exp_ptr, starts_ptr, capacity: tl.int32, out_ptr, N: tl.int32):
        # out_ptr[i] = (exp_ptr[i] - starts[exp_ptr[i]]) if (exp_ptr[i] - starts[exp_ptr[i]] < capacity) else -1
        for i in range(0, N):
            e = tl.load(exp_ptr + i)
            s = tl.load(starts_ptr + e)
            pos = e - s
            valid = pos < capacity
            # Write pos if valid, else -1
            val = pos if valid else -1
            tl.store(out_ptr + i, val)

    @triton.jit
    def row_bmm(A_ptr, B_ptr, C_ptr, H: tl.int32, M: tl.int32, stride_b0: tl.int32, stride_b1: tl.int32, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
        """
        Compute C[m] = sum_{h=0..H-1} A[h] * B[h, m] for m in [0, M).
        A: [H] (row vector)
        B: [H, M] (matrix)
        C: [M] (output vector)
        """

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M

        # Initialize accumulator
        acc = tl.zeros([BLOCK_M], dtype=tl.float32)  # accumulate in fp32

        # Loop over H in tiles
        for h_start in range(0, H, BLOCK_H):
            offs_h = h_start + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # Load A[h] and B[h, m] tiles
            a = tl.load(A_ptr + offs_h, mask=mask_h, other=0.0)
            b = tl.load(B_ptr + offs_h[:, None] * stride_b0 + offs_m[None, :] * stride_b1, mask=mask_h[:, None] & mask_m[None, :], other=0.0)

            # Multiply and reduce over H tile
            acc += tl.sum(a[:, None] * b, axis=0)

        # Store result
        tl.store(C_ptr + offs_m, acc, mask=mask_m)

    @triton.jit
    def silu_kernel(X_ptr, Y_ptr, N: tl.int32, BLOCK: tl.constexpr):
        """

        offs = pid * BLOCK + tl.arange(0, BLOCK)

        x = tl.load(X_ptr + offs, mask=mask, other=0.0)

        D: [M, H] (matrix)
        C: [M] (input vector)
        A: [H] (output vector)
        """

        offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

            grid_silu = (triton.cdiv(M, 256),)
            silu_kernel[grid_silu](gate_out, silu_out, M, 256)

            activated = silu_out * up_out  # keep as fp32

            result[i] = final_row
