import math
import torch
import triton
import triton.language as tl


@triton.jit
def triton_bmm(X_ptr, W_ptr, Y_ptr,
                B, H, M,
                X_stride_b, X_stride_h,
                W_stride_h, W_stride_m,
                Y_stride_b, Y_stride_m,
                BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    """
    Triton batched matmul: for each batch element b, compute Y[b, m] = sum_k X[b, k] * W[k, m]
    Shapes:
      X: (B, H)
      W: (H, M)
      Y: (B, M)
    Strides are in elements (not bytes).
    """
    b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # tile along M
    pid_h = tl.program_id(2)  # tile along H (we reduce over H)

    m_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    h_offsets = tl.arange(0, BLOCK_H)  # we loop H in steps of BLOCK_H

    # Initialize accumulator per tile (BLOCK_M x BLOCK_H). We'll reduce over H and write per M.
    acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Loop over H dimension in chunks
    for k in range(0, H, BLOCK_H):
        k_offsets = k + h_offsets

        mask_m = m_offsets < M
        mask_h = k_offsets < H

        # Load X[b, k_offsets] -> shape (BLOCK_H,)
        x_ptrs = X_ptr + b * X_stride_b + k_offsets * X_stride_h
        x = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)  # (BLOCK_H,)

        # Load W[k_offsets, m_offsets] -> shape (BLOCK_H, BLOCK_M)
        w_ptrs = W_ptr + k_offsets[:, None] * W_stride_h + m_offsets[None, :] * W_stride_m
        w = tl.load(w_ptrs, mask=mask_h[:, None] & mask_m[None, :], other=0.0).to(tl.float32)  # (BLOCK_H, BLOCK_M)

        # Accumulate: acc += x[:, None] * w  -> (BLOCK_M, BLOCK_H) += (BLOCK_M, 1) * (1, BLOCK_H)
        # Note: Triton supports broadcasting in elementwise ops; we accumulate across K.
        acc += x[:, None] * w  # broadcast multiply, then accumulate

    # Now we have acc per tile (BLOCK_M, BLOCK_H). We write out Y[b, m_offsets].
    # We need to sum across BLOCK_H to get Y[b, m_offsets] vector (size BLOCK_M).
    # Compute the result by summing over the BLOCK_H dimension.
    y_vec = tl.sum(acc, axis=1)  # (BLOCK_M,)

    # Store results
    y_ptrs = Y_ptr + b * Y_stride_b + m_offsets * Y_stride_m
    tl.store(y_ptrs, y_vec, mask=mask_m)


@triton.jit
def triton_silu_mul(Z_ptr, U_ptr, Y_ptr,
                    N,
                    Z_stride, U_stride, Y_stride,
                    BLOCK: tl.constexpr):
    """
    Elementwise activation: Y[i] = silu(Z[i]) * U[i], for i in [0, N)
    silu(x) = x * sigmoid(x)
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    z = tl.load(Z_ptr + offsets * Z_stride, mask=mask, other=0.0).to(tl.float32)
    u = tl.load(U_ptr + offsets * U_stride, mask=mask, other=0.0).to(tl.float32)

    # sigmoid(z) = 1 / (1 + exp(-z))
    sig = 1.0 / (1.0 + tl.exp(-z))
    y = z * sig * u

    tl.store(Y_ptr + offsets * Y_stride, y, mask=mask)


@triton.jit
def triton_atomic_add_weighted(inp_ptr, weight_ptr, out_ptr, N,
                                inp_stride, weight_stride, out_stride,
                                BLOCK: tl.constexpr):
    """
    Atomic add contributions to out_ptr without using torch.index_add.
    We assume inp_ptr[N], weight_ptr[N] are vectors of length N and out_ptr is the result vector (length H).
    Each element in out_ptr is a float32 accumulator. We do: out[i] += weight[i] * inp[i] for i in [0, N).
    Note: This is a device-side atomic accumulation. The provided code uses dummy data here,
    because without torch ops (sort/bincount) we cannot determine valid positions.
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    inp = tl.load(inp_ptr + offsets * inp_stride, mask=mask, other=0.0).to(tl.float32)
    weight = tl.load(weight_ptr + offsets * weight_stride, mask=mask, other=0.0).to(tl.float32)

    contrib = inp * weight  # shape (BLOCK,)
    # Atomic add each contribution to corresponding out element
    for i in range(BLOCK):
        idx = offsets[i]
        if mask[i]:
            tl.atomic_add(out_ptr + idx * out_stride, contrib[i])

import math
import torch
import triton
import triton.language as tl


@triton.jit
def triton_bmm(X_ptr, W_ptr, Y_ptr,
                B, H, M,
                X_stride_b, X_stride_h,
                W_stride_h, W_stride_m,
                Y_stride_b, Y_stride_m,
                BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    """
    Triton batched matmul:
      Inputs:
        X: pointer to [B, H] (row-major), float32
        W: pointer to [H, M] (row-major), float32
      Output:
        Y: pointer to [B, M] (row-major), float32
      Computation: for each b, Y[b, m] = sum_{k=0..H-1} X[b, k] * W[k, m]
    """
    b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # tile along M
    pid_h = tl.program_id(2)  # tile along H reduction

    m_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # Accumulator for this tile (BLOCK_M x BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Reduce over H in chunks
    for k in range(0, H, BLOCK_H):
        k_offsets = k + h_offsets

        mask_m = m_offsets < M
        mask_h = k_offsets < H

        # Load X[b, k_offsets] -> (BLOCK_H,)
        x_ptrs = X_ptr + b * X_stride_b + k_offsets * X_stride_h
        x = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)  # (BLOCK_H,)

        # Load W[k_offsets, m_offsets] -> (BLOCK_H, BLOCK_M)
        w_ptrs = W_ptr + k_offsets[:, None] * W_stride_h + m_offsets[None, :] * W_stride_m
        w = tl.load(w_ptrs, mask=mask_h[:, None] & mask_m[None, :], other=0.0).to(tl.float32)  # (BLOCK_H, BLOCK_M)

        # Accumulate: acc += x[:, None] * w  -> (BLOCK_M, BLOCK_H)
        acc += x[:, None] * w

    # Sum across reduction dimension to get Y[b, m_offsets]
    y_vec = tl.sum(acc, axis=1)  # (BLOCK_M,)

    # Store results
    y_ptrs = Y_ptr + b * Y_stride_b + m_offsets * Y_stride_m
    tl.store(y_ptrs, y_vec, mask=mask_m)

import torch
import triton
import triton.language as tl


@triton.jit
def triton_bmm(X_ptr, W_ptr, Y_ptr,
                B, H, M,
                X_stride_b, X_stride_h,
                W_stride_h, W_stride_m,
                Y_stride_b, Y_stride_m,
                BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    """
    Triton batched matmul:
      X: [B, H], row-major
      W: [H, M], row-major
      Y: [B, M], row-major
    For each b in [0, B), Y[b, m] = sum_{k=0..H-1} X[b, k] * W[k, m].
    Accumulation is done in float32.
    """
    b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # tile along M
    pid_h = tl.program_id(2)  # tile along H (reduction)

    m_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k in range(0, H, BLOCK_H):
        k_offsets = k + h_offsets
        mask_m = m_offsets < M
        mask_h = k_offsets < H

        x_ptrs = X_ptr + b * X_stride_b + k_offsets * X_stride_h  # (BLOCK_H,)
        x = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)

        w_ptrs = W_ptr + k_offsets[:, None] * W_stride_h + m_offsets[None, :] * W_stride_m  # (BLOCK_H, BLOCK_M)
        w = tl.load(w_ptrs, mask=mask_h[:, None] & mask_m[None, :], other=0.0).to(tl.float32)

        acc += x[:, None] * w

    y_vec = tl.sum(acc, axis=1)  # (BLOCK_M,)
    y_ptrs = Y_ptr + b * Y_stride_b + m_offsets * Y_stride_m
    tl.store(y_ptrs, y_vec, mask=mask_m)


@triton.jit
def triton_silu_mul(Z_ptr, U_ptr, Y_ptr,
                    N,
                    Z_stride, U_stride, Y_stride,
                    BLOCK: tl.constexpr):
    """
    Elementwise activation:
      Y[i] = silu(Z[i]) * U[i], for i in [0, N)
      silu(x) = x * sigmoid(x), sigmoid(x) = 1 / (1 + exp(-x))
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    z = tl.load(Z_ptr + offsets * Z_stride, mask=mask, other=0.0).to(tl.float32)
    u = tl.load(U_ptr + offsets * U_stride, mask=mask, other=0.0).to(tl.float32)

    sig = 1.0 / (1.0 + tl.exp(-z))
    y = z * sig * u

    tl.store(Y_ptr + offsets * Y_stride, y, mask=mask)


@triton.jit
def triton_atomic_add_row(In_ptr, Weight_ptr, Out_ptr, N,
                           In_stride, Weight_stride, Out_stride_row,
                           BLOCK: tl.constexpr):
    """
    Atomic add rows into Out:
      For i in [0, N), atomic add In[i] * Weight[i] into Out[i, :].
      In: [N], Weight: [N], Out: [num_tokens, hidden_size] row-major
    """
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    in_vals = tl.load(In_ptr + offsets * In_stride, mask=mask, other=0.0).to(tl.float32)
    weights = tl.load(Weight_ptr + offsets * Weight_stride, mask=mask, other=0.0).to(tl.float32)

    contrib = in_vals * weights  # (BLOCK,)
    # Atomic add into Out[i, :] for each i in offsets
    for i in range(BLOCK):
        idx = offsets[i]
        if mask[i]:
            tl.atomic_add(Out_ptr + idx * Out_stride_row, contrib[i])
