- Triton kernels:

     - Compute in float32 for stability; outputs are float32.
  2) silu_mul_triton: elementwise activation. Computes Z = SiLU(U) * V. Both inputs are float32 vectors; output is float32.

  4) fill_zeros_triton: writes zeros to an output buffer (used to initialize result in forward without using torch.ones).

```python
import torch
import triton
import triton.language as tl


@triton.jit
def bmm_row_triton(X_ptr, W_ptr, Y_ptr,
                    H, M,
                    X_stride, W_stride0, W_stride1, Y_stride,
                    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    # Single-row matmul: Y = X @ W, X is 1xH row, W is HxM, Y is 1xM
    # We implement a simple accumulation over H in tiles.
    # Note: This kernel is called with grid=(1,) for a single output row.
    offs_m = tl.arange(0, BLOCK_M)
    # Accumulator for output row
    acc = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # Loop over H in chunks
    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load X slice (1xH): X_ptr has row index = 0 (single-row), col = offs_h
        x = tl.load(X_ptr + offs_h * X_stride, mask=mask_h, other=0.0)

        # Load W tile: rows = offs_h, cols = offs_m
        w = tl.load(W_ptr + offs_h[:, None] * W_stride0 + offs_m[None, :] * W_stride1,
                    mask=mask_h[:, None], other=0.0)

        # acc += sum_h x[h] * w[h, :]
        acc += tl.sum(w * x[:, None], axis=0)

    # Store acc to Y
    mask_m = offs_m < M
    tl.store(Y_ptr + offs_m * Y_stride, acc, mask=mask_m)


@triton.jit
def silu_mul_triton(U_ptr, V_ptr, Z_ptr,
                    N,
                    U_stride, V_stride, Z_stride,
                    BLOCK: tl.constexpr):
    # Elementwise Z = SiLU(U) * V
    offs = tl.arange(0, BLOCK)
    mask = offs < N
    u = tl.load(U_ptr + offs * U_stride, mask=mask, other=0.0)
    v = tl.load(V_ptr + offs * V_stride, mask=mask, other=0.0)
    z = tl.sigmoid(u) * u  # SiLU(u) = u * sigmoid(u)
    z = z * v
    tl.store(Z_ptr + offs * Z_stride, z, mask=mask)


@triton.jit
def atomic_accum_triton(IN_ptr, W_ptr, OUT_ptr,
                        N,
                        IN_stride, OUT_stride,
                        BLOCK: tl.constexpr):
    # OUT[row] += W * IN, where W is a scalar at W_ptr[0]
    offs = tl.arange(0, BLOCK)
    mask = offs < N
    in_vec = tl.load(IN_ptr + offs * IN_stride, mask=mask, other=0.0)
    w = tl.load(W_ptr)  # scalar
    out_vec = in_vec * w
    # Atomic add to OUT row (single-row grid, row index implied by launch)
    tl.atomic_add(OUT_ptr + offs * OUT_stride, out_vec, mask=mask)


@triton.jit
def fill_zeros_triton(OUT_ptr, N, OUT_stride, BLOCK: tl.constexpr):
    offs = tl.arange(0, BLOCK)
    mask = offs < N
    # Store zeros
    tl.store(OUT_ptr + offs * OUT_stride, 0.0, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, hidden_states, selected_experts, routing_weights,
                expert_gate_weights, expert_up_weights, expert_down_weights):
        # Ensure inputs are on CUDA device (Triton requires CUDA tensors)
        assert hidden_states.is_cuda and selected_experts.is_cuda and routing_weights.is_cuda \
               and expert_gate_weights.is_cuda and expert_up_weights.is_cuda and expert_down_weights.is_cuda, \
            "All inputs must be CUDA tensors."

        num_tokens, hidden_size = hidden_states.shape
        num_experts, H, M = expert_gate_weights.shape
        # num_experts_per_tok is inferred from selected_experts
        num_experts_per_tok = selected_experts.shape[1]

        # Prepare output result (float32 for accumulation; cast to bfloat16 at return)
        result = torch.empty((num_tokens, hidden_size), dtype=torch.float32, device=hidden_states.device)

        # Initialize result to zeros using Triton (avoid torch.ones)
        fill_zeros_triton[(1,)](result, hidden_size, result.stride(1), BLOCK=128)

        # Loop over tokens and selected experts
        # Note: selected_experts is [num_tokens, num_experts_per_tok], dtype int64 (expert indices)
        # routing_weights is [num_tokens, num_experts_per_tok], dtype bfloat16
        for t in range(num_tokens):
            selected = selected_experts[t]  # [num_experts_per_tok]
            for j in range(num_experts_per_tok):
                e = int(selected[j].item())  # expert index

                # hidden_state for this token: [hidden_size]
                x = hidden_states[t]  # bfloat16, 1D

                # 1) Compute gate_out = hidden_states[t] @ expert_gate_weights[e] -> [M]
                w_gate = expert_gate_weights[e]  # [H, M], bfloat16
                gate_out = torch.empty((M,), dtype=torch.float32, device=hidden_states.device)
                grid_gate = (1,)
                bmm_row_triton[grid_gate](
                    x, w_gate, gate_out,
                    H, M,
                    x.stride(0), w_gate.stride(0), w_gate.stride(1), gate_out.stride(0),
                    BLOCK_M=128, BLOCK_H=64
                )

                # 2) Compute up_out = hidden_states[t] @ expert_up_weights[e] -> [M]
                w_up = expert_up_weights[e]  # [H, M], bfloat16
                up_out = torch.empty((M,), dtype=torch.float32, device=hidden_states.device)
                grid_up = (1,)
                bmm_row_triton[grid_up](
                    x, w_up, up_out,
                    H, M,
                    x.stride(0), w_up.stride(0), w_up.stride(1), up_out.stride(0),
                    BLOCK_M=128, BLOCK_H=64
                )

                # 3) Activation: activated = SiLU(gate_out) * up_out
                activated = torch.empty((M,), dtype=torch.float32, device=hidden_states.device)
                silu_mul_triton[(triton.cdiv(M, 128),)](
                    gate_out, up_out, activated,
                    M, gate_out.stride(0), up_out.stride(0), activated.stride(0),
                    BLOCK=128
                )

                # 4) Compute final_out = activated @ expert_down_weights[e] -> [hidden_size]
                w_down = expert_down_weights[e]  # [M, hidden_size], bfloat16
                final_out = torch.empty((hidden_size,), dtype=torch.float32, device=hidden_states.device)
                grid_down = (triton.cdiv(hidden_size, 128),)
                bmm_row_triton[grid_down](
                    activated, w_down, final_out,
                    M, hidden_size,
                    activated.stride(0), w_down.stride(0), w_down.stride(1), final_out.stride(0),
                    BLOCK_M=128, BLOCK_H=64
                )

                # 5) Accumulate: result[t] += routing_weights[t, e] * final_out
                # routing_weights[t, j] is bfloat16; load as scalar and cast to float32 for compute
                w_scalar = routing_weights[t, j]
                # Create a 1-element device tensor with the scalar value (no torch in host)
                # Triton will see this tensor on device; atomic_accum requires a pointer to the scalar.
                w_tensor = w_scalar.reshape(1).to(torch.float32)  # 1-element tensor on device
                atomic_accum_triton[(1,)](
                    final_out, w_tensor, result[t],
                    hidden_size,
                    final_out.stride(0), result[t].stride(0),
                    BLOCK=128
                )

        # Cast result to bfloat16 to match original model's output dtype
        result = result.to(torch.bfloat16)
        return result
```
