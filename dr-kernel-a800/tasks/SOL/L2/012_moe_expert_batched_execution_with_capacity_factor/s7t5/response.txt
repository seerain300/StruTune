def bincount_kernel(keys_ptr, counts_ptr, T: tl.constexpr, NUM_EXPERTS: tl.constexpr):
    e = tl.program_id(0)
    total = tl.zeros((), dtype=tl.int32)
    for t in range(0, T):
        key = tl.load(keys_ptr + t)
        if key == e:
            total += 1
    tl.store(counts_ptr + e, total)


# Kernel 3: Exclusive prefix sum (cumsum) of counts to get starts per expert. One program computes the scan.
@triton.jit
def prefix_sum_starts_kernel(counts_ptr, starts_ptr, NUM_EXPERTS: tl.constexpr):
    acc = tl.zeros((), dtype=tl.int32)
    for e in range(0, NUM_EXPERTS):
        cnt = tl.load(counts_ptr + e)
        acc += cnt
        tl.store(starts_ptr + e, acc - cnt)


# Kernel 4: Compute within-group positions: pos = global_sorted_index - starts[sorted_expert].
@triton.jit
def compute_positions_kernel(sorted_keys_ptr, starts_ptr, pos_ptr, T: tl.constexpr):
    idx = tl.program_id(0)
    if idx >= T:
        return
    key = tl.load(sorted_keys_ptr + idx)
    start = tl.load(starts_ptr + key)
    tl.store(pos_ptr + idx, idx - start)


# Kernel 5: Build expert_inputs by filling hidden_states rows at (e, pos). We scatter hidden_states[tok] into expert_inputs[e, pos, :].
# This kernel is launched with grid (num_experts, capacity). Each program handles one (e, pos) and checks validity via a separate mask.
# For simplicity, we pass a validity mask and write only when valid.
@triton.jit
def fill_expert_inputs_kernel(expert_inputs_ptr, hidden_states_ptr,
                               v_exp_ptr, v_pos_ptr, v_tok_ptr, valid_ptr,
                               num_experts: tl.constexpr, capacity: tl.constexpr,
                               hidden_size: tl.constexpr):
    e = tl.program_id(0)
    pos = tl.program_id(1)
    valid = tl.load(valid_ptr + e * capacity + pos)  # int32 flag 0/1
    if valid != 0:
        # Load tok and get hidden vector
        tok = tl.load(v_tok_ptr + e * capacity + pos)  # int64 token id
        # Compute base pointer offset for expert_inputs[e, pos, :]
        base = e * capacity * hidden_size + pos * hidden_size
        # Copy hidden_states[tok, :] into expert_inputs[e, pos, :]
        # Loop over hidden_size
        for j in range(0, hidden_size):
            val = tl.load(hidden_states_ptr + tok * hidden_size + j)
            tl.store(expert_inputs_ptr + base + j, val)


# Kernel 6: gate_out = expert_inputs @ expert_gate_weights for a fixed (e, n). Tiled over K (hidden_size), produces [J] vector.
@triton.jit
def bmm_gate_kernel(
    A_ptr,  # expert_inputs flattened as [NUM_EXPERTS * capacity * hidden_size]
    B_ptr,  # expert_gate_weights[num_experts, hidden_size, intermediate_size]
    C_ptr,  # gate_out[num_experts, capacity, intermediate_size] flattened as [NUM_EXPERTS * capacity * J]
    NUM_EXPERTS: tl.constexpr, capacity: tl.constexpr, hidden_size: tl.constexpr, intermediate_size: tl.constexpr,
    stride_A_m, stride_A_k,
    stride_B_e, stride_B_k, stride_B_j,
    stride_C_e, stride_C_n, stride_C_j,
    BLOCK_K: tl.constexpr, BLOCK_J: tl.constexpr,
):
    e = tl.program_id(0)
    n = tl.program_id(1)
    # Base offsets
    m_offset = e * capacity + n
    base_a = m_offset * hidden_size
    base_c = e * capacity * intermediate_size + n * intermediate_size

    acc = tl.zeros((BLOCK_J,), dtype=tl.bfloat16)

    for k0 in range(0, hidden_size, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < hidden_size
        # Load A[m, k] = expert_inputs[e, n, k]
        a_ptrs = A_ptr + base_a + k_idx * stride_A_k
        a = tl.load(a_ptrs, mask=mask_k, other=0.0)
        # Load B[e, k, j] tile
        b_ptrs = B_ptr + e * stride_B_e + k_idx[:, None] * stride_B_k + tl.arange(0, BLOCK_J)[None, :] * stride_B_j
        b = tl.load(b_ptrs, mask=(mask_k[:, None]), other=0.0)
        # Accumulate
        acc += tl.sum(b * a[:, None], axis=0)

    # Store C[e, n, :]
    c_ptrs = C_ptr + base_c + tl.arange(0, BLOCK_J) * stride_C_j
    tl.store(c_ptrs, acc, mask=(tl.arange(0, BLOCK_J) < intermediate_size))
