# Kernel 0: Flatten selected_experts into int32 (input int64 -> output int32)
@triton.jit
def flatten_experts_kernel(
    src_ptr,           # *int64, shape [num_tokens, num_experts_per_tok]
    dst_exp_ptr,       # *int32, shape [E]
    num_tokens: tl.constexpr,
    num_experts_per_tok: tl.constexpr,
    E: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < E
    val = tl.load(src_ptr + offsets, mask=mask, other=0)  # int64
    val = val.to(tl.int32)
    tl.store(dst_exp_ptr + offsets, val, mask=mask)


# Kernel 1: Odd-even stable sort by expert id (permute original positions: sorted_token_ids and sorted_exp, sorted_weights)
@triton.jit
def odd_even_stable_sort_by_exp_kernel(
    exp_ptr,                 # *int32, length E
    wt_ptr,                  # *bf16,  length E
    sorted_exp_ptr,          # *int32, length E (output)

    sorted_wt_ptr,           # *bf16,  length E (output)
    E: tl.constexpr,
    BLOCK: tl.constexpr,
):

    # Each phase iterates over disjoint neighbor pairs to make them sorted in ascending expert id.
    # We keep token indices unchanged for ties in expert id by only swapping when expert id differs.
    for phase in range(0, E):

                    b = exp_ptr[i + 1]

                        tl.store(sorted_wt_ptr + i, wb)
                        tl.store(sorted_wt_ptr + (i + 1), wa)
                    else:
                        tl.store(sorted_exp_ptr + i, a)
                        tl.store(sorted_exp_ptr + (i + 1), b)
                        tl.store(sorted_tok_ptr + i, i // num_experts_per_tok)
                        tl.store(sorted_tok_ptr + (i + 1), (i + 1) // num_experts_per_tok)
                        tl.store(sorted_wt_ptr + i, wa)
                        tl.store(sorted_wt_ptr + (i + 1), wb)
        else:
            # Odd phase: pairs (1,2), (3,4), ...

                i = start
                if i + 1 < E:
                    a = exp_ptr[i]
                    b = exp_ptr[i + 1]
                    wa = wt_ptr[i]
                    wb = wt_ptr[i + 1]
                    if a > b or (a == b and wa > wb):
                        tl.store(sorted_exp_ptr + i, b)
                        tl.store(sorted_exp_ptr + (i + 1), a)
                        tl.store(sorted_tok_ptr + i, i // num_experts_per_tok)
                        tl.store(sorted_tok_ptr + (i + 1), (i + 1) // num_experts_per_tok)
                        tl.store(sorted_wt_ptr + i, wb)
                        tl.store(sorted_wt_ptr + (i + 1), wa)
                    else:
                        tl.store(sorted_exp_ptr + i, a)
                        tl.store(sorted_exp_ptr + (i + 1), b)
                        tl.store(sorted_tok_ptr + i, i // num_experts_per_tok)
                        tl.store(sorted_tok_ptr + (i + 1), (i + 1) // num_experts_per_tok)
                        tl.store(sorted_wt_ptr + i, wa)
                        tl.store(sorted_wt_ptr + (i + 1), wb)


# Kernel 2: Bincount of flattened expert ids (int32 -> int32 counts per expert)
@triton.jit
def bincount_experts_kernel(
    exp_ptr,            # *int32, length E
    counts_ptr,         # *int32, length num_experts
    num_experts: tl.constexpr,
    E: tl.constexpr,
    BLOCK: tl.constexpr,
):
    e = tl.program_id(axis=0)  # expert index
    if e >= num_experts:
        return
    total = tl.zeros((), dtype=tl.int32)
    for start in range(0, E, BLOCK):
        offsets = start + tl.arange(0, BLOCK)
        mask = offsets < E
        vals = tl.load(exp_ptr + offsets, mask=mask, other=0)  # int32
        # Sum occurrences of 'e' in this chunk
        total += tl.sum(((vals == e) & mask).to(tl.int32))
    tl.store(counts_ptr + e, total)


# Kernel 3: Cumsum of counts to compute per-expert starts (int32 -> int32 starts[expert])
@triton.jit
def cumsum_starts_kernel(
    counts_ptr,     # *int32, length num_experts
    starts_ptr,     # *int32, length num_experts
    num_experts: tl.constexpr,
):
    # Sequentially compute inclusive scan (prefix sum)
    running = tl.zeros((), dtype=tl.int32)
    for e in range(0, num_experts):
        c = tl.load(counts_ptr + e)
        running += c
        tl.store(starts_ptr + e, running)


# Kernel 4: Compute within_pos for each flattened assignment: within_pos = i - starts[expert]
@triton.jit
def compute_within_pos_kernel(
    sorted_exp_ptr,   # *int32, length E
    starts_ptr,       # *int32, length num_experts
    within_ptr,       # *int32, length E
    num_experts: tl.constexpr,
    E: tl.constexpr,
):
    for i in range(0, E):
        e = tl.load(sorted_exp_ptr + i)  # int32
        start = tl.load(starts_ptr + e)  # int32
        tl.store(within_ptr + i, i - start)


# Kernel 5: Build validity mask: valid if within_pos < capacity (int32 -> int32 mask)
@triton.jit
def build_valid_mask_kernel(
    within_ptr,            # *int32, length E
    capacity,              # int32 scalar
    valid_mask_ptr,        # *int32, length E
    E: tl.constexpr,
):
    for i in range(0, E):
        w = tl.load(within_ptr + i)
        # 1 if valid, else 0
        v = tl.where(w < capacity, 1, 0)
        tl.store(valid_mask_ptr + i, v)
