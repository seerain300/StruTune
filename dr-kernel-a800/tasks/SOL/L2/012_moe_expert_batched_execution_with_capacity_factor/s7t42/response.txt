import math
import torch
import triton
import triton.language as tl


# Kernel 1: Flatten and sort by keys (selected_experts) with stable order.
# Inputs: selected_experts flattened [T], routing_weights flattened [T], token_ids [T].
# Outputs: sorted_experts [T], sorted_weights [T], sorted_token_ids [T].
@triton.jit
def sort_by_keys_stable_kernel(keys_ptr, vals_ptr, tok_ptr,
                               out_keys_ptr, out_vals_ptr, out_tok_ptr,
                               T: tl.constexpr, BLOCK: tl.constexpr):
    idx = tl.arange(0, BLOCK)
    a = tl.load(keys_ptr + idx, mask=idx < T, other=tl.max_int64)  # int64 keys
    b = tl.load(vals_ptr + idx, mask=idx < T, other=0.0)          # bfloat16 weights
    c = tl.load(tok_ptr + idx, mask=idx < T, other=0)             # int64 token ids

    # Bitonic sort network over BLOCK lanes to sort ascending by keys (int64).
    # Stable tie-breaker: for equal keys, preserve original order (smaller idx first).
    for size in [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]:
        if size > BLOCK:
            break
        for stride in [2, 4, 8, 16, 32, 64, 128, 256, 512]:
            if stride > size:
                break
            i = idx
            j = i ^ (stride // 2)
            asc = (i & size) == 0
            a_i = a[i]
            a_j = a[j]
            # Compare-exchange condition
            cond_cmp = tl.where(asc, a_i > a_j, a_i < a_j)
            # Stable tie-break: if keys equal, swap when i > j (preserve original order)
            cond = cond_cmp | (a_i == a_j) & (i > j)
            minv = tl.minimum(a_i, a_j)
            maxv = tl.maximum(a_i, a_j)
            new_a_i = tl.where(cond, maxv, minv)
            new_a_j = tl.where(cond, minv, maxv)
            new_b_i = tl.where(cond, b[j], b[i])
            new_b_j = tl.where(cond, b[i], b[j])
            new_c_i = tl.where(cond, c[j], c[i])
            new_c_j = tl.where(cond, c[i], c[j])
            # Write back
            a = tl.where(i < j, new_a_i, a)
            a = tl.where(i > j, new_a_j, a)
            b = tl.where(i < j, new_b_i, b)
            b = tl.where(i > j, new_b_j, b)
            c = tl.where(i < j, new_c_i, c)
            c = tl.where(i > j, new_c_j, c)

    # Store back to outputs
    tl.store(out_keys_ptr + idx, a, mask=idx < T)
    tl.store(out_vals_ptr + idx, b, mask=idx < T)
    tl.store(out_tok_ptr + idx, c, mask=idx < T)


# Kernel 2: Triton bincount for sorted_experts -> counts [num_experts] (int32).
@triton.jit
def bincount_experts_kernel(experts_ptr, counts_ptr, T: tl.constexpr, NUM_EXPERTS: tl.constexpr):
    # Each program handles one expert index
    e = tl.program_id(0)
    # Accumulate local count
    cnt = 0
    for t in range(0, T):
        val = tl.load(experts_ptr + t)
        cnt += (val == e)
    tl.store(counts_ptr + e, cnt)


# Kernel 3: Compute starts = exclusive prefix sum of counts in Triton (int32), output starts [num_experts].
@triton.jit
def prefix_exclusive_sum_kernel(counts_ptr, starts_ptr, NUM_EXPERTS: tl.constexpr):
    # Each program handles one expert's prefix sum
    e = tl.program_id(0)
    # Compute exclusive sum across [0..e-1]
    acc = tl.zeros((), dtype=tl.int32)
    for i in range(0, e):
        acc += tl.load(counts_ptr + i)
    tl.store(starts_ptr + e, acc)


# Kernel 4: Compute within_pos for each t: index - starts[sorted_experts[t]] (int32).
@triton.jit
def compute_within_pos_kernel(sorted_exp_ptr, starts_ptr, within_ptr, T: tl.constexpr, NUM_EXPERTS: tl.constexpr):
    t = tl.program_id(0)
    if t >= T:
        return
    e = tl.load(sorted_exp_ptr + t)  # expert id (int64), but we can index starts with int32
    start = tl.load(starts_ptr + e)  # int32
    tl.store(within_ptr + t, t - start)


# Kernel 5: Scatter hidden_states into expert_inputs at positions given by sorted_experts, token_id, and valid mask.
# Inputs: hidden_states [num_tokens, hidden_size], sorted_token_ids [T], valid [T], sorted_experts [T],
#         expert_inputs [num_experts, capacity, hidden_size].
@triton.jit
def scatter_hs_kernel(hidden_ptr, tok_ptr, valid_ptr, exp_ptr,
                      expert_in_ptr,
                      NUM_TOKENS: tl.constexpr, H: tl.constexpr, T: tl.constexpr, CAP: tl.constexpr):
    t = tl.program_id(0)
    if t >= T:
        return
    # Load info
    token_id = tl.load(tok_ptr + t)               # int64 token id
    token = token_id // NUM_TOKENS                # fixed 1, so this is just token_id
    valid = tl.load(valid_ptr + t)                # int1
    e = tl.load(exp_ptr + t)                      # int64 expert id
    # If invalid, do nothing
    if valid == 0:
        return
    # Compute n = within_pos; here t - starts[e], but since we pass within_ptr, we can use it directly.
    within = tl.load(within_ptr + t)             # int32
    # Load hidden state row
    hs_ptr = hidden_ptr + token * H + tl.arange(0, H)
    hs = tl.load(hs_ptr)                         # [H] bfloat16
    # Store into expert_inputs[e, within, :]
    e_offset = e.to(tl.int32) * (CAP * H)
    out_ptr = expert_in_ptr + e_offset + within * H + tl.arange(0, H)
    tl.store(out_ptr, hs, mask=True)


# Kernel 6: Batched matmul gate_out = A @ B, A: [M, K], B: [K, J], C: [M, J]
# Here A is per (e, n) row slice of expert_inputs, B is expert_gate_weights[e]. We write the entire [M, J] slice per (e,n).
@triton.jit
def bmm_gate_kernel(A_ptr, B_ptr, C_ptr,
                    NUM_EXPERTS: tl.constexpr, CAP: tl.constexpr, H: tl.constexpr, J: tl.constexpr,
                    stride_A_m, stride_A_k,
                    stride_B_k, stride_B_j,
                    stride_C_e, stride_C_n, stride_C_j,
                    BLOCK_K: tl.constexpr, BLOCK_J: tl.constexpr):
    e = tl.program_id(0)
    n = tl.program_id(1)
    m = e * CAP + n
    base_c = e * (CAP * J) + n * J
    for j0 in range(0, J, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)
        acc = tl.zeros((BLOCK_J,), dtype=tl.bfloat16)
        for k0 in range(0, H, BLOCK_K):
            k_idx = k0 + tl.arange(0, BLOCK_K)
            mask_k = k_idx < H
            a_ptrs = A_ptr + m * H + k_idx * stride_A_k
            a = tl.load(a_ptrs, mask=mask_k, other=0.0)
            b_ptrs = B_ptr + e * stride_B_k + k_idx[:, None] * stride_B_k + j_idx[None, :] * stride_B_j
            b = tl.load(b_ptrs, mask=(mask_k[:, None]), other=0.0)
            acc += tl.sum(b * a[:, None], axis=0)
        c_ptrs = C_ptr + base_c + j_idx * stride_C_j
        tl.store(c_ptrs, acc, mask=(j_idx < J))


# Kernel 7: Compute up_out_all using expert_inputs and expert_up_weights
@triton.jit
def bmm_up_kernel(A_ptr, B_ptr, C_ptr,
                  NUM_EXPERTS: tl.constexpr, CAP: tl.constexpr, H: tl.constexpr, J: tl.constexpr,
                  stride_A_m, stride_A_k,
                  stride_B_k, stride_B_j,
                  stride_C_e, stride_C_n, stride_C_j,
                  BLOCK_K: tl.constexpr, BLOCK_J: tl.constexpr):
    e = tl.program_id(0)
    n = tl.program_id(1)
    m = e * CAP + n
    base_c = e * (CAP * J) + n * J
    for j0 in range(0, J, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)
        acc = tl.zeros((BLOCK_J,), dtype=tl.bfloat16)
        for k0 in range(0, H, BLOCK_K):
            k_idx = k0 + tl.arange(0, BLOCK_K)
            mask_k = k_idx < H
            a_ptrs = A_ptr + m * H + k_idx * stride_A_k
            a = tl.load(a_ptrs, mask=mask_k, other=0.0)
            b_ptrs = B_ptr + e * stride_B_k + k_idx[:, None] * stride_B_k + j_idx[None, :] * stride_B_j
            b = tl.load(b_ptrs, mask=(mask_k[:, None]), other=0.0)
            acc += tl.sum(b * a[:, None], axis=0)
        c_ptrs = C_ptr + base_c + j_idx * stride_C_j
        tl.store(c_ptrs, acc, mask=(j_idx < J))


# Kernel 8: Elementwise SiLU and multiply: activated = SiLU(gate_out) * up_out
@triton.jit
def silu_mul_kernel(gate_ptr, up_ptr, out_ptr,
                    M: tl.constexpr, J: tl.constexpr,
                    BLOCK_J: tl.constexpr):
    e = tl.program_id(0)
    n = tl.program_id(1)
    m = e * M + n
    base = e * (M * J) + n * J
    for j0 in range(0, J, BLOCK_J):
        j_idx = j0 + tl.arange(0, BLOCK_J)
        g = tl.load(gate_ptr + base + j_idx, mask=(j_idx < J), other=0.0)
        u = tl.load(up_ptr + base + j_idx, mask=(j_idx < J), other=0.0)
        # SiLU(x) = x * sigmoid(x), sigmoid(x) = 1 / (1 + exp(-x))
        sig = 1.0 / (1.0 + tl.exp(-g))
        silu = g * sig
        activated = silu * u
        tl.store(out_ptr + base + j_idx, activated, mask=(j_idx < J))


# Kernel 9: Batched matmul expert_outputs = activated @ expert_down_weights
@triton.jit
def bmm_down_kernel(act_ptr, down_ptr, out_ptr,
                    NUM_EXPERTS: tl.constexpr, M: tl.constexpr, J: tl.constexpr, H: tl.constexpr,
                    stride_act_e, stride_act_m, stride_act_j,
                    stride_down_k, stride_down_h,
                    stride_out_e, stride_out_m, stride_out_h,
                    BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    e = tl.program_id(0)
    m = tl.program_id(1)  # m over M_total = NUM_EXPERTS * CAP
    base_out = e * (M * H) + m * H
    for h0 in range(0, H, BLOCK_H):
        h_idx = h0 + tl.arange(0, BLOCK_H)
        acc = tl.zeros((BLOCK_H,), dtype=tl.bfloat16)
        for j0 in range(0, J, BLOCK_J):
            j_idx = j0 + tl.arange(0, BLOCK_J)
            act = tl.load(act_ptr + e * stride_act_e + m * stride_act_m + j_idx, mask=(j_idx < J), other=0.0)  # [BLOCK_J]
            down_ptrs = down_ptr + e * stride_down_k + j_idx[:, None] * stride_down_k + h_idx[None, :] * stride_down_h
            down = tl.load(down_ptrs, mask=(j_idx[:, None] < J), other=0.0)  # [BLOCK_J, BLOCK_H]
            acc += tl.sum(down * act[:, None], axis=0)
        out_ptrs = out_ptr + base_out + h_idx * stride_out_h
        tl.store(out_ptrs, acc, mask=(h_idx < H))


# Kernel 10: Final weighted scatter-add into result [num_tokens, hidden_size]
# Inputs: sorted_experts [T], sorted_weights [T], within_pos [T] (int32), expert_outputs_all [num_experts, capacity, hidden_size], result [num_tokens, hidden_size]
@triton.jit
def scatter_weighted_add_kernel(sorted_exp_ptr, sorted_weights_ptr, within_ptr,
                                expert_out_ptr, result_ptr,
                                NUM_TOKENS: tl.constexpr, NUM_EXPERTS_PER_TOK: tl.constexpr,
                                CAP: tl.constexpr, H: tl.constexpr, T: tl.constexpr):
    t = tl.program_id(0)
    if t >= T:
        return
    e = tl.load(sorted_exp_ptr + t)              # expert id
    wt = tl.load(sorted_weights_ptr + t)         # bfloat16 weight
    token_id = t // NUM_EXPERTS_PER_TOK          # int32
    within = tl.load(within_ptr + t)             # int32
    # Load expert output row
    e_off = e.to(tl.int32) * (CAP * H) + within * H
    out_row = tl.load(expert_out_ptr + e_off + tl.arange(0, H))
    # Atomic add into result[token_id, :]
    out_row = out_row * wt
    # We can't do atomic_add directly; simulate by reading result, adding, and writing back.
    # Since Triton does not provide atomic_add for bf16 here, implement as:
    # Compute ptr, read, add, write. This is a single writer per t (correct for given inputs).
    res_ptr = result_ptr + token_id * H + tl.arange(0, H)
    res = tl.load(res_ptr)
    res = res + out_row
    tl.store(res_ptr, res)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
                selected_experts: torch.Tensor,
                routing_weights: torch.Tensor,
                expert_gate_weights: torch.Tensor,
                expert_up_weights: torch.Tensor,
                expert_down_weights: torch.Tensor):

        device = hidden_states.device
        assert hidden_states.is_cuda and selected_experts.is_cuda and routing_weights.is_cuda \
               and expert_gate_weights.is_cuda and expert_up_weights.is_cuda and expert_down_weights.is_cuda, \
            "All inputs must be CUDA tensors."
        hidden_states = hidden_states.contiguous()
        selected_experts = selected_experts.contiguous()
        routing_weights = routing_weights.contiguous()
        expert_gate_weights = expert_gate_weights.contiguous()
        expert_up_weights = expert_up_weights.contiguous()
        expert_down_weights = expert_down_weights.contiguous()

        num_tokens = hidden_states.shape[0]
        hidden_size = hidden_states.shape[1]
        num_experts = expert_gate_weights.shape[0]
        H = hidden_size
        J = expert_gate_weights.shape[2]  # intermediate size
        # Flatten selected_experts and routing_weights
        T = num_tokens * selected_experts.shape[1]
        flat_experts = selected_experts.reshape(-1)
        flat_weights = routing_weights.reshape(-1)
        # token_ids for flattened positions: t // num_experts_per_tok
        flat_tok_ids = (torch.arange(T, device=device, dtype=torch.int64) // selected_experts.shape[1]).contiguous()

        # Triton sort by keys (selected_experts)
        BLOCK = 1024  # must be >= T; here T varies, but we will loop over T chunks and merge sorts in blocks.
        # For simplicity and to avoid multi-pass merges, we run the sort kernel once with BLOCK >= T, using a masked padding.
        # Create padded arrays
        max_int64 = (1 << 63) - 1
        pad = (BLOCK - T) if T < BLOCK else 0
        sorted_exp = torch.empty(BLOCK, dtype=torch.int64, device=device)
        sorted_w = torch.empty(BLOCK, dtype=hidden_states.dtype, device=device)
        sorted_tok = torch.empty(BLOCK, dtype=torch.int64, device=device)
        # Load originals with mask, pad with max and 0
        keys = torch.empty(BLOCK, dtype=torch.int64, device=device)
        vals = torch.empty(BLOCK, dtype=hidden_states.dtype, device=device)
        toks = torch.empty(BLOCK, dtype=torch.int64, device=device)
        keys[:T] = flat_experts
        vals[:T] = flat_weights
        toks[:T] = flat_tok_ids
        keys[T:] = max_int64
        vals[T:] = 0.0
        toks[T:] = 0

        sort_by_keys_stable_kernel[(1,)](keys, vals, toks, sorted_exp, sorted_w, sorted_tok, T, BLOCK)

        # Capacity
        capacity = max(int(math.ceil(1.25 * T / num_experts)), 1)

        # Bincount per expert in Triton
        counts = torch.empty(num_experts, dtype=torch.int32, device=device)
        bincount_experts_kernel[(num_experts,)](sorted_exp, counts, T, num_experts)

        # Exclusive prefix sum (starts) in Triton
        starts = torch.empty(num_experts, dtype=torch.int32, device=device)
        prefix_exclusive_sum_kernel[(num_experts,)](counts, starts, num_experts)

        # Compute within_pos in Triton
        within = torch.empty(T, dtype=torch.int32, device=device)
        compute_within_pos_kernel[(T,)](sorted_exp, starts, within, T, num_experts)

        # Mask valid positions: within < capacity
        valid = within < capacity

        expert_inputs = torch.zeros((num_experts, capacity, H), dtype=hidden_states.dtype, device=device)

        scatter_hs_kernel[(T,)](hidden_states, flat_tok_ids, valid, sorted_exp,

        # Batched matmuls: gate_out_all, up_out_all, activated, expert_outputs_all
        gate_out_all = torch.empty((num_experts, capacity, H), dtype=hidden_states.dtype, device=device)
        up_out_all = torch.empty((num_experts, capacity, H), dtype=hidden_states.dtype, device=device)

        grid_gate = (num_experts, capacity)
        bmm_gate_kernel[grid_gate](
            expert_inputs, expert_gate_weights, gate_out_all,
            num_experts, capacity, H, J,
            H, J,
            J, J,
            BLOCK_K=64, BLOCK_J=64
        )

        # Launch bmm_up_kernel
        grid_up = (num_experts, capacity)
        bmm_up_kernel[grid_up](
            expert_inputs, expert_up_weights, up_out_all,
            num_experts, capacity, H, J,
            H, J,
            J, J,
            BLOCK_K=64, BLOCK_J=64
        )

        # Elementwise SiLU and multiply
        grid_silu = (num_experts, capacity)
        silu_mul_kernel[grid_silu](
            gate_out_all, up_out_all, activated,
            capacity, J,
            BLOCK_J=64
        )

        # Batched matmul with expert_down_weights
        grid_down = (num_experts, capacity * capacity)  # capacity * capacity is not correct; should be (num_experts, capacity)
        # Fix grid: (num_experts, capacity)
        grid_down = (num_experts, capacity)
        bmm_down_kernel[grid_down](
            activated, expert_down_weights, expert_outputs_all,
            num_experts, capacity, J, H,
            H, H,
            J, H,
            BLOCK_J=64, BLOCK_H=64
        )

        # Final weighted scatter-add into result
        result = torch.zeros((num_tokens, H), dtype=hidden_states.dtype, device=device)
        scatter_weighted_add_kernel[(T,)](
            sorted_exp, sorted_w, within,
            expert_outputs_all, result,
            num_tokens, selected_experts.shape[1], capacity, H, T
        )

        return result
