import torch
import triton
import triton.language as tl


# Kernel 1: Stable sort of flattened (expert_id, token_id) by expert_id (tie-break by token_id).
# Implement odd-even transposition sort in a single Triton program using scratch indices.
@triton.jit
def _stable_sort_by_expert_id(
    exp_ptr,        # int32* [P]
    tok_ptr,        # int32* [P]
    idx_ptr,        # int32* [P] output indices after sort
    P: tl.constexpr,
):
    # Initialize idx_ptr to [0..P-1]
    for i in range(P):
        tl.store(idx_ptr + i, i)
    # Perform odd-even transposition sort for P phases
    for phase in range(P):
        if (phase % 2) == 0:
            start = 0
            step = 2
        else:
            start = 1
            step = 2
        for i in range(start, P - 1, step):
            a = tl.load(exp_ptr + idx_ptr[i])
            b = tl.load(exp_ptr + idx_ptr[i + 1])
            t = tl.load(tok_ptr + idx_ptr[i])
            u = tl.load(tok_ptr + idx_ptr[i + 1])
            swap = (a > b) | ((a == b) & (t > u))
            tmp = tl.load(idx_ptr + i)
            tl.store(idx_ptr + i, tl.load(idx_ptr + i + 1), mask=swap)
            tl.store(idx_ptr + i + 1, tmp, mask=swap)


# Kernel 2: Bincount of expert_ids to get counts per expert (int32 counts)
@triton.jit
def _bincount_experts(
    exp_ptr,        # int32* [P]
    counts_ptr,     # int32* [E]
    P: tl.constexpr,
    E: tl.constexpr,
):
    for e in range(E):
        count = 0
        for i in range(P):
            if tl.load(exp_ptr + i) == e:
                count += 1
        tl.store(counts_ptr + e, count)


# Kernel 3: Inclusive cumsum of counts to get starts (global offsets per expert) into starts_ptr [E]
@triton.jit
def _cumsum_starts(
    counts_ptr,     # int32* [E]
    starts_ptr,     # int32* [E]
    E: tl.constexpr,
):
    tl.store(starts_ptr + 0, tl.load(counts_ptr + 0))
    for e in range(1, E):
        tl.store(starts_ptr + e, tl.load(counts_ptr + e) + tl.load(starts_ptr + e - 1))


# Kernel 4: Compute within_pos and valid masks for each i, and produce v_exp, v_tok, v_weight, v_pos, valid, within_pos
@triton.jit
def _compute_valid_positions(
    idx_ptr,        # int32* [P]
    exp_ptr,        # int32* [P]
    tok_ptr,        # int32* [P]
    wt_ptr,         # float32* [P]
    v_exp_ptr,      # int32* [P]
    v_tok_ptr,      # int32* [P]
    v_weight_ptr,   # float32* [P]
    v_pos_ptr,      # int32* [P]
    valid_ptr,      # int8* [P]
    within_ptr,      # int32* [P]
    starts_ptr,      # int32* [E]
    capacity,        # int32 scalar
    P: tl.constexpr,
    E: tl.constexpr,
):
    for i in range(P):
        idx = idx_ptr[i]
        e = tl.load(exp_ptr + idx)
        starts_e = tl.load(starts_ptr + e)
        within_i = idx - starts_e
        tl.store(within_ptr + i, within_i)
        valid = within_i < capacity
        tl.store(valid_ptr + i, valid.to(tl.int8))
        tl.store(v_exp_ptr + i, e)
        tl.store(v_tok_ptr + i, tl.load(tok_ptr + idx))
        tl.store(v_weight_ptr + i, tl.load(wt_ptr + idx))
        tl.store(v_pos_ptr + i, within_i)


# Kernel 5: Scatter hidden_states rows into expert_inputs[e, pos, :] for valid positions
# Inputs: hidden_ptr [T, hidden], v_tok_ptr [P], v_exp_ptr [P], v_pos_ptr [P], valid_ptr [P],
#         exp_inputs_ptr [E, capacity, hidden], T, hidden_dim, capacity, P
@triton.jit
def _scatter_hidden(
    hidden_ptr,     # float32* [T, hidden]
    v_tok_ptr,      # int32* [P]
    v_exp_ptr,      # int32* [P]
    v_pos_ptr,      # int32* [P]
    valid_ptr,      # int8* [P]
    exp_inputs_ptr, # float32* [E, capacity, hidden]
    T: tl.constexpr,
    hidden_dim: tl.constexpr,
    capacity: tl.constexpr,
    P: tl.constexpr,
):
    for i in range(P):
        if tl.load(valid_ptr + i) != 0:
            tok = tl.load(v_tok_ptr + i)
            e = tl.load(v_exp_ptr + i)
            pos = tl.load(v_pos_ptr + i)
            # Copy row hidden_states[tok, :] to exp_inputs[e, pos, :]
            for j in range(hidden_dim):
                val = tl.load(hidden_ptr + tok * hidden_dim + j)
                tl.store(exp_inputs_ptr + e * capacity + pos * hidden_dim + j, val)


# Kernel 6: Batched GEMM for gate_out = expert_inputs @ expert_gate_weights -> [E, capacity, N_gate]
# Accumulate across hidden_dim (K) and N_gate (N). One Triton program per row (e, pos), tiles across N.
@triton.jit
def _batched_gemm_gate(
    A_ptr,           # float32* [E*capacity, hidden]
    B_ptr,           # float32* [E, hidden, N_gate]
    C_ptr,           # float32* [E*capacity, N_gate]
    E: tl.constexpr,
    capacity: tl.constexpr,
    hidden_dim: tl.constexpr,
    N_gate: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)  # row id in [0, E*capacity)
    e = pid // capacity
    pos = pid % capacity
    if e >= E:
        return
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    # Loop over K dimension tiles
    for k_start in range(0, hidden_dim, BLOCK_K):
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < hidden_dim
        A_row = tl.load(A_ptr + pid * hidden_dim + k_idx, mask=mask_k, other=0.0)
        # Loop over N columns in tiles
        for n_start in range(0, N_gate, BLOCK_N):
            n_idx = n_start + tl.arange(0, BLOCK_N)
            mask_n = n_idx < N_gate
            # B[e, k_idx, n_idx] -> pointers: B_ptr + e*(hidden*N_gate) + k_idx*N_gate + n_idx
            B_ptrs = B_ptr + e * (hidden_dim * N_gate) + k_idx[:, None] * N_gate + n_idx[None, :]
            B_block = tl.load(B_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
            acc += tl.sum(A_row[:, None] * B_block, axis=0)
    # Store acc to C[pid, :]
    tl.store(C_ptr + pid * N_gate + tl.arange(0, BLOCK_N), acc, mask=(tl.arange(0, BLOCK_N) < N_gate))


# Kernel 7: Elementwise SiLU and multiply over rows: out = SiLU(a) * b, where a,gate_out; b,up_out
@triton.jit
def _silu_mul_rows(
    a_ptr,           # float32* [rows, N_gate]
    b_ptr,           # float32* [rows, N_gate]
    out_ptr,         # float32* [rows, N_gate]
    rows: tl.constexpr,
    N_gate: tl.constexpr,
    BLOCK: tl.constexpr,
):
    # 1D grid: each program handles a row
    pid_row = tl.program_id(0)
    for n_start in range(0, N_gate, BLOCK):
        cols = n_start + tl.arange(0, BLOCK)
        mask = cols < N_gate
        a = tl.load(a_ptr + pid_row * N_gate + cols, mask=mask, other=0.0)
        b = tl.load(b_ptr + pid_row * N_gate + cols, mask=mask, other=0.0)
        sig = 1.0 / (1.0 + tl.exp(-a))
        tl.store(out_ptr + pid_row * N_gate + cols, sig * b, mask=mask)


# Kernel 8: Batched GEMM for down: expert_outputs[e, pos, :] = sum_j activated[e, pos, j] * expert_down_weights[e, j, :]
@triton.jit
def _batched_gemm_down(
    activated_ptr,   # float32* [E*capacity, N_gate]
    D_ptr,           # float32* [E, N_gate, hidden]
    C_ptr,           # float32* [E*capacity, hidden]
    E: tl.constexpr,
    capacity: tl.constexpr,
    N_gate: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)  # row id in [0, E*capacity)
    e = pid // capacity
    pos = pid % capacity
    if e >= E:
        return
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)  # acc over hidden_dim
    # Loop over K dimension (N_gate) tiles
    for n_start in range(0, N_gate, BLOCK_N):
        n_idx = n_start + tl.arange(0, BLOCK_N)
        mask_n = n_idx < N_gate
        # activated row [BLOCK_N]
        act_row = tl.load(activated_ptr + pid * N_gate + n_idx, mask=mask_n, other=0.0)
        # Loop over K (hidden_dim) reduction
        for k_start in range(0, hidden_dim, BLOCK_K):
            k_idx = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_idx < hidden_dim
            # D[e, n_idx, k_idx] -> pointers: D_ptr + e*(N_gate*hidden_dim) + n_idx*hidden_dim + k_idx
            D_ptrs = D_ptr + e * (N_gate * hidden_dim) + n_idx[:, None] * hidden_dim + k_idx[None, :]
            D_block = tl.load(D_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
            acc += tl.sum(act_row[:, None] * D_block, axis=0)
    # Store acc to C[pid, :]
    tl.store(C_ptr + pid * hidden_dim + tl.arange(0, BLOCK_N), acc, mask=(tl.arange(0, BLOCK_N) < hidden_dim))


# Kernel 9: Scatter-add: For each i, add out[i, :] * weight[i] into result[tok[i], :].
# Inputs: out_ptr [P, N], v_tok_ptr [P], v_weight_ptr [P], result_ptr [T, N], P, N, BLOCK
@triton.jit
def _scatter_add_row(
    out_ptr,         # float32* [P, N]
    v_tok_ptr,       # int32* [P]
    v_weight_ptr,    # float32* [P]
    result_ptr,      # float32* [T, N]
    P: tl.constexpr,
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)
    col_start = pid_col * BLOCK
    cols = col_start + tl.arange(0, BLOCK)
    mask = cols < N
    out_row = tl.load(out_ptr + pid_row * N + cols, mask=mask, other=0.0)
    weight = tl.load(v_weight_ptr + pid_row)
    tok = tl.load(v_tok_ptr + pid_row)
    tl.store(result_ptr + tok * N + cols, out_row * weight, mask=mask)
