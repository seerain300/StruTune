# Triton kernel: Stable sort of flattened (expert_id, token_id) pairs.
# Inputs:
#   exp_ptr: int32*, length P (flattened selected_experts)
#   tok_ptr: int32*, length P (flattened token_ids)
#   ind_ptr: int32*, length P (scratch / output permutation indices)
# The kernel performs odd-even transposition sort on (exp, tok) and writes the permutation into ind_ptr.
@triton.jit
def _stable_sort_expert_pairs(exp_ptr, tok_ptr, ind_ptr, P: tl.constexpr):
    # We sort pairs (exp_ptr[i], tok_ptr[i]) lexicographically, stably on tok when exp is equal.
    # Implement odd-even transposition sort: P phases.
    # We do not use any torch ops in host; kernel specializes per P.
    for phase in range(P):
        if (phase % 2) == 0:
            # even phase: (0,1), (2,3), ...
            for i in range(0, P, 2):
                a = i
                b = i + 1
                if b >= P:
                    break
                ea = tl.load(exp_ptr + tl.load(ind_ptr + a))
                eb = tl.load(exp_ptr + tl.load(ind_ptr + b))
                ta = tl.load(tok_ptr + tl.load(ind_ptr + a))
                tb = tl.load(tok_ptr + tl.load(ind_ptr + b))
                # compare-swap with tie-break on tok
                swap = (ea > eb) | ((ea == eb) & (ta > tb))
                if swap:
                    tmp = tl.load(ind_ptr + a)
                    tl.store(ind_ptr + a, tl.load(ind_ptr + b))
                    tl.store(ind_ptr + b, tmp)
        else:
            # odd phase: (1,2), (3,4), ...
            for i in range(1, P, 2):
                a = i
                b = i + 1
                if b >= P:
                    break
                ea = tl.load(exp_ptr + tl.load(ind_ptr + a))
                eb = tl.load(exp_ptr + tl.load(ind_ptr + b))
                ta = tl.load(tok_ptr + tl.load(ind_ptr + a))
                tb = tl.load(tok_ptr + tl.load(ind_ptr + b))
                swap = (ea > eb) | ((ea == eb) & (ta > tb))
                if swap:

                    tl.store(ind_ptr + a, tl.load(ind_ptr + b))
                    tl.store(ind_ptr + b, tmp)

# Triton kernel: Scatter hidden states into expert_inputs[e, pos, :] for valid entries.

            pos = i - tl.load(starts_ptr + e)  # starts_ptr is available via kernel argument. For clarity, we pass it.
            # Copy hidden[tok, :] into expert_inputs[e, pos, :]
            # We perform this by iterating over hidden dimension; however, Triton kernels are efficient at vectorized loads/stores; we can load and store vectors.

# Triton kernel: Batched GEMM row for gate_out = hidden_inputs @ expert_gate_weights.
# Inputs:
#   A_ptr: float32*, [E, capacity, K] (expert_inputs)
#   B_ptr: float32*, [E, K, N_gate] (expert_gate_weights reshaped)
#   C_ptr: float32*, [E, capacity, N_gate]
# We launch with grid (E*capacity, cdiv(N_gate, BLOCK_N)) and process rows of C (for each e, pos).
# Note: This kernel accumulates over K in tiles and writes C row. We pass K and N_gate as constexpr for specialization.
@triton.jit
def _gemm_gate_row(A_ptr, B_ptr, C_ptr, E: tl.constexpr, capacity: tl.constexpr, K: tl.constexpr, N_gate: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr):
    pid_row = tl.program_id(0)  # over E*capacity
    pid_col = tl.program_id(1)  # over tiles of N_gate
    e = pid_row // capacity
    pos = pid_row % capacity
    # Compute base pointers
    A_row_base = A_ptr + e * (capacity * K) + pos * K
    B_row_base = B_ptr + e * K  # B is [E, K, N_gate], row base per e
    # We will write C[e, pos, col_start:col_start+BLOCK_N]
    col_start = pid_col * BLOCK_N
    for n in range(col_start, col_start + BLOCK_N, BLOCK_N):
        # Accumulator for this N tile
        acc = tl.zeros([BLOCK_N], dtype=tl.float32)
        for k in range(0, K, BLOCK_K):
            k_offs = k + tl.arange(0, BLOCK_K)
            a_vec = tl.load(A_row_base + k_offs, mask=k_offs < K, other=0.0)  # [BLOCK_K]
            # Load corresponding B block: [BLOCK_K, BLOCK_N]
            b_block = tl.load(B_row_base + k_offs[:, None] * N_gate + (n + tl.arange(0, BLOCK_N))[None, :], mask=(k_offs[:, None] < K) & ((n + tl.arange(0, BLOCK_N))[None, :] < N_gate), other=0.0)

# Triton kernel: Elementwise SiLU and multiply.
# Inputs:

#   up_ptr: float32*, [E*capacity*N_gate]
#   out_ptr: float32*, [E*capacity*N_gate]
# We launch with grid (E*capacity, cdiv(N_gate, BLOCK)) and process rows over N tiles.
@triton.jit
def _silu_mul_kernel(gate_ptr, up_ptr, out_ptr, E: tl.constexpr, capacity: tl.constexpr, N_gate: tl.constexpr, BLOCK: tl.constexpr):
    pid_row = tl.program_id(0)  # over E*capacity
    pid_col = tl.program_id(1)  # over tiles of N_gate
    col_start = pid_col * BLOCK
    for n in range(col_start, col_start + BLOCK, BLOCK):
        # Load gate and up for this row and tile
        gate_row_base = gate_ptr + pid_row * N_gate
        up_row_base = up_ptr + pid_row * N_gate
        gate_vec = tl.load(gate_row_base + n + tl.arange(0, BLOCK), mask=(n + tl.arange(0, BLOCK)) < N_gate, other=0.0)
        up_vec = tl.load(up_row_base + n + tl.arange(0, BLOCK), mask=(n + tl.arange(0, BLOCK)) < N_gate, other=0.0)
        # SiLU(x) = x * sigmoid(x), where sigmoid(x) = 1 / (1 + exp(-x))
        sig = 1.0 / (1.0 + tl.exp(-gate_vec))
        y = gate_vec * sig
        out_vec = y * up_vec
        tl.store(out_ptr + pid_row * N_gate + n + tl.arange(0, BLOCK), out_vec, mask=(n + tl.arange(0, BLOCK)) < N_gate)


# Triton kernel: Batched GEMM row for down: expert_outputs = activated @ expert_down_weights.

# Inputs:

@triton.jit

    col_start = tl.program_id(1) * BLOCK
    for j in range(col_start, col_start + BLOCK, BLOCK):
        acc = tl.zeros([BLOCK], dtype=tl.float32)

        # We don't have direct access to starts here; we rely on P precomputed and v_pos_ptr is the stored within_pos.
        for i in range(0, P):

            pos = tl.load(v_pos_ptr + i)
            tok_i = tl.load(v_tok_ptr + i)
            wt = tl.load(v_wt_ptr + i)

            out_vals = tl.load(out_ptrs, mask=(j + tl.arange(0, BLOCK)) < hidden, other=0.0)
            acc += out_vals * wt
        # Atomically add into result[tok, j]
        result_ptrs = result_ptr + tok * hidden + (j + tl.arange(0, BLOCK))
        tl.atomic_add(result_ptrs, acc, mask=(j + tl.arange(0, BLOCK)) < hidden)
