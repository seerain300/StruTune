```python
import triton
import triton.language as tl

# Define the Triton kernel: one program per token
@triton.jit
def _single_expert_forward_kernel(hidden_ptr, gate_ptr, up_ptr, down_ptr, out_ptr,
                                  num_tokens,
                                  H: tl.constexpr, M: tl.constexpr):
    t = tl.program_id(axis=0)
    if t >= num_tokens:
        return

    # We specialize to num_experts = 1, so expert index is 0
    # Gate and Up weights are shaped [1, H, M]; Down is [1, M, H]

    # Compute gate_out[t, :] and up_out[t, :] vectors of length M
    gate_out = tl.zeros((M,), dtype=tl.float32)
    up_out = tl.zeros((M,), dtype=tl.float32)

    # Loop over hidden_size H to compute gate_out and up_out
    for h in range(0, H):
        # Load hidden_states[t, h] as scalar
        hs = tl.load(hidden_ptr + t * H + h)
        # Convert to float32 for numeric stability
        hs_f = hs.to(tl.float32)

        # Load gate and up vectors for this h across M
        # gate[0, h, m] and up[0, h, m]
        gate_vec = tl.zeros((M,), dtype=tl.float32)
        up_vec = tl.zeros((M,), dtype=tl.float32)
        for m in range(0, M):
            g = tl.load(gate_ptr + 0 * (H * M) + h * M + m).to(tl.float32)
            u = tl.load(up_ptr  + 0 * (H * M) + h * M + m).to(tl.float32)
            gate_vec[m] = g
            up_vec[m] = u

        gate_out += hs_f * gate_vec
        up_out += hs_f * up_vec

    # SiLU: x * sigmoid(x)
    silu_out = gate_out * tl.sigmoid(gate_out)
    activated = silu_out * up_out  # [M] vector

    # Compute final output for this token: sum over M of activated * down[0, m, h]
    out_row = tl.zeros((H,), dtype=tl.float32)
    for h_out in range(0, H):
        row_sum = tl.zeros((), dtype=tl.float32)
        for m in range(0, M):
            d = tl.load(down_ptr + 0 * (M * H) + m * H + h_out).to(tl.float32)
            row_sum += activated[m] * d
        out_row[h_out] = row_sum

    # Store result for this token: out[t, h] in bfloat16
    # Cast to bfloat16 for output
    for h_out in range(0, H):
        tl.store(out_ptr + t * H + h_out, out_row[h_out].to(tl.bfloat16))
```
