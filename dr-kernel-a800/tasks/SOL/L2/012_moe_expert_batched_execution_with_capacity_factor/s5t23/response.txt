Below is a corrected, single codeblock implementing ModelNew that:
- Generates flattened expert IDs and routing weights in Triton.
- Performs a stable odd-even sort by expert ID, permuting original token indices, in Triton.
- Computes per-expert counts (bincount) and starts (cumsum) in Triton.
- Computes capacity per expert with rounding up (ceil) in Triton.
- Computes gate_out, up_out, activated, and final expert_outputs via Triton batched matmuls (fp32 accumulation, bf16 output) for all valid (token, expert) positions.

import torch
import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Kernel 0: Flatten selected_experts into int32 (input int64 -> output int32)
@triton.jit
def flatten_experts_kernel(
    src_ptr,           # *int64, shape [num_tokens, num_experts_per_tok]
    dst_exp_ptr,       # *int32, shape [E]
    num_tokens: tl.constexpr,
    num_experts_per_tok: tl.constexpr,
    E: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < E
    val = tl.load(src_ptr + offsets, mask=mask, other=0)  # int64
    val = val.to(tl.int32)
    tl.store(dst_exp_ptr + offsets, val, mask=mask)


# Kernel 1: Odd-even stable sort by expert id (permute original positions: sorted_token_ids and sorted_exp, sorted_weights)
@triton.jit
def odd_even_stable_sort_by_exp_kernel(
    exp_ptr,                 # *int32, length E
    tok_ptr,                 # *int32, length E (original token indices, e.g., arange(num_tokens).repeat)
    wt_ptr,                  # *bf16,  length E
    sorted_exp_ptr,          # *int32, length E (output)
    sorted_tok_ptr,          # *int32, length E (output)
    sorted_wt_ptr,           # *bf16,  length E (output)
    E: tl.constexpr,
    BLOCK: tl.constexpr,
):
    # We perform E passes of odd-even sorting. Each pass does:
    # even: compare (0,1), (2,3), ...
    # odd:  compare (1,2), (3,4), ...
    # We use stable tie-breaking by original token index: swap only if a > b or (a == b and wa > wb).
    for _ in range(0, E):
        # Even phase
        for start in range(0, E, 2 * BLOCK):
            i = start
            while i >= 0 and i + 1 < E:
                a = exp_ptr[i]
                b = exp_ptr[i + 1]
                wa = tl.load(wt_ptr + i)
                wb = tl.load(wt_ptr + i + 1)
                ta = tl.load(tok_ptr + i)
                tb = tl.load(tok_ptr + i + 1)
                # Stable condition: swap only if a > b or (a == b and wa > wb)
                cond = (a > b) | ((a == b) & (wa > wb))
                # If swap, write b,a for positions i and i+1
                tl.store(sorted_exp_ptr + i, b, mask=cond)
                tl.store(sorted_exp_ptr + (i + 1), a, mask=cond)
                tl.store(sorted_wt_ptr + i, wb, mask=cond)
                tl.store(sorted_wt_ptr + (i + 1), wa, mask=cond)
                tl.store(sorted_tok_ptr + i, tb, mask=cond)
                tl.store(sorted_tok_ptr + (i + 1), ta, mask=cond)
                i += 2 * BLOCK
        # Odd phase
        for start in range(1, E, 2 * BLOCK):
            i = start
            while i >= 0 and i + 1 < E:
                a = exp_ptr[i]
                b = exp_ptr[i + 1]
                wa = tl.load(wt_ptr + i)
                wb = tl.load(wt_ptr + i + 1)
                ta = tl.load(tok_ptr + i)
                tb = tl.load(tok_ptr + i + 1)
                cond = (a > b) | ((a == b) & (wa > wb))
                tl.store(sorted_exp_ptr + i, b, mask=cond)
                tl.store(sorted_exp_ptr + (i + 1), a, mask=cond)
                tl.store(sorted_wt_ptr + i, wb, mask=cond)
                tl.store(sorted_wt_ptr + (i + 1), wa, mask=cond)
                tl.store(sorted_tok_ptr + i, tb, mask=cond)
                tl.store(sorted_tok_ptr + (i + 1), ta, mask=cond)
                i += 2 * BLOCK


# Kernel 2: Bincount of flattened expert ids (int32 -> int32 counts per expert)
@triton.jit
def bincount_experts_kernel(
    exp_ptr,            # *int32, length E
    counts_ptr,         # *int32, length num_experts
    num_experts: tl.constexpr,
    E: tl.constexpr,
    BLOCK: tl.constexpr,
):
    e = tl.program_id(axis=0)  # expert index
    if e >= num_experts:
        return
    total = tl.zeros((), dtype=tl.int32)
    for start in range(0, E, BLOCK):
        offsets = start + tl.arange(0, BLOCK)
        mask = offsets < E
        vals = tl.load(exp_ptr + offsets, mask=mask, other=0)  # int32
        # Count how many == e
        cnt = tl.sum(((vals == e) & mask).to(tl.int32))
        total += cnt
    tl.store(counts_ptr + e, total)


# Kernel 3: Cumsum of counts to compute per-expert starts (int32 -> int32)
@triton.jit
def cumsum_starts_kernel(
    counts_ptr,         # *int32, length num_experts
    starts_ptr,         # *int32, length num_experts
    num_experts: tl.constexpr,
):
    # Simple sequential scan
    for e in range(0, num_experts):
        if e == 0:
            tl.store(starts_ptr + e, 0)
        else:
            tl.store(starts_ptr + e, tl.load(starts_ptr + (e - 1)) + tl.load(counts_ptr + e))


# Kernel 4: Compute per-expert capacity: starts[e] + ceil(1.25 * counts[e]) - 1
@triton.jit
def compute_capacity_kernel(
    counts_ptr,         # *int32, length num_experts
    starts_ptr,         # *int32, length num_experts
    capacity_ptr,       # *int32, length num_experts
    num_experts: tl.constexpr,
):
    for e in range(0, num_experts):
        count = tl.load(counts_ptr + e)
        start = tl.load(starts_ptr + e)
        need = count * 1.25  # float
        ceil_need = tl.floor(need + 1.0)  # ceil via floor(x + 1)
        cap = start + ceil_need - 1  # int
        tl.store(capacity_ptr + e, cap)


# Kernel 5: Compute within_pos for each flattened assignment: within_pos = global_index - starts[expert]
@triton.jit
def compute_within_pos_kernel(
    global_index_ptr,   # *int32, length E
    starts_ptr,         # *int32, length num_experts
    valid_ptr,          # *int32, length E (0/1 mask)
    exp_ptr,            # *int32, length E (sorted experts)
    within_ptr,         # *int32, length E (output within_pos for valid entries)
    num_experts: tl.constexpr,
    E: tl.constexpr,
):
    for i in range(0, E):
        if tl.load(valid_ptr + i) == 1:
            # Map i to expert e and starts[e]
            e = tl.load(exp_ptr + i)
            start = tl.load(starts_ptr + e)
            wi = tl.load(global_index_ptr + i)
            wpos = wi - start
            tl.store(within_ptr + i, wpos)
