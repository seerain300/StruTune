# Triton kernel: row-wise matmul A[H] x B[H, M] -> C[M]
# Each program handles one output element j in [0, M). We vectorize across rows in tiles.
@triton.jit
def row_bmm(A_ptr, B_ptr, C_ptr,
            H, M,
            stride_A_row, stride_A_col,
            stride_B_row, stride_B_col,
            stride_C_out,
            BLOCK_ROWS: tl.constexpr, BLOCK_N: tl.constexpr):
    pid = tl.program_id(axis=0)  # program id across output columns
    j = pid * BLOCK_N + tl.arange(0, BLOCK_N)
    # Accumulator in fp32
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over rows in tiles
    for row_start in range(0, H, BLOCK_ROWS):
        i = row_start + tl.arange(0, BLOCK_ROWS)  # [BLOCK_ROWS]
        # Build pointers
        A_ptrs = A_ptr + i[:, None] * stride_A_row + j[None, :] * stride_A_col  # [BLOCK_ROWS, BLOCK_N]
        B_ptrs = B_ptr + i[:, None] * stride_B_row + j[None, :] * stride_B_col  # [BLOCK_ROWS, BLOCK_N]
        mask = (i[:, None] < H) & (j[None, :] < M)
        # Load and cast to fp32
        a = tl.load(A_ptrs, mask=mask, other=0.0).to(tl.float32)  # [BLOCK_ROWS, BLOCK_N]
        b = tl.load(B_ptrs, mask=mask, other=0.0).to(tl.float32)  # [BLOCK_ROWS, BLOCK_N]
        # acc += sum_i a[i, j] * b[i, j]
        # We need to compute dot across i dimension for each j
        # Implement via sum over BLOCK_ROWS: acc += tl.sum(a * b, axis=0)
        acc += tl.sum(a * b, axis=0)

    # Store results to C
    C_ptrs = C_ptr + j * stride_C_out
    mask_out = j < M
    tl.store(C_ptrs, acc, mask=mask_out)


# Triton kernel: elementwise SiLU over a vector
@triton.jit
def silu_kernel(X_ptr, Y_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    # SiLU: x * sigmoid(x)
    y = x * (1.0 / (1.0 + tl.exp(-x)))
    tl.store(Y_ptr + offs, y, mask=mask)

        j = j_start + tl.arange(0, BLOCK_N)  # [BLOCK_N]
