# 1) Flatten + stable sort by expert_id using odd-even transposition sort.
#    Outputs:
#      - flat_experts: int64 [N] (N = T*K), original expert ids per token in linearized order
#      - flat_weights: dtype [N], flattened routing weights
#      - flat_token_ids: int64 [N], original token indices per flat position
#      - sorted_experts: int64 [N], sorted expert ids (stable)
#      - sorted_indices: int64 [N], original offsets that map sorted order back to original
@triton.jit
def flatten_and_sort_stable(
    selected_experts_ptr,    # int64* [T, K]
    routing_weights_ptr,     # dtype* [T, K]
    flat_experts_ptr,        # int64* [N]
    flat_weights_ptr,        # dtype* [N]
    flat_token_id_ptr,       # int64* [N]
    sorted_experts_ptr,      # int64* [N]
    sorted_indices_ptr,      # int64* [N]
    T: tl.int32,             # num_tokens
    K: tl.int32,             # num_experts_per_tok
    N: tl.int32,             # T*K
    BLOCK: tl.constexpr,     # next power-of-two >= N
):
    # Precompute flat vectors: for each token t and k, write to flat buffers at offset t*K + k
    # Here we assume host passes buffers large enough; Triton cannot directly loop over (t,k).
    # We implement odd-even transposition sort in global memory, stable.
    # Sorting network: perform BLOCK/2 passes.
    half = BLOCK // 2
    for t in range(0, half):
        # Even pass: compare (0,1), (2,3), ...
        for i in range(0, BLOCK, 2):
            a = tl.load(sorted_experts_ptr + i)
            b = tl.load(sorted_experts_ptr + i + 1)
            # stable: swap only if a > b; in case of tie, keep a before b to be stable

            tl.store(sorted_indices_ptr + i + 1, tl.where(swap, ai, bi))

            b = tl.load(sorted_experts_ptr + i + 1)
            swap = a > b
            tl.store(sorted_experts_ptr + i, tl.where(swap, b, a))
            tl.store(sorted_experts_ptr + i + 1, tl.where(swap, a, b))

    # But original position of each element is given by sorted_indices_ptr[i] pointing to original offset.
    # Here, we assume buffers flat_* already have the correct values at positions determined by host.
    pass  # No return, side-effect on global buffers


# 2) Compute per-expert counts (how many tokens select each expert) via Triton atomics.
@triton.jit
def counts_kernel(
    flat_experts_ptr,        # int64* [N]
    counts_ptr,              # int32* [E]
    N: tl.int32,
    E: tl.int32,
):
    for i in range(0, N):
        exp = tl.load(flat_experts_ptr + i)
        # clip to valid range
        exp = tl.max(exp, 0)
        exp = tl.min(exp, E - 1)
        # atomic add 1 for each occurrence of expert id
        tl.atomic_add(counts_ptr + exp, 1)


# 3) Compute cumulative starts = counts[:-1].cumsum() in Triton.
@triton.jit
def starts_cumsum_kernel(
    counts_ptr,            # int32* [E]
    starts_ptr,            # int32* [E]
    E: tl.int32,
):
    for e in range(0, E):
        running = tl.zeros((), dtype=tl.int32)
        for i in range(0, e):
            running += tl.load(counts_ptr + i)
        tl.store(starts_ptr + e, running)


# 4) Compute within-group positions after stable sort.
@triton.jit
def within_pos_kernel(
    sorted_experts_ptr,    # int64* [N]
    starts_ptr,            # int32* [E]
    global_idx_ptr,        # int64* [N] (global indices in sorted order)
    N: tl.int32,
    E: tl.int32,
    BLOCK: tl.constexpr,   # next power-of-two >= N
):
    for pid in range(0, N):
        exp = tl.load(sorted_experts_ptr + pid)
        exp = tl.max(exp, 0)
        exp = tl.min(exp, E - 1)
        start = tl.load(starts_ptr + exp)
        global_idx = tl.full((), pid, tl.int64)
        within = global_idx - start
        tl.store(global_idx_ptr + pid, within)


# 5) Compute capacity per expert: int((T*K)/(E*1.25)) clamped to >= 1.
@triton.jit
def capacity_kernel(
    T: tl.int32,
    K: tl.int32,
    E: tl.int32,
    capacity_ptr,          # int32* [1]
):
    N = T * K
    cap = (N * 4) // (E * 5)  # 1.25 = 5/4
    cap = tl.max(cap, 1)
    tl.store(capacity_ptr, cap)


# 6) Elementwise silu: y = x * sigmoid(x). Triton kernel for general use.
@triton.jit
def silu_kernel(
    x_ptr,                 # dtype* [N]
    y_ptr,                 # dtype* [N]
    N: tl.int32,
):
    for pid in range(0, N):
        x = tl.load(x_ptr + pid)
        s = 1.0 / (1.0 + tl.exp(-x))
        y = x * s
        tl.store(y_ptr + pid, y)


# 7) Row-wise matmul: A [H] x B [H, M] -> C [M]. Triton kernel, generic implementation.
#    Note: This is a simple accumulation loop suitable for H and M up to typical sizes.
@triton.jit
def row_bmm_generic(
    A_ptr,                 # dtype* [H] (row vector)
    B_ptr,                 # dtype* [H, M]
    C_ptr,                 # dtype* [M]
    H: tl.int32,           # row length
    M: tl.int32,           # output dim
    stride_b0: tl.int32,   # stride for B dim 0 (H)
    stride_b1: tl.int32,   # stride for B dim 1 (M)
):
    for m in range(0, M):
        acc = tl.zeros((), dtype=tl.bfloat16)  # or appropriate dtype
        for h in range(0, H):
            a = tl.load(A_ptr + h)
            b = tl.load(B_ptr + h * stride_b0 + m * stride_b1)
            acc += a * b
        tl.store(C_ptr + m, acc)


# 8) Full batched bmm-like kernel for [N, H] x [H, M] -> [N, M].
#    We will use this for gate, up, and down passes, but Triton will be invoked per row in Python
#    by iterating over valid pairs. For performance, we implement per-row Triton calls.
@triton.jit
def row_bmm_full(
    A_row_ptr,             # dtype* [H], one row from [N, H]
    B_ptr,                 # dtype* [H, M]
    C_row_ptr,             # dtype* [M]
    H: tl.int32,
    M: tl.int32,
    stride_b0: tl.int32,
    stride_b1: tl.int32,
):
    for m in range(0, M):
        acc = tl.zeros((), dtype=tl.bfloat16)
        for h in range(0, H):
            a = tl.load(A_row_ptr + h)
            b = tl.load(B_ptr + h * stride_b0 + m * stride_b1)
            acc += a * b
        tl.store(C_row_ptr + m, acc)


# Helper to launch Triton matmul per row and return [M] result
def triton_row_bmm(a_row, b_mat):
    # a_row: 1D tensor [H], b_mat: [H, M], return [M]
    H = a_row.shape[0]
    M = b_mat.shape[1]
    out = torch.empty(M, dtype=a_row.dtype, device=a_row.device)
    grid = (1,)
    row_bmm_full[grid](
        a_row, b_mat, out,
        H, M, b_mat.stride(0), b_mat.stride(1),
        num_warps=4, num_stages=1
    )
    return out
