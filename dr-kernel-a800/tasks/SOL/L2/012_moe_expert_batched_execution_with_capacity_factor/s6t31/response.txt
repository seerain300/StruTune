@triton.jit
def row_bmm_kernel(
    A_ptr,           # *f16/bf16 pointer to A [H]
    B_ptr,           # *f16/bf16 pointer to B [H, M]
    C_ptr,           # *f16/bf16 pointer to C [M]
    H: tl.int32,     # hidden size
    M: tl.int32,     # intermediate size
    stride_A: tl.int32,        # stride for A rows: typically 1
    stride_B_row: tl.int32,    # stride for B row: typically M
    stride_B_col: tl.int32,    # stride for B col: typically 1
    stride_C: tl.int32,        # stride for C rows: typically 1
    BLOCK_H: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)  # one program per token row

    acc = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        a = tl.load(A_ptr + pid * stride_A + offs_h, mask=mask_h, other=0.0).to(tl.float32)  # [BLOCK_H]

        for m0 in range(0, M, BLOCK_M):
            offs_m = m0 + tl.arange(0, BLOCK_M)
            mask_m = offs_m < M
            # B is [H, M], load a BLOCK_H x BLOCK_M tile
            b = tl.load(
                B_ptr + offs_h[:, None] * stride_B_row + offs_m[None, :] * stride_B_col,
                mask=mask_h[:, None] & mask_m[None, :],
                other=0.0,

            # acc += sum_h(a[h] * b[h, :]) -> reduce over H

    out_mask = out_offs < M

    Y_ptr,    # *f16/bf16 output vector
    N: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    y = x / (1.0 + tl.exp(-x))
    tl.store(Y_ptr + offs, y, mask=mask)


# Kernel 3: per-row batched matmul A[M] x B[M, H] -> C[H]
@triton.jit
def row_bmm_down_kernel(
    A_ptr,           # *f16/bf16 pointer to A [M]
    B_ptr,           # *f16/bf16 pointer to B [M, H]
    C_ptr,           # *f16/bf16 pointer to C [H]
    M: tl.int32,     # activation length
    H: tl.int32,     # output hidden size
    stride_A: tl.int32,        # stride for A: typically 1
    stride_B_row: tl.int32,    # stride for B row: typically H
    stride_B_col: tl.int32,    # stride for B col: typically 1
    stride_C: tl.int32,        # stride for C: typically 1
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)  # one program per token
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    for m0 in range(0, M, BLOCK_M):
        offs_m = m0 + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M
        a = tl.load(A_ptr + pid * stride_A + offs_m, mask=mask_m, other=0.0).to(tl.float32)  # [BLOCK_M]

        for h0 in range(0, H, BLOCK_H):
            offs_h = h0 + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H
            b = tl.load(
                B_ptr + offs_m[:, None] * stride_B_row + offs_h[None, :] * stride_B_col,
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_M, BLOCK_H]
            acc += tl.sum(a[:, None] * b, axis=0)

    out_offs = tl.arange(0, BLOCK_H)
    out_mask = out_offs < H
    tl.store(C_ptr + pid * stride_C + out_offs, acc, mask=out_mask)


# Kernel 4: counts per expert (no torch.bincount)
@triton.jit
def counts_by_exp_kernel(
    EXP_ptr,    # *int64 flat experts [N]
    COUNT_ptr,  # *int32 per-expert counts [num_experts]
    N: tl.int32,
    num_experts: tl.int32,
    BLOCK: tl.constexpr,
):
    for e in range(0, num_experts):
        # Count how many times e appears in EXP_ptr
        cnt = tl.zeros((), dtype=tl.int32)
        for i in range(0, N, BLOCK):
            offs = i + tl.arange(0, BLOCK)
            mask = offs < N
            vals = tl.load(EXP_ptr + offs, mask=mask, other=tl.zeros([BLOCK], dtype=tl.int64))
            cnt += tl.sum((vals == e).to(tl.int32), axis=0)
        tl.store(COUNT_ptr + e, cnt)


# Kernel 5: inclusive prefix-sum for starts (manual implementation)
@triton.jit
def inclusive_prefix_sum_kernel(
    COUNT_ptr,   # *int32 [num_experts]
    START_ptr,   # *int32 [num_experts]
    num_experts: tl.int32,
    BLOCK: tl.constexpr,
):
    # Simple serial inclusive sum; num_experts is small in this task.
    total = tl.zeros((), dtype=tl.int32)
    for e in range(0, num_experts):
        val = tl.load(COUNT_ptr + e)
        total += val
        tl.store(START_ptr + e, total)


# Kernel 6: odd-even sort by expert ID (stable) for flat_experts
@triton.jit
def odd_even_sort_int64_kernel(
    EXP_ptr,       # *int64 [N]
    IDX_ptr,       # *int64 [N] (indices to sort)
    N: tl.int32,
    BLOCK: tl.constexpr,
):
    # Sort EXP_ptr into ascending order, stable, using odd-even transposition sort.
    # We maintain IDX_ptr as the permutation.
    for phase in range(0, 16):  # sufficient for typical N
        if phase % 2 == 0:
            # Even phase: pairs (0,1), (2,3), ...
            for i in range(0, N, 2):
                a = tl.load(EXP_ptr + i)
                b = tl.load(EXP_ptr + i + 1)
                ia = tl.load(IDX_ptr + i)
                ib = tl.load(IDX_ptr + i + 1)
                swap = a > b
                new_a = tl.where(swap, b, a)
                new_b = tl.where(swap, a, b)
                new_ia = tl.where(swap, ib, ia)
                new_ib = tl.where(swap, ia, ib)
                tl.store(EXP_ptr + i, new_a)
                tl.store(EXP_ptr + i + 1, new_b)
                tl.store(IDX_ptr + i, new_ia)
                tl.store(IDX_ptr + i + 1, new_ib)
        else:
            # Odd phase: pairs (1,2), (3,4), ...
            for i in range(1, N, 2):
                a = tl.load(EXP_ptr + i)
                b = tl.load(EXP_ptr + i + 1)
                ia = tl.load(IDX_ptr + i)
                ib = tl.load(IDX_ptr + i + 1)
                swap = a > b
                new_a = tl.where(swap, b, a)
                new_b = tl.where(swap, a, b)
                new_ia = tl.where(swap, ib, ia)
                new_ib = tl.where(swap, ia, ib)
                tl.store(EXP_ptr + i, new_a)
                tl.store(EXP_ptr + i + 1, new_b)
                tl.store(IDX_ptr + i, new_ia)
                tl.store(IDX_ptr + i + 1, new_ib)
