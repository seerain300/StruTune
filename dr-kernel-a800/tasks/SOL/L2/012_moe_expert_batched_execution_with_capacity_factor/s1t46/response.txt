import triton
import triton.language as tl

# Triton kernel: batched matmul for 1xH by HxM producing 1xM
@triton.jit
def _matmul_1xH_HxM(X_ptr, W_ptr, Y_ptr,
                     H: tl.constexpr, M: tl.constexpr,
                     stride_xm, stride_xk, stride_wk, stride_wm, stride_ym):
    # X is [1, H], W is [H, M], Y is [1, M]
    m = 0  # single row
    acc = tl.zeros((1, M), dtype=tl.float32)
    # Loop over K dimension
    for k in range(0, H):
        x = tl.load(X_ptr + m * stride_xm + k * stride_xk)  # scalar
        w_row = tl.load(W_ptr + k * stride_wk + tl.arange(0, M) * stride_wm)  # [M]
        # x is scalar, broadcast to [1, M] by multiplying with w_row
        acc += x * w_row
    # Store result
    tl.store(Y_ptr + m * stride_ym + tl.arange(0, M) * stride_ym, acc[0, :])

# Triton kernel: elementwise activation y = silu(z)
@triton.jit
def _silu_kernel(Z_ptr, Y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    z = tl.load(Z_ptr + offsets, mask=mask, other=0.0)  # fp32
    # sigmoid(z) = 1 / (1 + exp(-z))
    sig = 1.0 / (1.0 + tl.exp(-z))
    y = z * sig
    tl.store(Y_ptr + offsets, y, mask=mask)

# Triton kernel: elementwise multiply Y = A * B (same length)
@triton.jit
def _mul_kernel(A_ptr, B_ptr, Y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    a = tl.load(A_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(B_ptr + offsets, mask=mask, other=0.0)
    y = a * b
    tl.store(Y_ptr + offsets, y, mask=mask)

# Triton kernel: atomic add into result rows (for accumulation)
@triton.jit
def _atomic_accumulate_kernel(ROW_ptr, OUT_ptr, scalar, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    row = tl.load(ROW_ptr + offsets, mask=mask, other=0.0)  # [BLOCK]
    val = row * scalar  # scalar broadcast
    # Atomic add into OUT row 0 at positions 'offsets'
    out_ptrs = OUT_ptr + offsets  # row 0 is at base
    # Triton atomic_add expects fp32
    tl.atomic_add(out_ptrs, val, mask=mask)

        result = torch.zeros(num_tokens, hidden_size, dtype=torch.float32, device=hidden_states.device)

                N = gate_out.shape[1]  # equals gate_M

                down_w = expert_down_weights[e]  # [M, H_in], bf16

                # Prepare final_out vector via Triton matmul: X=[1, H_act], W=[H_act, H_in], Y=[1, H_in]
                final_out = torch.empty(1, H_in, dtype=torch.float32, device=hidden_states.device)

                rw = rw.to(torch.float32)
