@triton.jit
def _rand_int(seed: tl.constexpr, n: tl.constexpr, out_ptr):
    # Minimal LCG: state = (state * 1664525 + 1013904223) & 0xFFFFFFFF
    state = seed
    for _ in range(n):
        state = (state * 1664525 + 1013904223) & 0xFFFFFFFF
        tl.store(out_ptr + _, state)


@triton.jit
def _randn_normal(seed: tl.constexpr, n: tl.constexpr, out_ptr):
    # Box-Muller via polar method: produce 1 normal per iteration
    state = seed
    for _ in range(n):
        s1 = (state * 1664525 + 1013904223) & 0xFFFFFFFF
        s2 = (s1 * 1664525 + 1013904223) & 0xFFFFFFFF
        r = tl.sqrt(s1 * 3.141592653589793 * 4.656612873077393e-10 + s2 * 2.3283064365386963e-10)  # scaled to [0,1)
        u1 = tl.fmod(s1, 1.0) / 4294967296.0
        u2 = tl.fmod(s2, 1.0) / 4294967296.0
        v1 = tl.cos(2.0 * 3.141592653589793 * u1)
        v2 = tl.sin(2.0 * 3.14159265393 * u2)
        x = tl.sqrt(-2.0 * tl.log(r)) * v1
        y = tl.sqrt(-2.0 * tl.log(r)) * v2
        # We need exactly one value per iteration. Here we store x.
        tl.store(out_ptr + _, x)
