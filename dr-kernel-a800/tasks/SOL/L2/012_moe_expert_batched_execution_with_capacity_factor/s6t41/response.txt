import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: row-wise batched matmul A[H] x B[H, M] -> C[M]
# Each program instance computes one output row m for a given (row_id, m).
# A is [H], B is [H, M], C is [M].
@triton.jit
def row_bmm_kernel(
    A_ptr, B_ptr, C_ptr,
    H: tl.constexpr, M: tl.constexpr,
    stride_A_in, stride_B_row, stride_B_col, stride_C,
    row_id: tl.constexpr
):
    # one program computes output column m for this row_id
    # We need to compute sum_j A[row_id, j] * B[j, m] for all j in [0, H)
    # Implement as vectorized over j: we iterate j in blocks
    # But Triton doesn't support dynamic loops over H easily; instead we compute using B's row stride and A vectorized load.
    # However, to do an outer product accumulation, we'll use a small compile-time loop over j when H is small, or we restructure.
    # Better: since we need all M outputs for a given row, we launch grid=(M,). But Triton prefers a single grid. Fix: we process
    # one output element per launch; grid size will be M.
    m = tl.program_id(0)
    # Initialize accumulator
    acc = 0.0
    # Loop over j dimension (H) in blocks to keep code simple and legal
    BLOCK = 128  # works for typical hidden sizes; can be tuned
    for j0 in range(0, H, BLOCK):
        j = j0 + tl.arange(0, BLOCK)
        mask_j = j < H
        # Load A[row_id, j]
        a = tl.load(A_ptr + row_id * stride_A_in + j * 0, mask=mask_j, other=0.0)  # stride_A_in is stride for H dim
        # Load B[j, m]
        b = tl.load(B_ptr + j * stride_B_row + m * stride_B_col, mask=mask_j, other=0.0)
        acc += tl.sum(a * b, axis=0)
    # Store result to C[m]
    tl.store(C_ptr + m * stride_C, acc)


# Triton kernel: elementwise SiLU over a vector
@triton.jit
def silu_kernel(X_ptr, Y_ptr, N: tl.constexpr):
    idx = tl.program_id(0)
    x = tl.load(X_ptr + idx)
    # sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(Y_ptr + idx, y)


# Triton kernel: row-wise matmul down (C[M] x D[M, H] -> E[H])
# Each program instance computes one output row h for a given (row_id, h).
@triton.jit
def row_bmm_down_kernel(
    C_ptr, D_ptr, E_ptr,
    M: tl.constexpr, H: tl.constexpr,
    stride_C, stride_D_row, stride_D_col, stride_E,
    row_id: tl.constexpr
):
    h = tl.program_id(0)
    acc = 0.0
    BLOCK = 128
    for j0 in range(0, M, BLOCK):
        j = j0 + tl.arange(0, BLOCK)
        mask_j = j < M
        c = tl.load(C_ptr + j, mask=mask_j, other=0.0)
        d = tl.load(D_ptr + j * stride_D_row + h * stride_D_col, mask=mask_j, other=0.0)
        acc += tl.sum(c * d, axis=0)
    tl.store(E_ptr + h * stride_E, acc)


def triton_row_bmm(A: torch.Tensor, B: torch.Tensor, compute_dtype: torch.dtype = torch.float32):
    """
    Compute C = A @ B for A: [H], B: [H, M], returns C: [M].
    Launch row_bmm_kernel with grid=(M,).
    """
    assert A.dim() == 1 and B.dim() == 2, "A must be 1D, B must be 2D"
    H = A.shape[0]
    M = B.shape[1]
    # A and B may be of any dtype; we compute in compute_dtype
    A_in = A.to(compute_dtype)
    B_in = B.to(compute_dtype)
    C = torch.empty((M,), device=A.device, dtype=compute_dtype)
    # For row_bmm, we can set strides: A is 1D with stride H; B row stride is M, col stride is 1
    stride_A_in = H
    stride_B_row = M
    stride_B_col = 1
    stride_C = 1
    grid = (M,)
    row_bmm_kernel[grid](
        A_in, B_in, C,
        H, M,
        stride_A_in, stride_B_row, stride_B_col, stride_C,
        row_id=0  # row_id is not used since we launch grid=(M,) and compute per m
    )
    return C


def triton_silu(x: torch.Tensor, compute_dtype: torch.dtype = torch.float32):
    """
    Elementwise SiLU: y = x * sigmoid(x).
    Launch silu_kernel with grid=(x.numel(),).
    """
    N = x.numel()
    x_in = x.to(compute_dtype)
    y = torch.empty_like(x_in, device=x.device, dtype=compute_dtype)
    grid = (N,)
    silu_kernel[grid](x_in, y, N)
    return y


def triton_row_bmm_down(C: torch.Tensor, D: torch.Tensor, compute_dtype: torch.dtype = torch.float32):
    """
    Compute E = C @ D for C: [M], D: [M, H], returns E: [H].
    Launch row_bmm_down_kernel with grid=(H,).
    """
    assert C.dim() == 1 and D.dim() == 2, "C must be 1D, D must be 2D"
    M = C.shape[0]
    H = D.shape[1]
    C_in = C.to(compute_dtype)
    D_in = D.to(compute_dtype)
    E = torch.empty((H,), device=C.device, dtype=compute_dtype)
    stride_C = 1
    stride_D_row = M
    stride_D_col = 1
    stride_E = 1
    grid = (H,)
    row_bmm_down_kernel[grid](
        C_in, D_in, E,
        M, H,
        stride_C, stride_D_row, stride_D_col, stride_E,
        row_id=0
    )
    return E
