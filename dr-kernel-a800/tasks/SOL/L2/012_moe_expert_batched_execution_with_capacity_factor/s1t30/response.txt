def bmm_triton_kernel(X_ptr, W_ptr, Y_ptr,
                       B, H, M,

                       stride_wh, stride_wm,
                       stride_yb, stride_ym,
                       BLOCK_M: tl.constexpr):

    b = tl.program_id(0)
    offs_m = tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
