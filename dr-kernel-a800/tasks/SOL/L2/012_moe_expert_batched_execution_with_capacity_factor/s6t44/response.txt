import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: row-wise matmul A[H] x B[H, M] -> C[M]
# Each program handles one output element j in [0, M)
# A is row-major with stride_h, B is [H, M] row-major, C is [M]
@triton.jit
def row_bmm_out_cols(A_ptr, B_ptr, C_ptr,
                     H, M,
                     stride_a, stride_b, stride_c,
                     BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    j = pid  # each program computes one output element j
    # If j >= M, we can early exit, but Triton expects grid=M. We'll guard j<M.
    # We need to loop over H to accumulate: C[j] = sum_h A[h] * B[h, j]
    # We'll vectorize over H in chunks of BLOCK_M. Note: here we accumulate a scalar.
    # Initialize accumulator
    acc = tl.zeros((), dtype=tl.float32)
    for h_start in range(0, H, BLOCK_M):
        offs_h = h_start + tl.arange(0, BLOCK_M)
        mask_h = offs_h < H
        a = tl.load(A_ptr + offs_h * stride_a, mask=mask_h, other=0.0)  # shape [BLOCK_M]
        b = tl.load(B_ptr + offs_h * stride_b + j, mask=mask_h, other=0.0)  # shape [BLOCK_M]
        # b is scalar per h; broadcast a to [BLOCK_M, 1] to multiply and sum
        prod = a[:, None] * b[None, :]
        acc += tl.sum(prod, axis=0)
    # Store to C[j]
    tl.store(C_ptr + j * stride_c, acc)


# Triton kernel: row-wise matmul A[M] x B[M, H] -> C[H]
# Each program handles one output element i in [0, H)
@triton.jit
def row_bmm_down(A_ptr, B_ptr, C_ptr,
                 M, H,
                 stride_a, stride_b, stride_c,
                 BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    i = pid  # each program computes one output element i
    acc = tl.zeros((), dtype=tl.float32)
    for m_start in range(0, M, BLOCK_H):
        offs_m = m_start + tl.arange(0, BLOCK_H)
        mask_m = offs_m < M
        a = tl.load(A_ptr + offs_m * stride_a, mask=mask_m, other=0.0)  # shape [BLOCK_H]
        b = tl.load(B_ptr + offs_m * stride_b + i, mask=mask_m, other=0.0)  # shape [BLOCK_H]
        prod = a[:, None] * b[None, :]
        acc += tl.sum(prod, axis=0)
    tl.store(C_ptr + i * stride_c, acc)


# Triton kernel: elementwise SiLU for a vector
@triton.jit
def silu_kernel(X_ptr, Y_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    # SiLU: x * sigmoid(x) = x / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(Y_ptr + offs, y, mask=mask)


# Helper to invoke row_bmm_out_cols over all rows (one row per launch). This is a host-side loop.
# We pass A_row as a 1D tensor of length H, and B as [H, M].
def triton_row_bmm_cols(A_row, B, out, H, M, stride_a, stride_b, stride_c, BLOCK_M: int):
    grid = (M,)
    row_bmm_out_cols[grid](
        A_row, B, out,
        H, M,
        stride_a, stride_b, stride_c,
        BLOCK_M=BLOCK_M,
        num_warps=4,
    )


# Helper to invoke row_bmm_down over all rows (one row per launch).
def triton_row_bmm_down(A, B, out, M, H, stride_a, stride_b, stride_c, BLOCK_H: int):
    grid = (H,)
    row_bmm_down[grid](
        A, B, out,
        M, H,
        stride_a, stride_b, stride_c,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


def triton_silu(x, n_elements, BLOCK: int):
    x_f32 = x.to(torch.float32)
    y = torch.empty_like(x_f32)
    grid = (triton.cdiv(n_elements, BLOCK),)
    silu_kernel[grid](x_f32, y, n_elements, BLOCK=BLOCK, num_warps=4)
    return y.to(x.dtype)


class ModelNew(torch.nn.Module):
