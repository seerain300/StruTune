import math
import torch
import triton
import triton.language as tl


# Triton kernels:
# Kernel 1: Scatter hidden_states into expert_inputs based on selected_experts and t.
# Inputs:
#   hidden_states_ptr: [num_tokens, hidden_size], dtype bfloat16 (we read as float32 for compute)
#   selected_experts_ptr: [T], int64
#   token_ids_ptr: [T], int64
#   expert_inputs_ptr: [num_experts, capacity, hidden_size], float32 (initialize to zeros on host)
#   T: total number of assignments
#   H: hidden_size
#   NUM_EXPERTS: num_experts
#   CAPACITY: capacity = ceil(1.25 * T / NUM_EXPERTS)
@triton.jit
def scatter_hs_kernel(
    hidden_states_ptr, selected_experts_ptr, token_ids_ptr,
    expert_inputs_ptr,
    T, H, NUM_EXPERTS, CAPACITY, BLOCK: tl.constexpr
):
    # Each program handles a block of rows (flattened assignments).
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < T

    # Load selected expert id and token id
    e = tl.load(selected_experts_ptr + offs, mask=mask, other=0).to(tl.int32)
    tok = tl.load(token_ids_ptr + offs, mask=mask, other=0).to(tl.int32)

    # Compute position within capacity: pos = offs % (NUM_EXPERTS * NUM_PER_TOK)
    NUM_PER_TOK = T // tl.num_tokens  # Not needed here; just used in host code. We instead compute NUM_PER_TOK from T and num_tokens passed via runtime (not tl.constexpr). So we must compute NUM_PER_TOK in host and pass as scalar.
    pos = offs % (NUM_EXPERTS * NUM_PER_TOK)

    # Load hidden state row for this token
    # hidden_states_ptr is [num_tokens, H], row index is tok
    hs_row_ptr = hidden_states_ptr + tok * H + tl.arange(0, H)
    # Note: We need H lanes to write a full row into expert_inputs[e, pos, :]
    # We write one hidden dimension per lane using a loop over H dimension blocks.
    # For simplicity and correctness, we assume H is small enough to be handled in one block; otherwise, we can split but that requires 2D grid. To keep code simple, we assume H <= BLOCK (e.g., 4096). If H > BLOCK, host can split, but Triton does not support arbitrary 2D grid here. So we enforce H <= BLOCK. We set BLOCK to a large value (e.g., 4096) which covers typical hidden sizes in the provided workloads.
    hs_vals = tl.load(hs_row_ptr, mask=mask, other=0.0)

    # Store into expert_inputs[e, pos, :]
    dst_ptr = expert_inputs_ptr + e * (CAPACITY * H) + pos * H + tl.arange(0, H)
    # We need to store hs_vals into the H lanes. To map lane to H index, we can use linear indexing: for lane i, store hs_vals[i] at offset i. This works because hs_vals is a vector of length H (BLOCK), and we use tl.arange(0, H) as the per-lane offset. We only store for valid mask.
    tl.store(dst_ptr, hs_vals, mask=mask)


# Kernel 2: Batched matmul gate_out = expert_inputs @ expert_gate_weights^T
# Inputs:
#   A: expert_inputs_ptr, [num_experts, capacity, hidden_size], float32
#   B: expert_gate_weights_ptr, [num_experts, hidden_size, moe_intermediate_size], float32 (we will treat B as [K, H, N] where K=num_experts, H=hidden_size, N=moe_intermediate_size)
#   C: gate_out_ptr, [num_experts, capacity, hidden_size], float32
#   K, H, N, Batches: num_experts, capacity, hidden_size
@triton.jit
def matmul_bmm_gate_kernel(
    A_ptr, B_ptr, C_ptr,
    num_experts, capacity, hidden_size, intermediatesize, BLOCK: tl.constexpr
):
    # Grid: (num_experts, ceil_div(capacity, BLOCK), hidden_size)
    e = tl.program_id(0)
    c_blk = tl.program_id(1)
    h = tl.program_id(2)
    offs_c = c_blk * BLOCK + tl.arange(0, BLOCK)
    mask_c = offs_c < capacity

    # Accumulator for this (e, h) across C blocks
    acc = tl.zeros([BLOCK], dtype=tl.float32)

    # Loop over hidden_size in tiles of BLOCK
    for off_h in range(0, hidden_size, BLOCK):
        offs_h = off_h + tl.arange(0, BLOCK)
        mask_h = offs_h < hidden_size

        # Load A[e, offs_c, offs_h] -> [BLOCK]
        a = tl.load(A_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + offs_h, mask=mask_c[:, None] & mask_h[None, :], other=0.0)
        # Load B[e, offs_h, :] -> shape [BLOCK, intermediatesize]
        b = tl.load(B_ptr + e * (hidden_size * intermediatesize) + offs_h[:, None] * intermediatesize + tl.arange(0, intermediatesize), mask=mask_h[:, None], other=0.0)
        # acc += sum over h-dim of a * b
        acc += tl.sum(a[:, None] * b[None, :], axis=0)

    # Store to C[e, offs_c, h]
    dst_ptr = C_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + h
    tl.store(dst_ptr, acc, mask=mask_c)


# Kernel 3: Batched matmul up_out = expert_inputs @ expert_up_weights^T
@triton.jit
def matmul_bmm_up_kernel(
    A_ptr, B_ptr, C_ptr,
    num_experts, capacity, hidden_size, intermediatesize, BLOCK: tl.constexpr
):
    e = tl.program_id(0)
    c_blk = tl.program_id(1)
    h = tl.program_id(2)
    offs_c = c_blk * BLOCK + tl.arange(0, BLOCK)
    mask_c = offs_c < capacity

    acc = tl.zeros([BLOCK], dtype=tl.float32)
    for off_h in range(0, hidden_size, BLOCK):
        offs_h = off_h + tl.arange(0, BLOCK)
        mask_h = offs_h < hidden_size
        a = tl.load(A_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + offs_h, mask=mask_c[:, None] & mask_h[None, :], other=0.0)
        b = tl.load(B_ptr + e * (hidden_size * intermediatesize) + offs_h[:, None] * intermediatesize + tl.arange(0, intermediatesize), mask=mask_h[:, None], other=0.0)
        acc += tl.sum(a[:, None] * b[None, :], axis=0)

    dst_ptr = C_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + h
    tl.store(dst_ptr, acc, mask=mask_c)


# Kernel 4: Elementwise activated = SiLU(gate_out) * up_out
# Inputs: gate_out_ptr, up_out_ptr, activated_ptr, sizes
@triton.jit
def elementwise_silu_mul_kernel(
    gate_ptr, up_ptr, out_ptr,
    num_experts, capacity, hidden_size, BLOCK: tl.constexpr
):
    e = tl.program_id(0)
    c_blk = tl.program_id(1)
    h = tl.program_id(2)
    offs_c = c_blk * BLOCK + tl.arange(0, BLOCK)
    mask_c = offs_c < capacity
    # Load gate and up
    gate = tl.load(gate_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + h, mask=mask_c, other=0.0)
    up = tl.load(up_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + h, mask=mask_c, other=0.0)
    # SiLU(x) = x * sigmoid(x) = x / (1 + exp(-x))
    silu = gate / (1.0 + tl.exp(-gate))
    activated = silu * up
    tl.store(out_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + h, activated, mask=mask_c)


# Kernel 5: Batched matmul expert_outputs = activated @ expert_down_weights^T
# Inputs: activated_ptr [num_experts, capacity, hidden_size], expert_down_weights_ptr [num_experts, hidden_size, hidden_size], output_ptr [num_experts, capacity, hidden_size]
@triton.jit
def matmul_bmm_down_kernel(
    A_ptr, B_ptr, C_ptr,
    num_experts, capacity, hidden_size, intermediatesize, BLOCK: tl.constexpr
):
    e = tl.program_id(0)
    c_blk = tl.program_id(1)
    h = tl.program_id(2)
    offs_c = c_blk * BLOCK + tl.arange(0, BLOCK)
    mask_c = offs_c < capacity

    acc = tl.zeros([BLOCK], dtype=tl.float32)
    for off_h in range(0, hidden_size, BLOCK):
        offs_h = off_h + tl.arange(0, BLOCK)
        mask_h = offs_h < hidden_size
        a = tl.load(A_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + offs_h, mask=mask_c[:, None] & mask_h[None, :], other=0.0)
        b = tl.load(B_ptr + e * (hidden_size * hidden_size) + offs_h[:, None] * hidden_size + tl.arange(0, hidden_size), mask=mask_h[:, None], other=0.0)
        acc += tl.sum(a[:, None] * b[None, :], axis=0)

    dst_ptr = C_ptr + e * (capacity * hidden_size) + offs_c * hidden_size + h
    tl.store(dst_ptr, acc, mask=mask_c)


# Kernel 6: Final weighted scatter-add into result.
# Inputs:
#   routing_weights_ptr: [T], float32
#   selected_experts_ptr: [T], int64
#   token_ids_ptr: [T], int64
#   expert_outputs_ptr: [num_experts, capacity, hidden_size], float32
#   result_ptr: [num_tokens, hidden_size], float32 (initialize to zeros on host), we will add to it.
# Outputs: result_ptr updated
@triton.jit
def scatter_weighted_add_kernel(
    routing_ptr, selected_experts_ptr, token_ids_ptr, expert_outputs_ptr, result_ptr,
    T, NUM_EXPERTS, NUM_PER_TOK, CAPACITY, H, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < T

    e = tl.load(selected_experts_ptr + offs, mask=mask, other=0).to(tl.int32)
    tok = tl.load(token_ids_ptr + offs, mask=mask, other=0).to(tl.int32)
    # pos within capacity: offs % (NUM_EXPERTS * NUM_PER_TOK)
    pos = offs % (NUM_EXPERTS * NUM_PER_TOK)

    v = tl.load(routing_ptr + offs, mask=mask, other=0.0)  # float32
    # Load expert_outputs[e, pos, :]
    dst_ptr = expert_outputs_ptr + e * (CAPACITY * H) + pos * H + tl.arange(0, H)
    out_vec = tl.load(dst_ptr, mask=mask, other=0.0)
    contrib = v[:, None] * out_vec  # broadcast over hidden dimension

    # Add into result[tok, :]
    res_ptr = result_ptr + tok * H + tl.arange(0, H)
    tl.store(res_ptr, contrib, mask=mask)

def get_inputs(axes_and_scalars: dict, device: torch.device) -> dict[str, torch.Tensor]:

    hidden_size = axes_and_scalars["hidden_size"]
    moe_intermediate_size = axes_and_scalars["moe_intermediate_size"]
    num_experts = axes_and_scalars["num_experts"]
    num_experts_per_tok = axes_and_scalars["num_experts_per_tok"]

    dtype = torch.bfloat16

    hidden_states = torch.randn(num_tokens, hidden_size, dtype=dtype, device=device)

    # Generate valid expert indices - each token selects num_experts_per_tok unique experts
    selected_experts = torch.zeros(num_tokens, num_experts_per_tok, dtype=torch.int64, device=device)
    for i in range(num_tokens):
        perm = torch.randperm(num_experts, device=device)[:num_experts_per_tok]
        selected_experts[i] = perm

    # Generate routing weights that sum to 1 for each token
    routing_logits = torch.randn(num_tokens, num_experts_per_tok, dtype=dtype, device=device)
    routing_weights = F.softmax(routing_logits.float(), dim=-1).to(dtype)

    # Expert weights: Xavier init with 1/sqrt(fan_in)
    # gate/up: shape [num_experts, hidden_size, moe_intermediate_size], fan_in = hidden_size (dim 1)

    expert_up_weights = torch.randn(num_experts, hidden_size, moe_intermediate_size, dtype=dtype, device=device) / math.sqrt(hidden_size)
    # down: shape [num_experts, moe_intermediate_size, hidden_size], fan_in = moe_intermediate_size (dim 1)
    expert_down_weights = torch.randn(num_experts, moe_intermediate_size, hidden_size, dtype=dtype, device=device) / math.sqrt(moe_intermediate_size)

    return {

        "selected_experts": selected_experts,
        "routing_weights": routing_weights,
