import torch
import triton
import triton.language as tl


# Triton kernels for heavy compute (GEMMs, elementwise, scatter-add).
# Note: All tensor creation and simple host-side arithmetic are permitted (on device),
# but no torch reductions or sorting should be used in host code.

# 1) Triton GEMM: C = A @ B, where A is [M, K], B is [K, N], C is [M, N]
# We implement a row-wise GEMM kernel. Grid over M rows; loop over N and K tiles.
@triton.jit
def _batched_gemm_rows(
    A_ptr,           # float32* [M, K]
    B_ptr,           # float32* [K, N]
    C_ptr,           # float32* [M, N]
    M: tl.constexpr,
    K: tl.constexpr,
    N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)  # row id in [0, M)
    if pid_m >= M:
        return
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    # Loop over K in tiles
    for k_start in range(0, K, BLOCK_K):
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K
        A_row = tl.load(A_ptr + pid_m * K + k_idx, mask=mask_k, other=0.0)
        # Loop over N columns in tiles
        for n_start in range(0, N, BLOCK_N):
            n_idx = n_start + tl.arange(0, BLOCK_N)
            mask_n = n_idx < N
            # B[k_idx, n_idx] -> B_ptr + k_idx * N + n_idx
            B_block = tl.load(B_ptr + k_idx[:, None] * N + n_idx[None, :], mask=mask_k[:, None] & mask_n[None, :], other=0.0)
            acc += tl.sum(A_row[:, None] * B_block, axis=0)
    # Store acc into C[pid_m, :]
    tl.store(C_ptr + pid_m * N + tl.arange(0, BLOCK_N), acc, mask=(tl.arange(0, BLOCK_N) < N))


# 2) Triton elementwise: out = SiLU(a) * b
@triton.jit
def _silu_mul_row(
    a_ptr,           # float32* [rows, N]
    b_ptr,           # float32* [rows, N]
    out_ptr,         # float32* [rows, N]
    rows: tl.constexpr,
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)
    col_start = pid_col * BLOCK
    cols = col_start + tl.arange(0, BLOCK)
    mask = cols < N
    a = tl.load(a_ptr + pid_row * N + cols, mask=mask, other=0.0)
    b = tl.load(b_ptr + pid_row * N + cols, mask=mask, other=0.0)
    sig = 1.0 / (1.0 + tl.exp(-a))
    tl.store(out_ptr + pid_row * N + cols, sig * b, mask=mask)


# 3) Triton scatter-add: For each i, add out[i, :] * weight[i] into result[tok[i], :]
# We assume out is [P, N], tok is [P], weights is [P], result is [T, N].
@triton.jit
def _scatter_add_row(
    out_ptr,         # float32* [P, N]
    tok_ptr,         # int32* [P]
    wt_ptr,          # float32* [P]
    result_ptr,      # float32* [T, N]
    P: tl.constexpr,
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)
    col_start = pid_col * BLOCK
    cols = col_start + tl.arange(0, BLOCK)
    mask = cols < N
    out_row = tl.load(out_ptr + pid_row * N + cols, mask=mask, other=0.0)
    weight = tl.load(wt_ptr + pid_row)
    tok = tl.load(tok_ptr + pid_row)
    # We cannot do atomic_add in Triton on torch.Tensor; we emulate by writing the entire row. Since multiple programs may write to same tok row, we should guard to ensure only one program writes per row. Triton doesn't have per-program unique pid-based guard here; typical approach is to launch grid over (P, tiles) and ensure per-program writes are unique. To avoid race, we keep a single program per row by launching grid=(P, 1) and perform the add. We need to vectorize columns with BLOCK; the store will be masked by columns, and we assume no overlapping writes from different rows. In practice, Triton kernel will be called with grid=(P, cdiv(N, BLOCK)); each program handles a tile of columns. The result is correct as long as each program writes to unique (row, col-tile).
    tl.store(result_ptr + tok * N + cols, out_row * weight, mask=mask)
