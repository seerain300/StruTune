# Triton kernel: compute per-row batched matmul of A[H] with B[H, M] -> output[M]
# We launch with grid over M blocks.
@triton.jit
def row_bmm_out_vec_kernel(
    A_ptr, B_ptr, C_ptr,
    H, M,
    stride_A, stride_B_row, stride_B_col, stride_C
,
    BLOCK_H: tl.constexpr, BLOCK_M: tl.constexpr
):
    pid_m = tl.program_id(0)
    j = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M,), dtype=tl.float32)
    # Loop over H in blocks
    for h_off in range(0, H, BLOCK_H):
        h_idx = h_off + tl.arange(0, BLOCK_H)
        # Load A[i] for all i in this block
        a = tl.load(A_ptr + h_idx * stride_A, mask=h_idx < H, other=0.0)  # [BLOCK_H]
        # Load B[i, j] for all i in this block and j in this tile
        b = tl.load(
            B_ptr + h_idx[:, None] * stride_B_row + j[None, :] * stride_B_col,
            mask=(h_idx[:, None] < H) & (j[None, :] < M),
            other=0.0
        )  # [BLOCK_H, BLOCK_M]
        # Accumulate: acc[j] += sum_i a[i] * b[i, j]
        acc += tl.sum(a[:, None] * b, axis=0)
    # Store results
    tl.store(C_ptr + j * stride_C, acc, mask=j < M)


# Triton kernel: elementwise SiLU over input vector X[M] -> Y[M], y = x * sigmoid(x)
@triton.jit
def silu_kernel(
    X_ptr, Y_ptr,
    N,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    x = tl.load(X_ptr + offs, mask=offs < N, other=0.0)
    # sigmoid(x) = 1 / (1 + exp(-x))
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(Y_ptr + offs, y, mask=offs < N)
