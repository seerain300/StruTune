# A[M=hidden_size, K=hidden_size] @ W[K=hidden_size, N=intermediate_size] -> [hidden_size, intermediate_size]

def bmm_gate_up_kernel(

    W_ptr,             # *bf16, [hidden_size, intermediate_size]
    C_ptr,             # *bf16, [hidden_size, intermediate_size] (output)
    M: tl.constexpr,   # hidden_size
    K: tl.constexpr,   # hidden_size
    N: tl.constexpr,   # intermediate_size

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    # We implement a direct computation here: for each output column j in [0..N), compute dot(A_row, W_col_j)
    # across K dimension. Triton supports loops; we'll implement a vectorized reduction.
    # Note: Triton does not support general 2D matmul out-of-the-box, so this kernel computes one row-block of C.
    # For this task, we keep it simple: direct O(M*K*N) computation in fp32, store bf16.

    x_ptr,             # *bf16, length L
    y_ptr,             # *bf16, length L
    L: tl.constexpr,
    BLOCK: tl.constexpr,
):

        tl.store(y_ptr + offs, y.to(tl.bfloat16), mask=mask)

    K: tl.constexpr,   # intermediate_size
