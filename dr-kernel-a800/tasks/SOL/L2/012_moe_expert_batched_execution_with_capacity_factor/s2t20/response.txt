def _sort_pairs_by_exp_key(
    flat_experts_ptr, flat_weights_ptr, flat_token_ids_ptr, out_idx_ptr,
    N, BLOCK_SIZE: tl.constexpr,
):
    # Bitonic sort network over BLOCK_SIZE lanes (must be >= N).
    # We operate on a single program with BLOCK_SIZE lanes. Only first N elements are valid.
    offsets = tl.arange(0, BLOCK_SIZE)
    mask = offsets < N

    # Load data
    # Note: Triton supports int64 and float16; bf16 support depends on version, but we will pass float16 for weights.
    exp = tl.load(flat_experts_ptr + offsets, mask=mask, other=0).to(tl.int64)
    tok = tl.load(flat_token_ids_ptr + offsets, mask=mask, other=0).to(tl.int64)
    wgt = tl.load(flat_weights_ptr + offsets, mask=mask, other=0.0)

    # We sort lexicographically: (exp, tok) to achieve stable ordering when exp ties.
    # Bitonic sort for pairs (exp, tok) with direction depending on k.
    idx = offsets  # own lane index

    # Outer loop: k = 2, 4, ..., BLOCK_SIZE
    k = 2
    while k <= BLOCK_SIZE:
        # Inner loop: j = k/2, k/4, ..., 1
        j = k // 2
        while j >= 1:
            partner = idx ^ j  # XOR pairing
            e_partner = exp[partner]
            t_partner = tok[partner]
            w_partner = wgt[partner]

            asc = ((idx & k) == 0)
            # Compare (exp, tok) pairs
            cmp_exp = exp > e_partner
            cmp_tok = exp == e_partner
            # Among equal exp, sort by token id ascending to maintain stability
            greater = tl.where(cmp_exp, 1, 0) + tl.where(cmp_tok & (tok > t_partner), 1, 0)

            swap = greater & (not asc)
            new_exp = tl.where(swap, e_partner, exp)
            new_tok = tl.where(swap, t_partner, tok)
            new_wgt = tl.where(swap, w_partner, wgt)

            tok = new_tok
            wgt = new_wgt
