        mask_m = offs_m < M

        acc = tl.zeros([BLOCK_M], dtype=tl.float32)

        for h_start in range(0, H, BLOCK_H):
            offs_h = h_start + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            a = tl.load(A_ptr + offs_h, mask=mask_h, other=0.0)

            # Multiply-accumulate: a[:, None] * b[None, :] -> [BLOCK_H, BLOCK_M], reduce over H
            acc += tl.sum(a[:, None] * b, axis=0)

                v_i = tl.load(values_ptr + i)
                v_j = tl.load(values_ptr + i + 1)
                # Compare-and-swap condition: if v_j < v_i, swap them
                swap = v_j < v_i
                new_i = tl.where(swap, v_j, v_i)
                new_j = tl.where(swap, v_i, v_j)
                # Write back (use mask to avoid OOB, always in-range for i and i+1)
                tl.store(values_ptr + i, new_i)
                tl.store(values_ptr + i + 1, new_j)

                v_i = tl.load(values_ptr + i)
                v_j = tl.load(values_ptr + i + 1)
                swap = v_j < v_i

                new_j = tl.where(swap, v_i, v_j)

                tl.store(values_ptr + i + 1, new_j)

        pid_m = tl.program_id(0)
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M

        # Accumulator in fp32 for stability
        acc = tl.zeros([BLOCK_M], dtype=tl.float32)

        # Loop over H in tiles and accumulate
        for h_start in range(0, H, BLOCK_H):
            offs_h = h_start + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # Load A tile (vector along H)
            a = tl.load(A_ptr + offs_h, mask=mask_h, other=0.0)
            # Load B tile (matrix [BLOCK_H, BLOCK_M])
            b_ptrs = B_ptr + (offs_h[:, None] * stride_b0 + offs_m[None, :] * stride_b1)
            b = tl.load(b_ptrs, mask=mask_h[:, None] & mask_m[None, :], other=0.0)

            # Accumulate: for each h in tile, sum a[h] * B[h, m]
            # We reduce along H axis: a[:, None] * b[None, :] -> [BLOCK_H, BLOCK_M]
            acc += tl.sum(a[:, None] * b, axis=0)

        # Store result
        tl.store(C_ptr + offs_m, acc, mask=mask_m)
