        b_exp_tok = tl.load(pairs_ptr + i + 1)    # int64 pair

    up_ptr,        # float32* [E*cap, N_gate]
    out_ptr,       # float32* [E*cap, N_gate]
    E: tl.constexpr,
    cap: tl.constexpr,
    N_gate: tl.constexpr,
    row_id: tl.constexpr,
):

    pos = row_id % cap
    for n in range(0, N_gate):
