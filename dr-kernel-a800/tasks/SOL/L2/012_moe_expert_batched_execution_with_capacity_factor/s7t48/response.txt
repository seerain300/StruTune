Below is a Triton-only implementation that:

- Uses the provided selected_experts, routing_weights, hidden_states, and expert weights.
- Launches Triton kernels to:
  - scatter hidden states into expert_inputs
  - compute gate_out = expert_inputs @ expert_gate_weights^T

# Kernel 1: Build expert_inputs by scattering hidden_states.
# expert_inputs: [num_experts, capacity, hidden_size] (bf16)
# selected_experts: [T] int64, tok_ids: [T] int64, hidden_states: [num_tokens, hidden_size] (bf16)
# T = num_tokens * num_experts_per_tok
@triton.jit
def scatter_hidden_states_kernel(
    selected_ptr, tok_ptr, hidden_ptr, expert_inputs_ptr,
    T: tl.constexpr, num_experts: tl.constexpr, hidden_size: tl.constexpr, num_experts_per_tok: tl.constexpr,
    capacity: tl.constexpr,
):
    t = tl.program_id(0)  # one program per flattened index
    # Load selected expert and token id
    e = tl.load(selected_ptr + t)  # int64
    token = tl.load(tok_ptr + t)   # int64

    # within_pos = t % (num_experts_per_tok * num_experts)
    within = t % (num_experts_per_tok * num_experts)
    valid = within < capacity

    # Load hidden vector slice
    # We vectorize over hidden_size in chunks of BLOCK_H
    BLOCK_H = 64
    for off in range(0, hidden_size, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < hidden_size
        # Compute source offset: hidden_states[token, :]
        src_offset = token * hidden_size + h
        hs = tl.load(hidden_ptr + src_offset, mask=mask_h, other=0.0)

        # Compute destination offsets: expert_inputs[e, within, h]
        dest_offset = e * capacity * hidden_size + within * hidden_size + h
        tl.store(expert_inputs_ptr + dest_offset, hs, mask=mask_h & valid)


# Triton batched matmul: gate_out = expert_inputs @ expert_gate_weights^T
# Inputs:
#   expert_inputs: [num_experts, M, K] (bf16), M = capacity, K = hidden_size
#   expert_gate_weights: [num_experts, hidden_size, N] (bf16), N = moe_intermediate_size
#   gate_out: [num_experts, M, N] (bf16)
@triton.jit
def bmm_gate_kernel(
    expert_inputs_ptr, gate_w_ptr, gate_out_ptr,
    num_experts: tl.constexpr, M: tl.constexpr, K: tl.constexpr, N: tl.constexpr,
    stride_e_in, stride_m_in, stride_k_in,
    stride_e_w, stride_k_w, stride_n_w,
    stride_e_out, stride_m_out, stride_n_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    e = tl.program_id(0)  # expert id
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.bfloat16)

    # loop over K dimension
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # A: expert_inputs[e, offs_m, offs_k]
        a_ptrs = expert_inputs_ptr + e * stride_e_in + offs_m[:, None] * stride_m_in + offs_k[None, :] * stride_k_in
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.bfloat16)

        # B: gate_w[e, offs_k, offs_n] -> transpose is offs_k for rows, offs_n for cols
        b_ptrs = gate_w_ptr + e * stride_e_w + offs_k[:, None] * stride_k_w + offs_n[None, :] * stride_n_w
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.bfloat16)

        # acc += A @ B
        acc += tl.dot(a, b)

    # store result
    out_ptrs = gate_out_ptr + e * stride_e_out + offs_m[:, None] * stride_m_out + offs_n[None, :] * stride_n_out
    tl.store(out_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# Triton batched matmul: up_out = expert_inputs @ expert_up_weights^T

# Elementwise SiLU and multiply: activated = SiLU(gate_out) * up_out

def silu_mul_kernel(

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    e = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    for m0 in range(0, M, BLOCK_M):
        for n0 in range(0, N, BLOCK_N):
            m = m0 + offs_m
            n = n0 + offs_n
            mask_m = m < M
            mask_n = n < N
            mask = mask_m[:, None] & mask_n[None, :]

            g_ptrs = gate_out_ptr + e * stride_e_g + m[:, None] * stride_m_g + n[None, :] * stride_n_g
            u_ptrs = up_out_ptr + e * stride_e_u + m[:, None] * stride_m_u + n[None, :] * stride_n_u

            u = tl.load(u_ptrs, mask=mask, other=0.0).to(tl.bfloat16)

            # SiLU: x * sigmoid(x) = x * (1 / (1 + exp(-x)))
            sig = 1.0 / (1.0 + tl.exp(-g))
            out = g * sig * u

            a_ptrs = activated_ptr + e * stride_e_a + m[:, None] * stride_m_a + n[None, :] * stride_n_a
            tl.store(a_ptrs, out, mask=mask)
