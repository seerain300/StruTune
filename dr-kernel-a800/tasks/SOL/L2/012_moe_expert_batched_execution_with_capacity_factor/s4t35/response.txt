# Triton kernel: per-row matmul for A_row (length H) times B (H x M) producing C[H].
# We process one row of A via 1D grid; B is provided as a contiguous [H, M] tensor.
@triton.jit
def triton_row_matmul(C_ptr, A_ptr, B_ptr,
                       H: tl.constexpr, M: tl.constexpr,
                       BLOCK: tl.constexpr):
    # Only one program instance; it handles the entire row. For generality, we can
    # implement a loop over columns with BLOCK size, but here H and M are moderate.
    # We'll use a loop over columns to cover M.
    # Load A_row as a vector of length H.
    # Note: A_ptr is a 1D pointer, we read indices [0..H-1].
    # Triton requires static for-loops with constexpr bounds.
    # We will iterate over column blocks.
    for col_start in range(0, M, BLOCK):
        offs = col_start + tl.arange(0, BLOCK)
        acc = tl.zeros([BLOCK], dtype=tl.float32)
        # Compute dot product of A_row with corresponding B columns
        for i in range(0, H):
            a = tl.load(A_ptr + i)
            b = tl.load(B_ptr + i * M + offs)
            acc += a * b
        # Store acc to C
        tl.store(C_ptr + offs, acc, mask=offs < M)


# Triton elementwise SiLU: y = x * sigmoid(x) for vector X
@triton.jit
def triton_silu(X_ptr, Y_ptr, N: tl.constexpr, BLOCK: tl.constexpr):

        offs = start + tl.arange(0, BLOCK)
        x = tl.load(X_ptr + offs)
        # sigmoid(x) = 1 / (1 + exp(-x))
        sig = 1.0 / (1.0 + tl.exp(-x))
        y = x * sig
        tl.store(Y_ptr + offs, y, mask=offs < N)

    H = A_row_1d.numel()

    triton_row_matmul[(1,)](C, A_row_1d, B_mat, H=H, M=M, BLOCK=128)

# Triton kernel: per-row matmul for A_row (length H) times B (H x M) producing C[H].
# A_row is a 1D tensor, B is a 2D tensor. We process one row of A via a simple loop.
@triton.jit
def triton_row_matmul(C_ptr, A_ptr, B_ptr,
                       H: tl.constexpr, M: tl.constexpr, BLOCK: tl.constexpr):
    # We iterate over columns in blocks and accumulate dot products.
    for col_start in range(0, M, BLOCK):
        offs = col_start + tl.arange(0, BLOCK)
        acc = tl.zeros([BLOCK], dtype=tl.float32)
        for i in range(0, H):
            a = tl.load(A_ptr + i)
            b = tl.load(B_ptr + i * M + offs)
            acc += a * b
        tl.store(C_ptr + offs, acc, mask=offs < M)


# Triton elementwise SiLU: y = x * sigmoid(x) for vector X
@triton.jit
def triton_silu(X_ptr, Y_ptr, N: tl.constexpr, BLOCK: tl.constexpr):

        offs = start + tl.arange(0, BLOCK)
        x = tl.load(X_ptr + offs)
        sig = 1.0 / (1.0 + tl.exp(-x))
