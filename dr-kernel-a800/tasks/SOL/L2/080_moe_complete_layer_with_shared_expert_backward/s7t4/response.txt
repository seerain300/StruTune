)

    A_ptr, out_ptr,
    M, N,
    stride_am, stride_an,
    stride_out,
    BLOCK_K: tl.constexpr,
):
    row = tl.program_id(0)  # one program per row
    acc = 0.0
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
