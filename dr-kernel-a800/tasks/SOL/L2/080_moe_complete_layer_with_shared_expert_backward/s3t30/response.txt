import triton.language as tl


# Triton kernel: row-wise squared norm: Out[m] = sum_k grad_output[m, k]^2
@triton.jit
def reduce_sum_sq_kernel(
    A_ptr,            # *fp32 or *bf16, but we load as fp32 for stable accumulation
    Out_ptr,          # *fp32
    M: tl.int32,      # number of rows
    N: tl.int32,      # number of cols
    BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    tl.store(Out_ptr + m, 0.0)  # initialize
    # Loop over columns in chunks
    for n_start in range(0, N, BLOCK_N):
        cols = n_start + tl.arange(0, BLOCK_N)
        mask = cols < N
        # Load row m for cols, cast to float32 for accumulation
        a = tl.load(A_ptr + m * N + cols, mask=mask, other=0.0)
        a = a.to(tl.float32)
        sq = a * a
        tl.store(Out_ptr + m, tl.load(Out_ptr + m) + tl.sum(sq, axis=0))


# Triton kernel: dot product per column: Out[k] = sum_m A[m, k] * B[m]
# We need a separate variant for producing matrices: e.g., grad_shared_expert_gate_weight
# where A has shape [M, K], B has shape [M], and Out has shape [K].
# Here we implement the per-k computation vectorized; grid=(K,), each program handles one column.
@triton.jit
def dot_product_weight_grad_kernel(
    A_ptr,    # *bf16 or *fp32, pointer to [M, K]
    B_ptr,    # *bf16 or *fp32, pointer to [M]
    Out_ptr,  # *fp32, pointer to [K]
    M: tl.int32,
    K: tl.int32,
    BLOCK_M: tl.constexpr,
):
    k = tl.program_id(0)
    sum_val = 0.0
    # Loop over rows in chunks
    for m_start in range(0, M, BLOCK_M):
        rows = m_start + tl.arange(0, BLOCK_M)
        mask = rows < M

        a = a.to(tl.float32)
        b = b.to(tl.float32)

# Triton kernel: GEMV-like per-output-column dot product:
# Given A[M, K] and B[M], compute Out[K] where Out[k] = sum_{i=0..M-1} A[i, k] * B[i]
# We implement grid=(K,), each program handles one k. Loop over M in chunks (BLOCK_M).
@triton.jit
def dot_product_weight_grad_kernel(
    A_ptr,    # *bf16 or *fp32, points to [M, K]
    B_ptr,    # *bf16 or *fp32, points to [M]
    Out_ptr,  # *fp32 or *bf16, points to [K]
    M: tl.int32,
    K: tl.int32,
    BLOCK_M: tl.constexpr,
):
    k = tl.program_id(0)
    # Accumulator in fp32
    acc = 0.0
    # Loop over rows in chunks
    for m_start in range(0, M, BLOCK_M):
        rows = m_start + tl.arange(0, BLOCK_M)
        mask = rows < M
        # Load A rows for column k and B rows; mask out-of-bounds
        a = tl.load(A_ptr + rows * K + k, mask=mask, other=0.0)  # shape [BLOCK_M]
        b = tl.load(B_ptr + rows, mask=mask, other=0.0)          # shape [BLOCK_M]
        # Accumulate in fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.sum(a * b, axis=0)
    # Store as fp32 (we'll cast to bf16 in host if needed). For safety, we store fp32 and cast later.
    tl.store(Out_ptr + k, acc)

        rows = m_start + tl.arange(0, BLOCK_M)
        mask = rows < M
        a = tl.load(A_ptr + rows * K + k, mask=mask, other=0.0)  # [BLOCK_M]
        b = tl.load(B_ptr + rows, mask=mask, other=0.0)          # [BLOCK_M]
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.sum(a * b, axis=0)
    tl.store(Out_ptr + k, acc)


# Triton kernel: per-row squared norm of a matrix: Out[i] = sum_j A[i, j]^2
# A is [M, N], Out is [M]. We use grid=(M,), loop over N in chunks (BLOCK_N).
@triton.jit
def reduce_sum_sq_kernel(
    A_ptr,    # *bf16 or *fp32, points to [M, N]
    Out_ptr,  # *fp32, points to [M]
    M: tl.int32,
    N: tl.int32,
    BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    sum_val = 0.0
    for n_start in range(0, N, BLOCK_N):
        cols = n_start + tl.arange(0, BLOCK_N)
        mask = cols < N
        a = tl.load(A_ptr + m * N + cols, mask=mask, other=0.0)
        a = a.to(tl.float32)
        sq = a * a
        sum_val += tl.sum(sq, axis=0)
    tl.store(Out_ptr + m, sum_val)
