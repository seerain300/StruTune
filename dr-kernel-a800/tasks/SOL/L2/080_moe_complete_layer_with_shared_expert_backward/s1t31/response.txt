# GEMV for N=1408: y[b, e] = sum_h hidden[b, h] * W[e, h]
# hidden: [B, H] (input dtype), W: [N, H] (input dtype), y: [B, N] (float32 output)
@triton.jit
def gemv_linear_1408_kernel(
    hidden_ptr,   # *T, [B, H]
    W_ptr,        # *T, [N=1408, H]
    y_ptr,        # *f32, [B, N=1408]
    B: tl.constexpr,
    H: tl.constexpr,        # H = hidden_size (e.g., 4096)
    N: tl.constexpr,        # N = output size (1408)
    stride_h_b, stride_h_h,
    stride_W_e, stride_W_h,
    stride_y_b, stride_y_e,
    BLOCK_H: tl.constexpr,  # e.g., 256
):
    pid_b = tl.program_id(0)  # batch row index
    pid_e = tl.program_id(1)  # output column index (e in [0, N))
    acc = 0.0
    # Loop over hidden dimension in chunks
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # Load hidden vector segment for this batch row
        h_vals = tl.load(hidden_ptr + pid_b * stride_h_b + offs_h * stride_h_h, mask=mask_h, other=0.0)
        # Load weight vector segment for this output index
        w_vals = tl.load(W_ptr + pid_e * stride_W_e + offs_h * stride_W_h, mask=mask_h, other=0.0)
        # Accumulate dot product in f32
        acc += tl.sum(h_vals.to(tl.float32) * w_vals.to(tl.float32), axis=0)
    # Store as float32
    tl.store(y_ptr + pid_b * stride_y_b + pid_e * stride_y_e, acc)


# GEMV for N=4096: y[b, h] = sum_t a[b, t] * down[h, t]
# a: [B, M] (M=1408), down: [H=4096, M], y: [B, H] (float32)

def gemv_down_4096_kernel(

        down_vals = tl.load(down_ptr + pid_h * stride_down_h + offs_m * stride_down_m, mask=mask_m, other=0.0)
        acc += tl.sum(a_vals * down_vals.to(tl.float32), axis=0)
    tl.store(y_ptr + pid_b * stride_y_b + pid_h * stride_y_h, acc)


# Elementwise: y = silu(a) * b on flat vectors
# a: [B, N] (f32), b: [B, N] (f32), y: [B, N] (f32)
@triton.jit
def silu_mul_elemwise_kernel(a_ptr, b_ptr, y_ptr, total_elems: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    tl.store(y_ptr + offs, y, mask=mask)
