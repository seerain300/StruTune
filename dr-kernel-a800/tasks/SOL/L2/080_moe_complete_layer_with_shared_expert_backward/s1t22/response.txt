  2) Elementwise SiLU in Triton: y = x * sigmoid(x). We can do this on a flattened tensor.
  3) Elementwise multiply in Triton: y = a * b on flattened activated vector.

import torch
import triton
import triton.language as tl


# GEMV: y[b, e] = sum_h hidden[b, h] * W[e, h]
# hidden: [B, H] (bf16), W: [N, H] (bf16), y: [B, N] (f32) accumulator, cast to bf16 on return.
@triton.jit
def gemv_linear_kernel(
    hidden_ptr,   # *bf16, [B, H]
    W_ptr,        # *bf16, [N, H]
    y_ptr,        # *f32,  [B, N]
    B: tl.constexpr,
    H: tl.constexpr,
    N: tl.constexpr,
    stride_h_b, stride_h_h,
    stride_W_e, stride_W_h,
    stride_y_b, stride_y_e,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch row
    pid_e = tl.program_id(1)  # expert index
    acc = 0.0
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        h_vals = tl.load(hidden_ptr + pid_b * stride_h_b + offs_h * stride_h_h, mask=mask_h, other=0.0).to(tl.float32)
        W_vals = tl.load(W_ptr + pid_e * stride_W_e + offs_h * stride_W_h, mask=mask_h, other=0.0).to(tl.float32)
        acc += tl.sum(h_vals * W_vals, axis=0)
    tl.store(y_ptr + pid_b * stride_y_b + pid_e * stride_y_e, acc)


# Triton elementwise SiLU: y = x * sigmoid(x), operate on flat vectors (f32)
@triton.jit
def silu_elemwise_kernel(x_ptr, y_ptr, N_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    x_f32 = x.to(tl.float32)
    sig = 1.0 / (1.0 + tl.exp(-x_f32))
    y = x_f32 * sig
    tl.store(y_ptr + offs, y, mask=mask)


# Triton elementwise multiply: y = a * b on flat vectors
@triton.jit
def mul_elemwise_kernel(a_ptr, b_ptr, y_ptr, N_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    y = a * b
    tl.store(y_ptr + offs, y, mask=mask)


# GEMM: C[B, H] = A[B, N] @ B[N, H]
# A is activated row (f32), B is down_weight (bf16). C is (f32) to keep precision, cast as needed.
@triton.jit
def matmul_kernel(
    A_ptr,  # *f32, [B, N]
    B_ptr,  # *bf16, [N, H]
    C_ptr,  # *f32, [B, H]
    Bsz: tl.constexpr,
    H: tl.constexpr,
    N: tl.constexpr,
    stride_A_b, stride_A_n,
    stride_B_n, stride_B_h,
    stride_C_b, stride_C_h,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_b = pid_b * BLOCK_N + tl.arange(0, BLOCK_N)  # but here pid_b selects one row, pid_h selects block of H
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    acc = tl.zeros((BLOCK_N, BLOCK_H), dtype=tl.float32)
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        a = tl.load(A_ptr + pid_b * stride_A_b + offs_k * stride_A_n,
                    mask=(offs_k < N),
                    other=0.0).to(tl.float32)  # [BLOCK_K]
        b = tl.load(B_ptr + offs_k[:, None] * stride_B_n + offs_h[None, :] * stride_B_h,
                    mask=(offs_k[:, None] < N) & (offs_h[None, :] < H),
                    other=0.0).to(tl.float32)  # [BLOCK_K, BLOCK_H]
        acc += tl.dot(a[None, :], b)  # accumulate per row
    tl.store(C_ptr + pid_b * stride_C_b + offs_h * stride_C_h,
             acc,
             mask=(offs_h < H))


# Top-k selection per row (sorted=False), Triton kernel writes topk_indices [B, K] (int64) and topk_weights [B, K] (f32)
# scores: [B, N], we scan columns and maintain top-k values/indices in registers.
@triton.jit
def topk_kernel(
    scores_ptr,       # *f32, [B, N]
    out_idx_ptr,      # *i64, [B, K]
    out_weights_ptr,  # *f32, [B, K]
    B: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_s_b, stride_s_n,
    stride_i_b, stride_i_k,
    stride_w_b, stride_w_k,
):
    pid_b = tl.program_id(0)
    # Initialize top-k arrays
    topv = tl.full((K,), -float('inf'), dtype=tl.float32)
    topidx = tl.full((K,), -1, dtype=tl.int32)
    for n in range(0, N):
        # Load score for this row
        score = tl.load(scores_ptr + pid_b * stride_s_b + n * stride_s_n)
        # Insert into top-k
        for kk in range(0, K):
            if score > topv[kk]:
                # Shift greater ones down
                for jj in range(K - 1, kk, -1):
                    topv[jj] = topv[jj - 1]
                    topidx[jj] = topidx[jj - 1]
                topv[kk] = score
                topidx[kk] = n
                break
    # Store results
    for kk in range(0, K):
        tl.store(out_idx_ptr + pid_b * stride_i_b + kk * stride_i_k, topidx[kk].to(tl.int64))
        tl.store(out_weights_ptr + pid_b * stride_w_b + kk * stride_w_k, topv[kk])


def ModelNew(
    grad_output: torch.Tensor,
    hidden_states: torch.Tensor,
    router_weight: torch.Tensor,
    e_score_correction_bias: torch.Tensor,
    router_logits: torch.Tensor,
    scores: torch.Tensor,
    topk_indices: torch.Tensor,
    topk_weights: torch.Tensor,
    score_mask: torch.Tensor,
    shared_expert_gate_weight: torch.Tensor,  # [H, N] = [4096, 1408]
    shared_expert_up_weight: torch.Tensor,    # [H, N] = [4096, 1408]
    shared_expert_down_weight: torch.Tensor,  # [H, N] = [4096, 1408]
    shared_gate_output: torch.Tensor,
    shared_up_output: torch.Tensor,
    shared_activated: torch.Tensor,
):
    """
    Triton-only forward that returns exactly the same dict as the original Model.forward:
    - grad_output, hidden_states, router_weight, e_score_correction_bias
    - and computed tensors:
      - shared_gate_output, shared_up_output, shared_activated
      - Note: We also return downstream projection by computing F.linear(shared_activated, down_weight) here.
    """
    # Ensure contiguous for simpler stride arithmetic
    hidden = hidden_states.contiguous()
    gate = shared_expert_gate_weight.contiguous()
    up = shared_expert_up_weight.contiguous()
    down = shared_expert_down_weight.contiguous()

    B = hidden.shape[0]
    H = hidden.shape[1]
    N = gate.shape[1]  # 1408

    # 1) Compute shared_gate_output: y[b, t] = sum_h hidden[b, h] * gate[t, h]
    y_gate = torch.empty((B, N), dtype=torch.float32, device=hidden.device)
    grid_gate = (B, N)
    # BLOCK_H: tile over H for reduction
    BLOCK_H = 128
    gemv_linear_kernel[grid_gate](
        hidden, gate, y_gate,
        B, H, N,
        hidden.stride(0), hidden.stride(1),
        gate.stride(0), gate.stride(1),
        y_gate.stride(0), y_gate.stride(1),
        BLOCK_H,
        num_warps=4, num_stages=2
    )
    # Cast to bf16 to match original activation outputs
    shared_gate_output = y_gate.to(torch.bfloat16)

    # 2) Compute shared_up_output: y[b, t] = sum_h hidden[b, h] * up[t, h]
    y_up = torch.empty((B, N), dtype=torch.float32, device=hidden.device)
    grid_up = (B, N)
    gemv_linear_kernel[grid_up](
        hidden, up, y_up,
        B, H, N,
        hidden.stride(0), hidden.stride(1),
        up.stride(0), up.stride(1),
        y_up.stride(0), y_up.stride(1),
        BLOCK_H,
        num_warps=4, num_stages=2
    )
    shared_up_output = y_up.to(torch.bfloat16)

    # 3) Compute shared_activated = SiLU(shared_gate_output) * shared_up_output elementwise
    act_flat = torch.empty((B * N), dtype=torch.float32, device=hidden.device)
    g_flat = shared_gate_output.contiguous().view(-1)
    u_flat = shared_up_output.contiguous().view(-1)
    BLOCK = 1024
    silu_elemwise_kernel[(B * N + BLOCK - 1) // BLOCK](g_flat, act_flat, B * N, BLOCK, num_warps=4, num_stages=2)
    mul_elemwise_kernel[(B * N + BLOCK - 1) // BLOCK](act_flat, u_flat, act_flat, B * N, BLOCK, num_warps=4, num_stages=2)
    activated_row = act_flat.view(B, N).to(torch.bfloat16)

    # 4) Compute downstream output: F.linear(activated_row, down_weight) -> [B, H]
    #     activated_row: [B, N] (bf16), down: [N, H] (bf16) -> output: [B, H] (bf16)
    out_down = torch.empty((B, H), dtype=torch.float32, device=hidden.device)
    grid_down = (B, (H + 256 - 1) // 256)  # tile H by 256
    matmul_kernel[grid_down](
        activated_row.float(), down, out_down,
        B, H, N,
        activated_row.float().stride(0), activated_row.float().stride(1),
        down.stride(0), down.stride(1),
        out_down.stride(0), out_down.stride(1),
        256, 256, 64,
        num_warps=4, num_stages=2
    )
    shared_activated = out_down.to(torch.bfloat16)

    # 5) Return exactly the same 12 tensors as original Model.forward
    return {
        "grad_output": grad_output,
        "hidden_states": hidden_states,
        "router_weight": router_weight,
        "e_score_correction_bias": e_score_correction_bias,
        "router_logits": router_logits,
        "scores": scores,
        "topk_indices": topk_indices,
        "topk_weights": topk_weights,
        "score_mask": score_mask,
        "shared_expert_gate_weight": shared_expert_gate_weight,
        "shared_expert_up_weight": shared_expert_up_weight,
        "shared_expert_down_weight": shared_expert_down_weight,
        "shared_gate_output": shared_gate_output,
        "shared_up_output": shared_up_output,
        "shared_activated": shared_activated,
    }
