import triton
import triton.language as tl

@triton.jit
def triton_fill_normal(out_ptr, n_elements: tl.constexpr):

    def __init__(self):
        super().__init__()

    def forward(self, device: torch.device, axes_and_scalars: dict):
        # Extract dimensions from axes
        batch_seq_len = int(axes_and_scalars.get("batch_seq_len", 1))
        hidden_size = 4096
        E = 128  # number of routed experts
        K = 8    # top-k per token

        triton.run(triton_fill_normal, hidden_ptr.numel(), hidden_ptr.numel())
