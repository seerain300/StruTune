import torch
import triton
import triton.language as tl


# Triton RNG: fill buffer with random values (approximate N(0,1)) using a simple LCG.
@triton.jit
def triton_fill_normal(out_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = 1664525
    c = 1013904223
    m = 4294967296  # 2^32
    seed = 1234567
    z = seed
    for i in range(BLOCK):
        z = (z * a + c) & (m - 1)
        r = tl.float32(z) / m
        # crude approximation of N(0,1): r * sqrt(12) - 6
        val = r * 2.23606797749979 - 6.0
        tl.store(out_ptr + offs + i, val, mask=mask)


# GEMV: out[b, m] = dot(X[b, :], Wt[m, :]) where
#   X: [B, K] row-major (K = hidden_size)
#   Wt: [M, K] row-major (M = n_routed_experts), actually a transpose of original router_weight
#   Out: [B, M] row-major
@triton.jit
def gemv_out(X_ptr, Wt_ptr, Out_ptr,
             B: tl.constexpr, K: tl.constexpr, M: tl.constexpr,
             stride_xb, stride_xk,
             stride_wtm, stride_wtk,
             stride_ob, stride_om):
    b = tl.program_id(0)
    m = tl.program_id(1)
    acc = 0.0
    for k_start in range(0, K, 128):
        offs_k = k_start + tl.arange(0, 128)
        mask_k = offs_k < K
        x = tl.load(X_ptr + b * stride_xb + offs_k * stride_xk, mask=mask_k, other=0.0)
        wt = tl.load(Wt_ptr + m * stride_wtm + offs_k * stride_wtk, mask=mask_k, other=0.0)
        acc += tl.sum(x * wt, axis=0)
    tl.store(Out_ptr + b * stride_ob + m * stride_om, acc)


# Elementwise sigmoid: y = 1 / (1 + exp(-x))
@triton.jit
def triton_sigmoid(x_ptr, y_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(y_ptr + offs, y, mask=mask)


# Elementwise SiLU: y = x * sigmoid(x)
@triton.jit
def triton_silu(x_ptr, y_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + offs, y, mask=mask)
