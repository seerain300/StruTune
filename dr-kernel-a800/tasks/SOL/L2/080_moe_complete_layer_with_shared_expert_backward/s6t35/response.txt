import triton
import triton.language as tl


@triton.jit
def linear_rowwise_bf16_to_f32(
    X_ptr,  # *bfloat16, shape [M, K]
    W_ptr,  # *bfloat16, shape [K, N]
    Y_ptr,  # *float32,  shape [M, N]
    M: tl.constexpr,  # number of rows in X
    K: tl.constexpr,  # number of columns in X / rows in W
    N: tl.constexpr,  # number of columns in W / output width
    stride_xm,  # stride of X along rows
    stride_xk,  # stride of X along cols
    stride_wk,  # stride of W along rows (K dim)
    stride_wn,  # stride of W along cols (N dim)
):
    # Each program computes one row m
    m = tl.program_id(axis=0)
    if m >= M:
        return

    # Accumulator for this row across N columns
    acc = tl.zeros((N,), dtype=tl.float32)

    # Loop over K in chunks of BLOCK_K (e.g., 64)
    BLOCK_K = 64
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load x[m, offs_k] as bfloat16; if M is small, masks handle bounds
        x_vals = tl.load(
            X_ptr + m * stride_xm + offs_k * stride_xk,
            mask=mask_k,
            other=0.0
        )  # shape [BLOCK_K], dtype bfloat16

        # Accumulate contributions for all N columns
        for n in range(0, N):
            # Load W[offs_k, n] as bfloat16; vectorized along K
            w_vec = tl.load(
                W_ptr + offs_k * stride_wk + n * stride_wn,
                mask=mask_k,
                other=0.0
            )  # shape [BLOCK_K], dtype bfloat16

            # acc[n] += sum_k x_vals[k] * w_vec[k]
            acc[n] += tl.sum(x_vals.to(tl.float32) * w_vec.to(tl.float32), axis=0)

    # Store acc to Y[m, :]
    offs_n = tl.arange(0, N)
    tl.store(Y_ptr + m * N + offs_n, acc, mask=offs_n < N)


@triton.jit
def silu_mul_kernel(
    GateOut_ptr,  # *float32, shape [M, N]
    Up_ptr,       # *float32, shape [M, N]
    Y_ptr,        # *float32, shape [M, N]
    M: tl.constexpr,
    N: tl.constexpr,
    stride_gm, stride_gn,
    stride_um, stride_un,
    stride_ym, stride_yn,
):
    m = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    if m >= M or n >= N:
        return

    gate = tl.load(GateOut_ptr + m * stride_gm + n * stride_gn)
    up = tl.load(Up_ptr + m * stride_um + n * stride_un)

    # y = gate * sigmoid(gate) * up
    sig = 1.0 / (1.0 + tl.exp(-gate))
    y = gate * sig * up

    tl.store(Y_ptr + m * stride_ym + n * stride_yn, y)

        gate_weight = shared_expert_gate_weight.contiguous()  # [K, N], bfloat16
        up_weight = shared_expert_up_weight.contiguous()      # [K, N], bfloat16

        M, K = hidden_states.shape
        K_w, N = gate_weight.shape
        assert K == K_w, f"hidden_states K={K} must match gate_weight rows K={K_w}"
        assert hidden_states.dtype == torch.bfloat16
        assert gate_weight.dtype == torch.bfloat16
        assert up_weight.dtype == torch.bfloat16

        # Output buffers (float32 for accumulation)
        gate_out = torch.empty((M, N), dtype=torch.float32, device=hidden_states.device)
        up_out = torch.empty((M, N), dtype=torch.float32, device=hidden_states.device)

        )

        silu_mul_kernel[grid2](

            up_out.stride(0), up_out.stride(1),

- 计算结果与原来的 PyTorch 逻辑一致：gate_out 通过 F.linear(hidden_states, gate_weight.T, bias=None)，up_out 通过 F.linear(hidden_states, up_weight.T, bias=None)，然后 y = gate * sigmoid(gate) * up。

一旦确认此版本通过所有 workloads，我们可以进一步优化性能：
- 用 tiled matmul (2D tiling) 和 tl.dot 来加速 GEMV，并考虑 FP16/FP32 mixed precision accumulation.
- 将 SiLU 计算与 GEMV 结果直接结合，减少中间 tensors 的内存传输.
- 增加 num_warps/num_stages 以提升并行度.
