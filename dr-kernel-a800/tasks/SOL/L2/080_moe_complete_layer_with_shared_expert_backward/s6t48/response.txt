@triton.jit
def gemv_rowwise_bf16_to_f32(
    X_ptr,   # *bfloat16, shape [M, K]
    W_ptr,   # *bfloat16, shape [K, N]
    Y_ptr,   # *float32,  shape [M, N]
    M, K, N,  # int32
    stride_xm, stride_xk,  # strides for X
    stride_wk, stride_wn,  # strides for W
    BLOCK_K: tl.constexpr,  # tile size over K
    BLOCK_N: tl.constexpr,  # output N vector chunk
):
    # Each program handles one row m
    m = tl.program_id(0)

    # Initialize output accumulator vector acc[N] in float32
    n_offsets = tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K in chunks
    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        # x[m, k] as bfloat16, masked for k < K
        x_ptrs = X_ptr + m * stride_xm + k_offsets * stride_xk
        x_vec = tl.load(x_ptrs, mask=(k_offsets < K), other=0.0)

        # w[k, n] as bfloat16, masked for k < K and n < N
        w_ptrs = W_ptr + k_offsets[:, None] * stride_wk + n_offsets[None, :] * stride_wn
        w_tile = tl.load(w_ptrs, mask=((k_offsets[:, None] < K) & (n_offsets[None, :] < N)), other=0.0)

        # Promote x_vec to float32 and multiply with w_tile, reduce over k to accumulate acc[n]
        x_f32 = x_vec.to(tl.float32)
        # We need acc[n] += sum_k (x_f32[k] * w_tile[k, n])
        # Implement reduction manually:
        # Note: Triton does not broadcast multiply with 1D and 2D like numpy; we loop over k chunk.
        # Create a loop over k within the chunk. Triton allows such constructs for small BLOCK_K.
        # However, Triton does not support Python for-loops over runtime ranges directly inside kernel.
        # Instead, we use tl.dot for reduction. To do so, reshape w_tile to [BLOCK_K, BLOCK_N] and use tl.dot(x_f32[:, None], w_tile)[None, :] pattern.
        # Since x_vec is [BLOCK_K], we can compute contrib = x_f32[:, None] * w_tile and then reduce.

            acc += x_k * w_sub.to(tl.float32)

    # Define stride_y0 and stride_yn via parameters:

@triton.jit
def gemv_rowwise_bf16_to_f32(
    X_ptr,   # *bfloat16, shape [M, K]
    W_ptr,   # *bfloat16, shape [K, N]
    Y_ptr,   # *float32,  shape [M, N]
    M, K, N,  # int32
    stride_xm, stride_xk,  # strides for X
    stride_wk, stride_wn,  # strides for W
    stride_y0, stride_yn,  # strides for Y
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    m = tl.program_id(0)
    n_offsets = tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        x_ptrs = X_ptr + m * stride_xm + k_offsets * stride_xk
        x_vec = tl.load(x_ptrs, mask=(k_offsets < K), other=0.0)

        w_ptrs = W_ptr + k_offsets[:, None] * stride_wk + n_offsets[None, :] * stride_wn
        w_tile = tl.load(w_ptrs, mask=((k_offsets[:, None] < K) & (n_offsets[None, :] < N)), other=0.0)

        x_f32 = x_vec.to(tl.float32)
        # For each kk in chunk, accumulate acc += x_f32[kk] * w_tile[kk, :]
        for kk in range(BLOCK_K):
            k_idx = k0 + kk
            x_k = x_f32[kk]  # scalar float32
            w_sub = w_tile[kk, :]  # vector of size BLOCK_N, bfloat16
            acc += x_k * w_sub.to(tl.float32)

    # Store acc to Y[m, n_offsets]
    y_ptrs = Y_ptr + m * stride_y0 + n_offsets * stride_yn
    tl.store(y_ptrs, acc, mask=(n_offsets < N))


@triton.jit
def silu_mul_kernel(
    X_ptr,  # *float32, shape [M, N]
    U_ptr,  # *float32, shape [M, N]
    Y_ptr,  # *float32, shape [M, N]
    M, N,   # ints
    stride_x0, stride_x1,
    stride_u0, stride_u1,
    stride_y0, stride_y1,
):
    # 2D grid: (M, N)
    m = tl.program_id(0)
    n = tl.program_id(1)
    # bounds check
    if m >= M or n >= N:
        return
    x = tl.load(X_ptr + m * stride_x0 + n * stride_x1)
    u = tl.load(U_ptr + m * stride_u0 + n * stride_u1)
    # SiLU(x) = x * sigmoid(x)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig * u
    tl.store(Y_ptr + m * stride_y0 + n * stride_y1, y)

    def forward(self, hidden_states, shared_expert_gate_weight, shared_expert_up_weight):
        """
        Triton-only forward that computes:
          gate_out = hidden_states @ shared_expert_gate_weight.T
          up_out   = hidden_states @ shared_expert_up_weight.T
          shared_activated = SiLU(gate_out) * up_out
        Returns bfloat16 tensor of shape [batch_seq_len, N].
        """
        assert hidden_states.is_cuda and shared_expert_gate_weight.is_cuda and shared_expert_up_weight.is_cuda, "Tensors must be CUDA"
        assert hidden_states.dtype == torch.bfloat16, "hidden_states must be bfloat16"
        assert shared_expert_gate_weight.dtype == torch.bfloat16 and shared_expert_up_weight.dtype == torch.bfloat16, "Weight tensors must be bfloat16"

        # Shapes
        M, K = hidden_states.shape
        Kw, N = shared_expert_gate_weight.shape
        Ku, _ = shared_expert_up_weight.shape
        assert K == Kw == Ku, "Hidden K must match gate/up weight K"

        # Allocate outputs in float32 for numerical stability
        gate_out = torch.empty((M, N), dtype=torch.float32, device=hidden_states.device)
        up_out = torch.empty((M, N), dtype=torch.float32, device=hidden_states.device)

        # Launch GEMV kernels: one program per row m
        grid_gate = (M,)
        gemv_rowwise_bf16_to_f32[grid_gate](
            hidden_states, shared_expert_gate_weight, gate_out,
            M, K, N,
            hidden_states.stride(0), hidden_states.stride(1),
            shared_expert_gate_weight.stride(0), shared_expert_gate_weight.stride(1),
            gate_out.stride(0), gate_out.stride(1),
            BLOCK_K=256, BLOCK_N=128,
            num_warps=4, num_stages=2
        )

        grid_up = (M,)
        gemv_rowwise_bf16_to_f32[grid_up](
            hidden_states, shared_expert_up_weight, up_out,
            M, K, N,
            hidden_states.stride(0), hidden_states.stride(1),
            shared_expert_up_weight.stride(0), shared_expert_up_weight.stride(1),
            up_out.stride(0), up_out.stride(1),
            BLOCK_K=256, BLOCK_N=128,
            num_warps=4, num_stages=2
        )

        # Elementwise Triton kernel for SiLU(gate_out) * up_out
        # We launch a 2D grid over (M, N)
        M_out, N_out = M, N
        activated = torch.empty((M_out, N_out), dtype=torch.float32, device=hidden_states.device)

        )

        return activated.to(torch.bfloat16)
