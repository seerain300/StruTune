import triton.language as tl


# Random fill with normal distribution: out[i] = N(0,1)
# out_ptr: [N] contiguous, float32
@triton.jit
def triton_fill_normal(out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    # Triton doesn't provide a built-in RNG across versions; use tl.rand for per-element RNG.
    rnd = tl.rand(offs)  # uniform in [0,1)
    # Convert to normal via Box-Muller: N(0,1)
    angle = 2.0 * tl.pi * rnd
    rad = tl.sqrt(-2.0 * tl.log(1.0 - rnd))
    val = tl.sin(angle) * rad  # N(0,1)
    tl.store(out_ptr + offs, val, mask=mask)


# GEMV: out[b, m] = dot(hidden_states[b, :], W[m, :])
# Inputs:
#   - X: [B, K] (row-major), float32
#   - W: [M, K] (row-major), float32
#   - Out: [B, M] (row-major), float32
# Each Triton program handles one (b, m) pair and iterates over K in chunks of 128.
@triton.jit
def gemv_row(hidden_states_ptr, w_ptr, out_ptr,
             B, K, M,
             stride_xb, stride_xk,
             stride_wm, stride_wk,
             stride_ob, stride_om):
    b = tl.program_id(0)  # batch row index
    m = tl.program_id(1)  # output index in W
    # Initialize accumulator
    acc = 0.0
    # Loop over K dimension in chunks of 128
    for k_start in range(0, K, 128):
        offs_k = k_start + tl.arange(0, 128)
        mask_k = offs_k < K
        # Load X[b, offs_k] and W[m, offs_k]
        x = tl.load(hidden_states_ptr + b * stride_xb + offs_k * stride_xk, mask=mask_k, other=0.0)  # [128]
        w = tl.load(w_ptr + m * stride_wm + offs_k * stride_wk, mask=mask_k, other=0.0)            # [128]
        # Dot product for this chunk
        acc += tl.sum(x * w, axis=0)
    tl.store(out_ptr + b * stride_ob + m * stride_om, acc)

# x: float32 vector
# y: float32 vector (same shape as x)

    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(y_ptr + offs, y, mask=mask)


# Top-k per row for scores [B, N]: returns indices (int32) and values (float32)
# Inputs:
#   - scores_ptr: [B, N], float32
#   - idx_ptr: [B, K], int32
#   - val_ptr: [B, K], float32
# Outputs:
#   - idx_ptr: [B, K], int32
#   - val_ptr: [B, K], float32
# K is passed as a constexpr; N is runtime. We iterate K times, each time finding max and its index, and mask it to -inf.
@triton.jit
def triton_topk_row(scores_ptr, idx_ptr, val_ptr,
                    B, N,
                    stride_sb, stride_sn,
                    stride_ib, stride_iK,
                    stride_vb, stride_vk,
                    K: tl.constexpr):
    b = tl.program_id(0)
    # Iteratively select top-K
    for k in range(0, K):
        max_val = -float('inf')
        max_idx = 0
        # Scan N to find max
        for n in range(0, N):
            score = tl.load(scores_ptr + b * stride_sb + n * stride_sn)
            if score > max_val:
                max_val = score
                max_idx = n
        # Store k-th top value and index
        tl.store(idx_ptr + b * stride_ib + k * stride_iK, max_idx)
        tl.store(val_ptr + b * stride_vb + k * stride_vk, max_val)
        # Mask selected element to -inf for next iterations
        # Note: Writing -inf can be unsafe; but Triton will handle it in float32.
        tl.store(scores_ptr + b * stride_sb + max_idx * stride_sn, -float('inf'))


# SiLU elementwise: y = x * sigmoid(x)
# x: float32 vector
# y: float32 vector (same shape as x)
@triton.jit
def triton_silu(x_ptr, y_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + offs, y, mask=mask)
