    X_ptr,       # [N] bfloat16 input
    Out_ptr,     # [1] bfloat16 output (scalar)
    N: tl.constexpr,  # number of elements (compile-time for reduction simplification)
    BLOCK_SIZE: tl.constexpr
):
    """
    Sum all elements of X_ptr (bfloat16) and store as scalar (bfloat16).
    """
    acc = tl.zeros((), dtype=tl.bfloat16)
    # Loop over N in chunks of BLOCK_SIZE
    for i in range(0, N, BLOCK_SIZE):
        offs = i + tl.arange(0, BLOCK_SIZE)
        mask = offs < N
        x = tl.load(X_ptr + offs, mask=mask, other=0.0)
        acc += tl.sum(x.to(tl.float32), axis=0)  # accumulate in float32 for accuracy
    acc = acc.to(tl.bfloat16)
    tl.store(Out_ptr, acc)


@triton.jit
def dot_product_bf16_kernel(
    A_ptr,       # [M] bfloat16 vector
    B_ptr,       # [N] bfloat16 vector
    Out_ptr,     # [N] bfloat16 output
    M, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):

    Out[n] = sum_{m=0..M-1} A[m] * B[n], for n in [0, N).

    n_idx = pid * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_n = n_idx < N
    acc = tl.zeros((BLOCK_N,), dtype=tl.bfloat16)
