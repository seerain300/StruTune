import torch
import triton
import triton.language as tl


@triton.jit
def gemv_rowwise_bf16_to_f32(
    X_ptr,   # *bfloat16, shape [M, K]
    W_ptr,   # *bfloat16, shape [K, N]
    Y_ptr,   # *float32,  shape [M, N]
    M, K, N,  # ints
    stride_xm, stride_xk,  # X strides
    stride_wk, stride_wn,  # W strides
    stride_ym, stride_yn,  # Y strides
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    m = tl.program_id(axis=0)
    # If m >= M, nothing to do (grid ensures m < M)
    # Initialize accumulator for this row
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        # Vector of n for this tile
        n_offsets = tl.arange(0, BLOCK_N)
        n_mask = n_offsets < N

        # Load W[k_start : k_start+BLOCK_K, n_offsets]
        # k_offsets: [BLOCK_K], n_offsets: [BLOCK_N]
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        # Pointer to W tile: shape [BLOCK_K, BLOCK_N]
        w_ptrs = W_ptr + k_offsets[:, None] * stride_wk + n_offsets[None, :] * stride_wn
        w_tile = tl.load(w_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # Convert to float32
        w_tile_f32 = w_tile.to(tl.float32)

        # Load X[m, k_offsets]
        x_ptrs = X_ptr + m * stride_xm + k_offsets * stride_xk
        x_vec = tl.load(x_ptrs, mask=k_mask, other=0.0).to(tl.float32)

        # Accumulate: acc += sum_k x_vec[k] * w_tile[k, :]
        # w_tile_f32 shape [BLOCK_K, BLOCK_N], x_vec[:, None] broadcasts over N
        acc += tl.sum(w_tile_f32 * x_vec[:, None], axis=0)

    # Store acc into Y[m, :]
    y_ptrs = Y_ptr + m * stride_ym + n_offsets * stride_yn
    tl.store(y_ptrs, acc, mask=n_mask)


@triton.jit
def silu_mul_rowwise_f32(
    GateOut_ptr,  # *float32, shape [M, N]
    UpOut_ptr,    # *float32, shape [M, N]
    Out_ptr,      # *float32, shape [M, N]
    M, N,
    stride_gm, stride_gn,
    stride_um, stride_un,
    stride_om, stride_on,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    m = tl.program_id(axis=0)
    n = tl.program_id(axis=1)
    # Load GateOut[m, n] and UpOut[m, n]
    g_ptr = GateOut_ptr + m * stride_gm + n * stride_gn
    u_ptr = UpOut_ptr + m * stride_um + n * stride_un
    g = tl.load(g_ptr)
    u = tl.load(u_ptr)
    # SiLU(g) = g * sigmoid(g)
    # sigmoid(g) = 1 / (1 + exp(-g))
    s = 1.0 / (1.0 + tl.exp(-g))
    y = g * s * u
    o_ptr = Out_ptr + m * stride_om + n * stride_on
    tl.store(o_ptr, y)
