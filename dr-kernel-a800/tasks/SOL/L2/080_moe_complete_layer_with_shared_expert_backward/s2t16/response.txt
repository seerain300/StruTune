- The evaluator indicates 0/16 workloads are correct, and all failures are “RUNTIME_ERROR”. The likely causes:
  - Some Triton kernels in the previous attempt were misconfigured. A common source of runtime errors is using a non-constexpr n_elements for tl.arange (e.g., triton_sigmoid using n_elements as a runtime value). Triton expects the length of tl.arange to be a compile-time constant (tl.constexpr). If we pass runtime integers, Triton won’t generate the loop, leading to out-of-bounds loads/stores and crashes.
  - Another risk is missing or incorrect masks for loads/stores in GEMV and reductions, or using incorrect strides for row-major layouts. Even a single out-of-bounds access can cause runtime errors.
  - We need to ensure that all Triton kernels use tl.constexpr for their iteration sizes, and that we launch with grid sizes computed from these constexpr values. We also need to ensure we never use torch operations in forward (which we already avoid), but we must ensure that the Triton kernels are actually doing the core math (F.linear, elementwise ops, top-k), not leaving it to torch.

We’ll implement a per-row topk in Triton that performs K iterations, each scanning N elements, finding the max and its index, storing it, and then setting that element to -inf. We will support N=128 and K=8 for this problem. We’ll write indices as int32 and values as float32, then convert to the expected int64 and float32 tensors on the host.

import torch
import triton
import triton.language as tl


# GEMV kernel: out[b, m] = dot(hidden_states[b, :], W[m, :])
# Inputs:
#   - hidden_states: [B, K], row-major, float32
#   - W: [M, K], row-major, float32
#   - out: [B, M], row-major, float32
# Each Triton program handles one (b, m) pair and accumulates over K in chunks.
@triton.jit
def gemv_row(hidden_states_ptr, w_ptr, out_ptr,
             B: tl.constexpr, K: tl.constexpr, M: tl.constexpr,
             stride_xb, stride_xk,
             stride_wm, stride_wk,
             stride_ob, stride_om):
    b = tl.program_id(0)  # batch row index
    m = tl.program_id(1)  # output index in W
    acc = tl.zeros((), dtype=tl.float32)
    # Loop over K in chunks (BLOCK_K) with compile-time loop bounds
    BLOCK_K = 128  # must be constexpr; Triton will unroll
    for k_start in range(0, K, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        x = tl.load(hidden_states_ptr + b * stride_xb + offs_k * stride_xk, mask=mask_k, other=0.0)  # [BLOCK_K]
        w = tl.load(w_ptr + m * stride_wm + offs_k * stride_wk, mask=mask_k, other=0.0)            # [BLOCK_K]
        acc += tl.sum(x * w, axis=0)
    tl.store(out_ptr + b * stride_ob + m * stride_om, acc)


# Elementwise sigmoid: y = 1 / (1 + exp(-x)), for float32
@triton.jit
def triton_sigmoid(x_ptr, y_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(y_ptr + offs, y, mask=mask)


# Elementwise SiLU: y = x * sigmoid(x), for float32
@triton.jit
def triton_silu(x_ptr, y_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + offs, y, mask=mask)


# Top-k per row (N=128, K=8). Input row is float32[128], output indices int32[8], values float32[8].
@triton.jit
def triton_topk_row(values_ptr, indices_ptr, values_out_ptr, N: tl.constexpr, K: tl.constexpr):
    # Only one program per row is needed; we process the whole row. No grid needed; pass scalar.
    # We'll assume N == 128 and K == 8.
    # Initialize best[k] and idx[k]
    neg_inf = -1.0e30
    best = [neg_inf] * K
    idx = [0] * K
    # Current offsets we haven't yet chosen
    offs = tl.arange(0, N)  # [0..127]
    for kk in range(0, K):
        # Initialize current best for this iteration
        cur_best = neg_inf
        cur_idx = 0
        # Scan all N to find max
        for i in range(0, N):
            # Load scalar value
            val = tl.load(values_ptr + i)
            # Update current best
            if val > cur_best:
                cur_best = val
                cur_idx = i
        # Record this top selection
        best[kk] = cur_best
        idx[kk] = cur_idx
        # Mask out this element by setting its value to -inf
        tl.store(values_ptr + cur_idx, neg_inf)
    # Store results
    # Write values
    for kk in range(0, K):
        tl.store(values_out_ptr + kk, best[kk])
    # Write indices as int32
    for kk in range(0, K):
        tl.store(indices_ptr + kk, idx[kk])


# Random fill kernel (LCG) to populate tensors with N(0,1)-like values; we won't call this in forward since get_inputs is used,
# but it's shown here for reference if needed elsewhere. Forward should not use torch.randn, so we ensure we rely on Triton GEMV and elementwise kernels.
@triton.jit
def triton_fill_normal(buf_ptr, n_elements: tl.constexpr, BLOCK: tl.constexpr, seed: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    # Simple LCG: next = (a * x + c) % m
    # Map offsets to integers and generate a random float
    x = offs  # initial seed per offset (simple)
    a = 1664525
    c = 1013904223
    m = 4294967296  # 2**32
    # Generate random number
    # Note: Triton does not have %, but we can use bitwise operations for modulo
    # Implement modulo via floor division: rand = ((a * x + c) // m)
    rand_int = (a * x + c) // m
    # Convert to float in [0,1)
    rand = rand_int.to(tl.float32) / 4294967296.0
    # Gaussian approximation: Box-Muller transform using two uniforms (but we only have one here).
    # Use simpler: y = rand - 0.5, scale by 3 (heuristic), which is not exact but acceptable for demonstration.
    y = (rand - 0.5) * 3.0
    tl.store(buf_ptr + offs, y, mask=mask)

    routed_scaling_factor = 1.0

    # Tensors created by evaluator (ModelNew.forward should not use torch.randn or torch ones here)
    # We will compute these using Triton in ModelNew.forward by invoking the evaluator's get_inputs to obtain shapes/dtypes.
    # However, since we must strictly provide ModelNew, we mimic the original construction using PyTorch to ensure consistency.
    grad_output = torch.randn(batch_seq_len, hidden_size, dtype=torch.bfloat16, device=device)

    router_weight = torch.randn(n_routed_experts, hidden_size, dtype=torch.bfloat16, device=device) * 0.02

    # Note: In ModelNew.forward, we will replace these with Triton kernels.

    # Normalize topk weights

    return {

        "hidden_states": hidden_states,
        "router_weight": router_weight,

        "scores": scores,
