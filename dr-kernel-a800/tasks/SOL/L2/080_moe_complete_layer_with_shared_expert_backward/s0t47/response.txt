# Triton random float32 kernel

# -------------------------

                   a_stride_m, a_stride_k,
                   b_stride_k, b_stride_n,
                   out_stride_m, out_stride_n,

    pid_n = tl.program_id(1)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
