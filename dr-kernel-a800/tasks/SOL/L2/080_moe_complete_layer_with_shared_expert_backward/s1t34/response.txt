import torch
import triton
import triton.language as tl


# GEMV: y[b, e] = sum_h hidden[b, h] * W[e, h]
# hidden: [B, H] (bf16), W: [N, H] (bf16), y: [B, N] (f32)
@triton.jit
def linear_forward_kernel(
    hidden_ptr,   # *bf16, [B, H]
    W_ptr,        # *bf16, [N, H]
    y_ptr,        # *f32,  [B, N]
    B: tl.constexpr,
    H: tl.constexpr,
    N: tl.constexpr,
    stride_h_b, stride_h_h,
    stride_W_e, stride_W_h,
    stride_y_b, stride_y_e,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch row index
    pid_e = tl.program_id(1)  # output feature index in W
    acc = 0.0
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        h_vals = tl.load(hidden_ptr + pid_b * stride_h_b + offs_h * stride_h_h, mask=mask_h, other=0.0).to(tl.float32)
        W_vals = tl.load(W_ptr + pid_e * stride_W_e + offs_h * stride_W_h, mask=mask_h, other=0.0).to(tl.float32)
        acc += tl.sum(h_vals * W_vals, axis=0)
    tl.store(y_ptr + pid_b * stride_y_b + pid_e * stride_y_e, acc)


# GEMV backward: grad_W[e, h] = sum_b grad_y[b, e] * hidden[b, h]
# grad_y: [B, N], hidden: [B, H], grad_W: [N, H], f32
@triton.jit
def gemv_backward_weight_kernel(
    grad_y_ptr,   # *f32, [B, N]
    hidden_ptr,   # *bf16, [B, H]
    grad_W_ptr,   # *f32, [N, H]
    B: tl.constexpr,
    H: tl.constexpr,
    N: tl.constexpr,
    stride_gy_b, stride_gy_e,
    stride_h_b, stride_h_h,
    stride_gW_e, stride_gW_h,
    BLOCK_H: tl.constexpr,
):
    pid_e = tl.program_id(0)  # feature index in grad_y (output of forward)
    pid_h = tl.program_id(1)  # hidden dimension index
    acc = 0.0

        # grad_y[b, e]

        # hidden[b, h]
        h = tl.load(hidden_ptr + b * stride_h_b + pid_h * stride_h_h).to(tl.float32)
        acc += gy * h
    tl.store(grad_W_ptr + pid_e * stride_gW_e + pid_h * stride_gW_h, acc)


# Elementwise SiLU: y = x * sigmoid(x), output f32
@triton.jit
def silu_elemwise_kernel(x_ptr, y_ptr, N_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(y_ptr + offs, y, mask=mask)


# Elementwise multiply: y = a * b, output f32
@triton.jit
def mul_elemwise_kernel(a_ptr, b_ptr, y_ptr, N_elements: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    y = a * b
    tl.store(y_ptr + offs, y, mask=mask)

    grad_shared_expert_gate_weight = grad_shared_gate_weight.to(torch.bfloat16).contiguous()  # shape [H, 1408], bf16
    grad_shared_expert_up_weight = grad_shared_up_weight.to(torch.bfloat16).contiguous()      # shape [H, 1408], bf16
    grad_shared_expert_down_weight = grad_shared_down_weight.to(torch.bfloat16).contiguous()  # shape [H, 1408], bf16

    # Return 5 tensors:
    return (
        grad_hidden_states,
        grad_router_weight,
        grad_shared_expert_gate_weight,
        grad_shared_expert_up_weight,
        grad_shared_expert_down_weight,
    )
