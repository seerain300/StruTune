# Triton kernel: 2D convolution NCHW, kernel 3x3, stride=2, padding=1, bias, GELU (tanh approximation)
# Input:
#   x_ptr: [B, C_in, H, W], contiguous NCHW
#   w_ptr: [C_out, C_in, 3, 3], contiguous
#   b_ptr: [C_out]
# Output:
#   out_ptr: [B, C_out, H_out, W_out], contiguous NCHW
@triton.jit
def conv3x3_stride2_nchw_gelu(
    x_ptr, w_ptr, b_ptr, out_ptr,
    B, C_in, H, W, C_out, H_out, W_out,
    KSIZE: tl.constexpr, STRIDE: tl.constexpr, PAD: tl.constexpr,
    BLOCK_CO: tl.constexpr, BLOCK_OW: tl.constexpr,
):
    b = tl.program_id(0)  # batch id
    co_block = tl.program_id(1)  # block over output channels

    # offsets for output channels handled by this program
    co_offsets = co_block * BLOCK_CO + tl.arange(0, BLOCK_CO)
    co_mask = co_offsets < C_out  # [BLOCK_CO]

    # initialize accumulator for these co_offsets
    # shape [BLOCK_CO, BLOCK_OW] where BLOCK_OW is number of output width positions
    y_tile = tl.zeros((BLOCK_CO, BLOCK_OW), dtype=tl.float32)

    # Loop over output height positions
    # Note: we compute ow, then for each ow loop over ih, iw (3x3 neighborhood)
    for oh in range(0, H_out):
        # For each output width position handled by this program
        ow_block_start = (oh // 1) * BLOCK_OW  # dummy to form vector; will loop per oh
        # We can vectorize over ow using a single loop: compute per ow directly
        # To vectorize, we need to handle multiple ow in a block; but Triton expects fixed grid.
        # So we compute one ow at a time and accumulate to y_tile.
        # We'll iterate ow as scalar and store; Triton will unroll. For performance, one can use while.
        # But to be safe and correct, we compute per ow:
        # We'll create a loop over ow; Triton supports Python loops with runtime bounds.
        for ow in range(0, W_out):
            ih = oh * STRIDE - PAD  # top-left input pixel for this output
            # Accumulate over 3x3 neighborhood and input channels
            for kh in range(0, KSIZE):
                # kh is 3 in KSIZE, but KSIZE is constexpr; Triton can handle it
                for kw in range(0, KSIZE):
                    iw = ow * STRIDE - PAD + kw  # vectorized over kw
                    # validity masks
                    valid_h = (ih + kh) >= 0 and (ih + kh) < H
                    valid_w = (iw) >= 0 and (iw) < W
                    mask_vec = valid_h & valid_w  # scalar masks apply to this output position

                    # Load input x[b, ic, ih+kh, iw] for ic=0..C_in-1, then multiply with corresponding weights
                    # We need to accumulate over C_in and 3x3 neighborhood
                    # For each ic:
                    for ic in range(0, C_in):
                        # Input linear offset for x[b, ic, ih+kh, iw]
                        # x layout: ((b * C_in + ic) * H + (ih + kh)) * W + iw
                        x_off = ((b * C_in + ic) * H + (ih + kh)) * W + iw
                        # Cast to int32 for addressing
                        x_off = x_off.to(tl.int32)
                        # Load value (masked)
                        x_val = tl.load(x_ptr + x_off, mask=mask_vec, other=0.0)
                        # Load weights w[co, ic, kh, kw]
                        # w layout: [C_out, C_in, 3, 3]
                        # idx = co * (C_in * 9) + ic * 9 + kh * 3 + kw
                        # We need co_offsets here
                        # We will compute per co and multiply by x_val, then add to y_tile
                        # Iterate over co_offsets (vectorized)
                        for co_i in range(0, BLOCK_CO):
                            co = co_offsets[co_i]
                            co_valid = co < C_out
                            idx_w = co * (C_in * 9) + ic * 9 + kh * 3 + kw
                            w_val = tl.load(w_ptr + idx_w, mask=co_valid, other=0.0)
                            y_tile[co_i, 0] += w_val * x_val  # since we process single ow per loop

    # Add bias
    for co_i in range(0, BLOCK_CO):
        co = co_offsets[co_i]
        co_valid = co < C_out
        b_val = tl.load(b_ptr + co, mask=co_valid, other=0.0)
        y_tile[co_i, 0] += b_val

    # GELU (tanh approximation)
    # Apply to each co row
    c = 0.7978845608028654  # sqrt(2/pi)
    for co_i in range(0, BLOCK_CO):
        co = co_offsets[co_i]
        co_valid = co < C_out
        y_row = y_tile[co_i, 0]
        x3 = y_row * y_row * y_row
        tanh_arg = c * (y_row + 0.044715 * x3)
        tanh_val = tl.math.tanh(tanh_arg)
        y_tile[co_i, 0] = 0.5 * y_row * (1.0 + tanh_val)

    # Store output: out[b, co, oh, ow]
    # For each co, store scalar at out_ptr + ((b * C_out + co) * H_out + oh) * W_out + ow
    for co_i in range(0, BLOCK_CO):
        co = co_offsets[co_i]
        co_valid = co < C_out
        out_off = ((b * C_out + co) * H_out + oh) * W_out + ow
        out_off = out_off.to(tl.int32)
        # Cast back to original dtype (bfloat16)
        y_val = y_tile[co_i, 0]
        tl.store(out_ptr + out_off, y_val.to(tl.bfloat16), mask=co_valid)


# Triton kernel: batched linear projection y[b, t, d] = sum_k x[b, t, k] * W[d, k]
# Inputs:
#   x_ptr: [B, T, K], bfloat16 (we will read as fp32)
#   W_ptr: [N, K], float32 (N = d_model)
#   y_ptr: [B, T, N], float32
# We launch grid over (B, T). Each program computes all N outputs for that (b, t).
@triton.jit
def linear_proj_kernel(
    x_ptr, W_ptr, y_ptr,
    B, T, K, N,
    BLOCK_N: tl.constexpr,
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    # Vector of output d indices
    d = tl.arange(0, BLOCK_N)
    # Accumulator for N outputs
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K to compute dot product
    for k in range(0, K):
        # Load x[b, t, k]
        x_off = (b * T + t) * K + k
        x_val = tl.load(x_ptr + x_off).to(tl.float32)  # input is bfloat16, convert to fp32
        # Load W[d, k] vector
        w_off = d * K + k  # [BLOCK_N]
        w_vec = tl.load(W_ptr + w_off)  # float32
        # Accumulate: acc += w_vec * x_val
        acc += w_vec * x_val

    # Store y[b, t, d] over d
    y_off_base = (b * T * N) + t * N  # linear indexing assuming contiguous [B, T, N]
    # We can store vector acc at offsets y_off_base + d
    for i in range(0, BLOCK_N):
        tl.store(y_ptr + (y_off_base + i), acc[i])


# Triton kernel: add positional embedding to [B, T, d_model] tensor
# Inputs:
#   y_ptr: [B, T, N], float32
#   pos_ptr: [T, N], float32
#   B, T, N
@triton.jit
def add_pos_embedding_kernel(
    y_ptr, pos_ptr,
    B, T, N,
    BLOCK_N: tl.constexpr,
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    d = tl.arange(0, BLOCK_N)
    # Load y[b, t, d] vector
    y_off_base = (b * T * N) + t * N
    y_vec = tl.load(y_ptr + y_off_base + d)
    # Load pos[t, d] vector
    pos_vec = tl.load(pos_ptr + (t * N + d))
    # Add
    y_vec += pos_vec
    # Store back
    for i in range(0, BLOCK_N):
        tl.store(y_ptr + (y_off_base + i), y_vec[i])
