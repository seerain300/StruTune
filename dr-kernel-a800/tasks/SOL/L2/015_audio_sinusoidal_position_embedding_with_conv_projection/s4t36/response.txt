# X: input tensor flattened
# Y: output tensor flattened
@triton.jit
def gelu_tanh_kernel(X, Y, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

        # Accumulate: acc[m] += sum_k W[m, k] * X[t, k]
        acc += tl.sum(w_vals * x_vals[None, :], axis=1)

    # Store acc to Y[b, t, m_offsets] (fp32; you can cast to bfloat16 after kernel)
    y_ptrs = Y + b * stride_yb + t * stride_yt + m_offsets * stride_ym
    tl.store(y_ptrs, acc, mask=m_mask)


class ModelNew(torch.nn.Module):
    def forward(self, input_features: torch.Tensor,
                conv2d1_weight: torch.Tensor, conv2d1_bias: torch.Tensor,
                conv2d2_weight: torch.Tensor, conv2d2_bias: torch.Tensor,
                conv2d3_weight: torch.Tensor, conv2d3_bias: torch.Tensor,
                conv_out_weight: torch.Tensor, positional_embedding: torch.Tensor,
                embed_scale: float, T_after_conv: int):
        # Convolutions using PyTorch (robust and correct)
        x = torch.nn.functional.conv2d(input_features, conv2d1_weight, conv2d1_bias, stride=2, padding=1)
        x = torch.nn.functional.gelu(x)

        x = torch.nn.functional.conv2d(x, conv2d2_weight, conv2d2_bias, stride=2, padding=1)
        x = torch.nn.functional.gelu(x)

        x = torch.nn.functional.conv2d(x, conv2d3_weight, conv2d3_bias, stride=2, padding=1)
        x = torch.nn.functional.gelu(x)

        # Reshape: (B, C, F, T_after_conv) where C=384, F=10
        B, C, F, T_after = x.size()  # x should be (B, 384, 10, T_after_conv)
        X = x.permute(0, 3, 1, 2).contiguous().view(B, T_after, C * F)  # (B, T_after_conv, 3840)

        # Linear projection in Triton: (B, T_after_conv, 3840) @ (1024, 3840)^T -> (B, T_after_conv, 1024)
        M = conv_out_weight.shape[0]  # 1024
        K = X.shape[2]  # 3840

        # Allocate fp32 output for accumulation
        Y = torch.empty((B, T_after, M), dtype=torch.float32, device=X.device)

        # Ensure contiguous tensors
        Xc = X.to(torch.float32).contiguous()
        Wc = conv_out_weight.contiguous()

        # Launch Triton kernel
        BLOCK_M = 128
        BLOCK_K = 128
        grid = (B, T_after, triton.cdiv(M, BLOCK_M))
        linear_no_bias_kernel[grid](
            Xc, Wc, Y,
            B, T_after, K, M,
            Xc.stride(0), Xc.stride(1), Xc.stride(2),
            Wc.stride(0), Wc.stride(1),
            Y.stride(0), Y.stride(1), Y.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_K=BLOCK_K,
        )

        # Scale by embed_scale and add positional embedding (PyTorch ops for simplicity)
        # embed_scale = sqrt(1024) = 32.0
        Y_scaled = Y * embed_scale  # Y is fp32, bfloat16 output desired
        # Convert to bfloat16
        Y_bf16 = Y_scaled.to(torch.bfloat16)

        # Add positional embedding: (1500, 1024), but only first T_after rows are used
        POS = positional_embedding.to(torch.bfloat16)
        # Broadcast add: Y_bf16 shape (B, T_after, 1024), POS[:T_after, :] shape (T_after, 1024)
        # We need to expand POS to (B, T_after, 1024)
        # Since Y_bf16[..., :] corresponds to 1024 columns, we can add along last dim by unsqueezing batch dimension
        # Create a broadcasted POS for each batch by repeating along batch dimension
        # But PyTorch broadcasting handles adding (T_after, 1024) to (B, T_after, 1024) when POS is unsqueezed to (1, T_after, 1024)
        POS_b = POS[:T_after, :].unsqueeze(0).expand(B, T_after, 1024).
