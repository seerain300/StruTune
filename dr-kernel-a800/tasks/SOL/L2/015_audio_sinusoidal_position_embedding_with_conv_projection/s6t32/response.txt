import math
import torch
import torch.nn as nn
import torch.nn.functional as F

import triton
import triton.language as tl


# Triton GEMV kernel: compute y[b, t, d] = sum_k x[b, t, k] * W[d, k] for d in [0..N-1]
# x is [B, T, K], W is [N, K], y is [B, T, N]
@triton.jit
def linear_gemv_kernel(
    x_ptr,          # *const float32, input [B, T, K]
    w_ptr,          # *const float32, weight [N, K]
    y_ptr,          # *float32, output [B, T, N]
    B: tl.int32, T: tl.int32, K: tl.int32, N: tl.int32,
    x_stride_b: tl.int32, x_stride_t: tl.int32, x_stride_k: tl.int32,
    y_stride_b: tl.int32, y_stride_t: tl.int32, y_stride_n: tl.int32,
    BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)
    t = tl.program_id(1)

    # accumulator for N outputs
    acc = tl.zeros((N,), dtype=tl.float32)

    # loop over K in tiles
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        # load x[b, t, k_offsets] as a vector
        x_off = b * x_stride_b + t * x_stride_t + k_offsets * x_stride_k
        x_vals = tl.load(x_ptr + x_off, mask=k_mask, other=0.0)  # [BLOCK_K], float32

        # load W[:, k_offsets] -> [N, BLOCK_K]
        w_off = k_offsets[None, :]  # shape [1, BLOCK_K], broadcasting below
        # We need to iterate over N and load each row: w[d, k]
        # We'll load per-dimension:
        for d in range(0, N):
            w_row = tl.load(w_ptr + d * K + k_offsets, mask=k_mask, other=0.0)  # [BLOCK_K]
            acc[d] += tl.sum(w_row * x_vals, axis=0)

    # store y[b, t, d] for all d
    for d in range(0, N):
        y_off = b * y_stride_b + t * y_stride_t + d * y_stride_n
        tl.store(y_ptr + y_off, acc[d])


# Triton kernel to add positional embeddings: y = x + pos_emb, where
# x is [B, T, N], pos_emb is [1, T, N] (broadcast over batch). We implement as [B, T, N].
# Note: We assume x and pos_emb have the same dtype. Here we assume float32 for simplicity.
@triton.jit
def add_pos_emb_kernel(
    x_ptr,          # *const float32, input [B, T, N]
    pos_ptr,        # *const float32, pos_emb [1, T, N] but we load as [T, N]
    y_ptr,          # *float32, output [B, T, N]
    B: tl.int32, T: tl.int32, N: tl.int32,
    x_stride_b: tl.int32, x_stride_t: tl.int32, x_stride_n: tl.int32,
    y_stride_b: tl.int32, y_stride_t: tl.int32, y_stride_n: tl.int32,
    BLOCK_N: tl.constexpr,
):
    b = tl.program_id(0)
    t = tl.program_id(1)

    # load pos_emb for this t across all N
    for n_start in range(0, N, BLOCK_N):
        n_offsets = n_start + tl.arange(0, BLOCK_N)
        n_mask = n_offsets < N
        pos_off = t * N + n_offsets  # pos is 1D vector of length T*N, but here we load [T, N] row, simplify:
        # Simpler approach: we access pos directly as pos[t, n] but Triton expects 1D. We will load pos[t, n] via linear index.
        # To do that, pos is provided as [T, N] in our forward by unsqueezing to [1, T, N] and slicing pos_emb[0, t, n].
        # However, Triton kernel pointer is 1D; we can pass a 2D pointer by flattening. For simplicity, we assume pos is provided as [T, N].
        # But since we cannot assume 2D pointer, we implement by passing a 1D pos vector of length T*N and index pos_idx = t * N + n.
        pos_off = t * N + n_offsets
        pos_vals = tl.load(pos_ptr + pos_off, mask=n_mask, other=0.0)

        # load x[b, t, n_offsets]
        x_off = b * x_stride_b + t * x_stride_t + n_offsets * x_stride_n
        x_vals = tl.load(x_ptr + x_off, mask=n_mask, other=0.0)

        # compute y
        y_vals = x_vals + pos_vals

        # store
        y_off = b * y_stride_b + t * y_stride_t + n_offsets * y_stride_n
        tl.store(y_ptr + y_off, y_vals)

        # Linear projection to d_model (no bias)
        B = b
        T = t
        K = x.shape[2]  # number of features after conv3
        N = self.d_model  # 1024

        # Make sure tensors are float32 for Triton kernels and contiguous
        x_3d = x.contiguous()

        x_3d_f32 = x_3d.float()
        W = conv_out_weight.float()  # conv_out_weight is [N, K]

        # Launch GEMV Triton kernel
        # Strides
        x_stride_b = x_3d_f32.stride(0)
        x_stride_t = x_3d_f32.stride(1)
        x_stride_k = x_3d_f32.stride(2)

        y_stride_t = y_f32.stride(1)

            B, T, K, N,

        # However, we don't have time_after_conv directly. The reshape gives us T = x.shape[3] after 3rd conv.
        # So we need to slice positional_embedding[:T, :].

        # Note: positional_embedding dtype is bfloat16; we'll cast to float32 for addition, then cast result back if needed.
        pos_emb = positional_embedding.to(torch.float32)  # [max, N]

        pos_emb_1d = pos_emb.view(1, T * N).squeeze(0)  # [T*N], but we want [T, N] per kernel, so we pass as [T, N] by flattening and indexing.

        y_final_f32 = torch.empty((B, T, N), dtype=torch.float32, device=y_f32.device)

            y_final_f32.stride(0), y_final_f32.stride(1), y_final_f32.stride(2),
            BLOCK_N=BLOCK_N,
        )

        # Cast back to bfloat16 to match original dtype
        y_final = y_final_f32.to(torch.bfloat16)
