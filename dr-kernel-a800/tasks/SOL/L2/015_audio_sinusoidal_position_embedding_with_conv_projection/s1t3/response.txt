    Bsz, T_out, K_in, K_out, scale,  # K_in=15360, K_out=1024
    BLOCK_N: tl.constexpr,  # tile over K_out
    BLOCK_M: tl.constexpr,  # tile over T_out (we set 1, since B is small per workload)
):
    pid_b = tl.program_id(0)  # batch
    pid_n = tl.program_id(1)  # tiles over K_out

    n_start = pid_n * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < K_out

    # Initialize accumulator per (b, n)
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Iterate over K_in in tiles (BLOCK_K). Since K_in=15360, we loop over it.
    BLOCK_K = 128  # tile over K_in
    for k_start in range(0, K_in, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K_in

        # Load X tile for this batch and all T_out: X_ptr + (pid_b * K_in + k)*T_out + t_out
        # We'll use a loop over t_out_idx to load vectors for each t_out_idx, then do dot.
        # For simplicity, we vectorize over t_out with BLOCK_M=1. Since T_out varies, we process one t_out per iteration.

        # We need a 2D tile: A_tile of shape (BLOCK_K, 1) from X and B_tile of shape (1, BLOCK_N) from W.T.
        # But Triton supports elementwise; we'll compute per t_out_idx.

        # To keep things simpler, we implement a nested loop over t_out_idx. Triton supports this in kernels.
        # Compute acc for each n_offset across all t_out_idx:
        # For each t_out_idx, compute A_vec = X[(pid_b, t_out_idx, k_offsets)] and do dot with B_vec = W[n_offsets, k_offsets].
        # Accumulate into acc.

        # Since BLOCK_M is 1, we will iterate t_out_idx in the kernel. Triton allows loops with dynamic ranges.
        # We can keep it simple: for each n we accumulate across all t_out_idx using a loop.

        # We'll compute A_vec for each t_out_idx and multiply with corresponding B_vec, then accumulate into acc.
        # We'll do this in Python code sense: Triton kernel needs to loop over T_out.

        # However, Triton kernel does not accept Python loops over T_out because it is a runtime value. We need to use a grid over T_out.
        # Instead, we write acc in per-t_out fashion: we'll restructure to 3D grid, but Triton kernels only have 2D. To handle T_out, we compute per t_out in a loop.

        # The kernel design: compute per (b, n) vector acc across all t_out. We will load X per t_out and W per n, and do dot reduction. To do that, we restructure:
        # We'll compute acc across all t_out for this (b, n). Since T_out is passed as runtime, we'll loop over t_out in Triton (supported).

        # Compute acc by looping t_out_idx
        # For each t_out_idx, load A_vec (BLOCK_K elements) from X, and B_vec (BLOCK_N) from W.T, and do dot then accumulate.

        for t_out_idx in range(0, T_out):
            # A_vec: X[(pid_b, t_out_idx, k_offsets)]
            a_vec = tl.load(
                X_ptr + pid_b * K_in + k_offsets * T_out + t_out_idx,
                mask=mask_k,
                other=0.0
            ).to(tl.float32)

            # B_vec: W[n_offsets, k_offsets] but W is (K_out, K_in). We need W^T, which is (K_in, K_out), accessed as W_T[k_offsets, n_offsets].
            # W_T layout is W_ptr with rows k and cols n. We'll read W_ptr[k_offsets, n_offsets].
            b_vec = tl.load(
                W_ptr + k_offsets[:, None] * K_out + n_offsets[None, :],
                mask=(k_offsets[:, None] < K_in) & (n_offsets[None, :] < K_out),
                other=0.0
            ).to(tl.float32)

            # acc += sum_k (a_vec[k] * b_vec[k, n]) => elementwise product then sum over k
            acc += tl.sum(a_vec[:, None] * b_vec, axis=0)

    # Add scaled positional embedding: POS_ptr is (OH, K_out), OH=B. We add POS[pid_b, n_offsets] scaled by scale to acc.
    pos_vec = tl.load(POS_ptr + pid_b * K_out + n_offsets, mask=mask_n, other=0.0).to(tl.float32)
    acc = acc + scale * pos_vec
