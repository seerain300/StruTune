def conv3x3_stride2_nchw_fp32(
    x_ptr,           # *float32
    w_ptr,           # *float32
    b_ptr,           # *float32
    out_ptr,         # *float32
    B, C_in, H, W, C_out, H_out, W_out,
    BLOCK_OW: tl.constexpr,
    BLOCK_CO: tl.constexpr,
):
    # Grid: (B, H_out, W_out_blocks, C_out_blocks)
    b = tl.program_id(0)
    oh = tl.program_id(1)
    pid_w = tl.program_id(2)
    pid_c = tl.program_id(3)

    ow_start = pid_w * BLOCK_OW
    co_start = pid_c * BLOCK_CO

    ow_offsets = ow_start + tl.arange(0, BLOCK_OW)  # vector of width positions
    co_offsets = co_start + tl.arange(0, BLOCK_CO)  # vector of output channels

    ow_mask = ow_offsets < W_out
    co_mask = co_offsets < C_out

    # Initialize output tile to bias
    # We'll compute bias as a vector for co_offsets
    b_vec = tl.load(b_ptr + co_offsets, mask=co_mask, other=0.0)  # [BLOCK_CO]
    y_tile = tl.zeros((BLOCK_CO, BLOCK_OW), dtype=tl.float32)

    # Accumulate over input channels and 3x3 neighborhood
    for ic in range(0, C_in):
        # Loop over kernel height
        for kh in range(0, 3):
            ih = oh + kh - 1  # padding=1, kernel 3x3 -> ih in [oh-1, oh+1]
            valid_h = (ih >= 0) & (ih < H)
            # Loop over kernel width
            for kw in range(0, 3):
                iw = ow_offsets - 1 + kw  # padding=1 -> iw in [ow-1, ow+1]
                valid_w = (iw >= 0) & (iw < W)
                mask = ow_mask & valid_h & valid_w  # [BLOCK_OW]

                # Load input vector x[b, ic, ih, iw]
                x_off_vec = ((b * C_in + ic) * H + ih) * W + iw
                x_vec = tl.load(x_ptr + x_off_vec, mask=mask, other=0.0)  # [BLOCK_OW]

                # Load weight vector for these co_offsets: w[co, ic, kh, kw]
                # w layout: [C_out, C_in, 3, 3]
                base_w = co_offsets * (C_in * 9)  # since each co has C_in*9 elements
                idx_w = base_w + ic * 9 + kh * 3 + kw  # [BLOCK_CO]
                w_vec = tl.load(w_ptr + idx_w, mask=co_mask, other=0.0)  # [BLOCK_CO]

                # Outer product and accumulation: y_tile += w_vec[:, None] * x_vec[None, :]
                # We need to broadcast: w_vec shape [BLOCK_CO], x_vec shape [BLOCK_OW]
                # Triton requires elementwise ops; we can do it via loop over co_offsets and adding column-wise.
                # Easier: compute each co column by multiplying w_vec with x_vec and add to y_tile.
                # Implement by looping over BLOCK_CO (small) and adding one column at a time.
                for co_i in range(0, BLOCK_CO):
                    if co_i < BLOCK_CO:  # guard
                        w_elem = w_vec[co_i]  # scalar
                        y_tile[co_i, :] += w_elem * x_vec  # add to all ow columns

    # Add bias (already initialized)
    # Apply GELU (tanh approximation) to y_tile
    c = 0.7978845608028654  # sqrt(2/pi)
    for co_i in range(0, BLOCK_CO):
        if co_i < BLOCK_CO:
            x_vec = y_tile[co_i, :]
            x3 = x_vec * x_vec * x_vec
            tanh_arg = c * (x_vec + 0.044715 * x3)
            tanh_val = tl.math.tanh(tanh_arg)
            y_tile[co_i, :] = 0.5 * x_vec * (1.0 + tanh_val)
