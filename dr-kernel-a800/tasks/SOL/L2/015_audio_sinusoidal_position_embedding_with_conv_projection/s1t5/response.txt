# Reliable Triton implementation for conv2d using GEMM form:
# For each (b, oh), compute output vector of length T_out for each output channel cout:
# conv2d(x, w) with stride=2, padding=1, kernel 3x3, Cin=1 for conv1 and Cin=Cout for conv2/3:
# out[b, cout, oh, t_out] = sum_{cin=0..Cin-1} sum_{kh=0..2} sum_{kt=0..2} x[b, cin, ih, it] * w[cout, cin, kh, kt], where
# ih = oh + kh - 1, it = t_out + kt - 1; valid if 0 <= ih < H and 0 <= it < T.
# We implement this as a kernel that computes one t_out per program and vectorizes over cout with a tile.
# However, Triton supports loops over small constants, and T_out is typically moderate (e.g., 40). We can compute per t_out.

@triton.jit
def conv2d_3x3_stride2_padding1_v1(
    X_ptr,         # *const bfloat16, input: (B, Cin, H, T)
    W_ptr,         # *const bfloat16, weight: (Cout, Cin, 3, 3)
    BIAS_ptr,      # *const bfloat16, bias: (Cout,)
    Y_ptr,         # *bfloat16, output: (B, Cout, H_out, T_out)
    Bsz: tl.constexpr,      # B
    Cin: tl.constexpr,      # input channels (1 or 384)
    H: tl.constexpr,        # height (80)
    T: tl.constexpr,        # time dimension
    Cout: tl.constexpr,     # output channels (384)
    T_out: tl.constexpr,    # output time length
    BLOCK_C: tl.constexpr,  # tile size for output channels
):
    pid_m = tl.program_id(0)  # over B * H_out
    pid_c = tl.program_id(1)  # over tiles of Cout
    t_out_idx = tl.program_id(2)  # over T_out

    b = pid_m // H
    oh = pid_m % H

    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # Initialize accumulator for this (b, oh, t_out_idx)
    acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel, compute contribution for this t_out_idx
    for cin in range(Cin):
        for kh in range(3):
            ih = oh + kh - 1
            valid_ih = (ih >= 0) & (ih < H)
            for kt in range(3):
                it = t_out_idx + kt - 1
                valid_it = (it >= 0) & (it < T)
                # If invalid, contribution is 0
                if valid_ih & valid_it:
                    # For Cin=1 case: X offset for (b, 0, ih, it)
                    x_off = b * (Cin * H * T) + 0 * (H * T) + ih * T + it
                    x_val = tl.load(X_ptr + x_off, mask=True, other=0.0).to(tl.float32)
                else:
                    x_val = 0.0

                # For weights: weight[cin, cout, kh, kt] -> we need to accumulate over cout. Weight is (Cout, Cin, 3, 3)
                # To access w[cout, cin, kh, kt], note that w is laid out as (Cout, Cin, 3, 3). We can compute address as:
                # w_addr = cout * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt
                # We need to multiply x_val with all w across cout. Do this by reading W_ptr and multiplying into acc.
                # Iterate over cout tile and load w scalar for each cout in the tile.
                for j in range(BLOCK_C):  # j indexes the cout tile
                    cout_j = c_start + j
                    if cout_j < Cout:
                        w_addr = cout_j * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt
                        w_val = tl.load(W_ptr + w_addr, mask=True, other=0.0).to(tl.float32)
                        acc[j] += x_val * w_val

    # Add bias
    for j in range(BLOCK_C):
        cout_j = c_start + j
        if cout_j < Cout:
            b_val = tl.load(BIAS_ptr + cout_j, mask=True, other=0.0).to(tl.float32)
            acc[j] += b_val

    # Store result to Y[b, c_offsets, oh, t_out_idx] (note: oh is scalar for this kernel)
    # Output layout: Y is (B, Cout, H, T_out). We will flatten indexing using strides.
    # y_off = b * (Cout * H * T_out) + cout * (H * T_out) + oh * T_out + t_out_idx
    for j in range(BLOCK_C):
        cout_j = c_start + j
        if cout_j < Cout:
            y_off = b * (Cout * H * T_out) + cout_j * (H * T_out) + oh * T_out + t_out_idx
            # Store as bfloat16
            tl.store(Y_ptr + y_off, acc[j].to(tl.bfloat16), mask=True)


# Triton GELU kernel (exact, erf-based)
@triton.jit
def gelu_exact_kernel(
    X_ptr,          # *const bfloat16, input tensor
    Y_ptr,          # *bfloat16, output tensor
    N: tl.constexpr,  # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    # exact GELU: 0.5 * x * (1 + erf(x / sqrt(2)))
    inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
    y = 0.5 * x * (1.0 + tl.math.erf(x * inv_sqrt2))
    tl.store(Y_ptr + offsets, y.to(tl.bfloat16), mask=mask)


# Triton GEMM: A is (M, K_in) = x, B is (K_out, N) = conv_out_weight.T (15360, 1024), C is (M, N) = y
# Here, x is (B, T_out, 15360) -> we flatten to M = B * T_out, K_in = 15360, N = 1024.
@triton.jit
def gemm_mul_add_pos_kernel(
    A_ptr,            # *const bfloat16, A: (M, K_in)
    BT_ptr,           # *const bfloat16, BT: (K_out, N) where K_out=15360, N=1024
    BIAS_ptr,         # *const bfloat16, bias (if any); not used here
    POS_ptr,          # *const bfloat16, positional embedding: (OH, N) where OH=M (i.e., B * T_out3)
    C_ptr,            # *bfloat16, output: (M, N)
    M: tl.constexpr,  # total rows in A (B * T_out3)
    K_in: tl.constexpr,  # input feature dim (15360)
    N: tl.constexpr,     # output channels (1024)
    stride_Am, stride_Ak,
    stride_BTk, stride_BTn,
    stride_Cm, stride_Cn,
    scale: tl.constexpr,  # float32 scaling factor, e.g., sqrt(d_model) = 32
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile over M
    pid_n = tl.program_id(1)  # tile over N

    m_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = m_offsets < M
    mask_n = n_offsets < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k0 in range(0, K_in, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K_in

        # Load A_tile: (BLOCK_M, BLOCK_K)
        A_tile = tl.load(
            A_ptr + m_offsets[:, None] * stride_Am + k_offsets[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # Load BT_tile: (BLOCK_K, BLOCK_N)
        BT_tile = tl.load(
            BT_ptr + k_offsets[:, None] * stride_BTk + n_offsets[None, :] * stride_BTn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(A_tile, BT_tile)

    # Scale
    acc = acc * scale

    # Add scaled positional embedding: POS is (OH, N), with OH=M
    # For each m in m_offsets, load POS[m, n_offsets] and add to acc[m, :]
    for i in range(BLOCK_M):
        m_i = m_offsets[i]
        if m_i < M:
            pos_vec = tl.load(
                POS_ptr + m_i * N + n_offsets,
                mask=mask_n,
                other=0.0
            ).to(tl.float32)
            acc[i, :] += scale * pos_vec

    # Store
    tl.store(
        C_ptr + m_offsets[:, None] * stride_Cm + n_offsets[None, :] * stride_Cn,
        acc.to(tl.bfloat16),
        mask=mask_m[:, None] & mask_n[None, :]
    )


def _run_conv2d_triton(x, w, bias, out_channels, time_dim, block_c=64):
    """
    Launch conv2d_3x3_stride2_padding1_v1 kernel.
    x: (B, 1, 80, time_dim) bfloat16
    w: (out_channels, Cin, 3, 3) bfloat16, Cin=1 or out_channels
    bias: (out_channels,) bfloat16
    Returns y: (B, out_channels, 80, T_out) bfloat16
    """
    B, Cin, H, T = x.shape
    Cout = out_channels
    T_out = (T - 3) // 2 + 1
    # Allocate output
    y = torch.empty((B, Cout, H, T_out), dtype=torch.bfloat16, device=x.device)
    # Grid: (B * H, tiles over Cout, T_out)
    grid = (B * H, triton.cdiv(Cout, block_c), T_out)
    conv2d_3x3_stride2_padding1_v1[grid](
        x, w, bias, y,
        B=B, Cin=Cin, H=H, T=T, Cout=Cout, T_out=T_out, BLOCK_C=block_c
    )
    return y


def _gelu_triton(x):
    """
    Apply exact GELU in Triton, returns bfloat16 tensor.
    """
    x_contig = x.contiguous()
    y = torch.empty_like(x_contig, dtype=torch.bfloat16, device=x.device)
    n = x_contig.numel()
    # Use BLOCK=1024
    grid = (triton.cdiv(n, 1024),)
    gelu_exact_kernel[grid](x_contig, y, N=n, BLOCK=1024)
    return y


def _gemm_triton(x, bt, scale, pos, out):
    """
    x: (B, T_out, K_in) -> flatten to (M, K_in)
    bt: (K_out, N) = conv_out_weight.T, here K_out=15360, N=1024
    pos: (B * T_out, N), positional embedding sliced to OH=M
    out: (M, N) will be written
    """
    B, T_out, K_in = x.shape
    N = bt.shape[1]  # 1024
    M = B * T_out
    # Reshape x to (M, K_in)
    A = x.reshape(M, -1).contiguous()
    # Ensure bt is (K_in, N) where K_in=15360, N=1024. bt is already (K_in, N)
    # Prepare strides
    stride_Am = A.stride(0)
    stride_Ak = A.stride(1)
    stride_BTk = bt.stride(0)
    stride_BTn = bt.stride(1)
    # Output C (M, N)
    C = out.reshape(M, N).contiguous()  # we will fill it
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    gemm_mul_add_pos_kernel[grid](
        A, bt, None, pos, C,
        M=M, K_in=15360, N=1024,
        stride_Am=stride_Am, stride_Ak=stride_Ak,
        stride_BTk=stride_BTk, stride_BTn=stride_BTn,
        stride_Cm=C.stride(0), stride_Cn=C.stride(1),
        scale=float(scale), BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # out is (B, T_out, N); reshape C back to (M, N) -> (B, T_out, N)
    out.copy_(C.view(B, T_out, N))

    oh = pid_m % H

    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # Accumulator for this (b, oh, t_out_idx)
    acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel positions
    for cin in range(Cin):
        for kh in range(3):
            ih = oh + kh - 1  # padding=1
            valid_ih = (ih >= 0) & (ih < H)
            for kt in range(3):
                it = t_out_idx + kt - 1  # padding=1
                valid_it = (it >= 0) & (it < T)
                if valid_ih & valid_it:
                    # Load input X[b, cin, ih, it] as scalar
                    x_off = b * (Cin * H * T) + cin * (H * T) + ih * T + it
                    x_val = tl.load(X_ptr + x_off).to(tl.float32)
                else:
                    x_val = 0.0

                # Accumulate over output channels in tile
                for j in range(BLOCK_C):
                    cout_j = c_start + j
                    if cout_j < Cout:
                        # Weight address: W[cout_j, cin, kh, kt] in layout (Cout, Cin, 3, 3)
                        w_addr = cout_j * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt
                        w_val = tl.load(W_ptr + w_addr).to(tl.float32)
                        acc[j] += x_val * w_val

    # Add bias
    for j in range(BLOCK_C):
        cout_j = c_start + j
        if cout_j < Cout:
            b_val = tl.load(BIAS_ptr + cout_j).to(tl.float32)
            acc[j] += b_val

    # Store to Y[b, c_offsets, oh, t_out_idx]
    # Y layout: (B, Cout, H, T_out)
    for j in range(BLOCK_C):
        cout_j = c_start + j
        if cout_j < Cout:
            y_off = b * (Cout * H * T_out) + cout_j * (H * T_out) + oh * T_out + t_out_idx
            tl.store(Y_ptr + y_off, acc[j].to(tl.bfloat16))


# Triton GELU kernel (exact, erf-based)
@triton.jit
def gelu_exact_kernel(
    X_ptr,          # *const bfloat16, input tensor
    Y_ptr,          # *bfloat16, output tensor
    N: tl.constexpr,  # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    # exact GELU: 0.5 * x * (1 + erf(x / sqrt(2)))
    inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
    y = 0.5 * x * (1.0 + tl.math.erf(x * inv_sqrt2))
    tl.store(Y_ptr + offsets, y.to(tl.bfloat16), mask=mask)
