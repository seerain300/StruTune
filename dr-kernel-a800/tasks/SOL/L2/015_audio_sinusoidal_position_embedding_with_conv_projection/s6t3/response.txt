import math
import triton
import triton.language as tl


@triton.jit
def conv_gelu_inplace_bf16(x_ptr, w_ptr, b_ptr, y_ptr,
                            B, H_in, W_in, C_in, C_out,
                            K_h, K_w, stride, pad,
                            BLOCK_W: tl.constexpr, BLOCK_C: tl.constexpr):
    # Each program handles one output element y[b, oh, ow, oc]
    b = tl.program_id(0)
    oh = tl.program_id(1)
    ow = tl.program_id(2)
    oc = tl.program_id(3)

    # Guard: only launch within bounds; in practice, grid covers all, but add mask anyway
    mask_b = b < B
    mask_oh = oh < H_in
    mask_ow = ow < W_in
    mask_oc = oc < C_out
    if not (mask_b and mask_oh and mask_ow and mask_oc):
        return

    # Compute input coordinates for 3x3 kernel
    # For stride=2, padding=1: ih = oh * 2 + ky - 1, iw = ow * 2 + kx - 1
    # We will accumulate over in_channels=C_in
    acc = tl.zeros((), dtype=tl.bfloat16)

    # For each kernel position
    for ky in range(K_h):
        for kx in range(K_w):
            ih = oh * stride + ky - pad
            iw = ow * stride + kx - pad
            # Valid if 0 <= ih < H_in and 0 <= iw < W_in
            valid = (ih >= 0) & (ih < H_in) & (iw >= 0) & (iw < W_in)
            for ic in range(C_in):
                # Load input x[b, ic, ih, iw] as scalar; out-of-bounds inputs contribute 0
                # Linear indexing: ((b*C_in + ic)*H_in + ih)*W_in + iw
                x_off = ((b * C_in + ic) * H_in + ih) * W_in + iw
                x_val = tl.load(x_ptr + x_off, mask=valid, other=0.0)

                # Load weight w[oc, ic, ky, kx] as scalar
                # w layout: [C_out, C_in, K_h, K_w] contiguous
                w_off = oc * (C_in * (K_h * K_w)) + ic * (K_h * K_w) + ky * K_w + kx
                w_val = tl.load(w_ptr + w_off)

                # Accumulate
                acc += x_val * w_val

    # Add bias
    b_val = tl.load(b_ptr + oc, mask=mask_oc, other=0.0)
    acc = acc + b_val

    # GELU (tanh approximation)
    # gelu(x) = 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = acc * acc * acc
    inner = c * (acc + 0.044715 * x3)
    t = tl.tanh(inner)
    y_val = 0.5 * acc * (1.0 + t)

    # Store y[b, oh, ow, oc]
    y_off = ((b * C_out) + oc) * (H_in * W_in) + oh * W_in + ow
    tl.store(y_ptr + y_off, y_val)


@triton.jit
def linear_gemm_gelu_bf16(x_ptr, w_ptr, y_ptr,
                          B, T, K, N,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Compute y[b, t, n] = GELU(sum_k x[b, t, k] * w[n, k])
    # x_ptr: [B*T, K]
    # w_ptr: [N, K]
    # y_ptr: [B*T, N]
    pid_m = tl.program_id(0)  # over B*T
    pid_n = tl.program_id(1)  # over N

    b = pid_m // T
    t = pid_m % T
    n = pid_n

    # Accumulator
    acc = tl.zeros((), dtype=tl.bfloat16)

    # Tile over K
    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # Load x[b, t, k_offsets] vector of size BLOCK_K
        x_off = pid_m * K + k_offsets
        x_vec = tl.load(x_ptr + x_off, mask=mask_k, other=0.0)  # [BLOCK_K]

        # Load w[n, k_offsets] vector of size BLOCK_K
        w_off = n * K + k_offsets
        w_vec = tl.load(w_ptr + w_off, mask=mask_k, other=0.0)  # [BLOCK_K]

        # Dot-product accumulate
        acc += tl.sum(x_vec * w_vec, axis=0)

    # GELU
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = acc * acc * acc
    inner = c * (acc + 0.044715 * x3)
    t = tl.tanh(inner)
    y_val = 0.5 * acc * (1.0 + t)

    # Store y[b, t, n]
    y_off = pid_m * N + n
    tl.store(y_ptr + y_off, y_val)


@triton.jit
def add_pos_embed_bf16(y_ptr, pos_ptr, B, T, N):
    # y_ptr: [B, T, N], pos_ptr: [T, N]
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_b
    t = pid_t
    n = pid_n

    # Bounds
    mask_b = b < B
    mask_t = t < T
    mask_n = n < N
    if not (mask_b and mask_t and mask_n):
        return

    y_off = (b * T * N) + (t * N) + n
    pos_off = (t * N) + n

    y_val = tl.load(y_ptr + y_off)
    pos_val = tl.load(pos_ptr + pos_off)
    y_val = y_val + pos_val
    tl.store(y_ptr + y_off, y_val)

import triton
import triton.language as tl


@triton.jit
def conv_gelu_inplace_bf16(x_ptr, w_ptr, b_ptr, y_ptr,
                            B, H_in, W_in, C_in, C_out,
                            K_h, K_w, stride, pad,
                            BLOCK_W: tl.constexpr, BLOCK_C: tl.constexpr):
    # Each program computes one output element y[b, oc, oh, ow]
    b = tl.program_id(0)
    oh = tl.program_id(1)
    ow = tl.program_id(2)
    oc = tl.program_id(3)

    # Bounds
    if (b >= B) or (oh >= H_in) or (ow >= W_in) or (oc >= C_out):
        return

    acc = tl.zeros((), dtype=tl.bfloat16)

    # 3x3 kernel, stride=2, padding=1
    for ky in range(K_h):
        for kx in range(K_w):

            iw = ow * stride + kx - pad
