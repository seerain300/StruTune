# Triton kernel: GELU (tanh approximation), elementwise over N elements
@triton.jit
def gelu_tanh_kernel(X, Y, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X + offsets, mask=mask, other=0.0).to(tl.float32)
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    inner = c * (x + 0.044715 * x3)
    gelu = 0.5 * x * (1.0 + tl.math.tanh(inner))
    tl.store(Y + offsets, gelu, mask=mask)

):

    pid_t = tl.program_id(1)
    pid_m = tl.program_id(2)

    b = pid_b
    t = pid_t
    m_start = pid_m * BLOCK_M
    m_offsets = m_start + tl.arange(0, BLOCK_M)
    m_mask = m_offsets < M

    acc = tl.zeros((BLOCK_M,), dtype=tl.float32)

        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        x_vec = tl.load(x_ptrs, mask=k_mask, other=0.0).to(tl.float32)

        # Load W[:, k_offsets] for M tile -> matrix (BLOCK_M, BLOCK_K)
        w_ptrs = W + m_offsets[:, None] * stride_wm + k_offsets[None, :] * stride_wk
        w_mat = tl.load(w_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

# Triton kernel: Conv2D 3x3, stride=2, padding=1
# Input: X (B, C_in, H, W) — arbitrary strides
# Weights: W (C_out, C_in, 3, 3) — arbitrary strides
# Bias: BIAS (C_out) — contiguous
# Output: Y (B, C_out, H_out, W_out) where H_out = (H - 3)//2 + 1, W_out = (W - 3)//2 + 1
@triton.jit
def conv3x3_stride2_pad1_kernel(
    X, W, BIAS, Y,
    B, C_in, H, W_in, C_out, H_out, W_out,
    stride_xb, stride_xc, stride_xh, stride_xw,
    stride_wo, stride_wi, stride_wkh, stride_wkw,
    stride_yb, stride_yc, stride_yh, stride_yw,
    BLOCK_CO: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_hw = tl.program_id(1)  # tile over H_out * W_out
    pid_co = tl.program_id(2)  # tile over C_out

    h_out_idx = pid_hw // W_out
    w_out_idx = pid_hw % W_out

    co_start = pid_co * BLOCK_CO
    co_offsets = co_start + tl.arange(0, BLOCK_CO)
    co_mask = co_offsets < C_out

    acc = tl.zeros((BLOCK_CO,), dtype=tl.float32)

    # Reduce over input channels and 3x3 kernel
    for ci in range(0, C_in):
        for kh in range(0, 3):
            h_in = h_out_idx * 2 + 1 - kh  # padding=1
            # validity of h_in

                w_in = w_out_idx * 2 + 1 - kw  # padding=1

        # Reshape: (B, C, F, T) -> (B, T, C*F)
        # After conv3, shape is (B, 384, 10, T_out), where T_out = W_out3. In original, B=1, time_dim=T_in, T_out=T_in//8.
        B, C, F, T_out = x3_gelu.shape
        x3_gelu = x3_gelu.permute(0, 3, 1, 2).contiguous()  # (B, T_out, 384, 10)
        C_out_dim = C * F  # 3840

        # Linear projection to d_model (no bias) using Triton
        # X: (B, T_out, K=3840), W: (M=1024, K=3840), Y: (B, T_out, 1024)
        X = x3_gelu.view(B, T_out, C_out_dim).float()  # compute in fp32
        W = conv_out_weight.float()  # (1024, 3840)
        Y = torch.empty((B, T_out, 1024), dtype=torch.float32, device=device)

        BLOCK_M = 64
        BLOCK_K = 64
        grid = (B, triton.cdiv(T_out, 1), triton.cdiv(1024, BLOCK_M))
        linear_no_bias_kernel[grid](
            X, W, Y,
            B, T_out, C_out_dim, 1024,
            X.stride(0), X.stride(1), X.stride(2),
            W.stride(0), W.stride(1),
            Y.stride(0), Y.stride(1), Y.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_K=BLOCK_K
        )

        # Scale embeddings and add positional embedding
        D = 1024
        T_eff = T_out
        POS = positional_embedding[:T_eff, :].float().contiguous()  # (T_eff, D), bfloat16 but load as fp32

        # Scale Y
        Y_scaled = Y * embed_scale

        # Add POS
        add_scaled_pos_emb_kernel[(B, triton.cdiv(T_eff, 128), triton.cdiv(D, 128))](
            Y_scaled, POS, B, T_eff, D, 1.0, 128, 128
        )

        # Cast back to original dtype (bfloat16) for output
        return Y_scaled.to(torch.bfloat16)
