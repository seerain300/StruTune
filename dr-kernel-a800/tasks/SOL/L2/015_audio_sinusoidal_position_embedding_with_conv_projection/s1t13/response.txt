import math
import torch
import triton
import triton.language as tl


# Triton conv2d kernel: 3x3, stride=2, padding=1
# Input: X(B, Cin, H, T) bfloat16
# Weight: W(Cout, Cin, 3, 3) bfloat16
# Bias: Bias(Cout) bfloat16
# Output: Y(B, Cout, H, T_out) bfloat16, with T_out = floor((T - 3)/2 + 1)
@triton.jit
def conv2d_3x3_stride2_padding1_kernel(
    X_ptr,         # *const bfloat16
    W_ptr,         # *const bfloat16
    BIAS_ptr,      # *const bfloat16
    Y_ptr,         # *bfloat16
    Bsz, Cin, H, T, Cout, T_out,
    BLOCK_C: tl.constexpr,
):
    # Grid: (B*H, tiles over Cout, T_out)
    pid_m = tl.program_id(0)   # over B*H
    pid_c = tl.program_id(1)   # over tiles of Cout
    t_out_idx = tl.program_id(2)  # specific time index in output

    b = pid_m // H
    oh = pid_m % H

    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # accumulator per output channel
    acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    for cin in range(Cin):
        for kh in range(3):
            ih = oh + kh - 1  # padding=1
            valid_ih = (ih >= 0) & (ih < H)
            for kt in range(3):
                it = t_out_idx + kt - 1  # padding=1
                valid_it = (it >= 0) & (it < T)
                if valid_ih & valid_it:
                    # Compute input pointer for this (b, cin, ih, it)
                    x_off = b * Cin * H * T + cin * H * T + ih * T + it
                    x_val = tl.load(X_ptr + x_off, mask=True, other=0.0).to(tl.float32)
                    # Load weight for each output channel in this tile
                    # W shape: (Cout, Cin, 3, 3)
                    # For given (cin, kh, kt), we load weights across c_offsets
                    # w_off = c_offsets * (Cin*3*3) + cin*(3*3) + kh*3 + kt
                    w_off = c_offsets * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt
                    w_val = tl.load(W_ptr + w_off, mask=mask_c, other=0.0).to(tl.float32)
                    # FMA accumulate
                    acc += x_val * w_val

    # Add bias
    bias_val = tl.load(BIAS_ptr + c_offsets, mask=mask_c, other=0.0).to(tl.float32)
    acc += bias_val

    # Store to Y
    y_off = b * Cout * H * T_out + c_offsets * (H * T_out) + oh * T_out + t_out_idx
    tl.store(Y_ptr + y_off, acc.to(tl.bfloat16), mask=mask_c)


# Triton GELU (exact) in-kernel: gelu(x) = 0.5 * x * (1 + erf(x / sqrt(2)))
@triton.jit
def gelu_erf_kernel_flat(
    IN_ptr,        # *const bfloat16
    OUT_ptr,       # *bfloat16
    NUMEL: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < NUMEL
    x = tl.load(IN_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    # erf(x / sqrt(2))
    inv_sqrt2 = 0.7071067811865476
    y = 0.5 * x * (1.0 + tl.math.erf(x * inv_sqrt2))
    tl.store(OUT_ptr + offsets, y.to(tl.bfloat16), mask=mask)


# Triton GEMM + add scaled positional embedding
# Input X flattened as (B * T_out3 * K), Weight W (N, K) where N=1024, K=15360
# Output C (B, T_out3, N)
@triton.jit
def gemm_pos_add_kernel(
    X_ptr,         # *const bfloat16
    W_ptr,         # *const bfloat16 (N, K)
    POS_ptr,       # *const bfloat16 (1500, N)
    C_ptr,         # *bfloat16 (B, T_out3, N)
    Bsz, Tsz, Ksz, Nsz, scale,
    BLOCK_N: tl.constexpr,
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    n_start = tl.program_id(2) * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < Nsz

    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K dimension
    for k0 in range(0, Ksz, 1):  # we'll implement as while for generality
        # Note: Triton does not support for ... in range with runtime Ksz; use while
        k = k0
        while k < Ksz:
            # Load x[b, t, k]
            x_off = (b * Tsz * Ksz) + t * Ksz + k
            x_val = tl.load(X_ptr + x_off).to(tl.float32)
            # Load w[n, k] across BLOCK_N
            w_off = n_offsets * Ksz + k
            w_val = tl.load(W_ptr + w_off, mask=mask_n, other=0.0).to(tl.float32)
            acc += x_val * w_val
            k += 1

    # Scale
    acc = acc * scale

    # Add scaled positional embedding slice: pos[b, t, :]
    pos_off = b * Nsz + t * Nsz + n_offsets
    pos_val = tl.load(POS_ptr + pos_off, mask=mask_n, other=0.0).to(tl.float32)
    acc = acc + pos_val

    # Store to C
    c_off = b * Tsz * Nsz + t * Nsz + n_offsets
    tl.store(C_ptr + c_off, acc.to(tl.bfloat16), mask=mask_n)

        super().__init__()

        assert input_features.dtype == torch.bfloat16, "Expected bfloat16 tensors."

        # Shapes

        y1_gelu = torch.empty_like(y1_flat, dtype=torch.bfloat16, device=input_features.device)

        y1 = y1_gelu.view(B, Cout1, H, T1_out)

        # Stage 2: conv2

        # GELU conv2
