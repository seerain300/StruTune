```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: 2D conv stride=2, padding=1, 3x3, arbitrary C_in, OC
# X: (B, C_in, IH, IW) float32, W: (OC, C_in, 3, 3) float32, BIAS: (OC) float32, Y: (B, OC, OH, OW) float32
@triton.jit
def conv2d_stride2_kernel(
    X_ptr, W_ptr, BIAS_ptr, Y_ptr,
    B, C_in, IH, IW, OC,
    OH, OW,
):
    b = tl.program_id(0)
    oc = tl.program_id(1)
    oh = tl.program_id(2)
    ow = tl.program_id(3)

    acc = tl.zeros((), dtype=tl.float32)

    # Sum over input channels and 3x3 taps
    for ic in range(0, C_in):
        for kh in range(0, 3):
            ih = 2 * oh + kh - 1  # stride=2, padding=1
            for kw in range(0, 3):
                iw = 2 * ow + kw - 1
                # bounds check
                in_bounds = (ih >= 0) & (ih < IH) & (iw >= 0) & (iw < IW)
                if in_bounds:
                    x_offset = b * (C_in * IH * IW) + ic * (IH * IW) + ih * IW + iw
                    x_val = tl.load(X_ptr + x_offset)
                    w_offset = oc * (C_in * 9) + ic * 9 + kh * 3 + kw
                    w_val = tl.load(W_ptr + w_offset)
                    acc += x_val * w_val

    # add bias
    bias_val = tl.load(BIAS_ptr + oc)
    acc += bias_val

    # store
    y_offset = b * (OC * OH * OW) + oc * (OH * OW) + oh * OW + ow
    tl.store(Y_ptr + y_offset, acc)


# Triton GELU (tanh approximation) over 1D flattened tensor
@triton.jit
def gelu_kernel_1d(X_ptr, Y_ptr, N, alpha: tl.float32, beta: tl.float32):
    pid = tl.program_id(0)
    offs = pid * 1024 + tl.arange(0, 1024)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    # GELU tanh approximation: 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    gelu = 0.5 * x * (1.0 + tl.tanh(c * (x + 0.044715 * x3)))
    tl.store(Y_ptr + offs, gelu, mask=mask)


# Triton kernel: linear projection (no bias) + scale + positional embedding
# X: (B, T, N) float32, W: (M=1024, N) float32, POS: (T, M) float32, Y: (B, T, M) float32
@triton.jit
def linear_project_pos_kernel(
    X_ptr, W_ptr, POS_ptr, Y_ptr,
    B, T, N, M, scale: tl.float32,
):
    b = tl.program_id(0)
    t = tl.program_id(1)
    # Loop over M in tiles
    for m0 in range(0, M, 128):
        m_offsets = m0 + tl.arange(0, 128)
        mask_m = m_offsets < M
        acc = tl.zeros([128], dtype=tl.float32)
        # Loop over N in tiles of 256
        for n0 in range(0, N, 256):
            n_offsets = n0 + tl.arange(0, 256)
            mask_n = n_offsets < N
            # Load X[b, t, n_offsets]
            x_ptrs = X_ptr + b * (T * N) + t * N + n_offsets
            x_vals = tl.load(x_ptrs, mask=mask_n, other=0.0)  # [256]
            # Load W[m_offsets, n_offsets] as [128, 256]
            w_ptrs = W_ptr + m_offsets[:, None] * N + n_offsets[None, :]
            w_vals = tl.load(w_ptrs, mask=mask_m[:, None] & mask_n[None, :], other=0.0)  # [128, 256]
            # Accumulate: sum over n of x * w per m
            acc += tl.sum(w_vals * x_vals[None, :], axis=1)
        # Apply scale and add pos_emb[t, :]
        acc = acc * scale
        pos_vec = tl.load(POS_ptr + t * M + m_offsets, mask=mask_m, other=0.0)
        acc = acc + pos_vec
        # Store to Y[b, t, m_offsets]
        y_ptrs = Y_ptr + b * (T * M) + t * M + m_offsets
        tl.store(y_ptrs, acc, mask=mask_m)


class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()

    def forward(
        self,
        input_features,        # (B, 1, 80, T), bfloat16, but we cast to fp32
        conv2d1_weight,        # (384, 1, 3, 3), bfloat16
        conv2d1_bias,          # (384), bfloat16
        conv2d2_weight,        # (384, 384, 3, 3), bfloat16
        conv2d2_bias,          # (384), bfloat16
        conv2d3_weight,        # (384, 384, 3, 3), bfloat16
        conv2d3_bias,          # (384), bfloat16
        conv_out_weight,       # (1024, 3840), bfloat16 (but we compute in fp32)
        positional_embedding,  # (1500, 1024), bfloat16 (we slice)
        embed_scale,           # float
    ):
        """
        All computation must be done by Triton kernels; no torch ops in forward.
        """
        # Ensure all tensors are on CUDA and contiguous; cast to fp32 for compute
        B, C_in, IH, IW = input_features.shape
        # Cast to fp32 for kernels
        X0 = input_features.contiguous().to(torch.float32)

        # Stage 1 conv: (1, 80, T) -> (B, 384, 40, (T+1)//2)
        OC1 = 384
        IH1 = IH
        IW1 = IW
        OH1 = (IH1 - 1) // 2 + 1  # 40
        IW1_out = (IW1 - 1) // 2 + 1
        X1 = torch.empty((B, OC1, OH1, IW1_out), dtype=torch.float32, device=X0.device)
        conv2d_stride2_kernel[(B, OC1, OH1, IW1_out)](
            X0, conv2d1_weight.contiguous().to(torch.float32),
            conv2d1_bias.contiguous().to(torch.float32), X1,
            B, 1, IH1, IW1, OC1, OH1, IW1_out
        )
        # GELU after conv1
        Y1 = torch.empty_like(X1)
        N1 = X1.numel()
        gelu_kernel_1d[(triton.cdiv(N1, 1024),)](
            X1.reshape(-1), Y1.reshape(-1), N1, 0.7978845608, 0.044715
        )

        # Stage 2 conv: (384, 40, (T+1)//2) -> (B, 384, 20, ((T+1)//2)//2)
        OC2 = 384
        C_in2 = OC1
        IH2 = OH1
        IW2 = IW1_out
        OH2 = (IH2 - 1) // 2 + 1  # 20
        IW2_out = (IW2 - 1) // 2 + 1
        X2 = torch.empty((B, OC2, OH2, IW2_out), dtype=torch.float32, device=X0.device)
        conv2d_stride2_kernel[(B, OC2, OH2, IW2_out)](
            Y1, conv2d2_weight.contiguous().to(torch.float32),
            conv2d2_bias.contiguous().to(torch.float32), X2,
            B, C_in2, IH2, IW2, OC2, OH2, IW2_out
        )
        # GELU after conv2
        Y2 = torch.empty_like(X2)
        N2 = X2.numel()
        gelu_kernel_1d[(triton.cdiv(N2, 1024),)](
            X2.reshape(-1), Y2.reshape(-1), N2, 0.7978845608, 0.044715
        )

        # Stage 3 conv: (384, 20, ((T+1)//2)//2) -> (B, 384, 10, (((T+1)//2)//2)//2)
        OC3 = 384
        C_in3 = OC2
        IH3 = OH2
        IW3 = IW2_out
        OH3 = (IH3 - 1) // 2 + 1  # 10
        IW3_out = (IW3 - 1) // 2 + 1
        X3 = torch.empty((B, OC3, OH3, IW3_out), dtype=torch.float32, device=X0.device)
        conv2d_stride2_kernel[(B, OC3, OH3, IW3_out)](
            Y2, conv2d3_weight.contiguous().to(torch.float32),
            conv2d3_bias.contiguous().to(torch.float32), X3,
            B, C_in3, IH3, IW3, OC3, OH3, IW3_out
        )
        # GELU after conv3
        Y3 = torch.empty_like(X3)
        N3 = X3.numel()
        gelu_kernel_1d[(triton.cdiv(N3, 1024),)](
            X3.reshape(-1), Y3.reshape(-1), N3, 0.7978845608, 0.044715
        )

        # Permute to (B, time_after_conv, 384*10)
        # time_after_conv is provided by the evaluation harness; assume we take Y3.shape[-1]*Y3.shape[-2]
        T_after = Y3.shape[-1] * Y3.shape[-2]  # for given code, this is 384*10=3840
        X_lin = Y3.permute(0, 3, 1, 2).contiguous().view(B, T_after, 3840).to(torch.float32)

        # Final linear projection + scale + positional embedding
        # W shape (1024, 3840)
        M = 1024
        N = X_lin.shape[-1]
        Y_final = torch.empty((B, X_lin.shape[1], M), dtype=torch.float32, device=X_lin.device)
        pos_emb = positional_embedding[:X_lin.shape[1], :].contiguous().to(torch.float32)
        linear_project_pos_kernel[(B, X_lin.shape[1],)](
            X_lin.reshape(B, X_lin.shape[1], N),
            conv_out_weight.contiguous().to(torch.float32),
            pos_emb.reshape(X_lin.shape[1], M),
            Y_final.reshape(B, X_lin.shape[1], M),
            B, X_lin.shape[1], N, M, embed_scale
        )

        return Y_final
```

```python
        T_after = Y3.shape[-1] * Y3.shape[-2]  # equals 3840 per code, but harness may supply different time_after_conv; here we use actual conv result size
        X_lin = Y3.permute(0, 3, 1, 2).contiguous().view(B, T_after, 3840).to(torch.float32)

        # Final linear projection + scale + positional embedding
        M = 1024
        N = X_lin.shape[-1]
        Y_final = torch.empty((B, X_lin.shape[1], M), dtype=torch.float32, device=X_lin.device)
        pos_emb = positional_embedding[:X_lin.shape[1], :].contiguous().to(torch.float32)
        linear_project_pos_kernel[(B, X_lin.shape[1],)](
            X_lin.reshape(B, X_lin.shape[1], N),
            conv_out_weight.contiguous().to(torch.float32),
            pos_emb.reshape(X_lin.shape[1], M),
            Y_final.reshape(B, X_lin.shape[1], M),
            B, X_lin.shape[1], N, M, embed_scale
        )

        return Y_final
```

- The conv kernel grid is (B, OC, OH, OW), iterating over input channels and 3x3 taps; accumulation is fp32; bias is added per oc.
