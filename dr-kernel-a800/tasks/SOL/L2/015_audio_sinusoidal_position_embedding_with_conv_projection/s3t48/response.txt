- Input x0: (B, 1, 80, time_dim), dtype bfloat16.
- Conv1: (1 -> 384 channels), stride=2, padding=1. Output shape: (B, 384, 40, time_dim//2).
- Conv2: (384 -> 384), stride=2, padding=1. Output shape: (B, 384, 20, time_dim//4).
- Conv3: (384 -> 384), stride=2, padding=1. Output shape: (B, 384, 10, time_dim//8) = (B, 384, 10, Tafter).
- GELU activation on conv3 output. PyTorch default is erf-based; we implement exact GELU in Triton using tl.math.erf to match it.
- Permute: (B, 384, 10, Tafter) -> (B, Tafter, 384*10) = (B, Tafter, 3840). Implemented via Triton gather by mapping indices, no torch permute/view.
- Linear projection: (B, Tafter, 3840) @ (1024, 3840) -> (B, Tafter, 1024). Implemented in Triton: grid over (B, Tafter, 1024) and iterate k=0..3839 for accumulation.
- Scale by embed_scale = sqrt(1024) = 32.0. Implemented in Triton.
- Add positional embedding: pos is (1500, 1024) float32. We add pos[t, :] to every batch sample. Implemented in Triton, broadcasting across batch. Ensure Triton receives strides as int64 to avoid index type mismatch errors.

ModelNew.forward will allocate input tensor and launch:
1) conv2d_stride2_bias_kernel three times (conv1, conv2, conv3),
2) gelu_erf_kernel on conv3 output,
3) gather_x25_kernel to form (B, Tafter, 3840) from conv3_gelu,
4) linear_proj_kernel to compute final (B, Tafter, 1024),
5) scale_kernel to multiply by 32.0,

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def conv2d_stride2_bias_kernel(
    x_ptr,            # *bf16 input tensor
    w_ptr,            # *bf16 weight tensor
    b_ptr,            # *bf16 bias tensor
    y_ptr,            # *bf16 output tensor
    B: tl.constexpr,
    Ci: tl.constexpr,           # input channels
    H, W,                       # input spatial dims
    Co,                         # output channels
    Kh: tl.constexpr, Kw: tl.constexpr,  # kernel size
    Ho, Wo,                     # output spatial dims
    x_s0, x_s1, x_s2, x_s3,     # x strides (int64)
    w_s0, w_s1, w_s2, w_s3,     # w strides (int64)
    y_s0, y_s1, y_s2, y_s3,     # y strides (int64)
):
    b_id = tl.program_id(0)
    co_id = tl.program_id(1)
    ho_id = tl.program_id(2)
    wo_id = tl.program_id(3)

    acc = tl.zeros((), dtype=tl.float32)

    for ci in range(Ci):
        for kh in range(Kh):
            hi = ho_id * 2 + 1 - kh
            for kw in range(Kw):
                wi = wo_id * 2 + 1 - kw
                in_bounds = (hi >= 0) & (hi < H) & (wi >= 0) & (wi < W)
                x_off = b_id * x_s0 + ci * x_s1 + hi * x_s2 + wi * x_s3
                x_val = tl.load(x_ptr + x_off, mask=in_bounds, other=0.0).to(tl.float32)
                w_off = co_id * w_s0 + ci * w_s1 + kh * w_s2 + kw * w_s3
                w_val = tl.load(w_ptr + w_off).to(tl.float32)
                acc += x_val * w_val

    b_val = tl.load(b_ptr + co_id).to(tl.float32)
    acc += b_val

    y_off = b_id * y_s0 + co_id * y_s1 + ho_id * y_s2 + wo_id * y_s3
    tl.store(y_ptr + y_off, acc)


@triton.jit
def gelu_erf_kernel(
    x_ptr,  # *bf16
    y_ptr,  # *bf16
    B: tl.constexpr, Co: tl.constexpr, Ho: tl.constexpr, Wo: tl.constexpr,
    x_s0, x_s1, x_s2, x_s3,
    y_s0, y_s1, y_s2, y_s3,
):
    b_id = tl.program_id(0)
    co_id = tl.program_id(1)
    ho_id = tl.program_id(2)
    wo_id = tl.program_id(3)

    x_off = b_id * x_s0 + co_id * x_s1 + ho_id * x_s2 + wo_id * x_s3
    x_val = tl.load(x_ptr + x_off).to(tl.float32)

    inv_sqrt2 = 0.7071067811865476  # 1 / sqrt(2)
    gelu = 0.5 * x_val * (1.0 + tl.math.erf(x_val * inv_sqrt2))

    y_off = b_id * y_s0 + co_id * y_s1 + ho_id * y_s2 + wo_id * y_s3
    tl.store(y_ptr + y_off, gelu)


@triton.jit
def gather_x25_kernel(
    x_ptr,  # *bf16, conv3_gelu tensor (B, 384, 10, Tafter)
    y_ptr,  # *bf16, output tensor (B, Tafter, 384*10)
    B, Co, Ho, Wo,  # for conv3: Co=384, Ho=10, Wo=Tafter
    x_s0, x_s1, x_s2, x_s3,
    y_s0, y_s1, y_s2, y_s3,
    Tafter,  # equal to Wo
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)  # 0..Tafter-1
    k_id = tl.program_id(2)  # 0..Co*Ho-1

    co = k_id // Ho
    ho = k_id % Ho
    d = co * Ho + ho  # d in [0, Co*Ho)

    x_off = b_id * x_s0 + co * x_s1 + ho * x_s2 + t_id * x_s3
    x_val = tl.load(x_ptr + x_off).to(tl.float32)

    # y tensor is (B, Tafter, Co*Ho) with contiguous last dim
    y_off = b_id * y_s0 + t_id * y_s1 + d * y_s2
    tl.store(y_ptr + y_off, x_val)


@triton.jit
def linear_proj_kernel(
    x_ptr,    # *bf16, (B, Tafter, 3840)
    w_ptr,    # *bf16, (1024, 3840)
    y_ptr,    # *bf16, (B, Tafter, 1024)
    B, T, K,  # K=3840
    x_s0, x_s1, x_s2,
    w_s0, w_s1,
    y_s0, y_s1, y_s2,
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)
    d_id = tl.program_id(2)  # output feature dimension 0..1023

    acc = tl.zeros((), dtype=tl.float32)

    # Iterate over input features
    for k in range(0, K):
        x_off = b_id * x_s0 + t_id * x_s1 + k * x_s2
        x_val = tl.load(x_ptr + x_off).to(tl.float32)
        w_off = d_id * w_s0 + k * w_s1
        w_val = tl.load(w_ptr + w_off).to(tl.float32)
        acc += x_val * w_val

    y_off = b_id * y_s0 + t_id * y_s1 + d_id * y_s2
    tl.store(y_ptr + y_off, acc)


@triton.jit
def scale_kernel(
    x_ptr,  # *bf16
    y_ptr,  # *bf16
    B, T, D,
    scale,  # float32
    x_s0, x_s1, x_s2,
    y_s0, y_s1, y_s2,
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)
    d_id = tl.program_id(2)

    x_off = b_id * x_s0 + t_id * x_s1 + d_id * x_s2
    x_val = tl.load(x_ptr + x_off).to(tl.float32) * scale
    y_off = b_id * y_s0 + t_id * y_s1 + d_id * y_s2
    tl.store(y_ptr + y_off, x_val)


@triton.jit
def add_pos_embedding_kernel(
    x_ptr,   # *bf16, (B, Tafter, 1024)
    pos_ptr, # *fp32, (1500, 1024)
    B, T, D,
    pos_s0, pos_s1,  # strides for pos
    x_s0, x_s1, x_s2,
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)
    d_id = tl.program_id(2)

    # load pos[t, d] as fp32
    pos_off = t_id * pos_s0 + d_id * pos_s1
    pos_val = tl.load(pos_ptr + pos_off).to(tl.float32)

    x_off = b_id * x_s0 + t_id * x_s1 + d_id * x_s2
    x_val = tl.load(x_ptr + x_off).to(tl.float32)
    y_val = x_val + pos_val

    tl.store(x_ptr + x_off, y_val)


class ModelNew(torch.nn.Module):
    def __init__(self, conv2d1_weight, conv2d1_bias, conv2d2_weight, conv2d2_bias, conv2d3_weight, conv2d3_bias, conv_out_weight, positional_embedding, embed_scale):
        super().__init__()
        # Store weights and buffers
        self.register_buffer("conv2d1_weight", conv2d1_weight)  # (384, 1, 3, 3)
        self.register_buffer("conv2d1_bias", conv2d1_bias)      # (384,)
        self.register_buffer("conv2d2_weight", conv2d2_weight)  # (384, 384, 3, 3)
        self.register_buffer("conv2d2_bias", conv2d2_bias)      # (384,)
        self.register_buffer("conv2d3_weight", conv2d3_weight)  # (384, 384, 3, 3)
        self.register_buffer("conv2d3_bias", conv2d3_bias)      # (384,)
        self.register_buffer("conv_out_weight", conv_out_weight)  # (1024, 3840)
        # positional_embedding is float32 in the provided get_inputs; we keep it as float32 buffer
        self.register_buffer("positional_embedding", positional_embedding)  # (1500, 1024), float32
        self.embed_scale = float(embed_scale)  # sqrt(1024) = 32.0

    def forward(self, input_features):
        # Ensure input is contiguous
        x = input_features.contiguous()
        B, Ci, H, W = x.shape  # Ci=1

        # conv1: output (B, 384, 40, W1) with W1 = W//2
        Co1, Ci1, Kh, Kw = self.conv2d1_weight.shape
        Ho1 = (H + 2 * 1 - Kh) // 2 + 1
        Wo1 = (W + 2 * 1 - Kw) // 2 + 1

        x1 = torch.empty((B, Co1, Ho1, Wo1), device=x.device, dtype=x.dtype)
        grid1 = (B, Co1, Ho1, Wo1)
        conv2d_stride2_bias_kernel[grid1](
            x, self.conv2d1_weight, self.conv2d1_bias, x1,
            B, Ci, H, W, Co1, Kh, Kw, Ho1, Wo1,
            x.stride(0), x.stride(1), x.stride(2), x.stride(3),
            self.conv2d1_weight.stride(0), self.conv2d1_weight.stride(1), self.conv2d1_weight.stride(2), self.conv2d1_weight.stride(3),
            x1.stride(0), x1.stride(1), x1.stride(2), x1.stride(3),
            num_warps=4, num_stages=2,
        )

        # conv2: (384 -> 384), stride=2, padding=1
        Co2, Ci2, Kh, Kw = self.conv2d2_weight.shape
        assert Co2 == 384 and Ci2 == 384
        H2 = Ho1
        W2 = Wo1 // 2
        x2 = torch.empty((B, Co2, H2, W2), device=x.device, dtype=x.dtype)
        grid2 = (B, Co2, H2, W2)
        conv2d_stride2_bias_kernel[grid2](
            x1, self.conv2d2_weight, self.conv2d2_bias, x2,
            B, Ci2, H2, W2, Co2, Kh, Kw, H2, W2,
            x1.stride(0), x1.stride(1), x1.stride(2), x1.stride(3),
            self.conv2d2_weight.stride(0), self.conv2d2_weight.stride(1), self.conv2d2_weight.stride(2), self.conv2d2_weight.stride(3),
            x2.stride(0), x2.stride(1), x2.stride(2), x2.stride(3),
            num_warps=4, num_stages=2,
        )

        # conv3: (384 -> 384), stride=2, padding=1
        Co3, Ci3, Kh, Kw = self.conv2d3_weight.shape
        assert Co3 == 384 and Ci3 == 384
        H3 = H2
        W3 = W2 // 2
        Tafter = W3
        x3 = torch.empty((B, Co3, H3, W3), device=x.device, dtype=x.dtype)
        grid3 = (B, Co3, H3, W3)
        conv2d_stride2_bias_kernel[grid3](
            x2, self.conv2d3_weight, self.conv2d3_bias, x3,
            B, Ci3, H3, W3, Co3, Kh, Kw, H3, W3,
            x2.stride(0), x2.stride(1), x2.stride(2), x2.stride(3),
            self.conv2d3_weight.stride(0), self.conv2d3_weight.stride(1), self.conv2d3_weight.stride(2), self.conv2d3_weight.stride(3),
            x3.stride(0), x3.stride(1), x3.stride(2), x3.stride(3),
            num_warps=4, num_stages=2,
        )

        # GELU on conv3 output
        x3_gelu = torch.empty_like(x3, dtype=torch.float32)  # compute GELU in fp32 for accuracy, cast later
        grid_gelu = (B, Co3, H3, W3)
        gelu_erf_kernel[grid_gelu](
            x3, x3_gelu,
            B, Co3, H3, W3,
            x3.stride(0), x3.stride(1), x3.stride(2), x3.stride(3),
            x3_gelu.stride(0), x3_gelu.stride(1), x3_gelu.stride(2), x3_gelu.stride(3),
            num_warps=4, num_stages=2,
        )
        # Cast back to original dtype
        x3_gelu = x3_gelu.to(x.dtype)

        # Gather to form (B, Tafter, 3840): k = co * Ho + ho, d = co * Ho + ho
        K = Co3 * H3  # 384 * 10 = 3840
        x25 = torch.empty((B, Tafter, K), device=x.device, dtype=x.dtype)
        grid_gather = (B, Tafter, K)
        gather_x25_kernel[grid_gather](
            x3_gelu, x25,
            B, Co3, H3, W3,
            x3_gelu.stride(0), x3_gelu.stride(1), x3_gelu.stride(2), x3_gelu.stride(3),
            x25.stride(0), x25.stride(1), x25.stride(2),
            Tafter,
            num_warps=4, num_stages=2,
        )

        # Linear projection to 1024
        y_lin = torch.empty((B, Tafter, 1024), device=x.device, dtype=torch.float32)
        grid_linear = (B, Tafter, 1024)
        linear_proj_kernel[grid_linear](
            x25, self.conv_out_weight, y_lin,
            B, Tafter, K,
            x25.stride(0), x25.stride(1), x25.stride(2),
            self.conv_out_weight.stride(0), self.conv_out_weight.stride(1),
            y_lin.stride(0), y_lin.stride(1), y_lin.stride(2),
            num_warps=4, num_stages=2,
        )
        # Scale by embed_scale
        y_scaled = torch.empty_like(y_lin, dtype=torch.float32)  # keep fp32 for scaling
        scale_kernel[grid_linear](
            y_lin, y_scaled,
            B, Tafter, 1024,
            self.embed_scale,
            y_lin.stride(0), y_lin.stride(1), y_lin.stride(2),
            y_scaled.stride(0), y_scaled.stride(1), y_scaled.stride(2),
            num_warps=4, num_stages=2,
        )
        # Add positional embedding
        # Ensure strides are int64 for Triton
        pos_s0 = self.positional_embedding.stride(0)  # int64
        pos_s1 = self.positional_embedding.stride(1)  # int64
        add_pos_embedding_kernel[grid_linear](
            y_scaled, self.positional_embedding,
            B, Tafter, 1024,
            pos_s0, pos_s1,
            y_scaled.stride(0), y_scaled.stride(1), y_scaled.stride(2),
            num_warps=4, num_stages=2,
        )
        # Cast to original dtype for output if needed
        y_out = y_scaled.to(x.dtype)

        return y_out
```
