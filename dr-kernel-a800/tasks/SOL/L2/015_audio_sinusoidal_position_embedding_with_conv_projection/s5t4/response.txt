def conv2d1_stride2_bias_gelu_kernel(
    X_ptr, W_ptr, BIAS_ptr, B_ptr,
    N, T, Co,
    # input sizes and strides
    x_strideN, x_strideC, x_strideF, x_strideT,
    # output sizes
    F_out, T_out,
    # conv params
    STRIDE: tl.constexpr,  # 2
    PADDING: tl.constexpr, # 1
    # meta
    K: tl.constexpr,       # kernel size 3
    BLOCK_CO: tl.constexpr,  # tile over output channels (not used directly, vectorize over co)
):
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_t_out = tl.program_id(2)

    # Accumulate in fp32 for numeric stability
    acc = tl.zeros((), dtype=tl.float32)

    # Ci is 1 by design
    ci = 0

    # Loop over 3x3 kernel; padding=1 => input indices can be negative/positive, but
    # the original PyTorch conv2d with padding=1 will produce zero when out-of-bounds.
    # Here we handle bounds manually.
    for kh in range(K):
        for kw in range(K):
            f_in = pid_t_out * STRIDE - PADDING + kh
            t_in = pid_t_out * STRIDE - PADDING + kw
            in_bounds = (f_in >= 0) and (f_in < F_out) and (t_in >= 0) and (t_in < T_out)
            # Load input X[n, ci=0, f_in, t_in] if in-bounds, else 0
            x_off = pid_n * x_strideN + ci * x_strideC + f_in * x_strideF + t_in * x_strideT
            x_val = tl.load(X_ptr + x_off, mask=in_bounds, other=0.0)
            x_val = x_val.to(tl.float32)
            # Load weight W[co, ci, kh, kw] = W[co, 0, kh, kw]
            w_off = co * (1 * K * K) + kh * K + kw  # since ci=0 and KxK
            # W_ptr layout is [Co, Ci, Kh, Kw] with Ci=1
            w_val = tl.load(W_ptr + w_off)
            w_val = w_val.to(tl.float32)
            acc += x_val * w_val

    # Add bias
    bias_val = tl.load(BIAS_ptr + pid_co).to(tl.float32)
    acc = acc + bias_val

    # GELU: y = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    c0 = 0.7978845608028654  # sqrt(2/pi)
    x3 = acc * acc * acc
    gelu = 0.5 * acc * (1.0 + tl.tanh(c0 * (acc + 0.044715 * x3)))

    # Store to B[n, co, t_out]
    b_off = pid_n * (Co * T_out) + pid_co * T_out + pid_t_out
    # Original dtype is bfloat16; store as bfloat16
    tl.store(B_ptr + b_off, gelu.to(tl.bfloat16))

# X is [N, T, M], W is [K, M], Y is [N, T, K]
@triton.jit
def linear_bmm_kernel(
    X_ptr, W_ptr, Y_ptr,
    N, T, M, K,
    x_strideN, x_strideT, x_strideM,
    w_strideK, w_strideM,
    y_strideN, y_strideT, y_strideK,
    BLOCK_M: tl.constexpr,  # tile size over M
    BLOCK_K: tl.constexpr,  # output k tile size
):
    pid_nt = tl.program_id(0)  # linearized (n, t)
    pid_k = tl.program_id(1)

    n = pid_nt // T
    t = pid_nt % T

    k_start = pid_k * BLOCK_K
    offs_k = k_start + tl.arange(0, BLOCK_K)
    mask_k = offs_k < K

    acc = tl.zeros([BLOCK_K], dtype=tl.float32)

    # Iterate over M in blocks
    for m_start in range(0, M, BLOCK_M):
        offs_m = m_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M

        # Load X[n, t, offs_m]
        x_off = n * x_strideN + t * x_strideT + offs_m * x_strideM
        x_vals = tl.load(X_ptr + x_off, mask=mask_m, other=0.0).to(tl.float32)  # [BLOCK_M]

        # Load W[offs_k, offs_m] as [BLOCK_K, BLOCK_M]
        w_ptrs = W_ptr + (offs_k[:, None] * w_strideK + offs_m[None, :] * w_strideM)
        w_mask = mask_k[:, None] & mask_m[None, :]
        w_vals = tl.load(w_ptrs, mask=w_mask, other=0.0).to(tl.float32)  # [BLOCK_K, BLOCK_M]

        # Accumulate: acc[k] += sum_m w[k, m] * x[m]
        acc += tl.sum(w_vals * x_vals[None, :], axis=1)

    # Store Y[n, t, offs_k]
    y_off = n * y_strideN + t * y_strideT + offs_k * y_strideK
    tl.store(Y_ptr + y_off, acc.to(tl.bfloat16), mask=mask_k)

            in_bounds_f = (f_in >= 0) and (f_in < F_out)
            in_bounds_t = (t_in >= 0) and (t_in < T_out)
            in_bounds = in_bounds_f and in_bounds_t

            x_off = pid_n * x_strideN + ci * x_strideC + f_in * x_strideF + t_in * x_strideT
            x_val = tl.load(X_ptr + x_off, mask=in_bounds, other=0.0).to(tl.float32)

            # Load weight W[co, ci, kh, kw] = W[co, 0, kh, kw]
            # Weight layout: [Co, Ci=1, 3, 3]
            w_off = pid_co * (1 * K * K) + kh * K + kw
            w_val = tl.load(W_ptr + w_off).to(tl.float32)

            acc += x_val * w_val

    # Add bias
    bias_val = tl.load(BIAS_ptr + pid_co).to(tl.float32)
    acc = acc + bias_val

    # GELU: y = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    c0 = 0.7978845608028654  # sqrt(2/pi)
    x3 = acc * acc * acc
    gelu = 0.5 * acc * (1.0 + tl.tanh(c0 * (acc + 0.044715 * x3)))

    # Store to B[n, co, t_out] with strides: B is [N, Co, T_out] contiguous, so b_stride = Co*T_out, b_stride_co = T_out, b_stride_t = 1
    b_off = pid_n * (Co * T_out) + pid_co * T_out + pid_t_out
    tl.store(B_ptr + b_off, gelu.to(tl.bfloat16))


# Batched GEMV Triton kernel: compute Y[n, t, k] = sum_j X[n, t, j] * W[k, j]
# X is [N, T, M], W is [K, M], Y is [N, T, K]
@triton.jit
def linear_bmm_kernel(
    X_ptr, W_ptr, Y_ptr,
    N, T, M, K,
    x_strideN, x_strideT, x_strideM,
    w_strideK, w_strideM,
    y_strideN, y_strideT, y_strideK,
    BLOCK_M: tl.constexpr,  # tile size over M
    BLOCK_K: tl.constexpr,  # tile size over K
):
    pid_nt = tl.program_id(0)  # linearized (n, t)
    pid_k = tl.program_id(1)

    n = pid_nt // T
    t = pid_nt % T

    k_start = pid_k * BLOCK_K
    offs_k = k_start + tl.arange(0, BLOCK_K)
    mask_k = offs_k < K

    acc = tl.zeros([BLOCK_K], dtype=tl.float32)

    # Iterate over M in blocks
    for m_start in range(0, M, BLOCK_M):
        offs_m = m_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M

        # Load X[n, t, offs_m]
        x_off = n * x_strideN + t * x_strideT + offs_m * x_strideM
        x_vals = tl.load(X_ptr + x_off, mask=mask_m, other=0.0).to(tl.float32)  # [BLOCK_M]

        # Load W[offs_k, offs_m] as [BLOCK_K, BLOCK_M]
        w_ptrs = W_ptr + (offs_k[:, None] * w_strideK + offs_m[None, :] * w_strideM)
        w_mask = mask_k[:, None] & mask_m[None, :]
        w_vals = tl.load(w_ptrs, mask=w_mask, other=0.0).to(tl.float32)  # [BLOCK_K, BLOCK_M]

        # Accumulate: acc[k] += sum_m w[k, m] * x[m]
        acc += tl.sum(w_vals * x_vals[None, :], axis=1)

    # Store Y[n, t, offs_k]
    y_off = n * y_strideN + t * y_strideT + offs_k * y_strideK
    tl.store(Y_ptr + y_off, acc.to(tl.bfloat16), mask=mask_k)
