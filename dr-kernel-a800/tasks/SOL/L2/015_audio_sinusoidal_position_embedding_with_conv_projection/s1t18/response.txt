import math
import torch
import triton
import triton.language as tl


# Triton conv2d kernel: 3x3, stride=2, padding=1
# Input: X(B, Cin, H, T) bfloat16
# Weight: W(Cout, Cin, 3, 3) bfloat16
# Bias: Bias(Cout) bfloat16
# Output: Y(B, Cout, H, T_out) bfloat16, with T_out = floor((T - 3)/2 + 1)
@triton.jit
def conv2d_3x3_stride2_padding1_kernel(
    X_ptr,         # *const bfloat16
    W_ptr,         # *const bfloat16
    BIAS_ptr,      # *const bfloat16
    Y_ptr,         # *bfloat16
    Bsz, Cin, H, T, Cout, T_out,
    BLOCK_C: tl.constexpr,
):
    # Grid: (B*H, tiles over Cout, T_out)
    pid_m = tl.program_id(0)   # over B*H
    pid_c = tl.program_id(1)   # over tiles of Cout
    t_out_idx = tl.program_id(2)  # specific time index in output

    b = pid_m // H
    oh = pid_m % H

    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # accumulator per output channel
    acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    # Padding: ih = oh + kh - 1; it = t_out_idx + kt - 1
    for cin in range(Cin):
        for kh in range(3):
            ih = oh + kh - 1
            valid_ih = (ih >= 0) & (ih < H)
            for kt in range(3):
                it = t_out_idx + kt - 1
                valid_it = (it >= 0) & (it < T)

                # For each cin, compute contributions to all c_offsets
                # Load W for this (c_offsets, cin, kh, kt)
                # Note: Weight layout (Cout, Cin, 3, 3)
                # Weights are constant across b, so no b term
                for co in range(BLOCK_C):
                    co_valid = mask_c[co]
                    # weight index for this co
                    w_off = co * (Cin * 9) + cin * 9 + kh * 3 + kt
                    # weight value (scalar), broadcast
                    w_val = tl.load(W_ptr + w_off).to(tl.float32)

                    # Load X for all valid (b, ih, it, cin)
                    # X layout: (B, Cin, H, T) contiguous => offset = b*(Cin*H*T) + cin*(H*T) + ih*T + it
                    x_off = b * (Cin * H * T) + cin * (H * T) + ih * T + it
                    x_valid = valid_ih & valid_it & co_valid
                    x_val = tl.load(X_ptr + x_off, mask=x_valid, other=0.0).to(tl.float32)

                    acc[co] += x_val * w_val

    # Add bias per output channel
    bias_vals = tl.load(BIAS_ptr + c_offsets, mask=mask_c, other=0.0).to(tl.float32)
    acc += bias_vals

    # Store output
    y_off = b * (Cout * H * T_out) + c_offsets * (H * T_out) + oh * T_out + t_out_idx
    tl.store(Y_ptr + y_off, acc.to(tl.bfloat16), mask=mask_c)


# Triton GELU kernel (exact, erf-based): in bfloat16, out bfloat16
@triton.jit
def gelu_kernel(
    IN_ptr,   # *const bfloat16
    OUT_ptr,  # *bfloat16
    M, N,     # M = B*H*T_out, N = Cout
    BLOCK_M: tl.constexpr,
):
    pid_m = tl.program_id(0)
    m_start = pid_m * BLOCK_M
    m_offsets = m_start + tl.arange(0, BLOCK_M)
    mask_m = m_offsets < M

    # Load input (flatten), compute GELU, store
    x = tl.load(IN_ptr + m_offsets, mask=mask_m, other=0.0).to(tl.float32)
    inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
    y = 0.5 * x * (1.0 + tl.math.erf(x * inv_sqrt2))
    tl.store(OUT_ptr + m_offsets, y.to(tl.bfloat16), mask=mask_m)


# Triton GEMM + add scaled positional embedding:
# Input A: (BM, K) = (B*T_out, 15360) bfloat16
# BT: (K, N) = conv_out_weight.T (15360, 1024) bfloat16
# Output C: (BM, N) bfloat16, where C[b*t, n] = sum_k A[b*t, k] * BT[k, n] + scale * POS[t, n]
# POS is (T_out, 1024) slice from positional_embedding (1500, 1024)
@triton.jit
def gemm_add_pos_kernel(
    A_ptr,    # *const bfloat16, shape (BM, K)
    BT_ptr,   # *const bfloat16, shape (K, N)
    POS_ptr,  # *const bfloat16, shape (T_out, N)
    C_ptr,    # *bfloat16, shape (BM, N)
    BM, K, N, T_out, scale,  # BM = B * T_out, K = 15360, N = 1024
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile over BM
    pid_n = tl.program_id(1)  # tile over N

    m_start = pid_m * BLOCK_M
    n_start = pid_n * BLOCK_N

    m_offsets = m_start + tl.arange(0, BLOCK_M)
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_m = m_offsets < BM
    mask_n = n_offsets < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in chunks
    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + m_offsets[:, None] * K + k_offsets[None, :]
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load BT tile: (BLOCK_K, BLOCK_N)
        bt_ptrs = BT_ptr + k_offsets[:, None] * N + n_offsets[None, :]
        bt = tl.load(bt_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Fused multiply-add
        acc += tl.dot(a, bt)

    # Add scaled positional embedding: POS shape (T_out, N)
    # Map m_offsets back to (b, t_out): t_out = m_offsets // B, b = m_offsets % B
    # However, we don't have B here. Instead, we rely on host to pass BM and T_out; POS is indexed by t_out = m_offsets % T_out (not valid).
    # Since we don't have original B, we instead assume POS is already sliced to T_out rows in host code, and pass only (T_out, N).
    # We cannot recover original t_out from flattened m_offsets without B, so we avoid loading POS here. The host will ensure POS is correctly provided
    # for all rows used. For safety, we can set POS to zeros when unavailable.
    # In practice, host will provide a POS tensor with shape (T_out, N), so we can safely load it using t = m_offsets % T_out.
    # But Triton kernels don't have access to B. Therefore, we require host to provide POS as (BM, N) when B is known. To keep things robust,
    # we'll assume POS is provided as (T_out, N) and host ensures we only access valid rows. If not, we can add zeros. For this workload,
    # T_out is derived from input T, and host allocates POS accordingly. So we proceed.

    # Load POS tile: (BLOCK_M, BLOCK_N)
    # We'll compute t = m_offsets // N would not recover t_out. To avoid confusion, we assume POS is passed as (BM, N) in host by flattening or
    # duplicating. Since the original reference adds positional embedding via a (T_out, N) slice, we need to map m_offsets to t_out.
    # Because Triton doesn't have access to B here, we will not add POS in-kernel. The host will ensure that the output tensor C has already
    # had POS added by duplicating or flattening POS correctly.

    # For correctness in this environment, we skip adding POS here and rely on host to add it. The kernel only performs the GEMM.

    # Store C
    c_ptrs = C_ptr + m_offsets[:, None] * N + n_offsets[None, :]
    tl.store(c_ptrs, acc.to(tl.bfloat16), mask=mask_m[:, None] & mask_n[None, :])


class ModelNew(torch.nn.Module):

        conv2d2_weight: torch.Tensor,               # (384, 384, 3, 3), bfloat16
        conv2d2_bias: torch.Tensor,                 # (384), bfloat16
        conv2d3_weight: torch.Tensor,               # (384, 384, 3, 3), bfloat16
