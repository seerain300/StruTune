# Triton kernel: Conv2D 3x3, stride=2, padding=1
# X: (B, C_in, H, W), W: (C_out, C_in, 3, 3), BIAS: (C_out)
# Y: (B, C_out, H_out, W_out), where H_out = (H - 3)//2 + 1, W_out = (W - 3)//2 + 1
@triton.jit
def conv2d_3x3_stride2_pad1_kernel(
    X, W, BIAS, Y,
    B, C_in, H, W_in, C_out, H_out, W_out,
    stride_xb, stride_xc, stride_xh, stride_xw,
    stride_wo, stride_wi, stride_wkh, stride_wkw,
    stride_yb, stride_yc, stride_yh, stride_yw,
    BLOCK_CO: tl.constexpr,
):
    # program ids: batch, output location, output-channel tile
    pid_b = tl.program_id(0)
    pid_hw = tl.program_id(1)
    pid_co_tile = tl.program_id(2)

    # derive h_out, w_out indices from pid_hw
    # pid_hw = h_out * W_out + w_out
    h_out = pid_hw // W_out
    w_out = pid_hw % W_out

    # output channel tile
    co_start = pid_co_tile * BLOCK_CO
    co_offsets = co_start + tl.arange(0, BLOCK_CO)
    co_mask = co_offsets < C_out

    # accumulator for this tile (per batch, per (h_out, w_out), per co)
    acc = tl.zeros((BLOCK_CO,), dtype=tl.float32)

    # reduction over input channels and 3x3 kernel
    for ci in range(0, C_in):
        for kh in range(0, 3):
            for kw in range(0, 3):
                in_h = h_out * 2 + 1 + kh  # stride=2, padding=1
                in_w = w_out * 2 + 1 + kw
                # bounds check for input
                in_valid = (in_h >= 0) & (in_h < H) & (in_w >= 0) & (in_w < W_in)
                # X[b, ci, in_h, in_w] load
                x_ptrs = X + pid_b * stride_xb + ci * stride_xc + in_h * stride_xh + in_w * stride_xw
                x_val = tl.load(x_ptrs, mask=in_valid, other=0.0).to(tl.float32)
                # W[co, ci, kh, kw] load vector for this ci, kh, kw across co tile

                w_vec = tl.load(w_ptrs, mask=co_mask, other=0.0).to(tl.float32)
                # accumulate
                acc += x_val * w_vec

    # add bias
    bias_ptrs = BIAS + co_offsets
    bias = tl.load(bias_ptrs, mask=co_mask, other=0.0).to(tl.float32)
    acc += bias

    # store Y[b, co, h_out, w_out]
    y_ptrs = Y + pid_b * stride_yb + co_offsets * stride_yc + h_out * stride_yh + w_out * stride_yw
    tl.store(y_ptrs, acc, mask=co_mask)


# Triton kernel: Elementwise GELU (tanh approximation)
@triton.jit
def gelu_tanh_kernel(X, Y, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X + offsets, mask=mask, other=0.0).to(tl.float32)
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    inner = c * (x + 0.044715 * x3)
    gelu = 0.5 * x * (1.0 + tl.math.tanh(inner))
    tl.store(Y + offsets, gelu, mask=mask)


# Triton kernel: Linear projection (no bias) Y = X @ W^T
# X: (B, T, K) flattened to contiguous
# W: (M, K) flattened to contiguous
# Y: (B, T, M) flattened to contiguous
@triton.jit
def linear_no_bias_kernel(
    X_flat, W_flat, Y_flat,
    B, T, K, M,
    BLOCK_M: tl.constexpr,  # tile over output channels (M)
    BLOCK_K: tl.constexpr   # tile over reduction (K)
):
    # program ids for batch, time position, output channel tile
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_m = tl.program_id(2)

    m_start = pid_m * BLOCK_M
    m_offsets = m_start + tl.arange(0, BLOCK_M)
    m_mask = m_offsets < M

    # base index in flattened X for (b, t, :)
    base_x = (pid_b * T + pid_t) * K

    # accumulator for this (b, t, m-tile)
    acc = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # reduce over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        # load X[b, t, k_chunk] -> vector of size BLOCK_K
        x_ptrs = X_flat + base_x + k_offsets
        x_vec = tl.load(x_ptrs, mask=k_mask, other=0.0).to(tl.float32)

        # load W[m_offsets, k_offsets] -> matrix [BLOCK_M, BLOCK_K]
        w_ptrs = W_flat + m_offsets[:, None] * K + k_offsets[None, :]
        w_mat = tl.load(w_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # acc[m] += sum_k x_vec[k] * w_mat[m, k]
        acc += tl.sum(w_mat * x_vec[None, :], axis=1)

    # store Y[b, t, m_offsets]
    y_base = (pid_b * T + pid_t) * M
    y_ptrs = Y_flat + y_base + m_offsets
    tl.store(y_ptrs, acc, mask=m_mask)


# Triton kernel: add scaled positional embedding
# Y: (B, T, M), POS: (M, ), SCALE: scalar
@triton.jit
def add_scaled_pos_emb_kernel(Y, POS, SCALE, B, T, M):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_m = tl.program_id(2)
    m = pid_m
    # load pos scalar and scale
    pos_val = tl.load(POS + m).to(tl.float32)
    # Y[b, t, m] = Y[b, t, m] + scale * pos_val
    y_ptr = Y + pid_b * (T * M) + pid_t * M + m
    y_val = tl.load(y_ptr).to(tl.float32)
    y_val = y_val + SCALE * pos_val
    tl.store(y_ptr, y_val)
