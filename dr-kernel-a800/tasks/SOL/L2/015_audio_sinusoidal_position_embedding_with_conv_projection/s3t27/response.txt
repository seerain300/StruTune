```python
import math
import triton
import triton.language as tl


# Conv2d stride=2, padding=1
# Input: x (B, Ci, H, W), weights w (Co, Ci, Kh, Kw), bias (Co,)
# Output: y (B, Co, Ho, Wo), where Ho = floor((H + 2*pad - Kh)/2) + 1, Wo similarly
@triton.jit
def conv2d_stride2_kernel(
    x_ptr, w_ptr, b_ptr, y_ptr,
    B, Ci, H, W, Co, Kh, Kw, Ho, Wo,
    x_s0, x_s1, x_s2, x_s3,
    w_s0, w_s1, w_s2, w_s3,
    y_s0, y_s1, y_s2, y_s3,
):
    b_id = tl.program_id(0)  # batch
    co_id = tl.program_id(1) # output channel
    ho_id = tl.program_id(2) # output height
    wo_id = tl.program_id(3) # output width

    acc = tl.zeros((), dtype=tl.float32)

    # Accumulate over input channels and kernel
    for ci in range(Ci):
        for kh in range(Kh):
            hi = ho_id * 2 + 1 - kh
            for kw in range(Kw):
                wi = wo_id * 2 + 1 - kw
                in_bounds = (hi >= 0) & (hi < H) & (wi >= 0) & (wi < W)
                x_off = b_id * x_s0 + ci * x_s1 + hi * x_s2 + wi * x_s3
                x_val = tl.load(x_ptr + x_off, mask=in_bounds, other=0.0).to(tl.float32)
                w_off = co_id * w_s0 + ci * w_s1 + kh * w_s2 + kw * w_s3
                w_val = tl.load(w_ptr + w_off).to(tl.float32)
                acc += x_val * w_val

    # add bias
    b_val = tl.load(b_ptr + co_id).to(tl.float32)
    acc += b_val

    # store
    y_off = b_id * y_s0 + co_id * y_s1 + ho_id * y_s2 + wo_id * y_s3
    tl.store(y_ptr + y_off, acc)  # store as fp32; y tensor dtype is fp32 in this context


# GELU (tanh approximation): gelu(x) = 0.5*x*(1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
@triton.jit
def gelu_tanh_kernel(
    x_ptr, y_ptr,
    B, Co, Ho, Wo,
    x_s0, x_s1, x_s2, x_s3,
    y_s0, y_s1, y_s2, y_s3,
):
    b_id = tl.program_id(0)
    co_id = tl.program_id(1)
    ho_id = tl.program_id(2)
    wo_id = tl.program_id(3)

    x_off = b_id * x_s0 + co_id * x_s1 + ho_id * x_s2 + wo_id * x_s3
    x_val = tl.load(x_ptr + x_off).to(tl.float32)

    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x_val * x_val * x_val
    gelu = 0.5 * x_val * (1.0 + tl.math.tanh(c * (x_val + 0.044715 * x3)))

    y_off = b_id * y_s0 + co_id * y_s1 + ho_id * y_s2 + wo_id * y_s3
    tl.store(y_ptr + y_off, gelu)


# Gather conv3_gelu[b, co, ho, wo] into x_gather[b, t, k], where k = co*10 + ho and t = wo
# conv3_gelu shape: (B, Co=384, Ho=10, Wo=Tafter)
# x_gather shape:  (B, Tafter, Co*Ho = 3840)
@triton.jit
def gather_x25_kernel(
    x3_gelu_ptr, xgather_ptr,
    B, Co, Ho, Wo,
    xg_s0, xg_s1, xg_s2,  # (B, T, K) strides
    x3_s0, x3_s1, x3_s2, x3_s3,  # (B, Co, Ho, Wo) strides
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)  # time index = output width Wo
    k_id = tl.program_id(2)  # flattened (co, ho): k in [0..Co*Ho-1]

    co = k_id // Ho
    ho = k_id % Ho

    # read conv3_gelu[b, co, ho, t]
    x_off = b_id * x3_s0 + co * x3_s1 + ho * x3_s2 + t_id * x3_s3
    x_val = tl.load(x3_gelu_ptr + x_off).to(tl.float32)

    # write to xgather[b, t, k]
    xg_off = b_id * xg_s0 + t_id * xg_s1 + k_id * xg_s2
    tl.store(xgather_ptr + xg_off, x_val)


# Linear projection: out[b, t, d] = sum_k xgather[b, t, k] * W[d, k], where W is (D=1024, K=3840)
@triton.jit
def linear_proj_kernel(
    xg_ptr, w_ptr, out_ptr,
    B, T, D, K,
    xg_s0, xg_s1, xg_s2,  # (B, T, K)
    w_s0, w_s1,            # (D, K)
    out_s0, out_s1, out_s2,  # (B, T, D)
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)
    d_id = tl.program_id(2)

    acc = tl.zeros((), dtype=tl.float32)
    # sum over K
    for k in range(K):
        xg_off = b_id * xg_s0 + t_id * xg_s1 + k * xg_s2
        x_val = tl.load(xg_ptr + xg_off).to(tl.float32)

        w_off = d_id * w_s0 + k * w_s1
        w_val = tl.load(w_ptr + w_off).to(tl.float32)

        acc += x_val * w_val

    out_off = b_id * out_s0 + t_id * out_s1 + d_id * out_s2
    tl.store(out_ptr + out_off, acc)


# Multiply by embed_scale in Triton
@triton.jit
def scale_kernel(
    inp_ptr, out_ptr,
    B, T, D,
    inp_s0, inp_s1, inp_s2,  # (B, T, D)
    out_s0, out_s1, out_s2,  # (B, T, D)
    scale: tl.float32,
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)
    d_id = tl.program_id(2)

    in_off = b_id * inp_s0 + t_id * inp_s1 + d_id * inp_s2
    val = tl.load(inp_ptr + in_off).to(tl.float32)
    val = val * scale
    out_off = b_id * out_s0 + t_id * out_s1 + d_id * out_s2
    tl.store(out_ptr + out_off, val)


# Broadcast-add positional embedding pos[t, d] to out_scaled[b, t, d], where pos is (Tmax, D)
@triton.jit
def add_pos_kernel(
    out_ptr, pos_ptr,
    B, T, D,
    out_s0, out_s1, out_s2,  # (B, T, D)
    pos_s0, pos_s1,          # (Tmax, D)
):
    b_id = tl.program_id(0)
    t_id = tl.program_id(1)
    d_id = tl.program_id(2)

    out_off = b_id * out_s0 + t_id * out_s1 + d_id * out_s2
    val = tl.load(out_ptr + out_off).to(tl.float32)

    pos_off = t_id * pos_s0 + d_id * pos_s1
    pos_val = tl.load(pos_ptr + pos_off).to(tl.float32)

    val = val + pos_val

    tl.store(out_ptr + out_off, val)


class ModelNew(torch.nn.Module):
    def __init__(self,
                 conv2d1_weight, conv2d1_bias,
                 conv2d2_weight, conv2d2_bias,
                 conv2d3_weight, conv2d3_bias,
                 conv_out_weight, positional_embedding, embed_scale):
        super().__init__()
        # Store buffers (no tensors need to be moved during forward; just pass pointers)
        # Note: original get_inputs returns bfloat16 tensors. We assume input dtype is bfloat16.
        self.register_buffer("conv2d1_weight", conv2d1_weight)  # (384, 1, 3, 3)
        self.register_buffer("conv2d1_bias", conv2d1_bias)      # (384,)
        self.register_buffer("conv2d2_weight", conv2d2_weight)  # (384, 384, 3, 3)
        self.register_buffer("conv2d2_bias", conv2d2_bias)      # (384,)
        self.register_buffer("conv2d3_weight", conv2d3_weight)  # (384, 384, 3, 3)
        self.register_buffer("conv2d3_bias", conv2d3_bias)      # (384,)
        self.register_buffer("conv_out_weight", conv_out_weight)  # (1024, 3840)
        self.register_buffer("positional_embedding", positional_embedding)  # (1500, 1024)
        self.embed_scale = float(embed_scale)  # sqrt(1024) = 32.0

    def forward(self, input_features):
        # input_features: (B, 1, 80, time_dim), bfloat16
        x0 = input_features.contiguous()
        B, Ci, H0, W0 = x0.shape
        Kh, Kw = 3, 3

        # Conv1: (1 -> 384), stride=2, padding=1
        Co1 = self.conv2d1_weight.shape[0]
        H1 = (H0 + 2 * 1 - Kh) // 2 + 1
        W1 = (W0 + 2 * 1 - Kw) // 2 + 1
        x1 = torch.empty((B, Co1, H1, W1), device=x0.device, dtype=x0.dtype)

        grid1 = (B, Co1, H1, W1)
        conv2d_stride2_kernel[grid1](
            x0, self.conv2d1_weight, self.conv2d1_bias, x1,
            B, Ci, H0, W0, Co1, Kh, Kw, H1, W1,
            x0.stride(0), x0.stride(1), x0.stride(2), x0.stride(3),
            self.conv2d1_weight.stride(0), self.conv2d1_weight.stride(1), self.conv2d1_weight.stride(2), self.conv2d1_weight.stride(3),
            x1.stride(0), x1.stride(1), x1.stride(2), x1.stride(3),
            num_warps=2,
        )

        # GELU on conv1
        x1_gelu = torch.empty_like(x1, dtype=torch.float32, device=x1.device)  # store GELU output in fp32
        grid1_gelu = (B, Co1, H1, W1)
        gelu_tanh_kernel[grid1_gelu](
            x1, x1_gelu,
            B, Co1, H1, W1,
            x1.stride(0), x1.stride(1), x1.stride(2), x1.stride(3),
            x1_gelu.stride(0), x1_gelu.stride(1), x1_gelu.stride(2), x1_gelu.stride(3),
            num_warps=2,
        )

        # Conv2: (384 -> 384), stride=2, padding=1
        Co2 = self.conv2d2_weight.shape[0]
        H2 = (H1 + 2 * 1 - Kh) // 2 + 1
        W2 = (W1 + 2 * 1 - Kw) // 2 + 1
        x2 = torch.empty((B, Co2, H2, W2), device=x1_gelu.device, dtype=torch.float32)

        grid2 = (B, Co2, H2, W2)
        conv2d_stride2_kernel[grid2](
            x1_gelu, self.conv2d2_weight, self.conv2d2_bias, x2,
            B, Co1, H1, W1, Co2, Kh, Kw, H2, W2,
            x1_gelu.stride(0), x1_gelu.stride(1), x1_gelu.stride(2), x1_gelu.stride(3),
            self.conv2d2_weight.stride(0), self.conv2d2_weight.stride(1), self.conv2d2_weight.stride(2), self.conv2d2_weight.stride(3),
            x2.stride(0), x2.stride(1), x2.stride(2), x2.stride(3),
            num_warps=2,
        )

        # GELU on conv2
        x2_gelu = torch.empty((B, Co2, H2, W2), device=x2.device, dtype=torch.float32)
        grid2_gelu = (B, Co2, H2, W2)
        gelu_tanh_kernel[grid2_gelu](
            x2, x2_gelu,
            B, Co2, H2, W2,
            x2.stride(0), x2.stride(1), x2.stride(2), x2.stride(3),
            x2_gelu.stride(0), x2_gelu.stride(1), x2_gelu.stride(2), x2_gelu.stride(3),
            num_warps=2,
        )

        # Conv3: (384 -> 384), stride=2, padding=1
        Co3 = self.conv2d3_weight.shape[0]
        H3 = (H2 + 2 * 1 - Kh) // 2 + 1
        W3 = (W2 + 2 * 1 - Kw) // 2 + 1
        x3_gelu = torch.empty((B, Co3, H3, W3), device=x2_gelu.device, dtype=torch.float32)

        grid3 = (B, Co3, H3, W3)
        conv2d_stride2_kernel[grid3](
            x2_gelu, self.conv2d3_weight, self.conv2d3_bias, x3_gelu,
            B, Co2, H2, W2, Co3, Kh, Kw, H3, W3,
            x2_gelu.stride(0), x2_gelu.stride(1), x2_gelu.stride(2), x2_gelu.stride(3),
            self.conv2d3_weight.stride(0), self.conv2d3_weight.stride(1), self.conv2d3_weight.stride(2), self.conv2d3_weight.stride(3),
            x3_gelu.stride(0), x3_gelu.stride(1), x3_gelu.stride(2), x3_gelu.stride(3),
            num_warps=2,
        )

        # Compute Tafter = x3_gelu.shape[3]
        Tafter = x3_gelu.shape[3]

        # Prepare xgather: (B, Tafter, Co3*H3 = 3840) -> gather x3_gelu[b, co, ho, wo]
        K = Co3 * H3  # 384 * 10 = 3840
        xgather = torch.empty((B, Tafter, K), device=x3_gelu.device, dtype=torch.float32)

        grid_gather = (B, Tafter, K)
        gather_x25_kernel[grid_gather](
            x3_gelu, xgather,
            B, Co3, H3, Tafter,
            xgather.stride(0), xgather.stride(1), xgather.stride(2),
            x3_gelu.stride(0), x3_gelu.stride(1), x3_gelu.stride(2), x3_gelu.stride(3),
            num_warps=4,
        )

        # Linear projection: (B, Tafter, 3840) @ (1024, 3840) -> (B, Tafter, 1024)
        D = self.conv_out_weight.shape[0]  # 1024
        out = torch.empty((B, Tafter, D), device=xgather.device, dtype=torch.float32)

        grid_linear = (B, Tafter, D)
        linear_proj_kernel[grid_linear](
            xgather, self.conv_out_weight, out,
            B, Tafter, D, K,
            xgather.stride(0), xgather.stride(1), xgather.stride(2),
            self.conv_out_weight.stride(0), self.conv_out_weight.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            num_warps=4,
        )

        # Scale by embed_scale (sqrt(1024) = 32.0)
        out_scaled = torch.empty_like(out, dtype=torch.float32, device=out.device)

        scale_kernel[grid_linear](
            out, out_scaled,
            B, Tafter, D,
            out.stride(0), out.stride(1), out.stride(2),
            out_scaled.stride(0), out_scaled.stride(1), out_scaled.stride(2),
            self.embed_scale,
            num_warps=4,
        )

        # Add positional embedding: pos is (1500, 1024), only first Tafter rows are needed
        # Broadcast-add along batch dimension without torch ops
        out_final = torch.empty_like(out_scaled, dtype=torch.float32, device=out_scaled.device)

        add_pos_kernel[grid_linear](
            out_scaled, self.positional_embedding,
            B, Tafter, D,
            out_scaled.stride(0), out_scaled.stride(1), out_scaled.stride(2),
            self.positional_embedding.stride(0), self.positional_embedding.stride(1),
            num_warps=4,
        )

        return out_final
```
