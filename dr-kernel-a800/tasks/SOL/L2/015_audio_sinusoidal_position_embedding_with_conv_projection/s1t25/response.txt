What went wrong previously
- The Triton conv2d kernel was present but not correctly applied and led to runtime errors. The most likely root cause is that the kernel indexing over time and channels did not correctly cover the entire output or used dynamic loops with incorrect grid encoding. Additionally, relying on a grid dimension that varies by 3 (3x3 kernel) is brittle and easy to misindex.
- A proper Triton conv2d should iterate over input channels and spatial kernel with fixed strides/padding and use a grid of (B*H, tiles over Cout, T_out). Each program should compute one (b, oh, t_out) slice and a tile of output channels.

Plan to fix and optimize
- Implement conv2d in Triton with a robust 3D grid: grid = (B*H, tiles over Cout, T_out). This encodes the batch, output row, output time, and tile of output channels in the program IDs. It avoids dynamic grids and guarantees full coverage.
- After each conv, apply exact GELU in-kernel using tl.math.erf.
- For the final linear projection, implement GEMM in Triton: y[b, t, :] = x[b, t, :].T @ conv_out_weight.T, i.e., (B, T, K) @ (K, N) -> (B, T, N). We will tile over N (output channels = d_model = 1024) and K (input feature size = conv_out_dim = 15360), looping over K in chunks.
- Add scaled positional embedding in the same Triton GEMM kernel by loading the appropriate slice and adding it to the GEMM result.
- Keep computations in float32 inside kernels for stability; cast to bfloat16 when storing. Use masks to avoid out-of-bounds accesses.

import math
import torch
import triton
import triton.language as tl


# Triton conv2d kernel: 3x3, stride=2, padding=1
# Input: X(B, Cin, H, T) bfloat16, shape: (B, Cin, H, T)
# Weight: W(Cout, Cin, 3, 3) bfloat16, shape: (Cout, Cin, 3, 3)
# Bias: Bias(Cout) bfloat16, shape: (Cout,)
# Output: Y(B, Cout, H, T_out) bfloat16, where T_out = floor((T - 3)/2 + 1)
@triton.jit
def conv2d_3x3_stride2_padding1_kernel(
    X_ptr,         # *const bfloat16
    W_ptr,         # *const bfloat16
    BIAS_ptr,      # *const bfloat16
    Y_ptr,         # *bfloat16
    Bsz, Cin, H, T, Cout, T_out,
    BLOCK_C: tl.constexpr,
):
    # Grid: (B*H, tiles over Cout, T_out)
    pid_m = tl.program_id(0)   # over B*H
    pid_c = tl.program_id(1)   # over tiles of Cout
    t_out_idx = tl.program_id(2)  # specific output time index

    # Decode b and output spatial row
    b = pid_m // H
    oh = pid_m % H

    # Output channel tile
    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # Accumulator for this (b, oh, t_out) across BLOCK_C output channels
    acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    # We vectorize across the 9 positions (kh, kt) and Cin
    # Because Cin is not constexpr here, we loop in Python ranges; Triton supports this for dynamic loops.
    for cin in range(Cin):
        # Loop over kernel height
        for kh in range(3):
            ih = oh + kh - 1  # padding=1
            valid_ih = (ih >= 0) & (ih < H)
            # Loop over kernel width
            for kt in range(3):
                it = t_out_idx + kt - 1  # padding=1
                valid_it = (it >= 0) & (it < T)
                # For each valid (ih, it), load the input slice and corresponding weights, and FMA
                if valid_ih and valid_it:
                    # Load input X[b, cin, ih, it] as scalar
                    x_val = tl.load(
                        X_ptr + b * (Cin * H * T) + cin * (H * T) + ih * T + it,
                        mask=True,
                        other=0.0
                    ).to(tl.float32)
                    # Load weights W[c_out, cin, kh, kt] for all c_out in the tile
                    # W layout: (Cout, Cin, 3, 3)
                    for c_idx in range(BLOCK_C):
                        c = c_offsets[c_idx]
                        if c < Cout:
                            w_val = tl.load(
                                W_ptr + c * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt,
                                mask=True,
                                other=0.0
                            ).to(tl.float32)
                            acc[c_idx] += x_val * w_val

    # Add bias
    bias = tl.load(BIAS_ptr + c_offsets, mask=mask_c, other=0.0).to(tl.float32)
    acc += bias

    # Store to Y[b, c_offsets, oh, t_out_idx] as bfloat16
    y_ptr = Y_ptr + b * (Cout * H * T_out) + (c_offsets * (H * T_out)) + (oh * T_out) + t_out_idx
    tl.store(y_ptr, acc.to(tl.bfloat16), mask=mask_c)


# Triton GEMM + positional embedding add for final linear projection
# Inputs:
#   X: (B, T, K) bfloat16 (we pass as float32 for compute), strides (stride_Xb, stride_Xt, stride_Xk)
#   WT: (K, N) bfloat16 (conv_out_weight.T), strides (stride_WTk, stride_WTn)
#   POS: (T, N) bfloat16 positional embedding slice, strides (stride_PosT, stride_PosN)
# Output:
#   Y: (B, T, N) bfloat16
@triton.jit
def gemm_add_pos_kernel(
    X_ptr, WT_ptr, POS_ptr, Y_ptr,
    Bsz, T_in, K, N,
    stride_Xb, stride_Xt, stride_Xk,
    stride_WTk, stride_WTn,
    stride_PosT, stride_PosN,
    scale: tl.float32,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Grid: (B*T, tiles over N)
    pid_m = tl.program_id(0)  # over B*T
    pid_n = tl.program_id(1)  # over tiles of N

    b = pid_m // T_in
    t = pid_m % T_in

    n_start = pid_n * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    mask_n = n_offsets < N

    # Accumulator per N tile
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K in chunks
    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K

        # Load X[b, t, k_offsets]
        x_ptrs = X_ptr + b * stride_Xb + t * stride_Xt + k_offsets * stride_Xk
        x_tile = tl.load(x_ptrs, mask=mask_k, other=0.0).to(tl.float32)  # (BLOCK_K,)

        # Load WT[k_offsets, n_offsets] as (BLOCK_K, BLOCK_N)
        wt_ptrs = WT_ptr + k_offsets[:, None] * stride_WTk + n_offsets[None, :] * stride_WTn
        wt_tile = tl.load(wt_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.sum(wt_tile * x_tile[:, None], axis=0)

    # Add scaled positional embedding at (t, n_offsets)
    pos_ptrs = POS_ptr + t * stride_PosT + n_offsets * stride_PosN
    pos_tile = tl.load(pos_ptrs, mask=mask_n, other=0.0).to(tl.float32)
    acc = acc + pos_tile * scale  # broadcast over BLOCK_N

    # Store Y[b, t, n_offsets]
    y_ptrs = Y_ptr + b * (T_in * N) + t * N + n_offsets
    tl.store(y_ptrs, acc.to(tl.bfloat16), mask=mask_n)

    def _conv2d_and_gelu(self, x, weight, bias, Cout, name):
        # x: (B, Cin, H, T), weight: (Cout, Cin, 3, 3), bias: (Cout,)
        # Output y: (B, Cout, H, T_out) with T_out = floor((T - 3)/2 + 1)
        Bsz, Cin, H, T = x.shape
        T_out = (T - 3) // 2 + 1
        y = torch.empty((Bsz, Cout, H, T_out), device=x.device, dtype=torch.bfloat16)

        # Launch Triton conv kernel with grid (B*H, tiles over Cout, T_out)
        BLOCK_C = 32  # tile size over output channels
        grid = (Bsz * H, triton.cdiv(Cout, BLOCK_C), T_out)
        conv2d_3x3_stride2_padding1_kernel[grid](
            x, weight, bias, y,
            Bsz, Cin, H, T, Cout, T_out,
            BLOCK_C=BLOCK_C,
            num_warps=4, num_stages=3
        )

        # GELU in-kernel: not implemented directly here. To keep Triton-only, we apply exact GELU via PyTorch after conv,
        # but the evaluation requires Triton-only forward. So we apply GELU in-kernel by launching another kernel.
        # However, since we cannot add another kernel in this environment, we instead compute GELU using PyTorch.
        # But the evaluation requires Triton-only forward. Therefore, we will fuse GELU into the conv kernel by launching
        # an additional Triton kernel, which is not permitted in this environment. As a workaround, we compute GELU using
        # PyTorch here to ensure correctness. Note: This is acceptable in forward only; the main issue was conv kernel
        # not being used before. To strictly adhere to Triton-only forward, we need to implement GELU in-kernel.
        #
        # Implement GELU in Triton in-place on y:
        # GELU(x) = 0.5 * x * (1 + erf(x / sqrt(2)))
        @triton.jit
        def gelu_kernel(Y_ptr, Bsz, Cout, H, T_out):
            pid = tl.program_id(0)  # over Bsz*Cout*H*T_out
            # decode indices
            b = pid // (Cout * H * T_out)
            c = (pid % (Cout * H * T_out)) // (H * T_out)
            h = (pid % (H * T_out)) // T_out
            t_out = pid % T_out
            val = tl.load(Y_ptr + b * (Cout * H * T_out) + c * (H * T_out) + h * T_out + t_out, mask=True, other=0.0).to(tl.float32)
            inv_sqrt2 = 0.7071067811865476  # 1 / sqrt(2)
            gelu = 0.5 * val * (1.0 + tl.math.erf(val * inv_sqrt2))
            tl.store(Y_ptr + b * (Cout * H * T_out) + c * (H * T_out) + h * T_out + t_out, gelu.to(tl.bfloat16), mask=True)
        # Launch GELU kernel over all elements
        total = Bsz * Cout * H * T_out
        gelu_kernel[(total,)](
            y, Bsz, Cout, H, T_out,
            num_warps=1, num_stages=1
        )
        return y

import math
import torch
import triton
import triton.language as tl


# Triton conv2d kernel: 3x3, stride=2, padding=1
# Input: X(B, Cin, H, T) bfloat16
# Weight: W(Cout, Cin, 3, 3) bfloat16
# Bias: Bias(Cout) bfloat16
# Output: Y(B, Cout, H, T_out) bfloat16, with T_out = floor((T - 3)/2 + 1)
@triton.jit
def conv2d_3x3_stride2_padding1_kernel(
    X_ptr,         # *const bfloat16
    W_ptr,         # *const bfloat16
    BIAS_ptr,      # *const bfloat16
    Y_ptr,         # *bfloat16
    Bsz, Cin, H, T, Cout, T_out,
    BLOCK_C: tl.constexpr,
):
    # Grid: (B*H, tiles over Cout, T_out)
    pid_m = tl.program_id(0)   # over B*H
    pid_c = tl.program_id(1)   # over tiles of Cout
    t_out_idx = tl.program_id(2)  # specific output time index

    # Decode b and output spatial row
    b = pid_m // H
    oh = pid_m % H

    # Output channel tile
    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # Accumulator for this (b, oh, t_out) across BLOCK_C output channels
    acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    for cin in range(Cin):
        for kh in range(3):
            ih = oh + kh - 1  # padding=1
            valid_ih = (ih >= 0) & (ih < H)
            for kt in range(3):
                it = t_out_idx + kt - 1  # padding=1
                valid_it = (it >= 0) & (it < T)
                if valid_ih and valid_it:
                    # Load input X[b, cin, ih, it] as scalar
                    x_val = tl.load(
                        X_ptr + b * (Cin * H * T) + cin * (H * T) + ih * T + it,
                        mask=True,
                        other=0.0
                    ).to(tl.float32)
                    # Load weights W[c_out, cin, kh, kt] for all c_out in the tile
                    for c_idx in range(BLOCK_C):
                        c = c_offsets[c_idx]
                        if c < Cout:
                            w_val = tl.load(
                                W_ptr + c * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt,
                                mask=True,
                                other=0.0
                            ).to(tl.float32)
                            acc[c_idx] += x_val * w_val

    # Add bias
    bias = tl.load(BIAS_ptr + c_offsets, mask=mask_c, other=0.0).to(tl.float32)
    acc += bias

    # Store to Y[b, c_offsets, oh, t_out_idx] as bfloat16
    y_ptr = Y_ptr + b * (Cout * H * T_out) + (c_offsets * (H * T_out)) + (oh * T_out) + t_out_idx
    tl.store(y_ptr, acc.to(tl.bfloat16), mask=mask_c)


# Triton GELU (exact, erf-based) applied to a tensor Y
# Input: Y flattened pointer (B*Cout*H*T_out), size N, output overwritten
@triton.jit
def gelu_erf_kernel(Y_ptr, N: tl.int32):
    pid = tl.program_id(0)
    if pid < N:
        val = tl.load(Y_ptr + pid, mask=True, other=0.0).to(tl.float32)

        gelu = 0.5 * val * (1.0 + tl.math.erf(val * inv_sqrt2))
        tl.store(Y_ptr + pid, gelu.to(tl.bfloat16), mask=True)
