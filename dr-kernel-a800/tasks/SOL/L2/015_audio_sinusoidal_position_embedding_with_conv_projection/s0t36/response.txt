죄송합니다. 이전 제안이 Triton kernel의 구현이 잘못되어 모든 작업이 실패했습니다. 이번에는 TRITON-ONLY 요구사항을 엄격히 준수하며, ModelNew.forward가 Triton kernel을 필수적으로 호출하고, 모든 수치 연산이 Triton에서 이루어지도록 정확한 구현을 제공합니다.

전체 계획

- Convolution 3개와 GELU: stride=2, padding=1, kernel=3x3. 이를 Triton kernel conv3x3_s2_p1_gelu로 구현합니다. 입력 X는 [B, IC, F_in, T_in], weight W는 [OC, IC, 3, 3], bias는 [OC], 출력 Y는 [B, OC, F_out, T_out]. 각 output tile에 대해 3x3 input 창을 순회하며, IC루프를 통해 weight와 input을 곱한 후 acc에 누적합니다. bias를 더하고 GELU를 적용합니다.
- Reshape: [B, channels, F, T] -> [B, T, channels*F]는 PyTorch view(메타데이터 변환)입니다. 계산이 아닙니다.
- Linear projection: [B*T3, 3840] x [3840, 1024]를 Triton matmul kernel로 계산합니다.
- Scaling and positional embedding add: 두 과정을 Triton elementwise kernel로 처리합니다. embed_scale(=sqrt(1024)=32.0) 곱하고, positional_embedding[:T3, :]을 add합니다.

ModelNew.forward에서 모든 수치 연산은 Triton kernel이 실행되도록 하겠습니다. Triton kernel은 float32로 동작하며, get_inputs에서 제공된 bfloat16 tensors는 float32로 변환하여 전달합니다.

import math
import torch
import torch.nn as nn

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: Conv2d 3x3, stride=2, padding=1, generic IC -> OC
# X: [B, IC, F_in, T_in], W: [OC, IC, 3, 3], bias: [OC], Y: [B, OC, F_out, T_out]
@triton.jit
def conv3x3_s2_p1_gelu(
    X_ptr, W_ptr, BIAS_ptr, Y_ptr,
    B, IC, F_in, T_in, OC, F_out, T_out,
    x_sN, x_sC, x_sF, x_sT,
    w_sOC, w_sIC, w_sKH, w_sKW,
    y_sN, y_sOC, y_sF, y_sT,
    BLOCK_F: tl.constexpr, BLOCK_T: tl.constexpr,
):
    # Launch grid: 1D over (b, oc, f_block, t_block)
    total_blocks = B * OC * ((F_out + BLOCK_F - 1) // BLOCK_F) * ((T_out + BLOCK_T - 1) // BLOCK_T)
    pid = tl.program_id(0)

    # Decode pid into (b, oc, f_block, t_block)
    grid_f_blocks = (F_out + BLOCK_F - 1) // BLOCK_F
    grid_t_blocks = (T_out + BLOCK_T - 1) // BLOCK_T

    b = pid // (OC * grid_f_blocks * grid_t_blocks)
    rem = pid % (OC * grid_f_blocks * grid_t_blocks)
    oc = rem // (grid_f_blocks * grid_t_blocks)
    f_block = rem // grid_t_blocks % grid_f_blocks
    t_block = rem % grid_t_blocks

    # Compute output indices for this tile
    f_out_start = f_block * BLOCK_F
    t_out_start = t_block * BLOCK_T
    f_out_idx = f_out_start + tl.arange(0, BLOCK_F)[:, None]  # [BF, 1]
    t_out_idx = t_out_start + tl.arange(0, BLOCK_T)[None, :]  # [1, BT]
    out_mask = (f_out_idx < F_out) & (t_out_idx < T_out)

    # Accumulator
    acc = tl.zeros((BLOCK_F, BLOCK_T), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    for ic in range(IC):
        # For each 3x3 tap, compute input coordinates
        for kh in range(3):
            for kw in range(3):
                # padding=1, so input index = (f_out_idx*2 - 1 + kh, t_out_idx*2 - 1 + kw)
                f_in_idx = f_out_idx * 2 - 1 + kh
                t_in_idx = t_out_idx * 2 - 1 + kw

                # Compute pointers for X[b, ic, f_in_idx, t_in_idx]
                x_ptrs = X_ptr + b * x_sN + ic * x_sC + f_in_idx * x_sF + t_in_idx * x_sT
                load_mask = out_mask
                x_vals = tl.load(x_ptrs, mask=load_mask, other=0.0)  # [BF, BT], float32

                # Load W[oc, ic, kh, kw] (scalar)
                w_ptr = W_ptr + oc * w_sOC + ic * w_sIC + kh * w_sKH + kw * w_sKW
                w_val = tl.load(w_ptr)  # float32

                acc += x_vals * w_val

    # Add bias
    b_val = tl.load(BIAS_ptr + oc)
    acc += b_val

    # GELU (approx)
    # y = 0.5 * x * (1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    # Compute in float32
    c0 = 0.7978845608028654  # sqrt(2/pi)
    c1 = 0.044715
    x3 = acc * acc * acc
    inner = acc + c1 * x3
    gelu = 0.5 * acc * (1.0 + tl.math.tanh(c0 * inner))

    # Store output
    y_ptrs = Y_ptr + b * y_sN + oc * y_sOC + f_out_idx * y_sF + t_out_idx * y_sT
    tl.store(y_ptrs, gelu, mask=out_mask)


# Triton matmul kernel: A[M, K] @ B[K, N] -> Y[M, N]
@triton.jit
def matmul_bf(
    A_ptr, B_ptr, Y_ptr,
    M, N, K,
    a_sM, a_sK,
    b_sK, b_sN,
    y_sM, y_sN,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    m_block = pid // grid_n
    n_block = pid % grid_n

    m_start = m_block * BLOCK_M
    n_start = n_block * BLOCK_N
    offs_m = m_start + tl.arange(0, BLOCK_M)
    offs_n = n_start + tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, K, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        a_ptrs = A_ptr + offs_m[:, None] * a_sM + offs_k[None, :] * a_sK
        b_ptrs = B_ptr + offs_k[:, None] * b_sK + offs_n[None, :] * b_sN
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)

    y_ptrs = Y_ptr + offs_m[:, None] * y_sM + offs_n[None, :] * y_sN
    tl.store(y_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# Triton elementwise kernel: Y[:] = X[:] * scale
@triton.jit
def scale_vec(
    X_ptr, Y_ptr, N,
    scale,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    y = x * scale
    tl.store(Y_ptr + offs, y, mask=mask)


# Triton elementwise kernel: Y[m, n] = X[m, n] + P[n]
@triton.jit
def add_pos_emb_vec(
    X_ptr, P_ptr, Y_ptr,
    M, N,
    x_sM, x_sN,
    y_sM, y_sN,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = M * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    # recover m, n
    n_idx = offs % N
    m_idx = offs // N
    x_ptrs = X_ptr + m_idx * x_sM + n_idx * x_sN
    p_ptrs = P_ptr + n_idx
    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
    p_vals = tl.load(p_ptrs, mask=mask, other=0.0)
    y_vals = x_vals + p_vals
    y_ptrs = Y_ptr + m_idx * y_sM + n_idx * y_sN
    tl.store(y_ptrs, y_vals, mask=mask)

            out_scaled.stride(0), out_scaled.stride(1),

    out_mask = (f_out_idx < F_out) & (t_out_idx < T_out)

    # Accumulator
    acc = tl.zeros((BLOCK_F, BLOCK_T), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    for ic in range(IC):
        # For each 3x3 tap, compute input coordinates (padding=1)
        for kh in range(3):
            for kw in range(3):
                f_in_idx = f_out_idx * 2 - 1 + kh  # [BF, 1]
                t_in_idx = t_out_idx * 2 - 1 + kw  # [1, BT]
                x_ptrs = X_ptr + b * x_sN + ic * x_sC + f_in_idx * x_sF + t_in_idx * x_sT
                load_mask = out_mask
                x_vals = tl.load(x_ptrs, mask=load_mask, other=0.0)  # [BF, BT], float32

                # Load W[oc, ic, kh, kw] (scalar)
                w_ptr = W_ptr + oc * w_sOC + ic * w_sIC + kh * w_sKH + kw * w_sKW
                w_val = tl.load(w_ptr)  # float32

                acc += x_vals * w_val

    # Add bias
    b_val = tl.load(BIAS_ptr + oc)
    acc += b_val

    # GELU (approx)
    # y = 0.5 * x * (1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
    c0 = 0.7978845608028654  # sqrt(2/pi)
    c1 = 0.044715
    x3 = acc * acc * acc
    inner = acc + c1 * x3
    gelu = 0.5 * acc * (1.0 + tl.math.tanh(c0 * inner))

    # Store output
    y_ptrs = Y_ptr + b * y_sN + oc * y_sOC + f_out_idx * y_sF + t_out_idx * y_sT
    tl.store(y_ptrs, gelu, mask=out_mask)


# Triton matmul kernel: A[M, K] @ B[K, N] -> Y[M, N]
@triton.jit
def matmul_bf(
    A_ptr, B_ptr, Y_ptr,
    M, N, K,
    a_sM, a_sK,
    b_sK, b_sN,
    y_sM, y_sN,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    m_block = pid // grid_n
    n_block = pid % grid_n

    m_start = m_block * BLOCK_M
    n_start = n_block * BLOCK_N
    offs_m = m_start + tl.arange(0, BLOCK_M)
    offs_n = n_start + tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, K, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        a_ptrs = A_ptr + offs_m[:, None] * a_sM + offs_k[None, :] * a_sK
        b_ptrs = B_ptr + offs_k[:, None] * b_sK + offs_n[None, :] * b_sN
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)

    y_ptrs = Y_ptr + offs_m[:, None] * y_sM + offs_n[None, :] * y_sN
    tl.store(y_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# Triton elementwise kernel: Y[:] = X[:] * scale
@triton.jit
def scale_vec(
    X_ptr, Y_ptr, N,
    scale,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    y = x * scale
    tl.store(Y_ptr + offs, y, mask=mask)


# Triton elementwise kernel: Y[m, n] = X[m, n] + P[n]
@triton.jit
def add_pos_emb_vec(
    X_ptr, P_ptr, Y_ptr,
    M, N,
    x_sM, x_sN,
    y_sM, y_sN,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = M * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    n_idx = offs % N
    m_idx = offs // N
    x_ptrs = X_ptr + m_idx * x_sM + n_idx * x_sN
    p_ptrs = P_ptr + n_idx
    x_vals = tl.load(x_ptrs, mask=mask, other=0.0)
    p_vals = tl.load(p_ptrs, mask=mask, other=0.0)
    y_vals = x_vals + p_vals
    y_ptrs = Y_ptr + m_idx * y_sM + n_idx * y_sN
    tl.store(y_ptrs, y_vals, mask=mask)
