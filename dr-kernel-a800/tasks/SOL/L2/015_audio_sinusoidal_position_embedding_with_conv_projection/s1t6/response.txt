# Triton GELU kernel (exact, erf-based), elementwise
@triton.jit
def gelu_exact_kernel(
    X_ptr,          # *const bfloat16
    Y_ptr,          # *bfloat16
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0).to(tl.float32)

    y = 0.5 * x * (1.0 + tl.math.erf(x * inv_sqrt2))
    tl.store(Y_ptr + offsets, y.to(tl.bfloat16), mask=mask)

    BT_ptr,           # *const bfloat16, BT: (K_out, N) = conv_out_weight.T
    POS_ptr,          # *const bfloat16, positional embedding: (OH, N), OH=M
    C_ptr,            # *bfloat16, output: (M, N)

    K_in: tl.constexpr,  # 15360
    N: tl.constexpr,     # 1024
    stride_Am, stride_Ak,
    stride_BTk, stride_BTn,
    stride_Cm, stride_Cn,
    scale: tl.constexpr,  # float32 scaling factor, e.g., 32.0
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)  # tile over N

            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0

        acc += tl.dot(A_tile, BT_tile)

        POS_ptr + m_offsets[:, None] * stride_Cm + n_offsets[None, :] * stride_Cn,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0

        C_ptr + m_offsets[:, None] * stride_Cm + n_offsets[None, :] * stride_Cn,
        acc.to(tl.bfloat16),
        mask=mask_m[:, None] & mask_n[None, :]
    )

Notes on correctness and performance
- Convolution: The Triton conv kernel uses stride=2, padding=1, 3x3 kernels. We loop over input channels and 3x3 positions, perform masked loads for padding, accumulate in float32, add bias, and store in bfloat16. The grid is (B*H, tiles over Cout, T_out) so each program computes one output pixel (b, oh, c_tile, t_out_idx). This ensures we cover all outputs. Although the kernel loops over Cin=384, Triton unrolls such loops when Cin is a constexpr. We pass Cin appropriately at launch for conv2/3 and Cin=1 for conv1.
- GELU: Exact GELU using erf is implemented in-kernel. We apply it to each conv output.
- GEMM: We implement y = x_flat @ conv_out_weight.T directly in Triton. We also add scaled positional embeddings per row and multiply by embed_scale in the kernel. This avoids any PyTorch matmul or embedding addition. The output y is (B, T_out3, 1024).
- Triton kernels are actually launched from ModelNew.forward; no PyTorch ops are used in forward beyond data movement and allocation. This should address the “decoy kernel” issue and ensure the evaluation environment runs the Triton kernels.

If any workloads previously failed, this version uses a proper grid and masks for conv outputs and exact GELU. The previous decoy issue is resolved by launching conv2d_3x3_stride2_padding1_kernel for each conv stage. The GELU and GEMM kernels are also launched.

usuario
A previous attempt replaced torch.conv2d with a Triton kernel that processed only a single output pixel per program. That design will launch O(B*H*T_out * tiles_over_Cout * T_out) programs — for the largest case in the benchmark (B=64, T=512, H=80, Cout=384, T_out≈256), that’s about 262 million programs, each doing 3*C*3=27 loads. This is far too many programs and will run extremely slowly.

The proper Triton approach is to compute tiles of the output image at once. For conv2d with stride=2, padding=1, 3x3, per output tile:
- Output spatial dims: OH=H, OW=floor((T-3)/2+1). We tile OH and OW with BLOCK_H, BLOCK_W, and loop over Cin and the 3x3 kernel window, accumulating into a (BLOCK_C, BLOCK_H, BLOCK_W) accumulator.
- Using tile-based kernels reduces program count dramatically. For example, with BLOCK_H=8, BLOCK_W=8, BLOCK_C=64, grid is roughly (B*80/8, 256/8, 384/64) = (80, 32, 6) per conv stage, i.e., about 160*32*6=30,720 programs instead of 262 million.
- This is still substantial, but acceptable for the given problem size, and should avoid the prior “too many programs” issue. It also avoids PyTorch ops entirely, satisfying the Triton-only requirement.

I will implement:
- A Triton conv2d kernel that computes a tile over (Cout, OH, OW), loops over Cin and 3x3, with masked loads for padding, accumulates in float32, adds bias, and stores bfloat16.
- An exact GELU kernel (erf-based) applied in-kernel after each conv.
- A Triton GEMM kernel for y = x_flat @ conv_out_weight.T with elementwise addition of scaled positional embeddings. We pass POS as (M, N), where M=B*T_out3 and N=1024, and add it in the kernel.

This design launches far fewer programs, should be correct, and avoids the previous decoy or excessive program count issues.

import math
import torch
import triton
import triton.language as tl


# Triton conv2d kernel: 3x3, stride=2, padding=1, output (B, Cout, H, T_out)
# Tiling over output channels and output spatial tile (OH, OW).
@triton.jit
def conv2d_3x3_stride2_padding1_tiled(
    X_ptr,         # *const bfloat16, input (B, Cin, H, T)
    W_ptr,         # *const bfloat16, weight (Cout, Cin, 3, 3)
    BIAS_ptr,      # *const bfloat16, bias (Cout,)
    Y_ptr,         # *bfloat16, output (B, Cout, H, T_out)
    Bsz, Cin, H, T, Cout, T_out,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_W: tl.constexpr,
):
    # Grid: (B, tiles over Cout, tiles over OH, tiles over OW)
    pid_b = tl.program_id(0)
    pid_ct = tl.program_id(1)
    pid_th = tl.program_id(2)
    pid_tw = tl.program_id(3)

    oh_start = pid_th * BLOCK_H
    ow_start = pid_tw * BLOCK_W

    c_start = pid_ct * BLOCK_C

    # Output indices for the tile
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    oh_offsets = oh_start + tl.arange(0, BLOCK_H)
    ow_offsets = ow_start + tl.arange(0, BLOCK_W)

    mask_c = c_offsets < Cout
    mask_oh = oh_offsets < H
    mask_ow = ow_offsets < T_out

    # Accumulator (BLOCK_C, BLOCK_H, BLOCK_W)
    acc = tl.zeros((BLOCK_C, BLOCK_H, BLOCK_W), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel window
    for cin in range(Cin):
        for kh in range(3):
            ih = oh_offsets[:, None] + kh - 1  # shape (BLOCK_H, 1)
            valid_ih = (ih >= 0) & (ih < H)
            for kt in range(3):
                it = ow_offsets[None, :] + kt - 1  # shape (1, BLOCK_W)
                valid_it = (it >= 0) & (it < T)

                # Compute X offsets: ((b*Cin + cin)*H + ih)*T + it
                # Broadcast to (BLOCK_H, BLOCK_W)
                x_off = ((pid_b * Cin + cin) * H + ih) * T + it

                # Mask for X load
                load_mask = (mask_oh[:, None] & mask_ow[None, :] &
                             valid_ih[:, None] & valid_it[None, :])

                x_val = tl.load(
                    X_ptr + x_off,
                    mask=load_mask,
                    other=0.0
                ).to(tl.float32)  # shape (BLOCK_H, BLOCK_W)

                # Load W for this (cin, kh, kt) across output channels tile
                # W layout: (Cout, Cin, 3, 3). For fixed (cin, kh, kt), W vector is length Cout.
                # Index is (c_offsets * Cin + cin) * 9 + (kh*3 + kt)
                w_off = (c_offsets * Cin + cin) * 9 + (kh * 3 + kt)
                w_vec = tl.load(W_ptr + w_off, mask=mask_c, other=0.0).to(tl.float32)  # (BLOCK_C,)
                # Broadcast w_vec to (BLOCK_C, BLOCK_H, BLOCK_W) and accumulate
                acc += w_vec[:, None, None] * x_val[None, :, :]

    # Add bias
    for j in range(BLOCK_C):
        cout_j = c_start + j
        if cout_j < Cout:
            bval = tl.load(BIAS_ptr + cout_j).to(tl.float32)
            acc[j, :, :] += bval

    # Store Y[b, c_offsets, oh_offsets, ow_offsets]
    # Y linear offset: b*(Cout*H*T_out) + c*H*T_out + oh*T_out + ow
    for j in range(BLOCK_C):
        cout_j = c_start + j
        if cout_j < Cout:
            y_tile_ptr = Y_ptr + pid_b * (Cout * H * T_out) + cout_j * (H * T_out)
            # Store acc[j, :, :] at positions (oh, ow)
            # Broadcast y_tile_ptr across (BLOCK_H, BLOCK_W)
            tl.store(
                y_tile_ptr + oh_offsets[:, None] * T_out + ow_offsets[None, :],
                acc[j, :, :].to(tl.bfloat16),
                mask=mask_oh[:, None] & mask_ow[None, :]
            )


# Triton GELU kernel (exact, erf-based), elementwise
@triton.jit
def gelu_exact_kernel(
    X_ptr,          # *const bfloat16
    Y_ptr,          # *bfloat16
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
    y = 0.5 * x * (1.0 + tl.math.erf(x * inv_sqrt2))
    tl.store(Y_ptr + offsets, y.to(tl.bfloat16), mask=mask)


# Triton GEMM: A is (M, K_in) = x_flat (B*T_out3, 15360), BT is (K_out, N) = conv_out_weight.T (15360, 1024),
# C is (M, N) = y (B*T_out3, 1024). We also add scaled positional embedding per row: POS (M, N).
@triton.jit
def gemm_mul_add_pos_kernel(
    A_ptr,            # *const bfloat16, A: (M, K_in)
    BT_ptr,           # *const bfloat16, BT: (K_out, N) = conv_out_weight.T
    POS_ptr,          # *const bfloat16, positional embedding: (M, N)
    C_ptr,            # *bfloat16, output: (M, N)
    M: tl.constexpr,  # total rows in A (B * T_out3)
    K_in: tl.constexpr,  # 15360
    N: tl.constexpr,     # 1024
    stride_Am, stride_Ak,
    stride_BTk, stride_BTn,
    stride_Cm, stride_Cn,
    scale: tl.constexpr,  # float32 scaling factor, e.g., 32.0
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile over M
    pid_n = tl.program_id(1)  # tile over N

    m_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = m_offsets < M
    mask_n = n_offsets < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks
    for k0 in range(0, K_in, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < K_in

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        A_tile = tl.load(
            A_ptr + m_offsets[:, None] * stride_Am + k_offsets[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # Load BT tile: shape (BLOCK_K, BLOCK_N)
        BT_tile = tl.load(
            BT_ptr + k_offsets[:, None] * stride_BTk + n_offsets[None, :] * stride_BTn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(A_tile, BT_tile)

    # Scale
    acc = acc * scale

    # Add scaled positional embedding: POS is (M, N)
    pos_tile = tl.load(
        POS_ptr + m_offsets[:, None] * stride_Cm + n_offsets[None, :] * stride_Cn,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0
    ).to(tl.float32)
    acc = acc + pos_tile  # broadcasting over BLOCK_M, BLOCK_N

    # Store
    tl.store(
        C_ptr + m_offsets[:, None] * stride_Cm + n_offsets[None, :] * stride_Cn,
        acc.to(tl.bfloat16),
        mask=mask_m[:, None] & mask_n[None, :]
    )
