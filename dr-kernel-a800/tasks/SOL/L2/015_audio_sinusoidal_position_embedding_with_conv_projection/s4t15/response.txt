import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# Triton kernel: GELU (tanh approximation), elementwise over N elements.
# X: input, Y: output, N: total elements
@triton.jit
def gelu_tanh_kernel(X, Y, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X + offsets, mask=mask, other=0.0).to(tl.float32)
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    inner = c * (x + 0.044715 * x3)
    gelu = 0.5 * x * (1.0 + tl.math.tanh(inner))
    tl.store(Y + offsets, gelu, mask=mask)


# Triton kernel: Linear projection (no bias) Y = X @ W^T
# X: (B, T, K), W: (M, K), Y: (B, T, M)
# We tile over M (output channels) and loop over K.
@triton.jit
def linear_no_bias_kernel(
    X, W, Y,
    B, T, K, M,
    stride_xb, stride_xt, stride_xk,
    stride_wm, stride_wk,
    stride_yb, stride_yt, stride_ym,
    BLOCK_M: tl.constexpr,   # tile size over M
    BLOCK_K: tl.constexpr    # tile size over reduction dimension
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_m = tl.program_id(2)

    b = pid_b
    t = pid_t
    m_start = pid_m * BLOCK_M
    m_offsets = m_start + tl.arange(0, BLOCK_M)
    m_mask = m_offsets < M

    # Accumulator for this (b, t) over a block of M outputs
    acc = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # Reduce over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        # Load X[b, t, k_offsets]
        x_ptrs = X + b * stride_xb + t * stride_xt + k_offsets * stride_xk
        x_vec = tl.load(x_ptrs, mask=k_mask, other=0.0).to(tl.float32)  # shape: (BLOCK_K,)

        # Load W[:, k_offsets] for M tile: shape (BLOCK_M, BLOCK_K)
        w_ptrs = W + m_offsets[:, None] * stride_wm + k_offsets[None, :] * stride_wk
        w_mat = tl.load(w_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Accumulate: acc_m += sum_k (x_vec[k] * w_mat[m, k])
        acc += tl.sum(w_mat * x_vec[None, :], axis=1)

    # Store results
    y_ptrs = Y + b * stride_yb + t * stride_yt + m_offsets * stride_ym
    tl.store(y_ptrs, acc, mask=m_mask)


# Triton kernel: Add scaled positional embedding to Y.
# Y: (B, T, M), POS: (T, M) (pre-sliced), SCALE: scalar
@triton.jit
def add_scaled_pos_emb_kernel(
    Y, POS, SCALE,
    B, T, M,
    stride_yb, stride_yt, stride_ym,
    stride_pos_t, stride_pos_m
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_m = tl.program_id(2)

    b = pid_b
    t = pid_t
    m_start = pid_m * 64
    m_offsets = m_start + tl.arange(0, 64)
    m_mask = m_offsets < M

    # Load current Y slice
    y_ptrs = Y + b * stride_yb + t * stride_yt + m_offsets * stride_ym
    y_vals = tl.load(y_ptrs, mask=m_mask, other=0.0).to(tl.float32)

    # Load pos embedding slice for this t over m_offsets
    pos_ptrs = POS + t * stride_pos_t + m_offsets * stride_pos_m
    pos_vals = tl.load(pos_ptrs, mask=m_mask, other=0.0).to(tl.float32)

    # Add scaled
    y_vals = y_vals + SCALE * pos_vals

    # Store back
    tl.store(y_ptrs, y_vals, mask=m_mask)


def triton_gelu(x: torch.Tensor) -> torch.Tensor:
    # x: (B, C, F, T) — flatten to 1D
    # Allocate output
    y = torch.empty_like(x)
    N = x.numel()
    # Choose block size
    BLOCK = 1024
    grid = (triton.cdiv(N, BLOCK),)
    # Launch
    gelu_tanh_kernel[grid](x, y, N, BLOCK=BLOCK)
    return y


def triton_linear_no_bias(x: torch.Tensor, w: torch.Tensor) -> torch.Tensor:
    # x: (B, T, K), w: (M, K)
    B, T, K = x.shape
    M = w.shape[0]
    # Ensure contiguous
    x_c = x.contiguous()
    w_c = w.contiguous()
    y = torch.empty((B, T, M), dtype=torch.float32, device=x.device)  # compute in fp32, cast later
    # Strides
    stride_xb, stride_xt, stride_xk = x_c.stride()
    stride_wm, stride_wk = w_c.stride()
    stride_yb, stride_yt, stride_ym = y.stride()
    # Tile sizes
    BLOCK_M = 64
    BLOCK_K = 64
    grid = (B, triton.cdiv(T, BLOCK_M), triton.cdiv(M, BLOCK_M))
    linear_no_bias_kernel[grid](
        x_c, w_c, y,
        B, T, K, M,
        stride_xb, stride_xt, stride_xk,
        stride_wm, stride_wk,
        stride_yb, stride_yt, stride_ym,
        BLOCK_M=BLOCK_M, BLOCK_K=BLOCK_K,
    )
    # Cast to original dtype (bfloat16) if needed
    if y.dtype != x.dtype:
        y = y.to(x.dtype)
    return y


def triton_add_scaled_pos_emb(y: torch.Tensor, pos_slice: torch.Tensor, scale: float):
    # y: (B, T, M), pos_slice: (T, M) sliced from positional_embedding
    B, T, M = y.shape
    y_c = y.contiguous()
    pos_c = pos_slice.contiguous()

    stride_pos_t, stride_pos_m = pos_c.stride()
    grid = (B, T, triton.cdiv(M, 64))
    add_scaled_pos_emb_kernel[grid](
        y_c, pos_c, scale,
        B, T, M,
        stride_yb, stride_yt, stride_ym,
        stride_pos_t, stride_pos_m,
        num_warps=4, num_stages=2
    )
    return y_c

# Triton kernel: Conv2D 3x3, stride=2, padding=1
# X: (B, C_in, H, W), W: (C_out, C_in, 3, 3), BIAS: (C_out), Y: (B, C_out, H_out, W_out)
# H_out = (H - 3)//2 + 1, W_out = (W - 3)//2 + 1
@triton.jit
def conv3x3_stride2_pad1_kernel(
    X, W, BIAS, Y,
    B, C_in, H, W as in_h, W as in_w,
    C_out, H_out, W_out,
    stride_xb, stride_xc, stride_xh, stride_xw,
    stride_wo, stride_wi, stride_wkh, stride_wkw,
    stride_yb, stride_yc, stride_yh, stride_yw,
    BLOCK_CO: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_w = tl.program_id(2)

    h_out = pid_h
    w_out = pid_w

    # Output channel tile
    co_start = tl.program_id(3) * BLOCK_CO
    co_offsets = co_start + tl.arange(0, BLOCK_CO)
    co_mask = co_offsets < C_out

    # Accumulator for this tile
    acc = tl.zeros((BLOCK_CO,), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel taps
    for ci in range(0, C_in):
        for kh in range(0, 3):
            in_h = h_out * 2 + kh - 1  # padding=1
            for kw in range(0, 3):
                in_w = w_out * 2 + kw - 1

                # Valid check for in-bounds input spatial location
                valid_hw = (in_h >= 0) & (in_h < H) & (in_w >= 0) & (in_w < W)
                # Build pointers for X[b, ci, in_h, in_w]
                x_ptrs = X + pid_b * stride_xb + ci * stride_xc + in_h * stride_xh + in_w * stride_xw
                x_val = tl.load(x_ptrs, mask=valid_hw, other=0.0).to(tl.float32)

                # Load weight vector W[co, ci, kh, kw] for all co in tile
                w_ptrs = W + co_offsets * stride_wo + ci * stride_wi + kh * stride_wkh + kw * stride_wkw
                w_vec = tl.load(w_ptrs, mask=co_mask, other=0.0).to(tl.float32)

                acc += x_val * w_vec

    # Add bias
    bias_ptrs = BIAS + co_offsets
    bias = tl.load(bias_ptrs, mask=co_mask, other=0.0).to(tl.float32)
    acc += bias

    # Store to Y
    y_ptrs = Y + pid_b * stride_yb + co_offsets * stride_yc + h_out * stride_yh + w_out * stride_yw
    tl.store(y_ptrs, acc, mask=co_mask)


# Triton kernel: GELU (tanh approximation), elementwise
@triton.jit
def gelu_tanh_kernel(X, Y, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(X + offsets, mask=mask, other=0.0).to(tl.float32)
    c = 0.7978845608028654  # sqrt(2/pi)
    x3 = x * x * x
    inner = c * (x + 0.044715 * x3)
    gelu = 0.5 * x * (1.0 + tl.math.tanh(inner))
    tl.store(Y + offsets, gelu, mask=mask)
