```python
@triton.jit
def conv2d_stride2_bias_gelu_kernel(
    x_ptr,           # input [N, Ci, Fi, Ti] in fp32
    weight_ptr,      # weights [Co, Ci, 3, 3] in fp32
    bias_ptr,        # bias [Co] in fp32
    output_ptr,      # output [N, Co, F_out, T_out] in fp32
    N, Ci, Co, Fi, Ti, F_out, T_out,
    x_sN, x_sCi, x_sF, x_sT,        # strides for x
    w_sCo, w_sCi, w_sKh, w_sKw,     # strides for weight
    out_sN, out_sCo, out_sF, out_sT,  # strides for output
    # GELU constants
    gelu_sqrt2_over_pi: tl.constexpr, gelu_coef: tl.constexpr,
):
    # Grid: (N, Co, F_out, T_out)
    n = tl.program_id(0)
    co = tl.program_id(1)
    f_out = tl.program_id(2)
    t_out = tl.program_id(3)

    # Initialize accumulator for output
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over input channels and 3x3 kernel
    for ci in range(0, Ci):
        for kh in range(0, 3):
            for kw in range(0, 3):
                fi = f_out * 2 - kh + 1
                ti = t_out * 2 - kw + 1

                # Load with mask

                w_val = tl.load(weight_ptr + w_idx)  # scalar

    # Add bias

    b_val = tl.load(bias_ptr + b_idx)
    acc += b_val

    inner = gelu_sqrt2_over_pi * (x_g + gelu_coef * x3)
    gelu_val = 0.5 * x_g * (1.0 + tl.math.tanh(inner))

    # Store output
    out_idx = n * out_sN + co * out_sCo + f_out * out_sF + t_out * out_sT
    tl.store(output_ptr + out_idx, gelu_val)

    w_ptr,            # weights [M, K] in fp32 (we pass transposed conv_out_weight: [3840, 1024])
    y_ptr,            # output [N, T_out3, K] in fp32
    N, T_out3, M, K,
    x_sN, x_sT, x_sM,   # strides for x: x[n, t, j]
    w_sM, w_sK,          # strides for w: w[j, k]
    y_sN, y_sT, y_sK,    # strides for y: y[n, t, k]
    BLOCK_K: tl.constexpr,
):

    t = tl.program_id(1)

        T2_out = (T1_out - 3) // 2 + 1

        x2 = torch.empty((N, 384, F2_out, T2_out), device=device, dtype=torch.float32)

        # Launch conv2 kernel
        grid2 = (N, 384, F2_out, T2_out)
        conv2d_stride2_bias_gelu_kernel[grid2](
            x1, w2, b2, x2,
            N, 384, 384, F1_out, T1_out, F2_out, T2_out,
            x1.stride(0), x1.stride(1), x1.stride(2), x1.stride(3),
            w2.stride(0), w2.stride(1), w2.stride(2), w2.stride(3),
            x2.stride(0), x2.stride(1), x2.stride(2), x2.stride(3),
            0.7978845608028654, 0.044715
        )

        # conv3: in (384), out (384), F_out3 = (F2_out - 3)//2 + 1 = 8, T_out3 = (T2_out - 3)//2 + 1
        F3_out = (F2_out - 3) // 2 + 1  # 8
        T3_out = (T2_out - 3) // 2 + 1

        x3 = torch.empty((N, 384, F3_out, T3_out), device=device, dtype=torch.float32)

        # Launch conv3 kernel
        grid3 = (N, 384, F3_out, T3_out)
        conv2d_stride2_bias_gelu_kernel[grid3](
            x2, w3, b3, x3,
            N, 384, 384, F2_out, T2_out, F3_out, T3_out,
            x2.stride(0), x2.stride(1), x2.stride(2), x2.stride(3),
            w3.stride(0), w3.stride(1), w3.stride(2), w3.stride(3),
            x3.stride(0), x3.stride(1), x3.stride(2), x3.stride(3),
            0.7978845608028654, 0.044715
        )

        # Reshape to [N, T3_out, C3*F3_out] where C3=384, F3_out=8, so M=C3*F3_out=384*8=3072
        x3_reshaped = x3.permute(0, 3, 1, 2).contiguous().view(N, T3_out, 384 * F3_out)

        # Prepare conv_out_weight as [M, K] where K=1024, M=3840 (given in get_inputs). In the provided get_inputs, conv_out_weight is [1024, 3840], so we use that.
        # Transpose to [M=3840, K=1024] for GEMV kernel
        M_linear = 3840  # per get_inputs
        K_linear = 1024
        W_t = cout_w.transpose(0, 1).contiguous()  # [3840, 1024]

        # Allocate output of GEMV
        y = torch.empty((N, T3_out, K_linear), device=device, dtype=torch.float32)

        # Launch GEMV kernel: grid = (N, T3_out, ceil_div(K, BLOCK_K))
        BLOCK_K = 128
        grid_linear = (N, T3_out, (K_linear + BLOCK_K - 1) // BLOCK_K)
        linear_gemv_kernel[grid_linear](
            x3_reshaped, W_t, y,
            N, T3_out, M_linear, K_linear,
            x3_reshaped.stride(0), x3_reshaped.stride(1), x3_reshaped.stride(2),
            W_t.stride(0), W_t.stride(1),
            y.stride(0), y.stride(1), y.stride(2),
            BLOCK_K=BLOCK_K
        )

        # Scale by embed_scale and add positional embedding
        # Note: positional_embedding is [T_max, 1024]. We only need the first T3_out rows.
        # Launch scale+pos kernel
        grid_scale = (N, T3_out, K_linear)
        scale_add_pos_kernel[grid_scale](
            y, pos_emb[:T3_out, :], float(embed_scale),
            N, T3_out, K_linear,
            y.stride(0), y.stride(1), y.stride(2),
            pos_emb.stride(0), pos_emb.stride(1)
        )

        return y
```
