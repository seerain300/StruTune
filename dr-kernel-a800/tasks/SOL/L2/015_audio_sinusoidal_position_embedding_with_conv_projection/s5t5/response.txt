# Conv2d 3x3 stride=2, padding=1, for input channels Ci=1, output [N, Co, F_out, T_out]
# Inputs:
#   X: [N, 1, F, T] bf16
#   W: [Co, 1, 3, 3] bf16
#   BIAS: [Co] bf16
# Output:
#   B: [N, Co, F_out, T_out] bf16
@triton.jit
def conv2d_ci1_stride2_bias_kernel(
    X_ptr, W_ptr, BIAS_ptr, B_ptr,
    N, F, T,
    Co, F_out, T_out,
    # strides
    x_strideN, x_strideC, x_strideF, x_strideT,
    w_strideCo, w_strideCi, w_strideH, w_strideW,
    b_stride0, b_stride1, b_stride2, b_stride3,
    # meta
):
    pid_n = tl.program_id(0)
    pid_co = tl.program_id(1)
    pid_t_out = tl.program_id(2)

    # For output time t_out, input time index is 2*t_out + 1 due to padding=1
    t_in = pid_t_out * 2 + 1

    # Initialize accumulator for GELU
    acc = tl.zeros((), dtype=tl.float32)

    # Reduce over 3x3 kernel and input channel Ci=1
    # h_off = -1,0,1 ; w_off = -1,0,1
    for h_off in range(3):
        h = h_off - 1  # because padding=1
        h_idx = h + t_in
        for w_off in range(3):
            w = w_off - 1
            w_idx = w + (t_in % F)  # T dimension doesn't mix F here; we only care about time
            # Note: here we treat F as time-dimension stride for Ci=1 conv; for Ci>1, we need to map to F_out.
            # Since Ci=1, F_in=F and F_out=(F-1)//2+1, and we compute h_idx = h + t_in; we must ensure h_idx in [0, F-1].
            # If h_idx < 0 or >= F-1, we skip (padding). In PyTorch, padding sets out-of-bounds to 0.
            # We'll implement that by masked load.
            if (h_idx < 0) or (h_idx >= F):
                x_val = 0.0
            else:
                x_ptr = X_ptr + pid_n * x_strideN + 0 * x_strideC + h_idx * x_strideF + t_in * x_strideT
                x_val = tl.load(x_ptr).to(tl.float32)

            # Load weight scalar for co, ci=0, h_off, w_off
            w_ptr = W_ptr + pid_co * w_strideCo + 0 * w_strideCi + h_off * w_strideH + w_off * w_strideW
            w_val = tl.load(w_ptr).to(tl.float32)

            acc += x_val * w_val

    # Add bias
    b_ptr = BIAS_ptr + pid_co * b_stride0
    b_val = tl.load(b_ptr).to(tl.float32)
    acc += b_val

    # GELU activation: y = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
    c0 = 0.7978845608028654  # sqrt(2/pi)
    x3 = acc * acc * acc
    gelu = 0.5 * acc * (1.0 + tl.tanh(c0 * (acc + 0.044715 * x3)))

    # Store result to B[n, co, f_out=0..39, t_out]
    # We launch with grid (N, Co, T_out); for each t_out, f_out is implied by pid_t_out
    b_store_ptr = B_ptr + pid_n * 0 + pid_co * 0 + pid_t_out * 0 + pid_n * (Co * F_out) + pid_co * F_out + pid_t_out
    tl.store(b_store_ptr, gelu.to(tl.bfloat16))

@triton.jit
def linear_bmm_kernel(

    w_strideK, w_strideM,
    y_strideN, y_strideT, y_strideK,
):

    pid_t = tl.program_id(1)
    pid_k = tl.program_id(2)

        offs_m = m + tl.arange(0, 128)

    y_ptr = Y_ptr + pid_n * y_strideN + pid_t * y_strideT + pid_k * y_strideK

        scale_embed_kernel[grid_scale](Y_scaled, scale, N_dummy, T_dummy, K, y_strideN, y_strideT, y_strideK)

        # Step 3: add positional embedding using Triton
        # positional_embedding is [max_source_positions=1500, d_model=1024], dtype bfloat16
        # We will use first T_dummy rows
        P = positional_embedding[:T_dummy, :].to(torch.bfloat16)
        # Launch Triton add_pos_emb_kernel
        # Y_scaled strides for [N, T, K]
        y_strideN2 = T_dummy * K
        y_strideT2 = K
        y_strideK2 = 1
        # P strides
        p_stride0 = 1024
        p_stride1 = 1
        grid_pos = (N_dummy * T_dummy * K,)
        add_pos_emb_kernel[grid_pos](Y_scaled, P, N_dummy, T_dummy, K, y_strideN2, y_strideT2, y_strideK2, p_stride0, p_stride1)

        return Y_scaled
