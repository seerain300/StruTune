# Triton conv2d kernel: 3x3, stride=2, padding=1
# Input: X(B, Cin, H, T) bfloat16
# Weight: W(Cout, Cin, 3, 3) bfloat16
# Bias: Bias(Cout) bfloat16
# Output: Y(B, Cout, H, T_out) bfloat16, with T_out = floor((T - 3)/2 + 1)
@triton.jit
def conv2d_3x3_stride2_padding1_kernel(
    X_ptr,         # *const bfloat16
    W_ptr,         # *const bfloat16
    BIAS_ptr,      # *const bfloat16
    Y_ptr,         # *bfloat16
    B: tl.int32, Cin: tl.int32, H: tl.int32, T: tl.int32, Cout: tl.int32, T_out: tl.int32,
    BLOCK_C: tl.constexpr,
):
    # Grid: (B*H, tiles over Cout, T_out)
    pid_m = tl.program_id(0)   # over B*H
    pid_c = tl.program_id(1)   # tiles over Cout
    pid_t = tl.program_id(2)   # output time index

    # Decode b and oh from pid_m
    b = pid_m // H
    oh = pid_m % H

    # Output time index
    t_out_idx = pid_t

    # Tile of output channels
    c_start = pid_c * BLOCK_C
    c_offsets = c_start + tl.arange(0, BLOCK_C)
    mask_c = c_offsets < Cout

    # Accumulator for this tile
    acc = tl.zeros([BLOCK_C], dtype=tl.float32)

    # Iterate over kernel 3x3 and Cin
    for kh in range(3):
        for kt in range(3):
            ih = oh + kh - 1  # padding=1
            it = t_out_idx + kt - 1  # padding=1, stride=2 so output length is floor((T-3)/2)+1
            # Only compute if ih and it are in bounds
            in_bounds = (ih >= 0) & (ih < H) & (it >= 0) & (it < T_out)
            # For each input channel
            for cin in range(Cin):
                # Load input X[b, cin, ih, it]
                x_ptr = X_ptr + b * (Cin * H * T) + cin * (H * T) + ih * T + it
                x_val = tl.load(x_ptr, mask=in_bounds, other=0.0).to(tl.float32)

                # Load weights W[c, cin, kh, kt] for all c in the tile
                # W is laid out as (Cout, Cin, 3, 3)
                w_ptr = W_ptr + c_offsets * (Cin * 3 * 3) + cin * (3 * 3) + kh * 3 + kt
                w_val = tl.load(w_ptr, mask=mask_c, other=0.0).to(tl.float32)

                acc += x_val * w_val

    # Add bias
    bias = tl.load(BIAS_ptr + c_offsets, mask=mask_c, other=0.0).to(tl.float32)
    acc += bias

    # GELU exact (erf-based)
    inv_sqrt2 = 0.7071067811865476  # 1/sqrt(2)
    gelu = 0.5 * acc * (1.0 + tl.math.erf(acc * inv_sqrt2))

    # Store to Y[b, c_offsets, oh, t_out_idx] as bfloat16
    y_ptr = Y_ptr + b * (Cout * H * T_out) + (c_offsets * (H * T_out)) + (oh * T_out) + t_out_idx
    tl.store(y_ptr, gelu.to(tl.bfloat16), mask=mask_c)


# Triton GELU (exact, erf-based) applied to a flattened tensor Y

# Output: overwrites Y_ptr with GELU results

    tl.store(y_ptr, y_val.to(tl.bfloat16), mask=mask_n)

    )

    )
