# conv2d_stride2_bias_gelu_kernel:
#   x_in: [N, Ci, Fi, Ti] float32 input
#   weight: [Co, Ci, 3, 3] float32
#   bias: [Co] float32
#   output: [N, Co, F_out, T_out] float32 (post GELU)
@triton.jit
def conv2d_stride2_bias_gelu_kernel(
    x_ptr,           # *float32, input [N, Ci, Fi, Ti]
    weight_ptr,      # *float32, weights [Co, Ci, 3, 3]
    bias_ptr,        # *float32, bias [Co]
    output_ptr,      # *float32, output [N, Co, F_out, T_out]
    N, Ci, Co, Fi, Ti, F_out, T_out,
    x_sN, x_sCi, x_sF, x_sT,
    w_sCo, w_sCi, w_sK_h, w_sK_w,
    o_sN, o_sCo, o_sF, o_sT,
    BLOCK_CO: tl.constexpr,
):
    # Grid: (N, F_out, T_out)
    n = tl.program_id(0)
    f = tl.program_id(1)
    t = tl.program_id(2)

    # Loop over output channels in blocks
    for co_start in range(0, Co, BLOCK_CO):
        co_offsets = co_start + tl.arange(0, BLOCK_CO)
        mask_co = co_offsets < Co

        # Accumulator for this (n, f, t), vector over BLOCK_CO
        acc = tl.zeros((BLOCK_CO,), dtype=tl.float32)

        # Convolution: sum over input channels and 3x3 kernel
        for ci in range(0, Ci):
            for kh in range(0, 3):
                for kw in range(0, 3):
                    # input index with padding=1 and stride=2
                    ih = f * 2 - 1 + kh  # since f is output, ih maps to input via stride=2 and padding=1
                    it = t * 2 - 1 + kw
                    # valid mask
                    valid = (ih >= 0) & (ih < Fi) & (it >= 0) & (it < Ti)
                    # Compute input offsets
                    x_off = n * x_sN + ci * x_sCi + ih * x_sF + it * x_sT
                    # Load input (masked), default 0.0
                    x_val = tl.load(x_ptr + x_off, mask=valid, other=0.0)

                    # Load weights for this (co block, ci, kh, kw)
                    w_off = co_offsets * w_sCo + ci * w_sCi + kh * w_sK_h + kw * w_sK_w
                    w_val = tl.load(weight_ptr + w_off, mask=mask_co, other=0.0)

                    # Outer product accumulate
                    acc += w_val * x_val

        b_off = co_offsets * 0 + bias_ptr  # bias_ptr is *float32
        b_val = tl.load(bias_ptr + co_offsets, mask=mask_co, other=0.0)
        acc += b_val

        # Fused GELU approximation: 0.5 * x * (1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
        # Compute in fp32
        c0 = 0.5
        c1 = 0.7978845608028654  # sqrt(2/pi)
        c2 = 0.044715
        x3 = acc * acc * acc
        inner = c1 * (acc + c2 * x3)
        gelu = c0 * acc * (1.0 + tl.tanh(inner))

        # Store to output
        o_off = n * o_sN + co_offsets * o_sCo + f * o_sF + t * o_sT
        tl.store(output_ptr + o_off, gelu, mask=mask_co)


# linear_gemv_kernel:
#   x_ptr: [N, T, M] float32, x flattened as [N*T, M]
#   w_ptr: [M, K] float32, weights [M, K] where M = channels * post-conv3 spatial
#   y_ptr: [N, T, K] float32
@triton.jit
def linear_gemv_kernel(
    x_ptr,   # *float32, input [N, T, M]
    w_ptr,   # *float32, weights [M, K]
    y_ptr,   # *float32, output [N, T, K]
    N, T, M, K,
    x_sN, x_sT, x_sM,
    w_sM, w_sK,
    y_sN, y_sT, y_sK,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_k = tl.program_id(2)

    # Reduce over M in tiles
    # We will compute one output element per program across (n, t, k)
    n = pid_n
    t = pid_t
    k_start = pid_k * BLOCK_K
    k_offsets = k_start + tl.arange(0, BLOCK_K)
    mask_k = k_offsets < K

    acc = tl.zeros((BLOCK_K,), dtype=tl.float32)

    for m_start in range(0, M, BLOCK_M):
        m_offsets = m_start + tl.arange(0, BLOCK_M)
        mask_m = m_offsets < M

        # Load x[n, t, m_offsets] -> shape (BLOCK_M,)
        x_off = n * x_sN + t * x_sT + m_offsets * x_sM
        x_val = tl.load(x_ptr + x_off, mask=mask_m, other=0.0)  # (BLOCK_M,)

        # Load w[m_offsets, k_offsets] -> shape (BLOCK_M, BLOCK_K)
        w_off = m_offsets[:, None] * w_sM + k_offsets[None, :] * w_sK
        w_val = tl.load(w_ptr + w_off, mask=(mask_m[:, None] & mask_k[None, :]), other=0.0)  # (BLOCK_M, BLOCK_K)

        # Accumulate: sum over m of x_val[m] * w_val[m, k]
        # Equivalent to matmul-like reduction along m: acc += sum(w_val * x_val[:, None], axis=0)
        # Triton doesn't have direct matmul, so we sum across m dimension
        # Convert to explicit sum for clarity
        for mm in range(0, BLOCK_M):
            m_valid = mask_m[mm]
            if m_valid:
                acc += w_val[mm, :] * x_val[mm]

    # Store y[n, t, k_offsets]
    y_off = n * y_sN + t * y_sT + k_offsets * y_sK
    tl.store(y_ptr + y_off, acc, mask=mask_k)


# scale_add_pos_kernel:
#   y_ptr: [N, T, K] float32
#   pos_ptr: [T, K] float32 positional embedding
#   scale: float
#   output stored in y_ptr (in-place add)
@triton.jit
def scale_add_pos_kernel(
    y_ptr,       # *float32, [N, T, K]
    pos_ptr,     # *float32, [T, K]
    scale,       # float32
    N, T, K,
    y_sN, y_sT, y_sK,
    p_sT, p_sK,
    BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_k = tl.program_id(2)

    n = pid_n
    t = pid_t
    k_start = pid_k * BLOCK_K
    k_offsets = k_start + tl.arange(0, BLOCK_K)
    mask_k = k_offsets < K

    # Load y[n, t, k]
    y_off = n * y_sN + t * y_sT + k_offsets * y_sK
    y_val = tl.load(y_ptr + y_off, mask=mask_k, other=0.0)

    # Load pos[t, k]
    p_off = t * p_sT + k_offsets * p_sK
    pos_val = tl.load(pos_ptr + p_off, mask=mask_k, other=0.0)

    # Scale and add
    y_val = y_val * scale + pos_val

    # Store back
    tl.store(y_ptr + y_off, y_val, mask=mask_k)

            BLOCK_M=32, BLOCK_K=32,

            N, T3_out, K,
            y.stride(0), y.stride(1), y.stride(2),
            pos_slice.stride(0), pos_slice.stride(1),
            BLOCK_K=64,
        )

        return y

        x_val = tl.load(x_ptr + x_off, mask=mask_m, other=0.0)  # (BLOCK_M,)

        # Load w[m_offsets, k_offsets] -> shape (BLOCK_M, BLOCK_K)
        w_off = m_offsets[:, None] * w_sM + k_offsets[None, :] * w_sK
        w_val = tl.load(w_ptr + w_off, mask=(mask_m[:, None] & mask_k[None, :]), other=0.0)  # (BLOCK_M, BLOCK_K)

        # Accumulate: sum over m of x_val[m] * w_val[m, k]
        for mm in range(0, BLOCK_M):
            if mask_m[mm]:
                acc += w_val[mm, :] * x_val[mm]

    # Store y[n, t, k_offsets]
    y_off = n * y_sN + t * y_sT + k_offsets * y_sK
    tl.store(y_ptr + y_off, acc, mask=mask_k)


# scale_add_pos_kernel:
#   y_ptr: [N, T, K] float32
#   pos_ptr: [T, K] float32 positional embedding
#   scale: float
#   output stored in y_ptr (in-place add)
@triton.jit
def scale_add_pos_kernel(
    y_ptr,       # *float32, [N, T, K]
    pos_ptr,     # *float32, [T, K]
    scale,       # float32
    N, T, K,
    y_sN, y_sT, y_sK,
    p_sT, p_sK,
    BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)
    pid_t = tl.program_id(1)
    pid_k = tl.program_id(2)

    n = pid_n
    t = pid_t
    k_start = pid_k * BLOCK_K
    k_offsets = k_start + tl.arange(0, BLOCK_K)
    mask_k = k_offsets < K

    # Load y[n, t, k]
    y_off = n * y_sN + t * y_sT + k_offsets * y_sK
    y_val = tl.load(y_ptr + y_off, mask=mask_k, other=0.0)

    # Load pos[t, k]
    p_off = t * p_sT + k_offsets * p_sK
    pos_val = tl.load(pos_ptr + p_off, mask=mask_k, other=0.0)

    # Scale and add
    y_val = y_val * scale + pos_val

    # Store back
    tl.store(y_ptr + y_off, y_val, mask=mask_k)
