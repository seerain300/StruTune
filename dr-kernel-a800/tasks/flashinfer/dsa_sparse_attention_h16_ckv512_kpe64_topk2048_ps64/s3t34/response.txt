import math
import torch
import triton
import triton.language as tl


@triton.jit
def matmul_row(A_ptr, B_ptr, C_ptr,
               N, M: tl.constexpr, BLOCK_M: tl.constexpr):
    """
    Compute C[j] = sum_i A[i] * B[j, i] for j in [0, N).
    A is 1D vector length M. B is contiguous [N, M].
    """
    j_vec = tl.arange(0, BLOCK_M)
    acc = tl.zeros([BLOCK_M], dtype=tl.float32)
    for m0 in range(0, M, BLOCK_M):
        j = m0 + j_vec
        mask = j < M
        a_chunk = tl.load(A_ptr + j, mask=mask, other=0.0)
        # B is [N, M] contiguous, so index = j * M + i
        b_block = tl.load(B_ptr + j[:, None] * M + (m0 + j_vec)[None, :],
                          mask=mask[:, None], other=0.0)
        # acc[j] += sum over ii of a_chunk[ii] * b_block[ii, j]
        for ii in range(BLOCK_M):
            i = m0 + ii
            ai = a_chunk[ii]
            row = b_block[ii, :]
            acc += ai * row
    store_j = tl.arange(0, N)
    tl.store(C_ptr + store_j, acc, mask=store_j < N)


@triton.jit
def softmax_logsumexp2_row(X_ptr, Valid_ptr, Out_ptr, LSE_ptr,
                            N: tl.constexpr, TOPK: tl.constexpr):
    """
    Compute stable softmax for TOPK entries with Valid mask (int32 0/1),
    and write base-2 logsumexp to LSE_ptr[head].
    Out_ptr is [TOPK], only entries with Valid!=0 contribute.
    """
    # Pass 1: find max over valid
    m = -float("inf")
    for j in range(0, TOPK):
        valid_j = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        if valid_j != 0:
            m = tl.maximum(m, xj)

    # Pass 2: sum of exp(xj - m) over valid
    s = 0.0
    for j in range(0, TOPK):
        valid_j = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        if valid_j != 0:
            s += tl.exp(xj - m)

    # LSE = m + log(s) / ln(2)
    lse_val = m + tl.log(s) * 1.4426950408889634  # 1/ln(2)
    tl.store(LSE_ptr + 0, lse_val)

    # Pass 3: write normalized softmax to Out_ptr for valid entries
    inv_s = 1.0 / s
    for j in range(0, TOPK):
        valid_j = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        if valid_j != 0:
            out_j = tl.exp(xj - m) * inv_s
        else:
            out_j = 0.0
        tl.store(Out_ptr + j, out_j)


@triton.jit
def reduction_row(Attn_ptr, B_ptr, Out_ptr,
                  N, OUT: tl.constexpr, BLOCK_OUT: tl.constexpr, BLOCK_TOPK: tl.constexpr):
    """
    Compute Out[k] = sum_{i=0..TOPK-1} Attn[i] * B[k, i] for k in [0, OUT).
    Attn is [N] with N <= TOPK (we use N=num_valid). B is [N, OUT] contiguous.
    """
    k_vec = tl.arange(0, BLOCK_OUT)
    acc = tl.zeros([BLOCK_OUT], dtype=tl.float32)
    for k0 in range(0, OUT, BLOCK_OUT):
        ks = k0 + k_vec
        mask_k = ks < OUT
        # reduce over TOPK in chunks
        for t0 in range(0, TOPK, BLOCK_TOPK):
            ts = t0 + tl.arange(0, BLOCK_TOPK)
            mask_t = ts < N  # N is passed as runtime, but we guard in host
            attn_chunk = tl.load(Attn_ptr + ts, mask=mask_t, other=0.0)
            b_block = tl.load(B_ptr + ks[:, None] * N + ts[None, :],
                              mask=mask_k[:, None] & mask_t[None, :], other=0.0)
            # sum over BLOCK_TOPK dimension
            acc += tl.sum(b_block * attn_chunk[None, :], axis=1)
    tl.store(Out_ptr + k_vec, acc, mask=mask_k)

This version keeps the Triton-only requirement strictly: no torch matmul, softmax, or reductions in forward. All heavy math is performed by Triton kernels, and the forward function only allocates tensors, gathers data, and launches kernels.

Analysis and plan
- Original per-token, per-head logic:
  - From flattened caches:
    - Kc_all: [num_pages*64, 512] → gather valid rows using sparse_indices to get Kc_rows: [num_valid, 512]
    - Kp_all: [num_pages*64, 64] → gather to get Kp_rows: [num_valid, 64]
  - qn = q_nope[t, h] → [512], qp = q_pe[t, h] → [64]
  - Compute two dot products:
    - logits_qn = qn @ Kc_rows.T → [num_valid]
    - logits_qp = qp @ Kp_rows.T → [num_valid]
    - logits = logits_qn + logits_qp
  - Compute per-head base-2 logsumexp:
    - lse = logsumexp_base2(logits_scaled)
  - Compute attention vector:
    - attn = softmax(logits_scaled)
  - Compute output:
    - output[t, h] = attn @ Kc_rows → [512], stored as bfloat16

- Triton kernels:
  1) matmul_row(A, B_rows, C):
     - A is a 1D vector (length M), B_rows is [N, M] contiguous, C is [N].
     - Used for qn @ Kc_rows.T (M=512) and qp @ Kp_rows.T (M=64).
     - Accumulates in float32 with BLOCK-sized loops.
  2) softmax_logsumexp2_row(X, Valid, Out, LSE_ptr):
     - X: float32 vector of length TOPK (2048).
     - Valid: int32 mask (0/1) for entries.
     - Pass 1: find max over valid.
     - Pass 2: compute sum of exp(x - max) over valid.
     - LSE = max + log(sum) / ln(2). Write to LSE_ptr[head].
     - Pass 3: write normalized softmax probabilities for valid entries into Out.
  3) reduction_row(Attn, B_rows, Out):
     - Attn: float32 vector [N] (attention scores).
     - B_rows: [N, OUT] contiguous (OUT=512).
     - Compute Out[k] = sum_i Attn[i] * B_rows[k, i] for k in [0, OUT).
     - Accumulate in float32 with BLOCK-sized chunks.

- Forward flow:
  - For each token t:
    - Build valid_indices from sparse_indices[t] (mask out -1).
    - Gather Kc_rows and Kp_rows accordingly.
    - For each head h:
      - Launch matmul_row to get logits_qn and logits_qp.
      - Concatenate into X and Valid with TOPK=2048 (mask padded entries).
      - Launch softmax_logsumexp2_row to get attn and lse[t, h].
      - Launch reduction_row to compute output[t, h].
    - Store outputs and lse.

- Important: This expects q_nope and q_pe shaped as [num_tokens, num_qo_heads, dim] where num_tokens is the first axis. If your environment supplies [1, 16, ...], get_inputs can adjust accordingly.

import math
import torch
import triton
import triton.language as tl


@triton.jit
def matmul_row(A_ptr, B_ptr, C_ptr,
               N, M: tl.constexpr, BLOCK_M: tl.constexpr):
    """
    Compute C[j] = sum_i A[i] * B[j, i] for j in [0, N).
    A is 1D vector of length M. B is a contiguous [N, M] matrix.
    """
    j_vec = tl.arange(0, BLOCK_M)
    acc = tl.zeros([BLOCK_M], dtype=tl.float32)
    for m0 in range(0, M, BLOCK_M):
        j = m0 + j_vec
        mask = j < M
        a_chunk = tl.load(A_ptr + j, mask=mask, other=0.0)
        b_block = tl.load(B_ptr + j[:, None] * M + (m0 + j_vec)[None, :],
                          mask=mask[:, None], other=0.0)
        for ii in range(BLOCK_M):
            i = m0 + ii
            ai = a_chunk[ii]
            row = b_block[ii, :]
            acc += ai * row
    store_j = tl.arange(0, N)
    tl.store(C_ptr + store_j, acc, mask=store_j < N)


@triton.jit
def softmax_logsumexp2_row(X_ptr, Valid_ptr, Out_ptr, LSE_ptr,
                            N: tl.constexpr, TOPK: tl.constexpr):
    """
    Compute stable softmax over TOPK entries in X with validity mask Valid (int32 0/1),
    write softmax probabilities to Out_ptr (only valid entries get probability),
    and write base-2 logsumexp to LSE_ptr[0].
    """
    # Pass 1: find max over valid
    m = -float("inf")
    for j in range(0, TOPK):
        valid_j = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        if valid_j != 0:
            m = tl.maximum(m, xj)

    # Pass 2: sum of exp(xj - m) over valid
    s = 0.0
    for j in range(0, TOPK):
        valid_j = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        if valid_j != 0:
            s += tl.exp(xj - m)

    # LSE = m + log(s) / ln(2)
    lse_val = m + tl.log(s) * 1.4426950408889634  # 1/ln(2)
    tl.store(LSE_ptr + 0, lse_val)

    # Pass 3: write normalized softmax to Out_ptr for valid entries
    inv_s = 1.0 / s
    for j in range(0, TOPK):
        valid_j = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        if valid_j != 0:
            out_j = tl.exp(xj - m) * inv_s
        else:
            out_j = 0.0
        tl.store(Out_ptr + j, out_j)
