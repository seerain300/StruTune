    out_ptr, lse_ptr,

    sm_scale, inv_log2,
):

    t = tl.program_id(0)

    base_idx = t * topk
    offs = tl.arange(0, topk)

    valid_mask = idx_vals != -1

    inv_log2 = inv_log2

        # Online logsumexp variables for this head

        sum_exp = tl.full((), 0.0, dtype=tl.float32)  # sum of exp(logit_scaled - m)

        for j in range(0, topk):

                Kp_row_ptr = Kp_all_ptr + tok_idx * Dp
