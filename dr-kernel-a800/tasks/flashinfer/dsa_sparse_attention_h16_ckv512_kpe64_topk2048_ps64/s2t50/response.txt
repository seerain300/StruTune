import torch
import triton
import triton.language as tl


@triton.jit
def attention_allheads_token_kernel(
    q_nope_ptr, q_pe_ptr, Kc_all_ptr, Kp_all_ptr, sparse_idx_ptr,
    out_ptr, lse_ptr,
    N, H, Dk, Dp, topk,
    sm_scale, inv_log2,
):
    # One program per token t
    t = tl.program_id(0)

    # Load indices for this token; sparse_idx_ptr is flattened [N*topk]
    base_idx = t * topk
    offs = tl.arange(0, topk)
    idx_vals = tl.load(sparse_idx_ptr + base_idx + offs, mask=offs < topk, other=-1)  # int32
    valid_mask = idx_vals != -1

    # For each head h: compute lse and then output
    for h in range(0, H):
        # Online logsumexp for logits_scaled[h, :] across candidates
        m = tl.full((), -1.0e30, dtype=tl.float32)
        sum_exp = tl.full((), 0.0, dtype=tl.float32)

        # First pass: compute m and sum_exp
        for j in range(0, topk):
            # if invalid, skip
            valid = valid_mask[j]
            if valid:
                j_idx = idx_vals[j]
                # Load Kc[j] and Kp[j] vectors
                # Kc_all_ptr: [num_pages*page_size, Dk] float32
                Kc_j = tl.load(Kc_all_ptr + j_idx * Dk + tl.arange(0, Dk))
                Kp_j = tl.load(Kp_all_ptr + j_idx * Dp + tl.arange(0, Dp))

                # q_nope[h, :] = row h of q_nope[t], float32
                # q_nope_ptr layout: [N, H, Dk] -> row offset = t * (H*Dk) + h * Dk
                qno_row_offset = t * (H * Dk) + h * Dk
                qno_vec = tl.load(q_nope_ptr + qno_row_offset + tl.arange(0, Dk), mask=True, other=0.0)
                qpe_vec = tl.load(q_pe_ptr + t * (H * Dp) + h * Dp + tl.arange(0, Dp), mask=True, other=0.0)

                # logits[h, j] = sum(qno_vec @ Kc_j) + sum(qpe_vec @ Kp_j)
                prod1 = (qno_vec * Kc_j).sum()
                prod2 = (qpe_vec * Kp_j).sum()
                logit = prod1 + prod2

                logit_scaled = logit * sm_scale
                # Online update: m, sum_exp
                is_greater = logit_scaled > m
                # if logit_scaled > m: new m = logit_scaled; sum_exp = sum_exp * exp(m - logit_scaled) + 1
                # else: sum_exp += exp(logit_scaled - m)
                # Note: tl.where does not branch; we compute both and select with masks
                new_sum_gt = sum_exp * tl.exp(m - logit_scaled) + 1.0
                new_sum_le = sum_exp + tl.exp(logit_scaled - m)
                sum_exp = tl.where(is_greater, new_sum_gt, new_sum_le)
                m = tl.where(is_greater, logit_scaled, m)

        # Compute lse[h] = m + log(sum_exp)/ln(2)
        lse_h = m + sum_exp * inv_log2
        tl.store(lse_ptr + t * H + h, lse_h)

        # Second pass: compute attn[h, :] and accumulate output
        # out_ptr is [N, H, Dk] float32, row-major: base = t * (H*Dk) + h * Dk
        out_base = t * (H * Dk) + h * Dk
        # Initialize output row
        for d in range(0, Dk):
            tl.store(out_ptr + out_base + d, 0.0)

        for j in range(0, topk):
            valid = valid_mask[j]
            if valid:
                j_idx = idx_vals[j]
                Kc_j = tl.load(Kc_all_ptr + j_idx * Dk + tl.arange(0, Dk))
                Kp_j = tl.load(Kp_all_ptr + j_idx * Dp + tl.arange(0, Dp))

                qno_vec = tl.load(q_nope_ptr + qno_row_offset + tl.arange(0, Dk), mask=True, other=0.0)
                qpe_vec = tl.load(q_pe_ptr + t * (H * Dp) + h * Dp + tl.arange(0, Dp), mask=True, other=0.0)

                prod2 = (qpe_vec * Kp_j).sum()
                logit = prod1 + prod2
                logit_scaled = logit * sm_scale

                attn_j = tl.exp(logit_scaled - lse_h)
                # out[h, :] += attn_j * Kc_j

import torch
import triton
import triton.language as tl


@triton.jit
def attention_allheads_token_kernel(
    q_nope_ptr, q_pe_ptr, Kc_all_ptr, Kp_all_ptr, sparse_idx_ptr,
    out_ptr, lse_ptr,
    N, H, Dk, Dp, topk,
    sm_scale, inv_log2,
):
    # One program per token t
    t = tl.program_id(0)

    # Load indices for this token; sparse_idx_ptr is flattened [N*topk]
    base_idx = t * topk
    offs = tl.arange(0, topk)
    idx_vals = tl.load(sparse_idx_ptr + base_idx + offs, mask=offs < topk, other=-1)  # int32
    valid_mask = idx_vals != -1  # [topk] boolean

    # Compute base offsets for q_nope, q_pe
    qno_base = t * (H * Dk)  # row start for this token in q_nope
    qpe_base = t * (H * Dp)  # row start for this token in q_pe

    for h in range(0, H):
        # Prepare base offset for head h
        qno_h_base = qno_base + h * Dk
        qpe_h_base = qpe_base + h * Dp

        # Vectorized computation over candidates
        j = tl.arange(0, topk)
        valid = valid_mask[j]  # [topk] boolean

        # Load Kc and Kp vectors for each candidate (masked)
        j_idx_f = idx_vals.to(tl.float32)  # cast to float for pointer arithmetic
        Kc_vecs = tl.load(Kc_all_ptr + j_idx_f * Dk + tl.arange(0, Dk), mask=valid, other=0.0)  # [topk, Dk]
        Kp_vecs = tl.load(Kp_all_ptr + j_idx_f * Dp + tl.arange(0, Dp), mask=valid, other=0.0)  # [topk, Dp]

        # Load q_nope row for head h and q_pe row for head h (lengths Dk and Dp)
        qno_vec = tl.load(q_nope_ptr + qno_h_base + tl.arange(0, Dk))
        qpe_vec = tl.load(q_pe_ptr + qpe_h_base + tl.arange(0, Dp))

        # Compute logits for each candidate: qno_vec @ Kc_vec + qpe_vec @ Kp_vec
        # Kc_vecs: [topk, Dk], qno_vec: [Dk] -> [topk]
        dot1 = (Kc_vecs * qno_vec[None, :]).sum(axis=1)  # [topk]
        dot2 = (Kp_vecs * qpe_vec[None, :]).sum(axis=1)  # [topk]
        logits = dot1 + dot2  # [topk], float32

        # Scale
        logit_scaled = logits * sm_scale  # [topk]

        # Online logsumexp: m = max, sum_exp = sum(exp(logit_scaled - m))
        # Triton reductions over vector
        m = tl.max(logit_scaled, axis=0)  # scalar
        sum_exp = tl.sum(tl.exp(logit_scaled - m), axis=0)  # scalar

        # lse[h] = m + log(sum_exp)/ln(2)
        lse_h = m + tl.log(sum_exp) * inv_log2
        tl.store(lse_ptr + t * H + h, lse_h)  # per head, per token

        # Compute output row: out[h, :] = sum_j attn_j * Kc_j
        # attn_j = exp(logit_scaled - lse_h)
        # We can accumulate directly using reductions without materializing attn.
        attn_scaled = logit_scaled - lse_h  # [topk]
        attn = tl.exp(attn_scaled)  # [topk]

        # Sum of attn * Kc_j over valid j
        prod = (attn[None, :] * Kc_vecs)  # [topk, Dk]
        sum_prod = tl.sum(prod, axis=0)   # [Dk]
        # Normalize by sum_exp
        out_row = sum_prod / sum_exp

        # Store output row h for token t
        out_base = t * (H * Dk) + h * Dk
        tl.store(out_ptr + out_base + tl.arange(0, Dk), out_row)
