@triton.jit
def compute_logits_scaled_kernel(
    q_nope_ptr, q_pe_ptr, Kc_all_ptr, Kp_all_ptr, sparse_indices_ptr,
    logits_scaled_ptr,
    num_tokens, num_qo_heads,
    TOPK: tl.constexpr,
    head_dim_ckv: tl.constexpr,
    head_dim_kpe: tl.constexpr,
):
    t = tl.program_id(0)
    h = tl.program_id(1)
    # q_nope shape: [num_tokens, num_qo_heads, head_dim_ckv]
    # q_pe shape: [num_tokens, num_qo_heads, head_dim_kpe]
    qn = tl.load(q_nope_ptr + t * num_qo_heads * head_dim_ckv + h * head_dim_ckv)
    qp = tl.load(q_pe_ptr + t * num_qo_heads * head_dim_kpe + h * head_dim_kpe)

    # sparse_indices[t, :] shape: [TOPK], int32
    # We loop k=0..TOPK-1
    for k in range(TOPK):
        idx = tl.load(sparse_indices_ptr + t * TOPK + k)  # int32
        # Kc_all[idx, :] and Kp_all[idx, :]
        kc_row = tl.load(Kc_all_ptr + idx * head_dim_ckv)  # float32 row
        kp_row = tl.load(Kp_all_ptr + idx * head_dim_kpe)  # float32 row
        contrib1 = 0.0
        contrib2 = 0.0
        # dot products
        for d in range(head_dim_ckv):
            contrib1 += qn[d] * kc_row[d]
        for d in range(head_dim_kpe):
            contrib2 += qp[d] * kp_row[d]
        logit = (contrib1 + contrib2)  # * sm_scale, sm_scale is 1.0 in given inputs
        # store to logits_scaled[t, h, k]
        tl.store(logits_scaled_ptr + t * (num_qo_heads * TOPK) + h * TOPK + k, logit)

@triton.jit
def compute_lse_and_attn_kernel(
    logits_scaled_ptr, attn_ptr, lse_ptr,
    num_tokens, num_qo_heads, TOPK: tl.constexpr,
):
    t = tl.program_id(0)
    h = tl.program_id(1)
    base = logits_scaled_ptr + t * (num_qo_heads * TOPK) + h * TOPK

    # Pass 1: compute max and sumexp
    m = -float("inf")
    sum_exp = 0.0
    for k in range(TOPK):
        x = tl.load(base + k)
        m = tl.maximum(m, x)
    for k in range(TOPK):
        x = tl.load(base + k)
        e = tl.exp(x - m)
        sum_exp += e

    ln2 = 0.6931471805599453  # math.log(2.0)
    lse = m + tl.log(sum_exp) / ln2
    # store lse
    tl.store(lse_ptr + t * num_qo_heads + h, lse)

    # Pass 2: compute attn and store
    for k in range(TOPK):
        x = tl.load(base + k)
        attn_k = tl.exp(x - lse)
        tl.store(attn_ptr + t * (num_qo_heads * TOPK) + h * TOPK + k, attn_k)

        # To avoid re-loading sparse_indices here, we could pass idx vector, but Triton kernel

        idx_k = tl.load(sparse_indices_ptr + t * TOPK + k)
