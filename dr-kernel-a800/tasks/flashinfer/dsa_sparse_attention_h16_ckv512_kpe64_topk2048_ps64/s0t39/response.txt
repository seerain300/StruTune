Root-cause analysis and plan
- The last submission failed to compile. The Triton JIT error indicates the kernel defined with a for-loop over dynamic ranges is not valid. Triton requires static loops; using for v_start in range(0, TOPK, v_tile) is invalid when TOPK is a runtime value. Even if we intended to declare TOPK as tl.constexpr, the loop variable must be known at compile time. Fix: remove dynamic loops, process one “v” at a time inside a single program per head. TOPK=2048 is small enough to handle this.
- We also should avoid any host-side torch computation; all math must be inside Triton kernels. The original code does dot products, max, sum, exp, softmax, and matmul. We’ll implement:
  - compute_logits: logits[h, v] = dot(qn[h], Kc[v]) + dot(qp[h], Kp[v])
  - lse base-2: lse[h] = m + log(sum(exp(logits_scaled[h,:] - m))) / ln(2), where m = max(logits_scaled[h,:])
  - softmax and output: output[h, c] = sum_v softmax_scaled[h, v] * Kc[v, c], with softmax_scaled = exp(logits_scaled[h, v] - lse[h]) / sum_v exp(logits_scaled[h, v] - lse[h])
- Ensure robust handling:
  - Flatten Kc_all/Kp_all from [num_pages, 64, D] to [num_pages*64, D]. We don’t need to do this in Python; the forward will pass already flattened tensors.
  - Process each token t: gather valid indices, compute outputs for all 16 heads, store output_t as fp32 (then cast to bfloat16 on return). Compute lse per head per token using a Triton kernel, then use that in a Triton matmul kernel to produce the final output for that token. This avoids using torch ops in forward.
- Performance considerations:
  - Per-token compute is dominated by 16 x 2048 dot products (Kc has 512 columns). Each dot is 512 multiplies-adds, so about 16*2048*512 ≈ 16.7M MACs per token, and we have num_tokens tokens. This is fine for modern GPUs.
  - We avoid loops over TOPK inside Triton; instead, we perform 2048 independent compute operations per head and let Triton vectorize over the head dimension. This keeps the kernel simple and avoids dynamic for-loops.

# Kernel 1: compute logits for one head across all valid positions v.
# Inputs:
#   qn_ptr: *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV], we index by h
#   qp_ptr: *fp32, [NUM_QO_HEADS, HEAD_DIM_KPE], we index by h

#   Kp_ptr: *fp32, [TOPK, HEAD_DIM_KPE]
#   logits_ptr: *fp32, [NUM_QO_HEADS, TOPK]
@triton.jit
def _compute_logits_row_kernel(
    qn_ptr,           # *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV]
    qp_ptr,           # *fp32, [NUM_QO_HEADS, HEAD_DIM_KPE]
    Kc_ptr,           # *fp32, [TOPK, HEAD_DIM_CKV]
    Kp_ptr,           # *fp32, [TOPQO_HEADS, TOPK]
    NUM_QO_HEADS: tl.constexpr,
    TOPK: tl.constexpr,
    HEAD_DIM_CKV: tl.constexpr,
    HEAD_DIM_KPE: tl.constexpr,
):
    # One program per head
    h = tl.program_id(axis=0)

    # Pointers to this head's q vectors
    qn_h_ptr = qn_ptr + h * HEAD_DIM_CKV
    qp_h_ptr = qp_ptr + h * HEAD_DIM_KPE

    # Initialize logits row for this head
    # We will write logits[h, v] for all v
    # Note: We use a scalar loop over TOPK to avoid dynamic for-loops in Triton
    for v in range(TOPK):
        # Accumulate dot products for both parts
        acc1 = 0.0  # scalar fp32
        acc2 = 0.0  # scalar fp32

        # Compute dot(qn[h], Kc[v, :])
        for c in range(HEAD_DIM_CKV):
            kc = tl.load(Kc_ptr + v * HEAD_DIM_CKV + c)  # scalar
            qc = tl.load(qn_h_ptr + c)                   # scalar

            kp = tl.load(Kp_ptr + v * HEAD_DIM_KPE + p)  # scalar
            qp = tl.load(qp_h_ptr + p)                   # scalar
            acc2 += kp * qp

        # Store logits[h, v]
        tl.store(logits_ptr + h * TOPK + v, acc1 + acc2)


# Kernel 2: compute per-head LSE in base-2: lse[h] = m + log(sum(exp(logits_scaled[h,:] - m))) / ln(2)
# Inputs:
#   logits_ptr: *fp32, [NUM_QO_HEADS, TOPK]
#   lse_ptr: *fp32, [NUM_QO_HEADS]
@triton.jit
def _lse_base2_kernel(
    logits_ptr,       # *fp32, [NUM_QO_HEADS, TOPK]
    lse_ptr,          # *fp32, [NUM_QO_HEADS]
    NUM_QO_HEADS: tl.constexpr,
    TOPK: tl.constexpr,
    LOG2E: tl.constexpr,  # log(2), provided as scalar
):
    h = tl.program_id(axis=0)

    # Compute maximum over TOPK for this head
    m = -float('inf')
    for v in range(TOPK):
        val = tl.load(logits_ptr + h * TOPK + v)
        if val > m:
            m = val

    # Compute sum of exp(logits_scaled - m)
    sum_exp = 0.0
    for v in range(TOPK):
        val = tl.load(logits_ptr + h * TOPK + v)
        sum_exp += tl.exp(val - m)

    lse = m + tl.log(sum_exp) * LOG2E
    tl.store(lse_ptr + h, lse)


# Kernel 3: compute output for one head using already computed lse[h]:
# out[h, :] = sum_v exp(logits_scaled[h, v] - lse[h]) * Kc[v, :]
# Inputs:
#   qn_ptr: *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV]
#   qp_ptr: *fp32, [NUM_QO_HEADS, HEAD_DIM_KPE]
#   Kc_ptr: *fp32, [TOPK, HEAD_DIM_CKV]
#   Kp_ptr: *fp32, [TOPK, HEAD_DIM_KPE]
#   logits_ptr: *fp32, [NUM_QO_HEADS, TOPK]
#   lse_ptr: *fp32, [NUM_QO_HEADS]
#   out_ptr: *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV]
@triton.jit
def _softmax_matmul_kernel(
    qn_ptr,           # *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV]
    qp_ptr,           # *fp32, [NUM_QO_HEADS, HEAD_DIM_KPE]
    Kc_ptr,           # *fp32, [TOPK, HEAD_DIM_CKV]
    Kp_ptr,           # *fp32, [TOPK, HEAD_DIM_KPE]
    logits_ptr,       # *fp32, [NUM_QO_HEADS, TOPK]
    lse_ptr,          # *fp32, [NUM_QO_HEADS]
    out_ptr,          # *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV]
    NUM_QO_HEADS: tl.constexpr,
    TOPK: tl.constexpr,
    HEAD_DIM_CKV: tl.constexpr,
    HEAD_DIM_KPE: tl.constexpr,
):
    h = tl.program_id(axis=0)

    # Compute lse for this head
    lse_h = tl.load(lse_ptr + h)

    # We will accumulate output[h, :] across all v
    out_row = tl.zeros([HEAD_DIM_CKV], dtype=tl.float32)

    for v in range(TOPK):
        val_scaled = tl.load(logits_ptr + h * TOPK + v)
        prob = tl.exp(val_scaled - lse_h)
        # contribution to each c from this v: prob * Kc[v, c]
        for c in range(HEAD_DIM_CKV):
            kc = tl.load(Kc_ptr + v * HEAD_DIM_CKV + c)  # scalar
            out_row[c] += prob * kc

    # Store the output row for this head
    tl.store(out_ptr + h * HEAD_DIM_CKV + tl.arange(0, HEAD_DIM_CKV), out_row)

    m = -float("inf")
    for v in range(TOPK):
        val = tl.load(logits_ptr + h * TOPK + v)
        if val > m:
            m = val
    sum_exp = 0.0
    for v in range(TOPK):
        val = tl.load(logits_ptr + h * TOPK + v)
        sum_exp += tl.exp(val - m)
    lse = m + tl.log(sum_exp) * LOG2E
    tl.store(lse_ptr + h, lse)


@triton.jit
def _softmax_matmul_kernel(
    qn_ptr,           # *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV]
    qp_ptr,           # *fp32, [NUM_QO_HEADS, HEAD_DIM_KPE]
    Kc_ptr,           # *fp32, [TOPK, HEAD_DIM_CKV]
    Kp_ptr,           # *fp32, [TOPK, HEAD_DIM_KPE]
    logits_ptr,       # *fp32, [NUM_QO_HEADS, TOPK]
    lse_val,          # scalar fp32, lse for this head
    out_ptr,          # *fp32, [NUM_QO_HEADS, HEAD_DIM_CKV] -> we'll store row h
    NUM_QO_HEADS: tl.constexpr,
    TOPK: tl.constexpr,
    HEAD_DIM_CKV: tl.constexpr,
    HEAD_DIM_KPE: tl.constexpr,
):
    h = tl.program_id(axis=0)
    # Compute output[h, :] = sum_v exp(logits_scaled[h, v] - lse_val) * Kc_sel[v, :]
    out_row = tl.zeros([HEAD_DIM_CKV], dtype=tl.float32)
    # Reconstruct softmax contributions across v and accumulate into out_row
    for v in range(TOPK):
        val_scaled = tl.load(logits_ptr + h * TOPK + v)
        prob = tl.exp(val_scaled - lse_val)
        # Multiply with Kc_sel[v, :] and accumulate
        for c in range(HEAD_DIM_CKV):
            kc = tl.load(Kc_ptr + v * HEAD_DIM_CKV + c)
            out_row[c] += prob * kc
    # Store the row
    tl.store(out_ptr + h * HEAD_DIM_CKV + tl.arange(0, HEAD_DIM_CKV), out_row)


def _run_triton(q_nope, q_pe, Kc_all, Kp_all, sparse_indices, sm_scale):
    # Triton-only forward: no torch ops for computation on tensors
    assert TRITON_AVAILABLE, "Triton is not available"
    assert q_nope.is_cuda and q_pe.is_cuda and Kc_all.is_cuda and Kp_all.is_cuda and sparse_indices.is_cuda, "All tensors must be on CUDA for Triton"

    num_tokens = q_nope.shape[0]
    device = q_nope.device

    # Output and lse buffers
    output = torch.empty((num_tokens, NUM_QO_HEADS, HEAD_DIM_CKV), dtype=torch.float32, device=device)
    lse = torch.empty((num_tokens, NUM_QO_HEADS), dtype=torch.float32, device=device)

    for t in range(num_tokens):
        indices = sparse_indices[t]  # [TOPK], int32
        valid = indices != -1
        valid_indices = indices[valid].to(torch.long)  # [num_valid]
        num_valid = valid_indices.numel()
        if num_valid == 0:
            # No valid entries for this token: output zeros, lse = -inf
            output[t].zero_()
            lse[t] = torch.full((NUM_QO_HEADS,), -float("inf"), dtype=torch.float32, device=device)
            continue

        # Gather selected rows from flattened caches
        # Ensure contiguous for Triton loads
        Kc_sel = Kc_all[valid_indices].contiguous()  # [num_valid, HEAD_DIM_CKV], fp32
        Kp_sel = Kp_all[valid_indices].contiguous()  # [num_valid, HEAD_DIM_KPE], fp32

        # Cast q vectors to fp32
        qn = q_nope[t].to(torch.float32).contiguous()  # [NUM_QO_HEADS, HEAD_DIM_CKV]
        qp = q_pe[t].to(torch.float32).contiguous()   # [NUM_QO_HEADS, HEAD_DIM_KPE]

        # Compute logits for all heads
        logits = torch.empty((NUM_QO_HEADS, TOPK), dtype=torch.float32, device=device)

        grid = (NUM_QO_HEADS,)
        _compute_logits_kernel[grid](
            qn, qp, Kc_sel, Kp_sel, logits,
            NUM_QO_HEADS=NUM_QO_HEADS, TOPK=TOPK,
            HEAD_DIM_CKV=HEAD_DIM_CKV, HEAD_DIM_KPE=HEAD_DIM_KPE,
        )

        # Compute per-head LSE in base-2
        lse_t = torch.empty((NUM_QO_HEADS,), dtype=torch.float32, device=device)
        _lse_base2_kernel[grid](
            logits, lse_t,
            NUM_QO_HEADS=NUM_QO_HEADS, TOPK=TOPK, LOG2E=1.4426950408889634,  # log(2)
        )
        lse[t] = lse_t

        # Compute output per head: out[h, :] = softmax_scaled @ Kc_sel
        for h in range(NUM_QO_HEADS):
            out_row = torch.empty((HEAD_DIM_CKV,), dtype=torch.float32, device=device)
            _softmax_matmul_kernel[grid](
                qn, qp, Kc_sel, Kp_sel, logits, lse[t, h], out_row,
                NUM_QO_HEADS=NUM_QO_HEADS, TOPK=TOPK,
                HEAD_DIM_CKV=HEAD_DIM_CKV, HEAD_DIM_KPE=HEAD_DIM_KPE,
            )
            output[t, h] = out_row

    # Return output in bfloat16, lse in float32
    return output.to(torch.bfloat16), lse


class ModelNew(torch.nn.Module):
    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, sparse_indices, sm_scale):
        # Triton-only forward: no torch ops for tensor computation
        return _run_triton(q_nope, q_pe, ckv_cache, kpe_cache, sparse_indices, sm_scale)
