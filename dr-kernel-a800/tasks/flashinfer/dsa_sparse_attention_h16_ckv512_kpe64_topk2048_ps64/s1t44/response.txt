import torch

    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True

# Grid: (num_tokens * num_qo_heads, total_valid_k)  where total_valid_k is the sum of valid indices across tokens

@triton.jit

    Kc_all_ptr,      # *float32 [num_pages * page_size, head_dim_ckv]
    Kp_all_ptr,      # *float32 [num_pages * page_size, head_dim_kpe]
    out_ptr,         # *float32 [num_tokens, num_qo_heads, total_valid_k]
    # dimensions
    num_tokens,      # int32
    num_qo_heads,    # int32
    head_dim_ckv,    # int32
    head_dim_kpe,    # int32
    total_valid_k,   # int32 (constexpr for compile-time loop)
    BLOCK: tl.constexpr,  # chunk size for reduction over dims
):

    pid_k = tl.program_id(1)  # over total_valid_k

    # Map pid_m to (t, h)
    t = pid_m // num_qo_heads
    h = pid_m % num_qo_heads

    # Bounds check for tokens
    if t >= num_tokens:
        return

    # Load q vectors for this (t, h)
    # q_nope_ptr layout: [num_tokens, num_qo_heads, head_dim_ckv]
    qn = tl.load(q_nope_ptr + t * (num_qo_heads * head_dim_ckv) + h * head_dim_ckv + tl.arange(0, BLOCK),
                 mask=tl.arange(0, BLOCK) < head_dim_ckv, other=0.0).to(tl.float32)
    # q_pe_ptr layout: [num_tokens, num_qo_heads, head_dim_kpe]
    qp = tl.load(q_pe_ptr + t * (num_qo_heads * head_dim_kpe) + h * head_dim_kpe + tl.arange(0, BLOCK),

    # Load Kc_all row and Kp_all row for index pid_k (already guaranteed valid index in host code)

                     mask=tl.arange(0, BLOCK) < head_dim_ckv, other=0.0).to(tl.float32)

                     mask=tl.arange(0, BLOCK) < head_dim_kpe, other=0.0).to(tl.float32)

    # Dot products (reduce over BLOCK chunks)
    contrib1 = 0.0
    contrib2 = 0.0
    for off in range(0, head_dim_ckv, BLOCK):
        qn_chunk = tl.load(q_nope_ptr + t * (num_qo_heads * head_dim_ckv) + h * head_dim_ckv + off + tl.arange(0, BLOCK),
                           mask=tl.arange(0, BLOCK) < (head_dim_ckv - off), other=0.0).to(tl.float32)
        Kc_chunk = tl.load(Kc_all_ptr + idx * head_dim_ckv + off + tl.arange(0, BLOCK),
                           mask=tl.arange(0, BLOCK) < (head_dim_ckv - off), other=0.0).to(tl.float32)
        contrib1 += tl.sum(qn_chunk * Kc_chunk, axis=0)

    for off in range(0, head_dim_kpe, BLOCK):
        qp_chunk = tl.load(q_pe_ptr + t * (num_qo_heads * head_dim_kpe) + h * head_dim_kpe + off + tl.arange(0, BLOCK),
                           mask=tl.arange(0, BLOCK) < (head_dim_kpe - off), other=0.0).to(tl.float32)
        Kp_chunk = tl.load(Kp_all_ptr + idx * head_dim_kpe + off + tl.arange(0, BLOCK),
                           mask=tl.arange(0, BLOCK) < (head_dim_kpe - off), other=0.0).to(tl.float32)
        contrib2 += tl.sum(qp_chunk * Kp_chunk, axis=0)

    # Store logits_scaled for (t, h, idx)
    tl.store(out_ptr + t * (num_qo_heads * total_valid_k) + h * total_valid_k + idx,
             (contrib1 + contrib2), eviction_policy="evict_last")

        q_pe:   [num_tokens, num_qo_heads, head_dim_kpe], bfloat16

            total_valid_k += int(valid.sum().item())

        BLOCK_KC = 128
        BLOCK_KP = 64

            idx = sparse_indices[t]  # [topk]
            valid = idx != -1
            K_actual = int(valid.sum().item())

            # We need to map each pid_k to its position in the slice. Triton didn't produce per-token slices; since we set K=total_valid_k, we can reconstruct using an index vector that matches original order: recompute valid positions in original topk order via idx[valid]. We can obtain this by calling the same _compute_logits_kernel for each t with K_actual instead of total_valid_k? However, Triton launch needs fixed K.             # Since correctness is paramount, replace Triton logits with PyTorch dot-products for this token to guarantee identical behavior.
            # Compute q_nope[t] @ Kc_all_valid^T and q_pe[t] @ Kp_all_valid^T, where valid = idx != -1
            # Build Kc_all_valid and Kp_all_valid based on idx[valid]

            # The Triton kernel above was trying to write to a fixed buffer sized for topk=2048. That can lead to incorrect order or missing entries. To ensure exact match, we will compute logits in PyTorch instead.

            # q_nope[t, h, :] shape: [head_dim_ckv]
            for h in range(num_qo_heads):

                qp = q_pe_f32[t, h, :]             # [head_dim_kpe]

                contrib1 = (qn @ Kc_valid.transpose(0, 1)).to(torch.float32)  # [K_actual]

                # lse = logsumexp(logits_scaled) / ln(2)
                m = torch.max(logits_scaled)                                  # scalar
                sum_exp = torch.sum(torch.exp(logits_scaled - m))            # scalar

                lse[t, h] = (m + math.log(sum_exp.item(), math.e)) / math.log(2.0, math.e)  # float32

                # attn = softmax(logits_scaled)
                attn = torch.softmax(logits_scaled, dim=0)                   # [K_actual]
                # output[t, h, :] = sum_k attn[k] * Kc_valid[k, :]
                out_vec = attn @ Kc_valid                                    # [head_dim_ckv]
                output[t, h, :] = out_vec.to(torch.bfloat16)

        return output, lse

import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: compute logits_scaled[t, h, k] = (q_nope[t, h] · Kc_cache[tok_idx, :]) + (q_pe[t, h] · Kp_cache[tok_idx, :])
# We will not precompute Kc_all/Kp_all; instead, for each k, we decode tok_idx and load from the original caches.
@triton.jit
def _compute_logits_from_indices_kernel(
    q_nope_ptr,         # *float32 [num_tokens, num_qo_heads, head_dim_ckv]
    q_pe_ptr,           # *float32 [num_tokens, num_qo_heads, head_dim_kpe]
    ckv_cache_ptr,      # *float32 [num_pages, page_size, head_dim_ckv]
    kpe_cache_ptr,      # *float32 [num_pages, page_size, head_dim_kpe]
    sparse_indices_ptr, # *int32   [num_tokens, topk]
    out_ptr,            # *float32 [num_tokens, num_qo_heads, topk]
    # dimensions
    num_tokens,         # int32
    num_qo_heads,       # int32
    head_dim_ckv,       # int32 (e.g., 512)
    head_dim_kpe,       # int32 (e.g., 64)
    topk,               # int32 (e.g., 2048)
    # reduction block sizes
    BLOCK_Q: tl.constexpr,   # 128
    BLOCK_KC: tl.constexpr,  # 128
    BLOCK_KP: tl.constexpr,  # 64
):
    # Grid: (num_tokens * num_qo_heads, topk)
    pid_m = tl.program_id(0)  # over tokens * heads
    k = tl.program_id(1)      # over topk indices

    # Map pid_m -> (t, h)
    t = pid_m // num_qo_heads
    h = pid_m % num_qo_heads
    if t >= num_tokens:
        return

    # Load idx from sparse_indices[t, k]
    idx = tl.load(sparse_indices_ptr + t * topk + k)  # int32
    # Valid if idx != -1
    is_valid = idx != -1

    # Load q vectors
    qn = tl.load(q_nope_ptr + t * (num_qo_heads * head_dim_ckv) + h * head_dim_ckv + tl.arange(0, BLOCK_Q),
                 mask=tl.arange(0, BLOCK_Q) < head_dim_ckv, other=0.0).to(tl.float32)
    qp = tl.load(q_pe_ptr + t * (num_qo_heads * head_dim_kpe) + h * head_dim_kpe + tl.arange(0, BLOCK_KP),
                 mask=tl.arange(0, BLOCK_KP) < head_dim_kpe, other=0.0).to(tl.float32)

    # Compute tok_idx for ckv/kpe: tok_idx = idx if valid else 0 (dummy)
    tok_idx = tl.where(is_valid, idx, 0)

    # Compute dot products only if valid; else contribute 0
    contrib1 = 0.0
    contrib2 = 0.0

    # Dot q_nope[t, h, :] · ckv_cache[tok_idx, :]
    for off in range(0, head_dim_ckv, BLOCK_Q):
        qn_chunk = tl.load(q_nope_ptr + t * (num_qo_heads * head_dim_ckv) + h * head_dim_ckv + off + tl.arange(0, BLOCK_Q),
                           mask=tl.arange(0, BLOCK_Q) < (head_dim_ckv - off), other=0.0).to(tl.float32)
        Kc_chunk = tl.load(ckv_cache_ptr + tok_idx * (page_size * head_dim_ckv) + off + tl.arange(0, BLOCK_Q),
                           mask=tl.arange(0, BLOCK_Q) < (head_dim_ckv - off), other=0.0).to(tl.float32)
        # Zero out when invalid
        qn_chunk = tl.where(is_valid, qn_chunk, 0.0)
        Kc_chunk = tl.where(is_valid, Kc_chunk, 0.0)
        contrib1 += tl.sum(qn_chunk * Kc_chunk, axis=0)

    # Dot q_pe[t, h, :] · kpe_cache[tok_idx, :]
    for off in range(0, head_dim_kpe, BLOCK_KP):
        qp_chunk = tl.load(q_pe_ptr + t * (num_qo_heads * head_dim_kpe) + h * head_dim_kpe + off + tl.arange(0, BLOCK_KP),
                           mask=tl.arange(0, BLOCK_KP) < (head_dim_kpe - off), other=0.0).to(tl.float32)
        Kp_chunk = tl.load(kpe_cache_ptr + tok_idx * (page_size * head_dim_kpe) + off + tl.arange(0, BLOCK_KP),
                           mask=tl.arange(0, BLOCK_KP) < (head_dim_kpe - off), other=0.0).to(tl.float32)
        # Zero out when invalid
        qp_chunk = tl.where(is_valid, qp_chunk, 0.0)
        Kp_chunk = tl.where(is_valid, Kp_chunk, 0.0)
        contrib2 += tl.sum(qp_chunk * Kp_chunk, axis=0)

    # Store logits_scaled[t, h, k] only if valid; else store 0.0
    out_val = contrib1 + contrib2
    out_val = tl.where(is_valid, out_val, 0.0)
    tl.store(out_ptr + t * (num_qo_heads * topk) + h * topk + k, out_val, mask=is_valid)
