        idx_k = tl.load(Sparse_ptr + t * topk + group_start + k)
        valid = idx_k != -1

        # Compute contrib = sum_d Qn[t,h,d] * Kc_all[idx_k,d] + sum_dp Qp[t,h,dp] * Kp_all[idx_k,dp]
        # First part over head_dim_ckv
        acc1 = 0.0
        d = 0
        BLOCK_D = 128
        while d < head_dim_ckv:
            offs = d + tl.arange(0, BLOCK_D)
            mask_d = (offs < head_dim_ckv) & valid
            qn_vals = tl.load(Qn_ptr + t * head_dim_ckv + h * 1 + offs, mask=mask_d, other=0.0)
            kc_vals = tl.load(Kc_all_ptr + idx_k * head_dim_ckv + offs, mask=mask_d, other=0.0)
            acc1 += tl.sum(qn_vals * kc_vals, axis=0)
            d += BLOCK_D

        # Second part over head_dim_kpe
        acc2 = 0.0
        dp = 0
        BLOCK_DP = 64
        while dp < head_dim_kpe:
            offs = dp + tl.arange(0, BLOCK_DP)
            mask_dp = (offs < head_dim_kpe) & valid
            qp_vals = tl.load(Qp_ptr + t * head_dim_kpe + h * 1 + offs, mask=mask_dp, other=0.0)
            kp_vals = tl.load(Kp_all_ptr + idx_k * head_dim_kpe + offs, mask=mask_dp, other=0.0)
            acc2 += tl.sum(qp_vals * kp_vals, axis=0)
            dp += BLOCK_DP

        acc += (acc1 + acc2)
        k += 1

    scaled = acc * sm_scale
    tl.store(Logits_ptr + t * (num_qo_heads * segments) + h * segments + j, scaled)


# Kernel 2: compute lse[t, h] = logsumexp over segments (each segment length=32)
@triton.jit
def compute_lse_per_group_kernel(
    Logits_ptr, LSE_ptr,
    num_tokens, num_qo_heads, segments,
):
    t = tl.program_id(0)
    h = tl.program_id(1)

    # Max across segments
    m = -float("inf")
    j = 0
    while j < segments:
        val = tl.load(Logits_ptr + t * (num_qo_heads * segments) + h * segments + j)
        m = tl.maximum(m, val)
        j += 1

    # Sum exp(logits - m)
    sum_exp = 0.0
    j = 0
    while j < segments:
        val = tl.load(Logits_ptr + t * (num_qo_heads * segments) + h * segments + j)
        sum_exp += tl.exp(val - m)
        j += 1

    lse = tl.log(sum_exp) + m
    tl.store(LSE_ptr + t * num_qo_heads + h, lse)


# Kernel 3: compute softmax per group (segments)
@triton.jit
def compute_softmax_per_group_kernel(
    Logits_ptr, LSE_ptr, Attn_ptr,
    num_tokens, num_qo_heads, segments,
):
    t = tl.program_id(0)
    h = tl.program_id(1)

    denom = 0.0
    j = 0
    while j < segments:
        val = tl.load(Logits_ptr + t * (num_qo_heads * segments) + h * segments + j)
        denom += tl.exp(val - tl.load(LSE_ptr + t * num_qo_heads + h))
        j += 1

    inv_denom = 1.0 / denom
    j = 0
    while j < segments:
        val = tl.load(Logits_ptr + t * (num_qo_heads * segments) + h * segments + j)
        lse = tl.load(LSE_ptr + t * num_qo_heads + h)
        attn_j = tl.exp(val - lse) * inv_denom
        tl.store(Attn_ptr + t * (num_qo_heads * segments) + h * segments + j, attn_j)
        j += 1
