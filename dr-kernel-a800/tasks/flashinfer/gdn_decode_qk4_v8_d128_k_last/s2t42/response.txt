import torch
import triton
import triton.language as tl


@triton.jit
def triton_update_kernel(
    q_ptr,             # *bfloat16, shape [B, QH, K]
    k_ptr,             # *bfloat16, shape [B, KH, K]
    v_ptr,             # *bfloat16, shape [B, VH, V]
    state_ptr,         # *float32, shape [B, H, V, K]
    g_ptr,             # *float32, shape [B, 1, H] (we use g[b, 0, h] = g[b, h])
    beta_ptr,          # *float32, shape [B, 1, H] (we use beta[b, 0, h] = beta[b, h])
    out_ptr,           # *float32, shape [B, H]
    B: tl.constexpr,   # batch size
    H: tl.constexpr,   # number of heads
    K: tl.constexpr,   # K dimension
    V: tl.constexpr,   # V dimension
    QH: tl.constexpr,  # num_q_heads (assumed 4)
    KH: tl.constexpr,  # num_k_heads (assumed 4)
    VH: tl.constexpr,  # num_v_heads (assumed 8)
    scale: tl.float32, # scalar
):
    # We launch one program per (b, h)
    pid = tl.program_id(axis=0)
    b = pid // H
    h = pid % H

    # For given h, q_idx = h, k_idx = h, v_idx = h due to repeat_interleave mapping (0->0,1->1,2->2,3->3)
    q_base = b * (QH * K) + h * K
    k_base = b * (KH * K) + h * K
    v_base = b * (VH * V) + h * V

    # Load q_h, k_h, v_h
    q_vec = tl.zeros([K], dtype=tl.float32)
    for j in range(0, K):
        q_j = tl.cast(tl.load(q_ptr + q_base + j), tl.float32)
        q_vec[j] = q_j

    k_vec = tl.zeros([K], dtype=tl.float32)
    for j in range(0, K):
        k_j = tl.cast(tl.load(k_ptr + k_base + j), tl.float32)
        k_vec[j] = k_j

    v_vec = tl.zeros([V], dtype=tl.float32)
    for j in range(0, V):
        v_j = tl.cast(tl.load(v_ptr + v_base + j), tl.float32)
        v_vec[j] = v_j

    # Load g and beta for this (b, h)
    g_val = tl.load(g_ptr + b * H + h)   # g[b, 0, h] -> g[b, h]
    beta_val = tl.load(beta_ptr + b * H + h)

    # Load state_old: shape [V, K] (row-major: v_idx * K + k_idx)
    state_old = tl.zeros([V, K], dtype=tl.float32)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            state_old[v_idx, k_idx] = tl.cast(tl.load(state_ptr + row_base + k_idx), tl.float32)

    # old_v = k_h @ (g * state_old)
    old_v = tl.zeros([K], dtype=tl.float32)
    for j in range(0, K):
        sum_val = tl.zeros((), dtype=tl.float32)
        for v_idx in range(0, V):
            # state_old[v_idx, j] is a scalar; g_val is scalar
            sum_val += state_old[v_idx, j] * g_val
        old_v[j] = tl.sum(k_vec * sum_val)  # k_vec[j] * sum_val? We need k_h @ (g * state_old)
        # Correct approach: for each j, old_v[j] = sum_v state_old[:, j] * g_val, then dot with k_vec
        # That is: old_v[j] = sum_v state_old[v, j] * g_val, where g_val is scalar multiplier per column. Then dot with k_vec.
        # Let's fix:
        # We need old_v[j] = sum_v state_old[v, j] * g_val, then multiply by k_vec[j]?
        # Actually, per formula, old_v[j] = sum_v state_old[v, j] * g_val. That's wrong; original code does:
        # old_state = g * state_old (elementwise), then old_v = k_h @ old_state
        # So old_v[j] = sum_v state_old[v, j] * g_val
        # But that would be wrong; correct is: new_v uses beta * v_h + (1 - beta) * old_v where old_v = k_h @ (g * state_old)
        # We must compute k_h @ (g * state_old) which is a dot of k_h with each column multiplied by g_val.
        # Let's do it correctly:
        # First, define g_state = state_old * g_val (elementwise)
        g_state = state_old * g_val
        old_v_j = tl.zeros((), dtype=tl.float32)
        for v_idx in range(0, V):
            old_v_j += g_state[v_idx, j]
        old_v[j] = tl.sum(k_vec * old_v_j)  # k_j * (sum_v g_state[v, j])

    # new_v = beta * v_h + (1 - beta) * old_v
    new_v = tl.zeros([V], dtype=tl.float32)
    for v_idx in range(0, V):
        new_v[v_idx] = beta_val * v_vec[v_idx] + (1.0 - beta_val) * old_v[v_idx]

    # state_remove = k_h @ old_v, state_update = k_h @ new_v (scalars)
    state_remove = tl.zeros((), dtype=tl.float32)
    for j in range(0, K):
        state_remove += old_v[j] * k_vec[j]

    state_update = tl.zeros((), dtype=tl.float32)
    for j in range(0, K):
        state_update += new_v[j] * k_vec[j]

    # Update h_state_new = (g * state_old) - state_remove + state_update
    h_state_new = (state_old * g_val) - state_remove + state_update  # broadcasting of scalar and [V,K] tensor

    # Compute output scalar: output = scale * q_h @ h_state_new
    q_h = q_vec  # already [K]
    out_scalar = tl.zeros((), dtype=tl.float32)
    for j in range(0, K):
        # For each column j, sum over rows
        col_sum = tl.sum(h_state_new[:, j])
        out_scalar += q_h[j] * col_sum

    out_scalar = out_scalar * scale

    # Store output
    tl.store(out_ptr + b * H + h, out_scalar)


@triton.jit
def triton_invsqrt_scale_kernel(
    K,                # int32
    out_scale_ptr     # *float32, single element
):
    # Compute 1/sqrt(K) and store
    scale = 1.0 / tl.sqrt(tl.cast(K, tl.float32))
    tl.store(out_scale_ptr, scale)

        """
        assert q.is_cuda and k.is_cuda and v.is_cuda and state.is_cuda, "All tensors must be on CUDA for Triton kernels."
        assert q.shape[0] == 1 and k.shape[0] == 1 and v.shape[0] == 1, "Only batch_size=1 supported in this example."
        B, T, QH, K = q.shape
        _, _, KH, _ = k.shape
        _, _, VH, V = v.shape
        # Since T == 1 in provided get_inputs, we squeeze it
        # Derive h mapping: repeat_interleave(2) for q and k (from 4 -> 8), v remains 8
        # We'll use h directly for q,k,v because repeat_interleave maps original indices 0->0,1->1,2->2,3->3.
        H = VH  # number of heads is num_v_heads, equals 8 in provided inputs

        # Ensure contiguous
        q = q.contiguous()
        k = k.contiguous()
        v = v.contiguous()
        state = state.contiguous()

        # Compute scale in Triton (or host scalar). We use Triton for the heavy kernel, and host scalar for scale to keep things simple.
        scale_tensor = torch.empty(1, dtype=torch.float32, device=q.device)
        triton_invsqrt_scale_kernel[(1,)](K, scale_tensor)

        # Prepare output tensor
        out = torch.empty(B * H, dtype=torch.float32, device=q.device)  # [B*H] linearized

        # Launch Triton kernel for each (b,h)
        grid = (B * H,)
        triton_update_kernel[grid](
            q, k, v, state,  # inputs
            a.float().squeeze(1), dt_bias.float(),  # g and beta computation can be precomputed in PyTorch since Triton host is not required
            out,

        # Reshape output to (B, H) float32 then cast to bfloat16 for return signature consistency
        out = out.view(B, H).to(torch.bfloat16).unsqueeze(1)  # unsqueeze dim-1 to (B, 1, H)
