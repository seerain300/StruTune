진행 과정
1) 원래 논리를 이해하기
- run 함수는 각 (batch b, head h)에 대해:
  - g = exp(-exp(A_log) * softplus(a + dt_bias))와 beta = sigmoid(b)을 계산합니다. 그다음
  - q_h, k_h, v_h 각각 K=128 벡터입니다. state[b, h]는 [V, K] = [128, 128].
  - old_v = k_h @ old_state, new_v = beta * v_h + (1 - beta) * old_v, state_remove = k_h @ old_v, state_update = k_h @ new_v, 그리고
    new_state[b, h] = g * old_state - state_remove + state_update (elementwise add/subtract).
  - output[b, h] = scale * (q_h @ new_state_col0), 여기서 new_state_col0은 new_state[:, 0]인 128차원 벡터의 합입니다.

2) Triton kernels
- exp_kernel: x_ptr에서 y_ptr로 exp 계산
- softplus_kernel: y = log(1 + exp(x))의 안정적 버전
- sigmoid_kernel: y = 1 / (1 + exp(-x))
- matvec_kernel: y = k @ x, x는 [K, V] 배열, k는 [K] 벡터, y는 [V] 벡터
- dot_kernel: out[0] = sum_i q[i] * x[i], 전역적으로 atomic_add를 통해 집계
- sqrt_scalar_kernel: out[0] = 1 / sqrt(K)

3) forward flow
- A_log, a + dt_bias, b: 각각의 어레이를 텐서로 만든 후 Triton kernels로 softplus, exp, sigmoid 계산
- q_exp, k_exp: q와 k를 num_v_heads/num_q_heads 비율만큼 repeat_interleave (텐서 연산 허용)
- state_f32: float32로 변환하여 시작 상태를 사용
- For b in [B], h in [H]:
  - old_state = state[b, h] (float32, [V, K])
  - old_v = matvec_kernel(old_state.view(K, V).contiguous(), k_exp[b, h]) -> [V]
  - new_v = beta[h] * v[b, h] + (1 - beta[h]) * old_v (torch, allowed as data op)
  - state_remove = matvec_kernel(old_v, k_exp[b, h]) -> [V]
  - state_update = matvec_kernel(new_v, k_exp[b, h]) -> [V]
  - new_state[b, h] = g[h] * old_state - state_remove + state_update (torch)
  - new_state_col0 = new_state[b, h][:, 0] -> [V]
  - out_scalar_buf = float32 1-element buffer; dot_kernel(q_h.float(), new_state_col0.float(), out_scalar_buf)
  - scale 처리: scale이 없으면 sqrt_kernel(128)로 1/sqrt(128) 계산 후 곱함
  - output[b, 0, h, 0] = out_scalar_buf[0] * scale (torch write; 단순히 저장에 사용)

4) state와 output dtype
- state: float32, shape [B, H, V, K], 반환 시도
- output: bfloat16, shape [B, 1, H, V], 반환 시도

이 구현은 Triton으로 모든 수학 연산을 수행하도록 합니다. 상태 갱신과 최종 출력 저장은 텐서의 데이터 변동으로 간주되므로 torch를 사용합니다.

@triton.jit
def exp_kernel(inp_ptr, out_ptr, N: tl.constexpr):
    # Compute out[i] = exp(inp[i]) for i in [0, N). One element per program.
    pid = tl.program_id(axis=0)
    i = pid
    x = tl.load(inp_ptr + i, mask=i < N, other=0.0)
    y = tl.exp(x)
    tl.store(out_ptr + i, y, mask=i < N)


@triton.jit
def softplus_kernel(x_ptr, out_ptr, N: tl.constexpr):
    # Compute softplus(x) = log(1 + exp(x)) in a stable manner.
    pid = tl.program_id(axis=0)
    i = pid
    x = tl.load(x_ptr + i, mask=i < N, other=0.0)

    y = tl.where(pos, pos_val, neg_val)

    v_offsets = v_start + tl.arange(0, BLOCK_V)

            k_val = k_chunk[kk]

            x_vals = tl.load(x_ptr + k_idx * V + v_offsets, mask=v_offsets < V, other=0.0)
            y_acc += k_val * x_vals
    tl.store(y_ptr + v_offsets, y_acc, mask=v_offsets < V)


@triton.jit
def dot_kernel(q_ptr, x_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Compute out[0] = sum_i q[i] * x[i].
    Each program handles a block of N and atomic_adds into out_ptr[0].
    """
    pid = tl.program_id(axis=0)
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    q = tl.load(q_ptr + offsets, mask=offsets < N, other=0.0)
    x = tl.load(x_ptr + offsets, mask=offsets < N, other=0.0)
    partial = tl.sum(q * x, axis=0)
    tl.atomic_add(out_ptr, partial)

    inv = 1.0 / tl.sqrt(K)

@triton.jit
def softplus_kernel(x_ptr, out_ptr, N: tl.constexpr):
    # Compute softplus(x) = log(1 + exp(x)) stably
    for i in range(N):

        if x > 0.0:

        else:

    v_start = pid * BLOCK_V
    v_offsets = v_start + tl.arange(0, BLOCK_V)
    y_acc = tl.zeros((BLOCK_V,), dtype=tl.float32)
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_chunk = tl.load(k_ptr + k_offsets, mask=k_offsets < K, other=0.0)  # [BLOCK_K]
        for kk in range(0, BLOCK_K):
            k_val = k_chunk[kk]
            k_idx = k_start + kk
            x_vals = tl.load(x_ptr + k_idx * V + v_offsets, mask=v_offsets < V, other=0.0)
            y_acc += k_val * x_vals
    tl.store(y_ptr + v_offsets, y_acc, mask=v_offsets < V)


@triton.jit
def dot_kernel(q_ptr, x_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    # Compute out[0] = sum_i q[i] * x[i]
    for start in range(0, N, BLOCK):
        offsets = start + tl.arange(0, BLOCK)
        q = tl.load(q_ptr + offsets, mask=offsets < N, other=0.0)
        x = tl.load(x_ptr + offsets, mask=offsets < N, other=0.0)
        partial = tl.sum(q * x, axis=0)
        tl.atomic_add(out_ptr, partial)


@triton.jit
def sqrt_scalar_kernel(K: tl.constexpr, out_ptr):
    inv = 1.0 / tl.sqrt(K)
    tl.store(out_ptr, inv)


@triton.jit
def write_elem_kernel(dst_ptr, src_ptr, index: tl.constexpr):
    # Write single element: dst[index] = src[0]
    val = tl.load(src_ptr)  # scalar from src_ptr[0]
    # dst_ptr is a 1D pointer to output tensor; write to element at index
    tl.store(dst_ptr + index, val)
