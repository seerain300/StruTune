@triton.jit
def _gate_g_and_beta_kernel(a_ptr, dt_bias_ptr, A_log_ptr, b_ptr, g_ptr, beta_ptr,
                            B: tl.constexpr, H: tl.constexpr):
    # Grid: (B, H)
    b_idx = tl.program_id(0)
    h = tl.program_id(1)

    # Load scalars
    a_val = tl.load(a_ptr + b_idx * H + h)     # a[b, h]
    dt_val = tl.load(dt_bias_ptr + h)          # dt_bias[h]
    A_val = tl.load(A_log_ptr + h)             # A_log[h]
    b_val = tl.load(b_ptr + b_idx * H + h)     # b[b, h]

    # Compute a_plus = a + dt
    a_plus = a_val + dt_val

    # softplus(a_plus) = log(1 + exp(a_plus))
    sp = tl.log(1.0 + tl.exp(a_plus))

    # g = exp(-exp(A_log) * softplus(a_plus))
    g_val = tl.exp(-tl.exp(A_val) * sp)

    # beta = 1 / (1 + exp(-b))
    beta_val = 1.0 / (1.0 + tl.exp(-b_val))

    # Store
    tl.store(g_ptr + b_idx * H + h, g_val)
    tl.store(beta_ptr + b_idx * H + h, beta_val)
