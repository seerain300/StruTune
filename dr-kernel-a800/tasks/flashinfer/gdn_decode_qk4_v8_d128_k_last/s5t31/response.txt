    """
    b = tl.program_id(0)
    h = tl.program_id(1)
    a_val = tl.load(a_ptr + b * stride_a_b + h * stride_a_h)
    dtb_val = tl.load(dt_bias_ptr + h)
    Alog_val = tl.load(A_log_ptr + h)
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(a_val + dtb_val))
    g_val = tl.exp(-tl.exp(Alog_val) * sp)
    tl.store(g_ptr + b * stride_a_b + h * stride_a_h, g_val)


@triton.jit
def _compute_beta_kernel(b_ptr, beta_ptr, H, stride_b_b, stride_b_h):
    """
    Compute beta[b,h] = sigmoid(b[b,1,h]) for all b,h.
    b_ptr: [B,H] flattened with given strides.
    beta_ptr: [B,H] flattened.
    """
    b = tl.program_id(0)
    h = tl.program_id(1)
    b_val = tl.load(b_ptr + b * stride_b_b + h * stride_b_h)
    beta_val = 1.0 / (1.0 + tl.exp(-b_val))
    tl.store(beta_ptr + b * stride_b_b + h * stride_b_h, beta_val)


@triton.jit
def _update_and_output_kernel(q_ptr, k_ptr, v_ptr, state_ptr, g_ptr, beta_ptr, out_ptr, new_state_ptr,
                               B, H, V, K, scale,
                               stride_q_b, stride_q_h, stride_k_b, stride_k_h,
                               stride_v_b, stride_v_h, stride_s_b, stride_s_h,
                               stride_o_b, stride_o_h,
                               stride_ns_b, stride_ns_h, stride_ns_v, stride_ns_k):
    """
    For each (b, h):
      - old_v = k_h @ state[b,h]         # reduce over K
      - new_v = beta[b,h] * v_h + (1 - beta[b,h]) * old_v
      - old_state = g[b,h] * state[b,h]
      - state_remove = k_h @ old_state   # reduce over K
      - state_update = k_h @ new_v       # reduce over K
      - new_state[b,h] = old_state - state_remove[:, None] + state_update[:, None]
      - output[b,h] = scale * (q_h @ new_state[b,h]) reduce over V
    """
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Load scalars
    g_val = tl.load(g_ptr + b * stride_q_b + h * stride_q_h)  # g[b,h]
    beta_val = tl.load(beta_ptr + b * stride_k_b + h * stride_k_h)  # beta[b,h]

    # Load vectors
    # q_h: [K]
    q_h = tl.load(q_ptr + b * stride_q_b + h * stride_q_h + tl.arange(0, K), mask=tl.arange(0, K) < K, other=0.0)
    # k_h: [K]
    k_h = tl.load(k_ptr + b * stride_k_b + h * stride_k_h + tl.arange(0, K), mask=tl.arange(0, K) < K, other=0.0)
    # v_h: [V]
    v_h = tl.load(v_ptr + b * stride_v_b + h * stride_v_h + tl.arange(0, V), mask=tl.arange(0, V) < V, other=0.0)

    # Load state[b,h] as [V,K]
    # Note: state is [B, H, V, K] with strides (stride_s_b, stride_s_h, stride_s_v, stride_s_k).
    # We access state at (b, h) slice: rows v in [0..V-1], cols k in [0..K-1]
    v_idx = tl.arange(0, V)[:, None]  # [V,1]
    k_idx = tl.arange(0, K)[None, :]  # [1,K]
    state_b_h = tl.load(
        state_ptr + b * stride_s_b + h * stride_s_h + v_idx * stride_s_v + k_idx * stride_s_k,
        mask=(v_idx < V) & (k_idx < K), other=0.0
    )  # [V,K]

    # Compute old_v = k_h @ state[b,h] over K -> scalar
    # Dot-reduce over K: sum_k k_h[k] * state_b_h[:, k] then sum over V
    dot_v = tl.sum(state_b_h * k_h[None, :], axis=1)  # [V]
    old_v = tl.sum(dot_v, axis=0)  # scalar

    # new_v = beta * v_h + (1 - beta) * old_v, broadcast to [V]
    new_v_vec = beta_val * v_h + (1.0 - beta_val) * old_v  # [V]

    # old_state = g_val * state[b,h]
    old_state_b_h = g_val * state_b_h  # [V,K]

    # state_remove = k_h @ old_state over K -> scalar
    dot_v_old = tl.sum(old_state_b_h * k_h[None, :], axis=1)  # [V]
    state_remove = tl.sum(dot_v_old, axis=0)  # scalar

    # state_update = k_h @ new_v_vec over K -> scalar
    dot_new = tl.sum(new_v_vec[None, :] * k_h[None, :], axis=1)  # [1,K] dot with [V] -> [1]
    state_update = tl.sum(dot_new, axis=0)  # scalar

    # new_state_b_h = old_state - state_remove[:, None] + state_update[:, None]
    # state_remove[:, None] broadcasts over K, state_update[:, None] broadcasts over K
    new_state_b_h = old_state_b_h - state_remove[:, None] + state_update[:, None]  # [V,K]

    # output[b,h] = scale * (q_h @ new_state_b_h) = scale * sum_{v,k} q_h[k] * new_state_b_h[v,k]
    dot_q = tl.sum(new_state_b_h * q_h[None, :], axis=1)  # [V]
    out_scalar = tl.sum(dot_q, axis=0)  # scalar
    tl.store(out_ptr + b * stride_o_b + h * stride_o_h, out_scalar * scale)

    # Store new_state[b,h] = new_state_b_h to [V,K]
    # We can store the matrix directly using pointer arithmetic
    # Note: new_state_ptr is [B, H, V, K] with strides (stride_ns_b, stride_ns_h, stride_ns_v, stride_ns_k)
    v_idx_store = tl.arange(0, V)[:, None]
    k_idx_store = tl.arange(0, K)[None, :]
    tl.store(
        new_state_ptr + b * stride_ns_b + h * stride_ns_h + v_idx_store * stride_ns_v + k_idx_store * stride_ns_k,
        new_state_b_h, mask=(v_idx_store < V) & (k_idx_store < K)
    )
