def _compute_g_kernel(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, B, H):
    """
    Compute g_vec[b*H + h] = exp(-exp(A_log[h]) * softplus(a[b, 1, h] + dt_bias[h])) for b in [0..B-1], h in [0..H-1].
    a_ptr: [B, H] flattened (we pass a.index_select(2, range(H)).flatten())
    dt_bias_ptr: [H]
    A_log_ptr: [H]
    g_ptr: [B*H]
    """
    b = tl.program_id(0)
    h = tl.program_id(1)

    # a_ptr has shape [B*H], so we directly load a[b, h]
    a_val = tl.load(a_ptr + b * H + h)  # fp32
    dt_bias_val = tl.load(dt_bias_ptr + h)  # fp32
    A_log_val = tl.load(A_log_ptr + h)  # fp32

    # softplus(x) = log(1 + exp(x))  (PyTorch softplus default beta=1, threshold=20)

    beta_val = 1.0 / (1.0 + tl.exp(-b_val))
    tl.store(beta_ptr + b_idx * H + h, beta_val)
