    dt_bias_ptr,    # *f32, [H]
    a_ptr,          # *f32, [B*H]
    A_log_ptr,      # *f32, [H]
    g_out_ptr,      # *f32, [B*H]
    H: tl.constexpr,
    B: tl.constexpr,
):

    b = pid // H
