    dt_bias_ptr: [H] float32 (we pass a[b,1,h] + dt_bias[h] flattened)
    A_log_ptr:   [H] float32
    g_ptr:       [H] float32
    """
    h = tl.program_id(0)
    x = dt_bias_ptr[h] + a_ptr[h]  # pass a_ptr[h] as (a[b,1,h] + dt_bias[h])
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x))
    g_val = tl.exp(-tl.exp(A_log_ptr[h]) * sp)
    tl.store(g_ptr + h, g_val)


@triton.jit
def _compute_beta_kernel(b_ptr, beta_ptr, H: tl.constexpr):
    """
    Compute beta[h] = 1 / (1 + exp(-b_ptr[h])) for h in [0..H-1]
    b_ptr:     [H] float32 (we pass b[b,1,h] flattened)
    beta_ptr:  [H] float32
    """
    h = tl.program_id(0)
    b_val = b_ptr[h]
    beta_val = 1.0 / (1.0 + tl.exp(-b_val))
    tl.store(beta_ptr + h, beta_val)


@triton.jit
def _update_all_kernel(q_ptr, k_ptr, v_ptr, state_ptr, g_ptr, beta_ptr,
                        new_state_ptr, out_ptr,
                        B, K, scale,
                        V: tl.constexpr, K_CONST: tl.constexpr):
    """
    For each (b,h), compute for all m in V:
      - old_v = sum_k k[b,h,k] @ state[b,h,m,:] (reduce over K)
      - new_v = beta[h] * v[b,h,m] + (1 - beta[h]) * old_v
      - k_sum = sum_k k[b,h,k]
      - delta_m = -old_v + (beta[h]*v[b,h,m] + (1-beta[h])*old_v) * k_sum
      - new_state[b,h,m,:] = (g[h] - 1) * state[b,h,m,:] + delta_m
      - output[b,h,m] = scale * q[b,h,:] @ new_state_row
    Writes:
      - new_state_ptr: flattened [B*H*V*K_CONST] float32
      - out_ptr:       [B*H*V] float32
    Assumes K_CONST=128, V_CONST=8, H unspecified; we pass H via program_id(1).
    """
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Load params
    g_val = tl.load(g_ptr + h)        # float32
    beta_val = tl.load(beta_ptr + h)  # float32

    # Base pointers
    # q: [B,4,K] -> linearized
    q_base = q_ptr + b * (4 * K)
    # k: [B,4,K] -> linearized
    k_base = k_ptr + b * (4 * K)
    # v: [B,8,V] -> linearized
    v_base = v_ptr + b * (8 * V)
    # state: [B,8,V,K_CONST] -> linearized
    state_base = state_ptr + b * (8 * V * K_CONST)

    # Precompute k_sum over K
    k_sum = 0.0
    for k_idx in tl.static_range(4):
        k_k_ptr = k_base + k_idx * K_CONST
        k_h = tl.load(k_k_ptr + tl.arange(0, K_CONST), mask=tl.arange(0, K_CONST) < K_CONST, other=0.0)  # [K_CONST]
        k_sum += tl.sum(k_h)

    # Loop over m in V
    for m in tl.static_range(V):
        # old_v = sum_k k @ state_row
        old_v = 0.0
        for k_idx in tl.static_range(4):
            k_k_ptr = k_base + k_idx * K_CONST
            k_h = tl.load(k_k_ptr + tl.arange(0, K_CONST), mask=tl.arange(0, K_CONST) < K_CONST, other=0.0)  # [K_CONST]
            row_start = h * V * K_CONST + m * K_CONST
            row = tl.load(state_base + row_start + tl.arange(0, K_CONST),
                          mask=tl.arange(0, K_CONST) < K_CONST, other=0.0)  # [K_CONST]
            old_v += tl.sum(k_h * row)

        # Load v[b,h,m]
        v_row = tl.load(v_base + m * V + tl.arange(0, V), mask=tl.arange(0, V) < V, other=0.0)  # [V]
        v_m = v_row[0]  # scalar

        # new_v scalar for this m
        new_v_scalar = beta_val * v_m + (1.0 - beta_val) * old_v

        # delta for this m
        delta_m = -old_v + (beta_val * v_m + (1.0 - beta_val) * old_v) * k_sum
        # new_state row update
        row_start = h * V * K_CONST + m * K_CONST
        row_old = tl.load(state_base + row_start + tl.arange(0, K_CONST),
                          mask=tl.arange(0, K_CONST) < K_CONST, other=0.0)  # [K_CONST]
        new_row = (g_val - 1.0) * row_old + delta_m  # [K_CONST]
        tl.store(new_state_ptr + b * (H * V * K_CONST) + (h * V + m) * K_CONST + tl.arange(0, K_CONST),
                 new_row, mask=tl.arange(0, K_CONST) < K_CONST)

        # output[b,h,m] = scale * q[b,h,:] @ new_row
        q_h = tl.load(q_base + h * K_CONST + tl.arange(0, K_CONST),
                      mask=tl.arange(0, K_CONST) < K_CONST, other=0.0)  # [K_CONST]
        out_val = tl.sum(q_h * new_row) * scale
        tl.store(out_ptr + b * (H * V) + h * V + m, out_val)
