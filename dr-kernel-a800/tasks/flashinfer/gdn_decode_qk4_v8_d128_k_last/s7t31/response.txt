import torch
import triton
import triton.language as tl


@triton.jit
def _compute_g_and_beta_kernel(a_ptr, dt_bias_ptr, b_ptr, A_log_ptr,
                                g_ptr, beta_ptr,
                                B: tl.constexpr, H: tl.constexpr):
    # One program per (b,h)
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H
    # Load scalars
    a_val = tl.load(a_ptr + b * H + h)  # a[b, h]  -> shape [B, H] in PyTorch but indexing here by b,H
    dt_val = tl.load(dt_bias_ptr + h)   # dt_bias[h] -> shape [H]
    A_val = tl.load(A_log_ptr + h)      # A_log[h]   -> shape [H]
    # Compute x = a + dt_bias
    x = a_val + dt_val
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x))
    # g = exp(-exp(A_log) * softplus(a + dt_bias))
    g = tl.exp(-tl.exp(A_val) * sp)
    # beta = sigmoid(b)
    b_val = tl.load(b_ptr + b * H + h)  # b[b, h] -> shape [B, H]
    beta = 1.0 / (1.0 + tl.exp(-b_val))
    # Store
    tl.store(g_ptr + b * H + h, g)
    tl.store(beta_ptr + b * H + h, beta)


@triton.jit
def _vec_matmul_flat(k_ptr, state_flat_ptr, out_ptr,
                     K: tl.constexpr, V: tl.constexpr):
    # out[k] = sum_{v=0..V-1} k[v] * state[v, k]
    for k in tl.static_range(0, K):
        acc = 0.0
        for m in tl.static_range(0, V * K):  # iterate over flattened M = V*K
            v = m // K
            kk = m % K
            if kk == k:
                k_val = tl.load(k_ptr + v)      # k[v]
                state_val = tl.load(state_flat_ptr + m)  # state[v, k]
                acc += k_val * state_val
        tl.store(out_ptr + k, acc)


@triton.jit
def _vec_scalar_k_vec(vec_ptr, k_ptr, out_ptr, N: tl.constexpr):
    # scalar = sum_{i=0..N-1} vec[i] * k[i]
    acc = 0.0
    for i in tl.static_range(0, N):
        vec_i = tl.load(vec_ptr + i)
        k_i = tl.load(k_ptr + i)
        acc += vec_i * k_i
    tl.store(out_ptr, acc)


@triton.jit
def _output_scalar_kernel(q_ptr, state_flat_ptr, out_ptr, scale_ptr,
                           K: tl.constexpr, V: tl.constexpr):
    # out = scale * sum_{k=0..K-1} q[k] * state[k], where state is flattened [V*K] => state[k] = state[k, 0]
    # Note: We only use the k=0 slice of state for simplicity here; if you need full q@state across K, revise accordingly.
    scale = tl.load(scale_ptr)
    acc = 0.0
    for k in tl.static_range(0, K):
        qk = tl.load(q_ptr + k)
        state_elem = tl.load(state_flat_ptr + k)  # corresponds to state[k, 0] flattened
        acc += qk * state_elem
    tl.store(out_ptr, acc * scale)


@triton.jit
def _sqrt_scale_kernel(scale_ptr, K: tl.constexpr):
    scale = 1.0 / tl.sqrt(K)
    tl.store(scale_ptr, scale)


class ModelNew(torch.nn.Module):
    def forward(self, q, k, v, state, A_log, a, dt_bias, b, scale):
        # All computations are performed inside Triton kernels; host code prepares tensors and launches kernels.
        assert q.dim() == 4 and k.dim() == 4 and v.dim() == 4 and state.dim() == 4
        assert q.shape[1] == 1 and k.shape[1] == 1 and v.shape[1] == 1 and state.shape[1] == 1
        B, _, Hq, K = q.shape
        _, _, Hk, _ = k.shape
        _, _, Hv, V = v.shape
        _, _, _, _ = state.shape  # state: [B, 1, Hv, V]
        # Provided shapes match: Hq=4, Hk=4, Hv=8, K=V=128
        # For generality, we use H=Hv, and assume Hq==Hk==Hv (as in reference).
        assert Hq == Hv and Hk == Hv, "Mismatch in head counts."
        Hv = Hq  # use q's head count for consistency

        # Prepare pointers and ensure contiguous float32
        device = q.device
        q_f = q.squeeze(1).to(torch.float32).contiguous()  # [B, Hq, K]
        k_f = k.squeeze(1).to(torch.float32).contiguous()  # [B, Hk, K]
        v_f = v.squeeze(1).to(torch.float32).contiguous()  # [B, Hv, V]
        state_f = state.to(torch.float32).contiguous()     # [B, Hv, V, K]
        A_log_f = A_log.to(torch.float32).contiguous()     # [Hv]
        a_f = a.squeeze(1).to(torch.float32).contiguous()  # [B, Hv]
        dt_bias_f = dt_bias.to(torch.float32).contiguous() # [Hv]
        b_f = b.squeeze(1).to(torch.float32).contiguous()  # [B, Hv]

        # Allocate outputs
        g_out = torch.empty(B * Hv, device=device, dtype=torch.float32)
        beta_out = torch.empty(B * Hv, device=device, dtype=torch.float32)
        output_f = torch.empty(B * Hv, device=device, dtype=torch.float32)
        new_state_f = torch.empty((B, Hv, V, K), device=device, dtype=torch.float32)

        # Launch g and beta computation kernel
        _compute_g_and_beta_kernel[(B * Hv,)](a_f, dt_bias_f, b_f, A_log_f,
                                              g_out, beta_out,
                                              B=B, H=Hv)

        # Compute scale = 1/sqrt(K) in Triton
        scale_buf = torch.empty(1, device=device, dtype=torch.float32)
        _sqrt_scale_kernel[(1,)](scale_buf, K=K)

        # Process each (b, h) pair: update state and compute output
        for b_idx in range(B):
            for h_idx in range(Hv):
                # Current head: h = h_idx
                # Prepare vectors
                k_vec = k_f[b_idx, h_idx]  # [K]
                v_vec = v_f[b_idx, h_idx]  # [V]
                state_mat = state_f[b_idx, h_idx]  # [V, K]
                q_vec = q_f[b_idx, h_idx]  # [K]

                # 1) old_v = k @ state_old (vector [K])
                old_v = torch.empty(K, device=device, dtype=torch.float32)
                _vec_matmul_flat[(1,)](k_vec, state_mat.reshape(-1), old_v,
                                       K=K, V=V)

                # 2) beta and g
                # Retrieve scalar beta and g using g_out[b_idx*Hv + h_idx]
                g_val = g_out[b_idx * Hv + h_idx]
                beta_val = beta_out[b_idx * Hv + h_idx]

                # 3) new_v = beta * v + (1 - beta) * old_v
                new_v = beta_val * v_vec + (1.0 - beta_val) * old_v  # [V], float32

                # 4) state_remove = k @ old_v (scalar)
                state_remove = torch.empty((), device=device, dtype=torch.float32)
                _vec_scalar_k_vec[(1,)](k_vec, old_v, state_remove, N=K)

                # 5) state_update = k @ new_v (scalar)
                state_update = torch.empty((), device=device, dtype=torch.float32)
                _vec_scalar_k_vec[(1,)](k_vec, new_v, state_update, N=K)

                # 6) new_state = g * state_old - state_remove + state_update
                state_old_flat = state_mat.reshape(-1)  # [V*K]
                new_state_row = (g_val * state_old_flat) - state_remove + state_update  # [V*K]
                new_state_f[b_idx, h_idx] = new_state_row.reshape(V, K)

                # 7) output = scale * (q @ new_state)
                # Flatten q_vec into [K] and new_state_row into [V*K]
                out_val = torch.empty((), device=device, dtype=torch.float32)
                _output_scalar_kernel[(1,)](q_vec, new_state_row, out_val, scale_buf,
                                             K=K, V=V)
                output_f[b_idx * Hv + h_idx] = out_val

        # Return output in bfloat16 and new_state (float32)
        output_bf16 = output_f.view(B, Hv).unsqueeze(1).to(torch.bfloat16)  # [B, 1, Hv]
        return output_bf16, new_state_f

    tl.store(beta_ptr + b * H + h, beta)


@triton.jit
def _vec_matmul_tile_vec(k_ptr, state_flat_ptr, out_ptr,
                         K: tl.constexpr):
    # out[k] = sum_{v=0..V-1} k[v] * state[v, k]
    # We don't have V here because V is implied by state_flat_ptr length V*K.
    # This kernel expects state_flat_ptr to be of length V*K and out_ptr of length K.
    for k in tl.static_range(0, K):
        acc = 0.0
        # We need to iterate over all v to accumulate contributions to out[k]
        # We can do that by iterating over m in [0, V*K), deriving v = m // K, kk = m % K.
        # When kk == k, add k[v] * state[v, k].
        for m in tl.static_range(0, K * V):
            v = m // K
            kk = m % K
            if kk == k:
                k_val = tl.load(k_ptr + v)       # k[v]
                state_val = tl.load(state_flat_ptr + m)  # state[v, k]
                acc += k_val * state_val
        tl.store(out_ptr + k, acc)


@triton.jit
def _vec_matmul_scalar(vec_ptr, k_ptr, out_ptr, N: tl.constexpr):
    # scalar = sum_{i=0..N-1} vec[i] * k[i]
    acc = 0.0
    for i in tl.static_range(0, N):
        vec_i = tl.load(vec_ptr + i)
        k_i = tl.load(k_ptr + i)
        acc += vec_i * k_i
    tl.store(out_ptr, acc)


@triton.jit
def _sqrt_scale_kernel(scale_ptr, K: tl.constexpr):
    # Compute scale = 1/sqrt(K)
    scale = 1.0 / tl.sqrt(K)
    tl.store(scale_ptr, scale)
