import torch
import triton
import triton.language as tl


@triton.jit
def triton_gate_beta_kernel(
    A_log_ptr,         # *float32, shape [H]
    a_ptr,             # *float32, shape [B, H], we index by (b,h)
    dt_bias_ptr,       # *float32, shape [H]
    b_ptr,             # *float32, shape [B, H], we index by (b,h)
    g_ptr,             # *float32, shape [B, H]
    beta_ptr,          # *float32, shape [B, H]
    B: tl.constexpr,   # batch size
    H: tl.constexpr,   # num heads (num_v_heads)
):
    pid = tl.program_id(axis=0)
    b = pid // H
    h = pid % H
    # Load a[b,h] and dt_bias[h]
    a_val = tl.load(a_ptr + b * H + h)
    dt_val = tl.load(dt_bias_ptr + h)
    x = a_val + dt_val  # float32
    # softplus(x) = max(x, 0) + log(1 + exp(-|x|))
    abs_x = tl.abs(x)
    sp = tl.maximum(x, 0.0) + tl.log(1.0 + tl.exp(-abs_x))
    A_log_val = tl.load(A_log_ptr + h)
    g_val = tl.exp(-tl.exp(A_log_val) * sp)
    # beta = sigmoid(b_val)
    b_val = tl.load(b_ptr + b * H + h)
    beta_val = 1.0 / (1.0 + tl.exp(-b_val))
    # Store g and beta at [b, h]
    tl.store(g_ptr + b * H + h, g_val)
    tl.store(beta_ptr + b * H + h, beta_val)


@triton.jit
def triton_update_kernel(
    q_ptr, k_ptr, v_ptr, state_ptr, g_ptr, beta_ptr,
    output_ptr, new_state_ptr,
    B: tl.constexpr, H: tl.constexpr, V: tl.constexpr, K: tl.constexpr,
    scale: tl.float32,
):
    pid = tl.program_id(axis=0)
    b = pid // H
    h = pid % H

    # Construct contiguous base vectors for q_h, k_h, v_h
    q_base = 0  # q is [B, H, K] => linear indexing base for (b,h)
    k_base = b * H * K
    v_base = b * H * V

    # Load g and beta for this (b,h)
    g_val = tl.load(g_ptr + b * H + h)  # float32
    beta_val = tl.load(beta_ptr + b * H + h)  # float32

    # Prepare vectors q_h, k_h, v_h (length K/V respectively)
    q_h = tl.zeros([K], dtype=tl.float32)
    k_h = tl.zeros([K], dtype=tl.float32)
    v_h = tl.zeros([V], dtype=tl.float32)

    # Fill q_h
    for j in range(0, K):
        q_j = tl.load(q_ptr + q_base + j)  # float32
        q_h[j] = q_j
    # Fill k_h
    for j in range(0, K):
        k_j = tl.load(k_ptr + k_base + j)
        k_h[j] = k_j
    # Fill v_h
    for j in range(0, V):
        v_j = tl.load(v_ptr + v_base + j)
        v_h[j] = v_j

    # Compute old_v = k_h @ (g * state_old) -> (K,)
    state_old = tl.zeros([V, K], dtype=tl.float32)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            state_old[v_idx, k_idx] = tl.load(state_ptr + row_base + k_idx)
    old_v = tl.zeros([K], dtype=tl.float32)
    for k_idx in range(0, K):
        sum_val = tl.zeros((), dtype=tl.float32)
        for v_idx in range(0, V):
            state_val = state_old[v_idx, k_idx]
            sum_val += state_val * g_val
        old_v[k_idx] = sum_val * k_h[k_idx]

    # Compute new_v = beta * v + (1 - beta) * old_v -> (V,)
    new_v = tl.zeros([V], dtype=tl.float32)
    for v_idx in range(0, V):
        new_v[v_idx] = beta_val * v_h[v_idx] + (1.0 - beta_val) * old_v[v_idx]

    # Compute state_remove and state_update: scalars k_h @ old_v and k_h @ new_v
    state_remove = tl.zeros((), dtype=tl.float32)
    state_update = tl.zeros((), dtype=tl.float32)
    for k_idx in range(0, K):
        state_remove += old_v[k_idx] * k_h[k_idx]
        state_update += new_v[k_idx] * k_h[k_idx]

    # Update h_state_new elementwise: h_state_new = (g * state_old) - state_remove + state_update
    h_state_new = tl.zeros([V, K], dtype=tl.float32)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            state_val = tl.load(state_ptr + row_base + k_idx)
            h_state_new[v_idx, k_idx] = state_val * g_val
    # Add/subtract scalars
    h_state_new = h_state_new - state_remove + state_update

    # Compute output scalar: output = scale * q_h @ h_state_new
    out_scalar = tl.zeros((), dtype=tl.float32)
    for k_idx in range(0, K):
        row_k = h_state_new[:, k_idx]  # [V] vector
        out_scalar += q_h[k_idx] * tl.sum(row_k)

    # Store output at [b, h] (scalar per (b,h))
    tl.store(output_ptr + b * H + h, out_scalar * scale)

    # Store new_state slice: write h_state_new back to state_ptr at (b,h)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            tl.store(new_state_ptr + b * (H * V * K) + h * V * K + v_idx * K + k_idx, h_state_new[v_idx, k_idx])

def triton_invsqrt_scale_kernel(K: tl.int32, out_ptr):

    tl.store(out_ptr, inv)

    # Load a[b,h] and dt_bias[h]
    a_val = tl.load(a_ptr + b * H + h)
    dt_val = tl.load(dt_bias_ptr + h)
    x = a_val + dt_val  # float32
    # softplus(x) = max(x, 0) + log(1 + exp(-|x|))
    abs_x = tl.abs(x)
    sp = tl.maximum(x, 0.0) + tl.log(1.0 + tl.exp(-abs_x))
    A_log_val = tl.load(A_log_ptr + h)
    g_val = tl.exp(-tl.exp(A_log_val) * sp)
    # beta = sigmoid(b_val)
    b_val = tl.load(b_ptr + b * H + h)
    beta_val = 1.0 / (1.0 + tl.exp(-b_val))
    # Store g and beta at [b, h]
    tl.store(g_ptr + b * H + h, g_val)
    tl.store(beta_ptr + b * H + h, beta_val)


@triton.jit
def triton_invsqrt_scale_kernel(K: tl.int32, out_ptr):
    inv = 1.0 / tl.sqrt(tl.cast(K, tl.float32))
    tl.store(out_ptr, inv)


@triton.jit
def triton_update_kernel(
    q_ptr, k_ptr, v_ptr, state_ptr, g_ptr, beta_ptr,
    output_ptr, new_state_ptr,
    B: tl.constexpr, H: tl.constexpr, V: tl.constexpr, K: tl.constexpr,
    scale: tl.float32,
):
    pid = tl.program_id(axis=0)
    b = pid // H
    h = pid % H

    # Construct contiguous base vectors for q_h, k_h, v_h
    q_base = 0  # q is [B, H, K] => linear indexing base for (b,h)
    k_base = b * H * K
    v_base = b * H * V

    # Load g and beta for this (b,h)
    g_val = tl.load(g_ptr + b * H + h)  # float32
    beta_val = tl.load(beta_ptr + b * H + h)  # float32

    # Prepare vectors q_h, k_h, v_h (length K/V respectively)
    q_h = tl.zeros([K], dtype=tl.float32)
    k_h = tl.zeros([K], dtype=tl.float32)
    v_h = tl.zeros([V], dtype=tl.float32)

    # Fill q_h
    for j in range(0, K):
        q_j = tl.load(q_ptr + q_base + j)  # float32
        q_h[j] = q_j
    # Fill k_h
    for j in range(0, K):
        k_j = tl.load(k_ptr + k_base + j)
        k_h[j] = k_j
    # Fill v_h
    for j in range(0, V):
        v_j = tl.load(v_ptr + v_base + j)
        v_h[j] = v_j

    # Compute old_v = k_h @ (g * state_old) -> (K,)
    state_old = tl.zeros([V, K], dtype=tl.float32)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            state_old[v_idx, k_idx] = tl.load(state_ptr + row_base + k_idx)
    old_v = tl.zeros([K], dtype=tl.float32)
    for k_idx in range(0, K):
        sum_val = tl.zeros((), dtype=tl.float32)
        for v_idx in range(0, V):
            state_val = state_old[v_idx, k_idx]
            sum_val += state_val * g_val
        old_v[k_idx] = sum_val * k_h[k_idx]

    # Compute new_v = beta * v + (1 - beta) * old_v -> (V,)
    new_v = tl.zeros([V], dtype=tl.float32)
    for v_idx in range(0, V):
        new_v[v_idx] = beta_val * v_h[v_idx] + (1.0 - beta_val) * old_v[v_idx]

    # Compute state_remove and state_update: scalars k_h @ old_v and k_h @ new_v
    state_remove = tl.zeros((), dtype=tl.float32)
    state_update = tl.zeros((), dtype=tl.float32)
    for k_idx in range(0, K):
        state_remove += old_v[k_idx] * k_h[k_idx]
        state_update += new_v[k_idx] * k_h[k_idx]

    # Update h_state_new elementwise: h_state_new = (g * state_old) - state_remove + state_update
    h_state_new = tl.zeros([V, K], dtype=tl.float32)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            state_val = tl.load(state_ptr + row_base + k_idx)
            h_state_new[v_idx, k_idx] = state_val * g_val
    # Add/subtract scalars
    h_state_new = h_state_new - state_remove + state_update

    # Compute output scalar: output = scale * q_h @ h_state_new
    out_scalar = tl.zeros((), dtype=tl.float32)
    for k_idx in range(0, K):
        row_k = h_state_new[:, k_idx]  # [V] vector
        out_scalar += q_h[k_idx] * tl.sum(row_k)

    # Store output at [b, h] (scalar per (b,h))
    tl.store(output_ptr + b * H + h, out_scalar * scale)

    # Store new_state slice: write h_state_new back to state_ptr at (b,h)
    for v_idx in range(0, V):
        row_base = b * (H * V * K) + h * V * K + v_idx * K
        for k_idx in range(0, K):
            tl.store(new_state_ptr + b * (H * V * K) + h * V * K + v_idx * K + k_idx, h_state_new[v_idx, k_idx])
