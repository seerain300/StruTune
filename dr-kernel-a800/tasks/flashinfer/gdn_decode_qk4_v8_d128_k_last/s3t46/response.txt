    k_ptr,        # [B, H, K] float32
    v_ptr,        # [B, H, V] float32
    state_ptr,    # [B, H, V, K] float32
    a_ptr,        # [B, H] float32
    dt_bias_ptr,  # [H] float32

    scale,        # float32 scalar

    H: tl.constexpr,

    h_idx = tl.program_id(axis=1)

    beta = 1.0 / (1.0 + tl.exp(-tl.load(b_ptr + b_idx * H + h_idx)))  # scalar

    # For each i in V, update all j in K
    for i in tl.static_range(0, V):

        # Initialize new row as copy of state[i,:]
        # (Note: Triton does not support pointer-to-pointer copy; we initialize by loading and storing.)
        old_v = 0.0
        # Compute dot k_row @ state[i,:]
        for jj in tl.static_range(0, K):
            old_v += k_row[jj] * tl.load(state_b_h_ptr + i * K + jj)
        new_v = beta * tl.load(v_b_h_ptr + i) + (1.0 - beta) * old_v

        # Update new_state[i, :] = state[i, :] - old_v + new_v * k_row
        for jj in tl.static_range(0, K):

            # Write new_state[i,jj] = state[i,jj] - old_v + new_v * k_row[jj]
            tl.store(row_new_ptrs + jj, val - old_v + new_v * k_row[jj])

    # After all rows updated, compute out[b,h] = scale * dot(q_row, new_state[b,h])
    acc = 0.0
    for jj in tl.static_range(0, K):
        acc += q_row[jj] * tl.load(new_state_b_h_ptr + jj)
    out_val = scale * acc
    tl.store(out_ptr + b_idx * H + h_idx, out_val)
