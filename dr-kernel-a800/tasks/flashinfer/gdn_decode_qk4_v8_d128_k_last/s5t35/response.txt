import math
import torch
import triton
import triton.language as tl


@triton.jit
def _update_all_kernel(
    q_ptr, k_ptr, v_ptr, state_ptr, out_ptr, new_state_ptr,
    a_ptr, dt_bias_ptr, A_log_ptr, b_ptr,
    B, H, V, K
):
    """
    Each program handles one (b, h):
      - b = program_id(0) // H
      - h = program_id(0) % H
    """
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H

    # Base offsets for this (b, h)
    # Note: q, k, v shapes: [B, F, K] flattened to [B*F*K]
    #       v index uses h directly in F=8, but we pass v_ptr for each b,h
    #       We'll compute q/k/v indices via b,h and K.
    q_offset = (b * H + h) * K
    k_offset = q_offset  # same K
    v_offset = b * H * V + h * V  # v is [B, H, V] flattened
    state_offset = b * H * V * K + h * V * K  # state is [B, H, V, K] flattened

    # Load parameters for this h
    a_val = tl.load(a_ptr + b * H + h)
    db_val = tl.load(dt_bias_ptr + h)
    A_log_val = tl.load(A_log_ptr + h)
    b_val = tl.load(b_ptr + b * H + h)

    # Compute g and beta in fp32
    # softplus(x) = log(1 + exp(x)), computed in fp32
    a_f32 = a_val.to(tl.float32)
    db_f32 = db_val.to(tl.float32)
    A_log_f32 = A_log_val.to(tl.float32)
    b_f32 = b_val.to(tl.float32)

    x = a_f32 + db_f32
    # softplus
    softplus = tl.log(1.0 + tl.exp(x))
    g = tl.exp(-tl.exp(A_log_f32) * softplus)
    beta = 1.0 / (1.0 + tl.exp(-b_f32))

    # Load q_h, k_h, v_h, state_old
    # q_h, k_h: [K]
    q_h = tl.load(q_ptr + q_offset + tl.arange(0, K), mask=tl.arange(0, K) < K)
    k_h = tl.load(k_ptr + k_offset + tl.arange(0, K), mask=tl.arange(0, K) < K)
    v_h = tl.load(v_ptr + v_offset + tl.arange(0, V), mask=tl.arange(0, V) < V)

    # state_old: [V, K]
    state_old = tl.zeros((V, K), dtype=tl.float32)
    for i in range(V):
        for j in range(K):
            state_old[i, j] = tl.load(state_ptr + state_offset + i * K + j)

    # old_v = k_h @ state_old (reduce over K)
    old_v = 0.0
    for j in range(K):
        old_v += k_h[j] * state_old[:, j]

    # new_v = beta * v_h + (1 - beta) * old_v (vector over V)
    new_v = beta * v_h + (1.0 - beta) * old_v

    # old_state = g * state_old
    old_state = g * state_old

    # state_remove = k_h @ old_state (reduce over K) -> [V]
    state_remove = tl.zeros((V,), dtype=tl.float32)
    for i in range(V):
        sum_k = 0.0
        for j in range(K):
            sum_k += k_h[j] * old_state[i, j]
        state_remove[i] = sum_k

    # state_update = k_h @ new_v (reduce over K) -> [V]
    state_update = tl.zeros((V,), dtype=tl.float32)
    for i in range(V):
        sum_k = 0.0
        for j in range(K):
            sum_k += k_h[j] * new_v[i]
        state_update[i] = sum_k

    # new_state[b,h] = old_state - state_remove[:, None] + state_update[:, None]  -> [V,K]
    new_state_tile = old_state - state_remove[:, None] + state_update[:, None]

    # Write new_state back to new_state_ptr
    for i in range(V):
        for j in range(K):
            tl.store(new_state_ptr + state_offset + i * K + j, new_state_tile[i, j])

    # Compute output[b,h] = scale * sum_j q_h[j] * new_state[b,h][j,:] (reduce over V)
    scale = 1.0 / tl.sqrt(K)  # default scale from original sample; use provided scale if needed
    out_val = 0.0
    for j in range(K):
        row_j = 0.0
        for i in range(V):
            row_j += q_h[i] * new_state_tile[i, j]
        out_val += scale * row_j
    # Store output to out_ptr[b*H + h]
    tl.store(out_ptr + b * H + h, out_val)


class ModelNew(torch.nn.Module):
    def forward(self, q, k, v, state, A_log, a, dt_bias, b, scale):
        """
        q: [B, 1, 4, K], k: [B, 1, 4, K], v: [B, 1, 8, V], state: [B, 8, V, K]
        A_log: [8], a: [B, 1, 8], dt_bias: [8], b: [B, 1, 8], scale: float
        Returns: output [B, H, V] bfloat16, new_state [B, H, V, K] float32
        """
        # Shapes
        B, _, Fq, K = q.shape  # Fq should be 4
        B2, _, Fk, _ = k.shape  # Fk should be 4
        B3, _, Fv, V = v.shape  # Fv should be 8
        B4, H, V2, K2 = state.shape  # H should be 8, V==V2, K==K2
        assert B == B2 == B3 == B4
        assert Fq == 4 and Fk == 4 and Fv == 8
        assert K == 128 and V == 128 and H == 8
        assert scale is not None and scale != 0.0

        # Cast parameters to float32 for Triton math
        a_f32 = a.to(torch.float32)      # [B, 1, H]
        b_f32 = b.to(torch.float32)      # [B, 1, H]
        A_log_f32 = A_log.to(torch.float32)  # [H]
        dt_bias_f32 = dt_bias.to(torch.float32)  # [H]

        # Flatten inputs for kernel
        # q, k: [B, F, K] -> flatten to [B*F*K]
        q_flat = q.squeeze(1).reshape(B, Fq * K).contiguous()
        k_flat = k.squeeze(1).reshape(B, Fk * K).contiguous()
        v_flat = v.squeeze(1).reshape(B, Fv * V).contiguous()  # [B, 8*128]
        state_flat = state.reshape(B, H * V * K).contiguous()  # [B, 8*128*128]

        # Allocate output [B, H] and new_state [B, H, V, K] (linearized)
        out = torch.empty(B * H, dtype=torch.float32, device=q.device)
        new_state = torch.empty(B * H * V * K, dtype=torch.float32, device=q.device)

        # Launch Triton kernel
        grid = (B * H,)
        _update_all_kernel[grid](
            q_flat, k_flat, v_flat, state_flat, out, new_state,
            a_f32.view(-1), dt_bias_f32, A_log_f32, b_f32.view(-1),
            B, H, V, K,
            num_warps=1,
        )

        # Reshape output to [B, H, V] and cast to bfloat16
        out_bf16 = out.view(B, H).unsqueeze(2).expand(B, H, V).contiguous().to(torch.bfloat16)
        new_state = new_state.view(B, H, V, K).contiguous()

        return out_bf16, new_state
