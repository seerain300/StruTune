import torch
import triton
import triton.language as tl


@triton.jit
def compute_g_beta_kernel(
    A_log_ptr,         # [H] float32
    a_ptr,             # [B,H] float32
    dt_bias_ptr,       # [H] float32
    b_ptr,             # [B,H] float32
    g_out_ptr,         # [B,H] float32
    beta_out_ptr,      # [B,H] float32
    B: tl.constexpr,   # number of batches (unused but in signature)
    H: tl.constexpr,   # number of heads
):
    # grid is (B*H,)
    pid = tl.program_id(axis=0)
    b_idx = pid // H
    h_idx = pid % H

    # Load scalars
    a_val = tl.load(a_ptr + b_idx * H + h_idx)
    dt_val = tl.load(dt_bias_ptr + h_idx)
    b_val = tl.load(b_ptr + b_idx * H + h_idx)
    A_log_val = tl.load(A_log_ptr + h_idx)

    # softplus(x) = log(1 + exp(-|x|)) + max(x, 0) for numerical stability
    x = a_val + dt_val
    abs_x = tl.abs(x)
    softplus = tl.log(1.0 + tl.exp(-abs_x)) + tl.maximum(x, 0.0)

    # g = exp(-exp(A_log[h]) * softplus)
    g = tl.exp(-tl.exp(A_log_val) * softplus)

    # beta = sigmoid(b) = 1 / (1 + exp(-b))
    beta = 1.0 / (1.0 + tl.exp(-b_val))

    # Store
    tl.store(g_out_ptr + b_idx * H + h_idx, g)
    tl.store(beta_out_ptr + b_idx * H + h_idx, beta)


@triton.jit
def state_update_kernel(
    state_ptr,          # [B,H,V,K] float32
    new_state_ptr,      # [B,H,V,K] float32
    k_ptr,              # [B,H,K] float32 (k expanded to H)
    v_ptr,              # [B,H,V] float32 (v expanded to H)
    beta_ptr,           # [B,H] float32
    B: tl.constexpr,
    H: tl.constexpr,
    V: tl.constexpr,    # e.g., 128
    K: tl.constexpr,    # e.g., 128
    stride_b: tl.constexpr,
    stride_h: tl.constexpr,
    stride_v: tl.constexpr,
    stride_k: tl.constexpr,
    stride_new_b: tl.constexpr,
    stride_new_h: tl.constexpr,
    stride_new_i: tl.constexpr,
    stride_new_j: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # over B*H
    b_idx = pid // H
    h_idx = pid % H

    # For each i in [0, V), update row i across j in [0, K)
    for i in tl.static_range(0, V):
        # Compute old_v = dot(k[b,h,:], state[b,h,i,:])
        sum_old_v = 0.0
        for j in tl.static_range(0, K):
            addr_k = b_idx * stride_b + h_idx * stride_h + j * stride_k
            k_val = tl.load(k_ptr + addr_k)  # [1] actually a scalar per (b,h)
            addr_state = b_idx * stride_b + h_idx * stride_h + i * stride_v + j * stride_k
            state_val = tl.load(state_ptr + addr_state)
            sum_old_v += k_val * state_val

        beta_val = tl.load(beta_ptr + b_idx * H + h_idx)

        # new_v = beta * v[b,h,i] + (1 - beta) * old_v
        addr_v = b_idx * stride_b + h_idx * stride_h + i * stride_v
        v_elem = tl.load(v_ptr + addr_v)
        new_v = beta_val * v_elem + (1.0 - beta_val) * sum_old_v

        # Update new_state[b,h,i,j] = state[b,h,i,j] - old_v + k[b,h,j] * new_v
        for j in tl.static_range(0, K):
            k_elem = tl.load(k_ptr + b_idx * stride_b + h_idx * stride_h + j * stride_k)
            addr_state_old = b_idx * stride_new_b + h_idx * stride_new_h + i * stride_new_i + j * stride_new_j
            state_old = tl.load(state_ptr + addr_state_old)
            addr_new = b_idx * stride_new_b + h_idx * stride_new_h + i * stride_new_i + j * stride_new_j
            # Note: Triton does not support multi-dim indexing; compute flat offsets as above.
            new_val = state_old - sum_old_v + k_elem * new_v
            tl.store(new_state_ptr + addr_new, new_val)


@triton.jit
def output_dot_kernel(
    q_exp_ptr,          # [B,H,K] float32 (q expanded to H)
    new_state_ptr,      # [B,H,V,K] float32
    out_ptr,            # [B,H] float32
    scale,              # float32
    B: tl.constexpr,
    H: tl.constexpr,
    V: tl.constexpr,    # e.g., 128
    K: tl.constexpr,    # e.g., 128
    stride_q_b: tl.constexpr,
    stride_q_h: tl.constexpr,
    stride_q_k: tl.constexpr,
    stride_ns_b: tl.constexpr,
    stride_ns_h: tl.constexpr,
    stride_ns_i: tl.constexpr,
    stride_ns_j: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # over B*H
    b_idx = pid // H
    h_idx = pid % H

    acc = 0.0
    for j in tl.static_range(0, K):
        addr_q = b_idx * stride_q_b + h_idx * stride_q_h + j * stride_q_k
        q_elem = tl.load(q_exp_ptr + addr_q)
        for i in tl.static_range(0, V):
            addr_ns = b_idx * stride_ns_b + h_idx * stride_ns_h + i * stride_ns_i + j * stride_ns_j
            ns_elem = tl.load(new_state_ptr + addr_ns)
            acc += q_elem * ns_elem

    out_val = scale * acc
    tl.store(out_ptr + b_idx * H + h_idx, out_val)


class ModelNew(torch.nn.Module):
    def forward(self, q, k, v, state, A_log, a, dt_bias, b, scale):
        """
        Triton-optimized implementation of the reference 'run' function.

        Args:
          q: [B, 1, num_q_heads, K], bfloat16
          k: [B, 1, num_k_heads, K], bfloat16
          v: [B, 1, num_v_heads, V], bfloat16
          state: [B, num_heads, V, K], float32
          A_log: [num_heads], float32
          a: [B, 1, num_heads], bfloat16
          dt_bias: [num_heads], float32
          b: [B, 1, num_heads], bfloat16
          scale: float
        Returns:
          output: [B, 1, H, 1], bfloat16
          new_state: [B, H, V, K], float32
        """
        device = q.device

        # Cast inputs to float32 for computation; keep 'state' as float32
        q_f32 = q.squeeze(1).to(torch.float32).contiguous()   # [B, num_q_heads, K]
        k_f32 = k.squeeze(1).to(torch.float32).contiguous()   # [B, num_k_heads, K]
        v_f32 = v.squeeze(1).to(torch.float32).contiguous()   # [B, num_v_heads, V]
        state_f32 = state.to(torch.float32).contiguous()      # [B, num_heads, V, K]

        # Expand q and k heads by repeat_interleave (ratio = num_v_heads // num_q_heads = 2 in provided tests)
        B = q_f32.shape[0]
        H = q_f32.shape[1]  # num_heads
        K = q_f32.shape[2]
        V = v_f32.shape[2]

        q_exp = q_f32.repeat_interleave(2, dim=1)  # [B, H, K]
        k_exp = k_f32.repeat_interleave(2, dim=1)  # [B, H, K]

        # Allocate outputs for g and beta
        g_out = torch.empty((B, H), dtype=torch.float32, device=device)
        beta_out = torch.empty((B, H), dtype=torch.float32, device=device)

        # Launch Triton kernel to compute g and beta per (b,h)
        grid = (B * H,)
        compute_g_beta_kernel[grid](
            A_log.to(torch.float32), a.squeeze(1).to(torch.float32), dt_bias.to(torch.float32), b.squeeze(1).to(torch.float32),
            g_out, beta_out,
            B=B, H=H,
        )

        # Allocate new_state
        new_state = torch.empty((B, H, V, K), dtype=torch.float32, device=device)

        # Get strides (in elements)
        stride_b, stride_h, stride_v, stride_k = state_f32.stride(0), state_f32.stride(1), state_f32.stride(2), state_f32.stride(3)
        stride_new_b, stride_new_h, stride_new_i, stride_new_j = new_state.stride(0), new_state.stride(1), new_state.stride(2), new_state.stride(3)

        # Launch Triton kernel to update new_state
        state_update_kernel[grid](
            state_f32, new_state, k_exp, v_f32, beta_out,
            B=B, H=H, V=V, K=K,
            stride_b=stride_b, stride_h=stride_h, stride_v=stride_v, stride_k=stride_k,
            stride_new_b=stride_new_b, stride_new_h=stride_new_h, stride_new_i=stride_new_i, stride_new_j=stride_new_j,
        )

        # Compute output: out[b,h] = scale * (q_exp[h] @ new_state[b,h]) for each (b,h)
        out = torch.empty((B, H), dtype=torch.float32, device=device)

        # Strides for q_exp and new_state
        stride_q_b, stride_q_h, stride_q_k = q_exp.stride(0), q_exp.stride(1), q_exp.stride(2)
        stride_ns_b, stride_ns_h, stride_ns_i, stride_ns_j = new_state.stride(0), new_state.stride(1), new_state.stride(2), new_state.stride(3)

        # Launch Triton kernel to compute outputs
        output_dot_kernel[grid](
            q_exp, new_state, out, float(scale),
            B=B, H=H, V=V, K=K,
            stride_q_b=stride_q_b, stride_q_h=stride_q_h, stride_q_k=stride_q_k,
            stride_ns_b=stride_ns_b, stride_ns_h=stride_ns_h, stride_ns_i=stride_ns_i, stride_ns_j=stride_ns_j,
        )

        # Return outputs with required dtypes and shapes: [B, 1, H, 1] bfloat16 and [B, H, V, K] float32
        output_bf16 = out.unsqueeze(1).unsqueeze(-1).to(torch.bfloat16)  # [B, 1, H, 1]
        new_state_f32 = new_state  # [B, H, V, K]
        return output_bf16, new_state_f32
