def gate_beta_scalar_kernel(g_out_ptr, A_log_ptr, a_ptr, dt_bias_ptr, B, H):

    pid = tl.program_id(0)
    # Map pid to (b,h) using flattened index

    h = pid % H

    b = pid // H
    h = pid % H

    tl.store(v_sum_ptr + pid, sum_val)
