Design recap:
- The original PyTorch run computes gate g and beta, then for each (b,h):
  - old_v = k @ state_bh
  - new_v = beta * v + (1 - beta) * old_v
  - state_remove = k @ (g * state_bh)
  - state_update = k @ new_v
  - h_state_vec[i] = sum_j state_bh[i,j] * g - state_remove + state_update
  - output[b,h] = scale * (q @ h_state_vec)
  - new_state[b,h] = h_state_vec reshaped [V, K]

Optimization strategy:
- Implement Triton kernels for:
  1) g = exp(-exp(A_log) * softplus(a + dt_bias))
  2) beta = sigmoid(b)
  3) old_v = dot(k, state_flat), state_remove = dot(k, g*state_flat), state_update = dot(k, beta*v + (1-beta)*old_v)
     (Note: Triton does not support arbitrary broadcasting of scalars into [V,K] in a single matmul, so we keep these as multiple small kernels.)
  4) h_state_vec[i] = sum_j state[i,j] * g - state_remove + state_update (compute per i)
  5) output[b,h] = scale * dot(q, h_state_vec)
  6) write new_state[b,h] as [V,K] by broadcasting h_state_vec across K.

- Forward will:
  - Convert q,k,v,a,b to float32 on device; dt_bias and A_log are float32.
  - Launch kernels to compute g and beta.
  - For each (b,h), allocate per-(b,h) output buffers and launch the remaining kernels to compute h_state_vec, output scalar, and write new_state.

This keeps all math in Triton, avoids Python-side loops over large dimensions, and ensures each Triton kernel is actually invoked.

import math
import torch
import triton
import triton.language as tl


# Triton kernel: compute g[b,h] = exp(-exp(A_log[h]) * softplus(a[b,h] + dt_bias[h]))
@triton.jit
def softplus_and_exp_kernel(
    dt_bias_ptr,    # float32 [H]
    a_ptr,          # float32 [B*H]
    A_log_ptr,      # float32 [H]
    g_out_ptr,      # float32 [B*H]
    B: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # 0..B*H-1
    h = pid % H
    a_val = tl.load(a_ptr + pid)
    dt_val = tl.load(dt_bias_ptr + h)
    x = a_val + dt_val
    s = tl.log(1.0 + tl.exp(x))
    e = tl.exp(tl.load(A_log_ptr + h))
    g = tl.exp(-e * s)
    tl.store(g_out_ptr + pid, g)


# Triton kernel: compute beta[b,h] = 1 / (1 + exp(-b[b,h]))
@triton.jit
def sigmoid_kernel(
    b_ptr,          # float32 [B*H]
    beta_out_ptr,   # float32 [B*H]
    B: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # 0..B*H-1
    b_val = tl.load(b_ptr + pid)
    beta = 1.0 / (1.0 + tl.exp(-b_val))
    tl.store(beta_out_ptr + pid, beta)


# Triton kernel: compute old_v = dot(k_vec, state_mat) where state_mat is flattened [V*K]
# Inputs: k_ptr [K], state_ptr [V*K], old_v_ptr [1]
@triton.jit
def dot_k_state_kernel(
    k_ptr,          # float32 [K]
    state_ptr,      # float32 [V*K]
    old_v_ptr,      # float32 [1]
    V: tl.constexpr,
    K: tl.constexpr,
):
    acc = 0.0
    for j in range(K):
        k_j = tl.load(k_ptr + j)
        for i in range(V):
            state_ij = tl.load(state_ptr + i * K + j)
            acc += state_ij * k_j
    tl.store(old_v_ptr, acc)


# Triton kernel: compute state_remove = dot(k_vec, g * state_mat) where state_mat is flattened [V*K]
@triton.jit
def dot_k_gstate_kernel(
    k_ptr,          # float32 [K]
    state_ptr,      # float32 [V*K]
    g_scalar,       # float32 scalar
    state_remove_ptr,  # float32 [1]
    V: tl.constexpr,
    K: tl.constexpr,
):
    acc = 0.0
    for j in range(K):
        k_j = tl.load(k_ptr + j)
        for i in range(V):
            state_ij = tl.load(state_ptr + i * K + j)
            acc += state_ij * k_j
    acc = acc * g_scalar
    tl.store(state_remove_ptr, acc)


# Triton kernel: compute state_update = dot(k_vec, beta*v + (1 - beta)*old_v), given old_v
@triton.jit
def dot_k_newv_kernel(
    k_ptr,          # float32 [K]
    v_ptr,          # float32 [V]
    old_v_scalar,   # float32 scalar
    beta_scalar,    # float32 scalar
    state_update_ptr,  # float32 [1]
    V: tl.constexpr,
    K: tl.constexpr,
):
    acc = 0.0
    for j in range(K):
        k_j = tl.load(k_ptr + j)
        for i in range(V):
            v_i = tl.load(v_ptr + i)
            acc += k_j * (beta_scalar * v_i + (1.0 - beta_scalar) * old_v_scalar)
    tl.store(state_update_ptr, acc)


# Triton kernel: compute h_state_vec[i] = sum_j state[i,j] * g - state_remove + state_update
@triton.jit
def h_state_vec_kernel(
    state_ptr,      # float32 [V*K]
    g_scalar,       # float32 scalar
    state_remove_scalar,  # float32 scalar
    state_update_scalar,  # float32 scalar
    h_state_ptr,    # float32 [V]
    V: tl.constexpr,
    K: tl.constexpr,
):
    # One program per i
    pid = tl.program_id(axis=0)  # 0..V-1
    acc = 0.0
    for j in range(K):
        acc += tl.load(state_ptr + pid * K + j)
    h_val = acc * g_scalar - state_remove_scalar + state_update_scalar
    tl.store(h_state_ptr + pid, h_val)


# Triton kernel: compute output[b,h] = scale * dot(q_vec, h_state_vec)
@triton.jit
def dot_q_hstate_kernel(
    q_ptr,          # float32 [K]
    h_state_ptr,    # float32 [V]
    scale,          # float32 scalar
    output_ptr,     # float32 [1]
    V: tl.constexpr,
    K: tl.constexpr,
):
    acc = 0.0
    for i in range(V):
        h_i = tl.load(h_state_ptr + i)
        for j in range(K):
            q_j = tl.load(q_ptr + j)
            acc += h_i * q_j
    acc = acc * scale
    tl.store(output_ptr, acc)


# Triton kernel: write new_state[b,h] as [V,K] by broadcasting h_state_vec across K
@triton.jit
def write_new_state_kernel(
    h_state_ptr,    # float32 [V]
    new_state_ptr,  # float32 [V*K]
    V: tl.constexpr,
    K: tl.constexpr,
):
    pid_row = tl.program_id(axis=0)  # 0..V-1
    pid_col = tl.program_id(axis=1)  # 0..K-1
    val = tl.load(h_state_ptr + pid_row)
    tl.store(new_state_ptr + pid_row * K + pid_col, val)

    beta_out_ptr,   # float32 [B*H]

# Triton kernel: compute output[b,h] = scale * (q @ h_state_vec)
# Also writes new_state[b,h] = h_state_vec reshaped as [V,K] (broadcast across K).
@triton.jit
def process_bh_kernel(
    q_ptr,          # float32 [K]
    k_ptr,          # float32 [K]
    v_ptr,          # float32 [V]
    state_ptr,      # float32 [V*K]
    g_scalar,       # float32 scalar
    beta_scalar,    # float32 scalar
    scale,          # float32 scalar
    output_ptr,     # float32 [1]
    new_state_ptr,  # float32 [V*K]
    V: tl.constexpr,
    K: tl.constexpr,
):
    # Accumulate h_state_vec[i] for all i
    h_state_vec = [0.0] * V  # Triton allows list scalars; we'll write them at the end
    # Compute old_v = k @ state, state_remove = k @ (g * state), state_update = k @ (beta*v + (1 - beta)*old_v)
    old_v = 0.0
    state_remove = 0.0
    state_update = 0.0

    # old_v
    for j in range(K):
        k_j = tl.load(k_ptr + j)
        for i in range(V):
            state_ij = tl.load(state_ptr + i * K + j)
            old_v += state_ij * k_j
    # state_remove
    for j in range(K):
        k_j = tl.load(k_ptr + j)
        for i in range(V):
            state_ij = tl.load(state_ptr + i * K + j)
            state_remove += state_ij * k_j
    state_remove = state_remove * g_scalar
    # state_update
    for j in range(K):
        k_j = tl.load(k_ptr + j)
        for i in range(V):
            v_i = tl.load(v_ptr + i)
            state_update += k_j * (beta_scalar * v_i + (1.0 - beta_scalar) * old_v)

    # Compute h_state_vec[i] = sum_j state[i,j] * g - state_remove + state_update
    for i in range(V):
        acc = 0.0
        for j in range(K):
            state_ij = tl.load(state_ptr + i * K + j)
            acc += state_ij
        h_state_vec[i] = acc * g_scalar - state_remove + state_update

    # Compute output[b,h] = scale * (q @ h_state_vec)
    output_bh = 0.0
    for i in range(V):
        h_i = h_state_vec[i]
        for j in range(K):

            output_bh += h_i * q_j

    for i in range(V):
        val = h_state_vec[i]
        for j in range(K):
