        a = tl.load(attn_ptr + attn_base + t)  # scalar attn value
        kc_row_ptr = Kc_ptr + t * head_dim_ckv
        # Loop over head_dim_ckv in chunks to keep vectorized loads

            offs = col + tl.arange(0, 32)
            mask = offs < head_dim_ckv

            out_vec += a * kc
    # Store out_vec to out_ptr[i, h, :]
    out_base = (i * H + h) * head_dim_ckv
    for j in range(0, head_dim_ckv):
        tl.store(out_ptr + out_base + j, out_vec[j])
    return
