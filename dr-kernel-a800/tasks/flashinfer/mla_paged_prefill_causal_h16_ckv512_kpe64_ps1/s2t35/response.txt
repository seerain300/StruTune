import torch
import math
import triton
import triton.language as tl


@triton.jit
def _forward_single_query_kernel(
    q_nope_ptr,       # *bf16, shape [Q_total, 16, 512]
    q_pe_ptr,         # *bf16, shape [Q_total, 16, 64]
    output_ptr,       # *bf16, shape [Q_total, 16, 512]
    lse_ptr,          # *f32,  shape [Q_total, 16]
    Kc_ptr,           # *f32,  shape [kv_len, 512]
    Kp_ptr,           # *f32,  shape [kv_len, 64]
    tok_len,          # int32: number of KV tokens for this batch element
    q_len,            # int32: number of queries in this batch element
    qo_indptr_ptr,    # *i32,  shape [len_indptr]
    kv_indptr_ptr,    # *i32,  shape [len_indptr]
    kv_indices_ptr,   # *i32,  shape [num_kv_indices]
    sm_scale,         # f32    scaling for logits
):
    # program ids: one per (batch, query)
    pid_b = tl.program_id(axis=0)  # batch index b
    pid_i = tl.program_id(axis=1)  # query index i

    # absolute query index
    qo_indptr_b = tl.load(qo_indptr_ptr + pid_b)  # int32
    q_abs = qo_indptr_b + pid_i
    # query count in this batch
    qo_indptr_next = tl.load(qo_indptr_ptr + pid_b + 1)
    if q_abs >= qo_indptr_next:
        # handle empty cases (no-op)
        return

    # Load qn and qp for all heads (h in 0..15) in fp32
    # q_nope_ptr and q_pe_ptr are bf16; we cast to fp32 for math
    # Output is per head, so loop over heads
    for h in range(16):
        # qn[h, :] = q_nope[q_abs, h, :]
        # Triton supports 3D pointer arithmetic via offsets; we manually compute offset:
        # offset = q_abs * (16 * 512) + h * 512
        qn_offset = q_abs * (16 * 512) + h * 512
        qn = tl.load(q_nope_ptr + qn_offset)
        qn = qn.to(tl.float32)  # [512], but we need scalar-by-vector form? We need to construct per-column, so better to load per feature via loop; however Triton allows tensor loads. Simpler: load a row vector by computing linear offsets for all 512 features.

        # Instead of row vector load, we can construct qn as a tensor by loading each feature:
        # But Triton load expects scalar pointer; to handle vector, we can compute dot via tl.dot with K vectors. However Triton's matmul here is limited to 2D matrices. So we load Kc_sel and Kp_sel as 2D and use tl.dot.

        # Prepare K_sel: Kc_sel and Kp_sel are [tok_len, 512] and [tok_len, 64] respectively.
        # We'll iterate j and load the corresponding rows.
        # Initialize logits and output accumulators
        logits = tl.zeros([tok_len], dtype=tl.float32)
        out_vec = tl.zeros([512], dtype=tl.float32)

        # Compute stable logsumexp and softmax
        max_logit = tl.full((), -1e20, tl.float32)
        # Loop over j tokens
        for j in range(tok_len):
            # dot with Kc_sel[j, :]
            Kc_row = tl.load(Kc_ptr + j * 512 + tl.arange(0, 512))  # vector of 512
            dot_qn = tl.sum(qn * Kc_row, axis=0)
            # dot with Kp_sel[j, :]
            Kp_row = tl.load(Kp_ptr + j * 64 + tl.arange(0, 64))   # vector of 64
            # We need to dot qp[h, :] with Kp_row. But we need qp[h, :], which is 64-dim. We can reconstruct it by loading q_pe at h. However q_pe is 64-dim per head.
            # q_pe_ptr: load for this head h
            qpe_offset = q_abs * (16 * 64) + h * 64
            qp = tl.load(q_pe_ptr + qpe_offset).to(tl.float32)  # [64]
            dot_qp = tl.sum(qp * Kp_row, axis=0)
            logits[j] = dot_qn + dot_qp

        # Apply causal mask: only j >= prefix_len + i are valid; prefix_len = tok_len - q_len
        prefix_len = tok_len - q_len
        valid = tl.arange(0, tok_len) >= (prefix_len + pid_i + 1)
        logits = tl.where(valid, logits, -1e20)

        # Scale logits
        logits = logits * sm_scale

        # Stable logsumexp
        m = tl.max(logits, axis=0)
        # exp and sum
        exp_logits = tl.exp(logits - m)
        sum_exp = tl.sum(exp_logits, axis=0)
        lse_val = (m + tl.log(sum_exp)) / tl.log(2.0)  # convert to log2
        tl.store(lse_ptr + q_abs * 16 + h, lse_val)

        # Softmax
        softmax = exp_logits / sum_exp  # shape [tok_len]

        # Accumulate attention output: out[h, :] += softmax[j] * Kc_sel[j, :]
        for j in range(tok_len):
            if valid[j]:
                Kc_row = tl.load(Kc_ptr + j * 512 + tl.arange(0, 512))
                out_vec += softmax[j] * Kc_row

        # Store output in bfloat16
        out_offset = q_abs * (16 * 512) + h * 512
        out_bf16 = out_vec.to(tl.bfloat16)
        tl.store(output_ptr + out_offset, out_bf16)

    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Check shapes
        assert q_nope.dim() == 3 and q_pe.dim() == 3
        total_q, num_qo_heads, head_dim_ckv = q_nope.shape
        _, num_qo_heads2, head_dim_kpe = q_pe.shape
        assert num_qo_heads == 16 and num_qo_heads == num_qo_heads2
        assert head_dim_ckv == 512
        assert head_dim_kpe == 64

        # Extract batch size and q_len per batch element
        batch_size = int(qo_indptr.shape[0]) - 1
        len_indptr = qo_indptr.shape[0]
        q_len_total = qo_indptr[-1].item()  # total queries
        qo_indptr = qo_indptr.to(torch.int32).contiguous()
        kv_indptr = kv_indptr.to(torch.int32).contiguous()
        kv_indices = kv_indices.to(torch.int32).contiguous()

        # Prepare output tensors
        output = torch.empty((q_len_total, 16, 512), dtype=torch.bfloat16, device=q_nope.device)
        lse = torch.empty((q_len_total, 16), dtype=torch.float32, device=q_nope.device)

        # For each batch element, select Kc_sel and Kp_sel and launch kernel
        for b in range(batch_size):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            q_len = q_end - q_start

            # If no queries in this batch element, skip
            if q_len == 0:
                continue

            # Token range for this batch
            tok_start = int(kv_indptr[b].item())
            tok_end = int(kv_indptr[b + 1].item())
            tok_len = tok_end - tok_start

            # Indices of KV tokens selected for this batch
            tok_idx = kv_indices[tok_start:tok_end].to(torch.int32).contiguous()  # [tok_len]

            # Select Kc_sel and Kp_sel per batch element from caches. Cast to fp32 for kernel.
            # ckv_cache: [num_pages, 1, 512], kpe_cache: [num_pages, 1, 64]
            Kc_sel = ckv_cache[tok_idx, 0].to(torch.float32)  # [tok_len, 512]
            Kp_sel = kpe_cache[tok_idx, 0].to(torch.float32)  # [tok_len, 64]

            # Launch Triton kernel: one program per query i in this batch
            grid = (1, q_len)
            _forward_single_query_kernel[grid](
                q_nope, q_pe, output, lse,
                Kc_sel, Kp_sel,
                tok_len, q_len,
                qo_indptr, kv_indptr, kv_indices,
                float(sm_scale),
                num_warps=4,
            )

        return output, lse

import math
import triton
import triton.language as tl


@triton.jit
def _forward_single_query_kernel(
    q_nope_ptr,       # *bf16, shape [Q_total, 16, 512]
    q_pe_ptr,         # *bf16, shape [Q_total, 16, 64]
    output_ptr,       # *bf16, shape [Q_total, 16, 512]
    lse_ptr,          # *f32,  shape [Q_total, 16]
    Kc_ptr,           # *f32,  shape [kv_len, 512]
    Kp_ptr,           # *f32,  shape [kv_len, 64]
    tok_len,          # int32: number of KV tokens for this batch element
    q_len,            # int32: number of queries in this batch element
    sm_scale,         # f32
):
    # program ids: one per (batch, query)
    pid_b = tl.program_id(axis=0)  # batch index b
    pid_i = tl.program_id(axis=1)  # query index i

    # absolute query index
    q_abs = pid_b * q_len + pid_i
    if q_abs < 0:
        return

    # Initialize per-head accumulators
    for h in range(16):
        # Compute logits vector for this head
        logits = tl.zeros([tok_len], dtype=tl.float32)

        # Compute qn[h, :] and qp[h, :] in fp32

        qpe_offset = q_abs * (16 * 64) + h * 64
        qp = tl.load(q_pe_ptr + qpe_offset).to(tl.float32)  # [64]

        # Compute logits for all j
        for j in range(tok_len):
            Kc_row = tl.load(Kc_ptr + j * 512 + tl.arange(0, 512))  # [512]
            dot_qn = tl.sum(qn * Kc_row, axis=0)

            Kp_row = tl.load(Kp_ptr + j * 64 + tl.arange(0, 64))  # [64]
            dot_qp = tl.sum(qp * Kp_row, axis=0)

            logits[j] = dot_qn + dot_qp

import torch
import math
import triton
import triton.language as tl


@triton.jit
def _forward_single_query_kernel(
    q_nope_ptr,       # *bf16, shape [Q_total, 16, 512]
    q_pe_ptr,         # *bf16, shape [Q_total, 16, 64]
    output_ptr,       # *bf16, shape [Q_total, 16, 512]
    lse_ptr,          # *f32,  shape [Q_total, 16]
    Kc_ptr,           # *f32,  shape [kv_len, 512]
    Kp_ptr,           # *f32,  shape [kv_len, 64]
    tok_len,          # int32: number of KV tokens for this batch element
    q_len,            # int32: number of queries in this batch element
    sm_scale,         # f32
):
    # program ids: one per (batch, query)
    pid_b = tl.program_id(axis=0)  # batch index b
    pid_i = tl.program_id(axis=1)  # query index i

    # absolute query index
    q_abs = pid_b * q_len + pid_i
    if q_abs < 0:
        return

    # Initialize per-head accumulators
    for h in range(16):
        # Compute logits vector for this head
        logits = tl.zeros([tok_len], dtype=tl.float32)

        # Compute qn[h, :] and qp[h, :] in fp32
        qn_offset = q_abs * (16 * 512) + h * 512
        qn = tl.load(q_nope_ptr + qn_offset).to(tl.float32)  # [512]

        qpe_offset = q_abs * (16 * 64) + h * 64
        qp = tl.load(q_pe_ptr + qpe_offset).to(tl.float32)  # [64]

        # Compute logits for all j
        for j in range(tok_len):
            Kc_row = tl.load(Kc_ptr + j * 512 + tl.arange(0, 512))  # [512]
            dot_qn = tl.sum(qn * Kc_row, axis=0)

            Kp_row = tl.load(Kp_ptr + j * 64 + tl.arange(0, 64))  # [64]
            dot_qp = tl.sum(qp * Kp_row, axis=0)

            logits[j] = dot_qn + dot_qp

        # Apply causal mask: j >= prefix_len + i + 1, where prefix_len = tok_len - q_len
        prefix_len = tok_len - q_len
        valid = tl.arange(0, tok_len) >= (prefix_len + pid_i + 1)
        logits = tl.where(valid, logits, -1e20)

        # Scale logits
        logits = logits * sm_scale

        # Stable logsumexp in fp32
        m = tl.max(logits, axis=0)
        exp_logits = tl.exp(logits - m)
        sum_exp = tl.sum(exp_logits, axis=0)
        lse_val = (m + tl.log(sum_exp)) / tl.log(2.0)  # logsumexp in log2
        tl.store(lse_ptr + q_abs * 16 + h, lse_val)

        # Softmax
        softmax = exp_logits / sum_exp  # [tok_len]

        # Accumulate attention output: out[h, :] += softmax[j] * Kc_sel[j, :]
        out_vec = tl.zeros([512], dtype=tl.float32)
        for j in range(tok_len):
            if valid[j]:
                Kc_row = tl.load(Kc_ptr + j * 512 + tl.arange(0, 512))  # [512]
                out_vec += softmax[j] * Kc_row

        # Store output in bfloat16
        out_offset = q_abs * (16 * 512) + h * 512
        out_bf16 = out_vec.to(tl.bfloat16)
        tl.store(output_ptr + out_offset, out_bf16)
