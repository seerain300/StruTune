Optimization strategy
- We will write a Triton kernel that processes one query per launch. The grid will be (len_indptr-1, q_len), i.e., one Triton program per batch element and per query. This avoids dynamic loops in the kernel and keeps it simple and robust for different workloads.
- Inside the kernel:
  - Compute absolute query index q_abs = q_start + i.
  - Load qn[h, :] and qp[h, :] from q_nope and q_pe for all heads, cast to float32 for computation.
  - Kc_sel and Kp_sel are provided as [kv_len, 512] and [kv_len, 64] in bfloat16 and cast to float32 in the kernel.
  - Compute logits per head by looping j from 0 to kv_len-1:
    - dot_qn = sum_k qn[h, k] * Kc_sel[j, k]
    - dot_qp = sum_k qp[h, k] * Kp_sel[j, k]
    - logits[h, j] = dot_qn + dot_qp
  - Scale logits by sm_scale, apply causal mask (j >= prefix_len + i + 1).
  - Compute logsumexp in log2: lse[h] = log(sum_j exp(logits[h, j])) * ln2_inv + max_j logits[h, j].
  - Compute softmax over j (stable) and accumulate out[h, :] = sum_j softmax[h, j] * Kc_sel[j, :].
  - Store output[q_abs, h, :] as bfloat16 and lse[q_abs, h] as float32.

- The host code in ModelNew.forward will:
  - Assert constants (num_qo_heads=16, head_dim_ckv=512, head_dim_kpe=64).

import torch
import math
import triton
import triton.language as tl


@triton.jit
def _forward_single_query_kernel(
    # Input pointers in original dtype
    q_nope_ptr,       # *bf16, shape [Q_total, 16, 512], row-major
    q_pe_ptr,         # *bf16, shape [Q_total, 16, 64], row-major
    Kc_sel_ptr,       # *bf16, shape [kv_len, 512], row-major (selected tokens)
    Kp_sel_ptr,       # *bf16, shape [kv_len, 64], row-major (selected tokens)
    output_ptr,       # *bf16, shape [Q_total, 16, 512], row-major
    lse_ptr,          # *fp32, shape [Q_total, 16]
    # runtime scalars
    q_start,          # int32, starting query index for this batch element
    # constexpr meta-parameters
    q_len: tl.constexpr,     # number of queries in this batch element (used for grid, not looped here)
    kv_len: tl.constexpr,    # number of selected KV tokens
    sm_scale: tl.constexpr,  # fp32 scaling factor
    ln2_inv: tl.constexpr,   # fp32 = 1.0 / ln(2.0)
    NUM_HEADS: tl.constexpr, # 16
    HEAD_DIM_CKV: tl.constexpr, # 512
    HEAD_DIM_KPE: tl.constexpr, # 64
):
    # Grid: (len_indptr-1, q_len). We decode b from program_id(0) and i from program_id(1).
    b = tl.program_id(0)
    i = tl.program_id(1)
    q_abs = q_start + i

    # Load qn[h, :] and qp[h, :] for all heads and cast to fp32
    # q layout: [Q_total, NUM_HEADS, HEAD_DIM_CKV] -> linear index = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV + k
    for h in range(NUM_HEADS):
        # Vector of k indices for this head
        k_vec = tl.arange(0, HEAD_DIM_CKV)
        # Compute base pointer for this h
        base = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV
        qn = tl.load(q_nope_ptr + base + k_vec, mask=k_vec < HEAD_DIM_CKV, other=0.0).to(tl.float32)  # [HEAD_DIM_CKV]
        kpe_vec = tl.arange(0, HEAD_DIM_KPE)
        base_pe = q_abs * (NUM_HEADS * HEAD_DIM_KPE) + h * HEAD_DIM_KPE
        qp = tl.load(q_pe_ptr + base_pe + kpe_vec, mask=kpe_vec < HEAD_DIM_KPE, other=0.0).to(tl.float32)  # [HEAD_DIM_KPE]

        # Initialize logits per head and accumulator out
        logits = tl.zeros((kv_len,), dtype=tl.float32)
        out_vec = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)

        # Compute logits for each j in [0, kv_len)
        for j in range(kv_len):
            # Load Kc_sel[j, :] and Kp_sel[j, :]
            k_vec2 = tl.arange(0, HEAD_DIM_CKV)
            Kc_j = tl.load(Kc_sel_ptr + j * HEAD_DIM_CKV + k_vec2, mask=k_vec2 < HEAD_DIM_CKV, other=0.0).to(tl.float32)  # [HEAD_DIM_CKV]
            kpe_vec2 = tl.arange(0, HEAD_DIM_KPE)
            Kp_j = tl.load(Kp_sel_ptr + j * HEAD_DIM_KPE + kpe_vec2, mask=kpe_vec2 < HEAD_DIM_KPE, other=0.0).to(tl.float32)  # [HEAD_DIM_KPE]

            # Dot products
            dot_qn = tl.sum(qn * Kc_j, axis=0)
            dot_qp = tl.sum(qp * Kp_j, axis=0)
            logits[j] = dot_qn + dot_qp  # scalar

        # Scale
        logits = logits * sm_scale

        # Causal mask: for query i, invalid positions j < (prefix_len + i + 1) should be -inf
        prefix_len = kv_len - q_len
        valid_start = prefix_len + i + 1
        causal_mask = tl.arange(0, kv_len) >= valid_start
        logits = tl.where(causal_mask, logits, -float("inf"))

        # logsumexp in log2
        m = tl.max(logits, axis=0)  # scalar
        sumexp = tl.sum(tl.exp(logits - m), axis=0)  # scalar
        lse_val = m + tl.log(sumexp) * ln2_inv  # scalar per head

        # Softmax over j
        logits = logits - m
        exp_logits = tl.exp(logits)
        sumexp = tl.sum(exp_logits, axis=0)  # scalar
        softmax = exp_logits / sumexp  # [kv_len] vector

        # Output: out[h, :] = sum_j softmax[j] * Kc_sel[j, :]
        for j in range(kv_len):
            Kc_j = tl.load(Kc_sel_ptr + j * HEAD_DIM_CKV + tl.arange(0, HEAD_DIM_CKV), mask=tl.arange(0, HEAD_DIM_CKV) < HEAD_DIM_CKV, other=0.0).to(tl.float32)
            out_vec += softmax[j] * Kc_j

        # Store output for this query and head as bfloat16
        out_store = out_vec.to(tl.bfloat16)
        base_out = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV
        # Store vector of length HEAD_DIM_CKV
        for k in range(HEAD_DIM_CKV):
            tl.store(output_ptr + base_out + k, out_store[k])

        # Store lse for this query and head as float32
        lse_base = q_abs * NUM_HEADS + h
        tl.store(lse_ptr + lse_base, lse_val)

        assert kpe_cache.shape[1] == 1 and kpe_cache.shape[2] == 64, "kpe_cache must have [num_pages, 1, 64]"

        total_q, num_qo_heads, head_dim_ckv = q_nope.shape
        head_dim_kpe = q_pe.shape[2]
        # No need to squeeze caches since we already enforce shape in asserts; select per batch
        len_indptr = qo_indptr.shape[0]
        batch_size = len_indptr - 1

        # Output and LSE buffers
        output = torch.empty((total_q, num_qo_heads, head_dim_ckv), dtype=torch.bfloat16, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Precompute ln2_inv
        ln2_inv = 1.0 / math.log(2.0)

        # Loop over batch elements and launch one Triton program per query
        for b in range(batch_size):
            # Number of queries in this batch element
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            q_len = q_end - q_start
            # Number of KV tokens in this batch element
            page_beg = int(kv_indptr[b].item())
            page_end = int(kv_indptr[b + 1].item())
            kv_len = page_end - page_beg
            tok_idx = kv_indices[page_beg:page_end].to(torch.int32).to(device)  # [kv_len]

            # Select Kc_sel and Kp_sel from caches (preserve dtype as bf16, cast to fp32 inside kernel)
            Kc_sel = ckv_cache[tok_idx]  # [kv_len, 512], bf16
            Kp_sel = kpe_cache[tok_idx]  # [kv_len, 64], bf16

            _forward_single_query_kernel[grid](
                q_nope, q_pe, Kc_sel, Kp_sel, output, lse,
                q_start,
                q_len=q_len, kv_len=kv_len,
                sm_scale=sm_scale,
                ln2_inv=ln2_inv,
                NUM_HEADS=16, HEAD_DIM_CKV=512, HEAD_DIM_KPE=64,
                num_warps=4, num_stages=2
            )

        return output, lse

    q_nope_ptr,       # *bf16, [Q_total, 16, 512]
    q_pe_ptr,         # *bf16, [Q_total, 16, 64]
    Kc_sel_ptr,       # *bf16, [kv_len, 512]
    Kp_sel_ptr,       # *bf16, [kv_len, 64]
    output_ptr,       # *bf16, [Q_total, 16, 512]
    lse_ptr,          # *fp32, [Q_total, 16]
    # runtime scalar
    q_start,          # int32
    # constexpr meta-parameters
    q_len: tl.constexpr,     # number of queries in this batch element (meta only, grid controls it)
    kv_len: tl.constexpr,    # number of KV tokens selected for this batch element
    sm_scale: tl.constexpr,  # fp32 scaling factor
    ln2_inv: tl.constexpr,   # fp32 = 1 / ln(2)
    NUM_HEADS: tl.constexpr, # 16
    HEAD_DIM_CKV: tl.constexpr, # 512
    HEAD_DIM_KPE: tl.constexpr, # 64
):
    # Grid: (len_indptr-1, q_len). Decode b and i
    b = tl.program_id(0)
    i = tl.program_id(1)
    q_abs = q_start + i

    # Compute prefix_len for causal mask
    prefix_len = kv_len - q_len
    valid_start = prefix_len + i + 1

    # Iterate over heads and compute everything per head
    for h in range(NUM_HEADS):
        # Load qn[h, :] and qp[h, :]
        k_vec = tl.arange(0, HEAD_DIM_CKV)
        base = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV
        qn = tl.load(q_nope_ptr + base + k_vec, mask=k_vec < HEAD_DIM_CKV, other=0.0).to(tl.float32)  # [512]

        kpe_vec = tl.arange(0, HEAD_DIM_KPE)
        base_pe = q_abs * (NUM_HEADS * HEAD_DIM_KPE) + h * HEAD_DIM_KPE
        qp = tl.load(q_pe_ptr + base_pe + kpe_vec, mask=kpe_vec < HEAD_DIM_KPE, other=0.0).to(tl.float32)  # [64]

        # Initialize logits and output accumulator
        logits = tl.zeros((kv_len,), dtype=tl.float32)
        out_vec = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)

        # Compute logits[j] for all j in [0, kv_len)
        for j in range(kv_len):
            k_vec2 = tl.arange(0, HEAD_DIM_CKV)
            Kc_j = tl.load(Kc_sel_ptr + j * HEAD_DIM_CKV + k_vec2, mask=k_vec2 < HEAD_DIM_CKV, other=0.0).to(tl.float32)  # [512]
            kpe_vec2 = tl.arange(0, HEAD_DIM_KPE)
            Kp_j = tl.load(Kp_sel_ptr + j * HEAD_DIM_KPE + kpe_vec2, mask=kpe_vec2 < HEAD_DIM_KPE, other=0.0).to(tl.float32)  # [64]

            # Dot products
            dot_qn = tl.sum(qn * Kc_j, axis=0)  # scalar
            dot_qp = tl.sum(qp * Kp_j, axis=0)  # scalar
            logits[j] = dot_qn + dot_qp

        # Scale
        logits = logits * sm_scale

        # Causal mask: j >= valid_start -> keep logits, else -inf
        j_vec = tl.arange(0, kv_len)
        causal_mask = j_vec >= valid_start
        logits = tl.where(causal_mask, logits, -float("inf"))

        # logsumexp in log2
        m = tl.max(logits, axis=0)  # scalar
        sumexp = tl.sum(tl.exp(logits - m), axis=0)  # scalar
        lse_val = m + tl.log(sumexp) * ln2_inv  # scalar per head

        # Softmax over j
        logits = logits - m
        exp_logits = tl.exp(logits)
        sumexp = tl.sum(exp_logits, axis=0)  # scalar
        softmax = exp_logits / sumexp  # [kv_len] vector

        # Output: out[h, :] = sum_j softmax[j] * Kc_sel[j, :]
        for j in range(kv_len):
            Kc_j = tl.load(Kc_sel_ptr + j * HEAD_DIM_CKV + tl.arange(0, HEAD_DIM_CKV), mask=tl.arange(0, HEAD_DIM_CKV) < HEAD_DIM_CKV, other=0.0).to(tl.float32)
            out_vec += softmax[j] * Kc_j

        # Store output vector for this query and head as bfloat16
        out_store = out_vec.to(tl.bfloat16)
        base_out = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV
        for k in range(HEAD_DIM_CKV):
            tl.store(output_ptr + base_out + k, out_store[k])

        # Store lse for this query and head as float32
        lse_base = q_abs * NUM_HEADS + h
        tl.store(lse_ptr + lse_base, lse_val)

    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Shape assertions matching original assumptions

        assert ckv_cache.shape[1] == 1 and ckv_cache.shape[2] == 512, "ckv_cache must be [num_pages, 1, 512]"
        assert kpe_cache.shape[1] == 1 and kpe_cache.shape[2] == 64, "kpe_cache must be [num_pages, 1, 64]"
        assert qo_indptr.dim() == 1 and kv_indptr.dim() == 1, "indptrs must be 1D"

        device = q_nope.device
        total_q, num_qo_heads, head_dim_ckv = q_nope.shape
        head_dim_kpe = q_pe.shape[2]
        len_indptr = qo_indptr.shape[0]
        batch_size = len_indptr - 1

        # Output and LSE buffers

        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Precompute ln2_inv
