@triton.jit
def compute_dot_qn_Kc(
    qn_ptr,           # *f32, shape [H, Dc], but we pass slices for h
    Kc_ptr,           # *f32, shape [L, Dc]
    dot_ptr,          # *f32, shape [H, L], 2D tensor to store results
    H: tl.constexpr,  # number of heads (here 16)
    Dc: tl.constexpr, # head_dim_ckv (512)
    L: tl.constexpr,  # number of tokens in this batch segment
    qn_h_offset: tl.constexpr,  # offset for h index (we'll pass h value)
    grid_dim0: tl.constexpr     # we'll set grid as (H, L)
):

    h = tl.program_id(0)  # not used due to constexpr grid, but ok
    t = tl.program_id(1)

    # Base offsets
    # qn_ptr is [H, Dc], but we pass per-h slice; here we don't need qn_ptr directly since we pass qn[h, :].
    # Instead, we load qn[h, :] once and compute dot.
    # However, Triton kernels receive raw pointers. We'll compute qn[h, :] by reading from qn_ptr base + h*Dc.

    # We will redesign to pass qn[h, :] as a 1D pointer, but Triton requires pointer arguments to be tensors. So we will use:
    # We'll pass qn_ptr as 1D pointer by flattening qn. For clarity, we'll pass qn[h, :] as a separate kernel with qn_ptr + h*Dc.

    # Redesign: We'll pass qn_ptr with shape (H, Dc) and index by h. Triton allows indexing with integer offsets.
    # So qn_ptr + h * Dc gives the base pointer for that head.

    qn_base = qn_ptr + h * Dc  # base pointer for this head
    Kc_row_ptr = Kc_ptr + t * Dc  # base pointer for this token row

    acc = 0.0

    tl.store(dot_ptr + h * L + t, acc)

@triton.jit
def compute_dot_qp_Kp(
    qp_ptr,           # *f32, shape [H, Dp]
    Kp_ptr,           # *f32, shape [L, Dp]
    dot_ptr,          # *f32, shape [H, L]

    Dp: tl.constexpr, # head_dim_kpe (64)
    L: tl.constexpr,
    qp_h_offset: tl.constexpr,  # offset for h index
    grid_dim0: tl.constexpr
):

    t = tl.program_id(1)

    qp_base = qp_ptr + h * Dp
    Kp_row_ptr = Kp_ptr + t * Dp

    acc = 0.0

        mask_d = offs_d < Dp
        qp_vec = tl.load(qp_base + offs_d, mask=mask_d, other=0.0)  # [64]

        acc += tl.sum(qp_vec * Kp_vec, axis=0)
    tl.store(dot_ptr + h * L + t, acc)


# Kernel 3: compute logits, lse, attn for each head h
@triton.jit
def compute_logits_lse_attn(
    dot_qn_ptr,       # *f32, shape [H, L]
    dot_qp_ptr,       # *f32, shape [H, L]
    lse_out_ptr,      # *f32, shape [H]
    attn_out_ptr,     # *f32, shape [H, L]
    H: tl.constexpr,
    L: tl.constexpr,
    sm_scale: tl.constexpr,  # float32 scalar
    i_abs: tl.constexpr      # absolute query position for causal mask
):
    h = tl.program_id(0)  # one program per head
    # Load dot_qn and dot_qp for this head
    row_base = 0
    logits = tl.zeros((L,), dtype=tl.float32)
    for t in range(0, L):
        v = tl.load(dot_qn_ptr + h * L + t) + tl.load(dot_qp_ptr + h * L + t)
        logits[t] = v
    # Causal mask: only tokens t <= i_abs allowed
    for t in range(0, L):
        if t > i_abs:
            logits[t] = -float('inf')

    # Compute lse per head: lse = log(sum(exp(logits)) * sm_scale)
    # Using stable: lse = log(sum(exp(logits - max))) + max + log(sm_scale)
    row_max = -float('inf')
    for t in range(0, L):
        row_max = tl.maximum(row_max, logits[t])
    sumexp = 0.0
    for t in range(0, L):
        sumexp += tl.exp(logits[t] - row_max)
    lse = row_max + math.log(2.0) * sm_scale  # base-2 logsumexp, but we use natural log; we'll multiply by log(2)
    # Correct: lse should be log(sumexp) / log(2) + row_max * sm_scale is not needed; we will compute lse = log(sumexp) + row_max, then scale.
    # We need base-2 lse: lse = (row_max + log(sumexp)) / log(2)
    lse = row_max + math.log(sumexp) / math.log(2.0)
    tl.store(lse_out_ptr + h, lse)

    # Compute attn = exp(logits - lse)
    denom = 0.0
    for t in range(0, L):
        denom += tl.exp(logits[t] - lse)
    for t in range(0, L):
        attn_out_ptr[h * L + t] = tl.exp(logits[t] - lse) / denom


# Kernel 4: output[i, h, :] = sum_t attn[h, t] * Kc[t, :]
@triton.jit
def matmul_vec_by_mat(
    attn_ptr,         # *f32, shape [H, L]
    Kc_ptr,           # *f32, shape [L, Dc]
    out_ptr,          # *f32, shape [Dc] (we'll store per h into output[i, h, :])
    H: tl.constexpr,
    L: tl.constexpr,
    Dc: tl.constexpr, # head_dim_ckv (512)
    i_idx: tl.constexpr,   # query index (for output[i, h, :], not used here directly)
    h_idx: tl.constexpr     # head index
):
    h = h_idx
    out_vec = tl.zeros((Dc,), dtype=tl.float32)
    for t in range(0, L, 64):
        offs_t = t + tl.arange(0, 64)
        mask_t = offs_t < L
        attn_vec = tl.load(attn_ptr + h * L + offs_t, mask=mask_t, other=0.0)  # [64]
        Kc_block_ptr = Kc_ptr + offs_t * Dc  # [64, Dc] but we load row-wise
        acc_partial = tl.zeros((Dc,), dtype=tl.float32)
        for d in range(0, Dc, 64):
            offs_d = d + tl.arange(0, 64)
            mask_d = offs_d < Dc
            Kc_row = tl.load(Kc_block_ptr + offs_d, mask=mask_d, other=0.0)  # [64]
            acc_partial += tl.sum(attn_vec[:, None] * Kc_row[None, :], axis=0)
        out_vec += acc_partial
    # Store into output[i, h, :]
    # We'll assume out_ptr points to the flattened output tensor: output[i, h, :] at linear offset i*H*Dc + h*Dc
    base_out = i_idx * H * Dc + h * Dc
    for d in range(0, Dc):
        tl.store(out_ptr + d, out_vec[d])

    L: tl.constexpr,
    sm_scale: tl.constexpr,
    i_abs: tl.constexpr  # absolute query position
):
    h = tl.program_id(0)  # one program per head
    logits = tl.zeros((L,), dtype=tl.float32)
    for t in range(0, L):
        logits[t] = tl.load(dot_qn_ptr + h * L + t) + tl.load(dot_qp_ptr + h * L + t)
    # causal mask
    for t in range(0, L):
        if t > i_abs:
            logits[t] = -float('inf')

    row_max = -float('inf')
    for t in range(0, L):
        row_max = tl.maximum(row_max, logits[t])
    sumexp = 0.0
    for t in range(0, L):
        sumexp += tl.exp(logits[t] - row_max)
    # base-2 logsumexp
    lse = row_max + math.log(sumexp) / math.log(2.0)
    tl.store(lse_out_ptr + h, lse)

    denom = 0.0
    for t in range(0, L):
        denom += tl.exp(logits[t] - lse)
    for t in range(0, L):
        attn_out_ptr[h * L + t] = tl.exp(logits[t] - lse) / denom


@triton.jit
def matmul_vec_by_mat(
    attn_ptr,         # *f32, shape [H, L], flattened
    Kc_ptr,           # *f32, shape [L, Dc], contiguous
    out_ptr,          # *f32, shape [Dc]
    H: tl.constexpr,
    L: tl.constexpr,
    Dc: tl.constexpr,
    h_idx: tl.constexpr
):
    h = h_idx
    out_vec = tl.zeros((Dc,), dtype=tl.float32)
    for t in range(0, L, 64):
        offs_t = t + tl.arange(0, 64)
        mask_t = offs_t < L
        attn_vec = tl.load(attn_ptr + h * L + offs_t, mask=mask_t, other=0.0)  # [64]
        acc_partial = tl.zeros((Dc,), dtype=tl.float32)
        for d in range(0, Dc, 64):
            offs_d = d + tl.arange(0, 64)
            mask_d = offs_d < Dc
            Kc_row = tl.load(Kc_ptr + offs_t * Dc + offs_d, mask=mask_d, other=0.0)  # [64]
            acc_partial += tl.sum(attn_vec[:, None] * Kc_row[None, :], axis=0)
        out_vec += acc_partial
    for d in range(0, Dc):
        tl.store(out_ptr + d, out_vec[d])

    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Move inputs to CUDA (assumed available)
        assert q_nope.is_cuda and q_pe.is_cuda and ckv_cache.is_cuda and kpe_cache.is_cuda, "Triton requires CUDA tensors"
        device = q_nope.device

        # Shapes

        _, _, Dp = q_pe.shape
        len_indptr = qo_indptr.numel()

        lse_out = torch.empty((total_q, H), dtype=torch.float32, device=device)  # placeholder, not used in return

        # For each batch element
        for b in range(batch_size):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            if q_start >= q_end:
                continue

            # Build tok_idx for this batch element: indices into kv_indptr window
            tok_idx = kv_indices[kv_indptr[b]: kv_indptr[b + 1]].to(torch.int32)  # [L]
