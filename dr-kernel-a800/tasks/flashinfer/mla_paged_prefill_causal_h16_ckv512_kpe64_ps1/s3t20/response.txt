import torch
import triton
import triton.language as tl


# Kernel: compute logits_scaled[h, :] for a given head h.
# It reads qn_vec (flattened) and qp_vec (flatted), and tok_idx_ptr (int32).
# For each l in [0, L), it accumulates over features:
#   logits[l] = dot(qn[h, :], Kc[tok_idx[l], :]) + dot(qp[h, :], Kp[tok_idx[l], :])
# Then scales by sm_scale and writes to logits_scaled_ptr[h, :].
@triton.jit
def compute_logits_kernel(
    qn_vec_ptr, qp_vec_ptr, Kc_ptr, Kp_ptr, tok_idx_ptr, logits_scaled_ptr,
    K: tl.constexpr,          # head_dim_ckv (512)
    Kp: tl.constexpr,         # head_dim_kpe (64)
    L: tl.constexpr,          # number of tokens in this segment
    sm_scale,                 # fp32 scalar
    head: tl.constexpr,       # which head (0..H-1)
):
    # Offsets for feature loops
    k_offsets = tl.arange(0, K)
    kp_offsets = tl.arange(0, Kp)
    l_offsets = tl.arange(0, L)

    # Accumulator for logits_scaled[h, :]
    logits = tl.zeros((L,), dtype=tl.float32)

    # For each token l, compute dot products
    for l in range(0, L):
        idx = tok_idx_ptr[l]
        # Accumulate over K features for q_nope
        # qn_vec layout: [H*K], we select row for head 'head'
        # Index qn_vec[head*K + k] for k in 0..K-1
        acc_qn = 0.0
        for k in range(0, K):
            qk = tl.load(qn_vec_ptr + head * K + k)
            kc = tl.load(Kc_ptr + idx * K + k)
            acc_qn += qk * kc

        # Accumulate over Kp features for q_pe
        acc_qp = 0.0
        for kp in range(0, Kp):
            qk = tl.load(qp_vec_ptr + head * Kp + kp)
            kpval = tl.load(Kp_ptr + idx * Kp + kp)
            acc_qp += qk * kpval

        logits[l] = acc_qn + acc_qp

    # Scale
    logits *= sm_scale

    # Store row h
    # logits_scaled_ptr is a 2D [H, L], row index is head (constexpr), col is l
    stride_log_h = L
    stride_log_l = 1
    for l in range(0, L):
        tl.store(logits_scaled_ptr + head * stride_log_h + l * stride_log_l, logits[l])


# Kernel: compute lse[h] = logsumexp(logits_scaled[h, :]) / ln(2).
# It zeros invalid positions before reduction to avoid NaNs.
@triton.jit
def compute_lse_kernel(
    logits_scaled_ptr, lse_ptr,
    L: tl.constexpr,
    sm_scale,                 # not used here, but kept for potential future usage
    head: tl.constexpr,
):
    prefix_len = L  # In our usage, prefix_len == L (no skipped tokens yet).
    # Initialize with -inf then override invalid positions with 0. We'll do explicit max and masked fill after.
    max_val = -float("inf")
    for l in range(0, L):
        val = tl.load(logits_scaled_ptr + head * L + l)
        if (l > (prefix_len - 1)):  # invalid position: causal mask
            val = 0.0
        if val > max_val:
            max_val = val

    sum_exp = 0.0
    ln2 = 1.4426950408889634  # log(2)
    for l in range(0, L):
        val = tl.load(logits_scaled_ptr + head * L + l)
        if l > (prefix_len - 1):
            val = 0.0
        sum_exp += tl.exp(val - max_val)
    lse = tl.log(sum_exp) + max_val
    lse = lse / ln2
    tl.store(lse_ptr + head, lse)


# Kernel: compute attn[h, :] = softmax(logits_scaled[h, :]) with invalid positions zeroed.
@triton.jit
def compute_softmax_kernel(
    logits_scaled_ptr, lse_ptr, attn_ptr,
    L: tl.constexpr,
    head: tl.constexpr,
):
    ln2 = 1.4426950408889634
    lse = tl.load(lse_ptr + head) * ln2
    for l in range(0, L):
        val = tl.load(logits_scaled_ptr + head * L + l)
        if l > (L - 1):  # all positions valid for L; but we keep syntax flexible
            val = 0.0
        attn = tl.exp(val - lse)
        tl.store(attn_ptr + head * L + l, attn)


# Kernel: compute output[h, :] = attn[h, :] @ Kc_all[tok_idx[:], :] (GEMV). Kc is [L, K].
@triton.jit
def gemv_out_kernel(
    attn_ptr, Kc_ptr, tok_idx_ptr, out_ptr,
    K: tl.constexpr, L: tl.constexpr,
    head: tl.constexpr,
):
    acc = tl.zeros((K,), dtype=tl.float32)
    # Compute acc = attn[h, :] dot Kc[tok_idx[:], :]
    # We need to know that attn[h, :] is a vector of length L. We'll iterate l and multiply with Kc rows.
    for l in range(0, L):
        attn_val = tl.load(attn_ptr + head * L + l)
        idx = tl.load(tok_idx_ptr + l)
        # Kc row: [K]
        kc_row = tl.load(Kc_ptr + idx * K + tl.arange(0, K))
        acc += attn_val * kc_row
    # Store output row for head h
    stride_out_h = K
    stride_out_k = 1
    for k in range(0, K):
        tl.store(out_ptr + head * stride_out_h + k * stride_out_k, acc[k])


def run(q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
    # Ensure on CUDA
    device = q_nope.device
    # Squeeze caches to [P, K] and [P, Kp]
    K = 512
    Kp = 64
    H = 16  # asserted in original code
    Kc_all = ckv_cache.squeeze(1).to(torch.float32)  # [P, K]
    Kp_all = kpe_cache.squeeze(1).to(torch.float32)  # [P, Kp]

    total_q = q_nope.shape[0]
    # Output buffers
    output = torch.empty((total_q, H, K), dtype=torch.bfloat16, device=device)
    lse = torch.empty((total_q, H), dtype=torch.float32, device=device)

    # Process each batch b: segments defined by qo_indptr and kv_indptr
    for b in range(0, qo_indptr.shape[0] - 1):
        q_start = int(qo_indptr[b].item())
        q_end = int(qo_indptr[b + 1].item())
        q_len = q_end - q_start

        if q_len <= 0:
            continue

        # Compute tok_idx for this batch segment
        tok_idx = kv_indices[int(kv_indptr[b].item()): int(kv_indptr[b + 1].item())].to(torch.int32).contiguous()  # [L]
        L = tok_idx.numel()

        # Iterate over queries in this batch
        for i in range(0, q_len):
            q_abs = q_start + i

            # Prepare flattened vectors for q_nope and q_pe at head dimension
            # q_nope shape: [Q, H, K], q_pe: [Q, H, Kp]
            qn_vec = q_nope[q_abs].reshape(-1).contiguous().to(torch.float32)  # length H*K = 8192
            qp_vec = q_pe[q_abs].reshape(-1).contiguous().to(torch.float32)    # length H*Kp = 1024

            # Allocate temporary buffers for per-head computations
            # Logits scaled for head h
            logits_scaled = torch.empty((H, L), dtype=torch.float32, device=device)

            # Launch compute_logits_kernel for each head h
            # Grid size: one program per head
            for h in range(0, H):
                compute_logits_kernel[(1,)](
                    qn_vec, qp_vec, Kc_all, Kp_all, tok_idx, logits_scaled,
                    K=K, Kp=Kp, L=L, sm_scale=float(sm_scale), head=h,
                )

            # Compute lse for each head using compute_lse_kernel
            for h in range(0, H):
                compute_lse_kernel[(1,)](
                    logits_scaled, lse[q_abs], L=L, sm_scale=float(sm_scale), head=h,
                )

            # Compute attn for each head using compute_softmax_kernel
            attn = torch.empty((H, L), dtype=torch.float32, device=device)
            for h in range(0, H):
                compute_softmax_kernel[(1,)](
                    logits_scaled, lse[q_abs], attn,
                    L=L, head=h,
                )

            # Compute output for each head using gemv_out_kernel
            out_row = torch.empty((K,), dtype=torch.float32, device=device)
            for h in range(0, H):
                gemv_out_kernel[(1,)](
                    attn[h], Kc_all, tok_idx, out_row,
                    K=K, L=L, head=h,
                )
                # Store output row h for this query
                output[q_abs, h] = out_row.to(torch.bfloat16)

    return output, lse

# - Computes out[h, :] = softmax(h, :) @ Kc[tok_idx[:], :] (GEMV for L tokens).
# Note: We only compute one head h per launch (grid=(1,)), but we can call it multiple times for different h.
@triton.jit
def fused_q_head_kernel(

    qp_vec_ptr,        # *fp32, length H*Kp

    Kp_ptr,            # *fp32, base pointer to [P, Kp] flattened

    kp_offsets = tl.arange(0, Kp)
    l_offsets = tl.arange(0, L)

    # First, compute logits_scaled[h, :] as a vector of length L
    logits_scaled = tl.zeros((L,), dtype=tl.float32)

- Avoid complex multi-dimensional Triton kernels and loops that may lead to missing argument or compilation issues. Use simple Triton kernels that each perform a single, clear operation:
  1) compute logits_scaled[h, :] for one head h and one query index i. This kernel loops over tokens L and accumulates dot products for q_nope and q_pe into a 1D vector of length L. It does not write to an [H, L] matrix; instead, forward computes per-head separately.
  2) compute softmax per head on logits_scaled[h, :]. To ensure Triton-only correctness, implement a masked softmax that zeros invalid positions (causal) and computes logsumexp safely without -inf. This kernel writes attn[h, :] to a 1D tensor.
  3) compute output row for one head h: out[h, :] = attn[h, :] @ Kc_selected_rows. Implement a small GEMV-like Triton kernel that loops over L and accumulates dot products with Kc rows for tok_idx. This kernel assumes K and L are constexpr; Triton can unroll for performance.

Forward logic:
- Ensure inputs are on CUDA and tensors are contiguous.
- Squeeze caches to [P, K] and [P, Kp].
- For each batch b and each query i:
  - Build tok_idx = kv_indices[kv_indptr[b]: kv_indptr[b+1]].to(torch.int32).contiguous() and pass to kernels.
  - Prepare qn_vec = q_nope[q_abs].reshape(-1).contiguous().to(torch.float32) and qp_vec = q_pe[q_abs].reshape(-1).contiguous().to(torch.float32) and pass to kernels.
  - For each head h:
    - Launch compute_logits_kernel to fill logits_scaled[h, :].
    - Launch compute_softmax_kernel to produce attn[h, :].
    - Launch gemv_out_kernel to compute out[h, :] and store in output[q_abs, h].

This avoids torch matmul/softmax/reduction in forward and ensures Triton is actually invoked. While not the fastest (many kernels per batch), it should be correct and addresses the earlier missing argument errors by matching kernel signatures and passing tok_idx_ptr, qn_vec_ptr, qp_vec_ptr, out_ptr.

import torch
import triton
import triton.language as tl


# Triton kernel: compute logits_scaled[h, :] for a single head h and query i.
# Inputs:
#   qn_vec_ptr: *fp32, length H*K (H=16, K=512), we read qn[h, :] as qn_vec[head*K + k]
#   qp_vec_ptr: *fp32, length H*Kp (Kp=64), we read qp[h, :] as qp_vec[head*Kp + kp]
#   Kc_ptr: *fp32, flattened [P, K], indexed by tok_idx[l] and feature k
#   Kp_ptr: *fp32, flattened [P, Kp], indexed by tok_idx[l] and feature kp
#   tok_idx_ptr: *int32, length L
#   out_vec_ptr: *fp32, length L (logits_scaled[h, :])
#   L: number of tokens in this segment
#   K: constexpr, 512
#   Kp: constexpr, 64
#   sm_scale: fp32 scalar
#   head: constexpr head index
@triton.jit
def compute_logits_kernel(
    qn_vec_ptr, qp_vec_ptr, Kc_ptr, Kp_ptr, tok_idx_ptr, out_vec_ptr,
    K: tl.constexpr,        # 512
    Kp: tl.constexpr,       # 64
    L,                      # runtime integer
    sm_scale,               # fp32 scalar
    head: tl.constexpr,     # head index (0..H-1)
):
    # Accumulate logits for each token l
    for l in range(0, L):
        idx = tl.load(tok_idx_ptr + l)  # int32 token index
        acc_qn = 0.0
        # sum over K features for q_nope
        for k in range(0, K):
            qk = tl.load(qn_vec_ptr + head * K + k)
            kc = tl.load(Kc_ptr + idx * K + k)
            acc_qn += qk * kc
        acc_qp = 0.0
        # sum over Kp features for q_pe
        for kp in range(0, Kp):
            qk = tl.load(qp_vec_ptr + head * Kp + kp)
            kpval = tl.load(Kp_ptr + idx * Kp + kp)
            acc_qp += qk * kpval
        logits_scaled = acc_qn + acc_qp
        logits_scaled *= sm_scale
        tl.store(out_vec_ptr + l, logits_scaled)


# Triton kernel: compute softmax per head on a vector of length L, with causal mask.
# We zero out invalid positions (l > prefix_len + i) before reduction.
# Inputs:
#   in_ptr: *fp32, length L (logits_scaled[h, :])
#   tok_idx_ptr: *int32, length L (not used for softmax, but kept for consistency)
#   out_ptr: *fp32, length L (attn[h, :])
#   L: runtime integer
#   prefix_len: runtime integer, number of valid tokens before this query (L - q_len)
#   sm_scale: fp32 scalar (unused)
#   head: constexpr (unused)
@triton.jit
def compute_softmax_kernel(
    in_ptr, tok_idx_ptr, out_ptr,
    L, prefix_len, sm_scale,
    head: tl.constexpr,
):
    # Compute max for numerical stability
    max_val = -float("inf")
    for l in range(0, L):
        val = tl.load(in_ptr + l)
        if l > (prefix_len - 1):  # causal: invalid, set to -inf
            val = -float("inf")
        if val > max_val:
            max_val = val

    sum_exp = 0.0
    for l in range(0, L):
        val = tl.load(in_ptr + l)
        if l > (prefix_len - 1):
            val = -float("inf")
        sum_exp += tl.exp(val - max_val)

    ln2 = 1.4426950408889634
    inv_ln2 = 1.0 / ln2
    for l in range(0, L):
        val = tl.load(in_ptr + l)
        if l > (prefix_len - 1):
            val = -float("inf")
        attn = tl.exp(val - max_val) / sum_exp
        attn *= inv_ln2  # original uses base-2 logsumexp; normalize by 1/ln(2)
        tl.store(out_ptr + l, attn)
