```python
import torch
import math

import triton
import triton.language as tl


# Triton kernels: all computational steps must be done by these kernels.
# 1) Row GEMV: y = q_row @ K.T, where q_row is [D_in], K is [M, D_in], y is [M]
@triton.jit
def row_gemv_kernel(q_ptr,  # *float32, row vector of length D_in
                    K_ptr,  # *float32, matrix [M, D_in]
                    y_ptr,  # *float32, output vector [M]
                    M: tl.constexpr,  # number of rows in K (number of tokens)
                    D_in: tl.constexpr,  # length of the row vector
                    BLOCK_M: tl.constexpr):
    # Each program handles one output element along M
    pid_m = tl.program_id(0)
    # Accumulator
    acc = tl.zeros([1], dtype=tl.float32)
    # Loop over the row dimension with BLOCK_M tiles
    # Note: Triton for-loop requires compile-time bounds; here we iterate from 0..M-1
    for off in range(0, M):
        # Load q_row element
        q = tl.load(q_ptr + off)  # q_ptr points to the row; load q[off]
        # Compute dot with K[off, :] (load K row at index off)
        # We need to load K[off, :] across D_in dimension
        dot = tl.zeros([1], dtype=tl.float32)
        for k in range(0, D_in):
            k_val = tl.load(K_ptr + off * D_in + k)
            dot += q * k_val
        acc += dot
    # Store result
    tl.store(y_ptr + pid_m, acc)


# 2) Scale 1D vector by a scalar
@triton.jit
def scale_1d_kernel(x_ptr,  # *float32
                    y_ptr,  # *float32
                    scale: tl.float32,
                    N: tl.constexpr):
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        yi = xi * scale
        tl.store(y_ptr + i, yi)


# 3) Apply causal mask: y = x if j > pos else -inf, for 1D vector
@triton.jit
def mask_neg_inf_1d_kernel(x_ptr,  # *float32
                            y_ptr,  # *float32
                            N: tl.constexpr,
                            pos: tl.int32):
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        keep = i > pos
        yi = tl.where(keep, xi, -float("inf"))
        tl.store(y_ptr + i, yi)


# 4) LSE of 1D masked vector: lse = log(sum(exp(x))) / ln(2)
@triton.jit
def lse_row_kernel(x_ptr,  # *float32, masked logits
                    out_ptr,  # *float32, 1-element tensor
                    N: tl.constexpr):
    # Compute max
    max_val = tl.full([1], -float("inf"), dtype=tl.float32)
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        max_val = tl.maximum(max_val, xi)
    # Compute sum(exp(x - max))
    sum_exp = tl.zeros([1], dtype=tl.float32)
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        sum_exp += tl.exp(xi - max_val)
    # LSE in base 2
    ln2 = 0.6931471805599453
    lse = tl.log(sum_exp) / ln2 + max_val
    tl.store(out_ptr, lse)


# 5) Softmax with causal mask: out[i] = exp(x[i] - max)/sum over kept elements
@triton.jit
def softmax_masked_row_kernel(x_ptr,  # *float32, masked logits
                               out_ptr,  # *float32
                               N: tl.constexpr,
                               pos: tl.int32):
    # Compute max
    max_val = tl.full([1], -float("inf"), dtype=tl.float32)
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        max_val = tl.maximum(max_val, xi)
    # Compute sum of exp for kept elements
    sum_exp = tl.zeros([1], dtype=tl.float32)
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        keep = i > pos
        sum_exp += tl.where(keep, tl.exp(xi - max_val), 0.0)
    # Compute normalized output
    for i in range(0, N):
        xi = tl.load(x_ptr + i)
        keep = i > pos
        numerator = tl.where(keep, tl.exp(xi - max_val), 0.0)
        out_i = numerator / sum_exp
        tl.store(out_ptr + i, out_i)


# 6) GEMV of 1D vector with 2D matrix (columns): out = softmax @ Kc.T
#    Here softmax is [M], Kc is [M, Dn], out is [Dn]
@triton.jit
def gemv_row_kernel(softmax_ptr,  # *float32, 1D vector [M]
                    K_ptr,         # *float32, matrix [M, D_in]
                    out_ptr,       # *float32, output vector [D_in]
                    M: tl.constexpr,
                    D_in: tl.constexpr,
                    BLOCK_M: tl.constexpr):
    # Each program handles one output element along D_in
    pid = tl.program_id(0)
    acc = tl.zeros([1], dtype=tl.float32)
    for off in range(0, M):
        s = tl.load(softmax_ptr + off)
        dot = tl.zeros([1], dtype=tl.float32)
        for k in range(0, D_in):
            k_val = tl.load(K_ptr + off * D_in + k)
            dot += s * k_val
        acc += dot
    tl.store(out_ptr + pid, acc)

        super().__init__()

        device = q_nope.device
        assert device.type == "cuda", "ModelNew.forward requires CUDA tensors"
        # Constants
        num_qo_heads = 16
        head_dim_ckv = 512
        head_dim_kpe = 64
        # Gather batch size and total_q
        len_indptr = qo_indptr.shape[0]
        batch_size = len_indptr - 1
        total_q = int(qo_indptr[-1].item())

        # Prepare output and lse
        output = torch.empty((total_q, num_qo_heads, head_dim_ckv), dtype=torch.bfloat16, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        for b in range(batch_size):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            q_len = q_end - q_start

            # kv range
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())
            kv_len = kv_end - kv_start

            # If nothing to do for this batch, skip
            if q_len <= 0 or kv_len <= 0:
                continue

            # Gather Kc and Kp for this batch using indices
            tok_idx = kv_indices[kv_start:kv_end].to(torch.int32).to(device)
            Kc = ckv_cache[tok_idx].contiguous().to(torch.float32)  # [kv_len, head_dim_ckv]
            Kp = kpe_cache[tok_idx].contiguous().to(torch.float32)  # [kv_len, head_dim_kpe]

            # prefix_len: number of previously cached tokens (tokens before current query)
            prefix_len = kv_len - q_len

            # Loop over queries in this batch
            for i in range(q_len):
                # Process each head h
                for h in range(num_qo_heads):
                    # Load qn and qp row vectors (float32 for kernels)
                    qn_vec = q_nope[q_start + i, h, :].to(torch.float32).contiguous()  # [512]
                    qp_vec = q_pe[q_start + i, h, :].to(torch.float32).contiguous()   # [64]

                    # Compute qn_logits and qp_logits via Triton GEMV
                    qn_logits = torch.empty(kv_len, dtype=torch.float32, device=device)
                    # Launch Triton GEMV for qn
                    row_gemv_kernel[(kv_len,)](qn_vec, Kc, qn_logits, kv_len, head_dim_ckv, BLOCK_M=128)

                    qp_logits = torch.empty(kv_len, dtype=torch.float32, device=device)
                    # Kp shape is [kv_len, 64]; treat D_in=64 for this vector
                    row_gemv_kernel[(kv_len,)](qp_vec, Kp, qp_logits, kv_len, head_dim_kpe, BLOCK_M=128)

                    logits = qn_logits + qp_logits  # [kv_len], float32

                    # Scale logits by sm_scale
                    logits_scaled = torch.empty(kv_len, dtype=torch.float32, device=device)
                    scale_1d_kernel[(kv_len,)](logits, logits_scaled, sm_scale, kv_len)

                    # Apply causal mask: j > (prefix_len + i)
                    query_abs_pos = prefix_len + i
                    logits_masked = torch.empty(kv_len, dtype=torch.float32, device=device)
                    mask_neg_inf_1d_kernel[(kv_len,)](logits_scaled, logits_masked, kv_len, query_abs_pos)

                    # Compute LSE (base 2): store per head
                    lse_val = torch.empty(1, dtype=torch.float32, device=device)
                    lse_row_kernel[(1,)](logits_masked, lse_val, kv_len)
                    lse[q_start + i, h] = lse_val[0]

                    # Softmax over masked logits
                    softmax_out = torch.empty(kv_len, dtype=torch.float32, device=device)
                    softmax_masked_row_kernel[(kv_len,)](logits_masked, softmax_out, kv_len, query_abs_pos)

                    # Final GEMV: out[h, :] = softmax_out @ Kc
                    out_row = torch.empty(head_dim_ckv, dtype=torch.float32, device=device)
                    gemv_row_kernel[(head_dim_ckv,)](softmax_out, Kc, out_row, kv_len, head_dim_ckv, BLOCK_M=128)

                    # Store output as bfloat16
                    output[q_start + i, h, :] = out_row.to(torch.bfloat16)

        return output, lse
```

```python
                    mask_neg_inf_1d_kernel[(kv_len,)](logits_scaled, logits_masked, kv_len, query_abs_pos)

                    # 5) Compute lse per head (base 2): lse[h] = logsumexp(logits_masked)
                    lse_val = torch.empty(1, dtype=torch.float32, device=device)
                    lse_row_kernel[(1,)](logits_masked, lse_val, kv_len)
                    lse[q_start + i, h] = lse_val[0]

                    # 6) Softmax over masked logits
                    softmax_out = torch.empty(kv_len, dtype=torch.float32, device=device)
                    softmax_masked_row_kernel[(kv_len,)](logits_masked, softmax_out, kv_len, query_abs_pos)

                    # 7) Final GEMV: out[h, :] = softmax_out @ Kc
                    out_row = torch.empty(head_dim_ckv, dtype=torch.float32, device=device)
                    gemv_row_kernel[(head_dim_ckv,)](softmax_out, Kc, out_row, kv_len, head_dim_ckv)

                    # Store output as bfloat16
                    output[q_start + i, h, :] = out_row.to(torch.bfloat16)

        return output, lse
```
