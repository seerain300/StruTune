        Then store into logits[i, h, :].
        Grid: (q_len, H)
        """
        i = tl.program_id(0)
        h = tl.program_id(1)

        # Read qn[h] and qp[h] for this query i
        qn = tl.load(q_nope + i * H * head_dim_ckv + h * head_dim_ckv + tl.arange(0, head_dim_ckv), mask=True, other=0.0)  # [512]
        qp = tl.load(q_pe + i * H * head_dim_kpe + h * head_dim_kpe + tl.arange(0, head_dim_kpe), mask=True, other=0.0)  # [64]

        acc = tl.zeros((head_dim_ckv,), dtype=tl.float32)

        # Loop over L tokens
        for t in range(L):
            Kc_row = tl.load(Kc_all + t * head_dim_ckv + tl.arange(0, head_dim_ckv), mask=True, other=0.0)  # [512]
            Kp_row = tl.load(Kp_all + t * head_dim_kpe + tl.arange(0, head_dim_kpe), mask=True, other=0.0)  # [64]
            dot_qn = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot_qp = tl.sum(qp * Kp_row, axis=0)  # scalar
            acc += dot_qn + dot_qp  # [512]

        # Store acc into logits[i, h, :]
        tl.store(logits + i * H * L + h * L + tl.arange(0, L), acc, mask=True)


if TRITON_AVAILABLE:
    @triton.jit
    def lse_and_attn_1d(logits, lse,
                        q_len: tl.constexpr, L: tl.constexpr, H: tl.constexpr,
                        SM_SCALE: tl.constexpr):
        """
        For each (i, h), compute:
          logits_scaled = logits[i, h, :] * SM_SCALE
          Apply causal mask: allow positions k > (L - q_len + i), else -inf
          lse[i, h] = log(sum(exp(logits_scaled - max))) / log(2)
        Grid: (q_len, H)
        """
        i = tl.program_id(0)
        h = tl.program_id(1)

        # Read logits[i, h, :]
        logits_vec = tl.load(logits + i * H * L + h * L + tl.arange(0, L), mask=True, other=-float("inf"))
        # Scale
        logits_vec = logits_vec * SM_SCALE
        # Max for stability
        max_val = tl.max(logits_vec, axis=0)
        logits_shift = logits_vec - max_val
        exp_vals = tl.exp(logits_shift)
        # Causal mask: k > query_abs_pos, where query_abs_pos = (L - q_len) + i
        query_abs_pos = L - q_len + i
        mask = tl.arange(0, L) > query_abs_pos
        exp_vals = tl.where(mask, exp_vals, 0.0)
        sumexp = tl.sum(exp_vals, axis=0)
        lse[i * H + h] = tl.log(sumexp) / math.log(2.0)


if TRITON_AVAILABLE:
    @triton.jit
    def matmul_vec_by_mat(softmax_vector, Kc, out,
                           q_len: tl.constexpr, H: tl.constexpr, L: tl.constexpr,
                           head_dim_ckv: tl.constexpr, BLOCK_COL: tl.constexpr):
        """
        For each (i, h), compute out[i, h, :] = softmax_vector[i, h, :] @ Kc.T
        softmax_vector[i, h, :] is length L (float32). Kc is [L, head_dim_ckv].
        Grid: (q_len, H, ceil_div(head_dim_ckv, BLOCK_COL))
        """
        i = tl.program_id(0)
        h = tl.program_id(1)
        col_block = tl.program_id(2)

        # Load softmax vector for this (i, h)
        vec = tl.load(softmax_vector + i * H * L + h * L + tl.arange(0, L), mask=True, other=0.0)  # [L]

        # Initialize output row for (i, h)
        out_row = tl.zeros((head_dim_ckv,), dtype=tl.float32)

        # Loop over Kc columns in tiles
        for j in range(0, L, BLOCK_COL):
            k_idx = j + tl.arange(0, BLOCK_COL)
            mask_k = k_idx < L
            Kc_tile = tl.load(Kc + k_idx * head_dim_ckv + tl.arange(0, head_dim_ckv), mask=mask_k, other=0.0)  # [BLOCK_COL, head_dim_ckv]
            vec_tile = vec[k_idx]  # [BLOCK_COL]
            # out_row += vec_tile[:, None] * Kc_tile[None, :]  (elementwise), then reduce along tile
            out_row += tl.sum(vec_tile[:, None] * Kc_tile[None, :], axis=0)

        # Store out_row into out[i, h, :] in column tiles
        out_cols = col_block * BLOCK_COL + tl.arange(0, BLOCK_COL)
        mask_out = out_cols < head_dim_ckv
        tl.store(out + (i * H + h) * head_dim_ckv + out_cols, out_row[out_cols], mask=mask_out)
