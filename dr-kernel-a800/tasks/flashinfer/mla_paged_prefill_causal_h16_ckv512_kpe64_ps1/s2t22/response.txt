import torch
import math
import triton
import triton.language as tl


@triton.jit
def _forward_single_query_kernel(
    # Pointers to input tensors (stored as original dtype)
    q_nope_ptr,     # *bf16, shape [Q_total, 16, 512]
    q_pe_ptr,       # *bf16, shape [Q_total, 16, 64]
    Kc_sel_ptr,     # *bf16, shape [kv_len, 512], row-major (gathered from ckv_cache)
    Kp_sel_ptr,     # *bf16, shape [kv_len, 64], row-major (gathered from kpe_cache)
    output_ptr,     # *bf16, shape [Q_total, 16, 512]
    lse_ptr,        # *fp32, shape [Q_total, 16]
    # runtime scalars (will be converted to fp32 inside kernel)
    q_abs,          # int32, absolute query index in q_nope
    sm_scale,       # fp32, scaling factor
    ln2_inv,        # fp32, 1.0 / ln(2.0)
    # meta-parameters (compile-time for this program)
    q_len: tl.constexpr,        # number of queries in this batch element (handled per program -> assumed 1)
    kv_len: tl.constexpr,       # number of selected KV tokens
    NUM_HEADS: tl.constexpr,    # 16
    HEAD_DIM_CKV: tl.constexpr, # 512
    HEAD_DIM_KPE: tl.constexpr  # 64
):
    # We process one query per program; q_abs identifies the query. For robustness we also use q_len=1 (common case).
    # Load qn[h, :] and qp[h, :] for all heads, cast to fp32 for math.
    # q_nope layout: row-major [Q_total, 16, 512], so stride along head and dim are fixed.
    # We'll index by (q_abs, h, k) using pointer math.

    # Allocate per-head vectors (fp32)
    for h in range(NUM_HEADS):
        # Load qn[h, :] -> [HEAD_DIM_CKV]
        qn = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
        for k in range(HEAD_DIM_CKV):
            # ptr offset: q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV + k
            ptr = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV + k
            val = tl.load(q_nope_ptr + ptr).to(tl.float32)
            qn[k] = val

        # Load qp[h, :] -> [HEAD_DIM_KPE]
        qp = tl.zeros((HEAD_DIM_KPE,), dtype=tl.float32)
        for k in range(HEAD_DIM_KPE):
            ptr = q_abs * (NUM_HEADS * HEAD_DIM_KPE) + h * HEAD_DIM_KPE + k
            val = tl.load(q_pe_ptr + ptr).to(tl.float32)
            qp[k] = val

        # Initialize logits[h, :] as empty vector, we'll fill by j loop
        logits_j = tl.zeros((kv_len,), dtype=tl.float32)

        # Compute logits[h, j] = dot(qn[h], Kc_sel[j, :]) + dot(qp[h], Kp_sel[j, :])
        for j in range(kv_len):
            # Load Kc_sel[j, :] -> [HEAD_DIM_CKV]
            Kc_j = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
            for k in range(HEAD_DIM_CKV):
                ptr = j * HEAD_DIM_CKV + k
                Kc_val = tl.load(Kc_sel_ptr + ptr).to(tl.float32)
                Kc_j[k] = Kc_val

            # Load Kp_sel[j, :] -> [HEAD_DIM_KPE]
            Kp_j = tl.zeros((HEAD_DIM_KPE,), dtype=tl.float32)
            for k in range(HEAD_DIM_KPE):
                ptr = j * HEAD_DIM_KPE + k
                Kp_val = tl.load(Kp_sel_ptr + ptr).to(tl.float32)
                Kp_j[k] = Kp_val

            # Dot products
            dot_qn = 0.0
            for k in range(HEAD_DIM_CKV):
                dot_qn += qn[k] * Kc_j[k]
            dot_qp = 0.0
            for k in range(HEAD_DIM_KPE):
                dot_qp += qp[k] * Kp_j[k]

            logits_j[j] = dot_qn + dot_qp

        # Scale
        logits_j = logits_j * sm_scale

        # Apply causal mask: valid j >= prefix_len + i + 1, where prefix_len = kv_len - q_len and i=0 here (one query per program)
        prefix_len = kv_len - 1  # since q_len=1 per program
        valid_start = prefix_len + 1  # absolute position: 1 (relative to prefix)
        for j in range(kv_len):
            if j < valid_start:
                logits_j[j] = -float("inf")

        # Compute per-head logsumexp in log2
        m = tl.max(logits_j, axis=0)  # scalar
        sumexp = 0.0
        for j in range(kv_len):
            sumexp += tl.exp(logits_j[j] - m)
        lse_val = (m + tl.log(sumexp) * ln2_inv)  # scalar

        # Softmax over j (stable): softmax[h, j] = exp(logits_j[j] - m) / sumexp
        softmax_j = tl.zeros((kv_len,), dtype=tl.float32)
        for j in range(kv_len):
            softmax_j[j] = tl.exp(logits_j[j] - m) / sumexp

        # Output vector: out[h, :] = sum_j softmax_j[j] * Kc_sel[j, :]
        out_vec = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
        for j in range(kv_len):
            Kc_j = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
            for k in range(HEAD_DIM_CKV):
                ptr = j * HEAD_DIM_CKV + k
                Kc_val = tl.load(Kc_sel_ptr + ptr).to(tl.float32)
                Kc_j[k] = Kc_val
            out_vec += softmax_j[j] * Kc_j

        # Store output vector as bfloat16 to output[q_abs, h, :]
        out_store = out_vec.to(tl.bfloat16)
        base_out = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV
        for k in range(HEAD_DIM_CKV):
            tl.store(output_ptr + base_out + k, out_store[k])

        # Store lse[q_abs, h] as float32
        lse_base = q_abs * NUM_HEADS + h
        tl.store(lse_ptr + lse_base, lse_val)

    kv_len: tl.constexpr,         # number of selected KV tokens (meta for specialization)
    NUM_HEADS: tl.constexpr,      # 16
    HEAD_DIM_CKV: tl.constexpr,   # 512
    HEAD_DIM_KPE: tl.constexpr    # 64
):
    # This kernel handles one specific query i within a batch element (but we will launch it once per i in host).
    # We need absolute position for causal mask: query_abs_pos = (kv_len - q_len) + (i + 1).
    # Since this kernel instance corresponds to a specific i, we pass q_len; q_abs identifies the query.
    prefix_len = kv_len - q_len
    query_abs_pos = prefix_len + 1  # i=0 for this program; when host launches for i>0, grid ensures q_abs=q_start+i

    # For each head h, compute logits[h, j], softmax, and output.
    for h in range(NUM_HEADS):
        # Load qn[h, :] -> [HEAD_DIM_CKV] (bf16 from q_nope), cast to fp32
        qn = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
        for k in range(HEAD_DIM_CKV):
            ptr = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV + k
            val = tl.load(q_nope_ptr + ptr).to(tl.float32)
            qn[k] = val

        # Load qp[h, :] -> [HEAD_DIM_KPE] (bf16 from q_pe), cast to fp32
        qp = tl.zeros((HEAD_DIM_KPE,), dtype=tl.float32)
        for k in range(HEAD_DIM_KPE):
            ptr = q_abs * (NUM_HEADS * HEAD_DIM_KPE) + h * HEAD_DIM_KPE + k
            val = tl.load(q_pe_ptr + ptr).to(tl.float32)
            qp[k] = val

        # Compute logits[h, j] for j in [0, kv_len)
        logits_j = tl.zeros((kv_len,), dtype=tl.float32)
        for j in range(kv_len):
            # Kc_sel[j, :] -> [HEAD_DIM_CKV]
            Kc_j = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
            for k in range(HEAD_DIM_CKV):
                ptr = j * HEAD_DIM_CKV + k
                Kc_val = tl.load(Kc_sel_ptr + ptr).to(tl.float32)
                Kc_j[k] = Kc_val

            # Kp_sel[j, :] -> [HEAD_DIM_KPE]
            Kp_j = tl.zeros((HEAD_DIM_KPE,), dtype=tl.float32)
            for k in range(HEAD_DIM_KPE):
                ptr = j * HEAD_DIM_KPE + k
                Kp_val = tl.load(Kp_sel_ptr + ptr).to(tl.float32)
                Kp_j[k] = Kp_val

            # Dot-products
            dot_qn = 0.0
            for k in range(HEAD_DIM_CKV):
                dot_qn += qn[k] * Kc_j[k]
            dot_qp = 0.0
            for k in range(HEAD_DIM_KPE):
                dot_qp += qp[k] * Kp_j[k]

            logits_j[j] = dot_qn + dot_qp

        # Scale by sm_scale
        logits_j = logits_j * sm_scale

        # Apply causal mask: absolute position is query_abs_pos; invalid j < query_abs_pos -> -inf
        for j in range(kv_len):
            if j < query_abs_pos:
                logits_j[j] = -float("inf")

        # logsumexp in log2: lse = log(sum(exp(logits - max))) * ln2_inv + max(logits)
        m = tl.max(logits_j, axis=0)  # scalar
        sumexp = 0.0
        for j in range(kv_len):
            sumexp += tl.exp(logits_j[j] - m)
        lse_val = (m + tl.log(sumexp) * ln2_inv)  # scalar

        # Softmax over j (stable)
        softmax_j = tl.zeros((kv_len,), dtype=tl.float32)
        for j in range(kv_len):
            softmax_j[j] = tl.exp(logits_j[j] - m) / sumexp

        # Output: out[h, :] = sum_j softmax_j[j] * Kc_sel[j, :]
        out_vec = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
        for j in range(kv_len):
            Kc_j = tl.zeros((HEAD_DIM_CKV,), dtype=tl.float32)
            for k in range(HEAD_DIM_CKV):
                ptr = j * HEAD_DIM_CKV + k
                Kc_val = tl.load(Kc_sel_ptr + ptr).to(tl.float32)
                Kc_j[k] = Kc_val
            out_vec += softmax_j[j] * Kc_j

        # Store output vector (bfloat16) to output[q_abs, h, :]
        out_store = out_vec.to(tl.bfloat16)
        base_out = q_abs * (NUM_HEADS * HEAD_DIM_CKV) + h * HEAD_DIM_CKV
        for k in range(HEAD_DIM_CKV):
            tl.store(output_ptr + base_out + k, out_store[k])

        # Store lse[q_abs, h] as float32
        lse_base = q_abs * NUM_HEADS + h
        tl.store(lse_ptr + lse_base, lse_val)
