Here is the full code with the required ModelNew entry point. It defines Triton kernels for computing logits, logsumexp with causal masking, attention softmax, and final output matmul. All heavy computation is performed inside Triton kernels and is actually launched from ModelNew.forward. There is no torch matmul or softmax in the forward path. Shapes and data types are handled explicitly; inputs are assumed to be on CUDA.

```python
import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Kernel 1: Compute logits for each (head h, token position l, query i)
# Grid: (H, L, 1). Each program computes logits[h, l] for a given (i).
@triton.jit
def compute_logits_heads_3d(
    qn_ptr,         # *[total_q, H, head_dim_ckv], float32
    qp_ptr,         # *[total_q, H, head_dim_kpe], float32
    Kc_ptr,         # *[L, head_dim_ckv], float32
    Kp_ptr,         # *[L, head_dim_kpe], float32
    logits_ptr,     # *[H, L], float32
    i: tl.constexpr,      # query index (program processes a specific i)
    H: tl.constexpr,
    head_dim_ckv: tl.constexpr,
    head_dim_kpe: tl.constexpr,
    L: tl.constexpr,
):
    h = tl.program_id(0)
    l = tl.program_id(1)

    # Compute qn[h] and qp[h] for this (i)
    qn_vec = tl.load(qn_ptr + (i * H * head_dim_ckv) + h * head_dim_ckv)  # [head_dim_ckv]
    qp_vec = tl.load(qp_ptr + (i * H * head_dim_kpe) + h * head_dim_kpe)  # [head_dim_kpe]

    # Load Kc[l] and Kp[l]
    Kc_vec = tl.load(Kc_ptr + l * head_dim_ckv)  # [head_dim_ckv]
    Kp_vec = tl.load(Kp_ptr + l * head_dim_kpe)  # [head_dim_kpe]

    # Dot-products
    acc1 = 0.0
    for d in range(head_dim_ckv):
        acc1 += qn_vec[d] * Kc_vec[d]

    acc2 = 0.0
    for d in range(head_dim_kpe):
        acc2 += qp_vec[d] * Kp_vec[d]

    logits_scalar = acc1 + acc2
    # Store
    tl.store(logits_ptr + h * L + l, logits_scalar)


# Kernel 2: Compute logsumexp (lse) per (i, h) and attention softmax
# Grid: (q_len, H). Each program computes lse and attn for a given (i, h).
@triton.jit
def lse_and_attn_1d(
    logits_ptr,     # *[H, L], float32
    attn_ptr,       # *[q_len, H, L], float32
    lse_ptr,        # *[q_len, H], float32
    SM_SCALE: tl.constexpr,
    q_len: tl.constexpr,
    H: tl.constexpr,
    L: tl.constexpr,
):
    i = tl.program_id(0)
    h = tl.program_id(1)

    # Load logits vector for this (h)
    logits_vec = tl.load(logits_ptr + h * L + tl.arange(0, L))  # [L]
    logits_scaled = logits_vec * SM_SCALE  # [L]

    # Apply causal mask: allow positions k > query_abs_pos; otherwise -inf.
    # For this (i, h), query_abs_pos = prefix_len + i; prefix_len = L - q_len
    prefix_len = L - q_len
    query_abs_pos = prefix_len + i
    causal_mask = tl.arange(0, L) > query_abs_pos
    logits_scaled = tl.where(causal_mask, logits_scaled, -float("inf"))

    # Max for numerical stability
    max_val = tl.max(logits_scaled, axis=0)
    sum_exp = 0.0
    for l in range(L):
        sum_exp += tl.exp(logits_scaled[l] - max_val)

    lse_val = (max_val + tl.log(sum_exp)) / math.log(2.0)
    tl.store(lse_ptr + i * H + h, lse_val)

    # Compute attn = softmax(logits_scaled) per position
    attn_vec = tl.zeros((L,), dtype=tl.float32)
    for l in range(L):
        attn_vec[l] = tl.exp(logits_scaled[l] - max_val) / sum_exp
    tl.store(attn_ptr + i * H * L + h * L + tl.arange(0, L), attn_vec)


# Kernel 3: Compute out[i, h, :] = attn[i, h, :] @ Kc.T, tiled over columns
# Grid: (q_len, H, ceil_div(head_dim_ckv, BLOCK_COL))
@triton.jit
def matmul_vec_by_mat(
    attn_ptr,       # *[q_len, H, L], float32
    Kc_ptr,         # *[L, head_dim_ckv], float32
    out_ptr,        # *[q_len, H, head_dim_ckv], float32
    q_len: tl.constexpr,
    H: tl.constexpr,
    L: tl.constexpr,
    head_dim_ckv: tl.constexpr,
    BLOCK_COL: tl.constexpr,
):
    i = tl.program_id(0)
    h = tl.program_id(1)
    col_block = tl.program_id(2)

    start_col = col_block * BLOCK_COL
    cols = start_col + tl.arange(0, BLOCK_COL)
    mask_cols = cols < head_dim_ckv

    acc = tl.zeros((BLOCK_COL,), dtype=tl.float32)

    # Compute attn_vec for this (i, h): attn[i, h, :]
    attn_vec = tl.load(attn_ptr + i * H * L + h * L + tl.arange(0, L))  # [L]

    for l in range(0, L):
        Kc_col = tl.load(Kc_ptr + l * head_dim_ckv + cols, mask=mask_cols, other=0.0)  # [BLOCK_COL]
        acc += attn_vec[l] * Kc_col

    tl.store(out_ptr + i * H * head_dim_ckv + h * head_dim_ckv + cols, acc, mask=mask_cols)


def _run_triton_only(q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
    # Ensure inputs are CUDA tensors
    device = q_nope.device
    total_q = q_nope.shape[0]
    H = q_nope.shape[1]
    head_dim_ckv = q_nope.shape[2]
    # Convert to float32 for compute; keep original to compare output later
    q_nope_f32 = q_nope.to(torch.float32)
    q_pe_f32 = q_pe.to(torch.float32)

    # qo_indptr and kv_indptr are int32; compute batch size
    batch_size = qo_indptr.shape[0] - 1
    L_all = ckv_cache.shape[0]  # number of tokens in cache
    # Prepare Kc and Kp flattened by batch b
    Kc_rows = ckv_cache.to(torch.float32)[:, 0, :].contiguous()  # [num_pages, 512]
    Kp_rows = kpe_cache.to(torch.float32)[:, 0, :].contiguous()  # [num_pages, 64]

    # Output buffers
    output = torch.empty((total_q, H, head_dim_ckv), dtype=torch.float32, device=device)  # final result will be bfloat16
    lse = torch.empty((total_q, H), dtype=torch.float32, device=device)

    # For each batch b, compute per-segment
    for b in range(batch_size):
        q_start = int(qo_indptr[b].item())
        q_end = int(qo_indptr[b + 1].item())
        if q_start >= q_end:
            continue

        # KV range
        kv_start = int(kv_indptr[b].item())
        kv_end = int(kv_indptr[b + 1].item())
        if kv_start >= kv_end:
            continue

        q_len = q_end - q_start
        L = kv_end - kv_start

        # Gather token indices and corresponding K rows
        tok_idx = kv_indices[kv_start:kv_end].to(torch.int32)  # [L]
        Kc_seg = Kc_rows[tok_idx]  # [L, 512]
        Kp_seg = Kp_rows[tok_idx]  # [L, 64]

        # Buffer for logits per (h, l)
        logits = torch.empty((H, L), dtype=torch.float32, device=device)

        # Kernel 1: compute logits[h, l] for each i in [0, q_len)
        # Launch grid over (H, L, 1) for each i
        for i in range(q_len):
            grid = (H, L, 1)
            compute_logits_heads_3d[grid](
                q_nope_f32, q_pe_f32, Kc_seg, Kp_seg, logits,
                i, H=H, head_dim_ckv=head_dim_ckv, head_dim_kpe=64, L=L,
            )

        # Kernel 2: lse and attn per (i, h)
        attn = torch.empty((q_len, H, L), dtype=torch.float32, device=device)
        grid_lse = (q_len, H)
        lse_and_attn_1d[grid_lse](
            logits, attn, lse,
            SM_SCALE=sm_scale, q_len=q_len, H=H, L=L,
        )

        # Kernel 3: out = attn @ Kc.T, tiled over columns
        BLOCK_COL = 128
        grid_out = (q_len, H, triton.cdiv(head_dim_ckv, BLOCK_COL))
        out_seg = torch.empty((q_len, H, head_dim_ckv), dtype=torch.float32, device=device)
        matmul_vec_by_mat[grid_out](
            attn, Kc_seg, out_seg,
            q_len=q_len, H=H, L=L, head_dim_ckv=head_dim_ckv, BLOCK_COL=BLOCK_COL,
        )

        # Merge out_seg into output at positions [q_start:q_end]
        for i in range(q_len):
            output[q_start + i] = out_seg[i]

    # Cast output to bfloat16 as in original code
    output_bf16 = output.to(torch.bfloat16)
    return output_bf16, lse


# Entry point
class ModelNew(torch.nn.Module):
    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Ensure CUDA tensors
        if not (q_nope.is_cuda and q_pe.is_cuda and ckv_cache.is_cuda and kpe_cache.is_cuda
                and qo_indptr.is_cuda and kv_indptr.is_cuda and kv_indices.is_cuda):
            raise RuntimeError("All inputs must be CUDA tensors for Triton execution.")
        return _run_triton_only(q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale)
```
