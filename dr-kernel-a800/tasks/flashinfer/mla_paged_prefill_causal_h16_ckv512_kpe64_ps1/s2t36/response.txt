import torch
import math
import triton
import triton.language as tl


@triton.jit
def _forward_single_query_kernel(
    q_nope_ptr,       # *bf16, [Q_total, 16, 512]
    q_pe_ptr,         # *bf16, [Q_total, 16, 64]
    output_ptr,       # *bf16, [Q_total, 16, 512]
    lse_ptr,          # *f32,  [Q_total, 16]
    ckv_cache_ptr,    # *bf16, [num_pages, 1, 512]
    kpe_cache_ptr,    # *bf16, [num_pages, 1, 64]
    qo_indptr_ptr,    # *i32,  [len_indptr]
    kv_indptr_ptr,    # *i32,  [len_indptr]
    kv_indices_ptr,   # *i32,  [num_kv_indices]
    q_len_total,      # i32
    tok_len,          # i32
    sm_scale,         # f32
):
    # Program ids: pid0 -> batch element b (len_indptr - 1), pid1 -> query i in [0, q_len_total)
    b = tl.program_id(0)
    i = tl.program_id(1)

    # Compute absolute query index
    q_start = tl.load(qo_indptr_ptr + b)  # int32
    q_abs = q_start + i
    q_len = tl.load(qo_indptr_ptr + b + 1) - q_start  # int32
    prefix_len = tok_len - q_len  # int32

    # Load qn and qp for all heads
    # qn[h, :] from q_nope[q_abs, h, :]
    # qp[h, :] from q_pe[q_abs, h, :]
    # We loop h = 0..15
    # For each head, compute out_vec and store
    for h in range(16):
        # Build offsets for q_nope and q_pe
        # Each head's entries are contiguous in memory: [q_abs * (16*512) + h*512 + k]
        # But q_nope is stored as [total_q, 16, 512] contiguous => offset = q_abs * (16*512) + h*512
        qn_offset = q_abs * (16 * 512) + h * 512
        qp_offset = q_abs * (16 * 64) + h * 64  # q_pe has last dim 64

        # Load qn and qp as float32
        qn_vec = tl.load(q_nope_ptr + qn_offset + tl.arange(0, 512), mask=(tl.arange(0, 512) < 512), other=0.0).to(tl.float32)
        # q_pe last dim is 64, we need vector of length 64, but original code uses q_len and heads only for q_nope,
        # so we skip loading q_pe here and instead compute output using Kc_sel only. However, original code uses qn and qp together; since q_len_total is total queries,
        # we can infer q_pe is unused in output calculation. To be faithful: the output uses Kc_sel and not q_pe.
        # Therefore, we don't need qp. We will compute logits without qp. But original code adds (qp @ Kp_sel.T).
        # We need qp[h, :] of length 64. Let's load it correctly:
        # q_pe layout: [total_q, 16, 64] => offset = q_abs * (16*64) + h*64
        qp_vec = tl.load(q_pe_ptr + (q_abs * (16 * 64) + h * 64) + tl.arange(0, 64), mask=(tl.arange(0, 64) < 64), other=0.0).to(tl.float32)

        # Prepare accumulators
        logits = tl.full([tok_len], -1e30, dtype=tl.float32)
        out_vec = tl.zeros([512], dtype=tl.float32)

        # Loop over j in [0, tok_len)
        for j in range(tok_len):
            # Load tok_idx for this batch
            tok_idx_j = tl.load(kv_indices_ptr + (b * tok_len + j))  # int32

            # Load Kc_sel[j, :] and Kp_sel[j, :]
            Kc_row = tl.load(ckv_cache_ptr + tok_idx_j * 512 + tl.arange(0, 512), mask=(tl.arange(0, 512) < 512), other=0.0).to(tl.float32)
            Kp_row = tl.load(kpe_cache_ptr + tok_idx_j * 64 + tl.arange(0, 64), mask=(tl.arange(0, 64) < 64), other=0.0).to(tl.float32)

            # Compute dot products
            dot_qn = tl.sum(qn_vec * Kc_row, axis=0)  # scalar
            dot_qp = tl.sum(qp_vec * Kp_row, axis=0)  # scalar

            logits[j] = (dot_qn + dot_qp) * sm_scale

        # Apply causal mask: j >= prefix_len + i + 1
        # Build mask vector
        j_range = tl.arange(0, tok_len)
        mask_j = (j_range >= (prefix_len + i + 1))
        logits = tl.where(mask_j, logits, -1e30)

        # Compute logsumexp in fp32, then convert to log2
        m = tl.max(logits, axis=0)
        sum_exp = tl.sum(tl.exp(logits - m), axis=0)
        lse_val = (m + tl.log(sum_exp)) / math.log(2.0)  # convert ln to log2
        tl.store(lse_ptr + q_abs * 16 + h, lse_val)

        # Compute softmax over logits
        exp_logits = tl.exp(logits - m)
        sum_exp = tl.sum(exp_logits, axis=0)
        softmax = exp_logits / sum_exp

        # Compute output for this head: out[h, :] = sum_j softmax[j] * Kc_sel[j, :]
        out_vec = tl.zeros([512], dtype=tl.float32)
        for j in range(tok_len):
            if mask_j[j]:
                Kc_row = tl.load(ckv_cache_ptr + tok_idx_j * 512 + tl.arange(0, 512), mask=(tl.arange(0, 512) < 512), other=0.0).to(tl.float32)
                out_vec += softmax[j] * Kc_row

        # Store output in bfloat16
        out_offset = q_abs * (16 * 512) + h * 512
        out_bf16 = out_vec.to(tl.bfloat16)
        tl.store(output_ptr + out_offset, out_bf16)
