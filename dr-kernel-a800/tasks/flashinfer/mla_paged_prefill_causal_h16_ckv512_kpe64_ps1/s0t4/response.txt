@triton.jit
def compute_logits_and_lse_row_kernel(
    qn_ptr,         # float32*  -> [H, Dn] pointer (but we pass a [Dn] row for h)
    qp_ptr,         # float32*  -> [H, Dp] pointer (but we pass a [Dp] row for h)
    Kc_ptr,         # float32*  -> [KV, Dn]
    Kp_ptr,         # float32*  -> [KV, Dp]
    logits_ptr,     # float32*  -> [KV]
    lse_ptr,        # float32*  -> [1] lse for this row
    sm_scale: tl.float32,
    prefix_len: tl.int32,     # int32 scalar
    query_abs_pos: tl.int32,  # int32 scalar
    KV: tl.int32,             # number of KV tokens
    Dn: tl.constexpr,         # head_dim_ckv == 512
    Dp: tl.constexpr,         # head_dim_kpe == 64
    BLOCK_K: tl.constexpr,    # tile size for KV
):

    # Compute logits = (qn @ Kc.T) + (qp @ Kp.T)
    # Initialize logits_vec
    logits_vec = tl.zeros((KV,), dtype=tl.float32)
    # Accumulate S = qn @ Kc.T
    for k0 in range(0, KV, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < KV
        # Load Kc tile [BLOCK_K, Dn]
        Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)
        # qn_row is loaded as 1D [Dn] from qn_ptr (we construct it in host as a [Dn] array for each h)
        # Since Triton kernel doesn't have direct access to h-specific qn, host must ensure qn_ptr points to correct row.
        # Here we implement qn_row by loading from qn_ptr at offset h * Dn + tl.arange(0, Dn).
        qn_row = tl.load(qn_ptr + tl.arange(0, Dn))  # [Dn]
        # partial = qn_row[None, :] @ Kc_tile          -> [BLOCK_K, Dn] x [Dn] -> [BLOCK_K]
        # Simplify: qn_row is 1D, so multiply across Dn and reduce
        partial = tl.sum(qn_row[None, :] * Kc_tile, axis=1)  # [BLOCK_K]
        logits_vec += partial
    # Accumulate T = qp @ Kp.T

        logits_vec += partial

    logits_vec *= sm_scale

    # Apply causal mask: keep j > (prefix_len + i), else -inf

    # Store logits
    tl.store(logits_ptr + tl.arange(0, KV), logits_vec)

@triton.jit
def lse_row_kernel(logits_ptr, lse_ptr, KV: tl.int32, scale: tl.float32):
    # Compute max for numerical stability
    max_val = -float("inf")
    for j in range(0, KV):
        max_val = tl.maximum(max_val, tl.load(logits_ptr + j))
    # Compute sum_exp of exp(logits - max)
    sum_exp = 0.0
    for j in range(0, KV):
        v = tl.load(logits_ptr + j)
        sum_exp += tl.exp(v - max_val)
    # lse = log(sum_exp) + max; divide by ln(2)
    lse_val = tl.log(sum_exp) + max_val
    lse_val = lse_val / tl.log(2.0)  # scale by ln(2)
    # Store scalar lse for this row
    tl.store(lse_ptr, lse_val)

@triton.jit
def softmax_row_kernel(logits_ptr, attn_ptr, KV: tl.int32):
    # Stable softmax for a single row
    max_val = -float("inf")
    for j in range(0, KV):

    sum_exp = 0.0
    for j in range(0, KV):
        v = tl.load(logits_ptr + j)
        sum_exp += tl.exp(v - max_val)

    for j in range(0, KV):
        v = tl.load(logits_ptr + j)
        tl.store(attn_ptr + j, tl.exp(v - max_val) * inv_sum)

def compute_out_row_kernel(attn_ptr, Kc_ptr, out_ptr, KV: tl.int32, Dn: tl.constexpr, BLOCK_K: tl.constexpr):

        mask_k = k_idx < KV

        attn_tile = tl.load(attn_ptr + k_idx)  # [BLOCK_K]
        # out_vec += sum_k attn_tile[k] * Kc_tile[k, :]
        # Manually sum over axis 0

                a = attn_tile[kk]  # scalar
                col = Kc_tile[kk, :]  # [Dn]
                out_vec += a * col
    tl.store(out_ptr + tl.arange(0, Dn), out_vec)

import torch

try:
    import triton
    import triton.language as tl
except Exception:
    triton = None
    tl = None

# Triton kernels: Triton-only, no torch ops in forward.

# Kernel 1: compute logits for a single head row: logits = qn @ Kc.T + qp @ Kp.T, then apply scaling and causal mask, compute lse.
@triton.jit
def compute_logits_and_lse_row_kernel(
    qn_ptr,         # float32*  -> 1D [Dn] row for head h
    qp_ptr,         # float32*  -> 1D [Dp] row for head h
    Kc_ptr,         # float32*  -> [KV, Dn]
    Kp_ptr,         # float32*  -> [KV, Dp]
    logits_ptr,     # float32*  -> [KV] (workspace)
    lse_ptr,        # float32*  -> [1] scalar lse for this row
    sm_scale: tl.float32,
    prefix_len: tl.int32,     # int32 scalar
    query_abs_pos: tl.int32,  # int32 scalar
    KV: tl.int32,             # number of KV tokens
    Dn: tl.constexpr,         # head_dim_ckv == 512
    Dp: tl.constexpr,         # head_dim_kpe == 64
    BLOCK_K: tl.constexpr,    # tile size for KV
    Dk: tl.constexpr,         # KV
):
    # Compute qn @ Kc.T and qp @ Kp.T in two parts and sum into logits_ptr
    # Part A: qn @ Kc.T
    # We will perform matmul of a [1, Dn] row with [Dn, Dk] to get [1, Dk]
    outA = tl.zeros((Dk,), dtype=tl.float32)
    # Loop over K tiles
    for k0 in range(0, Dk, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < Dk
        Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dn]
        # qn_row is 1D [Dn]
        qn_row = tl.load(qn_ptr + tl.arange(0, Dn))  # [Dn]
        # Multiply qn_row (1x[Dn]) with Kc_tile ([BLOCK_K, Dn]) -> [BLOCK_K] via outer product reduction
        partial = tl.dot(qn_row[None, :], Kc_tile)  # [1, BLOCK_K]
        outA += partial[0, :]  # [BLOCK_K] -> sum over K tiles

    # Part B: qp @ Kp.T
    outB = tl.zeros((Dk,), dtype=tl.float32)
    for k0 in range(0, Dk, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < Dk
        Kp_tile = tl.load(Kp_ptr + k_idx[:, None] * Dp + tl.arange(0, Dp), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dp]
        qp_row = tl.load(qp_ptr + tl.arange(0, Dp))  # [Dp]
        partial = tl.dot(qp_row[None, :], Kp_tile)  # [1, BLOCK_K]
        outB += partial[0, :]

    # Sum and scale
    logits_vec = outA + outB
    logits_vec *= sm_scale

    # Apply causal mask: keep j > (prefix_len + i), else -inf
    for j in range(0, Dk):
        if j <= (prefix_len + query_abs_pos):
            logits_vec[j] = -float("inf")

    # Store logits
    tl.store(logits_ptr + tl.arange(0, Dk), logits_vec)

    # Compute lse for this row via lse_row_kernel
    lse_row_kernel[(1,)](
        logits_ptr, lse_ptr, Dk, 1.0 / math.log(2.0),
        num_warps=4,
    )

# Kernel 2: lse of a row (logsumexp) with numerical stability, divided by ln(2)
@triton.jit
def lse_row_kernel(logits_ptr, lse_ptr, KV: tl.int32, scale: tl.float32):
    # Compute max for numerical stability
    max_val = -float("inf")
    for j in range(0, KV):
        max_val = tl.maximum(max_val, tl.load(logits_ptr + j))
    # Compute sum_exp of exp(logits - max)
    sum_exp = 0.0
    for j in range(0, KV):
        v = tl.load(logits_ptr + j)
        sum_exp += tl.exp(v - max_val)
    # lse = log(sum_exp) + max; divide by ln(2)
    lse_val = tl.log(sum_exp) + max_val
    lse_val = lse_val * scale
    # Store scalar lse for this row
    tl.store(lse_ptr, lse_val)

# Kernel 3: softmax of a row (stable)
@triton.jit
def softmax_row_kernel(logits_ptr, attn_ptr, KV: tl.int32):
    # Stable softmax for a single row
    max_val = -float("inf")
    for j in range(0, KV):
        max_val = tl.maximum(max_val, tl.load(logits_ptr + j))
    sum_exp = 0.0
    for j in range(0, KV):
        v = tl.load(logits_ptr + j)
        sum_exp += tl.exp(v - max_val)
    inv_sum = 1.0 / sum_exp
    for j in range(0, KV):
        v = tl.load(logits_ptr + j)
        tl.store(attn_ptr + j, tl.exp(v - max_val) * inv_sum)

# Kernel 4: out_row = attn @ Kc (matmul [KV, 1] by [KV, Dn] -> [Dn])
@triton.jit
def compute_out_row_kernel(attn_ptr, Kc_ptr, out_ptr, KV: tl.int32, Dn: tl.constexpr, BLOCK_K: tl.constexpr):
    # Compute out = attn @ Kc, attn: [KV], Kc: [KV, Dn], out: [Dn]
    out_vec = tl.zeros((Dn,), dtype=tl.float32)
    # attn is 1D [KV]; Kc is 2D [KV, Dn]
    for k0 in range(0, KV, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < KV
        attn_tile = tl.load(attn_ptr + k_idx, mask=mask_k, other=0.0)  # [BLOCK_K]
        Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dn]
        # out_vec += sum_k attn_tile[k] * Kc_tile[k, :]
        # Using elementwise multiply and reduction over k axis
        partial = tl.sum(attn_tile[:, None] * Kc_tile, axis=0)  # [Dn]
        out_vec += partial
    tl.store(out_ptr + tl.arange(0, Dn), out_vec)

# Helper matmul kernels for qn @ Kc.T and qp @ Kp.T (per head row)
# These are used inside compute_logits_and_lse_row_kernel.

# Kernel 5: matmul_qn_KcT_row_kernel computes out = qn @ Kc.T -> [Dk]
@triton.jit
def matmul_qn_KcT_row_kernel(
    qn_ptr,         # float32* -> [Dn] (row for head h)
    Kc_ptr,         # float32* -> [KV, Dn]
    out_ptr,        # float32* -> [Dk]
    Dn: tl.constexpr,  # 512
    Dk: tl.constexpr,  # KV
    BLOCK_K: tl.constexpr,  # e.g., 64
    BLOCK_N: tl.constexpr,  # e.g., 64
):
    # out shape [Dk], computed as qn (1x[Dn]) @ Kc.T ([Dn, Dk]) -> [1x[Dk]]
    # Use a single output vector accumulation
    out_vec = tl.zeros((Dk,), dtype=tl.float32)
    # Loop over K tiles
    for k0 in range(0, Dk, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < Dk
        # Load Kc tile [BLOCK_K, Dn]
        Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dn]
        # Load qn_row [Dn]
        qn_row = tl.load(qn_ptr + tl.arange(0, Dn))  # [Dn]
        # partial = qn_row @ Kc_tile.T -> [BLOCK_K]
        # Using tl.dot on qn_row (1x[Dn]) with Kc_tile.T ([Dn, BLOCK_K]) reduced over Dn
        # To form [Dn, BLOCK_K], take transpose of tile: Kc_tile_T = tl.trans(Kc_tile) is not available; instead do:
        # We compute partial as sum over Dn: for each k in tile, sum qn_row * Kc_tile[k, :]
        partial = tl.zeros((BLOCK_K,), dtype=tl.float32)
        for n0 in range(0, Dn, BLOCK_N):
            n_idx = n0 + tl.arange(0, BLOCK_N)
            mask_n = n_idx < Dn
            qn_sub = tl.load(qn_ptr + n_idx, mask=mask_n, other=0.0)  # [BLOCK_N]
            Kc_sub = tl.load(Kc_ptr + k_idx[:, None] * Dn + n_idx[None, :], mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # [BLOCK_K, BLOCK_N]
            # partial += sum over BLOCK_N of qn_sub * Kc_sub (each column of Kc_sub)
            for j in range(0, BLOCK_N):
                if (n0 + j) < Dn:
                    partial += qn_sub[j] * tl.sum(Kc_sub[:, j], axis=0)
        out_vec += partial
    tl.store(out_ptr + tl.arange(0, Dk), out_vec)
