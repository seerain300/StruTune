class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        device = q_nope.device
        total_q = q_nope.shape[0]
        num_qo_heads = q_nope.shape[1]
        head_dim_ckv = q_nope.shape[2]
        head_dim_kpe = q_pe.shape[2]
        assert num_qo_heads == 16
        assert head_dim_ckv == 512
        assert head_dim_kpe == 64

        # Kc_all and Kp_all have shape [num_pages, 1, D], we squeeze the 1-dim
        Kc_all = ckv_cache.squeeze(1).to(torch.float32)
        Kp_all = kpe_cache.squeeze(1).to(torch.float32)

        batch_size = kv_indptr.shape[0] - 1

        # Prepare output and lse
        output = torch.empty((total_q, num_qo_heads, head_dim_ckv), dtype=torch.bfloat16, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Precompute inv_ln2 (1 / ln(2))
        inv_ln2 = 1.4426950408889634  # 1 / math.log(2)

        for b in range(batch_size):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            if q_start >= q_end:
                continue

            q_len = q_end - q_start

            # KV block
            page_beg = int(kv_indptr[b].item())
            page_end = int(kv_indptr[b + 1].item())
            if page_beg >= page_end:
                continue

            tok_idx = kv_indices[page_beg:page_end].to(torch.int32)  # [L]
            L = tok_idx.shape[0]
            Kc_rows = Kc_all[tok_idx].to(torch.float32)  # [L, 512]
            Kp_rows = Kp_all[tok_idx].to(torch.float32)  # [L, 64]

            # For each query i in this batch
            for i in range(q_len):
                # Logits buffer [H, L], float32
                logits = torch.empty((num_qo_heads, L), dtype=torch.float32, device=device)

                # Launch compute_logits_kernel
                BLOCK_L = 128
                BLOCK_K = 64
                grid = (num_qo_heads, triton.cdiv(L, BLOCK_L))
                compute_logits_kernel[grid](
                    q_nope[q_start + i], q_pe[q_start + i], Kc_rows, Kp_rows, logits,
                    num_qo_heads, L, head_dim_ckv, head_dim_kpe,
                    q_nope.stride(0), q_nope.stride(1),
                    q_pe.stride(0), q_pe.stride(1),
                    Kc_rows.stride(0), Kc_rows.stride(1),
                    Kp_rows.stride(0), Kp_rows.stride(1),
                    logits.stride(0), logits.stride(1),
                    BLOCK_L=BLOCK_L, BLOCK_K=BLOCK_K,
                )

                # Launch lse_mask_kernel per head
                lse_b = torch.empty((num_qo_heads,), dtype=torch.float32, device=device)
                lse_mask_kernel[(num_qo_heads,)](
                    logits, lse_b, num_qo_heads, L, inv_ln2,
                    logits.stride(0), logits.stride(1),
                    BLOCK_L=BLOCK_L,
                )

                # Store lse
                lse[q_start + i, :] = lse_b

                # Compute output: softmax over L then attn @ Kc
                # Apply causal mask: positions j > (L - q_len + i) are not causal
                prefix_len = L - q_len  # number of previously cached tokens for this batch
                query_abs_pos = prefix_len + i
                causal = torch.arange(L, device=device) > query_abs_pos
                logits_masked = logits
                if causal.any():
                    logits_masked = logits.masked_fill_(causal.unsqueeze(0), -float('inf'))

                # Softmax over L for each head
                probs = torch.softmax(logits_masked, dim=-1)  # [H, L]

                # Output: out[h, :] = probs[h, :] @ Kc_rows[:, :]
                for h in range(num_qo_heads):
                    out_vec = probs[h] @ Kc_rows.float()  # [512]
                    output[q_start + i, h, :] = out_vec.to(torch.bfloat16)

        return output, lse
