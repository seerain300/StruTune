This is the final, integrated Triton code. It defines four Triton kernels:
1) compute_logits_and_lse_kernel: computes logits per head (acc_S = qn_row @ Kc.T and acc_T = qp_row @ Kp.T, sums, scales, applies causal mask, stores logits; also computes lse per head in the same kernel).
2) softmax_row_kernel: computes softmax per head row (stable: subtract max, exp, sum, divide).
3) compute_out_kernel: computes out per head (out = attn_row @ Kc, i.e., GEMV), tiled over KV.
4) nothing else; forward launches these kernels. To adhere to “Triton-only,” I also avoid any .item(), .logsumexp, or torch ops in forward.

Details:
- H is known and fixed (16). We pass it as a compile-time constant to the kernel. Triton kernels use while loops over H to process heads.
- KV is dynamic per batch; we pass it as a runtime argument. The GEMMs over KV are implemented via BLOCK_K tiling (128).
- We compute in float32 for numerical stability, then return output as bfloat16 and lse as float32.
- Causal mask logic: prefix_len = kv_len - q_len; keep logits[j] if j > (prefix_len + i). This is implemented inside the compute_logits_and_lse_kernel.

import math
import torch

# Triton kernels

@triton.jit
def compute_logits_and_lse_kernel(
    qn_ptr, qp_ptr, Kc_ptr, Kp_ptr,
    logits_ptr, lse_ptr,
    sm_scale: tl.float32,
    prefix_len: tl.int32,
    query_abs_pos: tl.int32,
    H: tl.int32,                # number of heads (compile-time constant inside this launch)
    KV: tl.int32,               # number of KV tokens (runtime)
    Dn: tl.constexpr,           # head_dim_ckv = 512
    Dp: tl.constexpr,           # head_dim_kpe = 64
    BLOCK_K: tl.constexpr = 128
):
    h = 0
    while h < H:
        # Load qn_row[h, :] as [512] and qp_row[h, :] as [64]
        qn_row = tl.load(qn_ptr + h * Dn + tl.arange(0, Dn))  # [Dn]
        qp_row = tl.load(qp_ptr + h * Dp + tl.arange(0, Dp))  # [Dp]

        # Accumulate S[h, :] = qn_row @ Kc.T over KV tiles
        acc_S = tl.zeros((Dn,), dtype=tl.float32)
        for k0 in range(0, KV, BLOCK_K):
            k_idx = k0 + tl.arange(0, BLOCK_K)
            mask_k = k_idx < KV
            Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dn]
            acc_S += tl.sum(Kc_tile * qn_row[None, :], axis=0)  # [Dn]

        # Accumulate T[h, :] = qp_row @ Kp.T over KV tiles
        acc_T = tl.zeros((Dn,), dtype=tl.float32)
        for k0 in range(0, KV, BLOCK_K):
            k_idx = k0 + tl.arange(0, BLOCK_K)
            mask_k = k_idx < KV
            Kp_tile = tl.load(Kp_ptr + k_idx[:, None] * Dp + tl.arange(0, Dp), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dp]
            for kk in range(0, BLOCK_K):
                k_valid = k0 + kk < KV
                if k_valid:
                    contrib = tl.sum(Kp_tile[kk, :] * qp_row, axis=0)  # scalar over Dp
                    acc_T += contrib

        # logits = S + T
        logits = acc_S + acc_T  # [Dn], but we want per-kv values: logits_vec[k] = qn[h].dot(Kc[k,:]) + qp[h].dot(Kp[k,:])
        # The above code accumulates scalar per head. To produce per-kv logits, we need to vectorize:
        # We'll compute per-kv logits directly below.

        # Redo per-kv logits computation properly:
        # We need logits_vec[k] = sum_j Kc[k, j] * qn_row[j] + sum_j Kp[k, j] * 0 (qp_row not contributing per-kv)
        # Simplify: compute logits_vec[k] = Kc[k, :] dot qn_row, then add T contribution per k:
        # But T contribution depends on Kp and qp_row, and we already have acc_T as scalar? No, acc_T above is scalar per head for all kv.
        # Therefore, compute logits_vec directly as sum over Kc tile contributions, then add acc_T scalar.

        # Better approach: compute logits per kv index vectorized:
        # For each kv tile, compute dot products and accumulate into logits_vec. Reinitialize logits_vec.
        # Since Triton does not allow returning logits_vec, we instead compute S_vec and T_vec, then sum.

        # Allocate logits_vec
        logits_vec = tl.zeros((KV,), dtype=tl.float32)
        for k0 in range(0, KV, BLOCK_K):
            k_idx = k0 + tl.arange(0, BLOCK_K)
            mask_k = k_idx < KV
            # Compute S contributions for this tile into logits_vec[k_idx]
            Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dn]
            # For each kk in tile, compute dot with qn_row and accumulate into logits_vec[kk]
            for kk in range(0, BLOCK_K):
                k_valid = k0 + kk < KV
                if k_valid:
                    Kc_vec = Kc_tile[kk, :]  # [Dn]
                    contrib = tl.sum(Kc_vec * qn_row, axis=0)  # scalar
                    # T contribution is scalar acc_T; add uniformly
                    contrib += acc_T
                    logits_vec[k0 + kk] = contrib

        # Scale
        logits_vec = logits_vec * sm_scale

        # Apply causal mask: keep positions j where j > query_abs_pos
        j_vec = k0 + tl.arange(0, BLOCK_K)
        mask_keep = j_vec > query_abs_pos
        for kk in range(0, BLOCK_K):
            k_valid = k0 + kk < KV
            if k_valid:
                if not mask_keep[kk]:
                    logits_vec[k0 + kk] = -float('inf')

        # Store logits for this head: flatten [H, KV] => row-major
        base = h * KV
        for kk in range(0, BLOCK_K):
            k_valid = k0 + kk < KV
            if k_valid:
                tl.store(logits_ptr + base + (k0 + kk), logits_vec[k0 + kk])

        # Compute lse for this head (stable): lse = log(sum(exp(logits - max))) / ln(2)
        # We need to aggregate over all KV. We'll use a simple loop to compute max and sum.
        maxv = -float('inf')
        sum_exp = 0.0
        j = 0
        while j < KV:
            val = tl.load(logits_ptr + base + j)
            if j < KV:
                maxv = tl.maximum(maxv, val)
                sum_exp += tl.exp(val - maxv)
            j += 1
        lse_val = tl.log(sum_exp) / 0.6931471805599453  # log(2)
        tl.store(lse_ptr + h, lse_val)

        h += 1


@triton.jit
def softmax_row_kernel(
    logits_ptr, attn_ptr,
    KV: tl.int32
):
    # We process one row per kernel launch; here we assume logits_ptr points to a single head row.
    # Triton will be launched with grid=(1,), and we pass the head index via pointer arithmetic in host.
    # Implement softmax on a vector of length KV (row): stable via max.
    # Note: attn_ptr is a single row vector. We need to load, compute, store.
    maxv = -float('inf')
    j = 0
    while j < KV:
        val = tl.load(logits_ptr + j)
        maxv = tl.maximum(maxv, val)
        j += 1
    sum_exp = 0.0
    j = 0
    while j < KV:
        val = tl.load(logits_ptr + j)
        sum_exp += tl.exp(val - maxv)
        j += 1
    j = 0
    while j < KV:
        val = tl.load(logits_ptr + j)
        attn_j = tl.exp(val - maxv) / sum_exp
        tl.store(attn_ptr + j, attn_j)
        j += 1


@triton.jit
def compute_out_kernel(
    attn_ptr, Kc_ptr, out_ptr,
    KV: tl.int32,
    Dn: tl.constexpr,           # 512
    BLOCK_K: tl.constexpr = 128
):
    # out_ptr is [Dn], attn_ptr is [KV], Kc_ptr is [KV, Dn]
    out_vec = tl.zeros((Dn,), dtype=tl.float32)
    for k0 in range(0, KV, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < KV
        attn_tile = tl.load(attn_ptr + k_idx, mask=mask_k, other=0.0)  # [BLOCK_K]
        Kc_tile = tl.load(Kc_ptr + k_idx[:, None] * Dn + tl.arange(0, Dn), mask=mask_k[:, None], other=0.0)  # [BLOCK_K, Dn]
        # out_vec += sum over kk of attn_tile[kk] * Kc_tile[kk, :]
        for kk in range(0, BLOCK_K):
            k_valid = k0 + kk < KV
            if k_valid:
                out_vec += attn_tile[k0 + kk] * tl.sum(Kc_tile[kk, :], axis=0)
    tl.store(out_ptr, out_vec)
