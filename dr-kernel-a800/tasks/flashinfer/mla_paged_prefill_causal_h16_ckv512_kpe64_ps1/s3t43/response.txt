- Inputs:

  - q_pe: [total_q, H, Kp], Kp=64
  - ckv_cache: [P, 1, K] → squeeze to [P, K]
  - kpe_cache: [P, 1, Kp] → squeeze to [P, Kp]
  - qo_indptr: [len_indptr], len_indptr = number of batches (e.g., 2). qo_indptr[0]=0, qo_indptr[1]=total_q.
  - kv_indptr: [len_indptr], kv_indptr[0]=0, kv_indptr[1]=num_tokens.
  - kv_indices: [num_tokens], token indices per batch.
  - sm_scale: float
- For each batch b and query i:
  - q_abs = q_start + i
  - qn = q_nope[q_abs] → [H, K], flatten to qn_vec_ptr length H*K
  - qp = q_pe[q_abs] → [H, Kp], flatten to qp_vec_ptr length H*Kp
  - For each head h:
    - tok_idx = kv_indices[kv_indptr[b]: kv_indptr[b+1]] → [L], int32
    - Compute logits_scaled[h, l] = sum_k qn[h, k] * Kc_all[tok_idx[l], k] + sum_kp qp[h, k'] * Kp_all[tok_idx[l], k'] for l in [0..L-1], scaled by sm_scale
    - Apply causal mask: invalid positions j > (L - q_len) + i; zero out invalid logits before reductions to avoid NaNs
    - lse[h] = logsumexp(logits_scaled[h, :]) / ln(2)
    - attn[h, l] = exp(logits_scaled[h, l] - lse[h]) for valid positions, invalid set to 0
    - out[h, :] = attn[h, :] @ Kc_all[tok_idx[:], :] (GEMV) → vector of length K, cast to bfloat16 and store

All math is done in Triton; forward does not use torch matmul, softmax, logsumexp, or elementwise reductions. Only allocations and launches remain in forward.

# Kernel 1: Compute per-head logits for tokens l in [0..L-1]
# qn_vec_ptr: *fp32, length H*K, q_nope[q_abs] flattened
# qp_vec_ptr: *fp32, length H*Kp, q_pe[q_abs] flattened
# Kc_ptr: *fp32, base pointer to Kc_all flattened (P*K)
# Kp_ptr: *fp32, base pointer to Kp_all flattened (P*Kp)
# tok_idx_ptr: *int32, length L
# L: number of tokens in this batch segment
# H, K, Kp: dimensions
# sm_scale: fp32 scalar
# out: logits_scaled for this head, length L (global pointer to buffer)
@triton.jit
def compute_logits_kernel(
    qn_vec_ptr, qp_vec_ptr, Kc_ptr, Kp_ptr, out_ptr,
    H, K, Kp, L,
    tok_idx_ptr,
    sm_scale,
):
    # out_ptr is a contiguous buffer of length H*L
    h = tl.program_id(0)
    base = h * L
    for j in range(0, L):
        # accumulate qn contribution
        acc = 0.0
        for k in range(0, K):
            qn_val = tl.load(qn_vec_ptr + h * K + k)
            idx = tl.load(tok_idx_ptr + j)
            kc_val = tl.load(Kc_ptr + idx * K + k)
            acc += qn_val * kc_val
        # accumulate qp contribution
        for kp in range(0, Kp):
            qp_val = tl.load(qp_vec_ptr + h * Kp + kp)
            idx = tl.load(tok_idx_ptr + j)
            kp_val = tl.load(Kp_ptr + idx * Kp + kp)
            acc += qp_val * kp_val
        acc *= sm_scale
        tl.store(out_ptr + base + j, acc)

# L: int
# lse_ptr: *fp32, scalar output
@triton.jit
def compute_lse_kernel(log_ptr, lse_ptr, L):
    # Compute max
    m = -1e20
    for j in range(0, L):
        v = tl.load(log_ptr + j)
        if v > m:
            m = v
    # Compute sum exp(v - m)
    s = 0.0
    for j in range(0, L):
        v = tl.load(log_ptr + j)
        s += tl.exp(v - m)

    lse_val = m + tl.log(s)
    # divide by ln(2)
    lse_val *= 1.4426950408889634  # 1 / ln(2)
    tl.store(lse_ptr, lse_val)


# Kernel 3: Compute softmax over a vector (scaled logits) and store attn in attn_ptr
# log_ptr: *fp32, input vector of length L (logits_scaled)
# lse_ptr: *fp32, scalar lse (per head)
# attn_ptr: *fp32, output vector of length L
@triton.jit
def compute_softmax_kernel(log_ptr, lse_ptr, attn_ptr, L):
    lse_val = tl.load(lse_ptr)
    for j in range(0, L):
        v = tl.load(log_ptr + j)
        a = tl.exp(v - lse_val)
        tl.store(attn_ptr + j, a)


# Kernel 4: GEMV for attn[h, :] @ Kc[tok_idx[:], :] → out_vec_ptr[h*K + k]
# attn_ptr: *fp32, vector of length L
# Kc_ptr: *fp32, base pointer to Kc_all flattened (P*K)
# tok_idx_ptr: *int32, length L
# K: int
# out_vec_ptr: *fp32, vector of length K, corresponds to output[h, :]
@triton.jit
def gemv_out_kernel(attn_ptr, Kc_ptr, tok_idx_ptr, out_vec_ptr, L, K):

    for k in range(0, K):
        acc = 0.0
        for l in range(0, L):
            attn_val = tl.load(attn_ptr + l)
            idx = tl.load(tok_idx_ptr + l)
            Kc_val = tl.load(Kc_ptr + idx * K + k)
