# We iterate over K and accumulate out[k] += attn[l] * Kc_rows[l, k].
@triton.jit

    attn_ptr, Kc_ptr, out_ptr,
    L, K,
    tok_idx_ptr,
    stride_K,
):
    # Triton cannot write out vectors directly from kernels; we call this per head in host loop.
    # Compute out[K] vector for a given head. We assume out_ptr is already allocated for this head.
    # attn_ptr: [L], Kc_ptr: [P, K]; we read rows using tok_idx_ptr.
    # We accumulate into out_ptr[k].
    for k in range(K):
        out_ptr[k] = tl.zeros((), dtype=tl.float32)
        for l in range(L):

            tok_l = tl.load(tok_idx_ptr + l)
            kc_val = tl.load(Kc_ptr + tok_l * stride_K + k)
            out_ptr[k] += attn_l * kc_val
