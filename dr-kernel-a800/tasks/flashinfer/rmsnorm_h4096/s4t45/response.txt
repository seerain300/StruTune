import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_weight_scale_kernel(
    hidden_ptr,       # *const input tensor (bfloat16/float16)
    weight_ptr,       # *const weight tensor
    out_ptr,          # *output tensor (same shape as hidden; dtype from allocation)
    B: tl.int32,      # batch size
    H: tl.int32,      # hidden size (expect 4096)
    EPS: tl.float32,  # epsilon
):
    # One program per row
    row_id = tl.program_id(0)
    # Compute offsets for this row
    # For a contiguous [B, H] tensor, row-major: offset = row_id * H + col
    # We'll create a vector of column indices
    cols = tl.arange(0, H)  # H is constexpr, 4096 in this task
    # Load x row: hidden_ptr is base pointer; row stride is H elements
    x = tl.load(hidden_ptr + row_id * H + cols)
    # Cast to float32 for math
    x_f32 = tl.cast(x, tl.float32)
    # Compute sum of squares
    sumsq = tl.sum(x_f32 * x_f32, axis=0)
    # Compute mean and inv_rms
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + EPS)
    # Load weight and cast to float32
    w = tl.load(weight_ptr + cols)
    w_f32 = tl.cast(w, tl.float32)
    # Compute output: y = x * inv_rms * w (all in f32)
    y_f32 = x_f32 * inv_rms * w_f32
    # Store; Triton will cast to the dtype of out_ptr if needed
    tl.store(out_ptr + row_id * H + cols, y_f32)

    def forward(self, hidden_states, weight, eps=1e-5):
        """
        hidden_states: [B, 4096], bfloat16 or float16
        weight: [4096], same dtype as hidden for performance; math is in float32
        """
        # Ensure CUDA tensors
        if not hidden_states.is_cuda or not weight.is_cuda:
            raise RuntimeError("ModelNew.forward requires CUDA tensors for Triton kernels.")
        # Ensure contiguous for simple indexing
        hidden = hidden_states.contiguous()
        weight = weight.contiguous()
        B, H = hidden.shape
        assert H == 4096, "This Triton kernel expects hidden size H=4096"
        # Allocate output tensor with same dtype and shape as hidden

        # Launch Triton kernel: one program per row
        grid = (B,)
        # Tune for best performance observed: num_warps=8, num_stages=2

            hidden, weight, out,
            B, H, eps,

        return out
