```python
import torch
import triton
import triton.language as tl

# Triton kernel: one program per row, vectorized across the full hidden_size (4096).
# Two-pass approach:
# - Pass 1: accumulate sum of squares across the row in float32, compute inv_rms.
# - Pass 2: scale and store output, multiplying by weight.
@triton.jit
def _row_fused_scale_kernel(
    hidden_ptr,       # *pointer* to input hidden states (original dtype)
    weight_ptr,       # *pointer* to weight (original dtype)
    out_ptr,          # *pointer* to output (original dtype)
    batch_size,       # int: number of rows
    EPS,              # float32 epsilon
    BLOCK_SIZE: tl.constexpr,  # number of columns per row (here 4096)
):
    pid = tl.program_id(axis=0)  # row id
    # Guard in case grid > batch_size (not expected, but safe)
    if pid >= batch_size:
        return

    # Base offsets for this row
    row_off = pid * BLOCK_SIZE

    # Pass 1: compute sum of squares and inv_rms
    sum_sq = 0.0
    offs = tl.arange(0, BLOCK_SIZE)
    # Loop over columns in chunks (here one chunk since BLOCK_SIZE == hidden_size)
    # Using a single chunk keeps code simple and avoids dynamic loops.
    x = tl.load(hidden_ptr + row_off + offs)
    x_f32 = x.to(tl.float32)
    sum_sq += tl.sum(x_f32 * x_f32)
    mean = sum_sq / BLOCK_SIZE
    inv_rms = tl.rsqrt(mean + EPS)

    # Pass 2: scale and store
    w = tl.load(weight_ptr)  # scalar weight, will broadcast
    y = (x_f32 * inv_rms) * w
    # Cast back to original dtype of hidden states for output
    y_cast = y.to(tl.float32)  # keep float32 output; the original code returns in hidden_states.dtype (bf16), but this kernel operates in f32 internally.
    # Note: The original PyTorch code casts back to hidden_states.dtype at the end of Python function, before returning.
    # To match that, we will allocate out as hidden_states.dtype and cast here accordingly.
    # Since Triton doesn't have direct access to the pointer's dtype, we rely on host to allocate out in desired dtype and let tl.store handle it.
    tl.store(out_ptr + row_off + offs, y_cast)  # store as float32; host will ensure out_ptr dtype matches desired final dtype.

# For this specific task, hidden_size is fixed at 4096. We choose BLOCK_SIZE=4096.
# The original Model.forward returns y.to(hidden_states.dtype), so we allocate output in that dtype on host.
class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor):
        # Ensure CUDA tensors
        assert hidden_states.is_cuda and weight.is_cuda, "Inputs must be CUDA tensors"
        # Shapes
        batch_size, hidden_size = hidden_states.shape
        assert hidden_size == 4096, "hidden_size must be 4096"
        # Work in float32 inside kernel for stability
        # Allocate output in the original dtype (match PyTorch behavior)
        out = torch.empty_like(hidden_states)
        EPS = 1e-5

        # Launch one program per row
        grid = (batch_size,)
        # Choose effective launch parameters; these worked best in prior runs
        _row_fused_scale_kernel[grid](
            hidden_states, weight, out,
            batch_size,
            EPS,
            BLOCK_SIZE=4096,
            num_warps=16,
            num_stages=1,
        )
        # Cast output back to hidden_states.dtype to match original behavior
        # The kernel stored float32; we need to cast to the desired dtype (likely bfloat16).
        # Since the original returns y.to(hidden_states.dtype), we perform this cast here.
        out = out.to(hidden_states.dtype)
        return out
```
