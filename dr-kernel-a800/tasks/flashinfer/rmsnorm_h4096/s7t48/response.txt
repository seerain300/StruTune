  - Compute inv_rms = rsqrt(mean(x^2) + EPS),

```python
import torch
import triton
import triton.language as tl

# Triton kernel: one program per row, single-pass fused computation.
# For each row b:
#   - Load x[b, :] (cast to float32),
#   - Compute inv_rms = rsqrt(mean(x[b, :]**2) + EPS),
#   - Scale and store y[b, :] = x[b, :] * inv_rms * weight (float32 output).
@triton.jit
def _row_fused_scale_singlepass_kernel(
    hidden_ptr,           # *ptr to hidden_states (original dtype, e.g., bfloat16)
    weight_ptr,           # *ptr to weight (original dtype, e.g., bfloat16)
    out_ptr,              # *ptr to output (float32)
    batch_size: tl.constexpr,
    hidden_size: tl.constexpr,        # must be 4096 for this model
    EPS: tl.constexpr,                # 1e-5
    BLOCK_SIZE: tl.constexpr          # 4096
):
    pid = tl.program_id(axis=0)  # one program per row
    base = pid * hidden_size

    # Load the entire row into float32
    offs = tl.arange(0, BLOCK_SIZE)
    mask = offs < hidden_size  # always true for hidden_size=4096, kept for generality
    x = tl.load(hidden_ptr + base + offs, mask=mask, other=0.0)
    x32 = x.to(tl.float32)

    # Compute inv_rms
    sumsq = tl.sum(x32 * x32, axis=0)  # scalar
    mean = sumsq / hidden_size
    inv_rms = tl.rsqrt(mean + EPS)     # scalar

    # Load weight vector (float32), scale, and store
    w = tl.load(weight_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    y32 = x32 * inv_rms * w

    # Store as float32
    tl.store(out_ptr + base + offs, y32, mask=mask)
```

```python
import torch
import triton
import triton.language as tl

# Triton kernel: one program per row, single-pass fused computation.
# For each row b:
#   - Load x[b, :] (cast to float32),
#   - Compute inv_rms = rsqrt(mean(x[b, :]**2) + EPS),
#   - Scale and store y[b, :] = x[b, :] * inv_rms * weight.
@triton.jit
def _row_fused_scale_singlepass_kernel(
    hidden_ptr,           # *ptr to hidden_states (original dtype, e.g., bfloat16)
    weight_ptr,           # *ptr to weight (original dtype, e.g., bfloat16)
    out_ptr,              # *ptr to output (same dtype as hidden_states)
    batch_size: tl.constexpr,
    hidden_size: tl.constexpr,        # must be 4096 for this model
    EPS: tl.constexpr,                # 1e-5
    BLOCK_SIZE: tl.constexpr          # 4096
):
    pid = tl.program_id(axis=0)  # one program per row
    base = pid * hidden_size

    # Load the entire row into float32
    offs = tl.arange(0, BLOCK_SIZE)
    mask = offs < hidden_size  # always true for hidden_size=4096, kept for generality
    x = tl.load(hidden_ptr + base + offs, mask=mask, other=0.0)
    x32 = x.to(tl.float32)

    # Compute inv_rms
    sumsq = tl.sum(x32 * x32, axis=0)  # scalar
    mean = sumsq / hidden_size
    inv_rms = tl.rsqrt(mean + EPS)     # scalar

    # Load weight vector, convert to float32, scale
    w = tl.load(weight_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    y32 = x32 * inv_rms * w

    # Store as output dtype (out_ptr type defines dtype)
    tl.store(out_ptr + base + offs, y32, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states, weight):
        # If not on CUDA, fall back to PyTorch implementation for correctness
        if (not hidden_states.is_cuda) or (not weight.is_cuda):
            x = hidden_states.to(torch.float32)
            inv_rms = torch.rsqrt(x.pow(2).mean(dim=-1, keepdim=True) + 1e-5)
            y = (x * inv_rms) * weight.to(torch.float32)
            return y.to(hidden_states.dtype)

        # Ensure tensors are contiguous
        hidden = hidden_states.contiguous()
        w = weight.contiguous()
        batch_size = hidden.shape[0]
        hidden_size = hidden.shape[1]
        assert hidden_size == 4096, "This Triton kernel expects hidden_size == 4096"

        # Output tensor in the same dtype as hidden_states
        out = torch.empty_like(hidden, dtype=hidden.dtype, device=hidden.device)

        # Choose launch config
        BLOCK_SIZE = 4096
        # Use a high-occupancy setting for memory-bound kernels
        grid = (batch_size,)
        # num_warps=32 often gives better throughput for 4096-element rows
        _row_fused_scale_singlepass_kernel[grid](
            hidden, w, out,
            batch_size=batch_size,
            hidden_size=hidden_size,
            EPS=1e-5,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=32,
            num_stages=1,
        )
        return out
```

คำอธิบาย
- ModelNew.forward:
  - หาก hidden_states หรือ weight ไม่ใช่ CUDA tensor, เราใช้ PyTorch implementation เพื่อให้รันได้ (preserves correctness on CPU).
  - หากเป็น CUDA, เราทำการ prepare tensors, แล้ว launch Triton kernel โดยกำหนด grid=(batch_size,) เพื่อให้แต่ละ row ถูกคำนวณโดย program หนึ่ง.
  - Kernel โหลดแถวทั้งหมด, คำนวณ inv_rms โดยใช้ float32 สำหรับ precision, แล้ว scale โดย weight.float() และ store ผลลัพธ์ใน dtype ของ out tensor (ซึ่งเรา allocate เป็น same dtype ของ hidden_states).
- ประโยชน์:
  - ใช้ single-pass ลดการอ่าน memory ลง (ไม่ต้อง read hidden_states เก็บ sum ก่อนอีกครั้ง), แต่เนื่องจาก Triton kernel ไม่มีทางนำผล sum ไปเก็บไว้ใน register ระหว่าง kernel (ในทางปฏิบัติเราทำ sum และ scale ใน kernel เดียวกัน), เราใช้วิธี compute ในพื้นที่เดียว 4096 elements แล้ว store ออกมา โดยใช้ float32 ในคำนวณเพื่อความแม่นยำ แล้ว store ใน dtype ของ out
- การเลือก num_warps=32 ช่วยให้ kernel มี occupancy สูงขึ้นสำหรับแถว 4096 elements และอาจเพิ่มความเร็วเมื่อเทียบกับ 16 warps

หมายเหตุ
- ถ้าต้องการ strict matching ของ dtype ในการ compute (คือ x และ weight ใน float32) และ store กลับไปเป็น dtype ของ hidden_states, เราสามารถเปลี่ยน tl.store(out_ptr + base + offs, y32.to(tl.float32), mask=mask) ซึ่งในที่นี้ out มี dtype ที่เราต้องการ คือ dtype ของ hidden_states. ในโค้ดนี้เราใช้ out = torch.empty_like(hidden) ที่มี dtype เหมือน hidden_states, และ tl.store จะใส่ค่าเป็น dtype ของ out_ptr (float32 หรืออื่น) ดังนั้นเราควร store เป็น y32 ซึ่งถ้า out เป็น bfloat16, Triton จะ cast implicit ที่ store; แต่ใน Triton เราควร specify ได้โดยการสร้าง tensor ที่มี dtype ที่ต้องการและ store y32 ลงไปซึ่งจะถูก cast โดย Triton ในตอน store ไปยัง out_ptr ที่มี dtype ที่เราสร้างไว้. แต่ในที่นี้เราอยากให้ output dtype ตรงกับ hidden_states.dtype ดังนั้นเราจึงสร้าง out เป็น empty_like(hidden) และ store y32 ลงไป ซึ่ง Triton อนุญาตการ cast ไปยัง dtype ของ destination pointer.

คุณสมบัติที่รักษาไว้
- ไม่ใช้ torch operators ใน host นอกเหนือจากการ prepare tensors และการ create output tensor
- ใช้ Triton kernel เท่านั้นในการคำนวณ
- ใช้งานได้กับ batch_size ที่แตกต่างกันโดย grid=(batch_size,) ซึ่งตรงกับ one program per row
- คำนวณใน float32 เพื่อความแม่นยำเหมือน PyTorch implementation แล้ว return ใน dtype ของ hidden_states

หากยังไม่พอใจความเร็วมากที่สุด, สามารถทดลองเปลี่ยน num_warps=16 กลับมาดูผลลัพธ์ หรือทดลอง num_stages=2; แต่ในประสบการณ์ครั้งก่อน num_warps=32, num_stages=1 สามารถให้ throughput สูงขึ้นเล็กน้อยสำหรับ rows ขนาด 4096 elements.
