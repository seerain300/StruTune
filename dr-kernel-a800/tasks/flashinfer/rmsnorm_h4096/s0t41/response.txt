import triton.language as tl

# Kernel 1: per-row sum of squares -> inv_rms
@triton.jit
def reduce_row_sumsq_kernel(x_ptr, inv_rms_ptr, B, H, EPS, BLOCK_SIZE: tl.constexpr):
    row_id = tl.program_id(axis=0)
    # Safety: if grid > B, mask out
    # We set grid=(B,) so row_id < B, but keep a guard for robustness
    if row_id >= B:
        return

    # Accumulator in fp32
    sumsq = tl.zeros((), dtype=tl.float32)

    # Loop over columns in tiles of size BLOCK_SIZE
    for start in range(0, H, BLOCK_SIZE):
        cols = start + tl.arange(0, BLOCK_SIZE)
        mask = cols < H
        # Compute pointer to the row segment
        x_off = row_id * H + cols
        x_vals = tl.load(x_ptr + x_off, mask=mask, other=0.0)
        x_vals = x_vals.to(tl.float32)
        sumsq += tl.sum(x_vals * x_vals, axis=0)

    # Compute inv_rms = 1 / sqrt(mean + EPS)
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + EPS)
    tl.store(inv_rms_ptr + row_id, inv_rms)
