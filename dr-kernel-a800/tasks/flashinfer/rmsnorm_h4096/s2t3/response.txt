def kernel_rms(
    x_ptr,                # *T, [B, H] (T can be bf16/fp16/fp32; cast to fp32 in kernel)
    out_inv_rms_ptr,      # *fp32, [B]
    B, H, EPS,            # int32, fp32
    stride_x_row, stride_x_col,
    BLOCK_N: tl.constexpr,
):

    if row >= B:
        return

    sum_sq = 0.0
    col_start = 0

        mask = cols < H

        col_start += BLOCK_N
