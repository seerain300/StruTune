Recap
- Per row:
  - Compute inv_rms = 1 / sqrt(mean(x^2) + eps), where mean is over the 4096 elements.
  - Output y = (x * inv_rms) * weight.
- Memory traffic is unavoidable: read each row twice and write once. Writing intermediates (like x^2 or inv_rms) to global memory would only add bandwidth and slow things down.

Plan
- Keep one Triton program per row, two passes:
  - First pass: accumulate sum of squares to derive inv_rms.
  - Second pass: re-read x, multiply by inv_rms and weight, and store output.
- Tiling and warps:
  - For hidden_size=4096, using BLOCK_SIZE=1024 yields 4 loop iterations; BLOCK_SIZE=512 yields 8 iterations. We’ll keep 1024 with 8 warps, which has performed well and reduced loop iterations.
  - num_stages=3 to pipeline loads.
- Compute precision: Keep compute in float32 and cast only at store, matching original behavior. Masking ensures correctness for any size, even though we assert 4096.

```python
import torch
import triton
import triton.language as tl

# Triton kernel: per-row RMS scaling with weight.
# One program per row. We read x twice: once to compute sum of squares, once to write output.
@triton.jit
def _rms_scale_kernel(
    hidden_ptr,        # *float32, pointer to hidden_states (cast to float32)
    weight_ptr,        # *float32, pointer to weight (cast to float32)
    out_ptr,           # *T_out, pointer to output (will store casted result)
    batch_size,        # int32
    hidden_size,       # int32
    eps,               # float32
    out_dtype_code: tl.constexpr,  # 0: fp32, 1: fp16, 2: bf16
    BLOCK_SIZE: tl.constexpr        # e.g., 1024
):
    row_id = tl.program_id(0)  # one program per row
    # Accumulate sum of squares across the row
    sumsq = 0.0
    # First pass
    for col in range(0, hidden_size, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < hidden_size
        x = tl.load(hidden_ptr + row_id * hidden_size + offs, mask=mask, other=0.0)
        sumsq += tl.sum(x * x, axis=0)
    mean = sumsq / hidden_size
    inv_rms = 1.0 / tl.sqrt(mean + eps)

    # Second pass: scale and write
    for col in range(0, hidden_size, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < hidden_size
        x = tl.load(hidden_ptr + row_id * hidden_size + offs, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        y = x * inv_rms * w
        # Cast to target dtype
        if out_dtype_code == 0:
            y_cast = y  # fp32
        elif out_dtype_code == 1:
            y_cast = y.to(tl.float16)
        elif out_dtype_code == 2:
            y_cast = y.to(tl.bfloat16)
        else:
            y_cast = y  # default fp32
        tl.store(out_ptr + row_id * hidden_size + offs, y_cast, mask=mask)

def run(hidden_states, weight):
    batch_size, hidden_size = hidden_states.shape
    assert hidden_size == 4096

    EPS = 1e-5

    # Ensure inputs are on CUDA and contiguous
    h = hidden_states.contiguous()
    w = weight.contiguous()

    # Compute in float32 for numerical stability
    h32 = h.to(torch.float32)
    w32 = w.to(torch.float32)

    out = torch.empty_like(h32)  # fp32 intermediate, will cast at store

    # Determine dtype code for casting
    if hidden_states.dtype == torch.float32:
        out_dtype_code = 0
    elif hidden_states.dtype == torch.float16:
        out_dtype_code = 1
    elif hidden_states.dtype == torch.bfloat16:
        out_dtype_code = 2
    else:
        out_dtype_code = 0  # default fp32

    # Launch Triton kernel: one program per row
    grid = (batch_size,)
    _rms_scale_kernel[grid](
        h32, w32, out,
        batch_size, hidden_size, EPS,
        out_dtype_code=out_dtype_code,
        BLOCK_SIZE=1024,
        num_warps=8,
        num_stages=3
    )

    # Cast output to original dtype and shape matches input
    return out.to(hidden_states.dtype)

def get_inputs():
    hidden_states = torch.randn([1, 4096], dtype=torch.bfloat16, device='cuda')
    weight = torch.randn([4096], dtype=torch.bfloat16, device='cuda')
    return [hidden_states, weight]

def fused_operator(tensor_0, tensor_1):
    _out = run(tensor_0, tensor_1)
    return list(_out) if isinstance(_out, (tuple, list)) else [_out]

class ModelNew(torch.nn.Module):
    def forward(self, *args):
        return run(*args)
```
