- My previous Triton implementation used three kernels:

  - Kernel 2: compute inv_rms from sums.
  - Kernel 3: compute final output y using inv_rms.
  - That’s three launches and an intermediate tensor for inv_rms, plus reading hidden_states for the output kernel again.

- Fuse steps 1 and 3 into a single Triton kernel that:
  - Computes per-row sum of squares in a first pass across tiles (columns).
  - Computes inv_rms[i] using EPS and H.
  - In a second pass across the same tiles, computes y = x * inv_rms[i] * w and writes to output.
- This reduces kernel launches from 3 to 1 per forward and avoids reading hidden_states twice. The tradeoff is two tiled passes per row within one kernel, which is still efficient because:
  - The hidden_states tensor is small (H=4096) and typical batch sizes are modest; one fused kernel per batch row keeps data in fast registers and avoids an intermediate write/read of inv_rms.
  - This approach minimizes host-side overhead and memory traffic.

- One Triton program per row (grid=(B,)).
- Inner loop over columns in tiles of BLOCK_SIZE (e.g., 256 or 512), vectorized using tl.arange.
- Masking handles tails when H is not a multiple of BLOCK_SIZE.
- num_warps tuned to 4 and num_stages to 2 for balance; H=4096 fits well with BLOCK_SIZE=256 or 512.

import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Fused Triton kernel:
# - Computes per-row sum of squares across all columns.
# - Computes inv_rms[i] = 1.0 / sqrt((sum_sq[i] / H) + EPS).
# - In a second pass, writes y[i, j] = x[i, j] * inv_rms[i] * w[j] to out_f32.
@triton.jit
def fused_rms_scale_kernel(
    hidden_ptr,         # *const float, [B, H]
    weight_ptr,         # *const float, [H]
    out_ptr,            # *float, [B, H]
    H: tl.constexpr,    # int, hidden size
    EPS: tl.constexpr,  # float
    BLOCK_SIZE: tl.constexpr,  # tile size over columns
):
    row = tl.program_id(axis=0)  # each program handles one row
    # Base pointers for this row
    # Triton expects element strides, so we just iterate with row * H + col
    # We'll compute offsets explicitly to be safe.
    # We'll use a loop over tiles of size BLOCK_SIZE to cover all columns.
    # Accumulate sum of squares in fp32
    sum_sq = 0.0
    # First pass: compute sum of squares for this row
    for col_start in range(0, H, BLOCK_SIZE):
        cols = col_start + tl.arange(0, BLOCK_SIZE)
        mask = cols < H
        x = tl.load(hidden_ptr + row * H + cols, mask=mask, other=0.0)
        sum_sq += tl.sum(x * x, axis=0)

    # Compute inv_rms for this row: 1 / sqrt(mean + EPS)
    mean = sum_sq / H
    inv_rms = 1.0 / tl.sqrt(mean + EPS)

    # Second pass: compute output y = x * inv_rms * weight and store
    for col_start in range(0, H, BLOCK_SIZE):
        cols = col_start + tl.arange(0, BLOCK_SIZE)
        mask = cols < H
        x = tl.load(hidden_ptr + row * H + cols, mask=mask, other=0.0)
        w = tl.load(weight_ptr + cols, mask=mask, other=0.0)
        y = x * inv_rms * w
        tl.store(out_ptr + row * H + cols, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states, weight):
        # If Triton not available or not on CUDA, fall back to the original PyTorch implementation.
        if (not TRITON_AVAILABLE) or (hidden_states.device.type != "cuda"):
            # Fallback: replicate original behavior
            batch_size, hidden_size = hidden_states.shape
            assert hidden_size == 4096
            x = hidden_states.to(torch.float32)
            EPS = 1e-5
            inv_rms = torch.rsqrt(x.pow(2).mean(dim=-1, keepdim=True) + EPS)
            y = (x * inv_rms) * weight.to(torch.float32)
            return y.to(hidden_states.dtype)

        # Ensure inputs are on CUDA and float32 contiguous
        orig_device = hidden_states.device
        x = hidden_states.contiguous().to(torch.float32)  # [B, H]
        w = weight.contiguous().to(torch.float32)        # [H]

        B, H = x.shape
        assert H == 4096, "This implementation assumes hidden_size == 4096."

        # Allocate output as float32
        out_f32 = torch.empty((B, H), dtype=torch.float32, device=x.device)

        # Choose tile size; 256 works well for H=4096, adjust warps/stages
        BLOCK_SIZE = 256
        # Launch fused kernel: one program per row
        grid = (B,)
        fused_rms_scale_kernel[grid](
            x, w, out_f32, H, 1e-5, BLOCK_SIZE, num_warps=4, num_stages=2
        )

        # Cast to original dtype and return on original device
        out = out_f32.to(hidden_states.dtype)
        if orig_device.type != "cuda":
            out = out.to(orig_device)
        return out
