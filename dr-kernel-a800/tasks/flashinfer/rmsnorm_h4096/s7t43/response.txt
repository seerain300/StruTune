import triton
import triton.language as tl

# Two-pass approach:
# - Pass 1: accumulate sum of squares across the row in float32, compute inv_rms.
# - Pass 2: scale and store output, multiplying by weight.

                             batch_size: tl.constexpr, hidden_size: tl.constexpr, EPS: tl.constexpr,
                             BLOCK_SIZE: tl.constexpr):

    pid = tl.program_id(axis=0)
    # Vector of column indices for the row
    offs = tl.arange(0, BLOCK_SIZE)

    # Pass 1: compute sum of squares in float32
    x = tl.load(hidden_ptr + pid * hidden_size + offs)  # x is in the original dtype
    x32 = x.to(tl.float32)
    sumsq = tl.sum(x32 * x32)
    mean = sumsq / hidden_size

    # Pass 2: scale and store output (cast to original dtype of hidden_states)
    w = tl.load(weight_ptr + offs).to(tl.float32)
    y32 = x32 * inv_rms * w

    # Ensure we store the scaled values.

    # In Python we'll allocate out as empty_like(hidden_states) so out_ptr dtype matches.
    tl.store(out_ptr + pid * hidden_size + offs, y32)
