- Operation recap:

  - For each row i:

    - inv_rms = rsqrt((sumsq / 4096) + EPS)
    - y[i, j] = x[i, j] * inv_rms * weight[j]

- Current strategy:
  - One Triton program per row. Load the entire row (4096 elements) into registers, compute sumsq, inv_rms, then compute output and store. This is a single pass and minimizes memory traffic.
- Performance observations:
  - The best performance in your environment used num_warps=8 and num_stages=2. num_warps=16 regressed slightly. We’ll keep those launch parameters.
  - Micro-optimizations:

    - Use rsqrt(mean + EPS).

    - Avoid unnecessary tensor reshapes or recomputation.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_weight_scale_kernel(
    hidden_ptr,       # *const input (e.g., bfloat16/float16)
    weight_ptr,       # *const weight (we'll load and cast to float32 in-kernel)
    out_ptr,          # *output (same dtype as hidden, e.g., bfloat16)
    B: tl.int32,      # batch size
    H: tl.int32,      # hidden size (4096 in this task)
    EPS: tl.float32,  # epsilon
    hidden_stride0: tl.int32,  # stride along batch dim for hidden
    hidden_stride1: tl.int32,  # stride along hidden dim for hidden
    out_stride0: tl.int32,     # stride along batch dim for output
    out_stride1: tl.int32,     # stride along hidden dim for output
    BLOCK_SIZE: tl.constexpr,  # set to H=4096 at launch
):
    # One program per row
    row_id = tl.program_id(axis=0)
    # Hints for alignment/vectorization: for contiguous tensors, strides are 1 for the last dim
    tl.multiple_of(hidden_stride1, 1)
    tl.multiple_of(out_stride1, 1)

    # Column offsets
    cols = tl.arange(0, BLOCK_SIZE)
    # Masks (always true for BLOCK_SIZE == H)
    mask = cols < H

    # Compute base pointers for this row
    hidden_row_ptr = hidden_ptr + row_id * hidden_stride0
    out_row_ptr = out_ptr + row_id * out_stride0

    # Load row values and cast to float32 for robust math
    x = tl.load(hidden_row_ptr + cols * hidden_stride1, mask=mask, other=0.0)
    x = tl.cast(x, tl.float32)

    # Compute sum of squares
    sumsq = tl.sum(x * x, axis=0)
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + EPS)

    # Load weight vector and cast to float32
    w = tl.load(weight_ptr + cols, mask=mask, other=0.0)  # weight is 1D contiguous
    w = tl.cast(w, tl.float32)

    # Compute output
    y = x * inv_rms * w

    # Store result (store will cast to out_ptr dtype automatically)
    tl.store(out_row_ptr + cols * out_stride1, y, mask=mask)


def _run_triton(hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of the PyTorch run function:
      y = (hidden_states.float() / rms) * weight.float()
      where rms = rsqrt(mean(hidden_states^2, dim=-1) + EPS), per row
    """
    assert hidden_states.ndim == 2, "hidden_states must be 2D [B, H]"
    assert weight.ndim == 1, "weight must be 1D [H]"
    B, H = hidden_states.shape
    assert H == 4096, "This Triton kernel expects H=4096"

    # Ensure contiguous
    hidden = hidden_states.contiguous()
    weight_f32 = weight.to(torch.float32).contiguous()

    out = torch.empty_like(hidden)

    # Launch configuration: one program per row
    grid = (B,)

    # Hints for codegen: BLOCK_SIZE must be H
    BLOCK_SIZE = H

    # The best-performing launch params in your environment were num_warps=8, num_stages=2
    _layernorm_weight_scale_kernel[grid](
        hidden, weight_f32, out,
        B, H, 1e-5,
        hidden.stride(0), hidden.stride(1),
        out.stride(0), out.stride(1),
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,  # empirically best
        num_stages=2, # empirically best
    )

    # Cast output back to original dtype if needed (PyTorch version casts to hidden_states.dtype at the end)
    # Note: out is already same shape; we keep dtype as hidden, which is fine since we return out directly.
    return out


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
        return _run_triton(hidden_states, weight)
```

- Launch Params는 이 환경에서 좋은 성능을 보여준 num_warps=8, num_stages=2를 고정합니다.
- Triton 내에서 원자적 행 처리와 일관된 메모리 접근 패턴을 유지합니다.

```python
import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_weight_scale_kernel(
    hidden_ptr,       # *const input tensor (e.g., bfloat16/float16)
    weight_ptr,       # *const weight (float32)
    out_ptr,          # *output tensor (same shape as hidden)
    B: tl.int32,      # batch size
    H: tl.int32,      # hidden size (4096)
    EPS: tl.float32,  # epsilon
    hidden_stride0: tl.int32,  # stride along batch dim for hidden
    hidden_stride1: tl.int32,  # stride along hidden dim for hidden
    out_stride0: tl.int32,     # stride along batch dim for output
    out_stride1: tl.int32,     # stride along hidden dim for output
    BLOCK_SIZE: tl.constexpr,  # columns processed per program (H)
):
    # One Triton program per row
    row_id = tl.program_id(axis=0)

    # Column indices
    cols = tl.arange(0, BLOCK_SIZE)
    mask = cols < H  # always true when BLOCK_SIZE == H

    # Base pointers for this row
    hidden_row_ptr = hidden_ptr + row_id * hidden_stride0
    out_row_ptr = out_ptr + row_id * out_stride0

    # Load the entire row in its native dtype and cast to float32 for robust math
    x = tl.load(hidden_row_ptr + cols * hidden_stride1, mask=mask, other=0.0)
    x = tl.cast(x, tl.float32)

    # Compute sum of squares across the row
    sumsq = tl.sum(x * x, axis=0)
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + EPS)

    # Load weight vector and cast to float32
    w = tl.load(weight_ptr + cols, mask=mask, other=0.0)  # weight is 1D contiguous
    w = tl.cast(w, tl.float32)

    # Compute output: y = x * inv_rms * w
    y = x * inv_rms * w

    # Store result; Triton will cast to out_ptr's element type if needed
    tl.store(out_row_ptr + cols * out_stride1, y, mask=mask)


def _run_triton(hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of:
      y = (hidden_states.float() / rsqrt(mean(hidden_states^2, dim=-1) + EPS)) * weight.float()
    Returns y with the same shape and dtype as hidden_states.
    """
    assert hidden_states.ndim == 2, "hidden_states must be 2D [B, H]"
    assert weight.ndim == 1, "weight must be 1D [H]"
    B, H = hidden_states.shape
    assert H == 4096, "This Triton kernel expects H=4096"

    # Ensure contiguous tensors
    hidden = hidden_states.contiguous()
    # Convert weight to float32 for numerics
    weight_f32 = weight.to(torch.float32).contiguous()

    out = torch.empty_like(hidden)

    # Launch: one program per row
    grid = (B,)

    # Empirically high-performance launch params in the evaluation environment
    _layernorm_weight_scale_kernel[grid](
        hidden, weight_f32, out,
        B, H, 1e-5,
        hidden.stride(0), hidden.stride(1),
        out.stride(0), out.stride(1),
        BLOCK_SIZE=H,
        num_warps=8,   # high performance
        num_stages=2,  # high performance
    )

    return out


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
        # Triton-only computation
        return _run_triton(hidden_states, weight)
```

설명
- 한 프로그램이 한 행을 처리하며, 전체 4096 열을 단일 패스로 처리합니다. 이는 연속된 메모리 접근을 활용하고, RMS 계산 후 즉시 스케일링을 수행하여 결과를 생성하는 간단하면서도 효과적인 방법입니다.
- Launch Params(num_warps=8, num_stages=2)는 이 환경에서 좋은 성능을 보여준 설정입니다. 이를 고정하여 재현성과 안정성을 높였습니다.
- 모든 수치 계산은 float32로 이루어지며, 결과는 hidden_states와 같은 dtype으로 반환됩니다.
- 추가적인 복잡한 최적화를 피하고, 이미 안정적인 성능을 유지합니다.
