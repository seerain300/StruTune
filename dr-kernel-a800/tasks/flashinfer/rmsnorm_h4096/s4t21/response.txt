```python
import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_weight_scale_kernel(
    hidden_ptr,       # *const input (e.g., bfloat16/float16)
    weight_ptr,       # *const weight (float32, host provides)
    out_ptr,          # *output
    B: tl.int32,      # batch size
    H: tl.int32,      # hidden size (expected 4096, but we handle general H with mask)
    EPS: tl.float32,  # epsilon
    hidden_stride0: tl.int32,  # stride along batch dim for hidden
    hidden_stride1: tl.int32,  # stride along hidden dim for hidden
    out_stride0: tl.int32,     # stride along batch dim for output
    out_stride1: tl.int32,     # stride along hidden dim for output
    BLOCK_SIZE: tl.constexpr,  # number of columns processed per program (must be constexpr)
):
    # Program id along batch dimension
    row_id = tl.program_id(0)
    # If grid > B, we could guard, but here we set grid=B, so no need

    # Compute row pointers (supporting non-contiguous via strides)
    row_hidden = hidden_ptr + row_id * hidden_stride0
    row_out = out_ptr + row_id * out_stride0

    # Column indices vector (constexpr size)
    cols = tl.arange(0, BLOCK_SIZE)

    # Mask for columns within H (robustness in case H != BLOCK_SIZE)
    mask = cols < H

    # Load x row and cast to float32 for math
    x = tl.load(row_hidden + cols * hidden_stride1, mask=mask, other=0.0)
    x = tl.cast(x, tl.float32)

    # Compute sum of squares across the row
    sumsq = tl.sum(x * x, axis=0)

    # Compute inv_rms
    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + EPS)

    # Load weight vector (float32), masked to BLOCK_SIZE, but we still multiply elementwise
    # If H < BLOCK_SIZE, extra weights are ignored via masking; if H == BLOCK_SIZE, all are used
    w = tl.load(weight_ptr + cols, mask=cols < H, other=0.0)  # weight_ptr is float32

    # Compute output: y = x * inv_rms * w
    y = x * inv_rms * w

    # Store result (Triton will cast to the pointer's element type if needed; we keep output dtype same as hidden)
    tl.store(row_out + cols * out_stride1, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
        # Ensure CUDA tensors and contiguity
        assert hidden_states.is_cuda and weight.is_cuda, "Inputs must be CUDA tensors"
        hidden = hidden_states.contiguous()
        weight = weight.to(torch.float32).contiguous()

        B, H = hidden.shape
        assert H == 4096, "This kernel expects hidden size to be 4096"

        out = torch.empty_like(hidden)

        # Launch one program per row
        grid = (B,)

        # Use the best-performing launch parameters in this environment
        _layernorm_weight_scale_kernel[grid](
            hidden, weight, out,
            B, H, 1e-5,
            hidden.stride(0), hidden.stride(1),
            out.stride(0), out.stride(1),
            BLOCK_SIZE=4096,  # constexpr: Triton requires this to be a compile-time constant
            num_warps=8,      # best-performing configuration in your environment
            num_stages=2,
        )

        return out
```

):
