Final code
```python
import torch
import triton
import triton.language as tl

# Triton kernel: one program per row, vectorized across the full hidden_size (4096).
# Two-pass approach (best for clarity and performance on varied GPUs):
# - Pass 1: accumulate sum of squares across the row in float32, compute inv_rms.
# - Pass 2: scale and store output, multiplying by weight.
@triton.jit
def _row_fused_scale_kernel(hidden_ptr, weight_ptr, out_ptr,
                             batch_size, hidden_size: tl.constexpr, EPS: tl.constexpr,
                             BLOCK_SIZE: tl.constexpr):
    # One program per row
    pid = tl.program_id(axis=0)
    # Vector of column offsets
    offs = tl.arange(0, BLOCK_SIZE)

    # Pass 1: compute sum of squares in float32 and derive inv_rms
    sumsq = tl.zeros((), dtype=tl.float32)
    # We iterate in chunks to ensure robustness even if hidden_size != BLOCK_SIZE
    for start in range(0, hidden_size, BLOCK_SIZE):
        idx = start + offs
        mask = idx < hidden_size
        # Load x as float32
        x = tl.load(hidden_ptr + pid * hidden_size + idx, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sumsq += tl.sum(x * x, axis=0)

    mean = sumsq / hidden_size
    inv_rms = tl.rsqrt(mean + EPS)

    # Pass 2: scale and store output
    for start in range(0, hidden_size, BLOCK_SIZE):
        idx = start + offs
        mask = idx < hidden_size
        # Load x and weight as float32
        x = tl.load(hidden_ptr + pid * hidden_size + idx, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        y = x * inv_rms * w
        # Store y (Triton will write float32; we'll cast on host if needed)
        tl.store(out_ptr + pid * hidden_size + idx, y, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
        # Ensure inputs are contiguous and on CUDA
        assert hidden_states.is_cuda and weight.is_cuda, "Inputs must be CUDA tensors."
        hidden_states = hidden_states.contiguous()
        weight = weight.contiguous()
        batch_size, hidden_size = hidden_states.shape
        assert hidden_size == 4096, "hidden_size must be 4096."

        # Output tensor in float32 (compute precision), cast to original dtype after
        out_fp32 = torch.empty_like(hidden_states, dtype=torch.float32)

        # Launch Triton kernel: one program per row
        grid = (batch_size,)
        # Fixed configuration that previously performed best on varied workloads
        _row_fused_scale_kernel[grid](
            hidden_states, weight, out_fp32,
            batch_size=batch_size,
            hidden_size=hidden_size,
            EPS=1e-5,
            BLOCK_SIZE=4096,
            num_warps=16,
            num_stages=1,
        )

        # Cast back to original dtype to match original behavior
        return out_fp32.to(hidden_states.dtype)
```

Notes
- This keeps correctness and avoids any torch computation in the host code other than allocation and casting. The kernel is the only computational component.
- The two-pass approach is simple and robust. The best performance observed was with num_warps=16 and num_stages=1. I’ve set the kernel to use that configuration. If further tuning is desired for specific hardware, num_warps can be adjusted, but this default is strong across typical GPUs.
- All math inside the kernel is performed in float32 for stability, then the output is cast back to the original dtype to match the original model’s behavior.
