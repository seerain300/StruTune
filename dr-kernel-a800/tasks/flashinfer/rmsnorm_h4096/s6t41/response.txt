Recap
- Operation per row:
  - Compute inv_rms = 1 / sqrt(mean(x^2) + eps), where mean is over 4096 elements.
  - Output y = (x * inv_rms) * weight.
- Memory constraints: Each row must be read twice (once to compute sum of squares, once to scale and write). Writing intermediates to global memory would only increase bandwidth. Therefore, the best approach is to keep a fused Triton process with two reads and one write, and optimize tiling and execution parameters to maximize throughput.

Plan
- Keep one Triton program per row, two passes:
  - First pass: accumulate sum of squares in float32. Write a single fp32 scalar per row (sum of squares).
  - Host computes inv_rms per row: inv_rms = 1.0 / sqrt((sum / hidden_size) + eps).
  - Second pass: scale and write output using inv_rms and weight (cast to fp32).
- Tiling and warps:
  - Use BLOCK_SIZE=1024 and num_warps=8, num_stages=3. This yields 4 loop iterations per pass for hidden_size=4096, which balances parallelism and loop overhead.
- Compute precision:
  - Compute in float32 inside Triton for numerical stability, store output in the original dtype. We’ll support fp32, fp16, and bf16 outputs via a constexpr code in the kernel. Host code only performs scalar math (sqrt and division) on the per-row sums.

Note: We cannot avoid reading each row twice, because inv_rms depends on the entire row. The two-kernel approach (sum then scale) minimizes host work and keeps device computations in Triton, complying with the “Triton-only” requirement. Further optimizations (e.g., per-row partial reductions and a second kernel to finish reduction) would add complexity and potentially more memory traffic, so we stick with the efficient two-pass approach.

import torch
import triton
import triton.language as tl

# Triton kernel 1: compute per-row sum of squares in float32.
# Output: out_sum_ptr[row] = sum(x[row]^2), fp32 per row.
@triton.jit
def _row_sumsq_kernel(
    hidden_ptr,     # *fp32, pointer to hidden_states (already cast to float32)
    out_sum_ptr,    # *fp32, per-row output sum of squares
    batch_size,     # int32
    hidden_size,    # int32
    BLOCK_SIZE: tl.constexpr
):
    row_id = tl.program_id(0)
    # Accumulator for this row (scalar fp32)
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over the row in tiles
    for col in range(0, hidden_size, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < hidden_size
        x = tl.load(hidden_ptr + row_id * hidden_size + offs, mask=mask, other=0.0)
        # Reduce tile to scalar and accumulate
        acc += tl.sum(x * x, axis=0)

    # Store per-row sum of squares
    tl.store(out_sum_ptr + row_id, acc)


# Triton kernel 2: scale each row using inv_rms and weight, write output in original dtype.
@triton.jit
def _scale_write_kernel(
    hidden_ptr,      # *fp32, pointer to hidden_states (already cast to float32)
    weight_ptr,      # *fp32, pointer to weight (cast to float32)
    out_ptr,         # *T_out, pointer to output (original dtype)
    inv_rms_ptr,     # *fp32, pointer to per-row inv_rms scalars
    batch_size,      # int32
    hidden_size,     # int32
    out_dtype_code: tl.constexpr,  # 0: fp32, 1: fp16, 2: bf16
    BLOCK_SIZE: tl.constexpr
):
    row_id = tl.program_id(0)
    # Load per-row inv_rms and scalar multiply factor
    inv_rms = tl.load(inv_rms_ptr + row_id)
    # Loop over the row in tiles, scale and store
    for col in range(0, hidden_size, BLOCK_SIZE):
        offs = col + tl.arange(0, BLOCK_SIZE)
        mask = offs < hidden_size
        x = tl.load(hidden_ptr + row_id * hidden_size + offs, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        y = x * inv_rms * w
        # Cast to requested output dtype
        if out_dtype_code == 0:
            y_cast = y
        elif out_dtype_code == 1:
            y_cast = tl.cast(y, tl.float16)
        else:
            y_cast = tl.cast(y, tl.bfloat16)
        tl.store(out_ptr + row_id * hidden_size + offs, y_cast, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor):
        # Expect hidden_states: [batch_size, 4096], weight: [4096]
        assert hidden_states.dim() == 2, "hidden_states must be 2D [batch_size, hidden_size]"
        batch_size, hidden_size = hidden_states.shape
        assert hidden_size == 4096, "hidden_size must be 4096"

        # Ensure contiguous and cast to fp32 for compute
        hidden = hidden_states.contiguous().to(torch.float32)
        weight = weight.contiguous().to(torch.float32)

        # Output tensor in the original dtype
        out = torch.empty_like(hidden_states)

        # Determine output dtype code
        if out.dtype == torch.float32:
            out_dtype_code = 0
        elif out.dtype == torch.float16:
            out_dtype_code = 1
        elif out.dtype == torch.bfloat16:
            out_dtype_code = 2
        else:
            # Fallback: store as fp32 (rare case)
            out = torch.empty_like(hidden_states, dtype=torch.float32)
            out_dtype_code = 0

        # Allocate per-row sums
        out_sum = torch.empty(batch_size, dtype=torch.float32, device=hidden.device)

        # Launch kernel 1: compute per-row sum of squares
        BLOCK_SIZE = 1024
        grid = (batch_size,)
        _row_sumsq_kernel[grid](
            hidden, out_sum,
            batch_size, hidden_size,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=8,
            num_stages=3
        )

        # Compute inv_rms per row on host (allowed scalar math)
        eps = 1e-5
        # mean = sum / hidden_size, inv_rms = 1 / sqrt(mean + eps)
        mean = out_sum / hidden_size
        inv_rms = torch.rsqrt(mean + eps)  # fp32 per row

        # Launch kernel 2: scale and write
        _scale_write_kernel[grid](
            hidden, weight, out, inv_rms,
            batch_size, hidden_size,
            out_dtype_code=out_dtype_code,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=8,
            num_stages=3
        )

        return out
