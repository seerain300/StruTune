import torch
import triton
import triton.language as tl


@triton.jit
def _row_rms_kernel(
    x_ptr,          # *pointer to hidden_states (contiguous)
    inv_rms_ptr,    # *pointer to output per-row inverse RMS
    B,              # batch size (rows)
    H,              # hidden size (columns)
    EPS,            # epsilon (float32)
    stride_x_row,   # stride for row in x
    stride_x_col,   # stride for col in x (should be 1 for contiguous)
    OUT_DTYPE: tl.constexpr,  # not used here, but kept for signature symmetry
    BLOCK_SIZE: tl.constexpr,
    VEC_ROWS: tl.constexpr,
):
    # Each program handles a block of rows; grid = (ceil_div(B, VEC_ROWS),)
    pid = tl.program_id(0)
    row_start = pid * VEC_ROWS
    rows = row_start + tl.arange(0, VEC_ROWS)  # [VEC_ROWS]
    # Initialize sum of squares per row
    sumsq = tl.zeros([VEC_ROWS], dtype=tl.float32)

    # Loop over columns in chunks of BLOCK_SIZE
    for col in range(0, H, BLOCK_SIZE):
        cols = col + tl.arange(0, BLOCK_SIZE)  # [BLOCK_SIZE]
        # 2D offsets: [VEC_ROWS, BLOCK_SIZE]
        offs = rows[:, None] * stride_x_row + cols[None, :] * stride_x_col
        mask = (rows[:, None] < B) & (cols[None, :] < H)
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x = x.to(tl.float32)
        sumsq += tl.sum(x * x, axis=1)

    mean = sumsq / H
    inv_rms = tl.rsqrt(mean + EPS)
    # Store per-row inv_rms
    tl.store(inv_rms_ptr + rows, inv_rms, mask=(rows < B))


@triton.jit
def _scale_row_kernel(
    x_ptr,          # *pointer to hidden_states (contiguous)
    weight_ptr,     # *pointer to weight (contiguous)
    inv_rms_ptr,    # *pointer to per-row inverse RMS
    out_ptr,        # *pointer to output (contiguous)
    B,              # batch size (rows)
    H,              # hidden size (columns)
    stride_x_row,   # stride for row in x
    stride_x_col,   # stride for col in x
    stride_out_row, # stride for row in out
    stride_out_col, # stride for col in out
    OUT_DTYPE: tl.constexpr,  # output dtype (tl.float16 or tl.bfloat16)
    BLOCK_SIZE: tl.constexpr,
    VEC_ROWS: tl.constexpr,
):
    # Each program handles a block of rows; grid = (ceil_div(B, VEC_ROWS),)
    pid = tl.program_id(0)
    row_start = pid * VEC_ROWS
    rows = row_start + tl.arange(0, VEC_ROWS)  # [VEC_ROWS]
    inv_rms_rows = tl.load(inv_rms_ptr + rows, mask=(rows < B), other=0.0).to(tl.float32)

    for col in range(0, H, BLOCK_SIZE):
        cols = col + tl.arange(0, BLOCK_SIZE)  # [BLOCK_SIZE]
        x_offs = rows[:, None] * stride_x_row + cols[None, :] * stride_x_col
        w_offs = cols
        mask = (rows[:, None] < B) & (cols[None, :] < H)
        x = tl.load(x_ptr + x_offs, mask=mask, other=0.0)
        w = tl.load(weight_ptr + w_offs, mask=(cols < H), other=0.0)
        x = x.to(tl.float32)
        y = x * inv_rms_rows[:, None] * w[None, :]
        # Cast to desired output dtype
        if OUT_DTYPE == tl.float16:
            y = y.to(tl.float16)
        else:
            # default keep as bfloat16
            y = y.to(tl.bfloat16)
        out_offs = rows[:, None] * stride_out_row + cols[None, :] * stride_out_col
        tl.store(out_ptr + out_offs, y, mask=mask)

        )
