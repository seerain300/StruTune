3) Strategy for robust correctness and performance
- Implement Triton kernels that are simple and compile reliably: avoid 2D broadcasted loads and inner loops over head_dim in Triton. Instead:
  - Host side constructs per-batch segments k_seg and v_seg as contiguous 2D tensors [num_kv_tokens, head_dim].
  - Triton kernels operate on 1D vectors: compute lse by looping over i in the segment (chunked), and compute output vector by looping over i similarly, using 1D loads.
- This keeps heavy computation inside Triton (no torch reductions in forward), and avoids complex Triton constructs that caused prior issues.

4) Triton kernels design
- Kernel A: compute_lse_kernel
  - Inputs: q_vec_ptr *fp32 [head_dim], k_seg_ptr *fp32 [num_kv_tokens, head_dim], lse_ptr *fp32 scalar, num_kv_tokens, sm_scale, head_dim, CHUNK.

- Kernel B: compute_output_kernel
  - Inputs: q_vec_ptr *fp32 [head_dim], k_seg_ptr *fp32 [num_kv_tokens, head_dim], v_seg_ptr *fp32 [num_kv_tokens, head_dim], lse_val fp32 scalar, out_ptr *fp32 [head_dim], num_kv_tokens, sm_scale, head_dim, CHUNK.
  - It recomputes logits_scaled for each i, computes attn = exp(logits_scaled - lse_val), and accumulates out += attn * v_seg[i, :]. Writes out vector to out_ptr.
- Forward:
  - Ensure tensors are on CUDA and contiguous; squeeze dim=1 on k_cache and v_cache.
  - For each batch segment b, compute qo and kv ranges, num_q_tokens and num_kv_tokens.
  - For each q_idx and each head h, compute max_kv_idx under the causal window, gather q vector, k_rows, v_rows (kv_head = h // 4), launch compute_lse_kernel to get lse_val, then launch compute_output_kernel to produce out_vec (fp32). Cast out_vec to bfloat16 and store into output, store lse_val into lse.

5) Dtypes and numerical stability
- All Triton computations use float32. We store output as bfloat16 to match original and lse as float32 (logsumexp in fp32 is stable). sm_scale = 1.0 / sqrt(head_dim) computed in host.

# Triton kernel: compute lse = logsumexp(logits_scaled) / ln(2) for a given q_vec and k_seg.
# logits_scaled[i] = (q_vec · k_seg[i, :]) * sm_scale, i in [0, max_kv_idx).
@triton.jit
def compute_lse_kernel(
    q_vec_ptr,        # *fp32, [head_dim]
    k_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    lse_ptr,          # *fp32, scalar output
    num_kv_tokens: tl.int32,
    sm_scale: tl.float32,
    head_dim: tl.constexpr,
    CHUNK: tl.constexpr
):
    # We only need the first q_idx + 1 + (num_kv_tokens - num_q_tokens) rows due to causal window,
    # but the original code computes across all num_kv_tokens for all q. For simplicity and correctness,
    # we compute across all num_kv_tokens here. If max_kv_idx < num_kv_tokens, we mask via index check.
    m = -float("inf")
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask_i = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim))
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            # valid_i = idx_i < num_kv_tokens  # already masked by mask_i
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim))
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits[i] = dot * sm_scale
        m = tl.maximum(m, tl.max(tl.where(mask_i, logits, -float("inf"))))
    l = 0.0
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask_i = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim))
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim))
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits[i] = dot * sm_scale
        l += tl.sum(tl.exp(tl.where(mask_i, logits - m, -float("inf"))))
    ln2 = 1.0 / math.log(2.0)
    lse_val = tl.log(l) / ln2
    tl.store(lse_ptr, lse_val)


# Triton kernel: compute output vector for a given q_vec, k_seg, v_seg, and lse_val.
# Output[i] = sum_j (attn[i] * v_seg[i, j]) where attn[i] = exp((q·k[i]) * sm_scale - lse).
@triton.jit
def compute_output_kernel(
    q_vec_ptr,        # *fp32, [head_dim]
    k_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    v_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    lse_val,          # float32 scalar
    out_ptr,          # *fp32, [head_dim]
    num_kv_tokens: tl.int32,
    sm_scale: tl.float32,
    head_dim: tl.constexpr,
    CHUNK: tl.constexpr
):
    # Initialize output
    for j in range(0, head_dim):
        tl.store(out_ptr + j, 0.0)
    # Compute output via accumulation per element j
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask_i = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim))
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim))
            dot = 0.0
            for j_local in range(0, head_dim):
                dot += q_vec[j_local] * k_row[j_local]
            logits[i] = dot * sm_scale
        attn = tl.exp(logits - lse_val)
        for i in range(0, CHUNK):
            idx_i = start + i
            mask_i_valid = idx_i < num_kv_tokens
            v_row_ptr = v_seg_ptr + idx_i * head_dim
            v_row = tl.load(v_row_ptr + tl.arange(0, head_dim), mask=mask_i_valid, other=0.0)
            for j_local in range(0, head_dim):
                tl.store(out_ptr + j_local, tl.load(out_ptr + j_local) + attn[i] * v_row[j_local])

class ModelNew(torch.nn.Module):
    def __init__(self):

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Ensure tensors are on CUDA and contiguous
        device = q.device
        q = q.to(device).contiguous()
        k_cache = k_cache.to(device).contiguous()
        v_cache = v_cache.to(device).contiguous()

        # Squeeze dim=1 (since original k_cache/v_cache have shape [num_pages, 1, num_kv_heads, head_dim])
        k_cache_flat = k_cache.squeeze(1)  # [num_pages, num_kv_heads, head_dim]
        v_cache_flat = v_cache.squeeze(1)  # [num_pages, num_kv_heads, head_dim]

        # Dimensions
        total_q, num_qo_heads, head_dim = q.shape
        num_kv_heads = k_cache.shape[2]
        assert num_qo_heads == 32, "num_qo_heads must be 32"
        assert num_kv_heads == 8, "num_kv_heads must be 8"
        assert head_dim == 128, "head_dim must be 128"
        assert total_q == int(qo_indptr[-1].item()), "Sum of qo_indptr must equal total_q."

        # Output and lse buffers
        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.bfloat16, device=device)
        lse = torch.full((total_q, num_qo_heads), -float("inf"), dtype=torch.float32, device=device)

        # GQA ratio
        gqa_ratio = num_qo_heads // num_kv_heads  # 4

        # Process each batch segment
        for b in range(0, qo_indptr.numel() - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            num_q_tokens = q_end - q_start
            num_kv_tokens = kv_end - kv_start

            if num_q_tokens <= 0 or num_kv_tokens <= 0:
                continue

            # Ensure kv_indices for this batch are on device
            kv_indices_b = kv_indices[kv_start:kv_end].to(device)  # [num_kv_tokens]

            for q_idx in range(num_q_tokens):
                global_q_idx = q_start + q_idx
                # Causal window: how many KV tokens this query can "see"
                delta = num_kv_tokens - num_q_tokens
                max_kv_idx = min(q_idx + 1 + delta, num_kv_tokens)
                if max_kv_idx <= 0:
                    continue

                # Loop over heads
                for h in range(num_qo_heads):
                    kv_head = h // gqa_ratio  # 0..7

                    # q vector for this head
                    q_vec = q[global_q_idx, h, :].to(torch.float32).contiguous()  # [head_dim]

                    # Gather k and v rows for this batch segment
                    k_rows = k_cache_flat[kv_indices_b[:max_kv_idx], kv_head, :].to(torch.float32).contiguous()  # [max_kv_idx, head_dim]
                    v_rows = v_cache_flat[kv_indices_b[:max_kv_idx], kv_head, :].to(torch.float32).contiguous()  # [max_kv_idx, head_dim]

                    # Compute lse for this head
                    lse_buf = torch.empty(1, dtype=torch.float32, device=device)
                    CHUNK = 128  # specialize for head_dim=128; mask handles smaller
                    compute_lse_kernel[(1,)](
                        q_vec,                 # *fp32 [head_dim]
                        k_rows,                # *fp32 [max_kv_idx, head_dim]
                        lse_buf,               # *fp32 scalar
                        num_kv_tokens=max_kv_idx,
                        sm_scale=float(sm_scale),
                        head_dim=head_dim,
                        CHUNK=CHUNK
                    )
                    lse_val = lse_buf[0]

                    # Compute output vector for this head
                    out_vec = torch.empty(head_dim, dtype=torch.float32, device=device)
                    compute_output_kernel[(1,)](
                        q_vec,                 # *fp32 [head_dim]
                        k_rows,                # *fp32 [max_kv_idx, head_dim]
                        v_rows,                # *fp32 [max_kv_idx, head_dim]
                        lse_val,               # float32
                        out_vec,               # *fp32 [head_dim]
                        num_kv_tokens=max_kv_idx,
                        sm_scale=float(sm_scale),
                        head_dim=head_dim,
                        CHUNK=CHUNK
                    )

                    # Store results
                    output[global_q_idx, h, :] = out_vec.to(torch.bfloat16)
                    lse[global_q_idx, h] = lse_val

        return output, lse

):
    # Compute max over logits_scaled for numerical stability
    m = -float("inf")
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask_i = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim))
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim))
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits[i] = dot * sm_scale
        m = tl.maximum(m, tl.max(tl.where(mask_i, logits, -float("inf"))))
    # Compute sum of exp(logits - m)
    l = 0.0
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask_i = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim))
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim))
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits[i] = dot * sm_scale
        l += tl.sum(tl.exp(tl.where(mask_i, logits - m, -float("inf"))))
    ln2 = 1.0 / math.log(2.0)
    lse_val = tl.log(l) / ln2
    tl.store(lse_ptr, lse_val)


# Triton kernel: compute output vector for a given q_vec, k_seg, v_seg, and lse_val.
# Output[i] = sum_j (attn[i] * v_seg[i, j]) where attn[i] = exp((q·k[i]) * sm_scale - lse).
@triton.jit
def compute_output_kernel(
    q_vec_ptr,        # *fp32, [head_dim]
    k_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    v_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    lse_val,          # float32 scalar
    out_ptr,          # *fp32, [head_dim]
    num_kv_tokens: tl.int32,
    sm_scale: tl.float32,
    head_dim: tl.constexpr,
    CHUNK: tl.constexpr
):
    # Initialize output vector
    for j in range(0, head_dim):
        tl.store(out_ptr + j, 0.0)
    # Accumulate output via per-element j
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask_i = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim))
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim))
            dot = 0.0
            for j_local in range(0, head_dim):
                dot += q_vec[j_local] * k_row[j_local]
            logits[i] = dot * sm_scale
        attn = tl.exp(logits - lse_val)
        for i in range(0, CHUNK):
            idx_i = start + i
            mask_i_valid = idx_i < num_kv_tokens
            v_row_ptr = v_seg_ptr + idx_i * head_dim
            v_row = tl.load(v_row_ptr + tl.arange(0, head_dim), mask=mask_i_valid, other=0.0)
            for j_local in range(0, head_dim):
                tl.store(out_ptr + j_local, tl.load(out_ptr + j_local) + attn[i] * v_row[j_local])
