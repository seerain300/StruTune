    v_ptr,             # *f32, [num_pages, 8, 128]

    MAX_Q: tl.constexpr,          # upper bound for q tokens per segment
    MAX_KV_TOKENS: tl.constexpr,  # upper bound for kv tokens per segment
    GQA_RATIO: tl.constexpr       # 32 // 8 = 4
):

    b = tl.program_id(0)

    # Load segment bounds
    q_start = tl.load(qo_indptr_ptr + b).to(tl.int32)
    q_end = tl.load(qo_indptr_ptr + b + 1).to(tl.int32)
    kv_start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    kv_end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)

    # Lengths (dynamic)
    num_q_tokens = q_end - q_start  # int32
    num_kv_tokens = kv_end - kv_start  # int32

    # Loop over queries in this segment up to MAX_Q
    for q_idx in range(0, MAX_Q):
        valid_q = q_idx < num_q_tokens
        global_q_idx = q_start + q_idx

        # If no valid query, skip
        if not valid_q:
            continue

        # q layout: [num_q, 32, 128] flattened by strides: q_offset = global_q_idx * (32*128) + h * 128
        q_pos = tl.zeros((128,), dtype=tl.float32)

        # We will load q_pos[h] vector for each h, but in Triton we can index q_ptr as 1D and compute offset.
        # Simpler: precompute base offset for q[global_q_idx, :, :] and read all 32 heads into a [32, 128] tensor.
        # However, we only need q_pos[h] for current h. So for each h, compute offset and load 128 dims.
        # We will load q_pos[h] into a vector of size 128 and use it for this q_idx.
        # We can do this by looping h and building q_pos[h]. But since we need it per h, better approach:
        # We can't have a vector q_pos, so we will recompute q_pos[h] each time for that h. This is fine.

        # We'll compute q_pos[h] by loading q_ptr at offset: global_q_idx * (num_qo_heads*head_dim) + h * head_dim
        # q_ptr is [num_q, 32, 128] flattened, so:
        q_offset_base = global_q_idx * (num_qo_heads * head_dim)
        q_pos = tl.zeros((head_dim,), dtype=tl.float32)

        for h in range(0, 32):

            # Linear index = q_offset_base + h * head_dim
            # We need to gather 128 elements starting at that offset. Triton allows vectorized loads via stride.
            # Compute offsets: offs = h * head_dim + arange(0, head_dim)

            q_vec = tl.load(q_ptr + q_offset_base + h * head_dim + offs)

            # We need q_pos vector of size head_dim for this h. Since q_vec is the whole head, we can use it.
            # But we only need q_pos[h] element vector, which is q_vec. We will compute logits per head h.
            # The above loop loads q_pos[h] for each h; we won't store q_pos, but we will load it per head below.

        # For each query head h
        for h in range(0, 32):
            # GQA mapping: kv_head = h // GQA_RATIO
            kv_head = h // GQA_RATIO  # 4

            k_list = tl.zeros((MAX_KV_TOKENS, head_dim), dtype=tl.float32)
            v_list = tl.zeros((MAX_KV_TOKENS, head_dim), dtype=tl.float32)

            # kv_indices_ptr points to segment-local indices: idx = kv_start + kk
            # For each kk in 0..MAX_KV_TOKENS, if kk < num_kv_tokens:
            # page_id = kv_indices[kv_start + kk]
            # Then k_ptr[row = page_id, head = kv_head, :] and v_ptr[row, head, :]
            # We need to gather rows for each kk. Triton allows gather via pointers; we'll construct addresses:
            # k_ptr address: (page_id * (num_kv_heads * head_dim)) + kv_head * head_dim + arange(0, head_dim)
            # Similarly for v_ptr.
            for kk in range(0, MAX_KV_TOKENS):
                valid_k = kk < num_kv_tokens
                idx = kv_start + kk
                # If idx out of range, skip
                # Load kv_index at idx
                page_id = tl.load(kv_indices_ptr + idx)

                base_k = page_id * (num_kv_heads * head_dim)
                base_v = page_id * (num_kv_heads * head_dim)
                k_row = tl.load(k_ptr + base_k + kv_head * head_dim + tl.arange(0, head_dim), mask=valid_k, other=0.0)
                v_row = tl.load(v_ptr + base_v + kv_head * head_dim + tl.arange(0, head_dim), mask=valid_k, other=0.0)
                # Place into k_list and v_list at position kk
                # Triton allows assignment to [kk, :] of 2D tensors; we construct via slicing:
                k_list[kk, :] = k_row
                v_list[kk, :] = v_row

            # Load q_pos[h]: vector of size head_dim
            offs_qh = h * head_dim + tl.arange(0, head_dim)
            q_vec = tl.load(q_ptr + q_offset_base + h * head_dim + offs_qh)  # shape [128]

            # Compute logits for each kk: q_vec · k_list[kk, :]
            # Initialize logits vector
            logits = tl.zeros((MAX_KV_TOKENS,), dtype=tl.float32)
            # For i in 0..MAX_KV_TOKENS-1: logits[i] = dot(q_vec, k_list[i, :])
            for i in range(0, MAX_KV_TOKENS):
                k_row = k_list[i, :]
                logits[i] = tl.sum(q_vec * k_row, axis=0)

            # Scale by sm_scale
            logits_scaled = logits * sm_scale

            # Apply causal mask: max_kv_idx = min(q_idx + 1 + (num_kv_tokens - num_q_tokens), num_kv_tokens)
            # For segment-local bounds, num_q_tokens_segment = num_q_tokens (since we compute within this segment).
            # But q_idx is local to this program’s segment loop, so we need global num_q_tokens_segment.
            # We can compute it by loading q_end - q_start which we already have as num_q_tokens.
            # Causal bound: max_kv_idx = min(q_idx + 1 + (num_kv_tokens - num_q_tokens), num_kv_tokens)
            # Note: q_idx is local loop index, not global. We need to know how many queries in this segment are processed.
            # num_q_tokens_segment is not available here. However, the original loop uses num_q_tokens = q_end - q_start,
            # and the causal bound uses num_q_tokens_segment == num_q_tokens. So we can use num_q_tokens.
            # Compute bound
            # delta = num_kv_tokens - num_q_tokens
            # bound = q_idx + 1 + delta, capped by num_kv_tokens
            delta = num_kv_tokens - num_q_tokens
            bound = q_idx + 1 + delta
            bound = tl.minimum(bound, num_kv_tokens)
            # Mask logits: set logits[kk] to -inf if kk >= bound
            for kk in range(0, MAX_KV_TOKENS):
                if kk >= bound:
                    logits_scaled[kk] = -float("inf")

            # Compute logsumexp in Triton: lse = log(sum(exp(logits_scaled))) / ln(2)
            # First, max for numerical stability
            max_val = tl.max(logits_scaled, axis=0)
            # exp(logits_scaled - max_val)
            exp_vals = tl.exp(logits_scaled - max_val)
            sum_exp = tl.sum(exp_vals, axis=0)
            lse_val = tl.log(sum_exp) / 0.6931471805599453  # 1 / ln(2)
            # Store lse to lse_ptr[global_q_idx, h]
            lse_offset = global_q_idx * 32 + h
            tl.store(lse_ptr + lse_offset, lse_val)

            # Compute softmax
            softmax_vals = exp_vals / sum_exp  # already masked by -inf for kk >= bound, so sum_exp excludes those

            # Compute output vector: out_vec = sum(softmax_vals * v_list[kk, :])
            out_vec = tl.zeros((head_dim,), dtype=tl.float32)
            for i in range(0, MAX_KV_TOKENS):
                v_row = v_list[i, :]
                out_vec += softmax_vals[i] * v_row

            # Store output vector to out_ptr[global_q_idx, h, :]
            out_offset_base = global_q_idx * (32 * head_dim) + h * head_dim
            # Store 128 elements
            for d in range(0, head_dim):
                tl.store(out_ptr + out_offset_base + d, out_vec[d])
