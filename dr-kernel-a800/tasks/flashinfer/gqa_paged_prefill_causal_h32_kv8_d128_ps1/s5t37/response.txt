    GQA_RATIO: tl.constexpr,   # 4 (32 // 8)

    qo_end = tl.load(qo_indptr_ptr + b + 1)

    kv_head = h // GQA_RATIO

    s = 0.0

        # Compute validity: i < candidate_max and i < num_kv_indices_in_b and kv_start + i < kv_end
        # However, Triton cannot index kv_indptr_ptr or qo_end directly inside the kernel without per-b values.
        # To handle this, we rely on the grid: program_id(1)=q_idx is in-bounds, and we pass per-b indptrs via loads.
        # We need num_kv_indices_in_b and candidate_max. Triton cannot access these directly, so we mask using i < qo_end - qo_start and i < kv_end - kv_start, but those are per-b scalars.
        # Therefore, we will use a scalar mask per i by recomputing using qo_end and kv_end via loads from indptr:

        kv_end_val = tl.load(kv_indptr_ptr + b + 1)

        num_kv_indices = kv_end_val - kv_start
        candidate_max = q_idx + 1 + (num_kv_indices - num_q_tokens)

        idx_i = kv_start + i
        # Load k_sub for kv_head: we load a 2D tile of shape [num_kv_heads, HEAD_DIM] using idx_i

        k_vec = tl.load(k_row_ptr + tl.arange(0, HEAD_DIM), mask=cond_i, other=0.0)

        # q_vec: [128], k_vec: [128]; dot = sum(q * k)

        for j in tl.static_range(0, HEAD_DIM):
            dot += q_vec[j] * k_vec[j]
        logits_scaled = dot * sm_scale

        # Update running max and sum for logsumexp
        m_new = tl.maximum(m, logits_scaled)
        s = s * tl.exp(m - m_new) + tl.exp(m_new - m) * tl.where(cond_i, tl.exp(logits_scaled - m_new), 0.0)

    lse_val = tl.log(s) + m  # natural logsumexp

        num_kv_indices = kv_end_val - kv_start
        candidate_max = q_idx + 1 + (num_kv_indices - num_q_tokens)

        idx_i = kv_start + i

        v_vec = tl.load(v_row_ptr + tl.arange(0, HEAD_DIM), mask=cond_i, other=0.0)

        out_vec += prob * v_vec
