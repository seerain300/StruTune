import torch
import math
import triton
import triton.language as tl


@triton.jit
def attn_kernel(
    q_ptr,              # *f32, [total_q, 32, 128]
    k_ptr,              # *f32, [num_pages, 8, 128] flattened view for gather
    v_ptr,              # *f32, [num_pages, 8, 128] flattened view for gather
    output_ptr,         # *f32, [total_q, 32, 128]
    output_lse_ptr,     # *f32, [total_q, 32]
    q_start,            # i32
    q_end,              # i32
    kv_start,           # i32
    kv_end,             # i32
    head_dim,           # i32, 128
    num_qo_heads,       # i32, 32
    num_kv_heads,       # i32, 8
    gqa_ratio,          # i32, 4
    sm_scale,           # f32 scalar
    ln2_inv,            # f32 scalar
    MAX_Q_SEG: tl.constexpr,
    MAX_KV_SEG: tl.constexpr,
):
    b = tl.program_id(0)

    # Segment bounds
    num_q_tokens_segment = q_end - q_start
    num_kv_tokens = kv_end - kv_start

    # Vector for head dim
    dim = tl.arange(0, head_dim)

    # Iterate over queries in this segment
    for q_i in range(0, MAX_Q_SEG):
        # Mask for valid queries
        q_valid = q_i < num_q_tokens_segment
        global_q_idx = q_start + q_i

        if not q_valid:
            continue

        # For each query head
        for h in range(0, num_qo_heads):
            # Map to KV head
            kv_head = h // gqa_ratio

            # Load q[h] vector
            q_vec = tl.load(q_ptr + global_q_idx * (num_qo_heads * head_dim) + h * head_dim + dim)

            # Compute logsumexp of logits_scaled over k list
            max_val = -float("inf")
            sum_exp = 0.0

            # First pass: accumulate logsumexp
            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = tl.load(kv_indices_ptr + (kv_start + kk))
                k_vec = tl.load(k_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + dim)
                prod = tl.sum(q_vec * k_vec, axis=0)  # scalar
                scaled = prod * sm_scale
                # numerically stable update
                if scaled > max_val:
                    sum_exp = sum_exp * tl.exp(max_val - scaled) + 1.0
                    max_val = scaled
                else:
                    sum_exp = sum_exp + tl.exp(scaled - max_val)

            # Compute lse = max + log(sum_exp), then scaled by 1/ln(2)
            lse_val = (max_val + tl.log(sum_exp)) * ln2_inv

            # Softmax denominator over the same k list
            sum_attn = 0.0

            # Second pass: compute softmax and output
            out_vec = 0.0
            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = tl.load(kv_indices_ptr + (kv_start + kk))
                k_vec = tl.load(k_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + dim)
                prod = tl.sum(q_vec * k_vec, axis=0)
                scaled = prod * sm_scale
                # causal mask: only consider kk < max_kv_idx
                max_kv_idx = tl.minimum(q_i + 1 + (num_kv_tokens - num_q_tokens_segment), num_kv_tokens)
                valid_k = kk < max_kv_idx
                attn = tl.exp(scaled - lse_val) * valid_k  # valid_k handled via zero attn when not valid
                v_vec = tl.load(v_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + dim)
                out_vec += attn * tl.sum(v_vec * k_vec, axis=0)  # this line is redundant, we want attn * v_vec

                # Accumulate sum of attn for normalization (softmax over valid keys)
                # If invalid, attn was zero due to valid_k, so sum_attn += 0.
                # We still need sum over valid keys; compute it separately.
                # For valid keys, attn = exp(scaled - lse_val); otherwise 0.
                # We can compute sum_attn in a third loop over valid keys. However, softmax requires the sum of
                # all valid attn. To keep one pass, we note that invalid keys contribute 0, so we can compute
                # sum_attn during valid key iterations using a conditional. Triton doesn't support dynamic if
                # for skipping accumulation; to ensure correctness, we do a separate small loop to sum valid attn.

            # Compute sum_attn over valid keys only: recompute sum of exp(scaled - lse_val) for kk < max_kv_idx
            sum_attn = 0.0
            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                max_kv_idx = tl.minimum(q_i + 1 + (num_kv_tokens - num_q_tokens_segment), num_kv_tokens)
                valid_k = kk < max_kv_idx
                k_idx = tl.load(kv_indices_ptr + (kv_start + kk))
                k_vec = tl.load(k_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + dim)
                prod = tl.sum(q_vec * k_vec, axis=0)
                scaled = prod * sm_scale
                attn_val = tl.exp(scaled - lse_val)
                sum_attn += attn_val * valid_k  # valid_k is 1 only when kk < max_kv_idx, else 0

            # Normalize out_vec by sum_attn (handle sum_attn==0 edge case: softmax zeros anyway)
            # Since we only added valid attn, sum_attn > 0 unless no valid keys; but with causal mask, there's always at least one if q_i + 1 + (num_kv_tokens - num_q_tokens_segment) >= 0.
            # In rare cases where max_kv_idx <= 0, all attn zeros and out_vec=0; we can skip division.
            out_vec = out_vec / sum_attn

            # Store output and lse
            tl.store(output_ptr + global_q_idx * (num_qo_heads * head_dim) + h * head_dim + dim, out_vec)
            tl.store(output_lse_ptr + global_q_idx * num_qo_heads + h, lse_val)

    def __init__(self):
        super().__init__()

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # q: [total_q, 32, 128], bfloat16 or float32
        # k_cache: [num_pages, 1, 8, 128], bfloat16
        # v_cache: [num_pages, 1, 8, 128], bfloat128
        # qo_indptr: [len_indptr], int32
        # kv_indptr: [len_indptr], int32
        # kv_indices: [num_kv_indices], int32
        # sm_scale: float32

        # Allocate outputs in fp32 (compute in fp32; final cast not done here per requirement)
        total_q, num_qo_heads, head_dim = q.shape
        assert num_qo_heads == 32 and head_dim == 128
        num_pages, _, num_kv_heads, _ = k_cache.shape
        assert num_kv_heads == 8 and head_dim == 128

        device = q.device

        # Flatten k_cache and v_cache for efficient gather (squeeze dim-1 to remove size 1)
        k_cache_flat = k_cache.squeeze(1).contiguous()  # [num_pages, 8, 128]
        v_cache_flat = v_cache.squeeze(1).contiguous()  # [num_pages, 8, 128]

        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Prepare inputs: pass scalar segment bounds to kernel
        len_indptr = qo_indptr.numel()
        grid = (len_indptr - 1,)

        # Convert tensors to fp32 for compute
        q_f32 = q.to(torch.float32)  # [total_q, 32, 128]
        k_ptr_f32 = k_cache_flat.to(torch.float32).view(-1)  # flatten into 1D for Triton loads (we'll recompute offsets in kernel)
        v_ptr_f32 = v_cache_flat.to(torch.float32).view(-1)

        # ln(2) inverse for LSE
        ln2_inv = 1.0 / math.log(2.0)

        # Launch Triton kernel
        attn_kernel[grid](
            q_f32, k_ptr_f32, v_ptr_f32, output, lse,
            qo_indptr[0].item(), qo_indptr[-1].item(), kv_indptr[0].item(), kv_indptr[-1].item(),
            head_dim, num_qo_heads, num_kv_heads, 4,  # gqa_ratio
            sm_scale, ln2_inv,
            MAX_Q_SEG=100,  # upper bound; masks handle actual num_q_tokens_segment
            MAX_KV_SEG=100, # upper bound; masks handle actual num_kv_tokens
            num_warps=4,
        )

        return output, lse
