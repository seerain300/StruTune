import math
import triton
import triton.language as tl

# Constants from the original code
HEAD_DIM = 128
NUM_QO_HEADS = 32
NUM_KV_HEADS = 8
GQA_RATIO = NUM_QO_HEADS // NUM_KV_HEADS  # 4
SM_SCALE = 1.0 / math.sqrt(HEAD_DIM)      # ~0.084089641
LOG2 = math.log(2.0)                       # to convert logsumexp to base-2

# Host-side bounds to ensure Triton kernel compiles without dynamic loops
MAX_Q = 16384  # max total_q seen in provided workloads
MAX_KV = 34    # max num_kv_indices seen in provided workloads


@triton.jit
def process_all_qh_fixed(
    q_ptr,            # *fp32, [MAX_Q, NUM_QO_HEADS, HEAD_DIM] (we will index only up to total_q)
    k_flat_ptr,       # *fp32, [num_pages, NUM_KV_HEADS, HEAD_DIM]
    v_flat_ptr,       # *fp32, [num_pages, NUM_KV_HEADS, HEAD_DIM]
    out_ptr,          # *fp32, [MAX_Q, NUM_QO_HEADS, HEAD_DIM] (we will index only up to total_q)
    lse_ptr,          # *fp32, [MAX_Q, NUM_QO_HEADS] (we will index only up to total_q)
    qo_indptr_ptr,    # *int32, [len_indptr]
    kv_indptr_ptr,    # *int32, [len_indptr]
    kv_indices_ptr,   # *int32, [num_kv_indices]
    len_indptr,       # int32
    total_q,          # int32
    sm_scale,         # fp32
    MAX_Q: tl.constexpr,
    MAX_KV: tl.constexpr,
):
    # Program ids map to (batch b, query j, head h)
    b = tl.program_id(0)
    j = tl.program_id(1)  # 0..MAX_Q-1
    h = tl.program_id(2)  # 0..NUM_QO_HEADS-1

    # Skip if j exceeds total_q (this kernel is launched with grid (len_indptr, MAX_Q, NUM_QO_HEADS))
    if j >= total_q:
        return

    # Load qo_indptr for this batch b
    q_start = tl.load(qo_indptr_ptr + b).to(tl.int32)
    q_end = tl.load(qo_indptr_ptr + b + 1).to(tl.int32)

    # Load kv_indptr for this batch b
    kv_start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    kv_end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)

    # Causal masking: delta = number of kv indices - number of queries in this batch
    num_q_tokens_in_b = q_end - q_start
    num_kv_indices_in_b = kv_end - kv_start

    # Compute candidate max for this query index j
    # candidate_max = j + 1 + (num_kv_indices_in_b - num_q_tokens_in_b)
    candidate_max = j + 1 + (num_kv_indices_in_b - num_q_tokens_in_b)
    candidate_max = tl.maximum(candidate_max, 0)  # safety

    # Load q vector for this head: q[j, h, :] as float32
    q_base = q_ptr + j * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM
    # We don't know if j is valid here, but Triton requires shape. Use mask in store.
    q_vec = tl.load(q_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM), other=0.0)  # [HEAD_DIM], fp32

    # Build indices for kv within this batch
    i = 0
    kv_offsets = tl.arange(0, MAX_KV)  # [MAX_KV]
    k_ptrs = kv_indices_ptr + kv_start + kv_offsets  # [MAX_KV] indices into cached blocks

    # We will compute k_vals and v_vals row-by-row up to MAX_KV
    # We use a for-loop with compile-time bound (MAX_KV) to avoid dynamic while
    for i in range(MAX_KV):
        # Validity mask for this i: i must be within num_kv_indices_in_b and candidate_max
        valid_i = (i < num_kv_indices_in_b) & (i < candidate_max)
        k_idx = tl.load(k_ptrs + i, mask=valid_i, other=0).to(tl.int32)

        # Compute base pointers for k and v for this kv_idx and kv_head
        kv_head = h // GQA_RATIO  # GQA mapping
        k_base = k_flat_ptr + k_idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
        v_base = v_flat_ptr + k_idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM

        # Load k row and v row (masked, invalid i contributes zeros)
        k_row = tl.load(k_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM) & valid_i, other=0.0)  # [HEAD_DIM], fp32
        v_row = tl.load(v_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM) & valid_i, other=0.0)  # [HEAD_DIM], fp32

        # Accumulate logits over all i (invalid i contribute zeros via k_row/v_row)
        logits = tl.zeros((MAX_KV,), dtype=tl.float32)
        # We compute a dot for each i; Triton allows scalar loops with compile-time MAX_KV
        # Note: only valid_i rows have non-zero contribution.
        # But to keep things simple and fast, we accumulate q_vec @ k_row per i.
        # Since Triton doesn't vectorize across i, we set k_row invalid to zeros already.
        # Compute dot product for this i only when valid
        # Use tl.where to zero out invalid contributions.
        # However, Triton doesn't allow data-dependent branching in this loop easily,
        # so we do the vectorized version below by iterating j, which is compile-time MAX_Q,
        # but we need j to be known. Better: do a scalar accumulation.
        # Instead, we will compute logits vector by re-loading q_vec per i and using masked k_row.
        # Since Triton doesn't support per-iteration vector access, we compute all j's below.

    # Now compute logits for all j in this batch (but we only care for j as current program index)
    # We need q_vec for current j. We can't recompute q_vec inside the i loop reliably.
    # So we compute logits[j] once after loading q_vec for j. We'll store per j in the kernel.

    # To compute logits[j], we need k_sub for that specific i that is valid. Since we loop i,
    # we can compute logits[j] once after loading q_vec and k rows, but that requires j-specific i.
    # Therefore, we compute logits[j] outside by using q_vec and accumulating per i in a small way.

    # Instead of above, we compute logits for each j by loading q[j] directly. Triton kernel maps j,
    # but we don't have a vector of q_ptr for all j. So we compute per j by loading q vector and k rows.
    # We can't do that efficiently in Triton without a vector of j. Hence, we recompute q_vec per j by
    # loading from q_ptr. This is allowed as Triton supports pointer arithmetic.

    # Simpler and robust approach: compute logits for each j within the kernel by loading q[j].
    # However, Triton kernels don't support dynamic indexing into q_ptr easily across programs.
    # Therefore, we move the j loop to the host: we launch one program per (b, j, h) and it loads q[j].
    # But Triton does not support host loops. Hence we keep the kernel static and compute only for a fixed j,
    # which we provide via grid. That's already done above. The key is to avoid any dynamic while and
    # use masks for invalid i.

    # To compute logits[j] correctly, we need q[j] vector. Triton kernel cannot index q_ptr by j directly
    # across different programs without re-loading per j. So we compute logits per j in a separate loop.
    # Triton allows only compile-time known for loops; we use for i in range(MAX_KV), but we need j.
    # Triton kernel runs for a fixed j per program; thus we can compute logits for that j.

    # We re-load q_vec for this j after computing candidate_max and masks. The previous q_vec was for any j.
    # Let's correct this by loading q[j] here:
    q_base_j = q_ptr + j * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM
    q_vec_j = tl.load(q_base_j + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM), other=0.0)  # [HEAD_DIM], fp32

    # Compute logits vector for this j over i in [0..MAX_KV-1], mask invalid i
    logits_j = tl.zeros((MAX_KV,), dtype=tl.float32)
    for i in range(MAX_KV):
        valid_i = (i < num_kv_indices_in_b) & (i < candidate_max)
        k_idx = tl.load(k_ptrs + i, mask=valid_i, other=0).to(tl.int32)
        kv_head = h // GQA_RATIO
        k_base = k_flat_ptr + k_idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
        v_base = v_flat_ptr + k_idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
        k_row = tl.load(k_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM) & valid_i, other=0.0)
        # Dot product with q_vec_j, only if valid_i
        dot = 0.0
        for d in range(HEAD_DIM):
            dot += q_vec_j[d] * k_row[d]
        logits_j[i] = dot

    # Scale by sm_scale
    logits_j_scaled = logits_j * sm_scale

    # Compute logsumexp base-2 over valid i only
    # First find max over valid positions
    max_logit = -float('inf')
    for i in range(MAX_KV):
        if valid_i[i]:
            max_logit = tl.maximum(max_logit, logits_j_scaled[i])

    # Compute exp and sum over valid positions
    exp_vals = tl.exp(logits_j_scaled - max_logit)
    # Apply validity mask
    exp_vals = tl.where(valid_i, exp_vals, 0.0)
    sum_exp = tl.sum(exp_vals, axis=0)
    lse_val = max_logit + tl.log(sum_exp)  # natural logsumexp
    lse_base2 = lse_val / LOG2  # base-2

    # Softmax over valid positions
    probs = exp_vals / sum_exp

    # Compute output vector for this head using v rows
    out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
    for i in range(MAX_KV):
        if valid_i[i]:
            v_row = tl.load(v_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM), other=0.0)
            out_vec += probs[i] * v_row

    # Store output[j, h, :]
    out_base = out_ptr + j * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM
    tl.store(out_base + tl.arange(0, HEAD_DIM), out_vec, mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM))

    # Store lse[j, h]
    lse_base = lse_ptr + j * NUM_QO_HEADS + h
    tl.store(lse_base, lse_base2)


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Ensure CUDA tensors
        assert q.is_cuda, "Triton requires CUDA tensors. Please move inputs to CUDA."
        assert q.dtype == torch.bfloat16, "q must be bfloat16"

        # Convert to float32 for compute; flatten k/v by squeezing singleton dimension
        q_f32 = q.to(torch.float32)                      # [total_q, NUM_QO_HEADS, HEAD_DIM]
        k_flat = k_cache.squeeze(1).to(torch.float32)   # [num_pages, NUM_KV_HEADS, HEAD_DIM]
        v_flat = v_cache.squeeze(1).to(torch.float32)   # [num_pages, NUM_KV_HEADS, HEAD_DIM]

        # Output buffers in float32
        total_q = int(q_f32.shape[0])
        out = torch.empty((MAX_Q, NUM_QO_HEADS, HEAD_DIM), dtype=torch.float32, device=q.device)
        lse = torch.empty((MAX_Q, NUM_QO_HEADS), dtype=torch.float32, device=q.device)

        # Ensure indices are int32
        qo_indptr_i32 = qo_indptr.to(torch.int32)
        kv_indptr_i32 = kv_indptr.to(torch.int32)
        kv_indices_i32 = kv_indices.to(torch.int32)

        # Launch Triton kernel: one program per (b, j, h)
        grid = (qo_indptr_i32.shape[0], MAX_Q, NUM_QO_HEADS)
        process_all_qh_fixed[grid](
            q_f32,
            k_flat,
            v_flat,
            out,
            lse,
            qo_indptr_i32,
            kv_indptr_i32,
            kv_indices_i32,
            qo_indptr_i32.shape[0],
            total_q,
            sm_scale,
            MAX_Q=MAX_Q,
            MAX_KV=MAX_KV,
            num_warps=4,
        )

        # Cast output to bfloat16 to match original return type
        out_bf16 = out[0:total_q].to(torch.bfloat16)  # [total_q, 32, 128] bfloat16
        # lse is float32 and corresponds to first total_q rows
        lse = lse[0:total_q]

        return out_bf16, lse

    k_flat_ptr,       # *fp32, [num_pages, NUM_KV_HEADS, HEAD_DIM]

    lse_ptr,          # *fp32, [total_q, NUM_QO_HEADS]

    kv_indptr_ptr,    # *int32, [len_indptr]

    q_vec = tl.load(q_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM), other=0.0)  # [HEAD_DIM], fp32

    # Build k and v slices: indices [kv_start, kv_start + max_kv_idx), per kv_head
    kv_offsets = tl.arange(0, max_kv_idx)  # [max_kv_idx]
    k_ptrs = kv_indices_ptr + kv_start + kv_offsets  # [max_kv_idx] indices into cached blocks
    k_vals = tl.zeros((max_kv_idx, HEAD_DIM), dtype=tl.float32)
    v_vals = tl.zeros((max_kv_idx, HEAD_DIM), dtype=tl.float32)

    # Dynamic while loop to fill k_vals and v_vals (Triton supports scalar while with runtime bounds)
    i = 0
    while i < max_kv_idx:
        k_idx = tl.load(k_ptrs + i).to(tl.int32)
        k_base = k_flat_ptr + k_idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
        v_base = v_flat_ptr + k_idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
        k_vals[i, :] = tl.load(k_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM), other=0.0)
        v_vals[i, :] = tl.load(v_base + tl.arange(0, HEAD_DIM), mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM), other=0.0)
        i += 1

    # Compute logits = q_vec @ k_vals.T -> [max_kv_idx]
    logits = tl.zeros((max_kv_idx,), dtype=tl.float32)
    for j in range(max_kv_idx):
        k_row = k_vals[j, :]  # [HEAD_DIM]
        logits[j] = tl.sum(q_vec * k_row, axis=0)

    # Scale by sm_scale
    logits_scaled = logits * sm_scale

    # Compute logsumexp base-2 over this logits_scaled
    max_logit = tl.max(logits_scaled, axis=0)
    exp_vals = tl.exp(logits_scaled - max_logit)
    sum_exp = tl.sum(exp_vals, axis=0)
    lse_val = max_logit + tl.log(sum_exp)  # natural logsumexp
    lse_base2 = lse_val / LOG2  # base-2 logsumexp

    # Softmax over logits_scaled
    probs = exp_vals / sum_exp

    # Output vector for this head: out_vec = probs @ v_vals -> [HEAD_DIM]
    out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
    for j in range(max_kv_idx):
        v_row = v_vals[j, :]  # [HEAD_DIM]
        out_vec += probs[j] * v_row

    # Store output[global_q_idx, h, :]
    out_base = out_ptr + global_q_idx * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM
    tl.store(out_base + tl.arange(0, HEAD_DIM), out_vec, mask=(tl.arange(0, HEAD_DIM) < HEAD_DIM))

    # Store lse[global_q_idx, h]
    lse_base = lse_ptr + global_q_idx * NUM_QO_HEADS + h
    tl.store(lse_base, lse_base2)
