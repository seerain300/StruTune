Analysis and plan

What the original code does
- It implements grouped single-head attention for each segment. Inputs:
  - q: [total_q, 32, 128], 32 query heads, head_dim=128
  - k_cache, v_cache: [num_pages, 1, 8, 128], squeezed to [num_pages, 8, 128]
  - qo_indptr, kv_indptr: 1D int32 arrays of length len_indptr (number of segments), with qo_indptr[-1] == total_q
  - kv_indices: int32 mapping each segment to a subset of “pages” (indices into k_cache/v_cache)
  - sm_scale = 1 / sqrt(128)
- For each segment b (len_indptr - 1):
  - q_start = qo_indptr[b], q_end = qo_indptr[b+1], kv_start = kv_indptr[b], kv_end = kv_indptr[b+1]
  - num_q_tokens_segment = q_end - q_start
  - num_kv_tokens = kv_end - kv_start
- For each query position q_idx in [q_start, q_end) and for each query head h in 0..31:
  - kv_head = h // 4 (GQA mapping; 32/8 = 4).
  - Build keys/values by reading kv_indices[kv_start:kv_end], then selecting k_cache/v_cache at those “page_ids” for kv_head.
  - Compute logits = q[h] · k_list.T for the keys present (length up to num_kv_tokens), scale by sm_scale, apply causal mask: max_kv_idx = min(q_idx + 1 + (num_kv_tokens - num_q_tokens_segment), num_kv_tokens); set logits[kk] to -inf for kk >= max_kv_idx.
  - Compute LSE = logsumexp(logits_scaled) / ln(2).
  - attn = softmax(logits_scaled) and zero out masked entries.
  - Compute out_vec = attn · v_list (sum over keys of attn[kk] * v_ptr[kk, kv_head, :]).
- Return output (bfloat16) and lse (float32).

Triton implementation strategy
- Define a single Triton kernel atencao_triton that:
  - Processes one segment per program (grid = (len_indptr - 1,)).
  - Loops over queries up to MAX_Q (upper bound) and masks out q_idx >= (q_end - q_start).
  - For each query head h:
    - kv_head = h // 4.
    - Accumulate logits as q_vec[h] · k_list across head_dim via nested loops over d in 0..127 and kk in 0..MAX_KV_TOKENS, masked by kk < num_kv_tokens.
    - Scale by sm_scale, apply causal mask, compute logsumexp and softmax, accumulate output vector via nested loops over d and kk, masked.
    - Store output and LSE.
- All computation is done inside Triton; forward only converts tensors to float32, ensures contiguity, allocates outputs, and launches the kernel.
- Use MAX_KV_TOKENS = num_pages as an upper bound, and MAX_Q = 4096 (sufficient for typical segment sizes). Triton JIT requires tl.constexpr bounds, so we pass these as meta-parameters.

Correctness considerations
- Important: The causal mask depends on num_q_tokens_segment = q_end - q_start, not the global q_end. We compute delta = num_kv_tokens - num_q_tokens_segment, then max_kv_idx = min(q_idx + 1 + delta, num_kv_tokens). This matches the original code.
- We preserve the unusual normalization: LSE divided by ln(2). sm_scale is 1/sqrt(128). All math in float32, outputs returned in bfloat16.

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def atencao_triton(
    q_ptr,           # *f32, [total_q, 32, 128]
    k_ptr,           # *f32, [num_pages, 8, 128]
    v_ptr,           # *f32, [num_pages, 8, 128]
    qo_indptr_ptr,   # *i32, [len_indptr]
    kv_indptr_ptr,   # *i32, [len_indptr]
    kv_indices_ptr,  # *i32, [num_kv_indices]
    out_ptr,         # *f32, [total_q, 32, 128]
    lse_ptr,         # *f32, [total_q, 32]
    total_q,         # i32
    num_qo_heads,    # i32 (32)
    head_dim,        # i32 (128)
    num_pages,       # i32
    num_kv_heads,    # i32 (8)
    gqa_ratio,       # i32 (4)
    sm_scale,        # f32
    LN2,             # f32 = log(2)
    MAX_Q: tl.constexpr,             # upper bound on queries per segment
    MAX_KV_TOKENS: tl.constexpr,     # upper bound on kv tokens (>= num_pages)
):
    # One program processes one segment (b)
    b = tl.program_id(axis=0)

    # Load segment bounds
    q_start = tl.load(qo_indptr_ptr + b)         # i32
    q_end = tl.load(qo_indptr_ptr + b + 1)       # i32
    kv_start = tl.load(kv_indptr_ptr + b)        # i32
    kv_end = tl.load(kv_indptr_ptr + b + 1)      # i32

    # Number of valid keys for this segment
    num_kv_tokens = kv_end - kv_start            # i32

    # For each query index within this segment
    for q_idx in range(MAX_Q):
        if q_idx >= (q_end - q_start):
            break
        global_q_idx = q_start + q_idx           # i32

        # Loop over query heads
        for h in range(num_qo_heads):
            kv_head = h // gqa_ratio             # 0..7

            # Compute logits vector for all keys in this segment
            logits = tl.zeros((MAX_KV_TOKENS,), dtype=tl.float32)
            for d in range(head_dim):
                # Load q component for this head
                q_off = global_q_idx * (num_qo_heads * head_dim) + h * head_dim
                q_val = tl.load(q_ptr + q_off + d)  # scalar
                # Accumulate logits[kk] += q_val * k_ptr[kk, kv_head, d]
                for kk in range(MAX_KV_TOKENS):
                    valid_k = kk < num_kv_tokens
                    kv_idx = kv_start + kk
                    idx = tl.load(kv_indices_ptr + kv_idx)  # int32
                    k_off = idx * (num_kv_heads * head_dim) + kv_head * head_dim + d
                    k_val = tl.load(k_ptr + k_off, mask=valid_k, other=0.0)  # scalar
                    logits[kk] += q_val * k_val

            # Scale logits by sm_scale
            logits_scaled = logits * sm_scale

            # Apply causal mask: max_kv_idx = min(q_idx + 1 + (num_kv_tokens - (q_end - q_start)), num_kv_tokens)
            num_q_tokens_segment = q_end - q_start
            delta = num_kv_tokens - num_q_tokens_segment
            delta = tl.maximum(delta, 0)
            max_kv_idx = tl.minimum(q_idx + 1 + delta, num_kv_tokens)
            # Mask logits: set to -inf beyond max_kv_idx
            for kk in range(MAX_KV_TOKENS):
                if kk >= max_kv_idx:
                    logits_scaled[kk] = -float("inf")

            # Compute LSE = logsumexp(logits_scaled) / LN2
            sum_exp = 0.0
            for kk in range(MAX_KV_TOKENS):
                sum_exp += tl.exp(logits_scaled[kk])
            lse = tl.log(sum_exp) / LN2

            # Compute attn = softmax(logits_scaled)
            attn = tl.zeros((MAX_KV_TOKENS,), dtype=tl.float32)
            for kk in range(MAX_KV_TOKENS):
                attn[kk] = tl.exp(logits_scaled[kk] - lse)
                if kk >= max_kv_idx:
                    attn[kk] = 0.0

            # Compute out_vec = attn @ v_ptr across keys for this head
            out_vec = tl.zeros((head_dim,), dtype=tl.float32)
            for d in range(head_dim):
                # Sum over keys: attn[kk] * v_ptr[kk, kv_head, d]
                for kk in range(MAX_KV_TOKENS):
                    valid_k = kk < num_kv_tokens
                    kv_idx = kv_start + kk
                    idx = tl.load(kv_indices_ptr + kv_idx)
                    v_off = idx * (num_kv_heads * head_dim) + kv_head * head_dim + d
                    v_val = tl.load(v_ptr + v_off, mask=valid_k, other=0.0)  # scalar
                    out_vec[d] += attn[kk] * v_val

            # Store output
            out_off = global_q_idx * (num_qo_heads * head_dim) + h * head_dim
            tl.store(out_ptr + out_off, out_vec)

            # Store LSE
            lse_off = global_q_idx * num_qo_heads + h
            tl.store(lse_ptr + lse_off, lse)


class ModelNew(torch.nn.Module):

        q = q.contiguous()

        v_cache = v_cache.contiguous()

        # Convert to float32 for numerics
        q_f32 = q.to(torch.float32)
        # Squeeze the "1" dimension from cache
        k_cache = k_cache.squeeze(1).contiguous()  # [num_pages, 8, 128]
        v_cache = v_cache.squeeze(1).contiguous()  # [num_pages, 8, 128]

        total_q, num_qo_heads, head_dim = q_f32.shape
        num_pages, num_kv_heads, _ = k_cache.shape
        assert num_qo_heads == 32, "num_qo_heads must be 32"

        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=device)

        return output_bf16, lse
```
