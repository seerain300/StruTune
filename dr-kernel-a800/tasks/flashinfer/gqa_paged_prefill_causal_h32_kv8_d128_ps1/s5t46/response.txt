```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def _attention_kernel(
    q_ptr,             # *float32, [total_q, 32, 128]
    k_ptr, v_ptr,      # *float32, [num_pages, 8, 128] (squeezed from original)
    qo_indptr_ptr,     # *int32, [len_indptr]
    kv_indptr_ptr,     # *int32, [len_indptr]
    kv_indices_ptr,    # *int32, [num_kv_indices]
    out_ptr,           # *float32, [total_q, 32, 128]
    lse_ptr,           # *float32, [total_q, 32]
    GQA_RATIO: tl.constexpr,     # 4
    MAX_Q: tl.constexpr,         # e.g., 4096
    MAX_KV: tl.constexpr,        # e.g., 256
    HEAD_DIM: tl.constexpr,      # 128
    sm_scale: tl.float32,        # scaling factor
):
    # Program ids: each program handles one (b, q_idx, h)
    b = tl.program_id(0)
    q_idx = tl.program_id(1)
    h = tl.program_id(2)

    # Load qo_indptr[b] and qo_indptr[b+1] to get sequence range for this batch
    qo_start = tl.load(qo_indptr_ptr + b)
    qo_end = tl.load(qo_indptr_ptr + b + 1)

    # Safety: skip if q range invalid
    # Triton requires static control flow; we mask by q_idx
    cond_q = (q_idx < (qo_end - qo_start)) & (q_idx >= 0) & (b < tl.load(qo_indptr_ptr + 0, eviction_policy='evict_last'))
    # Note: eviction_policy is not a Triton language feature; we keep it simple

    # Initialize lse and output accumulators
    # lse per (b, q_idx, h)
    lse_val = -1.0e20
    # We will compute lse via running max and sum in the first pass
    # and compute output in the second pass

    # First pass: compute logsumexp of scaled logits over candidate keys
    # Loop statically over kv indices with mask
    for i in tl.static_range(0, MAX_KV):
        # Determine if this i is within the candidate set for this query
        qo_end_val = tl.load(qo_indptr_ptr + b + 1)
        kv_end_val = tl.load(kv_indptr_ptr + b + 1)
        num_q_tokens = qo_end_val - qo_start
        num_kv_indices = kv_end_val - kv_start  # kv_start and kv_end are scalars loaded from indptr
        # kv_start and kv_end are not defined here; fix by loading them:
        kv_start = tl.load(kv_indptr_ptr + b)
        kv_end = tl.load(kv_indptr_ptr + b + 1)
        num_q_tokens = qo_end - qo_start
        num_kv_indices = kv_end - kv_start

        candidate_max = q_idx + 1 + (num_kv_indices - num_q_tokens)
        cond_i = (i < candidate_max) & (i < num_kv_indices)

        idx_i = kv_start + i  # int32
        kv_head = h // GQA_RATIO  # 0..7

        # Compute pointer offsets for k_row and v_row: [num_pages, 8, 128] -> flattened to [num_pages, 1024]
        # Offset = idx_i * (8 * HEAD_DIM) + kv_head * HEAD_DIM
        k_row_ptr = k_ptr + idx_i * (8 * HEAD_DIM) + kv_head * HEAD_DIM
        v_row_ptr = v_ptr + idx_i * (8 * HEAD_DIM) + kv_head * HEAD_DIM

        # Load k_row and v_row vectors
        # Note: Triton does not support load with dynamic pointer directly; but we can index a [HEAD_DIM] vector
        # Here we load q_sub[h] as a vector of HEAD_DIM
        q_row_ptr = q_ptr + (q_idx + qo_start) * (32 * HEAD_DIM) + h * HEAD_DIM
        # We need scalar q value at specific [q_idx] for this head; easier to load the entire [32,128] and slice if needed.
        # However, we only need q_sub[h] which is a scalar, not a vector. The original code uses q[global_q_idx, h, :],
        # but in our setup, we pass q as [total_q, 32, 128] flattened; we need to index by (q_idx + qo_start, h).
        # To keep it simple, we pass q as [total_q, 32, 128] contiguous and compute offsets accordingly.
        # But the kernel has limited access to q[h]; instead, we compute q_sub by loading q_ptr at the correct location.

        # Compute q_sub[h] vector
        q_sub_ptr = q_ptr + (q_idx + qo_start) * (32 * HEAD_DIM) + h * HEAD_DIM
        q_vec = tl.load(q_sub_ptr + tl.arange(0, HEAD_DIM), mask=cond_q, other=0.0)

        # Compute k_vec and v_vec
        k_vec = tl.load(k_row_ptr + tl.arange(0, HEAD_DIM), mask=cond_i, other=0.0)
        v_vec = tl.load(v_row_ptr + tl.arange(0, HEAD_DIM), mask=cond_i, other=0.0)

        # Compute dot product q_sub @ k_vec.T -> scalar
        dot = 0.0
        for j in tl.static_range(0, HEAD_DIM):
            dot += q_vec[j] * k_vec[j]
        logits_scaled = dot * sm_scale

        # Update running max and sum for logsumexp (m_new, s)
        # Keep scalar accumulators
        if cond_i:
            m_new = tl.maximum(lse_val, logits_scaled)
            s = s * tl.exp(lse_val - m_new) + tl.exp(m_new - lse_val) * tl.exp(logits_scaled - m_new)
            lse_val = m_new

    # Store lse for this (b, q_idx, h): Triton expects contiguous store to a 1D pointer
    lse_off = (b * (MAX_Q * 32)) + (q_idx * 32) + h
    tl.store(lse_ptr + lse_off, lse_val)

    # Second pass: compute output vector for head h
    out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
    total_sum = 0.0  # scalar accumulator

    for i in tl.static_range(0, MAX_KV):
        qo_end_val = tl.load(qo_indptr_ptr + b + 1)
        kv_end_val = tl.load(kv_indptr_ptr + b + 1)
        num_q_tokens = qo_end_val - qo_start
        num_kv_indices = kv_end_val - kv_start

        candidate_max = q_idx + 1 + (num_kv_indices - num_q_tokens)
        cond_i = (i < candidate_max) & (i < num_kv_indices)

        idx_i = kv_start + i
        kv_head = h // GQA_RATIO

        k_row_ptr = k_ptr + idx_i * (8 * HEAD_DIM) + kv_head * HEAD_DIM
        v_row_ptr = v_ptr + idx_i * (8 * HEAD_DIM) + kv_head * HEAD_DIM

        q_sub_ptr = q_ptr + (q_idx + qo_start) * (32 * HEAD_DIM) + h * HEAD_DIM
        q_vec = tl.load(q_sub_ptr + tl.arange(0, HEAD_DIM), mask=cond_q, other=0.0)

        k_vec = tl.load(k_row_ptr + tl.arange(0, HEAD_DIM), mask=cond_i, other=0.0)
        v_vec = tl.load(v_row_ptr + tl.arange(0, HEAD_DIM), mask=cond_i, other=0.0)

        dot = 0.0
        for j in tl.static_range(0, HEAD_DIM):
            dot += q_vec[j] * k_vec[j]
        logits_scaled = dot * sm_scale
        prob = tl.exp(logits_scaled - lse_val) * tl.where(cond_i, 1.0, 0.0)
        out_vec += prob * v_vec
        total_sum += tl.where(cond_i, prob, 0.0) * (prob * 0.0 + 1.0)  # ensure accumulation; but prob*v already contributes

    # Store output vector for this (b, q_idx, h) at out_ptr
    out_off = (b * (MAX_Q * 32 * HEAD_DIM)) + (q_idx * 32 + h) * HEAD_DIM
    for j in tl.static_range(0, HEAD_DIM):
        tl.store(out_ptr + out_off + j, out_vec[j])

    # Note: We are not storing output for all positions; the grid covers all (b, q_idx, h).
    # The host code will assemble the full output tensor using out_ptr. Here we store per (b,q_idx,h).
    # To assemble full output, host code uses out_ptr for each (b,q_idx,h) block. This kernel computes per block.
```
