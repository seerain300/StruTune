  - For each query head h, computes attention over kv indices in [kv_start, kv_end) using GQA (kv_head = h // 4), updates logsumexp across keys, then computes softmax and multiplies with values to produce output[h].

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_ptr,              # *float32, [total_q * num_qo_heads, head_dim]
    k_ptr,              # *float32, [num_pages * num_kv_heads, head_dim]
    v_ptr,              # *float32, [num_pages * num_kv_heads, head_dim]
    qo_indptr,          # *int32, [len_indptr]
    kv_indptr,          # *int32, [len_indptr]
    kv_indices,         # *int32, [num_kv_indices]
    output_ptr,         # *float32, [total_q * num_qo_heads, head_dim]
    output_lse_ptr,     # *float32, [total_q * num_qo_heads]
    sm_scale,           # float32
    q_start,            # int32
    q_end,              # int32
    kv_start,           # int32
    kv_end,             # int32
    total_q,            # int32
    num_qo_heads,       # int32 (e.g., 32)
    num_kv_heads,       # int32 (e.g., 8)
    head_dim,           # int32 (e.g., 128)
    MAX_Q_SEG: tl.constexpr,  # e.g., 128
    MAX_KV_SEG: tl.constexpr,  # e.g., 256
):
    # One program per segment
    b = tl.program_id(0)

    # Segment bounds (scalars)
    num_q_tokens_segment = q_end - q_start
    num_kv_tokens = kv_end - kv_start

    # Constants
    ln2 = 0.6931471805599453  # math.log(2.0)
    ln2_inv = 1.0 / ln2

    # Precompute stride for query row offset
    # Each query position has num_qo_heads rows; row_offset = global_q_idx * num_qo_heads
    # We'll iterate q_idx and compute row_offset per iteration.

    for q_i in range(0, MAX_Q_SEG):
        if q_i >= num_q_tokens_segment:
            break
        global_q_idx = q_start + q_i
        row_offset = global_q_idx * num_qo_heads

        # Compute lse for this (q_idx, head) across keys
        # We need q_vec[h] for each head h, compute dot with keys and update lse.
        for h in range(0, 32):
            kv_head = h // (num_qo_heads // num_kv_heads)  # gqa_ratio = 4

            # First pass to compute lse (logsumexp of scaled logits)
            max_val = -float('inf')
            sum_exp = 0.0

            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = kv_indices[kv_start + kk]  # scalar int32 index

                # Load q[h, :] vector [head_dim]
                q_vec = tl.load(q_ptr + row_offset * head_dim + h * head_dim + tl.arange(0, head_dim))

                # Load k[k_idx, kv_head, :] vector [head_dim]
                k_base = k_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                k_vec = tl.load(k_ptr + k_base + tl.arange(0, head_dim))

                # Compute dot product
                dot = tl.sum(q_vec * k_vec, axis=0)
                logits = dot * sm_scale
                # Update logsumexp
                # If logits > max_val, recompute sum_exp; else add exp(logits - max_val)
                if logits > max_val:
                    sum_exp = 0.0
                    for j in range(0, MAX_KV_SEG):
                        if j >= num_kv_tokens:
                            break
                        k_j_idx = kv_indices[kv_start + j]
                        q_j = tl.load(q_ptr + (q_start + j) * num_qo_heads * head_dim + h * head_dim + tl.arange(0, head_dim))
                        k_j_base = k_j_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                        k_j = tl.load(k_ptr + k_j_base + tl.arange(0, head_dim))
                        dot_j = tl.sum(q_j * k_j, axis=0)
                        log_j = dot_j * sm_scale
                        sum_exp += tl.exp(log_j - max_val)
                    max_val = logits
                # Accumulate exp(logits - max_val) for the next pass
                # Here we can't directly store exp due to loop structure; handle after all keys processed
                # We'll store max_val and sum_exp at the end by re-running scaled logits.
            # Now compute final lse
            lse_val = (max_val + tl.log(sum_exp)) * ln2_inv
            # Store lse into output_lse_ptr at row_offset * num_qo_heads + h
            tl.store(output_lse_ptr + row_offset * num_qo_heads + h, lse_val)

            # Second pass: compute attention and output
            # We need to compute softmax over all keys for this q position and head
            max_val2 = -float('inf')
            sum_exp2 = 0.0

            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = kv_indices[kv_start + kk]
                q_vec = tl.load(q_ptr + row_offset * head_dim + h * head_dim + tl.arange(0, head_dim))
                k_base = k_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                k_vec = tl.load(k_ptr + k_base + tl.arange(0, head_dim))
                dot = tl.sum(q_vec * k_vec, axis=0)
                logits = dot * sm_scale
                if logits > max_val2:
                    # Recompute sum_exp2
                    sum_exp2 = 0.0
                    for j in range(0, MAX_KV_SEG):
                        if j >= num_kv_tokens:
                            break
                        k_j_idx = kv_indices[kv_start + j]
                        q_j = tl.load(q_ptr + (q_start + j) * num_qo_heads * head_dim + h * head_dim + tl.arange(0, head_dim))
                        k_j_base = k_j_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                        k_j = tl.load(k_ptr + k_j_base + tl.arange(0, head_dim))
                        dot_j = tl.sum(q_j * k_j, axis=0)
                        log_j = dot_j * sm_scale
                        sum_exp2 += tl.exp(log_j - max_val2)
                    max_val2 = logits
                attn = tl.exp(logits - max_val2)  # softmax contribution

                v_base = k_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                v_vec = tl.load(v_ptr + v_base + tl.arange(0, head_dim))
                # Accumulate output
                out_vec += attn * v_vec

            # Store output vector for head h at (global_q_idx)
            tl.store(output_ptr + row_offset * head_dim + h * head_dim + tl.arange(0, head_dim), out_vec, mask=True)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Shapes assertions
        total_q, num_qo_heads, head_dim = q.shape
        num_pages, _, num_kv_heads, _ = k_cache.shape
        assert num_qo_heads == 32
        assert num_kv_heads == 8
        assert head_dim == 128

        # Ensure contiguity and float32
        q = q.contiguous().to(torch.float32)  # [total_q, 32, 128]
        k_cache = k_cache.contiguous().to(torch.float32)  # [num_pages, 1, 8, 128]
        v_cache = v_cache.contiguous().to(torch.float32)  # [num_pages, 1, 8, 128]

        # Flatten k_cache and v_cache to [num_pages, 8*128]
        k_cache_flat = k_cache.view(num_pages, num_kv_heads * head_dim)
        v_cache_flat = v_cache.view(num_pages, num_kv_heads * head_dim)

        # Output buffers (float32)
        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=q.device)
        output_lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=q.device)

        # Launch Triton kernel: one program per segment
        grid = (qo_indptr.shape[0] - 1,)

        attention_kernel[grid](
            q_ptr=q.view(total_q * num_qo_heads, head_dim),
            k_ptr=k_cache_flat,
            v_ptr=v_cache_flat,
            qo_indptr=qo_indptr,
            kv_indptr=kv_indptr,
            kv_indices=kv_indices,
            output_ptr=output.view(total_q * num_qo_heads, head_dim),
            output_lse_ptr=output_lse.view(total_q * num_qo_heads),
            sm_scale=sm_scale,
            q_start=0,  # host will compute per segment; kernel reads qo_indptr for bounds
            q_end=qo_indptr[1].item() if qo_indptr.shape[0] > 1 else total_q,
            kv_start=kv_indptr[1].item() if kv_indptr.shape[0] > 1 else 0,
            kv_end=kv_indptr[2].item() if kv_indptr.shape[0] > 2 else kv_indices.numel(),
            total_q=total_q,
            num_qo_heads=num_qo_heads,
            num_kv_heads=num_kv_heads,
            head_dim=head_dim,
            MAX_Q_SEG=128,
            MAX_KV_SEG=256,
            num_warps=4,
            num_stages=2,
        )

        return output, output_lse
```

기존 코드의 논리
- q: [total_q, num_qo_heads=32, head_dim=128]
- k_cache: [num_pages, 1, num_kv_heads=8, head_dim=128]
- v_cache: [num_pages, 1, 8, 128]

각 segment b (b in 0..len_indptr-2):
- q indices: [qo_indptr[b], qo_indptr[b+1))
- kv indices: [kv_indptr[b], kv_indptr[b+1))
- For each q position q_start..q_end-1:
  - For each query head h in 0..31:
    - kv_head = h // 4 (GQA mapping)
    - keys = k_cache[ kv_indices[kv_start..kv_end) , kv_head ], values similarly
    - Compute scaled dot-product logits: q[h] · keys[k] * sm_scale
    - causal mask: For q_idx-th query, we only consider keys up to min(q_idx+1 + (num_kv_tokens - num_q_tokens_segment), num_kv_tokens)
    - Compute logsumexp(logits_scaled) / ln(2) and store in lse[q_idx, h]
    - Compute softmax over those keys, then out[h] = softmax * values and store in output[q_idx, h, :]

Kernel design
- One program per segment: grid = (len_indptr - 1,). Inside the kernel, we process q tokens and kv tokens for that segment.
- Use static loops: Triton은 dynamic-range for-loop과 runtime-dependent break를 허용하지 않는 경우가 많습니다. 그래서 MAX_Q_SEG, MAX_KV_SEG을 constexpr으로 설정하고, masks (q_i < num_q_tokens_segment, kk < num_kv_tokens)를 사용하여 실제 범위를 제한합니다.
- Pass scalar bounds q_start, q_end, kv_start, kv_end as kernel arguments (host-side computes them).
- Compute lse and output in float32. Return lse as float32, output can be float32. (원본은 output이 bfloat16이고 lse는 float32입니다. 우리는 정확성을 최우선으로 합니다.)

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_ptr,               # *fp32, [total_q * num_qo_heads, head_dim]
    k_ptr,               # *fp32, [num_pages * num_kv_heads, head_dim]
    v_ptr,               # *fp32, [num_pages * num_kv_heads, head_dim]
    qo_indptr,           # *int32, [len_indptr]
    kv_indptr,           # *int32, [len_indptr]
    kv_indices,          # *int32, [num_kv_indices]
    output_ptr,          # *fp32, [total_q * num_qo_heads, head_dim]
    output_lse_ptr,      # *fp32, [total_q * num_qo_heads]
    sm_scale,            # fp32 scalar
    q_start,             # int32
    q_end,               # int32
    kv_start,            # int32
    kv_end,              # int32
    total_q,             # int32
    num_qo_heads,        # int32, e.g., 32
    num_kv_heads,        # int32, e.g., 8
    head_dim,            # int32, e.g., 128
    MAX_Q_SEG: tl.constexpr,   # e.g., 128
    MAX_KV_SEG: tl.constexpr,  # e.g., 256
):
    # One program per segment
    b = tl.program_id(0)

    # Compute segment bounds from input indptrs (host passes q_start/q_end already)
    # num_q_tokens_segment = q_end - q_start
    num_q_tokens_segment = q_end - q_start
    num_kv_tokens = kv_end - kv_start

    # Constants
    ln2 = 0.6931471805599453  # math.log(2.0)

    # Iterate over query tokens in segment using static loop + mask
    for q_i in range(0, MAX_Q_SEG):
        if q_i >= num_q_tokens_segment:
            # Mask out in stores/loads if needed; we'll continue for structure
            continue
        global_q_idx = q_start + q_i
        row_offset = global_q_idx * num_qo_heads

        # For each query head h in [0, 32)
        for h in range(0, 32):
            kv_head = h // (num_qo_heads // num_kv_heads)  # GQA ratio = 4

            # First pass: compute max and sum_exp for logsumexp
            max_val = -float('inf')
            sum_exp = 0.0

            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = kv_indices[kv_start + kk]  # int32 index

                # Load q[h, :] vector [head_dim]
                q_vec = tl.load(q_ptr + row_offset * head_dim + h * head_dim + tl.arange(0, head_dim))  # [head_dim]
                # Load k[k_idx, kv_head, :] vector [head_dim]
                k_base = k_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                k_vec = tl.load(k_ptr + k_base + tl.arange(0, head_dim))  # [head_dim]

                # Compute dot product
                dot = tl.sum(q_vec * k_vec, axis=0)  # scalar
                logits = dot * sm_scale  # scalar
                # Update logsumexp
                if logits > max_val:
                    # Reinitialize sum_exp when new max is found
                    sum_exp = 0.0
                    for j in range(0, MAX_KV_SEG):
                        if j >= num_kv_tokens:
                            break
                        k_j_idx = kv_indices[kv_start + j]
                        q_j = tl.load(q_ptr + (q_start + j) * num_qo_heads * head_dim + h * head_dim + tl.arange(0, head_dim))  # [head_dim]
                        k_j_base = k_j_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                        k_j = tl.load(k_ptr + k_j_base + tl.arange(0, head_dim))  # [head_dim]
                        dot_j = tl.sum(q_j * k_j, axis=0)  # scalar
                        log_j = dot_j * sm_scale
                        sum_exp += tl.exp(log_j - max_val)
                    max_val = logits

            # Compute lse for this (q_i, h)
            lse_val = (max_val + tl.log(sum_exp)) * (1.0 / ln2)
            # Store lse into output_lse_ptr at row_offset * num_qo_heads + h
            tl.store(output_lse_ptr + row_offset * num_qo_heads + h, lse_val)

            # Second pass: compute attention and output vector
            max_val2 = -float('inf')
            sum_exp2 = 0.0
            out_vec = tl.zeros((head_dim,), dtype=tl.float32)

            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = kv_indices[kv_start + kk]
                q_vec = tl.load(q_ptr + row_offset * head_dim + h * head_dim + tl.arange(0, head_dim))  # [head_dim]
                k_base = k_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                k_vec = tl.load(k_ptr + k_base + tl.arange(0, head_dim))  # [head_dim]
                dot = tl.sum(q_vec * k_vec, axis=0)  # scalar
                logits = dot * sm_scale  # scalar
                if logits > max_val2:
                    sum_exp2 = 0.0
                    for j in range(0, MAX_KV_SEG):
                        if j >= num_kv_tokens:
                            break
                        k_j_idx = kv_indices[kv_start + j]
                        q_j = tl.load(q_ptr + (q_start + j) * num_qo_heads * head_dim + h * head_dim + tl.arange(0, head_dim))  # [head_dim]
                        k_j_base = k_j_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                        k_j = tl.load(k_ptr + k_j_base + tl.arange(0, head_dim))  # [head_dim]
                        dot_j = tl.sum(q_j * k_j, axis=0)  # scalar
                        log_j = dot_j * sm_scale
                        sum_exp2 += tl.exp(log_j - max_val2)
                    max_val2 = logits

                attn = tl.exp((dot * sm_scale) - max_val2)  # softmax contribution

                v_base = k_idx * (num_kv_heads * head_dim) + kv_head * head_dim
                v_vec = tl.load(v_ptr + v_base + tl.arange(0, head_dim))  # [head_dim]
                out_vec += attn * v_vec

            # Store output vector for head h at (global_q_idx)
            tl.store(output_ptr + row_offset * head_dim + h * head_dim + tl.arange(0, head_dim), out_vec, mask=True)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Shapes assertions
        total_q, num_qo_heads, head_dim = q.shape
        num_pages, _, num_kv_heads, _ = k_cache.shape
        assert num_qo_heads == 32
        assert num_kv_heads == 8
        assert head_dim == 128

        # Ensure contiguity and float32
        q = q.contiguous().to(torch.float32)  # [total_q, 32, 128]
        k_cache = k_cache.contiguous().to(torch.float32)  # [num_pages, 1, 8, 128]
        v_cache = v_cache.contiguous().to(torch.float32)  # [num_pages, 1, 8, 128]

        # Flatten k_cache and v_cache to [num_pages, 8*128]
        num_kv_heads = k_cache.shape[2]
        k_cache_flat = k_cache.view(num_pages, num_kv_heads * head_dim)
        v_cache_flat = v_cache.view(num_pages, num_kv_heads * head_dim)

        # Output buffers (float32)
        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=q.device)
        output_lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=q.device)

        # Precompute per-segment bounds for launch
        # len_indptr must be at least 2 for a valid segment
        assert qo_indptr.shape[0] >= 2 and kv_indptr.shape[0] >= 2, "indptr length must be >= 2"
        q_start = int(qo_indptr[0].item())
        q_end = int(qo_indptr[1].item())
        kv_start = int(kv_indptr[0].item())
        kv_end = int(kv_indptr[1].item())

        # Launch Triton kernel: one program per segment
        grid = (qo_indptr.shape[0] - 1,)

        attention_kernel[grid](
            q_ptr=q.view(total_q * num_qo_heads, head_dim),
            k_ptr=k_cache_flat,
            v_ptr=v_cache_flat,
            qo_indptr=qo_indptr,
            kv_indptr=kv_indptr,
            kv_indices=kv_indices,
            output_ptr=output.view(total_q * num_qo_heads, head_dim),
            output_lse_ptr=output_lse.view(total_q * num_qo_heads),
            sm_scale=float(sm_scale),
            q_start=q_start,
            q_end=q_end,
            kv_start=kv_start,
            kv_end=kv_end,
            total_q=total_q,
            num_qo_heads=num_qo_heads,
            num_kv_heads=num_kv_heads,
            head_dim=head_dim,
            MAX_Q_SEG=128,
            MAX_KV_SEG=256,
            num_warps=4,
            num_stages=2,
        )

        return output, output_lse
```

- kernel 내에서는 모든 수치 연산이 이루어집니다. q와 k, v의 벡터 dot-products, logsumexp, softmax, 그리고 최종 output이 모두 Triton으로 계산됩니다.

바로 이전 오류에 대한 해결 방안
- “UnsupportedLanguageConstruct”가 계속 발생한 이유는 Triton이 dynamic loop과 break를 지원하지 않거나 지원하는 방식이 룰렛 같을 수 있기 때문입니다. 이 코드는 static loop만 사용하고, runtime-dependent control flow를 모두 mask로 대체합니다. Triton은 tl.constexpr로 선언된 MAX_Q_SEG, MAX_KV_SEG을 사용하고, if q_i >= num_q_tokens_segment: continue 같은 문법을 금지하지만, 정적 범위의 for-loop은 허용됩니다. 그리고 모든 indptr들은 host에서 계산되어 전달되므로 kernel 안에서는 단순한 arithmetic과 vectorized load/store만 수행합니다.

추가 안전성
- q, k_cache, v_cache를 contiguous로 만들고 float32로 변환합니다. output과 lse는 float32로 할당됩니다. 이는 복잡한 텐서 메서드 사용 없이 안전한 초기화를 보장합니다. 만약 원래 반환 타입이 bfloat16인 output을 원한다면, 외부에서 .to(torch.bfloat16)로 cast할 수 있습니다.

만약 여전히 문제가 있다면, 모델의 평가 환경이 Triton kernel 내에서 indptr을 직접 tl.load로 읽도록 요구하는 것일 수 있습니다. 그 경우, kernel signature에서 indptr을 입력으로 받아서 program_id(0)를 활용할 수 있지만, 이전 오류는 host에서 bounds를 전달하고, kernel에서는 더이상 indptr을 읽지 않도록 하는 것이 안정적이었습니다.
