  - kv_indptr: [len_indptr], int32 (cumsum over batch segments of keys/values)
  - kv_indices: [num_kv_indices], int32, mapping kv slots to k_cache indices
  - sm_scale: float32
- Assumptions:
  - num_qo_heads = 32, num_kv_heads = 8, head_dim = 128, gqa_ratio = 4

      - kv_head = h // 4

        - logits[k] = q[h] · k_cache[kv_indices[k], kv_head]

      - output[h] = attn · v_cache[...] (same k mapping)

Design for Triton kernel

- One Triton program per segment: grid = (len_indptr - 1,)
- Kernel arguments:
  - Pointers to q, k_cache, v_cache (contiguous tensors)
  - Pointers to qo_indptr, kv_indptr, kv_indices
  - Output pointers (output and lse)
  - Scalar segment bounds: q_start, q_end, kv_start, kv_end
  - Meta parameters: head_dim, num_qo_heads, num_kv_heads, gqa_ratio, MAX_Q_SEG, MAX_KV_SEG (constexpr), sm_scale
- Inside kernel:
  - Compute num_q_tokens_segment = q_end - q_start, num_kv_tokens = kv_end - kv_start
  - Loop over q_i in range(0, MAX_Q_SEG) with mask q_i < num_q_tokens_segment
    - global_q_idx = q_start + q_i
    - For each query head h in 0..31:
      - kv_head = h // 4
      - Compute lse:
        - max_val = -inf, sum_exp = 0.0
        - Loop kk in 0..MAX_KV_SEG, mask kk < num_kv_tokens
          - k_idx = kv_indices[kv_start + kk]
          - Load q_row[h] and k_row[k_idx, kv_head], compute dot = sum(q_row * k_row), logit = dot * sm_scale
          - Update logsumexp: new_max = max(max_val, logit), sum_exp = sum_exp * exp(max_val - new_max) + exp(logit - new_max), max_val = new_max
        - lse_val = (max_val + log(sum_exp)) / ln(2) and store to output_lse[global_q_idx, h]
      - Compute causal max_kv_idx = min(q_i + 1 + (num_kv_tokens - num_q_tokens_segment), num_kv_tokens)
      - Compute output vector out_vec:
        - Loop kk in 0..MAX_KV_SEG, mask kk < num_kv_tokens:
          - k_idx = kv_indices[kv_start + kk]
          - Compute logits[k] as above
          - attn = exp(logit - max_val) if kk < max_kv_idx else 0
          - Load v_row = v_cache[k_idx, kv_head, :]
          - out_vec += attn * v_row
      - Store output_vec to output[global_q_idx, h, :]
- No PyTorch ops inside host code; we allocate outputs as empty tensors and kernel fills them.

Why this fixes the prior failures

- We avoid unsupported Triton constructs by using only static loops with masks and no tl.load of dynamic pointers for bounds. Segment bounds are passed as scalar args.

import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_ptr,                    # *float32, shape [total_q * 32, 128]
    k_ptr,                    # *float32, shape [num_pages * 8, 128] (flattened)
    v_ptr,                    # *float32, shape [num_pages * 8, 128] (flattened)
    qo_indptr_ptr,            # *int32, shape [len_indptr]
    kv_indptr_ptr,            # *int32, shape [len_indptr]
    kv_indices_ptr,           # *int32, shape [num_kv_indices]
    output_ptr,               # *float32, shape [total_q * 32, 128]
    output_lse_ptr,           # *float32, shape [total_q * 32]
    sm_scale,                 # float32
    # segment bounds (scalars, passed from host)
    q_start, q_end,
    kv_start, kv_end,
    # meta-parameters
    head_dim: tl.constexpr,   # 128
    num_qo_heads: tl.constexpr,  # 32
    num_kv_heads: tl.constexpr,  # 8
    gqa_ratio: tl.constexpr,      # 4
    MAX_Q_SEG: tl.constexpr,
    MAX_KV_SEG: tl.constexpr,
):
    b = tl.program_id(axis=0)  # segment index
    # Compute segment lengths
    num_q_tokens_segment = q_end - q_start
    num_kv_tokens = kv_end - kv_start

    ln2 = 0.6931471805599453  # math.log(2.0)
    ln2_inv = 1.0 / ln2

    # Iterate over query tokens in this segment with static loop and mask
    for q_i in range(0, MAX_Q_SEG):
        if q_i >= num_q_tokens_segment:
            break
        global_q_idx = q_start + q_i
        row_offset = global_q_idx * num_qo_heads

        # Compute logsumexp for each query head
        for h in range(0, num_qo_heads):
            kv_head = h // gqa_ratio  # GQA mapping

            max_val = -float('inf')
            sum_exp = 0.0

            # Loop over KV tokens in this segment (masked by num_kv_tokens)
            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = kv_indices_ptr[kv_start + kk]  # int32

                # Load q[h, :] from q_ptr (row-major [total_q*32, head_dim])
                q_row = tl.load(q_ptr + row_offset + tl.arange(0, head_dim))
                # Load k[k_idx, kv_head, :] as a contiguous 128-vector
                k_row = tl.load(k_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + tl.arange(0, head_dim))

                # Dot product between q_row and k_row
                dot = tl.sum(q_row * k_row, axis=0)  # scalar
                logit = dot * sm_scale

                # Update logsumexp in a numerically stable way
                new_max = tl.maximum(max_val, logit)
                sum_exp = sum_exp * tl.exp(max_val - new_max) + tl.exp(logit - new_max)
                max_val = new_max

            # lse = logsumexp(logits_scaled) / ln(2)
            lse_val = (max_val + tl.log(sum_exp)) * ln2_inv
            tl.store(output_lse_ptr + row_offset + h, lse_val)

            # Causal attention: effective num of keys is max_kv_idx = min(q_i+1+delta, num_kv_tokens)
            delta = num_kv_tokens - num_q_tokens_segment
            max_kv_idx = q_i + 1 + delta
            max_kv_idx = tl.minimum(max_kv_idx, num_kv_tokens)

            # Compute output vector as sum of attn * v over valid kv indices
            out_vec = tl.zeros([head_dim], dtype=tl.float32)
            for kk in range(0, MAX_KV_SEG):
                if kk >= num_kv_tokens:
                    break
                k_idx = kv_indices_ptr[kv_start + kk]
                logit = (tl.sum(tl.load(q_ptr + row_offset + tl.arange(0, head_dim)) * tl.load(k_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + tl.arange(0, head_dim)), axis=0)) * sm_scale
                attn = tl.exp(logit - max_val) if kk < max_kv_idx else 0.0
                v_row = tl.load(v_ptr + k_idx * (num_kv_heads * head_dim) + kv_head * head_dim + tl.arange(0, head_dim))
                out_vec += attn * v_row

            tl.store(output_ptr + row_offset + tl.arange(0, head_dim), out_vec)
