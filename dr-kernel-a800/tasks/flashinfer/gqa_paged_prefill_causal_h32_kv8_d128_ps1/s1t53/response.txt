                            q_vec, k_vec, dot,
                            head_dim,
                            1, 1,

                        scaled[k_local] = dot[0] * self.sm_scale

                        scaled, lse_val,
                        num_all_k, head_dim,
                    )

                    lse[global_q_idx, h] = lse_val[0]

                    _softmax_matvec_kernel[(1,)](

                    # To write the full head vector, we replicate out_vec across d dimension:
