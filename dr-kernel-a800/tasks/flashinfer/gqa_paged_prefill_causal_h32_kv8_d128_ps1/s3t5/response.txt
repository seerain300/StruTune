import math
import torch
import triton
import triton.language as tl


# Fused Triton kernel: per (query, head), compute
# - logits_scaled = q @ k.T (q: [HEAD_DIM], k: [HEAD_DIM, HEAD_DIM])
# - lse = logsumexp(logits_scaled) / ln(2) (scalar)
# - attn = softmax(logits_scaled)
# - out = v @ attn (v: [HEAD_DIM, HEAD_DIM])
# Writes:
# - out_vec_ptr[0..HEAD_DIM-1] = out (float32), later cast to bfloat16
# - lse_ptr[0] = lse (float32)
# We pass k and v as full [HEAD_DIM, HEAD_DIM] matrices but only use the first NUM_KV rows where NUM_KV is determined by the batch segment.
# To keep correctness, we set k and v such that only the first max_kv_idx rows/columns contain valid data; others are zero. However, since we don't
# know max_kv_idx at compile time, we simply operate with loops up to HEAD_DIM and assume inputs are prepared accordingly. In host code, we will
# set k and v to have only the needed rows and zeros elsewhere.
@triton.jit
def _fused_perhead_kernel(q_vec_ptr, k_mat_ptr, v_mat_ptr, out_vec_ptr, lse_ptr,
                          HEAD_DIM: tl.constexpr, NUM_KV: tl.constexpr, SM_SCALE: tl.float32):
    # Compute logits_scaled = q @ k.T, length NUM_KV
    logits_scaled = tl.zeros([NUM_KV], dtype=tl.float32)
    for i in range(NUM_KV):
        acc = tl.zeros((), dtype=tl.float32)
        for j in range(HEAD_DIM):
            qj = tl.load(q_vec_ptr + j)
            ki_j = tl.load(k_mat_ptr + i * HEAD_DIM + j)  # k_mat is [HEAD_DIM, HEAD_DIM] row-major
            acc += qj * ki_j
        logits_scaled[i] = acc * SM_SCALE

    # Compute scalar logsumexp of logits_scaled
    m = logits_scaled[0]
    for j in range(1, NUM_KV):
        vj = logits_scaled[j]
        m = tl.maximum(m, vj)
    sum_exp = tl.zeros((), dtype=tl.float32)
    for j in range(0, NUM_KV):
        vj = logits_scaled[j]
        sum_exp += tl.exp(vj - m)
    lse = tl.log(sum_exp) + m
    tl.store(lse_ptr, lse / 0.6931471805599651)  # divide by ln(2)

    # Compute softmax over logits_scaled
    sum_exp = 0.0
    for j in range(NUM_KV):
        sum_exp += tl.exp(logits_scaled[j] - m)
    attn = tl.zeros([NUM_KV], dtype=tl.float32)
    for j in range(NUM_KV):
        attn[j] = tl.exp(logits_scaled[j] - m) / sum_exp

    # Compute out = v @ attn
    out = tl.zeros([HEAD_DIM], dtype=tl.float32)
    for j in range(NUM_KV):
        aj = attn[j]
        for r in range(HEAD_DIM):
            vj_r = tl.load(v_mat_ptr + r * HEAD_DIM + j)  # v_mat is [HEAD_DIM, HEAD_DIM] row-major
            out[r] += aj * vj_r
    # Store out vector
    for r in range(HEAD_DIM):
        tl.store(out_vec_ptr + r, out[r])


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Ensure CUDA tensors and contiguity
        device = q.device
        assert q.is_cuda and k_cache.is_cuda and v_cache.is_cuda and qo_indptr.is_cuda and kv_indptr.is_cuda and kv_indices.is_cuda, "All inputs must be on CUDA for Triton."
        q = q.contiguous()
        k_cache = k_cache.contiguous()
        v_cache = v_cache.contiguous()
        qo_indptr = qo_indptr.contiguous()
        kv_indptr = kv_indptr.contiguous()
        kv_indices = kv_indices.contiguous()

        # Flatten k_cache/v_cache by squeezing the singleton "page_size" dimension
        k_cache_flat = k_cache.squeeze(1).contiguous()  # [num_pages, 8, 128]
        v_cache_flat = v_cache.squeeze(1).contiguous()  # [num_pages, 8, 128]

        total_q, num_qo_heads, head_dim = q.shape
        assert num_qo_heads == 32, "num_qo_heads must be 32"
        assert head_dim == 128, "head_dim must be 128"

        num_pages, _, num_kv_heads, _ = k_cache.shape
        assert num_kv_heads == 8, "num_kv_heads must be 8"

        len_indptr = qo_indptr.shape[0]
        num_kv_indices = kv_indices.shape[0]

        # Output tensors (we'll produce output in float32 and cast to bfloat16; lse in float32)
        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Convert inputs to float32 for compute
        q_f32 = q.to(torch.float32).contiguous()                  # [total_q, 32, 128]
        k_cache_f32 = k_cache_flat.to(torch.float32).contiguous() # [num_pages, 8, 128]
        v_cache_f32 = v_cache_flat.to(torch.float32).contiguous() # [num_pages, 8, 128]

        # Loop over batch segments
        for b in range(len_indptr - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())

            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            if q_start >= q_end or kv_start >= kv_end:
                continue

            # Cached K/V groups for this batch: [num_kv_tokens]
            num_kv_tokens = kv_end - kv_start
            if num_kv_tokens <= 0:
                continue

            # Fetch K/V
            k_batch = k_cache_f32[kv_start:kv_end]  # [num_kv_tokens, 8, 128]
            v_batch = v_cache_f32[kv_start:kv_end]  # [num_kv_tokens, 8, 128]

            # Queries for this segment
            q_batch = q_f32[q_start:q_end]  # [num_q_tokens, 32, 128]
            num_q_tokens = q_batch.shape[0]

            # Causal-like max_kv_idx per query position
            # max_kv_idx = min(q_idx + 1 + (num_kv_tokens - num_q_tokens), num_kv_tokens)
            for q_idx in range(num_q_tokens):
                global_q_idx = q_start + q_idx
                max_kv_idx = min(q_idx + 1 + (num_kv_tokens - num_q_tokens), num_kv_tokens)
                if max_kv_idx <= 0:
                    continue

                # For each query head h
                for h in range(num_qo_heads):
                    kv_head = h // (num_qo_heads // num_kv_heads)  # h // 4

                    # Select vectors for this head (128-dim)
                    q_pos = q_batch[q_idx]  # [32, 128]
                    q_head = q_pos[h].contiguous()  # [128], float32

                    # Select K/V for this head
                    # We will build k_full and v_full as [HEAD_DIM, HEAD_DIM] with only first max_kv_idx rows valid, others zero.
                    # But to simplify, we directly use k_batch[:, kv_head] and v_batch[:, kv_head] and zero-pad to HEAD_DIM by constructing k_mat and v_mat.
                    # However, Triton kernel expects fixed-size matrices. A simpler approach:
                    # We construct k_mat and v_mat as [HEAD_DIM, HEAD_DIM] such that first max_kv_idx rows are k/v, others zeros.
                    # Allocate them as zeros and then slice.
                    k_mat = torch.zeros((head_dim, head_dim), dtype=torch.float32, device=device)
                    v_mat = torch.zeros((head_dim, head_dim), dtype=torch.float32, device=device)
                    # Fill k_mat[:, :max_kv_idx] and v_mat[:, :max_kv_idx]
                    k_group = k_batch[:, kv_head]  # [num_kv_tokens, 128]; but we only need up to max_kv_idx
                    v_group = v_batch[:, kv_head]

                    # Copy k_group and v_group into k_mat and v_mat columns 0..max_kv_idx-1, rows 0..max_kv_idx-1
                    # k_mat[r, j] = k_group[j, r] for r < max_kv_idx, j < max_kv_idx
                    # We can build k_mat and v_mat by copying k_group.T[:max_kv_idx, :max_kv_idx] and v_group.T[:max_kv_idx, :max_kv_idx]
                    if max_kv_idx > 0:
                        k_sub = k_group[:max_kv_idx]    # [max_kv_idx, 128]
                        v_sub = v_group[:max_kv_idx]    # [max_kv_idx, 128]
                        # Assign to k_mat[:, :max_kv_idx], v_mat[:, :max_kv_idx] using PyTorch slicing
                        # Since Triton cannot handle runtime slices, we ensure these tensors exist in PyTorch and are contiguous.
                        # The kernel will treat k_mat and v_mat as full [HEAD_DIM, HEAD_DIM] matrices; rows beyond max_kv_idx are zeros.

                    # Launch fused kernel for this (query, head)
                    out_vec = torch.empty(head_dim, dtype=torch.float32, device=device)
                    lse_elem = torch.empty(1, dtype=torch.float32, device=device)

                    _fused_perhead_kernel[(1,)](
                        q_head, k_mat, v_mat, out_vec, lse_elem,
                        HEAD_DIM=128,
                        NUM_KV=max_kv_idx,
                        SM_SCALE=sm_scale,
                    )

                    # Store output
                    output[global_q_idx, h] = out_vec.to(torch.bfloat16)

                    # Store lse (not used in output but kept for completeness)
                    # Note: We compute lse in Triton as well. The previous line does that; we can rely on lse_elem[0].
                    lse[global_q_idx, h] = lse_elem[0]

        return output, lse
