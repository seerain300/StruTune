import math
import torch

import triton
import triton.language as tl


@triton.jit
def _compute_one_triplet_kernel(
    q_ptr,            # *float32, [total_q, num_qo_heads, head_dim]
    k_ptr, v_ptr,     # *float32, [num_pages, num_kv_heads, head_dim] (squeezed from original)
    qo_indptr_ptr,    # *int32, [len_indptr]
    kv_indptr_ptr,    # *int32, [len_indptr]
    kv_indices_ptr,   # *int32, [num_kv_indices]
    out_ptr,          # *float32, [total_q, num_qo_heads, head_dim]
    lse_ptr,          # *float32, [total_q, num_qo_heads]
    GQA_RATIO: tl.constexpr,   # 4
    MAX_KV: tl.constexpr,      # e.g., 256
    sm_scale: tl.float32,      # scaling factor
):
    # Program ids: each program handles one (b, q_idx, h)
    b = tl.program_id(0)
    q_idx = tl.program_id(1)
    h = tl.program_id(2)

    # Load qo_indptr[b] and qo_indptr[b+1] to get sequence range for this batch
    qo_start = tl.load(qo_indptr_ptr + b)
    qo_end = tl.load(qo_indptr_ptr + b + 1)

    # GQA mapping: query head h uses kv head h // GQA_RATIO
    kv_head = h // GQA_RATIO

    # For this batch b, compute lengths
    num_q_tokens_in_b = qo_end - qo_start
    # num_kv_indices_in_b and kv_end are not directly available per program; but we can infer
    # We need kv_start and kv_end. However, for each program we only know b. The input arrays
    # are global. We need to access kv_indptr[b] and kv_indptr[b+1] similarly.
    kv_start = tl.load(kv_indptr_ptr + b)
    kv_end = tl.load(kv_indptr_ptr + b + 1)
    num_kv_indices_in_b = kv_end - kv_start

    # candidate_max and max_kv_idx are per (b, q_idx)
    candidate_max = q_idx + 1 + (num_kv_indices_in_b - num_q_tokens_in_b)
    # Clamp to MAX_KV for loop bounds
    # Triton doesn't support dynamic breaks; we'll use masked operations.
    # Prepare vectors for i in static range
    # We will build validity per i; Triton can't loop over i in Python, so we unroll up to MAX_KV.
    # But to avoid complicated per-i loads, we can load full [num_kv_heads, head_dim] tiles and slice kv_head.
    # To handle per-i indices, we instead compute a 1D offset and select idx = kv_indices[kv_start + i].
    # Triton supports loading a vector from a pointer with mask. So we'll load q_sub, k_vec, v_vec per i.

    # Compute base global_q_idx
    global_q_idx = qo_start + q_idx

    # Load q_sub vector for head h
    # q_ptr is [total_q, num_qo_heads, head_dim]; we can form offset via linear indexing
    q_off = global_q_idx * (1 * 128) + h * 128  # since num_qo_heads == 32, and we pass head_dim=128
    # Note: 1 * 128 is a placeholder; in Triton we can load with a single offset for this head
    # Triton requires contiguous pointers for simple linear indexing. Given q is [total_q, 32, 128] and contiguous,
    # we can linearize as: offset = global_q_idx * (32*128) + h * 128
    # But to keep it simple, we assume q is laid out as [T, H, D] with linear stride; we can load a 1D vector:
    # We need to load q[global_q_idx, h, :] where stride for h is 128, so linear offset:
    # offset_q = global_q_idx * (32*128) + h * 128; then load 128 elements with arange
    q_vec = tl.load(q_ptr + global_q_idx * (32 * 128) + h * 128 + tl.arange(0, 128))

    # Initialize logits_scaled and lse
    logits_scaled = tl.zeros((MAX_KV,), dtype=tl.float32)
    # We'll fill logits_scaled in a static loop with masks

    # First pass: compute logsumexp over valid i
    for i in tl.static_range(0, MAX_KV):
        # validity: i < candidate_max and i < num_kv_indices_in_b and (kv_start + i) < kv_end
        valid_i = (i < candidate_max) & (i < num_kv_indices_in_b) & ((kv_start + i) < kv_end)
        # idx for this i (safe to compute even if valid_i is False; we will mask loads)
        idx = tl.load(kv_indices_ptr + (kv_start + i))  # int32
        # Load k_vec and v_vec for this idx; cast to float32
        # k_ptr and v_ptr are [num_pages, num_kv_heads, head_dim]; squeezed
        # For a given idx (page), we select kv_head slice: k_ptr[idx, kv_head, :]
        # We need to construct a pointer for a 1D vector: use idx * (num_kv_heads * head_dim) + kv_head * head_dim
        # But Triton doesn't like this mixed arithmetic reliably. Instead, load full [num_kv_heads, head_dim] block
        # and slice kv_head. However, we don't have block sizes per idx. So we revert to loading directly with mask
        # using pointer arithmetic: k_ptr + idx * (num_kv_heads * head_dim) + kv_head * head_dim + arange(0, 128)
        # We can compute k_ptr stride by assuming original k_cache has shape [..., 8, 128] after squeeze(1).
        # Before squeeze(1), k_cache shape is [num_pages, 1, 8, 128]; stride for head_dim is 1. After squeeze(1),
        # it's [num_pages, 8, 128]. So stride between kv_heads is head_dim (128). For a fixed idx (page), we can
        # access k_ptr + idx * (8 * 128) + kv_head * 128 + arange(0, 128). This requires knowing 8; but we can
        # instead avoid complex pointer arithmetic by loading q only and handling k/v via idx and kv_head directly.
        # To simplify and avoid Triton pointer arithmetic pitfalls, we instead compute k_vec and v_vec per i using
        # direct addressing: assume k_ptr layout is contiguous [N, H, D] with N=num_pages*1*num_kv_heads (but it's not).
        # Given the earlier failures, we will avoid direct pointer arithmetic here and instead rely on known contiguity
        # by assuming idx points into a flattened [num_pages, 8, 128] tensor. Triton will handle masked loads.
        # Compute k_ptr address: idx selects a row in the flattened tensor of shape [num_pages, 8, 128]. The row base
        # address can be obtained by idx // 8, but that would require another tensor. To avoid, we instead compute
        # k_ptr address as: k_ptr + idx * (8 * 128) + kv_head * 128, which is correct if idx runs over
        # flattened (num_pages * 8). Since squeeze(1) removes dim 1, the layout is [num_pages, 8, 128] with contiguous
        # memory. Thus, idx can be interpreted as flattened index over N = num_pages * 8, and kv_head selects the
        # second dimension. Triton allows pointer + scalar, so this is valid. Similarly for v_ptr.
        k_vec = tl.load(k_ptr + idx * (8 * 128) + kv_head * 128 + tl.arange(0, 128), mask=valid_i, other=0.0)
        v_vec = tl.load(v_ptr + idx * (8 * 128) + kv_head * 128 + tl.arange(0, 128), mask=valid_i, other=0.0)

        # Compute dot(q_vec, k_vec) for this i
        # q_vec and k_vec are length 128 vectors
        dot = 0.0
        for j in tl.static_range(0, 128):
            dot += q_vec[j] * k_vec[j]
        scaled = dot * sm_scale
        logits_scaled[i] = tl.where(valid_i, scaled, -float("inf"))

    # Compute lse = logsumexp(logits_scaled) over valid positions (others are -inf)
    m = tl.max(logits_scaled, axis=0)
    s = tl.sum(tl.exp(logits_scaled - m), axis=0)
    lse_val = tl.log(s) + m  # natural log

    # Store lse to lse_ptr[global_q_idx, h]
    lse_off = global_q_idx * 32 + h  # since lse is [total_q, 32]
    tl.store(lse_ptr + lse_off, lse_val)

    # Second pass: compute output vector out[h] for this (b, q_idx, h)
    out_vec = tl.zeros((128,), dtype=tl.float32)
    for i in tl.static_range(0, MAX_KV):
        valid_i = (i < candidate_max) & (i < num_kv_indices_in_b) & ((kv_start + i) < kv_end)
        idx = tl.load(kv_indices_ptr + (kv_start + i))
        k_vec = tl.load(k_ptr + idx * (8 * 128) + kv_head * 128 + tl.arange(0, 128), mask=valid_i, other=0.0)
        v_vec = tl.load(v_ptr + idx * (8 * 128) + kv_head * 128 + tl.arange(0, 128), mask=valid_i, other=0.0)
        dot = 0.0
        for j in tl.static_range(0, 128):
            dot += q_vec[j] * k_vec[j]
        scaled = dot * sm_scale
        prob = tl.exp(scaled - lse_val)  # scaled is logits_scaled_i; using lse_val for softmax normalization
        out_vec += prob * v_vec

    # Store output vector to out_ptr[global_q_idx, h, :]
    out_off = global_q_idx * (32 * 128) + h * 128
    for j in tl.static_range(0, 128):
        tl.store(out_ptr + out_off + j, out_vec[j])


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.num_qo_heads = 32
        self.num_kv_heads = 8
        self.head_dim = 128
        self.GQA_RATIO = self.num_qo_heads // self.num_kv_heads  # 4
        self.MAX_KV = 256  # covers candidate_max up to 256 safely

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):
        # Ensure tensors are on CUDA and contiguous
        assert q.is_cuda and k_cache.is_cuda and v_cache.is_cuda and qo_indptr.is_cuda and kv_indptr.is_cuda and kv_indices.is_cuda, \
            "All inputs must be CUDA tensors for Triton kernels"
        # q: [total_q, 32, 128] in bfloat16
        q_f32 = q.to(torch.float32).contiguous()  # [total_q, 32, 128]
        # k_cache, v_cache: [num_pages, 1, 8, 128] -> squeeze dim=1 -> [num_pages, 8, 128]
        k_cache_flat = k_cache.squeeze(1).to(torch.float32).contiguous()  # [num_pages, 8, 128]
        v_cache_flat = v_cache.squeeze(1).to(torch.float32).contiguous()  # [num_pages, 8, 128]

        len_indptr = qo_indptr.shape[0]
        total_q = q_f32.shape[0]
        num_qo_heads = q_f32.shape[1]
        head_dim = q_f32.shape[2]
        num_pages, num_kv_heads, kv_dim = k_cache_flat.shape  # after squeeze(1), num_kv_heads=8, kv_dim=128
        assert num_kv_heads == 8 and kv_dim == 128, "k_cache/v_cache must have shape [num_pages, 8, 128] after squeeze(1)"

        # Allocate outputs
        out_f32 = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=q.device)
        lse_f32 = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=q.device)

        # Launch Triton: one program per (b, q_idx, h)
        grid = (len_indptr, total_q, num_qo_heads)
        _compute_one_triplet_kernel[grid](
            q_f32, k_cache_flat, v_cache_flat,
            qo_indptr, kv_indptr, kv_indices,
            out_f32, lse_f32,
            GQA_RATIO=self.GQA_RATIO,
            MAX_KV=self.MAX_KV,
            sm_scale=float(sm_scale),  # pass as Python float
            num_warps=4, num_stages=2
        )

        # Return output as bfloat16 and lse as float32, matching original signature
        return out_f32.to(torch.bfloat16), lse_f32
