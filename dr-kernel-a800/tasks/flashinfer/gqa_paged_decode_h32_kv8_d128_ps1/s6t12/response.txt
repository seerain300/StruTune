try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def _forward_kernel_bh(
    q_ptr,                  # *fp32, shape [B, H, D], contiguous
    k_ptr,                  # *fp32, shape [N_total, num_kv_heads, D], contiguous
    v_ptr,                  # *fp32, shape [N_total, num_kv_heads, D], contiguous
    kv_indices_ptr,         # *int32, shape [num_kv_indices]
    kv_indptr_ptr,          # *int32, shape [B+1]
    lse_ptr,                # *fp32, shape [B, H]
    out_ptr,                # *fp32, shape [B, H, D]
    B: tl.int32,
    H: tl.int32,
    D: tl.int32,
    num_kv_heads: tl.int32,
    sm_scale: tl.float32,   # scalar, e.g., 1/sqrt(128)
    N_TOTAL: tl.constexpr,  # loop bound, e.g., 128, mask iterations beyond actual_num_tokens
):
    # Program ids: one instance per (b, h)
    b = tl.program_id(0)  # 0..B-1
    h = tl.program_id(1)  # 0..H-1

    # Compute q vector for this (b, h): q[b, h, :]
    # We treat q as [B, H, D] contiguous: index = ((b * H) + h) * D + d
    q_base = q_ptr + (b * H + h) * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_vec[d] = tl.load(q_base + d).to(tl.float32)

    # Gather kv indices for this batch from kv_indptr
    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start  # scalar int32

    # Compute GQA kv head mapping: kv_head = h // (H // num_kv_heads) = h // 4
    kv_head = (h // (H // num_kv_heads)).to(tl.int32)

    # First pass: streaming logsumexp across tokens in base-2
    m = tl.full((), -float("inf"), tl.float32)
    sumexp = tl.full((), 0.0, tl.float32)
    ln2 = 0.6931471805599453  # ln(2)

    for nn in range(0, N_TOTAL):
        # If nn >= actual_num_tokens, skip by checking index validity
        idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)
        valid = start + nn < end
        if valid:
            # Base pointers for k and v for this token and kv_head
            k_base = k_ptr + idx * (num_kv_heads * D) + kv_head * D
            v_base = v_ptr + idx * (num_kv_heads * D) + kv_head * D

            # Load k_vec and compute dot = q_vec · k_vec
            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_base + d).to(tl.float32)

            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]

            logit = dot * sm_scale
            new_m = tl.maximum(m, logit)
            sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit - new_m)
            m = new_m

    # lse for this (b, h): logsumexp(logits_scaled) / ln(2)
    lse_val = m + tl.log(sumexp) / ln2
    tl.store(lse_ptr + b * H + h, lse_val)

    # Second pass: compute softmax per token and accumulate output
    out_base = out_ptr + (b * H + h) * D
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for nn in range(0, N_TOTAL):
        idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)
        valid = start + nn < end
        if valid:
            k_base = k_ptr + idx * (num_kv_heads * D) + kv_head * D
            v_base = v_ptr + idx * (num_kv_heads * D) + kv_head * D

            v_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                v_vec[d] = tl.load(v_base + d).to(tl.float32)

            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_base + d).to(tl.float32)

            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]

            logit = dot * sm_scale
            softmax = tl.exp(logit - m) / (sumexp * ln2)
            out_vec += softmax * v_vec

    # Store accumulated output
    for d in range(0, D):
        tl.store(out_base + d, out_vec[d])
