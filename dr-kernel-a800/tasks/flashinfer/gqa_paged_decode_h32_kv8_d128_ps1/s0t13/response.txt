Diagnosis
- Previous errors were due to Triton kernels using dynamic loops or incorrect pointer arithmetic. Triton does not allow Python for-loops with dynamic bounds. Also, pointer arithmetic like q_ptr + b*32*128 + h*128 is brittle and unnecessary if we pass appropriately shaped tensors and 2D/1D views.
- We must ensure Triton kernels write to output and lse buffers, and that forward launches them.

Plan
1) Precompute logits for all tokens (per batch, per head) using a Triton kernel:
   - Kernel _compute_logits_per_token: grid = (B, Hq, num_tokens). For each (b, h, t), compute q[b, h, :] · k[t, kv_head, :] scaled by sm_scale. This avoids dynamic loops inside kernels.

2) Accumulate output using Triton:
   - Kernel _accumulate_output_atomic_kernel: grid = (B, Hq). Each program handles (b, h). It iterates over all tokens t, computes scaled logits x_t, probability attn_t = exp(x_t) / sum_t exp(x_t), and atomically adds attn_t * v[t, kv_head, :] into out[b, h, :]. This avoids torch softmax.

3) Compute lse per (b, h) using Triton:
   - Kernel _lse_per_bh_kernel: grid = (B, Hq). Each program iterates over tokens t, recomputes q[b, h, :] · k[t, kv_head, :], and performs an online logsumexp update. Stores lse[b, h] = logsumexp / ln(2).

Forward signature
- Accept 7 positional args: q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale, _unused_param (ignore last to be safe).

import math
import torch
import triton
import triton.language as tl


# Kernel 1: compute per-token logits for all (b, h, t)
# Grid: (B, Hq, num_tokens)
# Inputs:
#   q_ptr:      *f32, [B, Hq, D] pointer to Q
#   k_ptr:      *f32, [num_tokens, D] pointer to K gathered per token
#   logits_ptr: *f32, [B*Hq*num_tokens] output buffer (flattened)
@triton.jit
def _compute_logits_per_token(
    q_ptr,          # *f32, [B, Hq, D]
    k_ptr,          # *f32, [num_tokens, D]
    logits_ptr,     # *f32, [B*Hq*num_tokens]
    B: tl.constexpr,
    Hq: tl.constexpr,
    num_tokens,     # i32
    D: tl.constexpr,
    sm_scale: tl.float32,
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    t = tl.program_id(2)
    if t >= num_tokens:
        return

    # Load q[b, h, :] and k[t, :]
    q_base = q_ptr + b * D + h * D
    k_vec = tl.load(k_ptr + t * D + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
    q_vec = tl.load(q_base + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
    acc = tl.sum(q_vec * k_vec, axis=0)
    x = acc * sm_scale
    idx = b * Hq * num_tokens + h * num_tokens + t
    tl.store(logits_ptr + idx, x)


# Kernel 2: accumulate output[b, h, :] across tokens using softmax over scaled logits
# Grid: (B, Hq)
# Inputs:
#   q_ptr:      *f32, [B, Hq, D] pointer to Q
#   k_ptr:      *f32, [num_tokens, D] pointer to K
#   v_ptr:      *f32, [num_tokens, D] pointer to V
#   out_ptr:    *f32, [B*Hq*D] pointer to output (flattened), initialized to 0
@triton.jit
def _accumulate_output_atomic_kernel(
    q_ptr,          # *f32, [B, Hq, D]
    k_ptr,          # *f32, [num_tokens, D]
    v_ptr,          # *f32, [num_tokens, D]
    out_ptr,        # *f32, [B*Hq*D], flattened, initialized to 0
    num_tokens,     # i32
    D: tl.constexpr,
    sm_scale: tl.float32,
):
    pid = tl.program_id(0)
    b = pid // Hq
    h = pid % Hq

    # Running max and sum for softmax
    max_val = -1e20
    sum_exp = 0.0

    # Precompute kv_head for GQA
    gqa_ratio = Hq // 8  # num_qo_heads = 32, num_kv_heads = 8, so 4
    kv_head = h // gqa_ratio

    # Loop over tokens to compute logsumexp over scaled logits
    t = 0
    while t < num_tokens:
        q_base = q_ptr + b * D + h * D
        k_vec = tl.load(k_ptr + t * D + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        q_vec = tl.load(q_base + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        acc = tl.sum(q_vec * k_vec, axis=0)
        x = acc * sm_scale
        if x > max_val:
            sum_exp = sum_exp * tl.exp(max_val - x) + 1.0
            max_val = x
        else:
            sum_exp = sum_exp + tl.exp(x - max_val)
        t += 1

    # Second pass: accumulate output
    for t in range(0, num_tokens):
        q_base = q_ptr + b * D + h * D
        k_vec = tl.load(k_ptr + t * D + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        q_vec = tl.load(q_base + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        acc = tl.sum(q_vec * k_vec, axis=0)
        x = acc * sm_scale
        p = tl.exp(x - max_val) / sum_exp  # softmax probability
        v_vec = tl.load(v_ptr + t * D + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        out_vec = p * v_vec
        # Write out[b, h, :] in flattened memory
        out_base = out_ptr + (b * Hq + h) * D
        idx = tl.arange(0, D)
        tl.store(out_base + idx, out_vec, mask=idx < D)


# Kernel 3: compute lse[b, h] = logsumexp(logits_scaled) / ln(2)
# Grid: (B, Hq)
@triton.jit
def _lse_per_bh_kernel(
    q_ptr,          # *f32, [B, Hq, D]
    k_ptr,          # *f32, [num_tokens, D]
    lse_ptr,        # *f32, [B*Hq]
    num_tokens,     # i32
    D: tl.constexpr,
    sm_scale: tl.float32,
):
    pid = tl.program_id(0)
    b = pid // Hq
    h = pid % Hq

    max_val = -1e20
    sum_exp = 0.0

    gqa_ratio = Hq // 8
    kv_head = h // gqa_ratio

    t = 0
    while t < num_tokens:
        q_base = q_ptr + b * D + h * D
        k_vec = tl.load(k_ptr + t * D + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        q_vec = tl.load(q_base + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0)
        acc = tl.sum(q_vec * k_vec, axis=0)
        x = acc * sm_scale
        if x > max_val:
            sum_exp = sum_exp * tl.exp(max_val - x) + 1.0
            max_val = x
        else:
            sum_exp = sum_exp + tl.exp(x - max_val)
        t += 1

    lse = max_val + tl.log(sum_exp)  # logsumexp of scaled logits
    # Divide by ln(2) to match original
    lse = lse / math.log(2.0)
    tl.store(lse_ptr + pid, lse)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale, _unused_param=None):
        # Ensure CUDA tensors
        assert q.is_cuda and k_cache.is_cuda and v_cache.is_cuda, "All inputs must be CUDA tensors."
        device = q.device

        # Shapes
        B, Hq, D = q.shape
        assert Hq == 32, "num_qo_heads must be 32"
        # k_cache, v_cache are [num_tokens, 1, Hk, D], but original asserts Hk=8 and uses only [num_tokens, D] per token
        # Here, inputs are [num_tokens, D] (as per get_inputs). We rely on that for k_cache/v_cache.
        num_tokens = kv_indices.shape[0]

        # Prepare flattened pointers: q as [B, Hq, D], k/v as [num_tokens, D]
        # Convert to float32 for compute
        q_f32 = q.contiguous().to(torch.float32)
        # k_cache and v_cache are given as [num_tokens, D] already in provided get_inputs; ensure contiguous
        k_flat = k_cache.contiguous().to(torch.float32)
        v_flat = v_cache.contiguous().to(torch.float32)

        # Output buffers
        output = torch.zeros((B, Hq, D), dtype=torch.float32, device=device)
        output_flat = output.view(-1)  # [B*Hq*D]
        lse = torch.empty((B, Hq), dtype=torch.float32, device=device)

        # Launch Triton kernels
        # 1) Compute logits per token
        grid1 = (B, Hq, num_tokens)
        _compute_logits_per_token[grid1](
            q_f32, k_flat, lse_flat, B, Hq, num_tokens, D, float(sm_scale),
        )
        # Note: We actually don't need logits stored; the accumulation kernel recomputes q·k. The above was for clarity,
        # but since Triton requires kernel usage, we can also compute logits in the accumulation kernel (already done in
        # _accumulate_output_atomic_kernel). For simplicity and compilation, we'll rely on recompute.

        # 2) Accumulate output
        grid2 = (B, Hq)
        _accumulate_output_atomic_kernel[grid2](
            q_f32, k_flat, v_flat, output_flat, num_tokens, D, float(sm_scale),
        )

        # 3) Compute lse
        grid3 = (B, Hq)
        _lse_per_bh_kernel[grid3](
            q_f32, k_flat, lse, num_tokens, D, float(sm_scale),
        )

        # Cast output to bfloat16 to match original
        output_bf16 = output.to(torch.bfloat16)

        return output_bf16, lse
