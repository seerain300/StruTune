import math
import torch
import triton
import triton.language as tl


@triton.jit
def softmax_and_attention_single_bh(
    q_ptr,                 # *fp32, pointer to q[b, h] vector, length = HEAD_DIM
    K_ptr,                 # *fp32, pointer to K tokens for this batch, shape [NUM_TOKENS, HEAD_DIM], contiguous
    V_ptr,                 # *fp32, pointer to V tokens for this batch, shape [NUM_TOKENS, HEAD_DIM], contiguous
    OUT_ptr,               # *fp32, output vector for this head [HEAD_DIM]
    LSE_ptr,               # *fp32, single scalar lse for this (b, h), 1-element tensor
    NUM_TOKENS: tl.constexpr,
    HEAD_DIM: tl.constexpr,
    SM_SCALE: tl.float32,       # 1.0 / sqrt(HEAD_DIM) (here 1/sqrt(128))
    LOG2_INVERSE: tl.float32,   # 1.0 / ln(2.0) ≈ 1.4426950408889634
):
    # One program instance per (b, h) where h = program_id(0)
    h = tl.program_id(0)

    # First pass: compute lse = logsumexp(logits_scaled) / ln(2)
    running_max = -float("inf")
    running_sum = 0.0

    for t in range(0, NUM_TOKENS):
        # Load q vector for this head: [HEAD_DIM]
        q_vec = tl.load(q_ptr + h * HEAD_DIM + tl.arange(0, HEAD_DIM))
        # Load k vector for token t: [HEAD_DIM]
        k_vec = tl.load(K_ptr + t * HEAD_DIM + tl.arange(0, HEAD_DIM))
        logits_t = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits_t * SM_SCALE
        # numerically stable update
        running_max = tl.maximum(running_max, scaled)
        # running_sum accumulates exp(scaled - running_max)
        running_sum += tl.exp(scaled - running_max)

    # Compute lse = logsumexp(scaled) / ln(2)
    # logsumexp(scaled) = log(running_sum) + running_max
    lse = tl.log(running_sum) + running_max
    lse = lse * LOG2_INVERSE  # divide by ln(2) via multiply by 1/ln(2)
    # Store scalar lse
    # Since we have only one head per grid dim, index it as 0
    tl.store(LSE_ptr + 0, lse)

    # Second pass: compute output[b, h] = sum_j exp(scaled_j - lse) * V_j
    out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
    for t in range(0, NUM_TOKENS):
        q_vec = tl.load(q_ptr + h * HEAD_DIM + tl.arange(0, HEAD_DIM))
        v_vec = tl.load(V_ptr + t * HEAD_DIM + tl.arange(0, HEAD_DIM))
        k_vec = tl.load(K_ptr + t * HEAD_DIM + tl.arange(0, HEAD_DIM))
        logits_t = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits_t * SM_SCALE
        attn = tl.exp(scaled - lse)  # softmax over tokens with ln(2) scaling already accounted in lse
        out_vec += attn * v_vec

    # Store output vector
    tl.store(OUT_ptr + h * HEAD_DIM + tl.arange(0, HEAD_DIM), out_vec)

- We will implement a Triton kernel that, for each (b, h), computes the lse and the output vector for that head:
  - First pass: loop over tokens, compute scaled logits, maintain running_max and running_sum for numerically stable logsumexp. At the end, lse = log(running_sum) + running_max, then divide by ln(2) by multiplying with 1/ln(2).
  - Second pass: loop again, recompute scaled logits, compute attn = exp(scaled - lse), and accumulate the 128-dim output vector by multiplying attn with v_vec and summing across tokens.
- Host code in ModelNew.forward will:
  - Ensure tensors are on CUDA and bfloat16.
  - Compute num_tokens per batch from kv_indptr.
  - Gather per-batch K_t and V_t by indexing k_cache and v_cache with kv_indices[start:start+num_tokens_b] (torch.gather). Cast to fp32 and make contiguous.
  - Launch the Triton kernel for each (b, h), passing q_vec, K_t_h, V_t_h, output[b, h], and lse[b, h].
  - Cast output to bfloat16 and return.

This ensures all computation is done in Triton kernels; the host code only performs allocations, .contiguous(), and launches kernels. It avoids previous errors like uninitialized variables and broadcasting issues by using explicit 1D pointer arithmetic and no torch reductions in the host code.

Final Triton implementation (ModelNew)

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def softmax_and_attention_single_bh(
    q_ptr,                 # *fp32, pointer to q[b, h] vector, length = HEAD_DIM
    K_ptr,                 # *fp32, pointer to K tokens for this batch, shape [NUM_TOKENS, HEAD_DIM], contiguous
    V_ptr,                 # *fp32, pointer to V tokens for this batch, shape [NUM_TOKENS, HEAD_DIM], contiguous
    OUT_ptr,               # *fp32, output vector for this head [HEAD_DIM]
    LSE_ptr,               # *fp32, single scalar lse for this (b, h), 1-element tensor
    NUM_TOKENS: tl.constexpr,
    HEAD_DIM: tl.constexpr,
    SM_SCALE: tl.float32,       # 1.0 / sqrt(HEAD_DIM) (here 1/sqrt(128))
    LOG2_INVERSE: tl.float32,   # 1.0 / ln(2.0) ≈ 1.4426950408889634
):
    # One program instance per (b, h) where h = program_id(0)
    h = tl.program_id(0)

    # First pass: compute lse = logsumexp(logits_scaled) / ln(2)
    running_max = -float("inf")
    running_sum = 0.0

    for t in range(0, NUM_TOKENS):
        # Load q vector for this head: [HEAD_DIM]
        q_vec = tl.load(q_ptr + h * HEAD_DIM + tl.arange(0, HEAD_DIM))
        # Load k vector for token t: [HEAD_DIM]
        k_vec = tl.load(K_ptr + t * HEAD_DIM + tl.arange(0, HEAD_DIM))
        logits_t = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits_t * SM_SCALE
        # numerically stable update
        running_max = tl.maximum(running_max, scaled)
        running_sum += tl.exp(scaled - running_max)

    # Compute lse = logsumexp(scaled) / ln(2)
    lse = tl.log(running_sum) + running_max
    lse = lse * LOG2_INVERSE  # divide by ln(2)
    # Store scalar lse at LSE_ptr[0]
    tl.store(LSE_ptr + 0, lse)

    # Second pass: compute output[b, h] = sum_j exp(scaled_j - lse) * V_j
    out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
    for t in range(0, NUM_TOKENS):
        q_vec = tl.load(q_ptr + h * HEAD_DIM + tl.arange(0, HEAD_DIM))
        v_vec = tl.load(V_ptr + t * HEAD_DIM + tl.arange(0, HEAD_DIM))
        k_vec = tl.load(K_ptr + t * HEAD_DIM + tl.arange(0, HEAD_DIM))
        logits_t = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits_t * SM_SCALE
        attn = tl.exp(scaled - lse)  # softmax over tokens
        out_vec += attn * v_vec

    # Store output vector
    tl.store(OUT_ptr + h * HEAD_DIM + tl.arange(0, HEAD_DIM), out_vec)

        assert q.dtype == torch.bfloat16 and k_cache.dtype == torch.bfloat16 and v_cache.dtype == torch.bfloat16
