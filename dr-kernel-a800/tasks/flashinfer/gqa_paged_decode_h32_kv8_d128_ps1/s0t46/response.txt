    def _compute_logits_bh_kernel(
        q_ptr,           # *f32, [B, Hq, D]
        k_ptr,           # *f32, [NUM_TOKS, Hk, D]
        logits_ptr,      # *f32, [B, Hq, NUM_TOKS]
        B: tl.constexpr,
        Hq: tl.constexpr,
        NUM_TOKS: tl.constexpr,  # number of tokens per batch element (runtime int)
        D: tl.constexpr,
        Hk: tl.constexpr,
        gqa_ratio: tl.constexpr,
    ):
        # Grid is (B, Hq)
        b = tl.program_id(0)
        h = tl.program_id(1)
        if b >= B or h >= Hq:
            return
        # Base offset for q[b, h, :]
        q_base = b * Hq * D + h * D
        # For each token t
        t = 0
        while t < NUM_TOKS:
            kv_head = h // gqa_ratio  # GQA mapping: query head to kv head
            # k base for (t, kv_head, :)
            k_base = t * Hk * D + kv_head * D
            dot_acc = 0.0
            # Loop over D dimension
            d = 0
            while d < D:
                q_val = tl.load(q_ptr + q_base + d)
                k_val = tl.load(k_ptr + k_base + d)
                dot_acc += q_val * k_val
                d += 1
            # Store logits[b, h, t]
            tl.store(logits_ptr + b * Hq * NUM_TOKS + h * NUM_TOKS + t, dot_acc)
            t += 1

    @triton.jit
    def _lse_per_bh_kernel(
        logits_ptr,       # *f32, [B, Hq, NUM_TOKS]
        lse_ptr,          # *f32, [B, Hq]
        B: tl.constexpr,
        Hq: tl.constexpr,
        NUM_TOKS: tl.constexpr,
    ):
        # Grid is (B, Hq)
        b = tl.program_id(0)
        h = tl.program_id(1)
        if b >= B or h >= Hq:
            return
        base = b * Hq * NUM_TOKS + h * NUM_TOKS
        m = -float("inf")
        t = 0
        while t < NUM_TOKS:
            val = tl.load(logits_ptr + base + t)
            if val > m:
                m = val
            t += 1
        sum_exp = 0.0
        t = 0
        while t < NUM_TOKS:
            val = tl.load(logits_ptr + base + t)
            sum_exp += tl.exp(val - m)
            t += 1
        lse = m + tl.log(sum_exp)
        # Divide by ln(2)
        lse = lse / 0.6931471805599453  # 1 / ln(2)
        tl.store(lse_ptr + b * Hq + h, lse)
