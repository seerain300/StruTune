import math

    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

    def _gqa_attention_kernel(
        q_ptr,           # *f32, shape [B, num_qo_heads, HEAD_DIM]
        k_ptr, v_ptr,    # *bf16 or *f16, shape [num_pages, 1, num_kv_heads, HEAD_DIM]
        kv_indptr_ptr,   # *i32, shape [B+1]
        out_ptr,         # *f32, shape [B, num_qo_heads, HEAD_DIM]
        lse_ptr,         # *f32, shape [B, num_qo_heads]

        num_qo_heads: tl.constexpr,
        num_kv_heads: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        sm_scale: tl.constexpr,
        gqa_ratio: tl.constexpr,
        ln2: tl.constexpr,  # 1.4426950408889634 (1 / ln(2))
    ):
        # One program per (b, h)
        pid = tl.program_id(0)
        b = pid // num_qo_heads
        h = pid % num_qo_heads
        if b >= B or h >= num_qo_heads:
            return

        kv_start = tl.load(kv_indptr_ptr + b)        # i32
        kv_end = tl.load(kv_indptr_ptr + b + 1)     # i32
        num_tokens = kv_end - kv_start              # i32 scalar (>= 0)

        # Load q vector for this (b, h) as float32
        q_offset = b * num_qo_heads * HEAD_DIM + h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_offset)  # [HEAD_DIM] f32

        # GQA mapping
        kv_head = h // gqa_ratio  # int

        # Initialize output vector and LSE
        out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
        lse = -float('inf')  # f32 scalar

        # Iterate over tokens in this batch
        # We loop in Python over a small range since num_tokens is a runtime value but Triton can handle this pattern.
        # Note: Triton supports such scalar-controlled loops; for very large num_tokens, you could block, but here workloads vary.
        for t in range(0, num_tokens):
            # Compute index for k/v
            # kv_indptr[b] is the starting token index for this batch; t is the offset within [0, num_tokens)
            idx = kv_start + t  # i32

            # Build offsets for k/v vectors: [num_kv_heads, HEAD_DIM]
            # Note: k_ptr layout is [num_pages, 1, num_kv_heads, HEAD_DIM] so stride for kv head is HEAD_DIM * 1 = HEAD_DIM
            # We load the [num_kv_heads, HEAD_DIM] slab for this idx, then select kv_head.
            # However, to keep it simple and efficient, we precompute offsets for the specific kv_head.
            # Address for k at (idx, 0, kv_head, :)
            k_offset = idx * (num_kv_heads * HEAD_DIM) + 0 * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM
            v_offset = idx * (num_kv_heads * HEAD_DIM) + 0 * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM

            # Load k_t and v_t as float32 (cast inside kernel)
            k_vec = tl.load(k_ptr + k_offset)  # [HEAD_DIM], bfloat16 -> float32
            v_vec = tl.load(v_ptr + v_offset)  # [HEAD_DIM], bfloat16 -> float32

            for d in range(0, HEAD_DIM):

                lse = maxv + tl.log(1.0 + tl.exp(minv - maxv))

            # Attention weight
            attn = tl.exp(scaled - lse)  # f32 scalar

            # Accumulate output vector
            for d in range(0, HEAD_DIM):
                out_vec[d] += attn * v_vec[d]

        # Store results
        out_offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        tl.store(out_ptr + out_offset, out_vec)  # out_ptr is float32

        # Store LSE divided by ln(2)
        lse_div = lse * ln2
        tl.store(lse_ptr + b * num_qo_heads + h, lse_div)  # lse_ptr is float32
