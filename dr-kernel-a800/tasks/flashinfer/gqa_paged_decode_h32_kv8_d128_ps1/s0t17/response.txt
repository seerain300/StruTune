import torch
import triton
import triton.language as tl


@triton.jit
def _compute_logits_bh_kernel(
    q_ptr,            # *f32, [B*Hq*D] contiguous
    k_ptr,            # *f32, [num_tokens * Hk * D] contiguous
    logits_ptr,       # *f32, [B*Hq*num_tokens] contiguous
    B: tl.constexpr,  # int
    Hq: tl.constexpr, # int
    D: tl.constexpr,  # int
    Hk: tl.constexpr, # int
    num_tokens,       # i32 (runtime)
    sm_scale,         # f32
):
    # program ids for batch and head
    b = tl.program_id(0)
    h = tl.program_id(1)

    # base offset for q[b, h, :]
    base_q = (b * Hq + h) * D
    # kv head for GQA
    kv_h = h // (Hq // Hk)

    # iterate tokens with scalar while loop
    t = 0
    while t < num_tokens:
        # base offset for k[t, kv_h, :]
        base_k = t * Hk * D + kv_h * D
        acc = 0.0
        i = 0
        while i < D:

            k_val = tl.load(k_ptr + base_k + i)
            acc += q_val * k_val
            i += 1
        # store scaled logits
        tl.store(logits_ptr + b * Hq * num_tokens + h * num_tokens + t, acc * sm_scale)
        t += 1


@triton.jit
def _lse_per_bh_kernel(
    logits_ptr,    # *f32, [B*Hq*num_tokens] contiguous
    lse_ptr,       # *f32, [B*Hq] contiguous
    B: tl.constexpr,  # int
    Hq: tl.constexpr, # int
    num_tokens,       # i32
):
    b = tl.program_id(0)
    h = tl.program_id(1)
    ptr = logits_ptr + b * Hq * num_tokens + h * num_tokens
    # online logsumexp
    m = -float('inf')
    s = 0.0
    t = 0
    while t < num_tokens:
        x = tl.load(ptr + t)
        if x > m:
            s = s * tl.exp(m - x) + 1.0
            m = x
        else:
            s += tl.exp(x - m)
        t += 1
    # divide by ln(2)
    lse = m + tl.log(s) / math.log(2.0)
    tl.store(lse_ptr + b * Hq + h, lse)


@triton.jit
def _accumulate_output_bh_kernel(
    q_ptr,            # *f32, [B*Hq*D] contiguous
    k_ptr,            # *f32, [num_tokens * Hk * D] contiguous
    v_ptr,            # *f32, [num_tokens * Hk * D] contiguous

    B: tl.constexpr,  # int
    Hq: tl.constexpr, # int
    D: tl.constexpr,  # int
    Hk: tl.constexpr, # int
    num_tokens,       # i32
    sm_scale,         # f32
):

    h = tl.program_id(1)
    # base offsets
    base_q = (b * Hq + h) * D
    kv_h = h // (Hq // Hk)

    # first pass: compute sum_exp = sum_t exp(logits_scaled[t]) for logsumexp
    sum_exp = 0.0
    t = 0
    while t < num_tokens:
        base_k = t * Hk * D + kv_h * D
        acc = 0.0
        i = 0
        while i < D:
            q_val = tl.load(q_ptr + base_q + i)
            k_val = tl.load(k_ptr + base_k + i)
            acc += q_val * k_val
            i += 1
        x = acc * sm_scale
        sum_exp += tl.exp(x)
        t += 1

    # second pass: accumulate output[b, h, :] += softmax(x) * v[token, kv_h, :]
    t = 0
    while t < num_tokens:
        base_k = t * Hk * D + kv_h * D
        base_v = t * Hk * D + kv_h * D

        acc = 0.0
        i = 0
        while i < D:
            q_val = tl.load(q_ptr + base_q + i)
            k_val = tl.load(k_ptr + base_k + i)
            acc += q_val * k_val
            i += 1

        attn = tl.exp(x) / sum_exp

        # v[token, kv_h, :]
        i = 0
        while i < D:

            tl.store(output_ptr + (b * Hq + h) * D + i, tl.load(output_ptr + (b * Hq + h) * D + i) + v_val * attn)
            i += 1
        t += 1
