- For each (batch b, query head h):

- It performs this in Python loops. We need to implement this in Triton.

Plan for Triton version:
- Launch one Triton program per (b, h) pair. This ensures correct per-head computation and avoids inter-program reductions.
- Avoid any 2D tensor constructs inside Triton. Use simple scalar and 1D vector loads.
- Use a while loop in Triton to iterate over tokens. Triton supports while loops with scalar/tensor conditions.
- In the first pass, iterate over tokens and accumulate sum_exp = sum(exp((q·k) * sm_scale)). This is needed to compute lse = log2(sum_exp) = log(sum_exp) / ln(2).
- In the second pass, iterate again over tokens, compute dot = q·k, attn = exp((dot - lse) * sm_scale), and accumulate out_vec += attn * v_row. Finally, store out_vec as bfloat16 and lse as fp32.

Key implementation details:
- Ensure all inputs to Triton are on CUDA and contiguous.
- Use fp32 for math inside Triton; store outputs in bfloat16.
- Compute base pointer offsets using Triton tensors and scalar math; avoid mixing Python ints incorrectly.
- Handle general batch_size, num_qo_heads=32, num_kv_heads=8, head_dim=128, GQA ratio 4.

import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def fused_gqa_row_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [Np, 1, K, D]
    v_ptr,            # *bf16, [Np, 1, K, D]
    indptr_ptr,       # *int32, [B+1]
    indices_ptr,      # *int32, [num_tokens_total]
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *fp32, [B, H]
    sm_scale,         # fp32 scalar
    B: tl.constexpr,  # batch size (not strictly needed as runtime, but ok as const)
    H: tl.constexpr,  # 32
    D: tl.constexpr,  # 128
    K: tl.constexpr,  # 8
    gqa_ratio: tl.constexpr,  # H // K == 4
):
    # program ids for (batch, head)
    pid_b = tl.program_id(0)  # int32
    pid_h = tl.program_id(1)  # int32

    # Load q vector for this (b, h) in fp32
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Load indptr[start, end] for this batch element
    start = tl.load(indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start  # scalar int32

    # Pass 1: compute sum_exp = sum(exp((q·k) * sm_scale)) across tokens
    sum_exp = 0.0  # scalar fp32
    t = 0
    while t < num_tokens:
        idx = tl.load(indices_ptr + start + t).to(tl.int32)  # scalar token index
        kv_head = pid_h // gqa_ratio  # 0..7

        # Load K row for this token and head (1D vector) from k_ptr
        k_row_base = idx * (K * D) + kv_head * D
        k_row_ptr = k_ptr + k_row_base
        k_row = tl.load(k_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]

        # dot(q_vec, k_row)
        dot = tl.sum(q_vec * k_row, axis=0)  # scalar fp32
        sum_exp += tl.exp(dot * sm_scale)
        t += 1

    # lse in base-2: log2(sum_exp) = log(sum_exp) / ln(2)
    lse_acc = tl.log(sum_exp) * 1.4426950408889634  # 1/ln(2)

    # Pass 2: compute output vector out[b, h, :] (fp32), then cast to bf16
    out_vec = tl.zeros((D,), dtype=tl.float32)
    t = 0
    while t < num_tokens:
        idx = tl.load(indices_ptr + start + t).to(tl.int32)
        kv_head = pid_h // gqa_ratio

        # Load K and V rows for this token and head
        k_row_base = idx * (K * D) + kv_head * D
        v_row_base = idx * (K * D) + kv_head * D

        k_row_ptr = k_ptr + k_row_base
        v_row_ptr = v_ptr + v_row_base

        k_row = tl.load(k_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]
        v_row = tl.load(v_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]

        # Compute logits_scaled and attention
        dot = tl.sum(q_vec * k_row, axis=0)  # scalar fp32
        attn = tl.exp((dot - lse_acc) * sm_scale)
        out_vec += attn * v_row
        t += 1

    # Store output as bfloat16
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))

    # Store lse (base-2)
    lse_offset = pid_b * H + pid_h
    tl.store(lse_ptr + lse_offset, lse_acc)
