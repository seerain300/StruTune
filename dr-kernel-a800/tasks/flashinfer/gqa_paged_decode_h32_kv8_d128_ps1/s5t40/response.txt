try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def gqa_row_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [Np, 1, K, D]
    v_ptr,            # *bf16, [Np, 1, K, D]
    kv_indptr_ptr,    # *int32, [B+1]
    kv_indices_ptr,   # *int32, [Np]  (though we only need indptr here)
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *float32, [B, H]
    sm_scale,         # float32 scalar
    B: tl.constexpr,  # unused but kept for clarity
    H: tl.constexpr,  # query heads
    D: tl.constexpr,  # head_dim
    K: tl.constexpr,  # kv_heads
    gqa_ratio: tl.constexpr,  # H//K
    MAX_TOKENS: tl.constexpr, # fixed iteration bound
):
    # One program per (b, h)
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)

    # Load q vector for this (b, h)
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Load indptr[start, end] for this batch element
    start = tl.load(kv_indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Pass 1: compute sum_exp = sum(exp((q·k) * sm_scale)) across tokens
    sum_exp = 0.0  # scalar float32
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        # For tokens within [start, end), we only need the token index
        idx = start + t  # guaranteed < end by guard
        kv_head = pid_h // gqa_ratio  # map query head to KV head under GQA

        # Compute pointers for k[idx, kv_head, :] and v[idx, kv_head, :]
        # Shapes: k is [Np, 1, K, D]; we ignore the 1-size dim and use idx for batch-like tokens
        k_base = k_ptr + idx * K * D  # idx selects the token within this batch's cache
        v_base = v_ptr + idx * K * D

        k_vec = tl.load(k_base + kv_head * D + tl.arange(0, D)).to(tl.float32)
        v_vec = tl.load(v_base + kv_head * D + tl.arange(0, D)).to(tl.float32)

        dot = tl.sum(q_vec * k_vec, axis=0)  # scalar
        sum_exp += tl.exp(dot * sm_scale)

    # lse in base-2
    lse_val = tl.log(sum_exp) / tl.log(2.0)  # scalar float32
    tl.store(lse_ptr + pid_b * H + pid_h, lse_val)

    # Pass 2: accumulate output vector
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = start + t
        kv_head = pid_h // gqa_ratio

        k_base = k_ptr + idx * K * D
        v_base = v_ptr + idx * K * D

        k_vec = tl.load(k_base + kv_head * D + tl.arange(0, D)).to(tl.float32)
        v_vec = tl.load(v_base + kv_head * D + tl.arange(0, D)).to(tl.float32)

        dot = tl.sum(q_vec * k_vec, axis=0)  # scalar
        attn = tl.exp((dot - lse_val) * sm_scale)  # scalar
        out_vec += attn * v_vec  # elementwise vector add across D

    # Store output as bfloat16
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))

):

    k_row_ptrs = k_ptr + (start + offs) * (K * D) + kv_head * D  # [MAX_TOKENS]
    v_row_ptrs = v_ptr + (start + offs) * (K * D) + kv_head * D  # [MAX_TOKENS]

    k_rows = tl.load(k_row_ptrs + tl.arange(0, D), mask=mask, other=0.0).to(tl.float32)  # [MAX_TOKENS, D]

    dot = tl.sum(q_vec[None, :] * k_rows, axis=1)  # [MAX_TOKENS]
