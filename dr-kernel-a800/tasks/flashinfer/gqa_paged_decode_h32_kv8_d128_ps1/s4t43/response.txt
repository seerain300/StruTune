import torch
import triton
import triton.language as tl


# Triton kernel: computes lse and output for a single (b, h)
# Inputs:
#   OUT_ptr: pointer to output[b, h, :] (float32, length head_dim)
#   LSE_ptr: pointer to lse[b, h] (float32, scalar)
#   Q_ptr: pointer to q[b, h, :] (float32, length head_dim)
#   K_ptr: pointer to K_t flattened (float32, length num_tokens*head_dim)
#   V_ptr: pointer to V_t flattened (float32, length num_tokens*head_dim)
# Runtime args:
#   num_tokens: number of tokens for this batch
#   sm_scale: scaling factor (float32)
#   LOG2_INVERSE: 1 / ln(2) (float32)
# Constexprs:
#   head_dim: compile-time constant (e.g., 128)
#   KV_HEAD: compile-time constant, kv_head index (e.g., h // gqa_ratio)
@triton.jit
def softmax_and_attention_single_bh(
    OUT_ptr, LSE_ptr,
    Q_ptr, K_ptr, V_ptr,
    num_tokens: tl.constexpr,      # number of tokens in this batch
    sm_scale: tl.constexpr,        # scaling factor
    LOG2_INVERSE: tl.constexpr,    # 1 / ln(2)
    head_dim: tl.constexpr,        # e.g., 128
    KV_HEAD: tl.constexpr,         # e.g., h // 4
):
    # This kernel handles one (b, h) pair per launch; grid will be set to (B, H) on host.
    # Since grid here is (1,1), we can't index by b/h directly; we expect forward to launch
    # this kernel with a 2D grid and pass LSE/OUT corresponding to that (b,h).
    # However, Triton doesn't allow indexing from program_id here easily; instead, we rely
    # on host passing the correct OUT_ptr/LSE_ptr for each (b,h).
    # To make it work: we simply assume OUT_ptr/LSE_ptr correspond to the current b,h as
    # specified by the launch. Triton requires static loops; we use while loop over tokens.
    running_max = -float("inf")
    running_sum = 0.0

    t = 0
    while t < num_tokens:
        # Compute dot(q, K_row)
        k_row = K_ptr + t * head_dim  # base pointer for this token's row
        v_row = V_ptr + t * head_dim  # base pointer for this token's row

        # Load q vector (head_dim elements)
        i = tl.arange(0, head_dim)
        q_vec = tl.load(Q_ptr + i)

        # Load k_row and v_row as vectors
        k_vec = tl.load(k_row + i)
        v_vec = tl.load(v_row + i)

        # logits = dot(q, k_row)
        logits = tl.sum(q_vec * k_vec, axis=0)
        logits_scaled = logits * sm_scale

        # numerically stable logsumexp update
        running_max_new = tl.maximum(running_max, logits_scaled)
        running_sum = running_sum * tl.exp(running_max - running_max_new) + tl.exp(logits_scaled - running_max_new)
        running_max = running_max_new

        t += 1

    # Compute lse = log(sum) + max, then divide by ln(2) via multiply with 1/ln(2)
    lse_val = tl.log(running_sum) + running_max
    lse_val = lse_val * LOG2_INVERSE

    # Store lse scalar (host expects [B, 32] vector)
    tl.store(LSE_ptr, lse_val)

    # Second pass: compute attention and accumulate output
    t = 0
    while t < num_tokens:
        k_row = K_ptr + t * head_dim
        v_row = V_ptr + t * head_dim

        q_vec = tl.load(Q_ptr + i)
        k_vec = tl.load(k_row + i)
        v_vec = tl.load(v_row + i)

        logits = tl.sum(q_vec * k_vec, axis=0)
        logits_scaled = logits * sm_scale

        attn = tl.exp(logits_scaled - lse_val)  # scalar

        # out_vec += attn * v_vec
        out_vec = tl.zeros((head_dim,), dtype=tl.float32)
        out_vec = out_vec + attn * v_vec
        tl.store(OUT_ptr + i, out_vec, mask=None)

        t += 1

        super().__init__()

        v_cache: same shape as k_cache

        """

                output[b].zero_()

                # Gather K_t and V_t for this batch and kv_head: [num_tokens_b, head_dim]
                # Note: k_cache and v_cache are [num_pages, num_kv_heads, head_dim].
                # Use indices to pick the right token's KV block.
                k_rows = k_cache[token_indices, kv_head, :]  # bfloat16
                v_rows = v_cache[token_indices, kv_head, :]  # bfloat16

                # Cast to float32 for Triton and make contiguous
                K_t = k_rows.to(torch.float32).contiguous()  # [num_tokens_b, head_dim]
                V_t = v_rows.to(torch.float32).contiguous()  # [num_tokens_b, head_dim]

                # Q vector for this head
                q_vec = q[b, h, :].to(torch.float32).contiguous()  # [head_dim]

                # Output vector and lse scalar for this (b, h)
                out_vec = torch.empty((self.head_dim,), dtype=torch.float32, device=device)
                lse_scalar = torch.empty((), dtype=torch.float32, device=device)

                # Launch Triton kernel for this (b, h)
                # Grid: Triton launch grid is not directly used to index outputs; we pass OUT_ptr/LSE_ptr for (b,h).
                # Use a single program instance, but we'll set meta parameters.
                softmax_and_attention_single_bh[(1,)](
                    out_vec, lse_scalar,
                    q_vec, K_t.view(-1), V_t.view(-1),
                    num_tokens_b,
                    float(sm_scale),
                    1.4426950408889634,  # LOG2_INVERSE = 1 / ln(2)
                    self.head_dim,       # constexpr
                    kv_head,             # constexpr kv head index
                )

                # Write results
                output[b, h, :] = out_vec
                lse[b, h] = lse_scalar

        # Cast output to bfloat16 as expected by the reference
        output = output.to(torch.bfloat16)
        return output, lse
