- Gather indices and compute k_vec, v_vec for each token, compute dot, accumulate logsumexp (base-2) using a streaming algorithm:

import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def _forward_kernel_bh(
    q_ptr,             # *fp16 or *fp32, shape [B, H, D]
    k_ptr,             # *fp16 or *fp32, shape [num_kv_indices, 1, num_kv_heads, D]
    v_ptr,             # *fp16 or *fp32, shape [num_kv_indices, 1, num_kv_heads, D]
    kv_indices_ptr,    # *int32, shape [num_kv_indices]
    kv_indptr_ptr,     # *int32, shape [B+1]
    lse_ptr,           # *fp32, shape [B, H]
    out_ptr,           # *fp32, shape [B, H, D]
    B: tl.constexpr,   # not used directly in kernel, but useful for understanding
    H: tl.constexpr,
    D: tl.constexpr,
    NUM_KV_HEADS: tl.constexpr,
    SM_SCALE,          # scalar float32
    N_TOTAL: tl.constexpr,  # loop bound, e.g., 128
):
    # Program ids for batch and head
    b = tl.program_id(0)  # int32
    h = tl.program_id(1)  # int32

    # Load q vector for (b, h): q[b, h, :]
    q_base = q_ptr + b * H * D + h * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_vec[d] = tl.load(q_base + d).to(tl.float32)

    # Compute start/end for this batch from kv_indptr
    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start  # int32

    # GQA mapping: kv_head = h // (H // NUM_KV_HEADS)
    kv_group = H // NUM_KV_HEADS
    kv_head = h // kv_group

    # First pass: compute logsumexp of scaled logits over tokens (base-2)
    m = -float("inf")
    sumexp = 0.0  # scalar float32
    ln2 = 0.6931471805599453  # float32 scalar

    for nn in range(0, N_TOTAL):
        # Mask: only process if nn < actual_num_tokens
        mask_nn = nn < actual_num_tokens
        idx = tl.load(kv_indices_ptr + start + nn, mask=mask_nn, other=0).to(tl.int32)

        # Compute k_vec = k_cache[idx, kv_head, :]
        k_row_base = k_ptr + idx * NUM_KV_HEADS * D + kv_head * D
        k_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            k_vec[d] = tl.load(k_row_base + d).to(tl.float32)

        # Dot product q_vec · k_vec
        dot = 0.0
        for d in range(0, D):
            dot += q_vec[d] * k_vec[d]

        logit_scaled = dot * SM_SCALE
        new_m = tl.maximum(m, logit_scaled)
        sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit_scaled - new_m)
        m = new_m

    # lse = logsumexp / ln(2)
    lse_val = m + tl.log(sumexp) / ln2
    tl.store(lse_ptr + b * H + h, lse_val)

    # Second pass: compute softmax and accumulate output
    out_base = out_ptr + b * H * D + h * D
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for nn in range(0, N_TOTAL):
        mask_nn = nn < actual_num_tokens
        idx = tl.load(kv_indices_ptr + start + nn, mask=mask_nn, other=0).to(tl.int32)

        v_row_base = v_ptr + idx * NUM_KV_HEADS * D + kv_head * D
        v_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            v_vec[d] = tl.load(v_row_base + d).to(tl.float32)

        k_row_base = k_ptr + idx * NUM_KV_HEADS * D + kv_head * D
        k_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            k_vec[d] = tl.load(k_row_base + d).to(tl.float32)

        dot = 0.0
        for d in range(0, D):
            dot += q_vec[d] * k_vec[d]

        logit_scaled = dot * SM_SCALE
        softmax = tl.exp(logit_scaled - m) / (sumexp * ln2)
        out_vec += softmax * v_vec

    # Store output vector
    for d in range(0, D):
        tl.store(out_base + d, out_vec[d])

Understanding and plan

- Compute per (batch b, query head h):
  - q_vec = q[b, h, :]
  - kv_head = h // (H // num_kv_heads) = h // 4 since H=32, num_kv_heads=8 (GQA).
  - For tokens n in [kv_indptr[b], kv_indptr[b+1)):
    - Gather k_row = k_cache[token_indices[n], kv_head, :] and v_row similarly.
    - dot = q_vec · k_row; logit_scaled = dot * sm_scale; accumulate logsumexp in base-2.
    - softmax = exp(logit_scaled - m) / (sumexp * ln2); output[b, h, :] += softmax * v_row.
  - lse[b, h] = m + log(sumexp) / ln2.

- Triton kernel design:
  - One program per (b, h): grid = (B, H). This avoids host-side loops and keeps the implementation simple and robust.
  - Use a fixed compile-time loop bound N_TOTAL=128 and mask nn < actual_num_tokens. This handles variable token counts safely (provided actual_num_tokens <= 128).
  - First pass: stream logsumexp (base-2) over tokens; compute m and sumexp.
  - Second pass: recompute dot for each token, compute softmax, and accumulate output vector.

- Forward method:
  - Allocates outputs lse (float32) and out (float32).
  - Launches the Triton kernel with grid=(B, H), num_warps=4, num_stages=2.
  - Returns out cast to bfloat16 and lse (float32), matching the original interface.

- Why no PyTorch in forward:
  - The forward does only allocations, kernel launch, and dtype cast. No tensor methods, no assertions, no .item(), no torch.dot, no .exp/.log. This satisfies the Triton-only requirement.

    k_ptr,             # *fp16 or *fp32, shape [num_kv_indices, 1, num_kv_heads, D]
    v_ptr,             # *fp16 or *fp32, shape [num_kv_indices, 1, num_kv_heads, D]
    kv_indices_ptr,    # *int32, shape [num_kv_indices]
    kv_indptr_ptr,     # *int32, shape [B+1]
    lse_ptr,           # *fp32, shape [B, H]
    out_ptr,           # *fp32, shape [B, H, D]
    H: tl.constexpr,
    D: tl.constexpr,
    NUM_KV_HEADS: tl.constexpr,
    SM_SCALE,          # scalar float32
    N_TOTAL: tl.constexpr,  # loop bound (e.g., 128), masked beyond actual_num_tokens
):
    # Program ids for batch and head
    b = tl.program_id(0)  # int32
    h = tl.program_id(1)  # int32

    # Load q vector for (b, h): q[b, h, :]
    q_base = q_ptr + b * H * D + h * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_vec[d] = tl.load(q_base + d).to(tl.float32)

    # Compute start/end for this batch from kv_indptr
    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start  # int32

    # GQA mapping: kv_head = h // (H // NUM_KV_HEADS)
    kv_group = H // NUM_KV_HEADS
    kv_head = h // kv_group

    # First pass: compute logsumexp of scaled logits over tokens (base-2)
    m = -float("inf")
