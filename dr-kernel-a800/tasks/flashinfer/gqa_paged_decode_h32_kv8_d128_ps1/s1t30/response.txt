  - Computes num_tokens from kv_indptr[b] and kv_indptr[b+1].

import math

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


if TRITON_AVAILABLE:

    def _attention_gqa_kernel(
        q_ptr,           # *f32, shape [B, num_qo_heads, HEAD_DIM]
        k_ptr, v_ptr,    # *f16 or *bf16, shape [num_pages, 1, num_kv_heads, HEAD_DIM]
        kv_indptr_ptr,   # *i32, shape [B+1]
        out_ptr,         # *f32, shape [B, num_qo_heads, HEAD_DIM]
        lse_ptr,         # *f32, shape [B, num_qo_heads]
        B,               # int32
        num_qo_heads: tl.constexpr,
        num_kv_heads: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        sm_scale: tl.constexpr,
        gqa_ratio: tl.constexpr,
        ln2: tl.constexpr,  # 1 / log(2.0)
        MAX_TOKENS: tl.constexpr,
    ):
        # One program per (b, h)
        pid = tl.program_id(0)
        b = pid // num_qo_heads
        h = pid % num_qo_heads
        if b >= B or h >= num_qo_heads:
            return

        # Compute token range for this batch
        kv_start = tl.load(kv_indptr_ptr + b)        # i32
        kv_end = tl.load(kv_indptr_ptr + b + 1)     # i32
        num_tokens = kv_end - kv_start              # i32 scalar

        # Load q vector for this (b, h) as float32
        q_offset = b * num_qo_heads * HEAD_DIM + h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_offset)  # [HEAD_DIM] f32

        # Initialize output vector and lse
        out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
        lse = tl.full((), -float("inf"), dtype=tl.float32)

        # Iterate over tokens up to MAX_TOKENS; for real workloads, num_tokens <= MAX_TOKENS
        for t in range(0, MAX_TOKENS):
            # If we've covered all tokens, break. Triton doesn't support break well; mask handles it.
            if t >= num_tokens:
                break

            # Compute the cache index v = kv_start + t
            v_idx = kv_start + t  # int32 scalar

            # Compute kv_head based on GQA mapping: kv_head = h // gqa_ratio
            kv_head = h // gqa_ratio

            # Build pointers for k_t and v_t: k_ptr has shape [num_pages, 1, num_kv_heads, HEAD_DIM]
            # Address: ((v_idx * (1*num_kv_heads*HEAD_DIM)) + (kv_head*HEAD_DIM)) -> linear index
            # Since k_ptr is [num_pages, 1, num_kv_heads, HEAD_DIM] contiguous:
            # offset = v_idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM
            k_off = v_idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM
            v_off = v_idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM

            # Load k_t and v_t as float32 for compute
            k_t = tl.load(k_ptr + k_off).to(tl.float32)  # [HEAD_DIM]
            v_t = tl.load(v_ptr + v_off).to(tl.float32)  # [HEAD_DIM]

            # Compute logits = q · k_t
            logits = 0.0
            for d in range(0, HEAD_DIM):
                logits += q_vec[d] * k_t[d]

            logits_scaled = logits * sm_scale

            # Numerically-stable LSE update
            # If lse == -inf, set lse = logits_scaled
            # Else: new_LSE = max(lse, logits_scaled) + log(1 + exp(-abs(lse - logits_scaled)))
            # We can implement piecewise for simplicity:
            new_lse = tl.where(
                lse == -float("inf"),
                logits_scaled,
                tl.log(1.0 + tl.exp(-tl.abs(lse - logits_scaled))) + tl.maximum(lse, logits_scaled),
            )
            lse = new_lse

            # attention = exp(logits_scaled - lse)
            attn = tl.exp(logits_scaled - lse)

            # Accumulate output: out += attn * v_t
            for d in range(0, HEAD_DIM):
                out_vec[d] += attn * v_t[d]

        # Store output vector (float32) and lse / ln(2)
        out_offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        tl.store(out_ptr + out_offset, out_vec)

        lse_scaled = lse * ln2
        tl.store(lse_ptr + b * num_qo_heads + h, lse_scaled)
