```python
import math

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


if TRITON_AVAILABLE:
    @triton.jit
    def _gqa_attention_kernel(
        q_ptr,           # *f32, shape [B, num_qo_heads, HEAD_DIM]
        k_ptr, v_ptr,    # *bf16 or *f16, shape [num_pages, 1, num_kv_heads, HEAD_DIM]
        kv_indptr_ptr,   # *i32, shape [B+1]
        out_ptr,         # *bf16, shape [B, num_qo_heads, HEAD_DIM]
        lse_ptr,         # *f32, shape [B, num_qo_heads]
        B: tl.constexpr,
        num_qo_heads: tl.constexpr,
        num_kv_heads: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        sm_scale: tl.constexpr,
        gqa_ratio: tl.constexpr,
        ln2: tl.constexpr,
    ):
        # One program per (b, h)
        b = tl.program_id(0)  # batch id
        h = tl.program_id(1)  # head id
        if b >= B or h >= num_qo_heads:
            return

        # Read token range for this batch
        kv_start = tl.load(kv_indptr_ptr + b)        # i32
        kv_end = tl.load(kv_indptr_ptr + b + 1)     # i32
        num_tokens = kv_end - kv_start              # i32 scalar

        # Load q[b, h] as float32
        q_offset = b * num_qo_heads * HEAD_DIM + h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_offset)  # [HEAD_DIM] f32

        # Initialize output vector and LSE
        out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
        lse = tl.full((), -float("inf"), dtype=tl.float32)

        # Loop over tokens
        # Note: Triton supports scalar loops; this is fine for typical num_tokens.
        for t in range(0, num_tokens):
            idx = kv_start + t  # token index into kv cache
            kv_head = h // gqa_ratio  # 0..num_kv_heads-1

            # Load k_t and v_t, cast to f32 for compute.
            # k_ptr and v_ptr are [num_pages, 1, num_kv_heads, HEAD_DIM]
            # For fixed idx and kv_head, we can load as 1D of length HEAD_DIM.
            k_off = idx * num_kv_heads * HEAD_DIM + kv_head * HEAD_DIM
            v_off = idx * num_kv_heads * HEAD_DIM + kv_head * HEAD_DIM

            # k_ptr and v_ptr are f16/bf16; cast to f32

            v_t = tl.load(v_ptr + v_off).to(tl.float32)  # [HEAD_DIM]

            # Else: diff = y - lse; if diff > 0: lse = lse + log(1 + exp(-diff))
            #       else: lse = lse + log(1 + exp(diff))
            if lse == -float("inf"):
                lse = y
            else:

                term = tl.log(1.0 + tl.exp(-tl.abs(diff)))
                lse = lse + tl.where(diff > 0.0, tl.log(1.0 + tl.exp(-diff)), tl.log(1.0 + tl.exp(diff)))

            attn = tl.exp(y - lse)

            # Accumulate output vector

        # Store LSE / ln(2) (float32)
        lse_scaled = lse / ln2
        tl.store(lse_ptr + b * num_qo_heads + h, lse_scaled)
```

```python
        kv_end = tl.load(kv_indptr_ptr + b + 1)     # i32
        num_tokens = kv_end - kv_start              # i32 scalar

        # Load q[b, h] as float32: linear offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        q_offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_offset)  # [HEAD_DIM] f32

        # Initialize output vector and LSE
        out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
        lse = tl.full((), -float("inf"), dtype=tl.float32)

        # Loop over tokens
        for t in range(0, num_tokens):
            idx = kv_start + t  # token index into kv cache
            kv_head = h // GQA_RATIO  # GQA mapping: 32/8 = 4 groups

            # Compute offsets for k_t and v_t
            # k_ptr and v_ptr are [num_pages, 1, num_kv_heads, HEAD_DIM]
            # For fixed idx and kv_head, offset = idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM
            base = idx * (num_kv_heads * HEAD_DIM)
            k_off = base + kv_head * HEAD_DIM
            v_off = base + kv_head * HEAD_DIM

            # Load k_t and v_t; assume bf16/f16 input, cast to f32 for compute
            k_t = tl.load(k_ptr + k_off).to(tl.float32)  # [HEAD_DIM]
            v_t = tl.load(v_ptr + v_off).to(tl.float32)  # [HEAD_DIM]

            # Dot product: q_vec · k_t
            dot_val = tl.zeros((), dtype=tl.float32)
            for i in range(HEAD_DIM):
                dot_val += q_vec[i] * k_t[i]

            # Scale logits
            y = dot_val * sm_scale

            # Stable LSE update
            if lse == -float("inf"):
                lse = y
            else:
                diff = y - lse
                # Merge: lse_new = lse + log(1 + exp(-abs(diff))) + sign(diff) * (y - lse)
                add_term = tl.where(diff > 0.0, tl.log(1.0 + tl.exp(-diff)), tl.log(1.0 + tl.exp(diff)))
                lse = lse + add_term

            # Compute attention weight
            attn = tl.exp(y - lse)

            # Accumulate output vector
            out_vec += attn * v_t

        # Store output as bfloat16
        out_offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        out_vec_bf16 = out_vec.to(tl.bfloat16)
        tl.store(out_ptr + out_offset, out_vec_bf16)

        # Store LSE / ln(2) (float32)
        lse_scaled = lse * ln2_inv
        tl.store(lse_ptr + b * num_qo_heads + h, lse_scaled)


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):
        # Ensure device and contiguity
        assert q.is_cuda and k_cache.is_cuda and v_cache.is_cuda and kv_indptr.is_cuda, \
            "All tensors must be on CUDA device for Triton kernel."
        q = q.contiguous()
        k_cache = k_cache.contiguous()
        v_cache = v_cache.contiguous()
        kv_indptr = kv_indptr.contiguous()

        B, num_qo_heads, HEAD_DIM = q.shape
        num_kv_heads = k_cache.shape[2]

        # Allocate outputs
        output = torch.empty((B, num_qo_heads, HEAD_DIM), dtype=torch.bfloat16, device=q.device)
        lse = torch.empty((B, num_qo_heads), dtype=torch.float32, device=q.device)

        # Launch Triton kernel: one program per (b, h)
        grid = (B, num_qo_heads)
        ln2_inv = 1.0 / math.log(2.0)
        GQA_RATIO = num_qo_heads // num_kv_heads

        _gqa_attention_kernel[grid](
            q.to(torch.float32),                # q_ptr: float32
            k_cache, v_cache,                  # k_ptr, v_ptr: original dtype (bf16/f16)
            kv_indptr,
            output,
            lse,
            B,
            num_qo_heads,
            num_kv_heads,
            HEAD_DIM,
            sm_scale,
            ln2_inv,
            GQA_RATIO,
            num_warps=4, num_stages=2,
        )

        return output, lse
```

Why this satisfies the requirements
- All numerical computation (dot products, scaling, LSE update, softmax via exp, accumulation) is done inside the Triton kernel. There are no torch matmul, softmax, logsumexp, or elementwise tensor math calls in ModelNew.forward.
- The kernel is invoked from ModelNew.forward with a grid of (B, 32). No host-side loops over tokens; the kernel loops over tokens for each (b, h).
- The output and lse shapes and dtypes match the original behavior: output is bfloat16, lse is float32 divided by ln(2).
- The implementation uses only Triton-supported operations: loads, stores, arithmetic, tl.exp, tl.log, tl.where, and loops. It does not rely on PyTorch tensor ops on GPU.

Performance notes
- Fusing the per-(b,h) attention computation into a single Triton kernel avoids Python overhead and repeated small torch ops, which is beneficial especially with many batches and heads. The kernel performs contiguous loads along the head dimension, which is friendly to memory access.
- Using num_warps=4 and num_stages=2 is a reasonable default for this type of per-program vector compute; you can tune these based on actual profiling.

Testing and benchmarking
- You can call ModelNew with the same inputs as get_inputs; ensure q, k_cache, v_cache, kv_indptr are on CUDA (e.g., .cuda()). The forward will run the Triton kernel and return the output and lse. Compare with the original PyTorch run to validate correctness.
