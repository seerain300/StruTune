import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: one program per (b, h)
# Computes lse[b, h] and output vector out[b, h, :] for this pair.
@triton.jit
def _forward_kernel_bh(
    q_ptr,          # *ptr to q, shape [B, H, D], dtype: float32
    k_ptr,          # *ptr to k_cache, shape [N_total, num_kv_heads, D], dtype: float32
    v_ptr,          # *ptr to v_cache, shape [N_total, num_kv_heads, D], dtype: float32
    kv_indices_ptr, # *ptr to int32 indices, shape [num_kv_indices]
    kv_indptr_ptr,  # *ptr to int32 indptr, shape [B+1]
    lse_ptr,        # *ptr to lse, shape [B, H], float32
    out_ptr,        # *ptr to output, shape [B, H, D], float32
    B: tl.constexpr,         # batch size
    H: tl.constexpr,         # num query heads
    D: tl.constexpr,         # head dim
    sm_scale,                # scalar float32
    N_TOTAL: tl.constexpr,   # compile-time loop bound (e.g., 128)
):
    # Program ids for batch and head
    b = tl.program_id(0)  # int32
    h = tl.program_id(1)  # int32

    # Constants
    num_kv_heads = 8
    kv_ratio = H // num_kv_heads  # 4 for H=32
    kv_head = h // kv_ratio       # GQA mapping: query head maps to kv head

    # Load q vector for this (b, h): q[b, h, :]
    q_base = q_ptr + b * H * D + h * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_vec[d] = tl.load(q_base + d)

    # Compute start and end for this batch from kv_indptr
    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start

    # First pass: streaming logsumexp across tokens
    m = -float("inf")
    sumexp = 0.0  # scalar float32
    ln2 = 0.6931471805599453  # log(2)
    for nn in range(0, N_TOTAL):
        # Mask out iterations beyond actual_num_tokens
        if nn < actual_num_tokens:
            idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)

            # Base pointers for k and v for this token and kv_head
            k_row_ptr = k_ptr + idx * num_kv_heads * D + kv_head * D
            v_row_ptr = v_ptr + idx * num_kv_heads * D + kv_head * D

            # Load k_vec and compute dot with q_vec
            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_row_ptr + d)

            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]

            logit = dot * sm_scale

            new_m = tl.maximum(m, logit)
            # sumexp = sumexp * exp(m - new_m) + exp(logit - new_m)
            sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit - new_m)
            m = new_m

    # Compute lse for this (b, h): logsumexp in base-2
    lse_val = m + tl.log(sumexp) / ln2
    tl.store(lse_ptr + b * H + h, lse_val)

    # Second pass: compute softmax per token and accumulate output
    out_base = out_ptr + b * H * D + h * D
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for nn in range(0, N_TOTAL):
        if nn < actual_num_tokens:
            idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)

            k_row_ptr = k_ptr + idx * num_kv_heads * D + kv_head * D
            v_row_ptr = v_ptr + idx * num_kv_heads * D + kv_head * D

            v_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                v_vec[d] = tl.load(v_row_ptr + d)

            # Recompute dot and logit_scaled
            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_row_ptr + d)
            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]
            logit = dot * sm_scale

            softmax = tl.exp(logit - m) / (sumexp * ln2)
            out_vec += softmax * v_vec

    # Store output vector
    for d in range(0, D):
        tl.store(out_base + d, out_vec[d])
