Analysis
- The original forward performs:
  - For each batch b and query head h: gather KV tokens via kv_indptr and kv_indices.
  - For each token i, compute logits = q[b,h] · k_i, scale by sm_scale, accumulate logsumexp across i, compute softmax, and produce out = sum_i attn_i * v_i.
  - Output[b,h,:] is bfloat16; lse[b,h] is float32.
- The previous attempts either didn’t define Triton kernels or included torch ops in forward, which violates the “TRITON-ONLY” requirement.

Plan
- Implement a Triton kernel that operates per (b, h):
  - Pass NUM_TOKS as a meta parameter and use masks (i < num_tokens_actual) to loop over token indices safely without scalar branching inside Triton.
  - Load q_vec[h,:], k_vec and v_vec per token, compute s = (q·k) * sm_scale.
  - First pass: compute max_s and sum_exp = sum(exp(s - max_s)) across tokens; lse = log(max_s) + log(sum_exp) * (1/ln(2)).
  - Second pass: recompute s, attn = exp(s - lse), accumulate out_vec += attn * v_vec.
  - Store out_vec to output and lse to lse_ptr.
- Host code in ModelNew.forward:
  - Ensure q, k_cache, v_cache are CUDA, contiguous, cast to float32 for compute.
  - Allocate output (float32) and lse (float32).
  - Launch the Triton kernel once per (b,h) pair. Pass num_tokens_actual as a runtime argument (which Triton supports), and mask loads/stores by that.
  - After kernel returns, cast output to bfloat16 to match original.

import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

if TRITON_AVAILABLE:
    @triton.jit
    def _attention_bh_kernel(
        q_ptr,            # *float32, [NUM_QO_HEADS, HEAD_DIM]
        k_ptr,            # *float32, [NUM_PAGES, NUM_KV_HEADS, HEAD_DIM]
        v_ptr,            # *float32, [NUM_PAGES, NUM_KV_HEADS, HEAD_DIM]
        kv_indptr_ptr,    # *int32, [BATCH_SIZE+1]
        kv_indices_ptr,   # *int32, [NUM_KV_INDICES]
        out_ptr,          # *float32, [BATCH_SIZE, NUM_QO_HEADS, HEAD_DIM]
        lse_ptr,          # *float32, [BATCH_SIZE, NUM_QO_HEADS]
        batch_size: tl.constexpr,
        B: tl.constexpr,  # = batch_size
        NUM_QO_HEADS: tl.constexpr,
        NUM_KV_HEADS: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        NUM_TOKS: tl.constexpr,         # max tokens we loop over
        sm_scale: tl.constexpr,         # float32 scalar
        half_ln2_inv: tl.constexpr,     # 1 / ln(2) as float32
    ):
        # program ids
        b = tl.program_id(0)  # batch index
        h = tl.program_id(1)  # query head index

        # Compute start and end for this batch from kv_indptr
        start = tl.load(kv_indptr_ptr + b)  # int32
        end = tl.load(kv_indptr_ptr + b + 1)  # int32
        num_tokens_actual = end - start  # int32

        # GQA mapping: kv_head = h // (32 // 8) = h // 4
        kv_ratio = NUM_QO_HEADS // NUM_KV_HEADS
        kv_head = h // kv_ratio  # 0..7

        # Load q[h, :]
        q_off = h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_off)  # [HEAD_DIM], float32

        # Accumulators for max and sum(exp(s - max))
        max_s = -float('inf')
        sum_exp = 0.0

        # Pass 1: compute max_s and sum_exp across tokens
        for i in range(NUM_TOKS):
            mask_i = i < num_tokens_actual
            idx = tl.load(kv_indices_ptr + start + i, mask=mask_i, other=0)  # int32

            # Compute offsets into k and v for this kv_head and token index
            # k_ptr layout: [NUM_PAGES, NUM_KV_HEADS, HEAD_DIM] contiguous
            k_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
            v_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM

            k_vec = tl.load(k_ptr + k_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32
            v_vec = tl.load(v_ptr + v_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32

            # Dot product: scalar logits
            logits = tl.sum(q_vec * k_vec, axis=0)  # float32
            s = logits * sm_scale

            # Update max and sum-exp only if valid
            if mask_i:
                max_s = tl.maximum(max_s, s)
                # sum_exp = sum_exp * exp(max_s - s) + 1.0 (shifted normalization)
                sum_exp = sum_exp * tl.exp(max_s - s) + 1.0

        # lse = log(max_s) + log(sum_exp) / ln(2)
        lse_val = tl.log(max_s) + tl.log(sum_exp) * half_ln2_inv  # float32
        # Store lse
        tl.store(lse_ptr + b * NUM_QO_HEADS + h, lse_val)

        # Accumulator for output vector
        out_vec = tl.zeros([HEAD_DIM], dtype=tl.float32)

        # Pass 2: recompute s, compute attn, accumulate out
        for i in range(NUM_TOKS):
            mask_i = i < num_tokens_actual
            idx = tl.load(kv_indices_ptr + start + i, mask=mask_i, other=0)  # int32

            k_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
            v_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM

            k_vec = tl.load(k_ptr + k_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32
            v_vec = tl.load(v_ptr + v_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32

            logits = tl.sum(q_vec * k_vec, axis=0)  # float32
            s = logits * sm_scale

            attn = tl.exp(s - lse_val)  # float32 scalar
            out_vec += attn * v_vec  # elementwise multiply and sum accumulates

        # Store output vector for this (b, h)
        out_off = b * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM
        tl.store(out_ptr + out_off, out_vec)

            num_warps=4,

Analysis

- The original PyTorch implementation does, for each (batch b, query head h):
  - Identify cached tokens via kv_indptr[b:b+1] and kv_indices.
  - For each cached token i:
    - Gather k_i and v_i for the appropriate KV head (GQA: kv_head = h // 4).
    - Compute logits = q[b, h] · k_i, scale by sm_scale = 1/sqrt(128).
    - Accumulate lse[b, h] = logsumexp(logits_scaled) / ln(2). Important: this is the logsumexp of the scaled logits, not the raw logits.
    - Compute attn_i = softmax(logits_scaled) and accumulate output[b, h, :] += attn_i * v_i.
  - Return output (bfloat16) and lse (float32).

- Prior attempts either failed to define/launch Triton kernels or used torch ops in forward, which is not allowed. Also, there was a subtle logical bug in the logsumexp update. The correct streaming update for logsumexp of s is:
  - m_new = max(m, s)
  - sum_new = sum * exp(m - m_new) + 1
  - We mistakenly used sum_exp = sum_exp * exp(m - s) + 1, which is incorrect for streaming.
  - Additionally, softmax requires normalization by the number of tokens (T). The original computes lse = logsumexp(s)/ln(2), then uses softmax(s) without dividing by T, which is incorrect. The correct approach is:
    - s_i = scale * (q·k_i)
    - lse = log(sum(exp(s_i))) / ln(2)  (without subtracting max), or with max trick for numerical stability:
      - m = max(s_i), sumexp = sum(exp(s_i - m)), lse = log(sumexp) / ln(2) + m.
    - attn_i = exp(s_i - lse), out += attn_i * v_i.
  - Note: Because output is computed via sum_i attn_i * v_i, the normalization by T cancels out in the final vector. But lse must still be consistent with the scaled logits.

  - For each token i:
    - Load k_i and v_i corresponding to kv_head = h // 4.
    - Compute logits = dot(q[h], k_i), s = logits * sm_scale.
    - First pass: compute m = max(s) and sumexp = sum(exp(s - m)), then lse = log(sumexp) / ln(2) + m.
    - Second pass: recompute s, compute attn = exp(s - lse), and accumulate out_vec += attn * v_i.
  - Store out_vec and lse[b, h].
- Host code:
  - Ensure q, k_cache, v_cache are CUDA tensors, contiguous, cast to float32 for compute.
  - Allocate output (float32) and lse (float32).
  - Launch kernel with grid (batch_size, num_qo_heads). No torch ops in forward.
  - Cast output to bfloat16 to match original signature.

import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True

if TRITON_AVAILABLE:
    @triton.jit
    def _attention_bh_kernel(
        q_ptr,            # *float32, [NUM_QO_HEADS, HEAD_DIM]
        k_ptr,            # *float32, [NUM_PAGES, NUM_KV_HEADS, HEAD_DIM]
        v_ptr,            # *float32, [NUM_PAGES, NUM_KV_HEADS, HEAD_DIM]
        kv_indptr_ptr,    # *int32, [BATCH_SIZE+1]
        kv_indices_ptr,   # *int32, [NUM_KV_INDICES]
        out_ptr,          # *float32, [BATCH_SIZE, NUM_QO_HEADS, HEAD_DIM]
        lse_ptr,          # *float32, [BATCH_SIZE, NUM_QO_HEADS]
        NUM_QO_HEADS: tl.constexpr,
        NUM_KV_HEADS: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        NUM_TOKS: tl.constexpr,         # upper bound of tokens
        sm_scale: tl.constexpr,         # float32 scalar
        half_ln2_inv: tl.constexpr,     # 1 / ln(2) as float32
    ):
        # program ids
        b = tl.program_id(0)  # batch index
        h = tl.program_id(1)  # query head index

        # Compute start and end from kv_indptr
        start = tl.load(kv_indptr_ptr + b)          # int32
        end = tl.load(kv_indptr_ptr + b + 1)        # int32
        num_tokens = end - start                     # int32

        # GQA mapping: kv_head = h // (32 // 8) = h // 4
        kv_ratio = NUM_QO_HEADS // NUM_KV_HEADS
        kv_head = h // kv_ratio  # 0..7

        # Load q[h, :]
        q_off = h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_off)  # [HEAD_DIM], float32

        # First pass: compute m = max(s) and sumexp = sum(exp(s - m))
        m = -float('inf')
        sumexp = 0.0

        for i in range(NUM_TOKS):
            mask_i = i < num_tokens
            idx = tl.load(kv_indices_ptr + start + i, mask=mask_i, other=0)  # int32

            # Offsets into k and v for this kv_head and token index
            k_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
            v_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM

            k_vec = tl.load(k_ptr + k_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32
            v_vec = tl.load(v_ptr + v_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32

            # Dot product: scalar logits
            logits = tl.sum(q_vec * k_vec, axis=0)  # float32
            s = logits * sm_scale

            # Streaming max and sumexp
            if mask_i:
                m_new = tl.maximum(m, s)
                sumexp = sumexp * tl.exp(m - m_new) + 1.0
                m = m_new

        # lse = log(sumexp) / ln(2) + m
        lse_val = tl.log(sumexp) * half_ln2_inv + m  # float32
        # Store lse
        tl.store(lse_ptr + b * NUM_QO_HEADS + h, lse_val)

        # Accumulator for output vector
        out_vec = tl.zeros([HEAD_DIM], dtype=tl.float32)

        # Second pass: recompute s, compute attn, accumulate out
        for i in range(NUM_TOKS):
            mask_i = i < num_tokens
            idx = tl.load(kv_indices_ptr + start + i, mask=mask_i, other=0)  # int32

            k_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM
            v_off = idx * (NUM_KV_HEADS * HEAD_DIM) + kv_head * HEAD_DIM

            k_vec = tl.load(k_ptr + k_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32
            v_vec = tl.load(v_ptr + v_off, mask=mask_i, other=0.0)  # [HEAD_DIM], float32

            logits = tl.sum(q_vec * k_vec, axis=0)  # float32
            s = logits * sm_scale

            attn = tl.exp(s - lse_val)  # float32 scalar
            out_vec += attn * v_vec

        # Store output vector for this (b, h)
        out_off = b * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM
        tl.store(out_ptr + out_off, out_vec)
