import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def fused_gqa_row_two_pass_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [Np, 1, K, D] (shape squeezed to [Np, K, D])
    v_ptr,            # *bf16, [Np, 1, K, D]
    indptr_ptr,       # *int32, [B+1]
    indices_ptr,      # *int32, [Np]
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *float32, [B, H]
    B: tl.constexpr,  # batch size (compile-time constant in kernel launch)
    H: tl.constexpr,  # num query heads
    D: tl.constexpr,  # head dim
    K: tl.constexpr,  # num kv heads
    GR: tl.constexpr, # gqa ratio (H // K)
    sm_scale,         # float32 scalar
    MAX_TOKENS: tl.constexpr,  # max iterations over tokens
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)

    # Compute start/end for this batch element
    start = tl.load(indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start  # runtime int32

    # Load q vector for this head (b, h)
    q_vec = tl.load(q_ptr + pid_b * H * D + pid_h * D + tl.arange(0, D)).to(tl.float32)  # [D]

    # Pass 1: accumulate sum_exp = sum(exp((q·k) * sm_scale)) over tokens
    sum_exp = 0.0
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = start + t
        # Determine kv head for GQA: kv_head = h // GR
        kv_head = pid_h // GR
        # Load k_row[idx, kv_head, :] as vector
        k_row = tl.load(
            k_ptr + idx * (K * D) + kv_head * D + tl.arange(0, D)
        ).to(tl.float32)  # [D]
        # Dot product over D
        dot = tl.sum(q_vec * k_row, axis=0)
        # Scale: we need to compute exp((dot * sm_scale) * (1/ln(2))) to get sum of exponentials
        # For logsumexp in base-2, we later subtract log(sum_exp)/ln(2). For now, accumulate sum_exp.
        sum_exp += tl.exp((dot * sm_scale) * 1.4426950408889634)  # 1 / ln(2)

    # Compute lse in base-2: lse = log(sum_exp) / ln(2)
    # Pass it to second pass via lse_ptr
    lse_val = (tl.log(sum_exp)) / 1.4426950408889634  # log2(sum_exp) = log(sum_exp) * (1/ln(2))

    # Pass 2: recompute dot for each token, compute attn, and accumulate output vector
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = start + t
        kv_head = pid_h // GR
        k_row = tl.load(
            k_ptr + idx * (K * D) + kv_head * D + tl.arange(0, D)
        ).to(tl.float32)  # [D]
        dot = tl.sum(q_vec * k_row, axis=0)
        # attn = exp((dot * sm_scale) - lse_val)
        attn = tl.exp((dot * sm_scale) - lse_val)
        v_row = tl.load(
            v_ptr + idx * (K * D) + kv_head * D + tl.arange(0, D)
        ).to(tl.float32)  # [D]
        out_vec += attn * v_row

    # Store output vector to out[b, h, :]
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))

    # Store lse in base-2 to lse[b, h]
    lse_offset = pid_b * H + pid_h
    tl.store(lse_ptr + lse_offset, lse_val)

    k_ptr,            # *bf16, [Np, K, D]
    v_ptr,            # *bf16, [Np, K, D]
    indptr_ptr,       # *int32, [B+1]
    out_ptr,          # *bf16, [B, H, D]

    sm_scale,         # float32 scalar
    MAX_TOKENS: tl.constexpr,  # max tokens to iterate

    pid_h = tl.program_id(1)

    # Pass 1: accumulate sum_exp in base-2
    sum_exp = 0.0
    for t in range(MAX_TOKENS):

        kv_head = pid_h // 4  # gqa ratio = H // K = 4
