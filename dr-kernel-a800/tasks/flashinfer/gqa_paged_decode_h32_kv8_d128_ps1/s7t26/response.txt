This can happen if the tensor is not 3D or if its shape tuple has unexpected length. To prevent this, we must not rely on Python tuple unpacking for tensor.shape in forward; instead, pass shapes explicitly as function arguments.
  - Subsequent “head_dim mismatch” assertions indicate the forward path did not preserve the expected head_dim. The code needs to strictly match the original assumptions.

- Triton integration: We must compute all numeric work inside Triton kernels. No torch matmul/softmax on host. We’ll implement:
  - lse_kernel: one program per (b, h) to compute lse[b, h] using a chunked loop over tokens. We compute logits_scaled = q·k_i * sm_scale, m = max, sum_exp = sum(exp(scaled - m)), lse = log(sum_exp) + m, and divide by log(2).
  - output_kernel: one program per (b, h) to produce output[b, h, :] by recomputing m and sum_exp, computing softmax, and accumulating soft_i * v_i[j] for each j in [0..D-1].

- Host-side changes:
  - Do not alter input tensors beyond what is necessary. Avoid .contiguous() or dtype conversions in forward. Compute shapes via torch calls (B, Nq, Nkv, D) and pass them as args to kernels. Keep q, k_cache, v_cache as originally received. Ensure we only use tensor.shape or element counts when we know they’re valid.

- Performance: This solution prioritizes correctness and robustness. It parallelizes across (b, h) and loops over tokens within each program. Further optimizations (vectorized reductions, fusing kernels) can be explored after correctness is established.

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def lse_kernel(
    q_ptr,          # *bf16, [B, Nq, D]
    k_ptr,          # *bf16, [Np, Nkv, D]
    indptr_ptr,     # *i32,  [B+1]
    indices_ptr,    # *i32,  [T_total]
    sm_scale,       # f32
    lse_ptr,        # *f32,  [B, Nq]
    B: tl.constexpr,          # batch size
    Nq: tl.constexpr,         # num_qo_heads
    Nkv: tl.constexpr,        # num_kv_heads
    D: tl.constexpr,          # head_dim
    gqa_ratio: tl.constexpr,  # Nq // Nkv (4)
    stride_q_b, stride_q_h, stride_q_d,
):
    # one program per (b, h)
    pid = tl.program_id(0)
    b = pid // Nq
    h = pid % Nq
    kv_head = h // gqa_ratio  # GQA mapping

    # Load q[b, h] (bf16), cast to f32 for compute
    q_base = q_ptr + b * stride_q_b + h * stride_q_h
    q_vec = tl.load(q_base + tl.arange(0, D) * stride_q_d).to(tl.float32)  # [D] f32

    # Token window [start, end)
    start = tl.load(indptr_ptr + b)         # int32
    end = tl.load(indptr_ptr + b + 1)       # int32
    T = end - start

    # Accumulate max and sum_exp for logsumexp
    m = tl.full((), -float("inf"), dtype=tl.float32)
    sum_exp = tl.zeros((), dtype=tl.float32)

    i = 0
    while i < T:
        idx = tl.load(indices_ptr + start + i).to(tl.int32)  # token index
        # k_base points to k_cache[idx, kv_head, :]
        k_base = k_ptr + idx * (Nkv * D) + kv_head * D
        k_i = tl.load(k_base + tl.arange(0, D) * 1).to(tl.float32)  # [D] f32
        attn = tl.sum(q_vec * k_i, axis=0)  # scalar
        scaled = attn * sm_scale
        m = tl.maximum(m, scaled)
        exp_term = tl.exp(scaled - m)
        sum_exp += exp_term
        i += 1

    # lse = log(sum_exp) + m, divide by log(2) per original
    log2 = 0.6931471805599453
    lse_val = tl.log(sum_exp) + m
    tl.store(lse_ptr + b * Nq + h, lse_val / log2)


@triton.jit
def output_kernel(
    q_ptr,          # *bf16, [B, Nq, D]
    k_ptr,          # *bf16, [Np, Nkv, D]
    v_ptr,          # *bf16, [Np, Nkv, D]
    indptr_ptr,     # *i32,  [B+1]
    indices_ptr,    # *i32,  [T_total]
    sm_scale,       # f32
    lse_ptr,        # *f32,  [B, Nq] (not directly used, but kept for potential reuse)
    out_ptr,        # *bf16, [B, Nq, D]
    B: tl.constexpr,          # batch size
    Nq: tl.constexpr,         # num_qo_heads
    Nkv: tl.constexpr,        # num_kv_heads
    D: tl.constexpr,          # head_dim
    gqa_ratio: tl.constexpr,  # Nq // Nkv (4)
    stride_q_b, stride_q_h, stride_q_d,
    stride_k_p, stride_k_h, stride_k_d,
    stride_v_p, stride_v_h, stride_v_d,
    stride_out_b, stride_out_h, stride_out_d,
):
    # one program per (b, h)
    pid = tl.program_id(0)
    b = pid // Nq
    h = pid % Nq
    kv_head = h // gqa_ratio  # GQA mapping

    # Load q[b, h] (bf16), cast to f32 for compute
    q_base = q_ptr + b * stride_q_b + h * stride_q_h
    q_vec = tl.load(q_base + tl.arange(0, D) * stride_q_d).to(tl.float32)  # [D] f32

    # Token window [start, end)
    start = tl.load(indptr_ptr + b)         # int32
    end = tl.load(indptr_ptr + b + 1)       # int32
    T = end - start

    # Recompute m and sum_exp to get softmax
    m = tl.full((), -float("inf"), dtype=tl.float32)
    sum_exp = tl.zeros((), dtype=tl.float32)

    i = 0
    while i < T:
        idx = tl.load(indices_ptr + start + i).to(tl.int32)  # token index
        k_base = k_ptr + idx * (Nkv * D) + kv_head * D
        k_i = tl.load(k_base + tl.arange(0, D) * 1).to(tl.float32)  # [D] f32
        attn = tl.sum(q_vec * k_i, axis=0)  # scalar
        scaled = attn * sm_scale
        m = tl.maximum(m, scaled)
        exp_term = tl.exp(scaled - m)
        sum_exp += exp_term
        i += 1

    # Compute output vector out[b, h, :]
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    j = 0
    while j < D:
        acc = tl.zeros((), dtype=tl.float32)
        i = 0
        while i < T:
            idx = tl.load(indices_ptr + start + i).to(tl.int32)  # token index
            k_base = k_ptr + idx * (Nkv * D) + kv_head * D
            k_i = tl.load(k_base + tl.arange(0, D) * 1).to(tl.float32)  # [D] f32
            attn = tl.sum(q_vec * k_i, axis=0)  # scalar
            scaled = attn * sm_scale
            p = tl.exp(scaled - m) / sum_exp  # scalar softmax
            v_base = v_ptr + idx * stride_v_p + kv_head * stride_v_h
            v_j = tl.load(v_base + j * stride_v_d).to(tl.float32)  # scalar f32
            acc += p * v_j
            i += 1
        # store as bfloat16
        tl.store(out_base + j * stride_out_d, acc.to(tl.bfloat16))
        j += 1


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):
        # We do NOT alter inputs; keep exactly as provided by the harness.
        # Only launch Triton kernels for numeric work; no torch ops for matmul/softmax on host.

        # Shapes (assumed to match the original fixed constants)
        B = q.shape[0]
        Nq = q.shape[1]
        D = q.shape[2]
        assert k_cache.shape[1] == 1 and v_cache.shape[1] == 1  # squeeze happens in kernels
        Np = k_cache.shape[0]
        Nkv = k_cache.shape[2]
        # Ensure on CUDA device; Triton requires CUDA
        assert q.is_cuda and k_cache.is_cuda and v_cache.is_cuda and kv_indptr.is_cuda and kv_indices.is_cuda

        # Output buffers
        out = torch.empty((B, Nq, D), dtype=torch.bfloat16, device=q.device)
        lse = torch.empty((B, Nq), dtype=torch.float32, device=q.device)

        # Grid: one program per (b, h)
        grid = (B * Nq,)

        # Launch lse_kernel
        lse_kernel[grid](
            q, k_cache, kv_indptr, kv_indices, sm_scale,
            lse,
            B, Nq, Nkv, D, Nq // Nkv,
            q.stride(0), q.stride(1), q.stride(2),
        )

        # Launch output_kernel
        output_kernel[grid](
            q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale,
            lse, out,
            B, Nq, Nkv, D, Nq // Nkv,
            q.stride(0), q.stride(1), q.stride(2),
            k_cache.stride(0), k_cache.stride(1), k_cache.stride(2),
            v_cache.stride(0), v_cache.stride(1), v_cache.stride(2),
            out.stride(0), out.stride(1), out.stride(2),
        )

        return out, lse
```
