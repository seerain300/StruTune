  3) Second pass recomputes q·k for each token, computes attn = exp((dot - lse) * sm_scale), and accumulates out_vec += attn * v into a fp32 output vector, stored as bfloat16 at the end.

This design avoids 2D tensors and dynamic loops and should compile reliably.

Final code:

```python
import math

@triton.jit
def fused_gqa_row_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [Np, 1, K, D]
    v_ptr,            # *bf16, [Np, 1, K, D]
    indptr_ptr,       # *int32, [B+1]
    indices_ptr,      # *int32, [num_tokens_total]
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *fp32, [B, H]
    sm_scale,         # fp32 scalar
    B: tl.constexpr,  # batch size
    H: tl.constexpr,  # 32
    D: tl.constexpr,  # 128
    K: tl.constexpr,  # 8
    gqa_ratio: tl.constexpr,  # H // K == 4
    BLOCK_T: tl.constexpr,     # block size for tokens
):
    pid_b = tl.program_id(0)  # batch index
    pid_h = tl.program_id(1)  # query head index

    # Load q vector for this (b, h) in fp32
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Load indptr[start, end] for this batch element
    start = tl.load(indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Pass 1: compute sum_exp = sum(exp((q·k) * sm_scale)) across tokens using fixed blocks
    sum_exp = 0.0
    for t in range(0, num_tokens, BLOCK_T):
        t_offsets = t + tl.arange(0, BLOCK_T)  # Triton vector
        mask_t = t_offsets < num_tokens
        idxs = tl.load(indices_ptr + start + t_offsets, mask=mask_t, other=0).to(tl.int32)  # [BLOCK_T]

        for i in range(BLOCK_T):
            if mask_t[i]:
                idx = idxs[i]  # scalar int32
                kv_head = pid_h // gqa_ratio  # 0..7
                k_row_ptr = k_ptr + idx * (K * D) + kv_head * D
                k_row = tl.load(k_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]
                dot = tl.sum(q_vec * k_row, axis=0)  # scalar fp32
                sum_exp += tl.exp(dot * sm_scale)

    # lse in base-2: log2(sum_exp) = log(sum_exp) / ln(2)
    lse_acc = tl.log(sum_exp) * 1.4426950408889634  # 1/ln(2)

    # Pass 2: compute output vector out[b, h, :] (fp32), then cast to bf16
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for t in range(0, num_tokens, BLOCK_T):
        t_offsets = t + tl.arange(0, BLOCK_T)
        mask_t = t_offsets < num_tokens
        idxs = tl.load(indices_ptr + start + t_offsets, mask=mask_t, other=0).to(tl.int32)  # [BLOCK_T]

        for i in range(BLOCK_T):
            if mask_t[i]:
                idx = idxs[i]
                kv_head = pid_h // gqa_ratio

                # Load K and V rows for this token and head
                k_row_ptr = k_ptr + idx * (K * D) + kv_head * D
                v_row_ptr = v_ptr + idx * (K * D) + kv_head * D

                k_row = tl.load(k_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]
                v_row = tl.load(v_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]

                # Compute logits_scaled and attention
                dot = tl.sum(q_vec * k_row, axis=0)  # scalar fp32
                attn = tl.exp((dot - lse_acc) * sm_scale)
                out_vec += attn * v_row

    # Store output as bfloat16
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))

    # Store lse (base-2)
    lse_offset = pid_b * H + pid_h
    tl.store(lse_ptr + lse_offset, lse_acc)
