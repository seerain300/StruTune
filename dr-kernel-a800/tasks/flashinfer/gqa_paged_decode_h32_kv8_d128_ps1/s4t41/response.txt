import math
import torch
import triton
import triton.language as tl


@triton.jit
def softmax_and_attention_single_bh(
    OUT_ptr,  # *float32, shape [head_dim]
    LSE_ptr,  # *float32, shape [1]
    Q_ptr,    # *float32, shape [head_dim]
    K_ptr,    # *float32, shape [num_tokens, head_dim]
    V_ptr,    # *float32, shape [num_tokens, head_dim]
    num_tokens,            # int
    sm_scale,              # float32
    LOG2_INVERSE,          # float32 (1 / ln(2))
    head_dim: tl.constexpr,         # compile-time constant
    num_qo_heads: tl.constexpr,     # not used in this kernel but kept for completeness
    num_kv_heads: tl.constexpr,     # for GQA ratio (not used here, but kept)
    KV_HEAD: tl.constexpr,          # compile-time constant = h // gqa_ratio
):
    # First pass: compute lse = logsumexp(scaled_logits) / ln(2)
    running_max = -float("inf")
    running_sum = 0.0

    t = 0
    while t < num_tokens:
        # Load q vector for head h (1D vector of length head_dim)
        i = tl.arange(0, head_dim)  # OK: head_dim is constexpr
        q_vec = tl.load(Q_ptr + i)

        # Load k vector for token t (1D vector of length head_dim)
        k_row = tl.load(K_ptr + t * head_dim + i)

        # Compute logits and scaled logits
        logits = tl.sum(q_vec * k_row, axis=0)  # scalar
        scaled = logits * sm_scale

        # Update running max and sum for numerically stable logsumexp
        running_max_new = tl.maximum(running_max, scaled)
        running_sum = running_sum * tl.exp(running_max - running_max_new) + tl.exp(scaled - running_max_new)
        running_max = running_max_new

        t += 1

    # Compute lse: log(sum(exp(scaled - max))) + max, then divide by ln(2)
    lse_value = tl.log(running_sum) + running_max
    lse_value = lse_value * LOG2_INVERSE
    tl.store(LSE_ptr, lse_value)

    # Second pass: compute output = sum_j softmax(scaled)_j * V_j
    out_vec = tl.zeros((head_dim,), dtype=tl.float32)
    t = 0
    while t < num_tokens:
        # Load q vector
        i = tl.arange(0, head_dim)
        q_vec = tl.load(Q_ptr + i)

        # Load k and v vectors
        k_row = tl.load(K_ptr + t * head_dim + i)
        v_row = tl.load(V_ptr + t * head_dim + i)

        # Compute scaled logits
        logits = tl.sum(q_vec * k_row, axis=0)
        scaled = logits * sm_scale

        # Compute attention for this token
        attn = tl.exp(scaled - lse_value)  # scalar

        # Accumulate output: out += attn * v_row
        out_vec += attn * v_row

        t += 1

    # Store output
    j = tl.arange(0, head_dim)
    tl.store(OUT_ptr + j, out_vec)

        assert num_kv_heads == 8
        assert head_dim == 128
        assert len_indptr == batch_size + 1
        assert num_kv_indices == kv_indptr[-1].item()

        # Output buffers

                q_vec = q[b, h, :].to(torch.float32).contiguous()  # [head_dim]

                out_vec = torch.empty((head_dim,), dtype=torch.float32, device=q.device)

                grid = (1, 1)  # single program instance per (b, h); adjust if needed
                # Note: The kernel takes OUT_ptr as a 1D pointer of length head_dim. We pass &out_vec[0].
                softmax_and_attention_single_bh[grid](

                    V_t,               # V_ptr
                    num_tokens_b,      # num_tokens

                    1.4426950408889634,  # LOG2_INVERSE = 1 / ln(2)
                    head_dim,           # tl.constexpr

                    num_kv_heads,       # tl.constexpr (unused in kernel but kept)

                )

                # Store output and lse
                output[b, h, :] = out_vec.to(torch.bfloat16)
                lse[b, h] = lse_scalar.item()  # store as float32
                # Note: We set lse for all heads; in the kernel, we computed only the lse for the selected head and stored it.
                # Since the kernel only computes one head at a time, this is correct.

        return output, lse

    - Second pass: loop over tokens again, recompute scaled logits, attn = exp(scaled - lse), and accumulate out_vec += attn * v_row.
  - Host code:
    - Compute num_tokens per batch: num_tokens_b = kv_indptr[b + 1] - kv_indptr[b].
    - For each batch b, build token_indices = kv_indices[kv_indptr[b]:kv_indptr[b+1]] -> shape [num_tokens_b].
    - Gather K and V per kv_head:
      - k_rows = k_cache[token_indices, kv_head, :] -> [num_tokens_b, head_dim]
      - v_rows = v_cache[token_indices, kv_head, :] -> [num_tokens_b, head_dim]
      - Cast to float32 and make contiguous.
    - Q vector per head: q_vec = q[b, h, :] as float32.
    - Launch the Triton kernel with grid=(1, 1) for each (b, h), passing Q, K, V, num_tokens_b, sm_scale, LOG2_INVERSE, and constexprs head_dim and KV_HEAD = h // 4.
    - Store output to bfloat16 and lse to float32.

Key fixes:
- Triton requires tl.arange bounds to be tl.constexpr. We declare head_dim and KV_HEAD as tl.constexpr in the kernel signature.
- We avoid any shape mismatches by directly indexing k_cache and v_cache with token indices and kv_head in host code, and passing K and V as [num_tokens_b, head_dim] contiguous tensors to the kernel. No torch reductions in forward beyond minimal casting and tensor preparation.

This satisfies the Triton-only requirement and eliminates the previous IndexError and compilation errors.

Final code
```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def softmax_and_attention_single_bh(
    OUT_ptr,   # *float32, shape [head_dim]
    LSE_ptr,   # *float32, scalar
    Q_ptr,     # *float32, shape [head_dim]
    K_ptr,     # *float32, shape [num_tokens_b, head_dim] viewed as flattened
    V_ptr,     # *float32, shape [num_tokens_b, head_dim] viewed as flattened
    num_tokens_b,            # int runtime: number of tokens for this batch
    sm_scale,                # float32
    LOG2_INVERSE,            # float32 (1 / ln(2))
    head_dim: tl.constexpr,         # compile-time constant
    KV_HEAD: tl.constexpr,          # compile-time constant: h // 4
):
    # First pass: compute lse = logsumexp(scaled_logits) / ln(2)
    running_max = -float("inf")
    running_sum = 0.0

    t = 0
    while t < num_tokens_b:
        # Load q vector for head h (1D of length head_dim)
        i = tl.arange(0, head_dim)
        q_vec = tl.load(Q_ptr + i)

        # Load k vector for token t: K_ptr is [num_tokens_b, head_dim] flattened
        k_row = tl.load(K_ptr + t * head_dim + i)

        # Compute logits and scaled logits
        logits = tl.sum(q_vec * k_row, axis=0)  # scalar
        scaled = logits * sm_scale

        # Update running max and sum (stable logsumexp)
        running_max_new = tl.maximum(running_max, scaled)
        running_sum = running_sum * tl.exp(running_max - running_max_new) + tl.exp(scaled - running_max_new)
        running_max = running_max_new

        t += 1

    # lse = log(sum(exp(scaled - max))) + max, divide by ln(2)
    lse_value = tl.log(running_sum) + running_max
    lse_value = lse_value * LOG2_INVERSE
    tl.store(LSE_ptr, lse_value)

    # Second pass: compute output = sum_j softmax(scaled)_j * V_j for this batch
    out_vec = tl.zeros((head_dim,), dtype=tl.float32)
    t = 0
    while t < num_tokens_b:
        i = tl.arange(0, head_dim)
        q_vec = tl.load(Q_ptr + i)
        k_row = tl.load(K_ptr + t * head_dim + i)
        v_row = tl.load(V_ptr + t * head_dim + i)

        logits = tl.sum(q_vec * k_row, axis=0)
        scaled = logits * sm_scale

        attn = tl.exp(scaled - lse_value)  # scalar attention for this token

        out_vec += attn * v_row

        t += 1

    # Store output vector
    j = tl.arange(0, head_dim)
    tl.store(OUT_ptr + j, out_vec)


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):
        # Shapes
        batch_size, num_qo_heads, head_dim = q.shape
        num_pages, num_kv_heads, _ = k_cache.shape  # k_cache: [num_pages, num_kv_heads, head_dim]
        len_indptr = kv_indptr.shape[0]
        num_kv_indices = kv_indices.shape[0]

        # Assertions as in original
        assert num_qo_heads == 32
        assert num_kv_heads == 8
        assert head_dim == 128
        assert len_indptr == batch_size + 1
        assert num_kv_indices == kv_indptr[-1].item()

        # Output buffers
        output = torch.empty((batch_size, num_qo_heads, head_dim), dtype=torch.bfloat16, device=q.device)
        lse = torch.empty((batch_size, num_qo_heads), dtype=torch.float32, device=q.device)

        gqa_ratio = num_qo_heads // num_kv_heads  # 4

        # Iterate over batch and heads; launch Triton kernel per (b, h)
        for b in range(batch_size):
            # Compute number of tokens for this batch
            start = int(kv_indptr[b].item())
            end = int(kv_indptr[b + 1].item())
            num_tokens_b = end - start

            # If no tokens, lse = -inf and output zero
            if num_tokens_b <= 0:
                lse[b, :] = -float("inf")
                output[b].zero_()
                continue

            # Gather token indices for this batch
            token_indices = kv_indices[start:end].to(torch.long).contiguous()  # [num_tokens_b]

            # For each head h
            for h in range(num_qo_heads):
                kv_head = h // gqa_ratio

                # Gather K and V for this batch and kv_head: [num_tokens_b, head_dim]
                k_rows = k_cache[token_indices, kv_head, :]  # bfloat16
                v_rows = v_cache[token_indices, kv_head, :]  # bfloat16

                # Cast to float32 for Triton
                K_t = k_rows.to(torch.float32).contiguous()  # [num_tokens_b, head_dim]
                V_t = v_rows.to(torch.float32).contiguous()  # [num_tokens_b, head_dim]

                # Q vector for this head
                q_vec = q[b, h, :].to(torch.float32).contiguous()  # [head_dim]

                # Output vector and lse scalar
                out_vec = torch.empty((head_dim,), dtype=torch.float32, device=q.device)
                lse_scalar = torch.empty((), dtype=torch.float32, device=q.device)

                # Launch Triton kernel for (b, h)
                grid = (1, 1)
                softmax_and_attention_single_bh[grid](
                    out_vec,                # OUT_ptr
                    lse_scalar,             # LSE_ptr
                    q_vec,                  # Q_ptr
                    K_t.view(-1),           # K_ptr flattened to [num_tokens_b * head_dim]
                    V_t.view(-1),           # V_ptr flattened
                    num_tokens_b,           # runtime: number of tokens for this batch
                    float(sm_scale),        # sm_scale
                    1.4426950408889634,     # LOG2_INVERSE
                    head_dim,               # constexpr
                    h // gqa_ratio,         # KV
