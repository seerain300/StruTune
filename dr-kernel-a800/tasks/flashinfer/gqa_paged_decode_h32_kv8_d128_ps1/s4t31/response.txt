Root cause and fixes

However, num_tokens per batch is derived from kv_indptr and kv_indices, and each token index corresponds to a particular “page” (i.e., the cache index in the first dimension). To handle this, we must gather the entire row k_cache[token_index, :, :] and v_cache[token_index, :, :], which gives us [8, 128] per token. Then, for GQA, we select kv_head = h // 4 and use the corresponding 128-dim vector.
- Triton pointer unpacking error: The earlier “too many values to unpack (expected 3)” came from the kernel signature expecting three pointers, but I only passed two. I’ll simplify by only passing q and the combined K/V, and load the necessary 128-dim vectors inside the kernel for each token. This avoids any pointer count mismatch and keeps all math inside the kernel.
- Host code must not perform any torch reductions or sqrt operations. All computation will be done by Triton kernels.

import torch
import triton
import triton.language as tl
import math

# Constants
HEAD_DIM = 128
NUM_QO_HEADS = 32
NUM_KV_HEADS = 8
GQA_RATIO = NUM_QO_HEADS // NUM_KV_HEADS  # = 4

LOG2_INV = 1.4426950408889634  # 1 / ln(2)


@triton.jit
def softmax_and_attention_single_bh(
    q_ptr,            # *fp32, shape [B, 32, 128]
    K_ptr,            # *fp32, shape [num_tokens_max, 8, 128] (will be masked by num_tokens)
    V_ptr,            # *fp32, shape [num_tokens_max, 8, 128]
    num_tokens_ptr,   # *int32, shape [B]
    lse_ptr,          # *fp32, shape [B*32]
    output_ptr,       # *fp32, shape [B, 32, 128]
    SM_SCALE: tl.constexpr,  # scalar float32
    HEAD_DIM: tl.constexpr,  # 128
    NUM_KV_HEADS: tl.constexpr,  # 8
    GQA_RATIO: tl.constexpr,  # 4
):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Load q[b, h, :]
    q_off = b * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM + tl.arange(0, HEAD_DIM)
    q_vec = tl.load(q_ptr + q_off)  # [128], fp32

    # Number of tokens for this batch
    num_tokens = tl.load(num_tokens_ptr + b)

    # First pass: compute lse = logsumexp(scaled_logits) / ln(2)
    running_max = -float("inf")
    running_sum = 0.0
    # Iterate up to num_tokens; Triton supports loops
    for t in range(0, 1000):  # upper bound; we will break when t >= num_tokens (but Triton loops are not breakable)
        # If t >= num_tokens, q*k = 0, so logsumexp ignores. To avoid out-of-bounds, we can mask via q*k but Triton doesn't support early break.
        # Instead, we rely on zero padding when t >= num_tokens (K_ptr/V_ptr are zero-padded beyond actual tokens).
        k_off = t * (NUM_KV_HEADS * HEAD_DIM) + (h // GQA_RATIO) * HEAD_DIM + tl.arange(0, HEAD_DIM)
        v_off = t * (NUM_KV_HEADS * HEAD_DIM) + (h // GQA_RATIO) * HEAD_DIM + tl.arange(0, HEAD_DIM)
        k_vec = tl.load(K_ptr + k_off)  # [128]
        v_vec = tl.load(V_ptr + v_off)  # [128]
        logits = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits * SM_SCALE
        running_max_new = tl.maximum(running_max, scaled)
        exp_term = tl.exp(scaled - running_max)
        running_sum = running_sum * tl.exp(running_max - running_max_new) + exp_term
        running_max = running_max_new

    # lse = log(running_sum) + running_max; divide by ln(2)
    lse = tl.log(running_sum) + running_max
    lse = lse * LOG2_INV
    tl.store(lse_ptr + b * NUM_QO_HEADS + h, lse)

    # Second pass: compute output[b, h, :]
    out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
    for t in range(0, 1000):
        k_off = t * (NUM_KV_HEADS * HEAD_DIM) + (h // GQA_RATIO) * HEAD_DIM + tl.arange(0, HEAD_DIM)
        v_off = t * (NUM_KV_HEADS * HEAD_DIM) + (h // GQA_RATIO) * HEAD_DIM + tl.arange(0, HEAD_DIM)
        k_vec = tl.load(K_ptr + k_off)  # [128]
        v_vec = tl.load(V_ptr + v_off)  # [128]
        logits = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits * SM_SCALE
        attn = tl.exp(scaled - lse)  # scalar
        out_vec += attn * v_vec  # elementwise multiply then sum over v_vec dimension? No: attn is scalar; out_vec += attn * v_vec

    # Store output[b, h, :]
    out_off = b * (NUM_QO_HEADS * HEAD_DIM) + h * HEAD_DIM + tl.arange(0, HEAD_DIM)
    tl.store(output_ptr + out_off, out_vec)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):
        # Ensure tensors are on the same device and contiguous
        assert q.is_cuda and k_cache.is_cuda and v_cache.is_cuda, "All inputs must be CUDA tensors"
        B, QH, HD = q.shape
        assert QH == NUM_QO_HEADS and HD == HEAD_DIM
        N, K, VD = k_cache.shape
        assert K == NUM_KV_HEADS and VD == HEAD_DIM, "k_cache must be [num_pages, 8, 128]"
        # We don't rely on the 1 in the middle dim; we treat as [N, 8, 128]
        # Compute per-batch num_tokens from kv_indptr and kv_indices
        # kv_indptr: shape [B+1], int32
        # kv_indices: shape [num_tokens], int32
        # Note: num_tokens per batch is kv_indptr[b+1] - kv_indptr[b]
        # We also need to know the starting index for each batch, which is kv_indptr[b]
        # But kv_indices is flat; we need to map tokens to actual cache rows: use kv_indptr[b] as start and count as num_tokens_b.
        # We'll build token_indices list per batch. However, Triton prefers pre-packed inputs. We can gather rows here.

        # Prepare num_tokens per batch (host-side)
        num_tokens = [int(kv_indptr[b + 1].item()) - int(kv_indptr[b].item()) for b in range(B)]
        # We'll pass num_tokens as int32 tensor
        num_tokens_tensor = torch.tensor(num_tokens, dtype=torch.int32, device=q.device)

        # Now build K_t and V_t for each batch: shape [num_tokens_b, 8, 128]
        # We need to gather rows from k_cache and v_cache using kv_indices[start + t].
        # But since k_cache shape is [N, 8, 128], we can build K_t by indexing k_cache with row_idx = kv_indices[start + t].
        # We'll pad to a maximum number of tokens to fit the Triton loop. From the provided workloads, num_tokens <= num_pages.
        # However, workloads may have large N; we can pick a safe MAX_TOKENS as 1000 for this kernel.

        # We'll pad K and V with zeros beyond actual num_tokens to avoid OOB; Triton mask can't be used like that directly.
        # Simpler: we won't rely on padding; instead, we keep the loop iteration to 1000 and guard loads by t < num_tokens.
        # But Triton doesn't support early break; we rely on the fact that after num_tokens, num_tokens_tensor ensures we don't loop further logically.

        # Build K_ptr and V_ptr of shape [MAX_TOKENS, 8, 128] with zeros beyond actual tokens.
        # We can construct K_t and V_t per batch and then concatenate across batches if needed, but Triton expects single pointers.
        # Instead, we'll create K_all and V_all for the maximum tokens in any batch (upper bound 1000), and fill zeros. But that's large.
        # Better: for each batch, create K_t and V_t of shape [num_tokens_b, 8, 128], then expand to [1000, 8, 128] by zeros.

        # For simplicity and to avoid excessive memory, we'll compute per batch directly: the kernel loops up to 1000 and uses num_tokens to guide the math.
        # That means for t >= num_tokens, q*k will be 0 (due to k loaded as zeros). So the first pass will be correct; second pass will accumulate zeros beyond num_tokens.

        # Allocate output and lse
        output = torch.empty((B, NUM_QO_HEADS, HEAD_DIM), dtype=torch.float32, device=q.device)
        lse = torch.empty((B, NUM_QO_HEADS), dtype=torch.float32, device=q.device)

        # Cast q, k_cache, v_cache to fp32 (inputs are bf16; we use fp32 for computation)
        q_fp32 = q.to(torch.float32)
        k_cache_fp32 = k_cache.to(torch.float32).contiguous()
        v_cache_fp32 = v_cache.to(torch.float32).contiguous()

        # Launch Triton kernel: grid over (batch, query_heads)
        grid = (B, NUM_QO_HEADS)
        softmax_and_attention_single_bh[grid](
            q_fp32,
            k_cache_fp32,
            v_cache_fp32,
            num_tokens_tensor,
            lse,
            output,
            SM_SCALE=sm_scale,  # scalar
            HEAD_DIM=HEAD_DIM,
            NUM_KV_HEADS=NUM_KV_HEADS,
            GQA_RATIO=GQA_RATIO,
            num_warps=4,
        )

        # Return output as bfloat16 and lse as float32 (matching original)
        return output.to(torch.bfloat16), lse
