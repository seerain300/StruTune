Plan:
- Implement three Triton kernels:
  1) load_kv_indices_kernel: Load kv_indices for the range [start, end) for a given batch b into a temporary tensor on device.
  2) pass1_reduce_kernel: For each (b, h), loop over tokens in that range using scalar iterations (MAX_TOKENS), load q[b, h, :], k[token, kv_head, :], v[token, kv_head, :], compute dot = sum(q · k) over D (constexpr loop), scale by sm_scale, and accumulate sum_exp. Compute lse in base-2 (log2).
  3) pass2_accumulate_kernel: Loop again over tokens, recompute dot, compute attn = exp((dot - lse) * sm_scale), and accumulate output[b, h, :] += attn * v[token, kv_head] over D (constexpr loop). Write output as bfloat16.
- ModelNew.forward will:
  - Ensure inputs are contiguous and on CUDA.
  - Launch load_kv_indices_kernel to materialize kv_indices for the current batch b.
  - Launch pass1_reduce_kernel to compute lse for all (b, h).
  - Launch pass2_accumulate_kernel to write the output for all (b, h).
- We pass sm_scale as a kernel argument and keep D, H, K, gqa_ratio as tl.constexpr to aid specialization. We avoid any tensor operations (torch.matmul, softmax, etc.) in host code, and we avoid tl.arange across dynamic token counts to prevent compilation failures.
- This satisfies the Triton-only requirement (all computation happens in Triton kernels), and it should compile and run on all given workloads.

import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Kernel 1: Load kv_indices for a given batch b into a contiguous device tensor
@triton.jit
def load_kv_indices_kernel(
    kv_indptr_ptr,       # *int32, shape [B+1]
    kv_indices_ptr,      # *int32, shape [num_kv_indices]
    tmp_ptr,             # *int32, shape [B+1], output buffer for indices range
    pid_b: tl.constexpr, # current batch index
    MAX_TOKENS: tl.constexpr,
):
    # For each t in [0, MAX_TOKENS), if t < num_tokens, write kv_indices[page_start + t]
    start = tl.load(kv_indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start
    # We cannot use Python range with runtime num_tokens; instead use scalar iterations guarded by if.
    # However, Triton requires loops to be compile-time; to keep it simple and robust, we avoid a while loop here.
    # Instead, we write only the [start, end) entries into tmp_ptr using if guards.
    # But since Triton doesn't support break, we emulate by not writing when idx out of range.
    # We will run a fixed number of iterations (MAX_TOKENS), but note: this kernel is only used to materialize indices, not to compute attention.
    # Here we just write the two scalars start and end to tmp_ptr[0] and tmp_ptr[1] to indicate range.
    tl.store(tmp_ptr + 0, start)
    tl.store(tmp_ptr + 1, end)


# Kernel 2: Compute lse (base-2 logsumexp) per (b, h) across tokens in [start, end)
@triton.jit
def pass1_reduce_kernel(
    q_ptr,               # *bf16, [B, H, D]
    k_ptr,               # *bf16, [Np, 1, K, D] but we address as [num_tokens, K, D] via indices
    v_ptr,               # *bf16, [Np, 1, K, D]
    tmp_ptr,             # *int32, [2], holds [start, end]
    output_ptr,          # *bf16, [B, H, D] (unused in this pass; we compute a scalar lse)
    lse_ptr,             # *float32, [B, H], output scalar lse per (b, h)
    sm_scale,            # float32 scalar
    pid_b: tl.constexpr, pid_h: tl.constexpr,
    B: tl.constexpr, H: tl.constexpr, D: tl.constexpr, K: tl.constexpr, gqa_ratio: tl.constexpr,
    MAX_TOKENS: tl.constexpr,
):
    # Load range [start, end]
    start = tl.load(tmp_ptr + 0).to(tl.int32)
    end = tl.load(tmp_ptr + 1).to(tl.int32)
    num_tokens = end - start

    # Load q vector for this head
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Accumulate sum_exp over tokens
    sum_exp = 0.0
    for t in range(MAX_TOKENS):
        if t < num_tokens:
            idx = start + t  # int32
            kvh = pid_h // gqa_ratio  # select corresponding kv head
            # Compute k and v offsets for this token and kv head
            # k_ptr layout is [num_tokens, K, D] when viewed via indices; we address directly:
            # We need k[idx, kvh, :] and v[idx, kvh, :]
            # Since we cannot vectorize over tokens, we iterate scalarly and use if guards above.
            # Here, we simply accumulate scalar dot products over D (constexpr loop).
            dot = 0.0
            for d in range(D):
                k_val = tl.load(k_ptr + idx * K * D + kvh * D + d).to(tl.float32)
                dot += q_vec[d] * k_val
            sum_exp += tl.exp(dot * sm_scale)
    # lse in base-2
    lse_val = tl.log(sum_exp) / tl.log(2.0)  # log base 2
    lse_addr = pid_b * H + pid_h
    tl.store(lse_ptr + lse_addr, lse_val)


# Kernel 3: Accumulate output vector for (b, h) across tokens, using lse from lse_ptr
@triton.jit
def pass2_accumulate_kernel(
    q_ptr,               # *bf16, [B, H, D]
    k_ptr,               # *bf16, [Np, 1, K, D]
    v_ptr,               # *bf16, [Np, 1, K, D]
    lse_ptr,             # *float32, [B, H]
    output_ptr,          # *bf16, [B, H, D]
    sm_scale,            # float32 scalar
    pid_b: tl.constexpr, pid_h: tl.constexpr,
    B: tl.constexpr, H: tl.constexpr, D: tl.constexpr, K: tl.constexpr, gqa_ratio: tl.constexpr,
    MAX_TOKENS: tl.constexpr,
):
    start = tl.load(tmp_ptr + 0).to(tl.int32)
    end = tl.load(tmp_ptr + 1).to(tl.int32)
    num_tokens = end - start

    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)

    lse_val = tl.load(lse_ptr + pid_b * H + pid_h).to(tl.float32)

    out_vec = tl.zeros((D,), dtype=tl.float32)
    for t in range(MAX_TOKENS):
        if t < num_tokens:
            idx = start + t
            kvh = pid_h // gqa_ratio
            dot = 0.0
            for d in range(D):
                k_val = tl.load(k_ptr + idx * K * D + kvh * D + d).to(tl.float32)
                dot += q_vec[d] * k_val
            attn = tl.exp((dot - lse_val) * sm_scale)
            for d in range(D):
                v_val = tl.load(v_ptr + idx * K * D + kvh * D + d).to(tl.float32)
                out_vec[d] += attn * v_val

    # Store output as bfloat16
    out_offset = pid_b * H * D + pid_h * D
    out_vals = out_vec.to(tl.bfloat16)
    # Store contiguous vector of length D
    # We can store each element via a loop
    for d in range(D):
        tl.store(output_ptr + out_offset + d, out_vals[d])


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):
        # Ensure Triton is available and tensors are on CUDA
        if not TRITON_AVAILABLE or not q.is_cuda:
            # Fallback: original PyTorch implementation (not used for evaluation here)
            batch_size, num_qo_heads, head_dim = q.shape
            _, _, num_kv_heads, _ = k_cache.shape
            assert num_qo_heads == 32 and num_kv_heads == 8 and head_dim == 128
            device = q.device
            output = torch.empty((batch_size, num_qo_heads, head_dim), dtype=torch.bfloat16, device=device)
            lse = torch.empty((batch_size, num_qo_heads), dtype=torch.float32, device=device)
            gqa_ratio = num_qo_heads // num_kv_heads
            k_cache_flat = k_cache.squeeze(1).to(torch.float32)  # [Np, K, D]
            v_cache_flat = v_cache.squeeze(1).to(torch.float32)
            for b in range(batch_size):
                start = int(kv_indptr[b].item())
                end = int(kv_indptr[b + 1].item())
                num_tokens = end - start
                k_batch = k_cache_flat[start:end]  # [num_tokens, K, D]
                v_batch = v_cache_flat[start:end]
                q_batch = q[b].to(torch.float32)  # [H, D]
                for h in range(num_qo_heads):
                    kvh = h // gqa_ratio
                    q_head = q_batch[h]  # [D]
                    logits = torch.matmul(q_head, k_batch[:, kvh].T)  # [num_tokens]
                    lse[b, h] = torch.logsumexp(logits * sm_scale, dim=-1) / math.log(2.0)

        q = q.contiguous()
        k_cache = k_cache.contiguous()
        v_cache = v_cache.contiguous()
        kv_indptr = kv_indptr.contiguous()
        kv_indices = kv_indices.contiguous()

        _, _, num_kv_heads, _ = k_cache.shape
        assert num_qo_heads == 32 and num_kv_heads == 8 and head_dim == 128

        lse = torch.empty((batch_size, num_qo_heads), dtype=torch.float32, device=device)

        # Temporary buffer to hold [start, end] for each batch
        tmp_ptr = torch.empty(2, dtype=torch.int32, device=device)

        # Grid: one program per (b, h)
        grid = (batch_size, num_qo_heads)

        # Launch 1: load kv indices for the range of each batch into tmp_ptr (here tmp_ptr holds [start, end]).
        load_kv_indices_kernel[grid](
            kv_indptr, kv_indices, tmp_ptr,
            pid_b=batch_size,  # placeholder; Triton will specialize per launch anyway
            MAX_TOKENS=1024,
            num_warps=1, num_stages=1,

        # Launch 2: reduce to lse per (b, h)
        pass1_reduce_kernel[grid](
            q, k_cache, v_cache, tmp_ptr, output, lse, sm_scale,
            pid_b=batch_size, pid_h=num_qo_heads,  # placeholders; Triton uses constexpr specialization
            B=batch_size, H=num_qo_heads, D=head_dim, K=num_kv_heads, gqa_ratio=4,
            MAX_TOKENS=1024,
            num_warps=1, num_stages=1,

        # Launch 3: accumulate output per (b, h)
        pass2_accumulate_kernel[grid](
            q, k_cache, v_cache, lse, output, sm_scale,
            pid_b=batch_size, pid_h=num_qo_heads,
            B=batch_size, H=num_qo_heads, D=head_dim, K=num_kv_heads, gqa_ratio=4,
            MAX_TOKENS=1024,
            num_warps=1, num_stages=1,
        )

        return output, lse

  - Load q[b, h, :] once and k[idx, kvh, :], v[idx, kvh, :] for that token, compute dot = sum(q · k) over D (constexpr loop), compute attn for that token, and update a global output vector for (b, h). Since each program handles a unique token, there is no need for reduction loops. We still need an lse computation; we can compute lse per (b, h) via a separate reduction kernel that uses one program per (b, h) with vectorized loads across D (which Triton supports when using tl.arange with constexpr D), but since D=128 is fixed, it’s safe.
- For simplicity and robustness:
  1) Kernel reduce_lse_kernel: one program per (b, h), loops over all tokens for that batch (scalar loop with if t < num_tokens), accumulates sum_exp and computes lse. This avoids vectorized token loops and is acceptable.
  2) Kernel update_output_kernel: one program per (b, h, token), computes dot, attn, and updates output[b, h, :] += attn * v[token, kvh]. Launch grid = (batch_size, num_qo_heads, num_tokens) so there are no dynamic loops. Triton will compile such a grid and run one program per token.
- This approach avoids the previous Triton language constructs errors (no tl.arange misuse, no Python dynamic loops), and ensures correct ranges by using program_id(0) for batch, program_id(1) for head, program_id(2) for token index. We still keep sm_scale as an argument, and all math in fp32.

# Kernel A: Compute lse (base-2 logsumexp) per (b, h) across all tokens in [kv_indptr[b], kv_indptr[b+1))
@triton.jit
def reduce_lse_kernel(
    q_ptr,               # *bf16, [B, H, D]
    k_ptr,               # *bf16, [Np, 1, K, D]
    v_ptr,               # *bf16, [Np, 1, K, D]
    kv_indptr_ptr,       # *int32, [B+1]
    lse_ptr,             # *float32, [B, H]
    sm_scale,            # float32 scalar
    pid_b: tl.constexpr, pid_h: tl.constexpr,
    B: tl.constexpr, H: tl.constexpr, D: tl.constexpr, K: tl.constexpr, gqa_ratio: tl.constexpr,
    MAX_TOKENS: tl.constexpr,
):
    # Load range [start, end] for this batch
    start = tl.load(kv_indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Load q vector for this head
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Accumulate sum_exp over tokens (scalar iterations guarded by num_tokens)
    sum_exp = 0.0
    for t in range(MAX_TOKENS):
        if t < num_tokens:
            idx = start + t  # token index for this batch
            kvh = pid_h // gqa_ratio  # select corresponding kv head
            dot = 0.0
            for d in range(D):
                k_val = tl.load(k_ptr + idx * K * D + kvh * D + d).to(tl.float32)
                dot += q_vec[d] * k_val
            sum_exp += tl.exp(dot * sm_scale)

    # lse in base-2
    lse_val = tl.log(sum_exp) / tl.log(2.0)  # log base 2
    lse_addr = pid_b * H + pid_h
    tl.store(lse_ptr + lse_addr, lse_val)

    sm_scale,            # float32 scalar

    B: tl.constexpr, H: tl.constexpr, D: tl.constexpr, K: tl.constexpr, gqa_ratio: tl.constexpr,
):
    start = tl.load(kv_indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Guard: ensure we only process valid tokens
    if pid_t < num_tokens:
        idx = start + pid_t  # token index for this batch
        kvh = pid_h // gqa_ratio  # select corresponding kv head

        q_offset = pid_b * H * D + pid_h * D
        q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

        dot = 0.0

            tl.store(output_ptr + out_offset + d, tl.load(output_ptr + out_offset + d).to(tl.float32) + attn * v_val)
