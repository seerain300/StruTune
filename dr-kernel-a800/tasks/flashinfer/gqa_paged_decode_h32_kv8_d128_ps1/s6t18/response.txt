```python
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: one program per (batch b, head h)
@triton.jit
def _forward_kernel_bh(
    q_ptr,           # *fp16 or *fp32, shape [B, H, D]
    k_ptr,           # *fp16 or *fp32, shape [N_total, num_kv_heads, D]
    v_ptr,           # *fp16 or *fp32, shape [N_total, num_kv_heads, D]
    kv_indices_ptr,  # *int32, shape [num_kv_indices]
    kv_indptr_ptr,   # *int32, shape [B+1]
    lse_ptr,         # *float32, shape [B, H]
    out_ptr,         # *float32, shape [B, H, D]
    # meta-parameters
    B: tl.constexpr,           # batch size (meta for shapes, not used in indexing)
    H: tl.constexpr,           # number of query heads
    D: tl.constexpr,           # head dimension
    num_kv_heads: tl.constexpr,  # number of kv heads (meta, not needed for mapping)
    sm_scale,                   # scalar float32 = 1.0 / sqrt(D)
    N_TOTAL: tl.constexpr,     # loop bound (e.g., 128), mask nn < actual_num_tokens
):
    # program ids for batch and head
    b = tl.program_id(0)  # int
    h = tl.program_id(1)  # int

    # Load q[b, h, :] as a D-length vector
    q_base = q_ptr + b * H * D + h * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_vec[d] = tl.load(q_base + d).to(tl.float32)

    # Compute kv_indptr[b] and kv_indptr[b+1] to get the range of tokens for this batch element
    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start

    # First pass: compute logsumexp of logits_scaled = q·k in base-2
    m = -float("inf")
    sumexp = 0.0
    ln2 = 0.6931471805599453  # log(2)
    for nn in range(0, N_TOTAL):
        # Mask: if nn >= actual_num_tokens, skip
        use = nn < actual_num_tokens
        # For masked iterations, dot=0 to avoid NaNs in exp
        dot = 0.0
        if use:
            idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)
            kv_head = h // 4  # GQA mapping: 32/8 => ratio 4
            k_row_ptr = k_ptr + idx * num_kv_heads * D + kv_head * D
            v_row_ptr = v_ptr + idx * num_kv_heads * D + kv_head * D
            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_row_ptr + d).to(tl.float32)
            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]
        logit = dot * sm_scale
        new_m = tl.maximum(m, logit)
        sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit - new_m)
        m = new_m

    # lse for this (b, h) in base-2
    lse_val = m + tl.log(sumexp) / ln2
    tl.store(lse_ptr + b * H + h, lse_val)

    # Second pass: compute softmax per token and accumulate output
    out_base = out_ptr + b * H * D + h * D
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for nn in range(0, N_TOTAL):
        use = nn < actual_num_tokens
        dot = 0.0
        if use:
            idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)
            kv_head = h // 4
            k_row_ptr = k_ptr + idx * num_kv_heads * D + kv_head * D
            v_row_ptr = v_ptr + idx * num_kv_heads * D + kv_head * D
            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_row_ptr + d).to(tl.float32)
            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]
        logit = dot * sm_scale
        softmax = tl.exp(logit - m) / (sumexp * ln2)  # softmax in base-2
        v_vec = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            v_vec[d] = tl.load(v_row_ptr + d).to(tl.float32)
        out_vec += softmax * v_vec
    tl.store(out_base, out_vec)
```
