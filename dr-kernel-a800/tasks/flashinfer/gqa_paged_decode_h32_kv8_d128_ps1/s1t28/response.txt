I will avoid any host-side torch operations (matmul, softmax, logsumexp, elementwise math) and ensure the kernel handles the full attention computation for each (batch b, query head h).

Analysis and plan
- The original code computes attention using a per-(b, h) loop over tokens derived from kv_indptr. It:
  - Asserts num_qo_heads=32, num_kv_heads=8, head_dim=128.
  - Uses GQA mapping: kv_head = h // (num_qo_heads // num_kv_heads) = h // 4.
  - For each b and h, it gathers k_t and v_t for tokens in [kv_indptr[b], kv_indptr[b+1]), computes logits = q[b, h] · k_t, updates a stable LSE, computes attention = softmax(logits_scaled), and accumulates out = sum_j attention_j * v_j.
- Triton-only requirement means:
  - Define and launch one or more @triton.jit kernels from ModelNew.forward.
  - Perform all heavy math (dot products, softmax, logsumexp, accumulation) in-kernel.
  - No host-side torch operations (torch.matmul, torch.softmax, torch.logsumexp, .log, .exp, .softmax, .dot).

Kernel strategy
- One Triton program per (batch b, query head h): grid size = (B * num_qo_heads,).
- Inside the program:
  - Read num_tokens = kv_indptr[b+1] - kv_indptr[b].
  - Initialize output vector (float32) of length head_dim and LSE scalar to -inf.
  - Loop t from 0 to num_tokens-1:
    - idx = kv_indptr[b] + t
    - Load k_t and v_t from k_cache/v_cache for kv_head.
    - Compute q_vec · k_t where q_vec = q[b, h], all in float32.
    - logits = dot(q_vec, k_t)
    - scaled = logits * sm_scale
    - Update LSE stably: new_LSE = max(LSE, scaled) + log(1 + exp(-abs(LSE - scaled)))  (to avoid overflow)
    - attention = exp(scaled - LSE)
    - out_vec += attention * v_t
  - Store out_vec to output[b, h] (cast to bfloat16), and store LSE / ln(2) to lse[b, h].
- We ensure tensors are contiguous and on CUDA, and allocate output tensors in forward.

import torch
import math

# Triton availability
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

# Define Triton kernel for the per-(b, h) attention computation
if TRITON_AVAILABLE:
    @triton.jit
    def _gqa_attention_kernel(
        q_ptr,           # *f32, shape [B, num_qo_heads, HEAD_DIM]
        k_ptr, v_ptr,    # *bf16 or *f16, shape [num_pages, 1, num_kv_heads, HEAD_DIM]
        kv_indptr_ptr,   # *i32, shape [B+1]
        out_ptr,         # *f32, shape [B, num_qo_heads, HEAD_DIM]
        lse_ptr,         # *f32, shape [B, num_qo_heads]
        B,               # int32
        num_qo_heads: tl.constexpr,
        num_kv_heads: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        sm_scale: tl.constexpr,
        gqa_ratio: tl.constexpr,  # num_qo_heads // num_kv_heads = 4
        ln2_inv: tl.constexpr,    # 1 / ln(2)
    ):
        # One program per (b, h)
        pid = tl.program_id(0)
        b = pid // num_qo_heads
        h = pid % num_qo_heads
        if b >= B or h >= num_qo_heads:
            return

        # Token range for this batch
        kv_start = tl.load(kv_indptr_ptr + b)        # i32
        kv_end = tl.load(kv_indptr_ptr + b + 1)     # i32
        num_tokens = kv_end - kv_start              # i32 scalar

        # Load q vector for this (b, h) as float32
        q_offset = b * num_qo_heads * HEAD_DIM + h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_offset)  # [HEAD_DIM] f32

        # Initialize output vector and LSE
        out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
        LSE = -float('inf')

        # Iterate over tokens
        for t in range(0, num_tokens):
            idx = kv_start + t  # token index into [kv_indptr[b], kv_indptr[b+1))

            # Compute kv_head for GQA
            kv_head = h // gqa_ratio  # int32

            # Load k_t and v_t for this token and kv_head; cast to float32
            # k_ptr layout: [num_pages, 1, num_kv_heads, HEAD_DIM] -> offset = idx * (1*num_kv_heads*HEAD_DIM) + kv_head * HEAD_DIM
            k_offset = idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM
            v_offset = idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM

            k_t = tl.load(k_ptr + k_offset)  # [HEAD_DIM], original dtype (bf16/f16), we'll cast in computation
            v_t = tl.load(v_ptr + v_offset)  # [HEAD_DIM]

            # Cast to float32 for compute
            k_t = k_t.to(tl.float32)
            v_t = v_t.to(tl.float32)

            # Compute logits = q_vec · k_t
            # Note: q_vec and k_t are vectors of length HEAD_DIM
            logits = 0.0
            for i in range(HEAD_DIM):
                logits += q_vec[i] * k_t[i]

            scaled = logits * sm_scale  # f32

            # Numerically stable LSE update: new_LSE = max(LSE, scaled) + log(1 + exp(-abs(LSE - scaled)))
            max_ls = tl.maximum(LSE, scaled)
            min_ls = tl.minimum(LSE, scaled)
            diff = max_ls - min_ls
            delta = tl.log(1.0 + tl.exp(-diff))  # log(1 + exp(-|LSE - scaled|))
            LSE = max_ls + delta

            # attention = exp(scaled - LSE)
            attn = tl.exp(scaled - LSE)

            # Accumulate output
            out_vec += attn * v_t

        # Store output and LSE / ln(2)
        out_offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        tl.store(out_ptr + out_offset, out_vec)

        lse_offset = b * num_qo_heads + h
        tl.store(lse_ptr + lse_offset, LSE * ln2_inv)


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):

        assert q.dim() == 3, "q must be [B, num_qo_heads, head_dim]"

        assert kv_indptr.dim() == 1, "kv_indptr must be 1D [B+1]"

        num_pages, _, num_kv_heads, _ = k_cache.shape
        assert num_qo_heads == 32, "num_qo_heads must be 32"

                lse,
                batch_size,

                num_warps=4,
            )
            # Cast output to bfloat16 as in original
            return output.to(torch.bfloat16), lse
        else:

            out = torch.empty_like(q, dtype=torch.bfloat16, device=device)
            for b in range(batch_size):

                kv_end = int(kv_indptr[b + 1].item())
                num_tokens = kv_end - kv_start
                if num_tokens <= 0:
                    out[b].zero_()
                    continue

                    q_vec = q[b, h].to(torch.float32)

                    attn = torch.softmax(logits_scaled, dim=0)

- Triton-Only 요구사항
  - ModelNew.forward는 Triton @triton.jit 커널을 정의하고 호출해야 합니다.
  - 모든 수치 연산 (dot product, softmax, logsumexp, accumulation)은 Triton 내에서 수행되어야 하며, host-side torch 연산 (torch.matmul, torch.softmax, torch.logsumexp, .exp, .log, .softmax)을 사용해서는 안 됩니다.
  - 이 예제는 q의 벡터와 k_t, v_t 간의 dot product와 exp/log를 직접 수행하여 어텐션을 계산합니다.

- 커널 설계
  - 하나의 프로그램이 각 (b, h)를 처리합니다. 그리드 크기는 (B * num_qo_heads,)입니다.
  - 프로그램 내에서:
    - num_tokens = kv_indptr[b+1] - kv_indptr[b]을 읽습니다.
    - q[b, h]를 float32로 로드합니다.
    - out_vec = 0 (float32), LSE = -inf로 초기화합니다.
    - t = 0부터 num_tokens-1까지 반복하며:
      - idx = kv_indptr[b] + t.
      - kv_head = h // 4 (GQA).
      - k_t = k_cache[idx, 0, kv_head, :], v_t = v_cache[idx, 0, kv_head, :]를 로드합니다. 이들은 원래 bf16/f16일 수 있으므로 float32로 캐스팅하여 연산합니다.
      - logits = q_vec · k_t (벡터 내적).
      - scaled = logits * sm_scale.
      - LSE를 numerically stable 방식으로 업데이트: new_LSE = max(LSE, scaled) + log(1 + exp(-abs(LSE - scaled))).
      - attn = exp(scaled - LSE).
      - out_vec += attn * v_t.
    - out_vec을 output[b, h]로 저장하고, LSE / ln(2)을 lse[b, h]로 저장합니다.

- 성능 고려
  - head_dim=128이므로, 벡터 내적은 128개의 반복 연산으로 처리됩니다. Triton은 이러한 로컬 벡터 연산을 지원합니다.
  - 전체 처리 프로그램 수는 B*32개로, 각 프로그램은 해당 (b, h)의 token range을 순회하며 독립적으로 계산합니다. 이는 병렬성을 제공합니다.
  - 만약 token range이 매우 커진다면, 추가로 분할할 수 있지만, eval workload에서는 일반적으로 작을 것으로 예상됩니다.

- 디버깅 및 robustness
  - 코드는 Triton이 사용 가능한 경우에만 커널을 실행합니다. 그렇지 않으면 fallback을 제공할 수 있습니다.
  - input tensors를 contiguous로 만들어 계산을 단순화합니다.
  - asserts로 원본 코드의 조건을 확인합니다.

import torch
import math

# Triton availability
try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False

# Triton kernel: per-(b, h) attention computation
if TRITON_AVAILABLE:
    @triton.jit
    def _gqa_attention_kernel(
        q_ptr,           # *f32, shape [B, num_qo_heads, HEAD_DIM]
        k_ptr, v_ptr,    # *bf16 or *f16, shape [num_pages, 1, num_kv_heads, HEAD_DIM]
        kv_indptr_ptr,   # *i32, shape [B+1]
        out_ptr,         # *f32, shape [B, num_qo_heads, HEAD_DIM]
        lse_ptr,         # *f32, shape [B, num_qo_heads]
        B,               # int32
        num_qo_heads: tl.constexpr,
        num_kv_heads: tl.constexpr,
        HEAD_DIM: tl.constexpr,
        sm_scale: tl.constexpr,
        gqa_ratio: tl.constexpr,   # = 4
        ln2_inv: tl.constexpr,     # 1 / ln(2)
    ):
        # One program per (b, h)
        pid = tl.program_id(0)
        b = pid // num_qo_heads
        h = pid % num_qo_heads
        if b >= B or h >= num_qo_heads:
            return

        # Token range for this batch
        kv_start = tl.load(kv_indptr_ptr + b)        # i32
        kv_end = tl.load(kv_indptr_ptr + b + 1)     # i32
        num_tokens = kv_end - kv_start              # i32 scalar

        # Load q vector for this (b, h) as float32
        q_offset = b * num_qo_heads * HEAD_DIM + h * HEAD_DIM
        q_vec = tl.load(q_ptr + q_offset)  # [HEAD_DIM] f32

        # Initialize output vector and LSE
        out_vec = tl.zeros((HEAD_DIM,), dtype=tl.float32)
        LSE = -float('inf')

        # Iterate over tokens
        for t in range(0, num_tokens):
            idx = kv_start + t  # token index into [kv_indptr[b], kv_indptr[b+1))

            # GQA mapping
            kv_head = h // gqa_ratio  # int32

            # Load k_t and v_t for this token and kv_head; cast to float32
            # k_ptr layout: [num_pages, 1, num_kv_heads, HEAD_DIM]
            # offset for k_ptr at (idx, 0, kv_head, :) is idx * (1*num_kv_heads*HEAD_DIM) + kv_head * HEAD_DIM
            k_offset = idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM
            v_offset = idx * (num_kv_heads * HEAD_DIM) + kv_head * HEAD_DIM

            k_t = tl.load(k_ptr + k_offset)  # [HEAD_DIM]
            v_t = tl.load(v_ptr + v_offset)  # [HEAD_DIM]

            k_t = k_t.to(tl.float32)
            v_t = v_t.to(tl.float32)

            # Compute logits = q_vec · k_t
            logits = 0.0
            for i in range(HEAD_DIM):
                logits += q_vec[i] * k_t[i]

            scaled = logits * sm_scale  # f32

            # Numerically stable LSE update: new_LSE = max(LSE, scaled) + log(1 + exp(-abs(LSE - scaled)))
            max_ls = tl.maximum(LSE, scaled)
            min_ls = tl.minimum(LSE, scaled)
            diff = max_ls - min_ls
            LSE = max_ls + tl.log(1.0 + tl.exp(-diff))

            # attention = exp(scaled - LSE)
            attn = tl.exp(scaled - LSE)

            # Accumulate output
            out_vec += attn * v_t

        # Store output and LSE / ln(2)
        out_offset = b * (num_qo_heads * HEAD_DIM) + h * HEAD_DIM
        tl.store(out_ptr + out_offset, out_vec)

        lse_offset = b * num_qo_heads + h
        tl.store(lse_ptr + lse_offset, LSE * ln2_inv)
