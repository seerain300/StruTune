- In pass 1, processes tokens in fixed-size blocks (BLOCK_T=128) using vectorized indices. We compute q_vec @ k_block where k_block is a [BLOCK_T, D] tensor of k rows for that block (loaded from memory and cast to fp32). This avoids per-token loops and should compile reliably.

- In pass 2, we loop over blocks again and compute attn per token, then accumulate out_vec += attn * v_row for that token. This keeps everything in Triton and avoids unsupported control flow.
- Passes sm_scale and uses tl.constexpr for H, D, K, gqa_ratio to help Triton specialize. While Triton doesn’t allow dynamic while loops, using BLOCK_T and tl.arange with masks allows vectorized processing without Python-level while loops for the inner dimension.

import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def fused_gqa_row_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [Np, 1, K, D]
    v_ptr,            # *bf16, [Np, 1, K, D]
    indptr_ptr,       # *int32, [B+1]
    indices_ptr,      # *int32, [num_tokens_total]
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *fp32, [B, H]
    B,                # int
    H: tl.constexpr,      # 32
    D: tl.constexpr,      # 128
    K: tl.constexpr,      # 8
    gqa_ratio: tl.constexpr,  # H // K == 4
    BLOCK_T: tl.constexpr,     # e.g., 128 (must divide num_tokens for ideal performance, but mask handles remainder)
    sm_scale,             # fp32 scalar
):
    pid_b = tl.program_id(0)  # batch index
    pid_h = tl.program_id(1)  # query head index

    # Load q vector for this (b, h) in fp32
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D] fp32 vector

    # Load indptr[start, end] for this batch element
    start = tl.load(indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Pass 1: compute sum_exp = sum(exp(logits_scaled)) across tokens in blocks
    sum_exp = 0.0
    t = 0
    while t < num_tokens:
        t_offsets = t + tl.arange(0, BLOCK_T)  # [BLOCK_T]
        mask_t = t_offsets < num_tokens        # [BLOCK_T]
        # Load token indices for this block
        idxs = tl.load(indices_ptr + start + t_offsets, mask=mask_t, other=0).to(tl.int32)  # [BLOCK_T]

        # Compute K_block: [BLOCK_T, D] in fp32
        kv_head = pid_h // gqa_ratio  # 0..7
        # Pointers for each row: shape [BLOCK_T]
        k_row_ptrs = k_ptr + idxs * (K * D) + kv_head * D  # [BLOCK_T]
        # Load rows and cast to fp32
        K_block = tl.load(k_row_ptrs[:, None] + tl.arange(0, D), mask=mask_t[:, None], other=0.0).to(tl.float32)  # [BLOCK_T, D]
        # Compute dot products for the block: [BLOCK_T]
        dots_block = tl.sum(K_block * q_vec[None, :], axis=1)  # [BLOCK_T]
        # sum_exp += sum(exp(dots * sm_scale)) over valid tokens
        sum_exp += tl.sum(tl.exp(dots_block * sm_scale), axis=0)
        t += BLOCK_T

    # lse in base-2: log2(sum_exp) = log(sum_exp) / ln(2)
    lse_acc = tl.log(sum_exp) * 1.4426950408889634  # 1/ln(2)

    # Pass 2: compute output vector out[b, h, :] (fp32), then cast to bf16
    out_vec = tl.zeros((D,), dtype=tl.float32)
    t = 0
    while t < num_tokens:
        t_offsets = t + tl.arange(0, BLOCK_T)  # [BLOCK_T]
        mask_t = t_offsets < num_tokens        # [BLOCK_T]
        idxs = tl.load(indices_ptr + start + t_offsets, mask=mask_t, other=0).to(tl.int32)  # [BLOCK_T]
        kv_head = pid_h // gqa_ratio

        # Pointers for each row: shape [BLOCK_T]
        k_row_ptrs = k_ptr + idxs * (K * D) + kv_head * D
        v_row_ptrs = v_ptr + idxs * (K * D) + kv_head * D

        # Load K and V rows for this block
        k_rows = tl.load(k_row_ptrs[:, None] + tl.arange(0, D), mask=mask_t[:, None], other=0.0).to(tl.float32)  # [BLOCK_T, D]
        v_rows = tl.load(v_row_ptrs[:, None] + tl.arange(0, D), mask=mask_t[:, None], other=0.0).to(tl.float32)  # [BLOCK_T, D]

        # Compute logits_scaled and attention for the block, accumulate into out_vec
        dots_block = tl.sum(k_rows * q_vec[None, :], axis=1)  # [BLOCK_T]
        attn_block = tl.exp((dots_block - lse_acc) * sm_scale)  # [BLOCK_T]
        # For each token in the block, add attn * v_row to out_vec
        # We do this elementwise across D:
        for di in range(D):
            out_vec[di] += tl.sum(attn_block * v_rows[:, di], axis=0)
        t += BLOCK_T

    # Store output as bfloat16
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))

    # Store lse (base-2)
    lse_offset = pid_b * H + pid_h
    tl.store(lse_ptr + lse_offset, (lse_acc * 1.4426950408889634))  # already base-2


class ModelNew(torch.nn.Module):
    def forward(self, q, k_cache, v_cache, kv_indptr, kv_indices, sm_scale):
        # Triton must be available and tensors on CUDA
        if not TRITON_AVAILABLE:
            raise RuntimeError("Triton is not available")
        device = q.device
        assert device.type == "cuda", "Inputs must be on CUDA device"

        # Shapes and asserts
        B, H, D = q.shape
        assert H == 32 and D == 128, "This Triton implementation expects H=32 and D=128"
        assert k_cache.shape[1] == 1 and v_cache.shape[1] == 1, "num_pages must be 1"
        Np, _, K, _ = k_cache.shape
        assert K == 8, "This Triton implementation expects K=8"
        assert kv_indptr.shape[0] == B + 1, "kv_indptr shape must be [B+1]"

        # Ensure contiguity
        q = q.contiguous()
        k_cache = k_cache.contiguous()
        v_cache = v_cache.contiguous()
        kv_indptr = kv_indptr.contiguous()
        kv_indices = kv_indices.contiguous()

        # Output buffers
        out = torch.empty((B, H, D), dtype=torch.bfloat16, device=device)
        lse = torch.empty((B, H), dtype=torch.float32, device=device)

        # Launch one program per (b, h)
        grid = (B, H)
        fused_gqa_row_kernel[grid](
            q, k_cache, v_cache, kv_indptr, kv_indices, out, lse,
            B, H=32, D=128, K=8, gqa_ratio=4, BLOCK_T=128, sm_scale=float(sm_scale),
        )
        return out, lse

import torch

    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def fused_gqa_row_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [Np, 1, K, D]
    v_ptr,            # *bf16, [Np, 1, K, D]
    indptr_ptr,       # *int32, [B+1]
    indices_ptr,      # *int32, [num_tokens_total]
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *fp32, [B, H]
    B,                # int
    H: tl.constexpr,      # 32
    D: tl.constexpr,      # 128
    K: tl.constexpr,      # 8
    gqa_ratio: tl.constexpr,  # 4
    sm_scale,             # fp32 scalar
):
    pid_b = tl.program_id(0)  # batch index
    pid_h = tl.program_id(1)  # query head index

    # Load q vector for this (b, h) in fp32
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D] fp32 vector

    # Load indptr[start, end] for this batch element
    start = tl.load(indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Pass 1: compute sum_exp = sum(exp(logits_scaled)) across tokens
    sum_exp = 0.0
    t = 0
    while t < num_tokens:
        idx = tl.load(indices_ptr + start + t).to(tl.int32)  # scalar token index
        kv_head = pid_h // gqa_ratio  # 0..7

        # Load K row for this token and head (1D vector) in fp32
        k_row_ptr = k_ptr + idx * (K * D) + kv_head * D
        k_row = tl.load(k_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]

        # dot(q_vec, k_row)
        dot = 0.0
        d = 0
        while d < D:
            dot += q_vec[d] * k_row[d]
            d += 1

        sum_exp += tl.exp(dot * sm_scale)
        t += 1

    # lse in base-2: log2(sum_exp) = log(sum_exp) / ln(2)
    lse_acc = tl.log(sum_exp) * 1.4426950408889634  # 1/ln(2)

    # Pass 2: compute output vector out[b, h, :] (fp32), then cast to bf16
    out_vec = tl.zeros((D,), dtype=tl.float32)
    t = 0
    while t < num_tokens:
        idx = tl.load(indices_ptr + start + t).to(tl.int32)
        kv_head = pid_h // gqa_ratio

        # Load K and V rows for this token and head
        k_row_ptr = k_ptr + idx * (K * D) + kv_head * D
        v_row_ptr = v_ptr + idx * (K * D) + kv_head * D

        k_row = tl.load(k_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]
        v_row = tl.load(v_row_ptr + tl.arange(0, D), mask=tl.arange(0, D) < D, other=0.0).to(tl.float32)  # [D]

        # Compute logits_scaled and attention
        dot = 0.0
        d = 0
        while d < D:
            dot += q_vec[d] * k_row[d]
            d += 1

        attn = tl.exp((dot - lse_acc) * sm_scale)

        # Accumulate output vector
        d = 0
        while d < D:
            out_vec[d] += attn * v_row[d]
            d += 1

        t += 1

    # Store output as bfloat16
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))

    # Store lse (base-2)
    lse_offset = pid_b * H + pid_h
    tl.store(lse_ptr + lse_offset, lse_acc * 1.4426950408889634)  # already base-2
