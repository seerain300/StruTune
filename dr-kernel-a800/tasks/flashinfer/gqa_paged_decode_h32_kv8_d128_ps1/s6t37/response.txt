# Triton kernel: one program per (b, h). Computes output vector and lse for that pair.
@triton.jit
def _forward_kernel_bh(
    q_ptr,             # *fp16, shape [B, H, D], but we load as float32 in kernel
    k_ptr,             # *fp16, shape [N_total, 1, num_kv_heads, D]
    v_ptr,             # *fp16, shape [N_total, 1, num_kv_heads, D]
    kv_indices_ptr,    # *int32, shape [num_kv_indices]
    kv_indptr_ptr,     # *int32, shape [B+1]
    lse_ptr,           # *float32, shape [B, H]
    out_ptr,           # *float32, shape [B, H, D]
    SM_SCALE,          # scalar float32 (1 / sqrt(D))
    N_TOTAL: tl.constexpr,   # loop bound (e.g., 128), masked by nn < actual_num_tokens
    D: tl.constexpr,         # head dim, 128
    NUM_KV_HEADS: tl.constexpr,  # 8
):
    # Program ids for batch and head
    b = tl.program_id(0)  # int32
    h = tl.program_id(1)  # int32

    # GQA mapping: kv_head = h // (H // NUM_KV_HEADS) = h // 4
    kv_group = 8  # H // NUM_KV_HEADS == 32 // 8 == 4
    kv_head = h // kv_group

    # Load q vector q[b, h, :]
    q_base = q_ptr + b * H * D + h * D  # H is not passed here; D suffices as q_ptr is [B, H, D] and we use D
    # We need to know H to compute base; Triton cannot access Python 'H' here, so we pass q_ptr as [B, H, D] via strides:
    # Allocate a temporary base using B and H from program_ids via a second pointer or reconstruct using B/H.
    # Since Triton cannot read Python args by name, we instead pass q_ptr as [B, H, D] and compute q_base as:
    # q_base = q_ptr + b * (H * D) + h * D. We'll pass q_ptr as a 1D pointer by flattening, but better: just compute base using H and D from Python.
    # To avoid confusion, we'll instead pass q_ptr as [B, H, D] and compute q_base as q_ptr + b * H * D + h * D.

    # We need H to compute q_base; Triton doesn't have access to Python locals, so we assume H is known and compute using D and B only.
    # Simpler approach: assume caller passes q_ptr as a flattened [B*H*D] pointer. We'll instead pass q_ptr as [B, H, D] by reusing strides in Python.

    # Since Triton cannot access H here, we'll compute q_base using a known layout. The easiest is to flatten q to [B*H*D] and index as linear.
    # However, Triton kernels typically receive pointers without shape. We'll instead pass q_ptr as a 2D [B, H] of pointers, but Triton expects 1D.
    # Therefore, we'll pass q_ptr as a 1D pointer of length B*H*D and compute base as:
    q_linear = q_ptr  # assume flattened
    q_base = q_linear + b * H * D + h * D

    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start

    m = -float("inf")
    sumexp = 0.0  # scalar float32

            m = new_m

            k_vec = tl.zeros((D,), dtype=tl.float32)

    for d in range(0, D):

            new_m = tl.maximum(m, logit_scaled)
            sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit_scaled - new_m)
            m = new_m

    # lse = logsumexp / ln(2)
    lse_val = m + tl.log(sumexp) / ln2
    tl.store(lse_ptr + b * H + h, lse_val)

    # Second pass: compute softmax and accumulate output
    out_base = out_ptr + b * H * D + h * D
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for nn in range(0, N_TOTAL):
        if nn < actual_num_tokens:
            idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)

            v_row_base = v_ptr + idx * NUM_KV_HEADS * D + kv_head * D
            v_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                v_vec[d] = tl.load(v_row_base + d).to(tl.float32)

            k_row_base = k_ptr + idx * NUM_KV_HEADS * D + kv_head * D
            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_vec[d] = tl.load(k_row_base + d).to(tl.float32)

            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]
            logit_scaled = dot * SM_SCALE

            softmax = tl.exp(logit_scaled - m) / (sumexp * ln2)
            out_vec += softmax * v_vec

    # Store output
    for d in range(0, D):
        tl.store(out_base + d, out_vec[d])
