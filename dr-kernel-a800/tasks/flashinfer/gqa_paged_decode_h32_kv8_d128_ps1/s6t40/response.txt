import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:

# Triton kernel: one program per (b, h)
# Computes:
# - q_vec = q[b, h, :]
# - For tokens n in [kv_indptr[b], kv_indptr[b+1)):
#     - k_row = k_cache[indices[n], kv_head, :]
#     - v_row = v_cache[indices[n], kv_head, :]
#     - dot = q_vec · k_row
#     - logit = dot * sm_scale
#     - Accumulate logsumexp in base-2: lse = logsumexp(logit) / ln(2)
#     - attn = softmax(logit)
#     - out[b, h, :] += attn * v_row
@triton.jit
def _forward_kernel_bh(
    q_ptr,                  # *fp16, [B, H, D]
    k_ptr,                  # *fp16, [N_total, 1, num_kv_heads, D] (but we gather by indices)
    v_ptr,                  # *fp16, [N_total, 1, num_kv_heads, D]
    kv_indices_ptr,         # *int32, [num_kv_indices]
    kv_indptr_ptr,          # *int32, [B+1]
    lse_ptr,                # *float32, [B, H]
    out_ptr,                # *float32, [B, H, D] (compute in fp32 and store as fp32; cast to fp16 in host after kernel)
    B: tl.int32,
    H: tl.int32,
    D: tl.constexpr,        # e.g., 128
    num_kv_heads: tl.int32, # e.g., 8
    sm_scale: tl.float32,   # 1/sqrt(D) in fp32
    N_TOTAL: tl.constexpr,  # e.g., 128, loop bound with mask
):
    b = tl.program_id(0)
    h = tl.program_id(1)

    # Load q[b, h, :] into fp32
    q_base = q_ptr + b * H * D + h * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_val = tl.load(q_base + d)
        q_vec[d] = q_val.to(tl.float32)

    # Compute start/end for this batch b
    start = tl.load(kv_indptr_ptr + b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + b + 1).to(tl.int32)
    actual_num_tokens = end - start  # int

    # Compute kv_head for GQA
    kv_head = h // (H // num_kv_heads)  # int

    # First pass: streaming logsumexp in base-2
    m = -float("inf")  # scalar
    sumexp = 0.0       # scalar
    ln2 = 0.6931471805599453
    for nn in range(0, N_TOTAL):
        valid = nn < actual_num_tokens
        # idx of the token
        idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)

        # k_row_ptr: k_cache[idx, kv_head, :]

        for d in range(0, D):
            k_val = tl.load(k_row_ptr + d)
            k_row[d] = k_val.to(tl.float32)

        dot = 0.0
        for d in range(0, D):
            dot += q_vec[d] * k_row[d]

        logit = dot * sm_scale
        new_m = tl.maximum(m, logit)
        sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit - new_m)
        m = new_m

    # lse[b, h] = logsumexp(logit) / ln(2) = m + log(sumexp) / ln2
    lse_val = m + tl.log(sumexp) / ln2
    tl.store(lse_ptr + b * H + h, lse_val)

    # Second pass: compute softmax and accumulate output
    out_base = out_ptr + b * H * D + h * D
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for nn in range(0, N_TOTAL):
        valid = nn < actual_num_tokens
        idx = tl.load(kv_indices_ptr + start + nn).to(tl.int32)
        k_row_ptr = k_ptr + idx * num_kv_heads * D + kv_head * D

        v_row = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            v_val = tl.load(v_row_ptr + d)
            v_row[d] = v_val.to(tl.float32)

        k_row = tl.zeros((D,), dtype=tl.float32)
        for d in range(0, D):
            k_val = tl.load(k_row_ptr + d)
            k_row[d] = k_val.to(tl.float32)
        dot = 0.0
        for d in range(0, D):
            dot += q_vec[d] * k_row[d]
        logit = dot * sm_scale

        # softmax over tokens
        softmax = tl.exp(logit - m) / (sumexp * ln2)
        out_vec += v_row * softmax

    # Store result
    for d in range(0, D):
        tl.store(out_base + d, out_vec[d])

            q_perm, k_cache, v_cache, kv_indices, kv_indptr, lse, output_fp32,
            H=H, D=D, B=B, num_kv_heads=num_kv_heads, sm_scale=float(sm_scale), N_TOTAL=128,
        )

        # Return output as bfloat16 (matching original), and lse as float32.
        output_bf16 = output_fp32.to(torch.bfloat16)
        return output_bf16, lse

我的实现
- 重新实现的核心是：对于每一个 (b, h) 对，计算 q_vec = q[b, h, :]，然后根据 kv_indptr[b:b+1] 确定token范围，遍历token indices，对于每个token n获取 k_row = k_cache[indices[n], kv_head, :], v_row = v_cache[indices[n], kv_head, :], 计算 dot = q_vec · k_row，logits_scaled = dot * sm_scale，更新 logsumexp in base-2，计算 softmax，然后累加 output[b, h, :] += attn * v_row。最后返回 output (bfloat16) 和 lse (float32)。

- Triton kernel _forward_kernel_bh: grid=(B, H). 一个程序实例处理一个 (b, h) 对。它：
  - 通过 q_ptr + b*H*D + h*D 加载 q_vec (fp32)。
  - 从 kv_indptr 读取 start/end，并使用 N_TOTAL (e.g., 128) 作为循环边界，nn < actual_num_tokens 作为mask，加载 k_row 和 v_row (fp16 -> fp32)。
  - 第一遍循环计算 logsumexp (base-2) 的 m 和 sumexp (streaming update)。
  - 存储 lse[b, h] = m + log(sumexp)/ln2。
  - 第二遍循环重新计算 logits，得到 softmax = exp(logits - m) / (sumexp * ln2) 并累加 output[b, h, :] += v_row * softmax。

- ModelNew.forward: 不进行任何 PyTorch math，在这里只做指针传递和 kernel[grid](...) 发射，并返回 bfloat16 output 和 float32 lse。

- 重要：不要 q.permute() 也不要做任何数据排列/变换 —— 直接基于传入的 q, k_cache, v_cache 的指针进行计算！
