import math
import torch
import triton
import triton.language as tl


@triton.jit
def softmax_and_attention_single_bh(
    q_ptr,             # *fp32 [B, num_qo_heads, D]
    k_ptr,             # *fp32 [num_tokens_b, H, D]
    v_ptr,             # *fp32 [num_tokens_b, H, D]
    out_ptr,           # *fp32 [B, num_qo_heads, D]
    lse_ptr,           # *fp32 [B, num_qo_heads]
    B: tl.constexpr,   # batch size (not used in kernel but for clarity)
    num_tokens_b,      # int32 scalar: number of tokens for this batch element
    sm_scale,          # fp32 scalar: 1/sqrt(D)
    H,                 # int32: num_kv_heads
    D: tl.constexpr,   # head_dim, compile-time for vector ops
    gqa_ratio: tl.constexpr,  # num_qo_heads // num_kv_heads (4)
):
    # program ids
    b = tl.program_id(0)
    h = tl.program_id(1)

    # compute kv head for GQA
    kv_head = h // gqa_ratio  # 0..7

    # pointers to q vector for this (b, h): q[b, h, :]
    # q is [B, num_qo_heads, D], row-major (we pass it as contiguous fp32)
    q_row_start = b * (num_qo_heads * D) + h * D
    q_vec = tl.load(q_ptr + q_row_start + tl.arange(0, D))

    # Initialize numerically stable logsumexp
    running_max = -float("inf")
    running_sum = 0.0

    # First pass: compute lse over scaled logits for all tokens
    t = 0
    while t < num_tokens_b:
        # K_t[t, kv_head, :] and V_t[t, kv_head, :]
        # k_ptr is [num_tokens_b, H, D], contiguous in last dim
        k_row_start = t * (H * D) + kv_head * D
        v_row_start = t * (H * D) + kv_head * D

        k_vec = tl.load(k_ptr + k_row_start + tl.arange(0, D))
        v_vec = tl.load(v_ptr + v_row_start + tl.arange(0, D))

        # logits_t = dot(q_vec, k_vec)
        logits = tl.sum(q_vec * k_vec, axis=0)  # scalar
        scaled = logits * sm_scale
        # update running_max and running_sum
        # if scaled > running_max: running_sum = running_sum * exp(running_max - scaled) + 1; running_max = scaled
        # else: running_sum += exp(scaled - running_max)
        if scaled > running_max:
            running_sum = running_sum * tl.exp(running_max - scaled) + 1.0
            running_max = scaled
        else:
            running_sum += tl.exp(scaled - running_max)
        t += 1

    # lse = log(running_sum) + running_max, then divide by ln(2)
    # ln(2) = 0.6931471805599453
    log2_inv = 1.4426950408889634
    lse_val = tl.log(running_sum) + running_max
    lse_val = lse_val * log2_inv
    tl.store(lse_ptr + b * num_qo_heads + h, lse_val)

    # Second pass: compute output[b, h, :] = sum_j exp(scaled_j) * v_j
    out_vec = tl.zeros([D], dtype=tl.float32)
    t = 0
    while t < num_tokens_b:
        k_row_start = t * (H * D) + kv_head * D
        v_row_start = t * (H * D) + kv_head * D

        k_vec = tl.load(k_ptr + k_row_start + tl.arange(0, D))
        v_vec = tl.load(v_ptr + v_row_start + tl.arange(0, D))

        logits = tl.sum(q_vec * k_vec, axis=0)
        scaled = logits * sm_scale
        attn = tl.exp(scaled - lse_val)  # scalar
        out_vec += attn * v_vec
        t += 1

    # store output vector
    out_row_start = b * (num_qo_heads * D) + h * D
    tl.store(out_ptr + out_row_start + tl.arange(0, D), out_vec)

        assert k_cache.shape[2] == D
        assert v_cache.shape == (num_pages, H, D)

        output = torch.empty((B, num_qo_heads, D), dtype=torch.float32, device=q.device)  # fp32 compute
        lse = torch.empty((B, num_qo_heads), dtype=torch.float32, device=q.device)

        # However, Triton requires scalar arguments; we can pass num_tokens_b from host by building a small per-batch table.
        # But since Triton while loops can use runtime scalars, we can compute num_tokens_b in Python and pass to kernel.

        # Compute num_tokens per batch element: num_tokens_b = kv_indptr[b+1] - kv_indptr[b]
        # Note: kv_indptr is typically int32, but may be int64; cast to int64 for safe arithmetic
        # We only need per-batch counts; create a list
        num_tokens_per_batch = []
        for b in range(B):
            start = int(kv_indptr[b].item())
            end = int(kv_indptr[b + 1].item())
            num_tokens_per_batch.append(end - start)
        # Prepare token_indices for each batch element: kv_indices[start:start+num_tokens_b]
        # We'll gather K/V per batch and pass to kernel.

        # Launch Triton kernel: one program per (b, h)
        grid = (B, num_qo_heads)
        softmax_and_attention_single_bh[grid](
            q.to(torch.float32),          # ensure fp32 inputs

            output,
            lse,
            B,

            sm_scale,
            H,
            D,
            num_qo_heads // H,            # gqa_ratio = 4
            num_warps=4,
        )

        # Cast output to bfloat16 to match original behavior
        output = output.to(torch.bfloat16)
        return output, lse
