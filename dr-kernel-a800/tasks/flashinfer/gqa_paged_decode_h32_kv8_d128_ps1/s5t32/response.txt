    k_ptr,          # *bf16, shape [Np, 1, K, D] (Np is runtime, we pass 1 as in inputs)
    v_ptr,          # *bf16, shape [Np, 1, K, D]
    kv_indptr_ptr,  # *int32, shape [B+1]
    kv_indices_ptr, # *int32, shape [num_kv_indices]
    out_ptr,        # *bf16, shape [B, H, D]
    lse_ptr,        # *float32, shape [B, H]
    sm_scale: tl.constexpr,    # scalar, e.g., 1/sqrt(128)
    B: tl.constexpr,           # batch size (constexpr for specialization)
    H: tl.constexpr,           # num_qo_heads (constexpr)
    D: tl.constexpr,           # head_dim (constexpr, 128)
    K: tl.constexpr,           # num_kv_heads (constexpr, 8)
    gqa_ratio: tl.constexpr,   # H // K (constexpr, 4)
    MAX_TOKENS: tl.constexpr,  # fixed loop bound (e.g., 1024)
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)

    # Load token range for this batch element
    start = tl.load(kv_indptr_ptr + pid_b).to(tl.int32)
    end = tl.load(kv_indptr_ptr + pid_b + 1).to(tl.int32)
    num_tokens = end - start

    # Load q vector for this (b, h) and cast to fp32 for computation
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Accumulator for sum_exp over tokens
    sum_exp = 0.0

    # Pass 1: compute sum_exp = sum(exp((q·k) * sm_scale))
    for t in range(MAX_TOKENS):
        if t < num_tokens:
            # idx = kv_indptr[b] + t
            idx = start + t
            k_idx = tl.load(kv_indices_ptr + idx).to(tl.int32)
            kv_h = pid_h // gqa_ratio
            # Load k_row [D]
            k_row = tl.load(k_ptr + k_idx * (K * D) + kv_h * D).to(tl.float32)
            # dot product
            dot = 0.0
            for d in range(D):
                dot += q_vec[d] * k_row[d]
            logit_scaled = dot * sm_scale
            sum_exp += tl.exp(logit_scaled)

    # Compute lse in base-2
    lse_val = tl.log(sum_exp) / 0.6931471805599453  # ln(2)

    # Accumulator for output vector
    out_vec = tl.zeros((D,), dtype=tl.float32)

    # Pass 2: compute output = sum_t exp((logit_scaled - lse) * sm_scale) * v_row

            idx = start + t

            logit_scaled = dot * sm_scale
            attn = tl.exp((logit_scaled - lse_val) * sm_scale)
            v_row = tl.load(v_ptr + k_idx * (K * D) + kv_h * D).to(tl.float32)
            for d in range(D):
                out_vec[d] += attn * v_row[d]

    # Store output vector (bf16) and lse (float32)
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))
    tl.store(lse_ptr + pid_b * H + pid_h, lse_val)
