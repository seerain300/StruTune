import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def gqa_row_triton_kernel(
    q_ptr,            # *bf16, shape [B, H, D]
    k_ptr,            # *bf16, shape [Np, 1, K, D]
    v_ptr,            # *bf16, shape [Np, 1, K, D]
    indptr_ptr,       # *int32, shape [B+1]
    indices_ptr,      # *int32, shape [num_kv_indices]
    out_ptr,          # *bf16, shape [B, H, D]
    lse_ptr,          # *float32, shape [B, H]
    sm_scale,         # float32 scalar
    B: tl.constexpr, H: tl.constexpr, D: tl.constexpr, K: tl.constexpr, GR: tl.constexpr, MAX_TOKENS: tl.constexpr,
):
    # program id corresponds to (b, h)
    pid = tl.program_id(axis=0)
    b = pid // H
    h = pid % H
    kv_head = h // GR

    # Load Q vector for this (b, h)
    q_offset = b * H * D + h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D], fp32 for stability

    # Load indptr[start, end] for this batch element
    start = tl.load(indptr_ptr + b).to(tl.int32)
    end = tl.load(indptr_ptr + b + 1).to(tl.int32)
    num_tokens = end - start

    # Pass 1: compute sum_exp over tokens
    sum_exp = 0.0
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = tl.load(indices_ptr + t).to(tl.int32)
        # k[:, kv_head, :] for this token idx -> shape [K, D] but we only need kv_head row
        # Memory layout: [Np, 1, K, D] -> contiguous over D
        k_row_base = idx * (1 * K * D)  # since squeeze(1) => shape [Np, K, D]
        k_vec = tl.load(k_ptr + k_row_base + kv_head * D + tl.arange(0, D)).to(tl.float32)
        logits = tl.sum(q_vec * k_vec, axis=0)
        sum_exp += tl.exp(logits * sm_scale)
    lse = tl.log(sum_exp) / 1.4426950408889634  # log2(sum_exp) = ln(sum_exp) / ln(2)

    # Prepare output vector
    out_vec = tl.zeros((D,), dtype=tl.float32)

    # Pass 2: compute output
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = tl.load(indices_ptr + t).to(tl.int32)
        k_row_base = idx * (1 * K * D)
        k_vec = tl.load(k_ptr + k_row_base + kv_head * D + tl.arange(0, D)).to(tl.float32)
        logits = tl.sum(q_vec * k_vec, axis=0)
        attn = tl.exp((logits - lse) * sm_scale)
        v_row_base = idx * (1 * K * D)
        v_vec = tl.load(v_ptr + v_row_base + kv_head * D + tl.arange(0, D)).to(tl.float32)
        out_vec += attn * v_vec

    # Store results
    out_offset = b * H * D + h * D
    tl.store(out_ptr + out_offset, out_vec.to(tl.bfloat16))
    tl.store(lse_ptr + b * H + h, lse)
