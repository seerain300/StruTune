import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


@triton.jit
def fused_gqa_row_kernel(
    q_ptr,          # *bf16, [B, H, D]
    k_ptr,          # *bf16, [Np, 1, K, D]
    v_ptr,          # *bf16, [Np, 1, K, D]
    tok_idx_ptr,    # *int32, [B], each entry is num_tokens for that batch
    tokens_ptr,     # *int32, [sum_b num_tokens[b]], flattened token indices per batch (see host for ordering)
    out_ptr,        # *float32, [B, H, D]
    lse_ptr,        # *float32, [B, H]
    sm_scale,       # float32 scalar
    B: tl.constexpr,
    H: tl.constexpr,
    D: tl.constexpr,
    K: tl.constexpr,
    gqa_ratio: tl.constexpr,
    MAX_TOKENS: tl.constexpr,  # large enough to cover typical num_tokens (e.g., 1024)
):
    pid_b = tl.program_id(0)  # batch id
    pid_h = tl.program_id(1)  # query head id

    # Bounds check for program ids (in case grid is larger)
    if (pid_b >= B) or (pid_h >= H):
        return

    # Load q vector for this (b, h)
    q_offset = pid_b * H * D + pid_h * D
    q_vec = tl.load(q_ptr + q_offset).to(tl.float32)  # [D]

    # Compute kv_head for GQA
    kv_head = pid_h // gqa_ratio

    # Load number of tokens for this batch element
    num_tokens = tl.load(tok_idx_ptr + pid_b).to(tl.int32)

    # Accumulators
    sum_exp = 0.0  # for logsumexp (pass 1)

    # Pass 1: compute sum_exp over all tokens
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        # Get token index for this batch
        tok_idx = tl.load(tokens_ptr + t).to(tl.int32)

        # Compute k_ptr offset: ((tok_idx * K) + kv_head) * D
        k_offset = tok_idx * K * D + kv_head * D
        k_vec = tl.load(k_ptr + k_offset).to(tl.float32)  # [D]

        # Dot product q·k
        dot = 0.0
        for d in range(D):
            dot += q_vec[d] * k_vec[d]
        # Scale and accumulate
        sum_exp += tl.exp(dot * sm_scale)

    # lse in base-2
    lse = tl.log(sum_exp) / math.log(2.0)
    # Store lse for this (b, h)
    tl.store(lse_ptr + pid_b * H + pid_h, lse)

    # Pass 2: compute output vector for this (b, h)
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        tok_idx = tl.load(tokens_ptr + t).to(tl.int32)
        k_offset = tok_idx * K * D + kv_head * D
        k_vec = tl.load(k_ptr + k_offset).to(tl.float32)  # [D]
        v_offset = tok_idx * K * D + kv_head * D
        v_vec = tl.load(v_ptr + v_offset).to(tl.float32)  # [D]

        dot = 0.0
        for d in range(D):
            dot += q_vec[d] * k_vec[d]
        attn = tl.exp((dot - lse) * sm_scale)  # softmax scaled attention for this token

        # Fused matmul: out_vec += attn * v_vec
        # This is elementwise multiply-add since both are 1D vectors
        for d in range(D):
            out_vec[d] += attn * v_vec[d]

    # Store output vector for this (b, h)
    out_offset = pid_b * H * D + pid_h * D
    tl.store(out_ptr + out_offset, out_vec)

        """

        # Precompute num_tokens[b] = kv_indptr[b+1] - kv_indptr[b]

        # Flatten token indices per batch in the order of kv_indptr
        # tokens_ptr[i] = kv_indices[kv_indptr[b]:kv_indptr[b+1]]
        tokens_list = []
        start = 0
        for b in range(B):

            tokens_list.extend(kv_indices[start:end].to(torch.int32()).tolist())

        # Ensure contiguity
        q = q.contiguous()
        k_cache = k_cache.contiguous()
        v_cache = v_cache.contiguous()
        kv_indptr = kv_indptr.contiguous()
        kv_indices = kv_indices.contiguous()
        num_tokens_b = num_tokens_b.contiguous()

        # Launch Triton kernel: grid = (B, H)
        grid = (B, H)
        fused_gqa_row_kernel[grid](
            q, k_cache, v_cache, num_tokens_b, tokens, out, lse,
            sm_scale,
            B=B, H=H, D=D, K=K, gqa_ratio=4, MAX_TOKENS=1024,
            num_warps=2, num_stages=1,
        )

        # Convert output to bfloat16 as expected by original function
        output = out.to(torch.bfloat16)
        return output, lse
