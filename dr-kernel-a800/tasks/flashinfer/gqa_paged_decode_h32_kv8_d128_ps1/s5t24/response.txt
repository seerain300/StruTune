import math
import torch

try:
    import triton
    import triton.language as tl
    TRITON_AVAILABLE = True
except Exception:
    TRITON_AVAILABLE = False


# Triton kernel: one program per (b, h)
@triton.jit
def gqa_scalar_kernel(
    q_ptr,            # *bf16, shape [B, H, D]
    k_ptr,            # *bf16, shape [Np, 1, K, D] but we index by token
    v_ptr,            # *bf16, shape [Np, 1, K, D] but we index by token
    indptr_ptr,       # *int32, shape [B+1]
    indices_ptr,      # *int32, shape [num_kv_indices]
    output_ptr,       # *bf16, shape [B, H, D]
    lse_ptr,          # *float32, shape [B, H]
    B: tl.constexpr, H: tl.constexpr, D: tl.constexpr,  # constexpr shapes
    sm_scale: tl.float32,  # scale factor
    MAX_TOKENS: tl.constexpr,
):
    # program id over (b, h)
    pid = tl.program_id(axis=0)
    # map to b,h
    b = pid // H
    h = pid % H
    if b >= B:
        return

    # Load indptr[start, end] for this batch element
    start = tl.load(indptr_ptr + b).to(tl.int32)
    end = tl.load(indptr_ptr + b + 1).to(tl.int32)
    num_tokens = end - start  # scalar int32

    # Compute kv_head for GQA
    kv_head = h // (H // 8)  # gqa_ratio = 4 since H=32, K=8

    # Pass 1: accumulate sum_exp for logsumexp in natural log
    sum_exp = 0.0
    LN2 = 0.6931471805599453  # ln(2)
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = start + t  # token index
        # Load q vector for this head: q[b, h, :]
        q_base = b * H * D + h * D
        q_vec = tl.load(q_ptr + q_base + tl.arange(0, D)).to(tl.float32)  # [D], vectorized only over constexpr D

        # Load k row for this token and kv_head: k[idx, kv_head, :]
        k_base = idx * (8 * D) + kv_head * D  # since K=8, but we load scalar by loop
        # We'll iterate over D with scalar loads
        dot = 0.0
        for i in range(D):
            # q[i]
            q_i = tl.load(q_ptr + q_base + i).to(tl.float32)
            # k[idx, kv_head, i]
            k_i = tl.load(k_ptr + k_base + i).to(tl.float32)
            dot += q_i * k_i
        # logits_scaled in base-2: q·k
        logits_scaled = dot * sm_scale
        # convert to natural log scale for sumexp
        sum_exp += math.exp(logits_scaled / LN2)
    # compute lse in base-2
    lse_val = math.log(sum_exp) * (1.0 / LN2)  # logsumexp in base-2
    # store lse to lse_ptr[b, h]
    tl.store(lse_ptr + b * H + h, tl.float32(lse_val))

    # Pass 2: compute output vector and store
    # Reinitialize output vector for this (b, h)
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for t in range(MAX_TOKENS):
        if t >= num_tokens:
            break
        idx = start + t
        q_base = b * H * D + h * D
        q_vec = tl.load(q_ptr + q_base + tl.arange(0, D)).to(tl.float32)  # [D]
        dot = 0.0
        for i in range(D):
            q_i = tl.load(q_ptr + q_base + i).to(tl.float32)
            k_i = tl.load(k_ptr + (idx * (8 * D) + kv_head * D) + i).to(tl.float32)
            dot += q_i * k_i
        logits_scaled = dot * sm_scale
        attn = math.exp((logits_scaled - lse_val) / LN2)  # convert to base-2 scaled attn then exponentiate

        # load v[idx, kv_head, :] and accumulate
        v_base = idx * (8 * D) + kv_head * D
        for i in range(D):
            v_i = tl.load(v_ptr + v_base + i).to(tl.float32)
            out_vec[i] += attn * v_i

    # Store output vector
    out_base = b * H * D + h * D
    tl.store(output_ptr + out_base + tl.arange(0, D), out_vec.to(tl.bfloat16))
