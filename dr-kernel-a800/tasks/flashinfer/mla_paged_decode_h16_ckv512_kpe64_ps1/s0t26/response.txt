import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_nope_ptr,           # *bf16 [B, H, Dc]
    q_pe_ptr,             # *bf16 [B, H, Dp]
    ckv_cache_ptr,        # *bf16 [N, 1, Dc]
    kpe_cache_ptr,        # *bf16 [N, 1, Dp]
    kv_indptr_ptr,        # *int32 [B+1]
    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *bf16 [B, H, Dc]
    lse_ptr,              # *float32 [B*H]
    B: tl.constexpr,      # batch_size
    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    BLOCK_SIZE: tl.constexpr,
    SM_SCALE: tl.constexpr,  # scaling factor for logits
):
    # One program per batch element
    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)          # int32 start
    end = tl.load(kv_indptr_ptr + b + 1)       # int32 end
    L_tokens = end - base                       # number of tokens in this segment (int32)

    # Process each head h
    for h in range(H):
        # Load q vectors for head h as float32
        qn_vec = tl.load(q_nope_ptr + b * H * Dc + h * Dc + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)
        qp_vec = tl.load(q_pe_ptr + b * H * Dp + h * Dp + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0).to(tl.float32)

        # Compute logits_scaled[h, :] of length L_tokens in chunks of BLOCK_SIZE
        logits_scaled = tl.full((BLOCK_SIZE,), -float("inf"), tl.float32)

        # Iterate tokens in vectorized chunks
        # Note: L_tokens is runtime, but we use a while-like pattern via for t in range(0, L_tokens, BLOCK_SIZE)
        # Triton supports this pattern; mask handles runtime tail.
        for t in range(0, L_tokens, BLOCK_SIZE):
            offs = t + tl.arange(0, BLOCK_SIZE)
            mask = offs < L_tokens
            # Gather indices for this chunk
            idxs = tl.load(kv_indices_ptr + base + offs, mask=mask, other=0).to(tl.int32)

            # Load keys for this chunk
            Kc_rows = tl.load(ckv_cache_ptr + idxs[:, None] * Dc + tl.arange(0, Dc)[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
            Kp_rows = tl.load(kpe_cache_ptr + idxs[:, None] * Dp + tl.arange(0, Dp)[None, :], mask=mask[:, None], other=0.0).to(tl.float32)

            # Compute dot products for this head
            dot1 = tl.sum(qn_vec[None, :] * Kc_rows, axis=1)  # [BLOCK_SIZE]
            dot2 = tl.sum(qp_vec[None, :] * Kp_rows, axis=1)  # [BLOCK_SIZE]
            val = (dot1 + dot2) * SM_SCALE

            # Update logits_scaled with mask
            is_valid = mask
            logits_scaled = tl.where(is_valid, val, logits_scaled)

        # Stable logsumexp across tokens: lse = log(sum(exp(val - max))) + max, divided by ln(2)
        max_val = tl.max(logits_scaled, axis=0)  # scalar
        sum_exp = tl.sum(tl.exp(logits_scaled - max_val), axis=0)  # scalar
        lse_val = tl.log(sum_exp) + max_val
        # Convert to base-2 logsumexp: divide by ln(2)
        lse_val = lse_val / tl.log(2.0)

        # Compute attention and final output vector out[h, :]
        out_vec = tl.zeros((Dc,), dtype=tl.float32)

        # Accumulate over tokens again, now with exact attn_i
        for t in range(0, L_tokens, BLOCK_SIZE):
            offs = t + tl.arange(0, BLOCK_SIZE)
            mask = offs < L_tokens
            idxs = tl.load(kv_indices_ptr + base + offs, mask=mask, other=0).to(tl.int32)

            Kc_rows = tl.load(ckv_cache_ptr + idxs[:, None] * Dc + tl.arange(0, Dc)[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
            Kp_rows = tl.load(kpe_cache_ptr + idxs[:, None] * Dp + tl.arange(0, Dp)[None, :], mask=mask[:, None], other=0.0).to(tl.float32)

            dot1 = tl.sum(qn_vec[None, :] * Kc_rows, axis=1)
            dot2 = tl.sum(qp_vec[None, :] * Kp_rows, axis=1)
            val = (dot1 + dot2) * SM_SCALE

            attn_vec = tl.exp(val - lse_val)  # [BLOCK_SIZE]
            out_vec += tl.sum(attn_vec[:, None] * Kc_rows, axis=0)

        # Store output for head h
        out_offset = b * H * Dc + h * Dc
        tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec)

        # Store lse per (b, h)
        lse_offset = b * H + h
        tl.store(lse_ptr + lse_offset, lse_val)

          * sm_scale: float32 scalar
        - Returns:

        q_nope = q_nope.contiguous()

        ckv_cache = ckv_cache.contiguous()
        kpe_cache = kpe_cache.contiguous()

        kv_indices = kv_indices.contiguous()

        B = q_nope.shape[0]
        H = q_nope.shape[1]
        Dc = q_nope.shape[2]
        Dp = q_pe.shape[2]

import triton.language as tl

def attention_kernel(

    q_pe_ptr,             # *bf16 [B, H, Dp]
    ckv_cache_ptr,        # *bf16 [N, 1, Dc]
    kpe_cache_ptr,        # *bf16 [N, 1, Dp]

    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *bf16 [B, H, Dc]
    lse_ptr,              # *float32 [B*H]

    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    SM_SCALE: tl.constexpr,  # scaling factor for logits
):

    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)          # int32 start index in kv_indices
    end = tl.load(kv_indptr_ptr + b + 1)       # int32 end index in kv_indices
    L_tokens = end - base                       # number of tokens in this segment

    # Process each head h

        qn_vec = tl.load(q_nope_ptr + b * H * Dc + h * Dc + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)

        # Streaming logsumexp over tokens to compute lse (base-2): lse = log(sum_exp) + max_val, then divide by ln(2)
        max_val = tl.full((), -float("inf"), tl.float32)
        sum_exp = tl.full((), 0.0, tl.float32)

        # Loop over tokens i
        # Note: Triton allows Python range with a scalar; this pattern is supported.
        for i in range(L_tokens):
            idx = tl.load(kv_indices_ptr + base + i)  # int32
            # Load Kc_row [Dc] and Kp_row [Dp] for this token
