import math
import torch
import triton
import triton.language as tl


@triton.jit
def _compute_b_all_heads_kernel(
    q_nope_ptr,           # [H*D1], float32
    q_pe_ptr,             # [H*D2], float32
    Kc_rows_ptr,          # [L_tokens*D1], float32
    Kp_rows_ptr,          # [L_tokens*D2], float32
    output_ptr,           # [B*H*D1], float32
    lse_ptr,              # [B*H], float32
    H: tl.constexpr,      # num heads (e.g., 16)
    D1: tl.constexpr,     # ckv head dim (e.g., 512)
    D2: tl.constexpr,     # kpe head dim (e.g., 64)
    L_tokens: tl.constexpr,
    sm_scale: tl.constexpr,
    MAX_T: tl.constexpr,  # loop bound (e.g., 2048)
):
    # one program per batch element
    b = tl.program_id(0)
    if b < 0:
        return

    # Initialize per-head outputs and lse accumulators
    for h in tl.static_range(0, H):
        out_base = b * H * D1 + h * D1
        # zero-init output vector
        for d in tl.static_range(0, D1):
            tl.store(output_ptr + out_base + d, 0.0)

        # per-column max and sum for lse across tokens
        token_max_vec = tl.full((D1,), -float("inf"), dtype=tl.float32)
        token_sum_vec = tl.zeros((D1,), dtype=tl.float32)

        # Iterate over tokens
        for t in tl.static_range(0, MAX_T):
            valid = t < L_tokens
            # Load Kc_row and Kp_row for this token (1D vectors)
            # Kc_rows_ptr has length L_tokens * D1
            Kc_row = tl.load(Kc_rows_ptr + t * D1 + tl.arange(0, D1), mask=valid, other=0.0).to(tl.float32)  # [D1]
            Kp_row = tl.load(Kp_rows_ptr + t * D2 + tl.arange(0, D2), mask=valid, other=0.0).to(tl.float32)  # [D2]

            # Load qn[h, :] and qp[h, :]
            qn = tl.load(q_nope_ptr + h * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
            qp = tl.load(q_pe_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

            # Compute scalar logits
            dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
            logits_scalar = (dot1 + dot2) * sm_scale  # scalar

            # Update max and sum for lse
            # For invalid tokens, skip (valid=False ensures masked loads used zeros above)
            token_max_vec = tl.maximum(token_max_vec, logits_scalar)
            # Sum of exp(logits - max) per column
            token_sum_vec += tl.where(valid, tl.exp(logits_scalar - token_max_vec), 0.0)

        # Compute lse: logsumexp / ln(2)
        lse_val = token_max_vec + tl.log(token_sum_vec) / math.log(2.0)  # scalar
        # Store lse for this head
        tl.store(lse_ptr + b * H + h, lse_val)

        # Second pass: accumulate output using softmax scaling
        for t in tl.static_range(0, MAX_T):
            valid = t < L_tokens
            Kc_row = tl.load(Kc_rows_ptr + t * D1 + tl.arange(0, D1), mask=valid, other=0.0).to(tl.float32)  # [D1]
            Kp_row = tl.load(Kp_rows_ptr + t * D2 + tl.arange(0, D2), mask=valid, other=0.0).to(tl.float32)  # [D2]

            qn = tl.load(q_nope_ptr + h * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
            qp = tl.load(q_pe_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

            dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
            logits_scalar = (dot1 + dot2) * sm_scale  # scalar

            # Compute softmax scalar for this token
            soft = tl.where(valid, tl.exp(logits_scalar - token_max_vec) / token_sum_vec, 0.0)
            out_vec = tl.load(output_ptr + out_base + tl.arange(0, D1)).to(tl.float32)  # [D1]
            out_vec += soft * Kc_row
            for d in tl.static_range(0, D1):
                tl.store(output_ptr + out_base + d, out_vec[d])

    # Done for this batch element

        kpe_cache: [N, 1, D2], bfloat16
        kv_indptr: [B+1], int32

        # Note: For typical num_kv_indices (tens), this is cheap.
