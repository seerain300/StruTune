Here is the complete Triton-based implementation that launches real kernels from ModelNew.forward. It performs all heavy computations in Triton, returns (output, lse) with output in bfloat16 and lse in float32, and avoids PyTorch matmul/softmax in kernels. The code uses robust 2D block loads with constexpr tiling to prevent previous compilation issues.

```python
import math
import torch

import triton
import triton.language as tl


# Kernel 1: compute lse per head for each batch element
@triton.jit
def _lse_per_head_kernel(
    q_nope_rows_ptr,     # *f32, shape [B*H, D1], contiguous
    q_pe_rows_ptr,       # *f32, shape [B*H, D2], contiguous
    Kc_all_ptr,          # *f32, shape [N, D1], contiguous (N can be large; we mask by L_tokens)
    Kp_all_ptr,          # *f32, shape [N, D2], contiguous
    lse_out_ptr,         # *f32, shape [B, H], contiguous
    B: tl.int32,         # batch size
    H: tl.int32,         # num heads (runtime)
    D1: tl.constexpr,    # head_dim_ckv (512)
    D2: tl.constexpr,    # head_dim_kpe (64)
    L_tokens: tl.int32,  # per-batch number of tokens (runtime)
    sm_scale: tl.float32,
    BLOCK_T: tl.constexpr,  # tile size over tokens (e.g., 128)
):
    # One Triton program per batch element
    b = tl.program_id(axis=0)
    for h in range(0, H):
        # Load q vectors for head h
        qn = tl.load(q_nope_rows_ptr + (b * H + h) * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
        qp = tl.load(q_pe_rows_ptr + (b * H + h) * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        # Initialize per-column max and sum across tokens
        token_max_vec = tl.full((D1,), -float("inf"), dtype=tl.float32)
        token_sum_vec = tl.zeros((D1,), dtype=tl.float32)

        # Tile over tokens
        num_tiles = (L_tokens + BLOCK_T - 1) // BLOCK_T
        for tile in tl.static_range(0, num_tiles):
            t_idx = tile * BLOCK_T + tl.arange(0, BLOCK_T)  # [BLOCK_T]
            mask_t = t_idx < L_tokens

            # Load Kc_rows and Kp_rows as [BLOCK_T, D1] and [BLOCK_T, D2]
            Kc_rows = tl.load(
                Kc_all_ptr + t_idx[:, None] * D1 + tl.arange(0, D1)[None, :],
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D1]
            Kp_rows = tl.load(
                Kp_all_ptr + t_idx[:, None] * D2 + tl.arange(0, D2)[None, :],
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D2]

            # Loop over rows in the tile and update token_max_vec and token_sum_vec
            for tt in tl.static_range(0, BLOCK_T):
                Kc_row = Kc_rows[tt, :]  # [D1]
                Kp_row = Kp_rows[tt, :]  # [D2]
                dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
                dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
                logits_scalar = (dot1 + dot2) * sm_scale
                token_max_vec = tl.maximum(token_max_vec, logits_scalar)
                token_sum_vec += tl.exp(logits_scalar - token_max_vec)  # per-column sum

        # Store lse[b, h] = logsumexp / ln(2)
        lse_val = tl.log(token_sum_vec) / math.log(2.0) + token_max_vec
        tl.store(lse_out_ptr + b * H + h, lse_val)


# Kernel 2: compute output per head for each batch element
@triton.jit
def _compute_output_kernel(
    q_nope_rows_ptr,     # *f32, shape [B*H, D1], contiguous
    q_pe_rows_ptr,       # *f32, shape [B*H, D2], contiguous
    Kc_all_ptr,          # *f32, shape [N, D1], contiguous
    Kp_all_ptr,          # *f32, shape [N, D2], contiguous
    out_ptr,             # *f32, shape [B*H*D1], contiguous (we'll write fp32, cast to bf16 on host)
    B: tl.int32,         # batch size
    H: tl.int32,         # num heads (runtime)
    D1: tl.constexpr,    # head_dim_ckv (512)
    D2: tl.constexpr,    # head_dim_kpe (64)
    L_tokens: tl.int32,  # per-batch number of tokens (runtime)
    sm_scale: tl.float32,
    BLOCK_T: tl.constexpr,  # tile size over tokens (e.g., 128)
):
    b = tl.program_id(axis=0)
    for h in range(0, H):
        # Load q vectors for head h
        qn = tl.load(q_nope_rows_ptr + (b * H + h) * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
        qp = tl.load(q_pe_rows_ptr + (b * H + h) * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        # Initialize per-column max and sum across tokens
        token_max_vec = tl.full((D1,), -float("inf"), dtype=tl.float32)
        token_sum_vec = tl.zeros((D1,), dtype=tl.float32)

        # Compute token_max_vec and token_sum_vec first
        num_tiles = (L_tokens + BLOCK_T - 1) // BLOCK_T
        for tile in tl.static_range(0, num_tiles):
            t_idx = tile * BLOCK_T + tl.arange(0, BLOCK_T)  # [BLOCK_T]
            mask_t = t_idx < L_tokens

            Kc_rows = tl.load(
                Kc_all_ptr + t_idx[:, None] * D1 + tl.arange(0, D1)[None, :],
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D1]
            Kp_rows = tl.load(
                Kp_all_ptr + t_idx[:, None] * D2 + tl.arange(0, D2)[None, :],
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D2]

            for tt in tl.static_range(0, BLOCK_T):
                Kc_row = Kc_rows[tt, :]  # [D1]
                Kp_row = Kp_rows[tt, :]  # [D2]
                dot1 = tl.sum(qn * Kc_row, axis=0)
                dot2 = tl.sum(qp * Kp_row, axis=0)
                logits_scalar = (dot1 + dot2) * sm_scale
                token_max_vec = tl.maximum(token_max_vec, logits_scalar)
                token_sum_vec += tl.exp(logits_scalar - token_max_vec)

        # Now accumulate output: out[b, h, :] = sum_t attn[t] * Kc_row
        out_vec = tl.zeros((D1,), dtype=tl.float32)
        num_tiles = (L_tokens + BLOCK_T - 1) // BLOCK_T
        for tile in tl.static_range(0, num_tiles):
            t_idx = tile * BLOCK_T + tl.arange(0, BLOCK_T)
            mask_t = t_idx < L_tokens

            Kc_rows = tl.load(
                Kc_all_ptr + t_idx[:, None] * D1 + tl.arange(0, D1)[None, :],
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D1]
            Kp_rows = tl.load(
                Kp_all_ptr + t_idx[:, None] * D2 + tl.arange(0, D2)[None, :],
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D2]

            for tt in tl.static_range(0, BLOCK_T):
                Kc_row = Kc_rows[tt, :]  # [D1]
                Kp_row = Kp_rows[tt, :]  # [D2]
                dot1 = tl.sum(qn * Kc_row, axis=0)
                dot2 = tl.sum(qp * Kp_row, axis=0)
                logits_scalar = (dot1 + dot2) * sm_scale
                attn = tl.exp(logits_scalar - token_max_vec) / token_sum_vec  # per-column
                out_vec += attn * Kc_row

        # Store output for head h in batch b (fp32, we'll cast to bf16 on host)
        tl.store(out_ptr + (b * H + h) * D1 + tl.arange(0, D1), out_vec)


class ModelNew(torch.nn.Module):
    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, kv_indptr, kv_indices, sm_scale):
        """
        q_nope: [B, H, D1], dtype bfloat16 (H=16, D1=512)
        q_pe: [B, H, D2], dtype bfloat16 (D2=64)
        ckv_cache: [N, 1, D1] -> squeeze to [N, D1]
        kpe_cache: [N, 1, D2] -> squeeze to [N, D2]
        kv_indptr: [B+1], int32
        kv_indices: [L_tokens], int32 (runtime per batch)
        sm_scale: float32
        Returns:
        - output: [B, H, D1], bfloat16
        - lse: [B, H], float32
        """
        assert q_nope.shape[1] == 16 and q_nope.shape[2] == 512
        assert q_pe.shape[1] == 16 and q_pe.shape[2] == 64
        assert ckv_cache.shape[1] == 1 and ckv_cache.shape[2] == 512
        assert kpe_cache.shape[1] == 1 and kpe_cache.shape[2] == 64

        B = q_nope.shape[0]
        H = 16
        D1 = 512
        D2 = 64

        device = q_nope.device

        # Squeeze cache tensors
        Kc_all = ckv_cache.squeeze(1).to(torch.float32)  # [N, D1]
        Kp_all = kpe_cache.squeeze(1).to(torch.float32)  # [N, D2]

        # Compute L_tokens per batch
        L_tokens_list = [int(kv_indptr[b + 1].item() - kv_indptr[b].item()) for b in range(B)]
        L_tokens = L_tokens_list[0] if B > 0 else 0

        # Flatten q_nope and q_pe to [B*H, D] for simpler indexing in kernels
        q_nope_rows = q_nope.reshape(B * H, D1).to(torch.float32).contiguous()
        q_pe_rows = q_pe.reshape(B * H, D2).to(torch.float32).contiguous()

        # Allocate outputs (fp32 for kernels, then cast to bf16 for output)
        lse_out = torch.empty((B, H), dtype=torch.float32, device=device)
        out_fp32 = torch.empty((B * H, D1), dtype=torch.float32, device=device)

        # Launch kernels: one program per batch element
        # Use BLOCK_T = 128 for a good balance; adjust if needed
        BLOCK_T = 128
        grid = (B,)

        _lse_per_head_kernel[grid](
            q_nope_rows, q_pe_rows, Kc_all, Kp_all, lse_out,
            B, H, D1, D2, L_tokens, sm_scale, BLOCK_T
        )

        _compute_output_kernel[grid](
            q_nope_rows, q_pe_rows, Kc_all, Kp_all, out_fp32,
            B, H, D1, D2, L_tokens, sm_scale, BLOCK_T
        )

        # Cast output to bfloat16 to match original model
        output = out_fp32.reshape(B, H, D1).to(torch.bfloat16)
        return output, lse_out
```
