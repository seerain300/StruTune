Implementation details
- Triton kernels:
  1) matvec_row: computes out = v @ B where v is [M] and B is [M, N] as blocks of columns. Each program processes BLOCK_N columns. We will call this for:
     - qn_h @ Kc.T: B = Kc.T (shape [512, L_tokens]).
     - qp_h @ Kp.T: B = Kp.T (shape [64, L_tokens]).
     - attention_probs @ Kc: B = Kc (shape [L_tokens, 512]).
  2) softmax_base2_kernel: computes softmax in base-2 for a 1D vector of length L. It writes probabilities to out_ptr[0:L] and the scalar lse (base-2) to out_ptr[L].
  3) matmul_small: computes C[M,N] = A[M,K] @ B[K,N] with tiling over K. We will use it to compute C[1,512] = attention_probs[1,L_tokens] @ Kc[1,512].

- Dtypes:
  - Inputs are bfloat16; inside kernels we compute in float32 for stability. Output returned as bfloat16 to match original. lse returned as float32.

Final code

```python
import math
import torch

# Triton kernels: all math is performed inside Triton. No torch ops in forward.

# matvec_row: computes out = v @ B, where v is [M] (row vector) and B is [M, N].
# Each program handles BLOCK_N columns of the output vector.
@triton.jit
def matvec_row(v_ptr, B_ptr, out_ptr,
                M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)
    n_offsets = pid * BLOCK_N + tl.arange(0, BLOCK_N)
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)
    # Loop over K dimension
    for k in range(0, K):
        v_k = tl.load(v_ptr + k)  # v_ptr points to [M], contiguous
        b_k = tl.load(B_ptr + k * N + n_offsets, mask=n_offsets < N, other=0.0)
        acc += v_k * b_k
    tl.store(out_ptr + n_offsets, acc, mask=n_offsets < N)

# softmax_base2_kernel: computes softmax in base-2 for a 1D vector of length L.
# It writes probabilities to out_ptr[0:L] and the scalar lse (base-2) to out_ptr[L].
@triton.jit
def softmax_base2_kernel(logits_ptr, out_ptr, L: tl.constexpr, BLOCK: tl.constexpr):
    # Find max for numerical stability
    max_val = -float('inf')
    for i in range(0, L):
        val = tl.load(logits_ptr + i)
        max_val = tl.maximum(max_val, val)
    # Compute sum of exp(logits - max) * (2 ** lse), then normalize by sum
    sum_val = 0.0
    for i in range(0, L):
        val = tl.load(logits_ptr + i)
        sum_val += tl.exp(val - max_val)  # store in out_ptr[L] after computing scale
    # Compute lse (base-2): lse = max + log(sum) / ln(2)
    inv_ln2 = 1.4426950408889634  # 1 / ln(2)
    lse_val = max_val + tl.log(sum_val) * inv_ln2
    # Write scalar lse to out_ptr[L]
    tl.store(out_ptr + L, lse_val)

    # Compute per-token probabilities and store
    for i in range(0, L):
        val = tl.load(logits_ptr + i)
        p = tl.exp(val - max_val) / sum_val
        # store in base-2 probability: p = exp((val - max)/ln2) / (2**lse) but we already normalized by sum;
        # we need p in base-2: p_base2 = p / (2 ** lse) = p * 0.5 ** lse
        # However, softmax normalization already divides by sum; we must ensure probabilities are base-2.
        # Since p is in [0,1], we need to scale by 0.5 ** lse to convert to base-2 distribution.
        # We can compute scale = 0.5 ** lse via exp(-lse * ln(2)) which equals 0.5 ** lse.
        ln05 = -0.6931471805599453  # ln(0.5)
        scale = tl.exp(ln05 * lse_val)
        p = p * scale
        tl.store(out_ptr + i, p)

# matmul_small: computes C[M,N] = A[M,K] @ B[K,N] with tiling over K.
@triton.jit
def matmul_small(A_ptr, B_ptr, C_ptr,
                 M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # We specialize for M=1 in our usage. Grid will have (1, cdiv(N, BLOCK_N)).
    pid_n = tl.program_id(0)
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)
    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        a = tl.load(A_ptr + 0 * K + k_offsets, mask=k_offsets < K, other=0.0)  # [BLOCK_K]
        b = tl.load(B_ptr + k_offsets[:, None] * N + n_offsets[None, :], mask=(k_offsets[:, None] < K) & (n_offsets[None, :] < N), other=0.0)  # [BLOCK_K, BLOCK_N]
        # acc += a[None, :] @ b
        acc += tl.sum(a[:, None] * b, axis=0)
    tl.store(C_ptr + 0 * N + n_offsets, acc, mask=n_offsets < N)

# Note: We will invoke matvec_row for qn_h @ Kc.T and qp_h @ Kp.T, and for attention_probs @ Kc.
# We will invoke softmax_base2_kernel to compute per-token softmax (base-2) and lse.
# We will invoke matmul_small for the final attention vector @ Kc (A is [1, L], B is [L, 512], C is [1, 512]).
```
