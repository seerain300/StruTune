    Dq: tl.constexpr, # 512
    Dp: tl.constexpr, # 64
    T: tl.constexpr,  # number of tokens
    BLOCK_T: tl.constexpr,  # tile over tokens (power-of-two, e.g., 128)
):

    # Preload qn[h, :] and qp[h, :]

    off_dp = tl.arange(0, Dp)  # position embedding feature
