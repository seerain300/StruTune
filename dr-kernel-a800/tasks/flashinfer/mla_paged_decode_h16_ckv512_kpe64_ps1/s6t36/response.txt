import torch
import triton
import triton.language as tl


@triton.jit
def matmul_qn_KcT_fp32_kernel(
    qn_ptr,     # *fp32, [H, D]
    Kc_ptr,     # *fp32, [L_tokens, D]
    acc_ptr,    # *fp32, [H, L_tokens]
    H: tl.constexpr,         # number of heads
    D: tl.constexpr,         # head_dim_ckv (compile-time constant, e.g., 512)
    L_tokens: tl.constexpr,  # number of tokens per batch element (compile-time for simplicity)
):
    # 2D grid: (h in 0..H-1, t in 0..L_tokens-1)
    h = tl.program_id(0)
    t = tl.program_id(1)

    acc = 0.0
    for kk in range(0, D):
        q = tl.load(qn_ptr + h * D + kk)  # qn[h, kk]
        k = tl.load(Kc_ptr + t * D + kk)  # Kc[t, kk]
        acc += q * k
    tl.store(acc_ptr + h * L_tokens + t, acc)


@triton.jit
def matmul_qp_KpT_fp32_kernel(
    qp_ptr,     # *fp32, [H, Dp]
    Kp_ptr,     # *fp32, [L_tokens, Dp]
    acc_ptr,    # *fp32, [H, L_tokens]
    H: tl.constexpr,         # number of heads
    Dp: tl.constexpr,        # head_dim_kpe (compile-time constant, e.g., 64)
    L_tokens: tl.constexpr,  # number of tokens per batch element (compile-time for simplicity)
):
    # 2D grid: (h in 0..H-1, t in 0..L_tokens-1)
    h = tl.program_id(0)
    t = tl.program_id(1)

    acc = 0.0
    for kk in range(0, Dp):
        q = tl.load(qp_ptr + h * Dp + kk)  # qp[h, kk]
        k = tl.load(Kp_ptr + t * Dp + kk)  # Kp[t, kk]
        acc += q * k
    tl.store(acc_ptr + h * L_tokens + t, acc)


@triton.jit
def add_two_accs_fp32_kernel(
    acc1_ptr,    # *fp32, [H, L_tokens]
    acc2_ptr,    # *fp32, [H, L_tokens]
    sum_ptr,     # *fp32, [H, L_tokens]
    H: tl.constexpr,
    L_tokens: tl.constexpr,
):
    h = tl.program_id(0)
    t = tl.program_id(1)
    v1 = tl.load(acc1_ptr + h * L_tokens + t)
    v2 = tl.load(acc2_ptr + h * L_tokens + t)
    tl.store(sum_ptr + h * L_tokens + t, v1 + v2)


@triton.jit
def scale_logits_fp32_kernel(
    sum_ptr,      # *fp32, [H, L_tokens]
    scale,        # fp32
    out_ptr,      # *fp32, [H, L_tokens]
    H: tl.constexpr,
    L_tokens: tl.constexpr,
):
    h = tl.program_id(0)
    t = tl.program_id(1)
    v = tl.load(sum_ptr + h * L_tokens + t)
    v = v * scale
    tl.store(out_ptr + h * L_tokens + t, v)


@triton.jit
def rowwise_lse_fp32_kernel(
    logits_ptr,   # *fp32, [H, L_tokens]
    lse_ptr,      # *fp32, [H]
    H: tl.constexpr,
    L_tokens: tl.constexpr,
):
    h = tl.program_id(0)
    # Compute logsumexp for row h
    # Initialize m and sum
    m = -float("inf")
    sum_exp = 0.0
    for t in range(0, L_tokens):
        val = tl.load(logits_ptr + h * L_tokens + t)
        # If val is -inf, continue; else update m and sum
        # Using tl.where to avoid NaNs
        # Note: Triton does not have fpclassify; treat -inf as below current m
        # Update m
        is_bigger = val > m
        m = tl.where(is_bigger, val, m)
        # Accumulate exp(val - m)
        # For val == -inf, exp(-inf) == 0
        exp_val = tl.exp(val - m)
        sum_exp += exp_val
    lse_val = m + tl.log(sum_exp)  # natural logsumexp
    ln2 = 0.6931471805599453
    lse_val = lse_val / ln2  # convert to base-2 logsumexp
    tl.store(lse_ptr + h, lse_val)


@triton.jit

    sm_ptr,       # *fp32, [H, L_tokens]
