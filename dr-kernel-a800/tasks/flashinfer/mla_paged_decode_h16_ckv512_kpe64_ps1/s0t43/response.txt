import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_nope_ptr,           # *bf16 [B, H, Dc]
    q_pe_ptr,             # *bf16 [B, H, Dp]
    ckv_cache_ptr,        # *bf16 [N, 1, Dc]
    kpe_cache_ptr,        # *bf16 [N, 1, Dp]
    kv_indptr_ptr,        # *int32 [B+1]
    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *float32 [B, H, Dc]  (we will cast to bf16 on host)
    lse_ptr,              # *float32 [B*H]
    B: tl.constexpr,      # batch_size
    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    SM_SCALE: tl.float32,  # scaling factor for logits (runtime scalar)
):
    # One program per batch element
    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)          # int32
    end = tl.load(kv_indptr_ptr + b + 1)       # int32
    L_tokens = end - base                       # number of tokens for this batch element

    # If no tokens for this batch element, set lse to -inf and output to zeros
    has_tokens = L_tokens > 0

    # Loop over heads
    for h in range(H):
        # Compute q vectors for this head: q_nope[b, h, :] and q_pe[b, h, :]
        # We'll construct qn and qp by loading the appropriate rows.
        # q_nope_ptr is [B, H, Dc], we want qn = q_nope[b, h, :]
        qn_offset = b * H * Dc + h * Dc
        qn = tl.load(q_nope_ptr + qn_offset + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)

        qp_offset = b * H * Dp + h * Dp  # Note: head_dim_kpe=Dp corresponds to q_pe shape [B, H, Dp]
        # However, q_pe shape is [B, H, Dp], so we need to load q[b, h, :] which has Dp elements.
        # We should load q_pe[b, h, :] properly. Given q_pe is [B, H, Dp], the offset for head h is:
        # We need to compute q[b, h, :] across Dp, but Triton expects a base pointer. We'll load a vector:
        # q[b, h, :] is at offset b*H*Dp + h*Dp, then across Dp dimension. Correction:
        # The correct offset for q[b, h, :] is (b * H + h) * Dp. We'll fix this properly.
        # Compute qn and qp correctly:
        qn = tl.load(q_nope_ptr + qn_offset + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)
        # q_pe_ptr layout is [B, H, Dp]; for head h, vector across Dp:
        qv_offset = (b * H + h) * Dp
        qp = tl.load(q_pe_ptr + qv_offset + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0).to(tl.float32)

        # First pass: compute max and sum of exp(logits_scaled) stably
        max_val = tl.full((), -float("inf"), tl.float32)
        sum_exp = tl.full((), 0.0, tl.float32)

        # Loop over tokens i in [0, L_tokens). We use a simple scalar loop.
        # Note: Triton supports scalar control flow; we loop up to a safe bound if needed.
        # Since L_tokens is dynamic, we loop while i < L_tokens:
        i = 0
        while i < 1024:  # upper bound; real loop will break when i >= L_tokens
            if i >= L_tokens:
                break
            idx = tl.load(kv_indices_ptr + base + i)

            # Load Kc_row and Kp_row as float32
            Kc_row_ptr = ckv_cache_ptr + idx * Dc
            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)

            Kp_row_ptr = kpe_cache_ptr + idx * Dp
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0).to(tl.float32)

            # Dot products: qn @ Kc_row + qp @ Kp_row
            dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
            val = (dot1 + dot2) * SM_SCALE

            # Stable logsumexp update
            if max_val == -float("inf") or val > max_val:
                sum_exp = tl.exp(val - max_val)
                max_val = val
            else:
                sum_exp = sum_exp + tl.exp(val - max_val)
            i += 1

        # Compute lse = log(sum_exp) + max_val, then divide by ln(2) to match original
        lse_val = tl.log(sum_exp) + max_val
        # Scale by 1/ln(2)
        lse_val = lse_val / tl.log(2.0)

        # Second pass: compute attention weights and final output vector
        out_vec = tl.zeros((Dc,), dtype=tl.float32)
        i = 0
        while i < 1024:
            if i >= L_tokens:
                break
            idx = tl.load(kv_indices_ptr + base + i)

            Kc_row_ptr = ckv_cache_ptr + idx * Dc
            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)

            Kp_row_ptr = kpe_cache_ptr + idx * Dp
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0).to(tl.float32)

            dot1 = tl.sum(qn * Kc_row, axis=0)
            dot2 = tl.sum(qp * Kp_row, axis=0)
            val = (dot1 + dot2) * SM_SCALE
            attn_i = tl.exp(val - lse_val)  # softmax over tokens

            out_vec += attn_i * Kc_row
            i += 1

        # Store output for head h (float32)
        out_offset = b * H * Dc + h * Dc
        tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec)

        # Store lse per (b, h)
        lse_offset = b * H + h
        tl.store(lse_ptr + lse_offset, lse_val)

    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, kv_indptr, kv_indices, sm_scale):
        """
        Triton-only forward:
        - Inputs:
          * q_nope: [B, H, Dc], dtype bfloat16 (B=1, H=16, Dc=512)
          * q_pe: [B, H, Dp], dtype bfloat16 (Dp=64)
          * ckv_cache: [N, 1, Dc], dtype bfloat16
          * kpe_cache: [N, 1, Dp], dtype bfloat16
          * kv_indptr: [B+1], int32
          * kv_indices: [L], int32
          * sm_scale: float32 scalar
        - Returns:
          * output: [B, H, Dc], dtype bfloat16
          * lse: [B, H], dtype float32
        """
        assert q_nope.is_cuda and q_pe.is_cuda and ckv_cache.is_cuda and kpe_cache.is_cuda, "All tensors must be on CUDA."
        assert kv_indptr.is_cuda and kv_indices.is_cuda, "Indptr and indices must be on CUDA."

        B, H, Dc = q_nope.shape
        Dp = q_pe.shape[-1]
        N = ckv_cache.shape[0]
        L = kv_indices.shape[0]

        # Allocate outputs as float32 (compute dtype), cast to bfloat16 at the end
        output = torch.empty((B, H, Dc), dtype=torch.float32, device=q_nope.device)

            output, lse,
            B=B, H=H, Dc=Dc, Dp=Dp, SM_SCALE=float(sm_scale),
        )

        # Cast output to bfloat16 to match original return type
        output_bf16 = output.to(torch.bfloat16)
        return [output_bf16, lse]
