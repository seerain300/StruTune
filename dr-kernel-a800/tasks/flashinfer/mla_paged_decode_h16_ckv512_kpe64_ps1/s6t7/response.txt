    qn_ptr,         # *fp32, shape [H, D], contiguous
    qp_ptr,         # *fp32, shape [H, Dp], contiguous
    Kc_ptr,         # *fp32, shape [L_tokens, D], contiguous
    Kp_ptr,         # *fp32, shape [L_tokens, Dp], contiguous

    L_tokens: tl.int32,
    b: tl.int32,
    sm_scale: tl.float32,
):

    h = tl.program_id(1)  # second grid dim is heads

    # Initialize max for logsumexp
    m = -float("inf")
    for t in range(0, L_tokens):
        # Load qn row for head h
        qn_row = tl.zeros((D,), dtype=tl.float32)
        for kk in range(0, D):
            qn_row[kk] = tl.load(qn_ptr + h * D + kk)

        # Load qp row for head h
        qp_row = tl.zeros((Dp,), dtype=tl.float32)
        for kk in range(0, Dp):
            qp_row[kk] = tl.load(qp_ptr + h * Dp + kk)

        # Load Kc row for token t

        for kk in range(0, D):
            tok_idx_t = tl.load(tok_idx_ptr + t)
            Kc_row[kk] = tl.load(Kc_ptr + tok_idx_t * D + kk)

            tok_idx_t = tl.load(tok_idx_ptr + t)
            Kp_row[kk] = tl.load(Kp_ptr + tok_idx_t * Dp + kk)

        # Compute dot products
        acc1 = 0.0
        for kk in range(0, D):
            acc1 += qn_row[kk] * Kc_row[kk]
        acc2 = 0.0
        for kk in range(0, Dp):
