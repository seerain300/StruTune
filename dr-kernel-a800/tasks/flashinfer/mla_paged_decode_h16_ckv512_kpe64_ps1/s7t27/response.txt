# Triton kernels: perform all computation

# matvec_row: out = v @ B, where v is [M] and B is [M, N].
@triton.jit
def matvec_row(v_ptr, B_ptr, out_ptr,
                M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                BLOCK_N: tl.constexpr):
    # One program computes up to BLOCK_N outputs
    n_offsets = tl.arange(0, BLOCK_N)
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)
    # Loop over K dimension
    for k in range(0, K):
        v_k = tl.load(v_ptr + k)  # v_ptr is [M], contiguous
        b_k = tl.load(B_ptr + k * N + n_offsets, mask=n_offsets < N, other=0.0)
        acc += v_k * b_k
    tl.store(out_ptr + n_offsets, acc, mask=n_offsets < N)

# softmax in base-2: computes softmax(logits) in base-2 and writes both probs and lse.
@triton.jit
def softmax_base2_kernel(logits_ptr, probs_ptr, lse_ptr,
                          L: tl.constexpr,
                          BLOCK: tl.constexpr):
    # We process all L elements in one program. BLOCK should be >= L for the simple approach.
    l_offsets = tl.arange(0, BLOCK)
    mask = l_offsets < L

    # Load logits
    logits = tl.load(logits_ptr + l_offsets, mask=mask, other=0.0).to(tl.float32)

    # Compute base-2 logsumexp
    # lse = log(sum(exp(x))) / log(2) = logsumexp(x) * (1 / ln(2))
    # We compute max for numerical stability.
    max_val = tl.max(logits, axis=0)
    x = logits - max_val
    exp_x = tl.exp(x)
    sum_exp = tl.sum(exp_x, axis=0)
    lse_scalar = tl.log(sum_exp) * (1.0 / 0.6931471805599453)  # 1 / ln(2)
    # Store scalar lse
    tl.store(lse_ptr, lse_scalar)

    # Compute probabilities: exp(x - lse) / sum_exp
    # Convert to base-2: p_base2 = p * (1 / ln(2))
    p = exp_x / sum_exp
    p_base2 = p * (1.0 / 0.6931471805599453)
    tl.store(probs_ptr + l_offsets, p_base2, mask=mask)

# matmul_small: computes C[M, N] = A[M, K] @ B[K, N]
@triton.jit
def matmul_small(A_ptr, B_ptr, C_ptr,
                  M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

@triton.jit

                M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                BLOCK_N: tl.constexpr):

    n_offsets = tl.arange(0, BLOCK_N)
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)

    for k in range(0, K):

        b_k = tl.load(B_ptr + k * N + n_offsets)  # B_ptr points to [M, N], contiguous row-major
        acc += v_k * b_k
    tl.store(out_ptr + n_offsets, acc)

@triton.jit
def softmax_base2_kernel(logits_ptr, probs_ptr, lse_ptr,
                          L: tl.constexpr,
                          BLOCK: tl.constexpr):

    l_offsets = tl.arange(0, BLOCK)
    mask = l_offsets < L

    logits = tl.load(logits_ptr + l_offsets, mask=mask, other=0.0).to(tl.float32)

    # Compute base-2 logsumexp for numerical stability
    max_val = tl.max(logits, axis=0)
    x = logits - max_val
    exp_x = tl.exp(x)
    sum_exp = tl.sum(exp_x, axis=0)
    # ln(2) = 0.6931471805599453

    tl.store(lse_ptr, lse_scalar)

    # Probabilities in base-2: p_base2 = exp(x - lse) / sum_exp * (1/ln(2))
    p = exp_x / sum_exp
    p_base2 = p * (1.0 / 0.6931471805599453)

@triton.jit
def matmul_small(A_ptr, B_ptr, C_ptr,
                  M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # We implement a single program that computes C[m, :] for m in 0..M-1 (here M=1).
    for m in range(0, M):

        acc = tl.zeros([BLOCK_N], dtype=tl.float32)
        for k0 in range(0, K, BLOCK_K):
            k_offsets = k0 + tl.arange(0, BLOCK_K)
            a_vec = tl.load(A_ptr + m * K + k_offsets, mask=k_offsets < K, other=0.0)  # [BLOCK_K]

        tl.store(C_ptr + m * N + n_offsets, acc, mask=n_offsets < N)
