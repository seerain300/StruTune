Here is the complete code implementing the Triton-optimized version. The entry point class is ModelNew, which invokes Triton kernels in forward to perform all computations, avoiding any PyTorch tensor operations on tensors.

```python
import torch
import math
import triton
import triton.language as tl


# Triton kernels: all heavy computations are implemented here
@triton.jit
def fused_logits_kernel(
    qn_ptr,        # *f32 [H, Dq]
    qp_ptr,        # *f32 [H, Dp]
    Kc_ptr,        # *f32 [T, Dq]
    Kp_ptr,        # *f32 [T, Dp]
    logits_ptr,    # *f32 [H, T]
    H: tl.constexpr,     # number of heads (16)
    T: tl.constexpr,     # number of tokens in this batch element
    Dq: tl.constexpr,    # 512
    Dp: tl.constexpr,    # 64
    BLOCK_T: tl.constexpr=128
):
    # Each program handles one (head, token-tile) pair
    h = tl.program_id(0)
    for t_start in range(0, T, BLOCK_T):
        offs_t = t_start + tl.arange(0, BLOCK_T)
        mask_t = offs_t < T

        # Initialize logits tile accumulator for this head
        logits_tile = tl.zeros((BLOCK_T,), dtype=tl.float32)

        # Compute dot(qn[h, :], Kc[t, :]) and dot(qp[h, :], Kp[t, :]) and sum
        for d_start in range(0, Dq, 128):
            offs_d = d_start + tl.arange(0, 128)
            mask_d = offs_d < Dq
            # qn[h, d] over offs_d
            qn_vec = tl.load(qn_ptr + h * Dq + offs_d, mask=mask_d, other=0.0)  # [128]
            # Kc[t, offs_d] for each t in tile
            Kc_tile = tl.load(
                Kc_ptr + offs_t[:, None] * Dq + offs_d[None, :],
                mask=mask_t[:, None] & mask_d[None, :],
                other=0.0
            )  # [BLOCK_T, 128]
            dot2 = tl.sum(Kc_tile * qn_vec[None, :], axis=1)  # [BLOCK_T]
            logits_tile += dot2

        for d_start in range(0, Dp, 128):
            offs_d = d_start + tl.arange(0, 128)
            mask_d = offs_d < Dp
            # qp[h, d] over offs_d
            qp_vec = tl.load(qp_ptr + h * Dp + offs_d, mask=mask_d, other=0.0)  # [128]
            # Kp[t, offs_d] for each t in tile
            Kp_tile = tl.load(
                Kp_ptr + offs_t[:, None] * Dp + offs_d[None, :],
                mask=mask_t[:, None] & mask_d[None, :],
                other=0.0
            )  # [BLOCK_T, 128]
            dot2 = tl.sum(Kp_tile * qp_vec[None, :], axis=1)  # [BLOCK_T]
            logits_tile += dot2

        tl.store(logits_ptr + h * T + offs_t, logits_tile, mask=mask_t)


@triton.jit
def softmax_row_kernel(
    x_ptr,         # *f32 [H, T]
    y_ptr,         # *f32 [H, T]
    H: tl.constexpr,
    T: tl.constexpr,
    BLOCK_T: tl.constexpr=128
):
    h = tl.program_id(0)
    for t_start in range(0, T, BLOCK_T):
        offs_t = t_start + tl.arange(0, BLOCK_T)
        mask_t = offs_t < T
        x = tl.load(x_ptr + h * T + offs_t, mask=mask_t, other=-float("inf"))
        x_max = tl.max(x, axis=0)
        x = x - x_max
        exp_x = tl.exp(x)
        sum_exp = tl.sum(exp_x, axis=0)
        y = exp_x / sum_exp
        tl.store(y_ptr + h * T + offs_t, y, mask=mask_t)


@triton.jit
def lse_row_kernel(
    x_ptr,         # *f32 [H, T]
    lse_ptr,       # *f32 [H]
    H: tl.constexpr,
    T: tl.constexpr,
    BLOCK_T: tl.constexpr=128
):
    h = tl.program_id(0)
    sum_log2 = 0.0
    for t_start in range(0, T, BLOCK_T):
        offs_t = t_start + tl.arange(0, BLOCK_T)
        mask_t = offs_t < T
        x = tl.load(x_ptr + h * T + offs_t, mask=mask_t, other=-float("inf"))
        x_max = tl.max(x, axis=0)
        exp_sum = tl.sum(tl.exp(x - x_max), axis=0)
        sum_log2 += tl.log(2.0) * (x_max + tl.log(exp_sum))
    tl.store(lse_ptr + h, sum_log2)


@triton.jit
def gemv_row_kernel(
    attn_ptr,      # *f32 [T]
    Kc_ptr,        # *f32 [T, Dq]
    out_ptr,       # *f32 [Dq]
    T: tl.constexpr,
    Dq: tl.constexpr,
    BLOCK_D: tl.constexpr=128,
    BLOCK_T: tl.constexpr=128
):
    # out = attn @ Kc for a single row vector attn of length T
    out_vec = tl.zeros((Dq,), dtype=tl.float32)
    for d_start in range(0, Dq, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < Dq
        for t_start in range(0, T, BLOCK_T):
            offs_t = t_start + tl.arange(0, BLOCK_T)
            mask_t = offs_t < T
            attn_tile = tl.load(attn_ptr + offs_t, mask=mask_t, other=0.0)  # [BLOCK_T]
            Kc_tile = tl.load(
                Kc_ptr + offs_t[:, None] * Dq + offs_d[None, :],
                mask=mask_t[:, None] & mask_d[None, :],
                other=0.0
            )  # [BLOCK_T, BLOCK_D]
            partial = tl.sum(Kc_tile * attn_tile[:, None], axis=0)  # [BLOCK_D]
            out_vec += partial
        tl.store(out_ptr + offs_d, out_vec[offs_d], mask=mask_d)


def run(q_nope, q_pe, ckv_cache, kpe_cache, kv_indptr, kv_indices, sm_scale):
    # Expect shapes: q_nope [B, 16, 512], q_pe [B, 16, 64], ckv_cache [N, 1, 512], kpe_cache [N, 1, 64]
    device = q_nope.device
    B = q_nope.shape[0]
    H = 16
    Dq = 512
    Dp = 64

    output = torch.empty((B, H, Dq), dtype=torch.bfloat16, device=device)
    lse = torch.empty((B, H), dtype=torch.float32, device=device)

    for b in range(B):
        # Compute number of tokens for this batch element
        L_tokens = int(kv_indptr[b + 1].item()) - int(kv_indptr[b].item())
        if L_tokens <= 0:
            # No KV cache for this batch element; set output to zeros and lse to -inf
            output[b].zero_()
            lse[b].fill_(-float("inf"))
            continue

        # Gather token indices and corresponding Kc/Kp
        tok_idx = kv_indices[int(kv_indptr[b].item()): int(kv_indptr[b + 1].item())].to(device)
        Kc_b = ckv_cache[tok_idx].squeeze(1).to(torch.float32)  # [L_tokens, 512]
        Kp_b = kpe_cache[tok_idx].squeeze(1).to(torch.float32)  # [L_tokens, 64]
        qn = q_nope[b].to(torch.float32)  # [16, 512]
        qp = q_pe[b].to(torch.float32)    # [16, 64]

        # Allocate intermediate tensors
        logits = torch.empty((H, L_tokens), dtype=torch.float32, device=device)

        # Launch fused logits kernel
        grid = (H,)
        fused_logits_kernel[grid](
            qn, qp, Kc_b, Kp_b, logits,
            H=H, T=L_tokens, Dq=Dq, Dp=Dp,
            BLOCK_T=128,
            num_warps=4, num_stages=2
        )

        # Scale and softmax per head
        logits_scaled = logits * sm_scale
        attn = torch.empty_like(logits_scaled)
        grid_softmax = (H,)
        softmax_row_kernel[grid_softmax](
            logits_scaled, attn, H=H, T=L_tokens, BLOCK_T=128, num_warps=4, num_stages=2
        )

        # Compute lse per head (logsumexp of scaled logits) / ln(2)
        lse[b] = torch.empty((H,), dtype=torch.float32, device=device)
        lse_row_kernel[grid_softmax](
            logits_scaled, lse[b], H=H, T=L_tokens, BLOCK_T=128, num_warps=4, num_stages=2
        )

        # Per-head matmul: out[h, :] = attn[h, :] @ Kc[:, :]
        for h_idx in range(H):
            out_vec = torch.empty((Dq,), dtype=torch.float32, device=device)
            attn_vec = attn[h_idx]  # [L_tokens]
            gemv_row_kernel[(1,)](
                attn_vec, Kc_b, out_vec,
                T=L_tokens, Dq=Dq, BLOCK_D=128, BLOCK_T=128,
                num_warps=4, num_stages=2
            )
            output[b, h_idx] = out_vec  # write into output tensor

    # Cast output to bfloat16 to match original run’s output dtype
    output_bf16 = output.to(torch.bfloat16)
    return output_bf16, lse.to(torch.float32)
```
