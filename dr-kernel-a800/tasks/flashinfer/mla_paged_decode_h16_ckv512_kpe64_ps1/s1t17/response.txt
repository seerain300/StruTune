@triton.jit
def gather_rows_c_kernel(cache_ptr, tok_idx_ptr, out_ptr,
                          num_tokens: tl.constexpr, Dc: tl.constexpr):

        return

#    Two-pass reduction: first max, then sum of exp, then lse = m + log(sum_exp) / ln(2).
@triton.jit

    if i >= H:
        return

    for t in range(0, L):
        val = tl.load(logits_ptr + i * L + t)
        m = tl.maximum(m, val)
    sum_exp = 0.0
    for t in range(0, L):

        sum_exp += tl.exp(val - m)

@triton.jit

        val = tl.load(logits_ptr + i * L + t)
        m = tl.maximum(m, val)

        val = tl.load(logits_ptr + i * L + t)
        sum_exp += tl.exp(val - m)
    # Write normalized softmax
    for t in range(0, L):
        val = tl.load(logits_ptr + i * L + t)
        p = tl.exp(val - m) / sum_exp
        tl.store(attn_ptr + i * L + t, p)


# 5) Matvec: out_vec[i, :] = attn_row[i, :] @ Kc_col[:, ] where attn_row[i] is [L_tokens], Kc_col is [L_tokens, Dc]
#    Write out_vec[i, :] into out_ptr[i * Dc + d] using chunks of Dc.
@triton.jit
def matvec_rows_kernel(attn_ptr, Kc_ptr, out_ptr,
                        H: tl.constexpr, L: tl.constexpr, Dc: tl.constexpr):
    i = tl.program_id(0)

        return

        acc = 0.0
        # Loop over L tokens

            a = tl.load(attn_ptr + i * L + t)
            k = tl.load(Kc_ptr + t * Dc + d)
            acc += a * k
        tl.store(out_ptr + i * Dc + d, acc)
