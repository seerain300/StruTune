import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_nope_ptr,           # *bf16 [B, H, Dc]
    q_pe_ptr,             # *bf16 [B, H, Dp]
    ckv_cache_ptr,        # *bf16 [N, 1, Dc]
    kpe_cache_ptr,        # *bf16 [N, 1, Dp]
    kv_indptr_ptr,        # *int32 [B+1]
    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *bf16 [B, H, Dc]
    lse_ptr,              # *float32 [B*H]
    B: tl.constexpr,      # batch_size
    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    L: tl.constexpr,      # total number of tokens (len(kv_indices))
    SM_SCALE: tl.constexpr,  # scaling factor
):
    # One program per batch element
    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)       # int32
    end = tl.load(kv_indptr_ptr + b + 1)    # int32
    L_tokens = end - base

    LOG2_INV = 1.0 / tl.log(2.0)

    # Loop over heads
    for h in range(H):
        # Prepare output vector (float32)
        out_vec = tl.zeros((Dc,), dtype=tl.float32)

        # If no tokens for this batch element, set output to zeros and lse to -inf
        if L_tokens <= 0:
            out_offset = b * H * Dc + h * Dc
            tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec)
            lse_offset = b * H + h
            tl.store(lse_ptr + lse_offset, -float("inf"))
            continue

        # Load q_nope and q_pe for this head (cast to float32)
        qn_ptr = q_nope_ptr + b * H * Dc + h * Dc
        qp_ptr = q_pe_ptr + b * H * Dp + h * Dp
        qn = tl.load(qn_ptr + tl.arange(0, Dc)).to(tl.float32)  # [Dc]
        qp = tl.load(qp_ptr + tl.arange(0, Dp)).to(tl.float32)  # [Dp]

        # Pass 1: compute max over scaled logits across tokens (numerically stable lse)
        max_val = tl.full((), -float("inf"), dtype=tl.float32)
        for i in range(L_tokens):
            idx = base + i
            tok_idx = tl.load(kv_indices_ptr + idx)  # token index for this segment
            Kc_row_ptr = ckv_cache_ptr + tok_idx * Dc
            Kp_row_ptr = kpe_cache_ptr + tok_idx * Dp

            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc)).to(tl.float32)  # [Dc]
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp)).to(tl.float32)  # [Dp]

            dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
            val = (dot1 + dot2) * SM_SCALE
            max_val = tl.maximum(max_val, val)

        # Pass 2: compute sum_exp to finalize lse and accumulate output
        sum_exp = tl.full((), 0.0, dtype=tl.float32)
        for i in range(L_tokens):
            idx = base + i
            tok_idx = tl.load(kv_indices_ptr + idx)  # token index for this segment
            Kc_row_ptr = ckv_cache_ptr + tok_idx * Dc
            Kp_row_ptr = kpe_cache_ptr + tok_idx * Dp

            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc)).to(tl.float32)  # [Dc]
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp)).to(tl.float32)  # [Dp]

            dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
            val = (dot1 + dot2) * SM_SCALE
            attn_i = tl.exp(val - max_val)
            sum_exp += attn_i

        lse_val = tl.log(sum_exp) + max_val
        lse_val = lse_val * LOG2_INV  # convert to base-2

        # Re-do passes with normalized attn to compute output vector
        for i in range(L_tokens):
            idx = base + i
            tok_idx = tl.load(kv_indices_ptr + idx)  # token index for this segment
            Kc_row_ptr = ckv_cache_ptr + tok_idx * Dc
            Kp_row_ptr = kpe_cache_ptr + tok_idx * Dp

            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc)).to(tl.float32)  # [Dc]
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp)).to(tl.float32)  # [Dp]

            dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
            dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
            val = (dot1 + dot2) * SM_SCALE
            attn_i = tl.exp(val - lse_val)
            out_vec += attn_i * Kc_row

        # Store output for head h
        out_offset = b * H * Dc + h * Dc
        tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec)

        # Store lse per (b, h)
        lse_offset = b * H + h
        tl.store(lse_ptr + lse_offset, lse_val)

          * ckv_cache: [N, 1, Dc], dtype bfloat16

          * kv_indices: [L], int32

        - Returns:
          * output: [B, H, Dc], dtype bfloat16
          * lse: [B, H], dtype float32
        """
        assert q_nope.is_cuda and q_pe.is_cuda and ckv_cache.is_cuda and kpe_cache.is_cuda, "Inputs must be CUDA tensors."
        B, H, Dc = q_nope.shape

        N, _, _ = ckv_cache.shape
        assert ckv_cache.shape[1] == 1 and kpe_cache.shape[1] == 1, "kv_cache must have shape [N, 1, D]"

        )

        return output_bf16, lse

            Kc_row_ptr = ckv_cache_ptr + tok_idx * Dc
            Kp_row_ptr = kpe_cache_ptr + tok_idx * Dp
            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc)).to(tl.float32)
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp)).to(tl.float32)
            dot1 = tl.sum(qn * Kc_row, axis=0)
            dot2 = tl.sum(qp * Kp_row, axis=0)
            val = (dot1 + dot2) * SM_SCALE
            sum_exp += tl.exp(val - max_val)

        lse_val = tl.log(sum_exp) + max_val
        lse_val = lse_val * LOG2_INV

        for i in range(L_tokens):
            idx = base + i
            tok_idx = tl.load(kv_indices_ptr + idx)
            Kc_row_ptr = ckv_cache_ptr + tok_idx * Dc
            Kp_row_ptr = kpe_cache_ptr + tok_idx * Dp
            Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc)).to(tl.float32)
            Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp)).to(tl.float32)
            dot1 = tl.sum(qn * Kc_row, axis=0)
            dot2 = tl.sum(qp * Kp_row, axis=0)
            val = (dot1 + dot2) * SM_SCALE
            attn_i = tl.exp(val - lse_val)
            out_vec += attn_i * Kc_row

        out_offset = b * H * Dc + h * Dc
        tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec)
        lse_offset = b * H + h
        tl.store(lse_ptr + lse_offset, lse_val)
