import torch
import triton
import triton.language as tl


@triton.jit
def matvec_row_kernel(
    A_ptr,       # *float32, pointer to q vector (length K), flattened to 1D
    B_ptr,       # *float32, pointer to B rows, shape [M, K], contiguous
    C_ptr,       # *float32, pointer to output, shape [M], contiguous
    M,           # int, number of rows in B (runtime)
    K: tl.constexpr,           # int, length of q and columns of B (compile-time)
    BLOCK_K: tl.constexpr = 64 # tile size along K
):
    # One program computes a single output element: C[i] = A @ B[i, :]
    i = tl.program_id(0)  # index over M
    acc = 0.0
    # Loop over K in tiles
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)  # vector of indices along K (constexpr)
        mask_k = offs_k < K
        # Load q chunk
        a = tl.load(A_ptr + offs_k, mask=mask_k, other=0.0)
        # Load B row chunk: row i, cols offs_k
        b = tl.load(B_ptr + i * K + offs_k, mask=mask_k, other=0.0)
        # Accumulate dot product of a and b
        acc += tl.sum(a * b, axis=0)
    # Store the result for this i
    tl.store(C_ptr + i, acc)


@triton.jit
def softmax_lse_kernel(
    logits_ptr,         # *float32, pointer to logits vector of length M
    lse_ptr,            # *float32, pointer to per-head lse, length num_qo_heads
    M: tl.constexpr,    # length of logits vector (compile-time for kernel specialization)
    scale: tl.constexpr,       # scaling factor (compile-time float)
    inv_ln2: tl.constexpr       # 1.0 / ln(2) (compile-time float)
):
    # Each program handles one head index h
    h = tl.program_id(0)
    # Compute max for numerical stability
    max_val = -float("inf")
    for m0 in range(0, M, 128):
        offs = m0 + tl.arange(0, 128)
        mask = offs < M
        x = tl.load(logits_ptr + offs, mask=mask, other=-float("inf"))
        cur_max = tl.max(x, axis=0)
        max_val = tl.maximum(max_val, cur_max)

    # Compute sum of exp(logits * scale)
    sum_val = 0.0
    for m0 in range(0, M, 128):
        offs = m0 + tl.arange(0, 128)
        mask = offs < M
        x = tl.load(logits_ptr + offs, mask=mask, other=-float("inf"))
        e = tl.exp((x - max_val) * scale)
        # Zero out contributions from masked positions
        e = tl.where(mask, e, 0.0)
        sum_val += tl.sum(e, axis=0)

    lse = (max_val + tl.log(sum_val)) * inv_ln2
    tl.store(lse_ptr + h, lse)


@triton.jit
def matvec_accum_kernel(
    attn_ptr,           # *float32, pointer to attn vector of length M
    B_ptr,              # *float32, pointer to B rows, shape [M, K], contiguous
    C_ptr,              # *float32, pointer to output vector, shape [K], contiguous
    M,                  # int, number of rows in B (runtime)
    K: tl.constexpr,            # int, length of output vector (compile-time)
    BLOCK_M: tl.constexpr = 128 # tile size along M
):
    # One program accumulates one output element c_out[k] over all M
    k = tl.program_id(0)  # index over K
    acc = 0.0
    for m0 in range(0, M, BLOCK_M):
        offs_m = m0 + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M
        # Load attn chunk for these M positions
        attn = tl.load(attn_ptr + offs_m, mask=mask_m, other=0.0)
        # Load corresponding B rows columns k: B_ptr is [M, K], col k
        b = tl.load(B_ptr + offs_m * K + k, mask=mask_m, other=0.0)
        acc += tl.sum(attn * b, axis=0)
    tl.store(C_ptr + k, acc)

        output = torch.zeros((batch_size, num_qo_heads, head_dim_ckv), dtype=torch.bfloat16, device=device)
        lse = torch.full((batch_size, num_qo_heads), -float("inf"), dtype=torch.float32, device=device)

        inv_ln2 = 1.4426950408889634  # 1 / ln(2)

        for b in range(batch_size):
            page_beg = int(kv_indptr[b].item())
            page_end = int(kv_indptr[b + 1].item())
            if page_beg >= page_end:

            Kp = Kp_all[tok_idx]  # [M, 64], float32
            Kc = Kc.contiguous()
            Kp = Kp.contiguous()

            # Per-head outputs
            for h in range(num_qo_heads):
                qn = q_nope[b, h, :].to(torch.float32).contiguous()  # [512]
                qp = q_pe[b, h, :].to(torch.float32).contiguous()   # [64]

                # Allocate logits buffer [M]
                logits = torch.empty((M,), dtype=torch.float32, device=device)

                # Compute logits = qn @ Kc.T + qp @ Kp.T
                # Grid over M, loop over K in tiles
                grid = (M,)
                matvec_row_kernel[grid](
                    qn, Kc, logits, M, K=Kc.shape[1], BLOCK_K=64, num_warps=4
                )
                # Add contribution from qp @ Kp.T
                # Recompute for each head to keep simplicity; Kp.T is [64, M], but we do row-wise above per qn chunk
                # Note: The above kernel computes qn @ Kc.T, now we add qp @ Kp.T similarly.
                # We'll re-run with qp and Kp, storing into logits.
                # However, since we already have qn @ Kc.T, we can compute qp @ Kp.T in the same call by swapping params.
                # To keep clear, we recompute by swapping q and B.
                # But simpler: use a second call for clarity; Triton allows repeated kernel launch.

                # We need to compute logits += qp @ Kp.T. The kernel above expects a q vector [K] and B [M,K].
                # We can compute qp @ Kp.T by swapping: run the same kernel with A=qp [64], B=Kp.T [M,64], output logit_len=M.
                # However, computing per-row matvec above does not include Kp contribution. Instead, we compute it separately.

                # Compute Kp_T as [M, 64] already available as Kp; we need per-row dot with qp. Let's do it explicitly:
                # We'll run matvec_row_kernel with A=qp, B=Kp, output into a separate logits_qp of shape [M] then add.
                logits_qp = torch.empty((M,), dtype=torch.float32, device=device)
                grid2 = (M,)
                matvec_row_kernel[grid2](
                    qp, Kp, logits_qp, M, K=Kp.shape[1], BLOCK_K=64, num_warps=4
                )
                logits = logits + logits_qp

                # Compute lse for this head
                # Run softmax_lse_kernel specialized for M
                # We need to pass M as constexpr. Triton requires constexpr for tl.arange. We handle this by launching per-M.
                # But softmax_lse_kernel expects M as constexpr. We can specialize by re-launching per M, which Triton supports here.
                softmax_lse_kernel[(1,)](
                    logits, lse[b], M, scale=float(sm_scale), inv_ln2=inv_ln2
                )

                # Compute out = attn @ Kc
                attn = torch.empty((M,), dtype=torch.float32, device=device)
                # attn[i] = exp((logits[i] - max) * scale) / sum_j exp((logits[j] - max) * scale)
                # We already have max from lse kernel? No, we need to compute it here.
                # We'll recompute max and sum here in Python using torch ops for simplicity and correctness.
                # This avoids Triton vector-reduction complexity. Since M is small per batch, this is acceptable and correct.
                logits_scaled = logits * float(sm_scale)
                max_val = torch.max(logits_scaled)
                sum_exp = torch.sum(torch.exp(logits_scaled - max_val))
                attn = torch.exp(logits_scaled - max_val) / sum_exp  # [M]

                # out[h, :] = sum_i attn[i] * Kc[i, :]
                out_vec = torch.empty((head_dim_ckv,), dtype=torch.float32, device=device)
                grid_acc = (head_dim_ckv,)
                matvec_accum_kernel[grid_acc](
                    attn, Kc, out_vec, M, K=head_dim_ckv, BLOCK_M=128, num_warps=4
                )
                output[b, h, :] = out_vec.to(torch.bfloat16)

        return output, lse
