Strategy in Triton
- Kernel 1 (_lse_per_head_kernel): Computes lse per head for batch b.
  - One Triton program per batch element b. Loop over heads h.
  - For each head, load qn and qp vectors once.
  - Iterate tokens in tiles (BLOCK_T) with a static loop; for each token tt in tile:
    - Load Kc_row and Kp_row (1D vector loads).
    - Compute logits_scalar = (sum(qn * Kc_row) + sum(qp * Kp_row)) * sm_scale
    - Maintain per-column token_max_vec and token_sum_vec to compute lse afterward: lse = token_max_vec + log(token_sum_vec) / ln(2).
- Kernel 2 (_compute_output_kernel): Computes output per head for batch b.
  - Same structure: loop over tiles, compute logits_scalar per token, compute softmax scaling factors, and accumulate out[h, :] += exp(logits - token_max_vec) / token_sum_vec * Kc_row.

This avoids the problematic 2D broadcasted loads and ensures all heavy computation is in Triton.

```python
import math
import torch

import triton
import triton.language as tl


@triton.jit
def _lse_per_head_kernel(
    q_nope_rows_ptr,     # *f32, shape [H, D1], contiguous
    q_pe_rows_ptr,       # *f32, shape [H, D2], contiguous
    Kc_sub_ptr,          # *f32, shape [L_tokens, D1], contiguous
    Kp_sub_ptr,          # *f32, shape [L_tokens, D2], contiguous
    lse_out_ptr,         # *f32, shape [B, H], contiguous
    H: tl.int32,         # number of heads (runtime)
    D1: tl.constexpr,    # head_dim_ckv (512)
    D2: tl.constexpr,    # head_dim_kpe (64)
    L_tokens: tl.int32,  # runtime
    sm_scale: tl.float32,
    BLOCK_T: tl.constexpr,
):
    # One Triton program per batch element
    b = tl.program_id(axis=0)
    for h in range(0, H):
        # Load q vectors for this head as 1D constexpr vectors
        qn = tl.load(q_nope_rows_ptr + h * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
        qp = tl.load(q_pe_rows_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        # Initialize per-column max and sum across tokens
        token_max_vec = tl.full((D1,), -float("inf"), dtype=tl.float32)
        token_sum_vec = tl.zeros((D1,), dtype=tl.float32)

        num_tiles = (L_tokens + BLOCK_T - 1) // BLOCK_T
        for tile in range(0, num_tiles):
            t0 = tile * BLOCK_T
            # Unroll over each token within the tile
            for tt in tl.static_range(0, BLOCK_T):
                t = t0 + tt
                valid = t < L_tokens
                # Load Kc_row and Kp_row as 1D vectors
                Kc_row = tl.load(Kc_sub_ptr + t * D1 + tl.arange(0, D1), mask=valid, other=0.0).to(tl.float32)  # [D1]
                Kp_row = tl.load(Kp_sub_ptr + t * D2 + tl.arange(0, D2), mask=valid, other=0.0).to(tl.float32)  # [D2]

                # Compute scalar logits for this token
                dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
                dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
                logits_scalar = (dot1 + dot2) * sm_scale  # scalar

                # Update per-column max and sum
                token_max_vec = tl.maximum(token_max_vec, logits_scalar)
                token_sum_vec += tl.where(valid, tl.exp(logits_scalar - token_max_vec), 0.0)

        # Compute lse and store: lse = max + log(sum exp(...)) / ln(2)
        lse_val = token_max_vec + tl.log(token_sum_vec) / 1.4426950408889634  # 1 / ln(2)
        tl.store(lse_out_ptr + b * H + h, lse_val)

    Kp_sub_ptr,          # *f32, shape [L_tokens, D2], contiguous

        qp = tl.load(q_pe_rows_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        token_max_vec = tl.full((D1,), -float("inf"), dtype=tl.float32)

        # First pass: compute token_max_vec and token_sum_vec

                dot1 = tl.sum(qn * Kc_row, axis=0)
                dot2 = tl.sum(qp * Kp_row, axis=0)
                logits_scalar = (dot1 + dot2) * sm_scale
                token_max_vec = tl.maximum(token_max_vec, logits_scalar)
                token_sum_vec += tl.where(valid, tl.exp(logits_scalar - token_max_vec), 0.0)

        # Second pass: accumulate output for each token
        for tile in range(0, num_tiles):
            t0 = tile * BLOCK_T
            for tt in tl.static_range(0, BLOCK_T):
                t = t0 + tt
                valid = t < L_tokens
                Kc_row = tl.load(Kc_sub_ptr + t * D1 + tl.arange(0, D1), mask=valid, other=0.0).to(tl.float32)  # [D1]

                dot1 = tl.sum(qn * Kc_row, axis=0)
                dot2 = tl.sum(qp * Kp_row, axis=0)
                logits_scalar = (dot1 + dot2) * sm_scale
                attn = tl.where(valid, tl.exp(logits_scalar - token_max_vec) / token_sum_vec, 0.0)
                # Store out[b, h, :] += attn * Kc_row
                out_row = tl.load(out_ptr + b * (H * D1) + h * D1 + tl.arange(0, D1), mask=True, other=0.0).to(tl.float32)
                out_row += attn * Kc_row
                tl.store(out_ptr + b * (H * D1) + h * D1 + tl.arange(0, D1), out_row)

        D2 = q_pe.shape[2]

        q_nope_rows = q_nope_rows.view(B, H, D1)

        lse = torch.empty((B, H), dtype=torch.float32, device=device)

        )
        # Note: The above call is a template. We need to recompute L_tokens per batch element dynamically.
        # To correctly compute L_tokens per element, we relaunch with dynamic args using a small loop over b.

        # Correct approach: iterate b and compute L_tokens for each, then launch per element. To do so, we can
        # either reinitialize kernels per b or restructure. Triton supports program_id per launch. We'll do it in two steps:
        # First, compute lse for each b by relaunching with correct L_tokens. Then compute out for each b.

        # However, Triton kernels are stateless and cannot share state across launches. We will compute lse per b,

        # Compute lse per b
        for b in range(B):
            # compute L_tokens for this batch element
            start = int(kv_indptr[b].item())
            end = int(kv_indptr[b + 1].item())
            L_tokens = end - start
            if L_tokens <= 0:
                continue
            # Slice Kc and Kp for this batch
            Kc_sub = Kc_all[kv_indices[start:end], :].contiguous()
            Kp_sub = Kp_all[kv_indices[start:end], :].contiguous()

            # We need to pass L_tokens. Triton expects constexpr tiling, but L_tokens is runtime.
            # Workaround: launch once per b with program_id=axis=0. Triton handles scalar parameters.

        return output, lse
```

```python
                H=H, D1=D1, D2=D2, L_tokens=L_tokens, sm_scale=sm_scale, BLOCK_T=BLOCK_T,
            )

            # Compute output for this b and each head
            _compute_output_per_b_kernel[(1,)](
                q_nope_rows[b], q_pe_rows[b], Kc_sub, Kp_sub, out[b],
                H=H, D1=D1, D2=D2, L_tokens=L_tokens, sm_scale=sm_scale, BLOCK_T=BLOCK_T,
            )

        # Cast output to bfloat16 to match original behavior
        output = out.to(torch.bfloat16)
        return output, lse
```

import math
import torch

import triton
import triton.language as tl


@triton.jit
def _lse_per_b_kernel(
    q_nope_rows_ptr,     # *f32, shape [H, D1], contiguous
    q_pe_rows_ptr,       # *f32, shape [H, D2], contiguous
    Kc_sub_ptr,          # *f32, shape [L_tokens, D1], contiguous
    Kp_sub_ptr,          # *f32, shape [L_tokens, D2], contiguous
    lse_out_ptr,         # *f32, shape [B, H], contiguous
    H: tl.int32,         # number of heads (runtime)
    D1: tl.constexpr,    # head_dim_ckv (512)
    D2: tl.constexpr,    # head_dim_kpe (64)
    L_tokens: tl.int32,  # runtime (per b)
    sm_scale: tl.float32,
    BLOCK_T: tl.constexpr,
):
    # One Triton program per batch element b
    b = tl.program_id(axis=0)

    for h in range(0, H):
        # Load q vectors for this head
        qn = tl.load(q_nope_rows_ptr + h * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
        qp = tl.load(q_pe_rows_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        # Initialize per-column max and sum across tokens
        token_max_vec = tl.full((D1,), -float("inf"), dtype=tl.float32)
        token_sum_vec = tl.zeros((D1,), dtype=tl.float32)

        # Tile over tokens; static unroll over BLOCK_T
        for tile in tl.static_range(0, (L_tokens + BLOCK_T - 1) // BLOCK_T):
            t0 = tile * BLOCK_T
            for tt in tl.static_range(0, BLOCK_T):
                t = t0 + tt
                valid = t < L_tokens
                # Load Kc_row and Kp_row as 1D vectors
                Kc_row = tl.load(Kc_sub_ptr + t * D1 + tl.arange(0, D1), mask=valid, other=0.0).to(tl.float32)  # [D1]
                Kp_row = tl.load(Kp_sub_ptr + t * D2 + tl.arange(0, D2), mask=valid, other=0.0).to(tl.float32)  # [D2]

                # Compute scalar logits for this token
                dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
                dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
                logits_scalar = (dot1 + dot2) * sm_scale  # scalar

                # Update per-column max and sum (masked by valid)
                token_max_vec = tl.maximum(token_max_vec, logits_scalar)
                token_sum_vec += tl.where(valid, tl.exp(logits_scalar - token_max_vec), 0.0)

        #
