import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_nope_ptr,           # *float32 [B, H, Dc]
    q_pe_ptr,             # *float32 [B, H, Dp]
    ckv_cache_ptr,        # *float32 [N, 1, Dc]
    kpe_cache_ptr,        # *float32 [N, 1, Dp]
    kv_indptr_ptr,        # *int32 [B+1]
    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *float32 [B, H, Dc]
    lse_ptr,              # *float32 [B*H]
    B: tl.constexpr,      # batch_size
    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    SM_SCALE: tl.constexpr,  # scaling factor
    MAX_TOKENS: tl.constexpr,  # upper bound for tokens per batch element
):
    # One program per batch element
    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)
    end = tl.load(kv_indptr_ptr + b + 1)
    L_tokens = end - base

    # Loop over heads
    for h in range(H):
        # Prepare vectors
        # Read q_nope[b, h, :] and q_pe[b, h, :]
        qn_ptr = q_nope_ptr + b * H * Dc + h * Dc
        qp_ptr = q_pe_ptr + b * H * Dp + h * Dp

        qn = tl.load(qn_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0)  # float32
        qp = tl.load(qp_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0)  # float32

        # Initialize logits_scaled vector with -inf
        logits_scaled = tl.full((MAX_TOKENS,), -float('inf'), dtype=tl.float32)

        # Compute logits for each token index (up to MAX_TOKENS)
        for i in range(MAX_TOKENS):
            if i < L_tokens:
                idx = tl.load(kv_indices_ptr + base + i)
                Kc_row_ptr = ckv_cache_ptr + idx * Dc
                Kp_row_ptr = kpe_cache_ptr + idx * Dp

                # Load Kc and Kp rows as float32
                Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0)
                Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0)

                # Dot products
                dot1 = tl.sum(qn * Kc_row, axis=0)
                dot2 = tl.sum(qp * Kp_row, axis=0)

                val = (dot1 + dot2) * SM_SCALE
                # Update logits_scaled at position i
                logits_scaled = tl.where(tl.arange(0, MAX_TOKENS) == i, val, logits_scaled)

        # Stable logsumexp over valid tokens
        max_val = tl.max(logits_scaled, axis=0)
        sum_exp = tl.sum(tl.exp(logits_scaled - max_val), axis=0)
        lse_val = tl.log(sum_exp) + max_val
        # Scale by 1/ln(2)
        lse_val = lse_val / tl.log(2.0)

        # Compute final output vector: out[b, h, :] = sum_i attn_i * Kc_rows[i]
        out_vec = tl.zeros((Dc,), dtype=tl.float32)
        for i in range(MAX_TOKENS):
            attn_i = tl.exp((logits_scaled[i] - lse_val))
            if i < L_tokens:
                idx = tl.load(kv_indices_ptr + base + i)

                Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0)
                out_vec += attn_i * Kc_row

        # Store output for head h
        out_offset = b * H * Dc + h * Dc
        tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec)

        # Store lse per (b, h)
        lse_offset = b * H + h
        tl.store(lse_ptr + lse_offset, lse_val)

The original PyTorch logic per batch element b and head h:
- Compute L_tokens = kv_indptr[b+1] - kv_indptr[b].
- Gather token indices tok_idx = kv_indices[kv_indptr[b]: kv_indptr[b+1]].
- Kc = ckv_cache[tok_idx] and Kp = kpe_cache[tok_idx].
- logits = (q_nope[b, h] @ Kc.T) + (q_pe[b, h] @ Kp.T).
- lse[b, h] = logsumexp(logits * sm_scale) / ln(2).
- out[b, h] = softmax(logits_scaled) @ Kc.

In Triton:
- One program per b. For each h:
  - Initialize acc = -inf (running max for logsumexp).
  - Loop i from 0 to MAX_TOKENS-1, mask i < L_tokens:
    - idx = kv_indices[base + i], where base = kv_indptr[b].
    - Load Kc_row and Kp_row as float32.
    - Compute dot1 = sum(qn * Kc_row), dot2 = sum(qp * Kp_row).
    - val = (dot1 + dot2) * sm_scale; acc = max(acc, val).
    - Maintain a small vector scaled_logits for later sum_exp (not vector update via tl.where).
  - Compute sum_exp = sum(exp(scaled_logits - acc)); lse_val = log(sum_exp) + acc; divide by ln(2).
  - Re-loop i to compute attn_i = exp(scaled_logits[i] - lse_val) and accumulate out_vec += attn_i * Kc_row.
  - Store out_vec to output[b, h, :] (float32), and store lse_val to lse[b, h].

This avoids vector assignment pitfalls and uses reductions Triton supports. It also ensures the Triton kernel is actually invoked.

Final code (ModelNew with Triton kernel launch)

```python
import torch
import triton
import triton.language as tl


@triton.jit
def attention_kernel(
    q_nope_ptr,           # *bf16 [B, H, Dc]
    q_pe_ptr,             # *bf16 [B, H, Dp]
    ckv_cache_ptr,        # *bf16 [N, 1, Dc]
    kpe_cache_ptr,        # *bf16 [N, 1, Dp]
    kv_indptr_ptr,        # *int32 [B+1]
    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *bf16 [B, H, Dc] (we store float32, host casts to bf16)
    lse_ptr,              # *float32 [B*H]
    B: tl.constexpr,      # batch_size
    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    SM_SCALE: tl.constexpr,  # scaling factor
    MAX_TOKENS: tl.constexpr,  # upper bound for tokens per batch element
):
    # One program per batch element
    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)        # int32
    end = tl.load(kv_indptr_ptr + b + 1)     # int32
    L_tokens = end - base                     # int32

    # Loop over heads
    for h in range(H):
        # Read q_nope[b, h, :] and q_pe[b, h, :]
        qn_ptr = q_nope_ptr + b * H * Dc + h * Dc
        qp_ptr = q_pe_ptr + b * H * Dp + h * Dp

        # Load as bf16, Triton will infer; we will use these values in math by casting in load calls
        # Note: Triton load from pointers returns the tensor dtype; q_nope/q_pe are bf16, we cast to f32 after load.
        qn = tl.load(qn_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)
        qp = tl.load(qp_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0).to(tl.float32)

        # Initialize running max and arrays for scaled logits
        acc = -float('inf')  # float32 scalar for max
        scaled_logits = tl.full((MAX_TOKENS,), -float('inf'), dtype=tl.float32)

        # Compute scaled logits for each token index (up to MAX_TOKENS) with mask
        for i in range(MAX_TOKENS):
            if i < L_tokens:
                idx = tl.load(kv_indices_ptr + base + i)  # int32 token index
                Kc_row_ptr = ckv_cache_ptr + idx * Dc
                Kp_row_ptr = kpe_cache_ptr + idx * Dp

                # Load Kc and Kp rows as float32
                Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dc), mask=tl.arange(0, Dc) < Dc, other=0.0).to(tl.float32)
                Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0).to(tl.float32)

                # Dot products
                dot1 = tl.sum(qn * Kc_row, axis=0)
                dot2 = tl.sum(qp * Kp_row, axis=0)

                val = (dot1 + dot2) * SM_SCALE
                # Update running max
                acc = tl.maximum(acc, val)
                # Store scaled logit for position i
                scaled_logits = tl.where(tl.arange(0, MAX_TOKENS) == i, val, scaled_logits)

        # Stable sum of exp over valid tokens using running max
        sum_exp = tl.zeros((), dtype=tl.float32)
        for i in range(MAX_TOKENS):
            if i < L_tokens:
                sum_exp += tl.exp(scaled_logits[i] - acc)

        lse_val = tl.log(sum_exp) + acc
        # Scale by 1/ln(2)
        lse_val = lse_val / tl.log(2
