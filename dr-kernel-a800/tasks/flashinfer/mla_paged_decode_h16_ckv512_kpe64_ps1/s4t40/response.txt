- Kc = ckv_cache[tokens, 0, :] → [M, 512]
- Kp = kpe_cache[tokens, 0, :] → [M, 64]

- logits_scaled = logits * sm_scale
- lse[h] = logsumexp(logits_scaled) / ln(2) computed per head

def matvec_row_kernel(
    A_ptr,          # *float32, pointer to q vector (length K), flattened to 1D
    B_ptr,          # *float32, pointer to B rows, shape [M, K], contiguous
    C_ptr,          # *float32, pointer to output, shape [M], contiguous
    M,              # int, number of rows in B (runtime)
    K: tl.constexpr,             # int, length of q and columns of B (compile-time)
    BLOCK_K: tl.constexpr = 64   # tile size along K
):
    # One program computes a single output element: C[i] = A @ B[i, :]
    i = tl.program_id(0)  # index over M
    acc = 0.0
    # Loop over K in tiles; K is tl.constexpr so this is compiled-time loop
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)  # vector along K
        mask_k = offs_k < K
        # Load q chunk
        a = tl.load(A_ptr + offs_k, mask=mask_k, other=0.0)
        # Load B row chunk: B_ptr is [M, K], row i, cols offs_k
        b = tl.load(B_ptr + i * K + offs_k, mask=mask_k, other=0.0)
        acc += tl.sum(a * b, axis=0)
    tl.store(C_ptr + i, acc)

@triton.jit
def matvec_row_kernel(
    A_ptr,          # *float32, pointer to q vector (length K), flattened to 1D
    B_ptr,          # *float32, pointer to B rows, shape [M, K], contiguous
    C_ptr,          # *float32, pointer to output, shape [M], contiguous
    M: tl.constexpr,              # number of rows in B (compile-time for specialization)
    K: tl.constexpr,              # int, length of q and columns of B (compile-time)
    BLOCK_K: tl.constexpr = 64    # tile size along K
):
    # One program computes a single output element: C[i] = A @ B[i, :]
    i = tl.program_id(0)  # index over M
    acc = 0.0
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)  # vector along K
        mask_k = offs_k < K
        a = tl.load(A_ptr + offs_k, mask=mask_k, other=0.0)
        b = tl.load(B_ptr + i * K + offs_k, mask=mask_k, other=0.0)
        acc += tl.sum(a * b, axis=0)
    tl.store(C_ptr + i, acc)


@triton.jit
def compute_out_from_attn_kernel(
    attn_ptr,       # *float32, pointer to attn vector [M]
    Kc_ptr,         # *float32, pointer to Kc rows [M, 512], contiguous
    out_ptr,        # *float32, pointer to output vector [512]
    M: tl.constexpr,       # number of tokens
    Kc_dim: tl.constexpr   # 512
):
    # Compute out = attn @ Kc. One program per output dimension kk.
    kk = tl.program_id(0)
    acc = 0.0
    for i in range(0, M):
        attn_i = tl.load(attn_ptr + i)
        # Kc[i, kk] load
        val = tl.load(Kc_ptr + i * Kc_dim + kk)
        acc += attn_i * val
    tl.store(out_ptr + kk, acc)

    def __init__(self):

        # No parameters; everything is runtime and Triton kernels.

    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, kv_indptr, kv_indices, sm_scale):

        device = q_nope.device
        assert q_nope.is_cuda and q_pe.is_cuda and ckv_cache.is_cuda and kpe_cache.is_cuda and kv_indptr.is_cuda and kv_indices.is_cuda, "All tensors must be on CUDA device for Triton."

        batch_size = q_nope.shape[0]
        num_qo_heads = q_nope.shape[1]
        head_dim_ckv = q_nope.shape[2]  # 512
        head_dim_kpe = q_pe.shape[2]    # 64
        num_pages = ckv_cache.shape[0]
        Kc_dim = head_dim_ckv
        Kp_dim = head_dim_kpe

        # Gather Kc_all and Kp_all as [num_pages, K] and cast to float32
        Kc_all = ckv_cache.squeeze(1).to(torch.float32)  # [num_pages, Kc_dim]
        Kp_all = kpe_cache.squeeze(1).to(torch.float32)  # [num_pages, Kp_dim]

        # Output buffers
        output = torch.zeros((batch_size, num_qo_heads, Kc_dim), dtype=torch.float32, device=device)  # we will return cast to bfloat16
        lse = torch.full((batch_size, num_qo_heads), -float("inf"), dtype=torch.float32, device=device)

        # Precompute attn for each head per batch element (we will compute per-batch in Triton)
        for b in range(batch_size):
            # tokens for this batch
            if kv_indptr.numel() <= 1:
                # No tokens in this batch; skip
                lse[b] = 0.0
                output[b] = torch.zeros((num_qo_heads, Kc_dim), dtype=torch.float32, device=device)
                continue

            page_beg = int(kv_indptr[b].item())
            page_end = int(kv_indptr[b + 1].item())

            if page_beg >= page_end:
                lse[b] = 0.0
                output[b] = torch.zeros((num_qo_heads, Kc_dim), dtype=torch.float32, device=device)
                continue

            M = page_end - page_beg  # number of tokens for this batch
            tokens = kv_indices[page_beg:page_end]  # [M]

            # Gather Kc and Kp for these tokens
