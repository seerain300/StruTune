Key challenges for Triton implementation
- Triton kernels need compile-time tile sizes for tl.arange. Dynamic loop bounds (runtime L_tokens) can cause compilation errors unless we tile and use tl.static_range or other patterns.
- We must avoid Python for-loops over runtime L_tokens. Instead, iterate over tiles of tokens and use masks for boundary checks.
- We must compute lse per head. While softmax typically needs all tokens, we can compute per-head max and sum(exp(.)) via reductions. However, to keep it simple and correct, we will compute the token-wise logits and then reduce. Triton can do this via loops over tiles and masks.

Plan to fix and optimize
- Keep all heavy computation in Triton kernels. No torch matmul/softmax in host code.
- Precompute and pass:

- Triton kernel 1: _compute_lse_per_head_kernel

    - Load qn[h, :] and qp[h, :] as vectors using tl.arange(0, D1) and tl.arange(0, D2).
    - Initialize logits_buf = zeros([D1], fp32).

- Triton kernel 2: _compute_out_kernel

    - Reload qn and qp.
    - Initialize out_vec = zeros([D1], fp32).
    - Initialize token_max = -inf (fp32).
    - First pass over tiles:
      - Same as above to compute token_max.
    - Second pass:
      - Recompute logits_buf (same as above), then compute exp(logits_buf - token_max). Multiply with corresponding Kc_row and accumulate into out_vec. Finally, store out_vec.to(bfloat16) to output[b, h, :].
- Host code in ModelNew.forward:
  - Compute q_nope_rows[b] and q_pe_rows[b] in fp32.

    q_pe_rows_ptr,         # *f32, shape [H, D2], contiguous
    Kc_sub_ptr,            # *f32, shape [L_tokens, D1], contiguous (per batch b)
    Kp_sub_ptr,            # *f32, shape [L_tokens, D2], contiguous (per batch b)
    lse_out_ptr,           # *f32, shape [B, H], contiguous
    H: tl.int32,           # number of heads (runtime)
    D1: tl.constexpr,      # head_dim_ckv (512), constexpr
    D2: tl.constexpr,      # head_dim_kpe (64), constexpr
    L_tokens: tl.int32,    # number of tokens in this batch element (runtime)
    sm_scale: tl.float32,  # scaling factor
    BLOCK_T: tl.constexpr, # tile size for tokens (e.g., 64)
):
    # One Triton program per batch element
    b = tl.program_id(axis=0)

    # Loop over heads; we assume H is passed as int, but Triton likes static loops here
    for h in range(0, H):
        # Load q vectors for this head
        qn = tl.load(q_nope_rows_ptr + h * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
        qp = tl.load(q_pe_rows_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        # Initialize per-column logits buffer
        logits_buf = tl.zeros((D1,), dtype=tl.float32)

        # Tile over tokens and accumulate
        # We'll iterate tiles of size BLOCK_T
        num_tiles = (L_tokens + BLOCK_T - 1) // BLOCK_T

        for tile in range(0, num_tiles):
            t_idx = tile * BLOCK_T + tl.arange(0, BLOCK_T)
            mask_t = t_idx < L_tokens

            # Compute Kc_rows and Kp_rows for this tile
            Kc_rows = tl.load(
                Kc_sub_ptr + t_idx * D1 + tl.arange(0, D1),
                mask=mask_t[:, None],  # [BLOCK_T, 1] broadcast over D1
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D1]
            Kp_rows = tl.load(
                Kp_sub_ptr + t_idx * D2 + tl.arange(0, D2),
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D2]

            # For each tt in the tile, accumulate into logits_buf
            # Using static_range to satisfy Triton's compile-time loop requirement
            for tt in tl.static_range(0, BLOCK_T):
                valid = mask_t[tt]
                Kc_row = Kc_rows[tt, :]   # [D1]
                Kp_row = Kp_rows[tt, :]   # [D2]

                dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
                dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
                logits_scalar = (dot1 + dot2) * sm_scale
                # Only add if valid
                logits_buf += tl.where(valid, logits_scalar, 0.0)

        # Compute lse for this head: logsumexp(logits_buf) / ln(2)
        token_max = tl.max(logits_buf, axis=0)  # scalar
        sum_exp = tl.sum(tl.exp(logits_buf - token_max), axis=0)  # scalar
        lse_h = token_max + tl.log(sum_exp) / tl.log(2.0)  # scalar
        # Store to lse_out[b, h]
        tl.store(lse_out_ptr + b * H + h, lse_h)

    Kc_sub_ptr,            # *f32, shape [L_tokens, D1], contiguous (per batch b)
    Kp_sub_ptr,            # *f32, shape [L_tokens, D2], contiguous (per batch b)
    lse_out_ptr,           # *f32, shape [B, H], contiguous
    H: tl.int32,           # number of heads (runtime)
    D1: tl.constexpr,      # head_dim_ckv (512), constexpr
    D2: tl.constexpr,      # head_dim_kpe (64), constexpr
    L_tokens: tl.int32,    # number of tokens in this batch element (runtime)
    sm_scale: tl.float32,  # scaling factor (not used in lse directly)
    BLOCK_T: tl.constexpr, # tile size for tokens (e.g., 64)
):
    # One Triton program per batch element
    b = tl.program_id(axis=0)

    for h in range(0, H):
        # Load q vectors for this head
        qn = tl.load(q_nope_rows_ptr + h * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
        qp = tl.load(q_pe_rows_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        # Per-column logits buffer across all tokens
        logits_buf = tl.zeros((D1,), dtype=tl.float32)

        # Tile over tokens and accumulate logits per column
        num_tiles = (L_tokens + BLOCK_T - 1) // BLOCK_T
        for tile in range(0, num_tiles):
            t_idx = tile * BLOCK_T + tl.arange(0, BLOCK_T)  # [BLOCK_T]
            mask_t = t_idx < L_tokens

            # Load Kc_rows and Kp_rows for this tile: [BLOCK_T, D1] and [BLOCK_T, D2]
            Kc_rows = tl.load(
                Kc_sub_ptr + t_idx * D1 + tl.arange(0, D1),
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D1]
            Kp_rows = tl.load(
                Kp_sub_ptr + t_idx * D2 + tl.arange(0, D2),
                mask=mask_t[:, None],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_T, D2]

            # For each tt in the tile, accumulate into logits_buf
            for tt in tl.static_range(0, BLOCK_T):
                valid = mask_t[tt]  # scalar bool
                Kc_row = Kc_rows[tt, :]  # [D1]
                Kp_row = Kp_rows[tt, :]  # [D2]
                dot1 = tl.sum(qn * Kc_row, axis=0)  # scalar
                dot2 = tl.sum(qp * Kp_row, axis=0)  # scalar
                logits_scalar = (dot1 + dot2) * sm_scale
                logits_buf += tl.where(valid, logits_scalar, 0.0)

        # Compute lse for this head: logsumexp(logits_buf) / ln(2)
        token_max = tl.max(logits_buf, axis=0)  # scalar
        sum_exp = tl.sum(tl.exp(logits_buf - token_max), axis=0)  # scalar
        lse_h = token_max + tl.log(sum_exp) / tl.log(2.0)  # scalar
        # Store to lse_out[b, h]
        tl.store(lse_out_ptr + b * H + h, lse_h)
