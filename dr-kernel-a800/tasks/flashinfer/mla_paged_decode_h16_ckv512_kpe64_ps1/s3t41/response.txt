import torch
import math
import triton
import triton.language as tl


@triton.jit
def fused_logits_kernel(
    qn_ptr,   # *f32 [H, Dq]
    qp_ptr,   # *f32 [H, Dp]
    Kc_ptr,   # *f32 [T, Dq]
    Kp_ptr,   # *f32 [T, Dp]
    out_ptr,  # *f32 [H, T]
    H: tl.constexpr,    # number of heads
    Dq: tl.constexpr,   # head_dim_ckv = 512
    Dp: tl.constexpr,   # head_dim_kpe = 64
    T: tl.constexpr,    # number of tokens L_tokens
    BLOCK_T: tl.constexpr
):
    # grid: (H, ceil_div(T, BLOCK_T))
    h = tl.program_id(0)
    t_block = tl.program_id(1)

    offs_t = t_block * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = offs_t < T

    # Load qn[h, :] and qp[h, :]
    qn = tl.load(qn_ptr + h * Dq + tl.arange(0, Dq), mask=True, other=0.0)  # [Dq]
    qp = tl.load(qp_ptr + h * Dp + tl.arange(0, Dp), mask=True, other=0.0)  # [Dp]

    # Accumulate dot products for this block of tokens
    acc = tl.zeros((BLOCK_T,), dtype=tl.float32)

    # Loop over Dq in tiles
    for d in range(0, Dq, BLOCK_T):  # Dq=512, BLOCK_T=128, so 4 iterations
        k_d = tl.arange(0, BLOCK_T)
        k_mask = (d + k_d) < Dq
        Kc_block = tl.load(Kc_ptr + (offs_t[:, None] * Dq + d + k_d[None, :]), mask=mask_t[:, None] & k_mask[None, :], other=0.0)  # [BLOCK_T, BLOCK_D]
        acc += tl.sum(Kc_block * qn[d : d + BLOCK_T][:, None], axis=0)  # sum over k_d

    # Repeat for Kp with Dp=64
    acc_p = tl.zeros((BLOCK_T,), dtype=tl.float32)
    for dp in range(0, Dp, BLOCK_T):  # Dp=64, BLOCK_T=128, masked beyond Dp
        k_p = tl.arange(0, BLOCK_T)
        k_mask_p = (dp + k_p) < Dp
        Kp_block = tl.load(Kp_ptr + (offs_t[:, None] * Dp + dp + k_p[None, :]), mask=mask_t[:, None] & k_mask_p[None, :], other=0.0)  # [BLOCK_T, BLOCK_DP]
        acc_p += tl.sum(Kp_block * qp[dp : dp + BLOCK_T][:, None], axis=0)

    acc_total = acc + acc_p
    tl.store(out_ptr + h * T + offs_t, acc_total, mask=mask_t)


@triton.jit
def softmax_row_kernel(
    logits_ptr,  # *f32 [H, T]
    out_ptr,     # *f32 [H, T]
    H: tl.constexpr,
    T: tl.constexpr,
    BLOCK_T: tl.constexpr
):
    # grid: (H, ceil_div(T, BLOCK_T))
    h = tl.program_id(0)
    t_block = tl.program_id(1)

    offs_t = t_block * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = offs_t < T

    x = tl.load(logits_ptr + h * T + offs_t, mask=mask_t, other=-float("inf"))
    x_max = tl.max(x, axis=0)
    x = x - x_max
    exp_x = tl.exp(x)
    sum_exp = tl.sum(exp_x, axis=0)
    attn = exp_x / sum_exp
    tl.store(out_ptr + h * T + offs_t, attn, mask=mask_t)


@triton.jit
def lse_row_kernel(
    logits_ptr,  # *f32 [H, T]
    out_ptr,     # *f32 [H]
    H: tl.constexpr,
    T: tl.constexpr,
    BLOCK_T: tl.constexpr
):
    # grid: (H,)
    h = tl.program_id(0)

    lse = -float("inf")
    for t_block in range(0, triton.cdiv(T, BLOCK_T)):
        offs_t = t_block * BLOCK_T + tl.arange(0, BLOCK_T)
        mask_t = offs_t < T
        x = tl.load(logits_ptr + h * T + offs_t, mask=mask_t, other=-float("inf"))
        x_max = tl.max(x, axis=0)
        sum_exp = tl.sum(tl.exp(x - x_max), axis=0)
        lse_candidate = tl.log(sum_exp) + x_max
        lse = tl.maximum(lse, lse_candidate)

    lse = lse / math.log(2.0)  # convert to base-2
    tl.store(out_ptr + h, lse)


@triton.jit
def gemv_row_kernel(
    attn_ptr,   # *f32 [H, T]
    Kc_ptr,     # *f32 [T, Dq]
    out_ptr,    # *f32 [H, Dq]
    H: tl.constexpr,    # number of heads
    Dq: tl.constexpr,   # head_dim_ckv = 512
    T: tl.constexpr,    # number of tokens
    BLOCK_T: tl.constexpr
):
    # grid: (H, ceil_div(Dq, BLOCK_D))
    h = tl.program_id(0)
    d_block = tl.program_id(1)

    offs_d = d_block * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_d = offs_d < Dq

    # Accumulator for out[h, offs_d]
    acc = tl.zeros((BLOCK_T,), dtype=tl.float32)

    # Loop over tokens in blocks
    for t_block in range(0, triton.cdiv(T, BLOCK_T)):
        offs_t = t_block * BLOCK_T + tl.arange(0, BLOCK_T)
        mask_t = offs_t < T
        attn_block = tl.load(attn_ptr + h * T + offs_t, mask=mask_t, other=0.0)  # [BLOCK_T]
        Kc_block = tl.load(Kc_ptr + (offs_t[:, None] * Dq + offs_d[None, :]), mask=mask_t[:, None] & mask_d[None, :], other=0.0)  # [BLOCK_T, BLOCK_D]
        # Multiply and reduce over tokens axis
        acc += tl.sum(Kc_block * attn_block[:, None], axis=0)

    tl.store(out_ptr + h * Dq + offs_d, acc, mask=mask_d)

            # Gather corresponding cache rows (squeeze dim=1 since there's only 1)
            Kc_b = ckv_cache[tok_idx].contiguous().to(torch.float32)  # [L_tokens, Dq]
            Kp_b = kpe_cache[tok_idx].contiguous().to(torch.float32)  # [L_tokens, Dp]

            qn = q_nope[b].contiguous().to(torch.float32)  # [H, Dq] but here H=16, we load per head inside kernel
            qp = q_pe[b].contiguous().to(torch.float32)    # [H, Dp]

            # Allocate intermediates
            logits = torch.empty((H, L_tokens), dtype=torch.float32, device=device)
            attn = torch.empty((H, L_tokens), dtype=torch.float32, device=device)

            # 1) Compute logits
            grid_logits = (H, triton.cdiv(L_tokens, BLOCK_T))
            fused_logits_kernel[grid_logits](

                H=H, Dq=Dq, Dp=Dp, T=L_tokens,

                num_warps=4, num_stages=2
            )

            # 2) Softmax over tokens per head

            grid_softmax = (H, triton.cdiv(L_tokens, BLOCK_T))

                logits_scaled, attn,
                H=H, T=L_tokens, BLOCK_T=BLOCK_T,
                num_warps=4, num_stages=2
            )

            # 3) Row-wise logsumexp divided by ln(2)
            grid_lse = (H,)
            lse_row_kernel[grid_lse](
                logits_scaled, lse[b],
                H=H, T=L_tokens, BLOCK_T=BLOCK_T,
                num_warps=4, num_stages=2
            )

            # 4) GEMV per head: out[h, :] = attn[h, :] @ Kc[:, :]
            grid_gemm = (H, triton.cdiv(Dq, BLOCK_T))

                attn, Kc_b, output[b],
                H=H, Dq=Dq, T=L_tokens, BLOCK_T=BLOCK_T,
                num_warps=4, num_stages=2
            )

        # Cast output to bfloat16 to match original, lse remains float32
        output = output.to(torch.bfloat16)
        return output, lse
