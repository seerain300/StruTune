import triton
import triton.language as tl

    q_nope_ptr,           # *bf16 [B, H, Dc]
    q_pe_ptr,             # *bf16 [B, H, Dp]
    ckv_cache_ptr,        # *bf16 [N, 1, Dc]
    kpe_cache_ptr,        # *bf16 [N, 1, Dp]
    kv_indptr_ptr,        # *int32 [B+1]
    kv_indices_ptr,       # *int32 [L]
    output_ptr,           # *bf16 [B, H, Dc]
    lse_ptr,              # *float32 [B*H]
    B: tl.constexpr,      # batch_size (unused but can be used for grid)
    H: tl.constexpr,      # num_qo_heads
    Dc: tl.constexpr,     # head_dim_ckv
    Dp: tl.constexpr,     # head_dim_kpe
    MAX_TOKENS: tl.constexpr,  # upper bound for tokens in segment
    SM_SCALE: tl.constexpr,     # scaling factor for logits (float)
):
    # One program per batch element
    b = tl.program_id(0)

    # Load token range for this batch element
    base = tl.load(kv_indptr_ptr + b)         # int32
    end = tl.load(kv_indptr_ptr + b + 1)      # int32
    L_tokens = end - base                      # number of tokens for this batch element

    # If no tokens, store zeros for output and -inf for lse
    if L_tokens <= 0:
        # We'll fill output with zeros after loop; for now, compute lse as -inf
        lse_val = -float('inf')
        tl.store(lse_ptr + b * H, lse_val)
        return

    # Loop over heads
    for h in range(H):
        # Load qn (bf16) and qp (bf16) for this batch and head, cast to f32 for math
        # q_nope: shape [B, H, Dc], q_pe: shape [B, H, Dp]
        qn_bf = tl.load(q_nope_ptr + b * H * Dc + h * Dc)            # *bf16 [Dc]
        qp_bf = tl.load(q_pe_ptr + b * H * Dp + h * Dp)              # *bf16 [Dp]
        qn = qn_bf.to(tl.float32)                                    # [Dc] float32
        qp = qp_bf.to(tl.float32)                                    # [Dp] float32

        # Running max for stable logsumexp and accumulation of sum of exp
        acc = -float('inf')  # scalar float32
        sum_exp = 0.0        # scalar float32

        # Compute scaled logits and accumulate sum of exp with max
        for i in range(MAX_TOKENS):
            if i < L_tokens:
                idx = tl.load(kv_indices_ptr + base + i)             # int32 token index
                # Load Kc and Kp rows as bf16 then cast to f32

                Kc_row = Kc_row_bf.to(tl.float32)
                Kp_row = Kp_row_bf.to(tl.float32)

                for d in range(Dc):
                    dot1 += qn[d] * Kc_row[d]
                dot2 = 0.0
                for d in range(Dp):
                    dot2 += qp[d] * Kp_row[d]

                val = (dot1 + dot2) * SM_SCALE
                acc = tl.maximum(acc, val)
                # Accumulate sum_exp for positions i < L_tokens using the current max
                sum_exp += tl.exp(val - acc)

        # Compute lse = log(sum_exp) + acc, scaled by 1/ln(2)
        lse_val = tl.log(sum_exp) + acc
        lse_val = lse_val / tl.log(2.0)
        tl.store(lse_ptr + b * H + h, lse_val)

        # Compute final output vector: out[b, h, :] = sum_i attn_i * Kc_all[i, :]

        for i in range(MAX_TOKENS):

                Kp_row_bf = tl.load(kpe_cache_ptr + idx * Dp + tl.arange(0, Dp))  # *bf16
                Kp_row = Kp_row_bf.to(tl.float32)

                for d in range(Dc):
                    dot1 += qn[d] * Kc_row[d]
                dot2 = 0.0

                    dot2 += qp[d] * Kp_row[d]

                val = (dot1 + dot2) * SM_SCALE
                attn_i = tl.exp(val - lse_val)

                out_vec += attn_i * Kc_row

        # Store output for head h as bfloat16
        out_offset = b * H * Dc + h * Dc
        tl.store(output_ptr + out_offset + tl.arange(0, Dc), out_vec.to(tl.bfloat16))
