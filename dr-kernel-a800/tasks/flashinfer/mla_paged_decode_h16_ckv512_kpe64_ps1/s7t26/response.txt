```python
import math
import torch

# Triton kernels: all computation inside Triton. No torch ops in forward.

# matvec_row: computes out = v @ B, where v is [M] and B is [M, N].
# We launch one program and set BLOCK_N to cover the entire N dimension.
@triton.jit
def matvec_row(v_ptr, B_ptr, out_ptr,
                M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
                BLOCK_N: tl.constexpr):
    n_offsets = tl.arange(0, BLOCK_N)
    acc = tl.zeros([BLOCK_N], dtype=tl.float32)
    # Loop over K dimension (tokens)
    for k in range(0, K):
        v_k = tl.load(v_ptr + k)  # v_ptr is [M], contiguous
        # B_ptr is [M, N] contiguous with row stride N
        b_col = tl.load(B_ptr + k * N + n_offsets, mask=n_offsets < N, other=0.0)
        acc += v_k * b_col
    tl.store(out_ptr + n_offsets, acc, mask=n_offsets < N)

# softmax_base2_kernel: computes softmax in base-2 for a 1D vector logits of length L.
# It writes probabilities (dim -1) into out_probs and also writes the scalar lse (base-2)
# into out_lse_ptr.
@triton.jit
def softmax_base2_kernel(logits_ptr, out_probs_ptr, out_lse_ptr,
                          L: tl.constexpr, scale: tl.constexpr):
    # Compute max for numerical stability
    max_val = -float("inf")
    for i in range(0, L):
        val = tl.load(logits_ptr + i)
        max_val = tl.maximum(max_val, val)

    sum_exp = 0.0
    for i in range(0, L):
        val = tl.load(logits_ptr + i)
        val = val - max_val
        exp_val = tl.exp(val * scale)
        sum_exp += exp_val
        # store probabilities
        tl.store(out_probs_ptr + i, exp_val)

    # lse = log(sum_exp) / log(2.0)
    # scale is 1.4426950408889634 (1 / ln(2))
    lse = tl.log(sum_exp) * scale
    tl.store(out_lse_ptr, lse)

# matmul_small: computes C[M, N] = A[M, K] @ B[K, N], writing into C_ptr.
@triton.jit
def matmul_small(A_ptr, B_ptr, C_ptr,
                  M: tl.constexpr, N: tl.constexpr, K: tl.constexpr):
    # One program computes the entire C (M,N), iterating K in tiles.
    # For generality, we support M, N, K as constexpr.
    # Note: Here we assume M=1 as in attention_probs[1, K] @ Kc[1, N].
    acc = tl.zeros([M, N], dtype=tl.float32)
    # Loop over K in blocks
    for k0 in range(0, K, 32):  # 32 is a reasonable tile size for small K
        for i in range(0, M):
            for j in range(0, N):
                s = 0.0
                for kk in range(0, 32):
                    k = k0 + kk
                    a = tl.load(A_ptr + i * K + k) if k < K else 0.0
                    b = tl.load(B_ptr + k * N + j) if k < K else 0.0
                    s += a * b
                acc[i, j] = s
    tl.store(C_ptr + i * N + j, acc[i, j], mask=True)  # Since M=1, we can store directly.
```
