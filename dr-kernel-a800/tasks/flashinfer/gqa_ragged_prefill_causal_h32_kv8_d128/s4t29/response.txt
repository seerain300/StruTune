- forward loop uses qo_indptr and kv_indptr of length len_indptr (typically 1 in the provided get_inputs). It processes each batch element b with slices:
  - q slice: q[q_start:q_end], where q_start = qo_indptr[b], q_end = qo_indptr[b+1]
  - kv slice: k[v_start:kv_end], v[v_start:kv_end], where kv_start = kv_indptr[b], kv_end = kv_indptr[b+1]
- For each q_token in that batch, it computes attention logits over kv positions: logits[q_token, qo_head, kv_pos] = (q[q_token, qo_head] · K_exp[kv_pos]) * sm_scale, where K_exp is k expanded with repeat_interleave(gqa_ratio, dim=1). In code, the mask used is kv_positions < q_positions + 1 + delta, but with delta = kv_end - kv_start and q_start=0, q_end=num_q_tokens, q_positions=q_token, this simplifies to kv_positions < q_token + 1 + (kv_end - kv_start - kv_start). Given the loop and masks, the effective constraint is j <= q_token.
- lse is logsumexp(logits, dim=-1) per (q_token, qo_head), then divided by log(2).
- output per (q_token, qo_head) is softmax(logits) · V_exp where V_exp is v expanded similarly.

# Kernel 1: compute lse for each (b, q_token, qo_head)
@triton.jit
def compute_lse_kernel(
    q_ptr, k_exp_ptr, lse_ptr,
    total_q: tl.int32, num_qo_heads: tl.int32, head_dim: tl.int32,
    sm_scale: tl.float32, out_len: tl.int32, gqa_ratio: tl.int32
):
    b = tl.program_id(0)
    q_token = tl.program_id(1)
    qo_head = tl.program_id(2)

    # If out of bounds, return (defensive; grid should ensure in-range)
    if (b >= 1 or q_token >= total_q or qo_head >= num_qo_heads):
        return

    # Load Q vector for this (q_token, qo_head)
    q_base = (q_token * num_qo_heads + qo_head) * head_dim
    q_vec = [tl.load(q_ptr + q_base + d) for d in range(head_dim)]

    # Initialize lse with -inf, then update via stable logsumexp in two passes
    lse_val = -float('inf')
    # First pass: max over logits
    max_val = -float('inf')
    for kv_pos in tl.static_range(0, out_len):
        j = kv_pos // gqa_ratio
        r = kv_pos % gqa_ratio
        # Load K_exp vector at position kv_pos, head qo_head
        k_base = kv_pos * head_dim + qo_head * head_dim
        k_vec = [tl.load(k_exp_ptr + k_base + d) for d in range(head_dim)]

        # Compute dot(Q, K_exp) * sm_scale
        dot = 0.0
        for d in range(0, head_dim):
            dot += q_vec[d] * k_vec[d]
        dot = dot * sm_scale

        # Apply causal mask: only if j <= q_token
        if j <= q_token:
            if dot > max_val:
                max_val = dot

    # Second pass: compute sum(exp(dot - max_val)) over valid kv positions
    sum_exp = 0.0
    for kv_pos in tl.static_range(0, out_len):
        j = kv_pos // gqa_ratio
        r = kv_pos % gqa_ratio
        k_base = kv_pos * head_dim + qo_head * head_dim
        k_vec = [tl.load(k_exp_ptr + k_base + d) for d in range(head_dim)]

        dot = 0.0
        for d in range(0, head_dim):
            dot += q_vec[d] * k_vec[d]
        dot = dot * sm_scale

        if j <= q_token:
            sum_exp += tl.exp(dot - max_val)

    # Compute lse = max + log(sum) / log(2)
    lse_val = max_val + tl.log(sum_exp) / math.log(2.0)

    # Store lse
    lse_offset = b * (total_q * num_qo_heads) + q_token * num_qo_heads + qo_head
    tl.store(lse_ptr + lse_offset, lse_val)

def compute_output_kernel(
    q_ptr, k_ptr, v_ptr, qo_indptr_ptr, kv_indptr_ptr, output_ptr,
    total_q: tl.int32, num_qo_heads: tl.int32, num_kv_heads: tl.int32, head_dim: tl.int32,
    sm_scale: tl.float32, len_indptr: tl.int32, gqa_ratio: tl.int32
):
    # Grid: (len_indptr, total_q, num_qo_heads)
    b = tl.program_id(0)
    q_token = tl.program_id(1)
    qo_head = tl.program_id(2)

    # Bounds check
    if b >= len_indptr or q_token >= total_q or qo_head >= num_qo_heads:
        return

    # Load qo_indptr[b] and kv_indptr[b] (int32)
    qo_start = tl.load(qo_indptr_ptr + b)
    kv_start = tl.load(kv_indptr_ptr + b)

    # Compute q_end and kv_end
    q_end = tl.load(qo_indptr_ptr + b + 1) if b + 1 < len_indptr else total_q + 1
    kv_end = tl.load(kv_indptr_ptr + b + 1) if b + 1 < len_indptr else total_kv + 1

    # If no queries or no KV in this batch slice, skip
    if q_token >= q_end or kv_start >= kv_end:
        return

    # Prepare output vector [head_dim] for this (b, q_token, qo_head)
    out_vec = tl.zeros([head_dim], dtype=tl.float32)
    offs = tl.arange(0, head_dim)

    # Load Q vector for this (q_token, qo_head)
    q_base = (q_token * num_qo_heads + qo_head) * head_dim
    q_vec = tl.load(q_ptr + q_base + offs)  # [head_dim]

    # Total expanded KV positions
    out_len = (kv_end - kv_start) * gqa_ratio

    # Compute normalization sum and accumulate output
    sum_exp = 0.0
    for kv_pos in tl.static_range(0, out_len):
        j = kv_pos // gqa_ratio
        r = kv_pos % gqa_ratio

        # Effective kv index in original k/v: kv_idx = j + kv_start
        kv_idx = kv_start + j

        # Load K_exp and V_exp vectors for this expanded position and qo_head
        k_base = kv_idx * (num_kv_heads * head_dim) + j * (head_dim) + qo_head * head_dim

        v_base = kv_idx * (num_kv_heads * head_dim) + j * (head_dim) + qo_head * head_dim
        v_vec = tl.load(v_ptr + v_base + offs)  # [head_dim]

        # Compute dot(Q, K_exp) * sm_scale
        dot = tl.sum(q_vec * k_vec, axis=0) * sm_scale

        # Apply causal mask: only if j <= q_token
        if j <= q_token:

            sum_exp += attn
            out_vec += attn * v_vec

    # Now write output: [b, q_token, qo_head, :]
    out_offset = b * (total_q * num_qo_heads * head_dim) + q_token * (num_qo_heads * head_dim) + qo_head * head_dim
    tl.store(output_ptr + out_offset + offs, out_vec)
