                        q_base = q_ptr + q_row * G * D + g * D
                        d = tl.arange(0, D)
                        Q_vec = tl.load(q_base + d, mask=valid_mm, other=0.0)
                        for nn_sub in range(0, BLOCK_N):
                            valid_nn = mask_nn[nn_sub]
                            k_base = k_ptr + (kv_start + (n0 + nn_sub)) * GH * D
                            score = tl.zeros((), dtype=tl.float32)
                            for hh in range(0, GH):
                                k_vec = tl.load(k_base + hh * D + d, mask=valid_nn, other=0.0)
                                score += tl.sum(Q_vec * k_vec, axis=0)
                            score *= SM_SCALE
                            causal = valid_nn and (n0 + nn_sub < (q_start + m0 + mm_sub + 1 + (N - M)))
                            logits_tile[nn_sub] = score if causal else -float("inf")

                    # Softmax over logits_tile
                    logits_tile = tl.maximum(logits_tile, -1e20)
                    max_logit = tl.max(logits_tile, axis=0)
                    logits_tile = logits_tile - max_logit
                    exp_logits = tl.exp(logits_tile)
                    denom = tl.sum(exp_logits, axis=0)
                    attn = exp_logits / denom  # [BLOCK_N]

                    # Accumulate output out_row[mm_sub, :] += attn * V
                    for mm_sub in range(0, BLOCK_M):
                        valid_mm = mask_mm[mm_sub]
                        if valid_mm:
                            q_row = q_start + m0 + mm_sub
                            v_base = v_ptr + (kv_start + (n0 + nn + 0)) * GH * D  # placeholder; use each nn_sub below
                            # For each nn_sub, multiply attn[nn_sub] * V_vec and add to out_row[mm_sub, :]
                            for nn_sub in range(0, BLOCK_N):
                                valid_nn = mask_nn[nn_sub]
                                v_base = v_ptr + (kv_start + (n0 + nn_sub)) * GH * D
                                # Compute V_vec for this nn_sub
                                V_vec = tl.zeros((D,), dtype=tl.float32)
                                for hh in range(0, GH):
                                    V_vec += tl.load(v_base + hh * D + d, mask=valid_nn, other=0.0)
                                out_row[mm_sub, :] += attn[nn_sub] * V_vec

            # Store out_row to out_ptr at (row=mm_sub, head=g)
            for mm_sub in range(0, BLOCK_M):
                valid_mm = mask_mm[mm_sub]
                if valid_mm:
                    q_row = q_start + m0 + mm_sub
                    out_base = out_ptr + q_row * G * D + g * D
                    # out_row is [D] for this mm_sub
                    # We need to store as [D] at out_base; but out_ptr is laid out as [M, G, D]. So store contiguously:
                    # The pointer arithmetic above ensures correctness.
                    # Triton requires explicit stores; we can store by iterating over d.
                    for dd in range(0, D):
                        tl.store(out_base + dd, out_row[mm_sub, dd])

    def forward(self, q, k, v, qo_indptr, kv_indptr, sm_scale):
        # If Triton isn't available or not on CUDA, fallback to original PyTorch implementation.
        if (not TRITON_AVAILABLE) or (q.device.type != "cuda"):
            total_q, num_qo_heads, head_dim = q.shape
            total_kv, num_kv_heads, _ = k.shape
            len_indptr = qo_indptr.shape[0]
            assert num_qo_heads == 32
            assert num_kv_heads == 8
            assert head_dim == 128
            assert total_q == int(qo_indptr[-1].item())
            assert total_kv == int(kv_indptr[-1].item())
            output = torch.zeros((total_q, num_qo_heads, head_dim), dtype=torch.bfloat16, device=q.device)
            lse = torch.full((total_q, num_qo_heads), -float("inf"), dtype=torch.float32, device=q.device)
            q_f32 = q.to(torch.float32)
            k_f32 = k.to(torch.float32)
            v_f32 = v.to(torch.float32)
            gqa_ratio = num_qo_heads // num_kv_heads
            for b in range(len_indptr - 1):
                q_start = int(qo_indptr[b].item())
                q_end = int(qo_indptr[b + 1].item())
                kv_start = int(kv_indptr[b].item())
                kv_end = int(kv_indptr[b + 1].item())
                if q_start >= q_end or kv_start >= kv_end:
                    continue
                q_batch = q_f32[q_start:q_end]
                k_batch = k_f32[kv_start:kv_end]
                v_batch = v_f32[kv_start:kv_end]
                num_q_tokens = q_batch.shape[0]
                num_kv_tokens = k_batch.shape[0]
                delta = num_kv_tokens - num_q_tokens
                k_expanded = k_batch.repeat_interleave(gqa_ratio, dim=1)
                v_expanded = v_batch.repeat_interleave(gqa_ratio, dim=1)
                logits = torch.einsum('qhd,khd->qhk', q_batch, k_expanded) * sm_scale
                q_positions = torch.arange(num_q_tokens, device=q.device)
                kv_positions = torch.arange(num_kv_tokens, device=q.device)
                causal_mask = kv_positions[None, :] < (q_positions[:, None] + 1 + delta)
                logits = logits.masked_fill(~causal_mask[:, None, :], float('-inf'))
                lse_batch = torch.logsumexp(logits, dim=-1) / math.log(2.0)
                lse[q_start:q_end] = lse_batch
                attn_weights = torch.softmax(logits, dim=-1)
                output_batch = torch.einsum('qhk,khd->qhd', attn_weights, v_expanded)
                output[q_start:q_end] = output_batch.to(torch.bfloat16)
            return output, lse

        device = q.device
        q_f32 = q.to(torch.float32).contiguous()
        k_f32 = k.to(torch.float32).contiguous()
        v_f32 = v.to(torch.float32).contiguous()

        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)
