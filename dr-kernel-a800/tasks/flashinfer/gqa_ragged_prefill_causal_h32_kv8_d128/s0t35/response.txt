```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def attention_gqa_single_kernel(
    q_ptr, k_ptr, v_ptr, out_ptr, lse_ptr,
    qo_indptr_ptr, kv_indptr_ptr,
    total_q, total_kv, sm_scale,
    NUM_SEGMENTS: tl.constexpr,  # expected to be 1 for these workloads
):
    """
    Triton kernel: one program per (i, h). Handles all segments by looping over NUM_SEGMENTS.
    q_ptr: float32 [total_q, 32, 128]
    k_ptr: float32 [total_kv, 8, 128]
    v_ptr: float32 [total_kv, 8, 128]
    out_ptr: float32 [total_q, 32, 128], initialized to zeros
    lse_ptr: float32 [total_q, 32], initialized to -inf
    qo_indptr_ptr: int32 [NUM_SEGMENTS+1]
    kv_indptr_ptr: int32 [NUM_SEGMENTS+1]
    """
    pid = tl.program_id(axis=0)
    i = pid // 32
    h = pid % 32

    # Process each segment
    for b in range(NUM_SEGMENTS):
        q_start = tl.load(qo_indptr_ptr + b)  # int32
        q_end = tl.load(qo_indptr_ptr + b + 1)  # int32
        kv_start = tl.load(kv_indptr_ptr + b)  # int32
        kv_end = tl.load(kv_indptr_ptr + b + 1)  # int32

        Nq = q_end - q_start
        Nk = kv_end - kv_start

        if Nq <= 0 or Nk <= 0:
            continue

        delta = Nk - Nq

        # Compute logits across 8 kv heads (per (i, h))
        logits = tl.zeros((8,), dtype=tl.float32)

        # Load q[i, h, :] for this segment (q indexing is absolute across segments, not per-b; but qo_indptr b gives range)
        q_base = (q_start + i) * 32 * 128 + h * 128
        q_vec = tl.load(q_ptr + q_base)  # [128] float32

        # For each kv head j in 0..7, compute dot product over Nk rows of k and apply causal mask
        for j in range(8):
            orig_h = h % 8  # map to original 8 kv heads
            # Accumulate dot over k rows: k_ptr[kv_start + t, orig_h, :]
            acc = 0.0
            t = 0
            while t < Nk:
                k_row_ptr = (kv_start + t) * 8 * 128 + orig_h * 128
                k_vec = tl.load(k_ptr + k_row_ptr)  # [128]
                acc += tl.sum(q_vec * k_vec)  # scalar
                t += 1
            acc *= sm_scale
            # Apply forward-causal mask: allow if j < (i + 1 + delta); else set -inf
            # For safety, we set logits[j] to acc if allowed, else -1e20; softmax will zero it.
            allow = j < (i + 1 + delta)
            logits[j] = tl.where(allow, acc, -1e20)

        # Compute base-2 logsumexp
        m = tl.max(logits, axis=0)
        sum_exp = tl.sum(tl.exp(logits - m), axis=0)
        lse_val = m + tl.log(sum_exp) / 0.6931471805599453  # ln(2)

        # Softmax across 8 positions
        soft = tl.exp(logits - lse_val)  # [8] float32

        # Accumulate output[i, h, :] = sum_j soft[j] * v_exp[:, j, :]
        out_base = (q_start + i) * 32 * 128 + h * 128
        # We will write by iterating j and adding
        for j in range(8):
            orig_h = h % 8
            v_base = (kv_start + j) * 32 * 128 + orig_h * 128
            v_vec = tl.load(v_ptr + v_base)  # [128]
            out_vec = out_ptr + out_base
            # out_ptr is float32, we can add directly
            # Note: Triton supports scalar addition to a pointer vector
            # We need to add v_vec * soft[j] to the 128 elements of out_ptr starting at out_base
            # Triton doesn't support elementwise vector update from scalar; we do it element by element.
            # Instead, we can load current out_vec, add, and store back. But Triton pointer arithmetic expects contiguous ranges.
            # Better: compute per-lane and store back. We'll reconstruct lanes using arange.
            offs = tl.arange(0, 128)
            current = tl.load(out_vec + offs)  # load current output
            current += v_vec * soft[j]
            tl.store(out_vec + offs, current)

        # Store lse[i, h]
        lse_index = (q_start + i) * 32 + h
        tl.store(lse_ptr + lse_index, lse_val)

    # Kernel finishes; out_ptr is updated in-place across segments. Nothing to return, forward returns PyTorch tensors.


class ModelNew(torch.nn.Module):
    def forward(self, q, k, v, qo_indptr, kv_indptr, sm_scale):
        # Ensure device and dtype compatibility
        assert q.is_cuda and k.is_cuda and v.is_cuda, "Inputs must be on CUDA for Triton."
        device = q.device
        num_qo_heads = 32
        num_kv_heads = 8
        head_dim = 128

        # Cast to float32 for computation; original code uses float32 internally
        q_f32 = q.to(torch.float32)
        k_f32 = k.to(torch.float32)
        v_f32 = v.to(torch.float32)

        # Expand k and v to 32 heads (GQA ratio = 4)
        k_expanded = k_f32.repeat_interleave(4, dim=1)  # [total_kv, 32, 128]
        v_expanded = v_f32.repeat_interleave(4, dim=1)  # [total_kv, 32, 128]

        # Output and LSE buffers
        total_q = q_f32.shape[0]
        total_kv = k_expanded.shape[0]
        output = torch.zeros((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=device)
        lse = torch.full((total_q, num_qo_heads), -float("inf"), dtype=torch.float32, device=device)

        # len_indptr is fixed at 2 in the evaluator, so NUM_SEGMENTS = 1
        NUM_SEGMENTS = qo_indptr.numel() - 1  # expect 1; guard for safety

        # Launch Triton kernel: one program per (i, h)
        grid = (total_q * num_qo_heads,)
        attention_gqa_single_kernel[grid](
            q_f32, k_expanded, v_expanded, output, lse,
            qo_indptr, kv_indptr,
            total_q, total_kv, sm_scale,
            NUM_SEGMENTS=NUM_SEGMENTS,
            num_warps=1,  # small kernels; can tune
        )

        # Return in original expected dtypes
        output_bf16 = output.to(torch.bfloat16)
        return output_bf16, lse
```

    total_q, total_kv, sm_scale,
    NUM_SEGMENTS: tl.constexpr,
):

    Triton kernel: one program per (i, h). Processes all segments in NUM_SEGMENTS.
    q_ptr: float32 [total_q, 32, 128]
    k_ptr: float32 [total_kv, 8, 128]
    v_ptr: float32 [total_kv, 8, 128]
    out_ptr: float32 [total_q, 32, 128], will be accumulated across segments
    lse_ptr: float32 [total_q, 32], will be set per segment
    qo_indptr_ptr: int32 [NUM_SEGMENTS+1]
    kv_indptr_ptr: int32 [NUM_SEGMENTS+1]
    """

    h = pid % 32

    for b in range(NUM_SEGMENTS):
