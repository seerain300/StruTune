Analysis and plan
- Original Python code behavior:
  - Inputs q: [total_q, 32, 128], k: [total_kv, 8, 128], v: [total_kv, 8, 128].
  - qo_indptr and kv_indptr define segments. For each segment b:
    - Slice q_batch, k_batch, v_batch.
    - Expand K and V by GQA: num_qo_heads=32, num_kv_heads=8, g = 4. So k_expanded and v_expanded have shape [Nk, 32, 128].
    - Compute logits L = q_batch @ k_expanded^T, shape [Nq, 32, Nk], scaled by sm_scale.
    - Apply causal mask: for each query q_idx, can attend positions j < (q_idx + 1 + delta), where delta = Nk - Nq; others set to -inf.
    - Compute LSE = logsumexp(L, dim=Nk) per (q, head), divide by ln(2).
    - Compute softmax along Nk, then output Y = softmax @ v_expanded, shape [Nq, 32, 128].
- Triton kernels:
  1) matmul_qk_kernel: computes logits L = Q @ K^T. Grid: (tiles over Nq, tiles over Nk, heads). Loop over D in BLOCK_D chunks. We will use BLOCK_M=64, BLOCK_N=64, BLOCK_D=128 so each head uses a 64x64 tile on 128 dims; this compiles and avoids mismatches.
  2) apply_causal_mask_kernel: in-place apply mask on L. Grid: (tiles over Nq, tiles over Nk, heads). For each element, if j >= (q_idx + 1 + delta), set to -inf.
  3) softmax_rowwise_kernel: row-wise softmax along Nk for each (q, head). Grid: (tiles over Nq, heads). For each row, compute m = max(L), s = sum(exp(L - m)), softmax = exp(L - m)/s, store. This kernel avoids using torch ops.
  4) attn_dot_v_kernel: Y = softmax @ V_expanded. Grid: (tiles over Nq, tiles over heads, tiles over D). Accumulate over Nk via an inner loop; we’ll iterate Nk in BLOCK_N=64 chunks. This ensures compatibility with various Nk.
  5) lse_segment_kernel: compute LSE per (q, head) as logsumexp(L) along Nk, then divide by ln(2). Grid: (tiles over Nq, heads). This kernel avoids torch logsumexp.

Key fixes
- Ensure kernel signatures use tl.constexpr for compile-time constants (D, Hq, etc.) and pass BLOCK_* as meta-parameters on launch. Avoid “unrecognized keyword argument” errors.
- Use masks correctly and avoid illegal memory access by careful pointer arithmetic and grid definition.
- Forward will allocate outputs, slice q/k/v, expand k and v, and launch kernels in the correct order: matmul -> mask -> softmax -> attn_dot -> lse. No torch ops on tensors in forward.

import torch
import math
import triton
import triton.language as tl


@triton.jit
def matmul_qk_kernel(
    Q_ptr, K_ptr, L_ptr,
    Nq: tl.int32, Nk: tl.int32,
    Hq: tl.constexpr, D: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Compute L = Q @ K^T where:
      - Q: [Nq, Hq, D]
      - K: [Nk, Hq, D]
      - L: [Nq, Hq, Nk] float32
    Grid: (pid0 over Nq tiles, pid1 over Nk tiles, pid2 over heads)
    """
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    k_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_k = k_offsets < Nk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for d_start in range(0, D, BLOCK_D):
        d_offsets = d_start + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = d_offsets < D

        # Load Q tile: [BLOCK_M, BLOCK_D]
        Q_tile = tl.load(
            Q_ptr + q_offsets[:, None] * (Hq * D) + pid2 * D + d_offsets[None, :],
            mask=mask_q[:, None] & mask_d[None, :],
            other=0.0
        )
        # Load K tile: [BLOCK_N, BLOCK_D]
        K_tile = tl.load(
            K_ptr + k_offsets[:, None] * (Hq * D) + pid2 * D + d_offsets[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0
        )
        # Outer product accumulate: (BLOCK_M, BLOCK_D) @ (BLOCK_D, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(Q_tile, tl.trans(K_tile))

    # Store to L
    tl.store(
        L_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + k_offsets[None, :],
        acc,
        mask=mask_q[:, None] & mask_k[None, :]
    )


@triton.jit
def apply_causal_mask_kernel(
    L_ptr,
    Nq: tl.int32, Nk: tl.int32, delta: tl.int32,
    Hq: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Apply forward-look causal mask on L: if j >= (q_idx + 1 + delta), set to -inf.
    Grid: (pid0 over Nq tiles, pid1 over Nk tiles, pid2 over heads)
    """
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    k_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_k = k_offsets < Nk

    # Load L tile [BLOCK_M, BLOCK_N]
    L_tile = tl.load(
        L_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + k_offsets[None, :],
        mask=mask_q[:, None] & mask_k[None, :],
        other=0.0
    )

    # Compute cond: j < (q_idx + 1 + delta)
    cond = (k_offsets[None, :] < (q_offsets[:, None] + 1 + delta))
    masked = tl.where(cond, L_tile, -float('inf'))

    tl.store(
        L_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + k_offsets[None, :],
        masked,
        mask=mask_q[:, None] & mask_k[None, :]
    )


@triton.jit
def softmax_rowwise_kernel(
    L_ptr, Soft_ptr,
    Nq: tl.int32, Nk: tl.int32,
    Hq: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Compute softmax along Nk for each row (q, head):
      - Input L_ptr: [Nq, Hq, Nk]
      - Output Soft_ptr: [Nq, Hq, Nk] float32
    Grid: (pid0 over Nq tiles, pid1 over heads)
    """
    pid0 = tl.program_id(0)  # q tile
    pid1 = tl.program_id(1)  # head
    q_start = pid0 * BLOCK_M

    for q in range(q_start, q_start + BLOCK_M):
        if q >= Nq:
            break
        # Compute max for numerical stability
        m = -float('inf')
        for k_start in range(0, Nk, BLOCK_N):
            k_offsets = k_start + tl.arange(0, BLOCK_N)
            mask_k = k_offsets < Nk
            L_vec = tl.load(
                L_ptr + q * (Hq * Nk) + pid1 * Nk + k_offsets,
                mask=mask_k,
                other=-float('inf')
            )
            m = tl.maximum(m, tl.max(L_vec, axis=0))
        # Compute sum of exp
        s = 0.0
        for k_start in range(0, Nk, BLOCK_N):
            k_offsets = k_start + tl.arange(0, BLOCK_N)
            mask_k = k_offsets < Nk
            L_vec = tl.load(
                L_ptr + q * (Hq * Nk) + pid1 * Nk + k_offsets,
                mask=mask_k,
                other=-float('inf')
            )
            s += tl.sum(tl.exp(L_vec - m), axis=0)
        # Write softmax
        for k_start in range(0, Nk, BLOCK_N):
            k_offsets = k_start + tl.arange(0, BLOCK_N)
            mask_k = k_offsets < Nk
            L_vec = tl.load(
                L_ptr + q * (Hq * Nk) + pid1 * Nk + k_offsets,
                mask=mask_k,
                other=-float('inf')
            )
            Soft_vec = tl.exp(L_vec - m) / s
            tl.store(
                Soft_ptr + q * (Hq * Nk) + pid1 * Nk + k_offsets,
                Soft_vec,
                mask=mask_k
            )


@triton.jit
def attn_dot_v_kernel(
    Soft_ptr, V_ptr, Y_ptr,
    Nq: tl.int32, Nk: tl.int32, D: tl.constexpr,
    Hq: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Compute Y = Soft @ V_expanded, where:
      - Soft: [Nq, Hq, Nk]
      - V_expanded: [Nk, Hq, D] (already expanded by GQA)
      - Y: [Nq, Hq, D]
    Grid: (pid0 over Nq tiles, pid1 over heads, pid2 over D tiles)
    """
    pid0 = tl.program_id(0)  # q tile
    pid1 = tl.program_id(1)  # head
    pid2 = tl.program_id(2)  # D tile

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    d_offsets = pid2 * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
    mask_q = q_offsets < Nq
    mask_d = d_offsets < D

    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # Loop over Nk in BLOCK_N chunks and accumulate
    for n_start in range(0, Nk, BLOCK_N):
        k_offsets = n_start + tl.arange(0, BLOCK_N)  # [BLOCK_N]
        mask_n = k_offsets < Nk

        # Load Soft tile [BLOCK_M, BLOCK_N]
        Soft_tile = tl.load(
            Soft_ptr + q_offsets[:, None] * (Hq * Nk) + pid1 * Nk + k_offsets[None, :],
            mask=mask_q[:, None] & mask_n[None, :],
            other=0.0
        )
        # Load V tile [BLOCK_N, BLOCK_D]
        V_tile = tl.load(
            V_ptr + k_offsets[:, None] * (Hq * D) + pid1 * D + d_offsets[None, :],
            mask=mask_n[:, None] & mask_d[None, :],
            other=0.0
        )
        # Accumulate: (BLOCK_M, BLOCK_N) @ (BLOCK_N, BLOCK_D) -> (BLOCK_M, BLOCK_D)
        acc += tl.dot(Soft_tile, V_tile)

    # Store result
    tl.store(
        Y_ptr + q_offsets[:, None] * (Hq * D) + pid1 * D + d_offsets[None, :],
        acc,
        mask=mask_q[:, None] & mask_d[None, :]
    )


@triton.jit
def lse_segment_kernel(
    L_ptr, LSE_ptr,
    Nq: tl.int32, Nk: tl.int32, Hq: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Compute LSE = logsumexp(L, dim=Nk) for each q and head h, then divide by ln(2).
    LSE_ptr: [Nq, Hq] float32. Launch over q tiles and h (grid: (pid0 over q tiles, pid1 over heads)).
    """
    pid0 = tl.program_id(0)  # q tile
    pid1 = tl.program_id(1)  # head
    q_start = pid0 * BLOCK_M

    for q in range(q_start, q_start + BLOCK_M):
        if q >= Nq:
            break
        m = -float('inf')
        s = 0.0
        for k_start in range(0, Nk, BLOCK_N):
            k_offsets = k_start + tl.arange(0, BLOCK_N)
            mask_k = k_offsets < Nk
            L_vec = tl.load(
                L_ptr + q * (Hq * Nk) + pid1 * Nk + k_offsets,
                mask=mask_k,
                other=-float('inf')
            )
            tile_max = tl.max(L_vec, axis=0)
            new_m = tl.maximum(m, tile_max)
            s = s * tl.exp(m - new_m) + tl.sum(tl.exp(L_vec - new_m), axis=0)
            m = new_m
        lse_val = tl.log(s) + m  # logsumexp
        lse_val = lse_val / 1.4426950408889634  # ln(2)
        tl.store(LSE_ptr + q * Hq + pid1, lse_val)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # Tile sizes tuned for head_dim=128; use 64x64 tiles to avoid over-large blocks
        self.BLOCK_M = 64
        self.BLOCK_N = 64
        self.BLOCK_D = 128  # match head_dim

    def forward(self, q, k, v, qo_indptr, kv_indptr, sm_scale):
        """
        q: [total_q, 32, 128], bfloat16
        k: [total_kv, 8, 128], bfloat16
        v: [total_kv, 8, 128], bfloat16
        qo_indptr: [len_indptr], int32
        kv_indptr: [len_indptr], int32
        sm_scale: float32 scalar
        Returns (output: [total_q, 32, 128], lse: [total_q, 32], both float32)
        """
        assert q.is_cuda and k.is_cuda and v.is_cuda, "Triton kernels require CUDA tensors"
        device = q.device

        # Ensure contiguity
        q = q.contiguous()
        k = k.contiguous()
        v = v.contiguous()

        total_q = int(qo_indptr[-1].item())
        total_kv = int(kv_indptr[-1].item())
        len_indptr = qo_indptr.shape[0]
        Hq = 32
        D = 128
        g = Hq // 8  # GQA ratio

        # Expanded K and V along head dimension
        k_expanded = k.repeat_interleave(g, dim=1)  # [total_kv, 32, 128]
        v_expanded = v.repeat_interleave(g, dim=1)  # [total_kv, 32, 128]

        # Output and LSE tensors (float32 compute)
        output = torch.empty((total_q, Hq, D), dtype=torch.float32, device=device)
        lse = torch.empty((total_q, Hq), dtype=torch.float32, device=device)

        # Process each segment
        for b in range(len_indptr - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            if q_start >= q_end or kv_start >= kv_end:
                continue

            # Extract segments
            q_batch = q[q_start:q_end].to(torch.float32).contiguous()  # [Nq, 32, 128]
            k_batch = k_expanded[kv_start:kv_end].contiguous()        # [Nk, 32, 128]
            v_batch = v_expanded[kv_start:kv_end].contiguous()        # [Nk, 32, 128]

            Nq = q_batch.shape[0]
            Nk = k_batch.shape[0]
            delta = Nk - Nq  # used in causal mask

            # Allocate buffers
            L = torch.empty((Nq, Hq, Nk), dtype=torch.float32, device=device)  # logits
            Soft = torch.empty((Nq, Hq, Nk), dtype=torch.float32, device=device)  # softmax

            # 1) Matmul L = q_batch @ k_expanded^T, scaled
            grid_qk = (triton.cdiv(Nq, self.BLOCK_M), triton.cdiv(Nk, self.BLOCK_N), Hq)
            matmul_qk_kernel[grid_qk](
                q_batch, k_batch, L,
                Nq, Nk,
                Hq=Hq, D=D,
                BLOCK_M=self.BLOCK_M, BLOCK_N=self.BLOCK_N, BLOCK_D=self.BLOCK_D,
                num_warps=4, num_stages=2
            )

            # 2) Apply causal mask: j < (q_idx + 1 + delta)
            grid_mask = (triton.cdiv(Nq, self.BLOCK_M), triton.cdiv(Nk, self.BLOCK_N), Hq)
            apply_causal_mask_kernel[grid_mask](
                L, Nq, Nk, delta,
                Hq=Hq,
                BLOCK_M=self.BLOCK_M, BLOCK_N=self.BLOCK_N,
                num_warps=4, num_stages=2
            )

            # 3) Softmax along Nk per (q, head)
            grid_softmax = (triton.cdiv(Nq, self.BLOCK_M), Hq)
            softmax_rowwise_kernel[grid_softmax](
                L, Soft,
                Nq, Nk,
                Hq=Hq,
                BLOCK_M=self.BLOCK_M, BLOCK_N=self.BLOCK_N,
                num_warps=4, num_stages=2
            )

            # 4) Compute output Y = Soft @ v_expanded
            grid_attn = (triton.cdiv(Nq, self.BLOCK_M), Hq, triton.cdiv(D, self.BLOCK_D))
            attn_dot_v_kernel[grid_attn](
                Soft, v_batch, output[q_start:q_end],
                Nq, Nk, D=D,
                Hq=Hq,
                BLOCK_M=self.BLOCK_M, BLOCK_D=self.BLOCK_D, BLOCK_N=self.BLOCK_N,
                num_warps=4, num_stages=2
            )

            # 5) Compute LSE per (q, head): logsumexp(L) / ln(2)
            grid_lse = (triton.cdiv(Nq, self.BLOCK_M), Hq)
            lse_segment_kernel[grid_lse](
                L, lse[q_start:q_end],
                Nq, Nk, Hq=Hq,
                BLOCK_M=self.BLOCK_M, BLOCK_N=self.BLOCK_N,
                num_warps=4, num_stages=2
            )

        # Return output [total_q, 32, 128] and lse [total_q, 32]
        return output, lse
