import math
import torch
import triton
import triton.language as tl


@triton.jit
def segment_attention_kernel(
    q_ptr, k_ptr, v_ptr, out_ptr, lse_ptr,
    q_start, q_end, kv_start, kv_end, sm_scale,
):
    """
    Triton kernel: processes one segment defined by [q_start, q_end) and [kv_start, kv_end).
    - q_ptr: float32 [total_q, 32, 128]
    - k_ptr: float32 [total_kv, 8, 128]
    - v_ptr: float32 [total_kv, 8, 128]
    - out_ptr: float32 [total_q, 32, 128], will be zero-initialized before call
    - lse_ptr: float32 [total_q, 32], will be -inf initialized before call
    """
    Nq = q_end - q_start
    Nk = kv_end - kv_start
    delta = Nk - Nq

    # One program per (i, h)
    # Triton grid will be (Nq * 32,)
    pid = tl.program_id(axis=0)
    i = pid // 32
    h = pid % 32

    if i >= Nq:
        return

    # Load q[i, h, :]
    q_vec_base = (q_start + i) * 32 * 128 + h * 128
    q_vec = tl.load(q_ptr + q_vec_base)  # [128], float32

    # Prepare logits vector of length 8
    logits = tl.zeros((8,), dtype=tl.float32)

    # Compute dot with each kv head j = 0..7
    for j in tl.static_range(8):
        k_base = (kv_start + j) * 8 * 128 + (h % 8) * 128  # map q head to original kv head: orig_h = h % 8
        k_vec = tl.load(k_ptr + k_base)  # [128]
        dot = tl.sum(q_vec * k_vec, axis=0) * sm_scale  # scalar
        logits[j] = dot

        # Apply forward-causal mask: if j >= (i + 1 + delta), set -inf
        if j >= (i + 1 + delta):
            logits[j] = -float("inf")

    # Base-2 logsumexp over 8 positions
    m = tl.max(logits, axis=0)
    sum_exp = tl.sum(tl.exp(logits - m), axis=0)
    lse_val = m + tl.log(sum_exp) / 0.6931471805599453  # ln(2)

    # Softmax across 8 positions
    soft = tl.exp(logits - lse_val)  # [8], float32

    # Accumulate output[i, h, :] += soft[j] * v[kv_start + j, orig_h, :] for j in 0..7
    orig_h = h % 8
    for j in tl.static_range(8):
        v_base = (kv_start + j) * 8 * 128 + orig_h * 128  # v has 8 heads, use orig_h
        v_vec = tl.load(v_ptr + v_base)  # [128]
        out_base = (q_start + i) * 32 * 128 + h * 128
        # Store accumulation; out_ptr is float32
        tl.store(out_ptr + out_base, v_vec * soft[j])

    # Store lse[i, h]
    lse_index = (q_start + i) * 32 + h
    tl.store(lse_ptr + lse_index, lse_val)

def segment_attention_kernel(
    q_ptr, k_ptr, v_ptr, out_ptr, lse_ptr,
    q_start, q_end, kv_start, kv_end, sm_scale,
):
    """
    Triton kernel: processes one segment [q_start, q_end) and [kv_start, kv_end).
    For each (i, h) pair, computes attention:
      - logits[j] = dot(q[i, h, :], k_exp[kv_start + j, h%8, :]) * sm_scale for j in 0..7
      - Apply forward-causal mask: logits[j] = -inf if j >= (i + 1 + delta), delta = Nk - Nq
      - lse[i, h] = logsumexp(logits) / ln(2)
      - soft = softmax(logits, axis=8)
      - out[i, h, :] = sum_j soft[j] * v_exp[kv_start + j, h%8, :]
    Writes output to out_ptr and lse to lse_ptr.
    """
    Nq = q_end - q_start
    Nk = kv_end - kv_start
    delta = Nk - Nq

    # One program per (i, h)
    pid = tl.program_id(axis=0)
    i = pid // 32
    h = pid % 32
    if i >= Nq:
        return

    # Load q[i, h, :]
    q_vec_base = (q_start + i) * 32 * 128 + h * 128
    q_vec = tl.load(q_ptr + q_vec_base)  # [128], float32

    # Prepare logits vector of length 8 (we will compute and store per j)
    logits = tl.zeros((8,), dtype=tl.float32)

    # Compute dot with each kv head j = 0..7 and apply mask
    for j in tl.static_range(8):
        orig_h = h % 8
        k_base = (kv_start + j) * 8 * 128 + orig_h * 128  # k has 8 heads; map q head to kv head
        k_vec = tl.load(k_ptr + k_base)  # [128]
        dot = tl.sum(q_vec * k_vec, axis=0) * sm_scale
        logits[j] = dot
        # Apply forward-causal mask: if j >= (i + 1 + delta), set -inf
        if j >= (i + 1 + delta):
            logits[j] = -float("inf")

    # Base-2 logsumexp over 8 positions
    m = tl.max(logits, axis=0)
    sum_exp = tl.sum(tl.exp(logits - m), axis=0)
    lse_val = m + tl.log(sum_exp) / 0.6931471805599453  # ln(2)

    # Softmax across 8 positions
    soft = tl.exp(logits - lse_val)  # [8], float32

    # Accumulate output[i, h, :] += soft[j] * v[kv_start + j, orig_h, :] for j in 0..7
    out_base = (q_start + i) * 32 * 128 + h * 128
    orig_h = h % 8
    for j in tl.static_range(8):
        v_base = (kv_start + j) * 8 * 128 + orig_h * 128
        v_vec = tl.load(v_ptr + v_base)  # [128]
        # Accumulate into out_ptr
        tl.store(out_ptr + out_base, v_vec * soft[j])

    # Store lse[i, h]
    lse_index = (q_start + i) * 32 + h
    tl.store(lse_ptr + lse_index, lse_val)
