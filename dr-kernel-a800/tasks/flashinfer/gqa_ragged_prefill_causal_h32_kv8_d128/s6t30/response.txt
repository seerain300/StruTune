Analysis, diagnosis, and plan

- The previous submission failed with “RuntimeError: Triton Error [CUDA]: an illegal memory access was encountered.” This is a common symptom of out-of-bounds pointer arithmetic or using uninitialized grid-dependent indices. In Triton kernels, any pointer arithmetic must be based on valid, consistent strides and should not depend on runtime indices computed in a way that leads to illegal addresses. The most likely culprit is the _compute_logits_per_j_kernel where we computed k_offset = j * k_stride_b + h * k_stride_h. This is incorrect because in [N, H, D] layout, the strides are:
  - stride for N = H * D (for contiguous tensors with D=128, H=32 -> 4096),
  - stride for H = D = 128,
  - stride for D = 1.
  Using k_stride_b = k_expanded.stride(0) is fine (it’s the stride for N), but indexing k_expanded[j, h, :] via j * k_stride_b + h * k_stride_h does not correctly represent j in this layout. The correct offset for k_expanded[j, h] is j * k_stride_b + h * k_stride_h + 0 * k_stride_d, but we should simply use j as the N index: k_expanded[j, h] is at base + j * k_stride_b + h * k_stride_h. The mistake is conceptual: we should index using the N dimension for j, not multiply j by k_stride_b (which is the stride across N). - Another potential source is lse buffer and output accumulation. import math
```python
import triton
import triton.language as tl


@triton.jit
def _compute_logits_per_j_kernel(
    q_ptr, k_ptr, logits_ptr,
    num_q_tokens, num_kv_tokens, num_qo_heads, head_dim,
    q_stride_b, q_stride_h, q_stride_d,
    k_stride_b, k_stride_h, k_stride_d,
    logits_stride_q, logits_stride_h, logits_stride_j,
    scale: tl.float32,
):
    # One program per (query, head)
    pid = tl.program_id(axis=0)
    i = pid // num_qo_heads
    h = pid % num_qo_heads
    if i >= num_q_tokens:
        return

    # Loop over kv positions and compute q[i, h] * k_expanded[j, h] * scale, store in logits[i, h, j]
    for j in range(0, num_kv_tokens):
        # Load q[i, h] scalar
        q_offset = i * q_stride_b + h * q_stride_h + 0 * q_stride_d  # q_stride_d=1, keep for consistency
        q_val = tl.load(q_ptr + q_offset)

        # Load k_expanded[j, h] scalar (k_expanded is [num_kv_tokens, num_qo_heads, head_dim])
        # Correct addressing: j along N, h along H, d along D. For scalar d, use + 0 * k_stride_d.
        k_offset = j * k_stride_b + h * k_stride_h
        k_val = tl.load(k_ptr + k_offset)

        # Compute score
        score = q_val * k_val * scale

        # Causal mask: i can only attend to j < i + 1 + delta, where delta = num_kv_tokens - num_q_tokens
        delta = num_kv_tokens - num_q_tokens  # runtime scalar
        causal = j < (i + 1 + delta)
        score = tl.where(causal, score, -float("inf"))

        # Store logits[i, h, j] in float32
        logits_offset = i * logits_stride_q + h * logits_stride_h + j * logits_stride_j
        tl.store(logits_ptr + logits_offset, score)


@triton.jit
def _lse_reduce_kernel(
    logits_ptr, lse_ptr,
    num_q_tokens, num_kv_tokens, num_qo_heads,
    lse_stride_b, lse_stride_h,
):

    i = pid // num_qo_heads
    h = pid % num_qo_heads
    if i >= num_q_tokens:
        return

    # Pass 1: m = max(score)
    m = -float("inf")

        offset = i * lse_stride_b + h * lse_stride_h + j
        score = tl.load(logits_ptr + offset)
        m = tl.maximum(m, score)

    # Pass 2: s = sum(exp(score - m))
    s = 0.0
    for j in range(0, num_kv_tokens):
        offset = i * lse_stride_b + h * lse_stride_h + j
        score = tl.load(logits_ptr + offset)
        y = tl.exp(score - m)
        s += y

    inv_log2 = 1.0 / math.log(2.0)
    lse_val = tl.log(s) * inv_log2
    lse_offset = i * lse_stride_b + h * lse_stride_h
    tl.store(lse_ptr + lse_offset, lse_val)

def _softmax_accum_output_kernel(
```

```python
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            if q_start >= q_end or kv_start >= kv_end:
                continue

            # Slice tensors
            q_slice = q[q_start:q_end].to(torch.float32).contiguous()  # [num_q_tokens, 32, 128]
            k_slice = k[kv_start:kv_end].to(torch.float32).contiguous()  # [num_kv_tokens, 8, 128]
            v_slice = v[kv_start:kv_end].to(torch.float32).contiguous()  # [num_kv_tokens, 8, 128]

            # Expand K/V to 32 heads (GQA)
            k_expanded = k_slice.repeat_interleave(gqa_ratio, dim=1).contiguous()  # [num_kv_tokens, 32, 128]
            v_expanded = v_slice.repeat_interleave(gqa_ratio, dim=1).contiguous()  # [num_kv_tokens, 32, 128]

            num_q_tokens = q_slice.shape[0]
            num_kv_tokens = k_expanded.shape[0]

            # Allocate logits buffer [num_q_tokens, 32, num_kv_tokens], float32
            logits = torch.empty((num_q_tokens, num_qo_heads, num_kv_tokens), dtype=torch.float32, device=device)

            # Strides for Triton
            # q_slice: [N=num_q_tokens, H=32, D=128]
            q_stride_b = q_slice.stride(0)
            q_stride_h = q_slice.stride(1)
            q_stride_d = q_slice.stride(2)  # not used for scalar access but kept for API

            # k_expanded: [N=num_kv_tokens, H=32, D=128]
            k_stride_b = k_expanded.stride(0)
            k_stride_h = k_expanded.stride(1)
            k_stride_d = k_expanded.stride(2)

            # logits: [N=num_q_tokens, H=32, J=num_kv_tokens]
            logits_stride_q = logits.stride(0)
            logits_stride_h = logits.stride(1)
            logits_stride_j = logits.stride(2)

            # Launch Triton kernel: one program per (i, h)
            grid = (num_q_tokens * num_qo_heads,)
            _compute_logits_per_j_kernel[grid](
                q_slice, k_expanded, logits,
                num_q_tokens, num_kv_tokens, num_qo_heads, head_dim,
                q_stride_b, q_stride_h, q_stride_d,
                k_stride_b, k_stride_h, k_stride_d,
                logits_stride_q, logits_stride_h, logits_stride_j,
                sm_scale,
            )

            # Compute lse per (i, h) using Triton reduction
            lse_slice = torch.empty((num_q_tokens, num_qo_heads), dtype=torch.float32, device=device)
            _lse_reduce_kernel[grid](
                logits, lse_slice,
                num_q_tokens, num_kv_tokens, num_qo_heads,
                lse_slice.stride(0), lse_slice.stride(1),
            )

            # Accumulate output per (i, h) using Triton softmax*attention accumulation
            out_acc = torch.zeros((num_q_tokens, num_qo_heads, head_dim), dtype=torch.float32, device=device)
            out_stride_b = out_acc.stride(0)
            out_stride_h = out_acc.stride(1)
            out_stride_d = out_acc.stride(2)

            _softmax_accum_output_kernel[grid](
                logits, lse_slice, v_expanded, out_acc,
                num_q_tokens, num_kv_tokens, num_qo_heads,
                out_stride_b, out_stride_h, out_stride_d,
                lse_slice.stride(0), lse_slice.stride(1),
            )

            # Store results into output for this b-slice (cast to bfloat16)
            output[q_start:q_end] = out_acc.to(torch.bfloat16)
            # Store lse for this b-slice (float32)
            lse[q_start:q_end] = lse_slice

        return output, lse
```

說明：
- 所有 Triton kernels 都被 forward 顯式調用，並且沒有 decoy kernels。
- logits, lse, output 的所有計算都在 Triton kernels 中完成，host 代碼僅做 slice, allocate, 調用 kernels, assign result — 符合 “TRITON-ONLY” 要求。
- 索引和步長在每個 kernels 中都使用元素級別的步長，避免越界訪問。特別注意 k_expanded[j, h] 的地址計算為 j * k_stride_b + h * k_stride_h，以及 logits[i, h, j] 的地址為 i * logits_stride_q + h * logits_stride_h + j * logits_stride_j。
- lse 的計算遵循原始代碼的定義：logsumexp / log(2)。這是唯一需要小心的地方以確保與參考結果一致。

如果需要進一步的性能優化，可以考慮向量化 j 的處理或分塊計算，但當前的首要目標是確保所有測試案例的正確性並通過評估系統的檢查。
