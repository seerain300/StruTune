    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    mask_k = k_offsets < Nk

def lse_softmax_lastdim_kernel(
    L_ptr, LSE_ptr,
    Nq: tl.int32, Nk: tl.int32, Hq: tl.constexpr,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    # Grid: (pid0 over Nq, pid1 over Hq)
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    q = pid0
    h = pid1

    # If q >= Nq, nothing to do
    if q >= Nq:
        return

    # Initialize max and sum for logsumexp
    max_val = -float('inf')
    sum_exp = 0.0

    for k_start in range(0, Nk, BLOCK_N):
        k_offsets = k_start + tl.arange(0, BLOCK_N)
        mask_k = k_offsets < Nk

        # Load logits for this (q,h) row: [BLOCK_N]
        L_ptrs = L_ptr + q * (Hq * Nk) + h * Nk + k_offsets
        x = tl.load(L_ptrs, mask=mask_k, other=-float('inf'))

        # Apply causal mask: kv < (q + 1 + delta)
        kv_positions = k_offsets
        cond = kv_positions < (q + 1 + delta)
        x = tl.where(cond, x, -float('inf'))

        # Numerical reduction for LSE
        local_max = tl.max(x, axis=0)
        # Accumulate sum of exp(x - max) over valid positions
        # We need to sum only where x is not -inf; tl.where can help:
        x_for_sum = tl.where(x == -float('inf'), -float('inf'), x)
        exp_vec = tl.exp(x_for_sum - local_max)
        # Zero out invalid positions
        exp_vec = tl.where(cond, exp_vec, 0.0)
        local_sum = tl.sum(exp_vec, axis=0)

        # Update global max and sum
        max_val = tl.maximum(max_val, local_max)
        sum_exp += local_sum

    # LSE = max + log(sum(exp(x - max))) / ln(2)
    lse_val = (max_val + tl.log(sum_exp)) * inv_ln2
    # Store [q, h]
    tl.store(LSE_ptr + q * Hq + h, lse_val)
