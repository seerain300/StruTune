@triton.jit
def apply_causal_mask_kernel(
    L_ptr,
    Nq: tl.int32, Nk: tl.int32,
    delta: tl.int32,
    sm_scale: tl.float32,  # not used here, but passed for consistency
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    In-place: set L[q, h, j] = -inf if j >= q + 1 + delta; else keep L.
    Grid: (Nq tiles, Nk tiles).
    """
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    j_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_j = j_offsets < Nk

    # Compute condition: j < (q + 1 + delta)
    cond = j_offsets[None, :] < (q_offsets[:, None] + 1 + delta)

    tile = tl.load(
        L_ptr + q_offsets[:, None] * (32 * Nk) + tl.arange(0, 32) * Nk + j_offsets[None, :],
        mask=mask_q[:, None] & mask_j[None, :],
        other=0.0
    )
    tile = tl.where(cond, tile, -float('inf'))
    tl.store(
        L_ptr + q_offsets[:, None] * (32 * Nk) + tl.arange(0, 32) * Nk + j_offsets[None, :],
        tile,
        mask=mask_q[:, None] & mask_j[None, :]
    )


@triton.jit
def masked_softmax_dimN_kernel(
    L_ptr, Soft_ptr,
    Nq: tl.int32, Nk: tl.int32,
    Hq: tl.constexpr,  # we use h as program_id(2)
    sm_scale: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Softmax along Nk for each (q, head). Soft_ptr: [Nq, Hq, Nk], float32.
    L_ptr: logits with potential -inf masked; soft output: same layout.
    """
    pid0 = tl.program_id(0)  # tile over Nq
    pid1 = tl.program_id(1)  # tile over Nk
    pid2 = tl.program_id(2)  # head index (0..Hq-1)

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    j_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_j = j_offsets < Nk

    # Load logits tile [BLOCK_M, BLOCK_N], scaled by sm_scale
    tile = tl.load(
        L_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + j_offsets[None, :],
        mask=mask_q[:, None] & mask_j[None, :],
        other=-float('inf')
    ) * sm_scale

    # Numerically stable softmax: subtract max per row
    tile_max = tl.max(tile, axis=1)  # [BLOCK_M]
    tile = tile - tile_max[:, None]

    exp_tile = tl.exp(tile)
    sum_exp = tl.sum(exp_tile, axis=1)  # [BLOCK_M]
    soft = exp_tile / sum_exp[:, None]

    tl.store(
        Soft_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + j_offsets[None, :],
        soft,
        mask=mask_q[:, None] & mask_j[None, :]
    )


@triton.jit
def attn_dot_v_kernel(
    Soft_ptr, V_ptr, Y_ptr,
    Nq: tl.int32, Nk: tl.int32,
    Hq: tl.constexpr, D: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Compute Y = Soft @ V_expanded^T, where:
      - Soft: [Nq, Hq, Nk] float32
      - V: [Nk, 8, 128], we expand to [Nk, 32, 128] logically but pass actual 8 heads; kernel assumes D=128.
      - Y: [Nq, Hq, 128] float32
    Grid: (Nq tiles, Nk tiles, Hq)
    """
    pid0 = tl.program_id(0)  # tile over Nq
    pid1 = tl.program_id(1)  # tile over Nk
    pid2 = tl.program_id(2)  # head index (0..Hq-1)

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    k_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_k = k_offsets < Nk

    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # We assume D=128; do a single 128-wide reduction
    for d_start in tl.static_range(0, 128, 128):
        d_offsets = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_offsets < 128

        # Load Soft tile [BLOCK_M, BLOCK_N]
        Soft_tile = tl.load(
            Soft_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + k_offsets[None, :],
            mask=mask_q[:, None] & mask_k[None, :],
            other=0.0
        )  # [BLOCK_M, BLOCK_N]

        # Load V tile as [BLOCK_N, BLOCK_D] (expanded V); V is [Nk, 8, 128] but we treat it as [Nk, 32, 128]
        # We need to select the expanded head index pid2. Here we assume we're using the expanded V with 32 heads.
        # Since Triton kernel is launched with Hq=32, we can index V_ptr directly at head index pid2.
        # However, V tensor actually has 8 heads; we need to expand logically in Python by repeating.
        # For simplicity and correctness, we will implement expansion in Python before kernel launch by repeating.
        # Not done here; instead, we ensure V passed is already expanded to 32 heads.
        V_tile = tl.load(
            V_ptr + k_offsets[:, None] * (32 * D) + pid2 * D + d_offsets[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0
        )  # [BLOCK_N, BLOCK_D]

        acc += tl.dot(Soft_tile, V_tile)

    tl.store(
        Y_ptr + q_offsets[:, None] * (Hq * D) + pid2 * D + d_offsets[None, :],
        acc,
        mask=mask_q[:, None] & mask_d[None, :]
    )

class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k, v, qo_indptr, kv_indptr, sm_scale):
        """
        q: [total_q, 32, 128], bfloat16
        k: [total_kv, 8, 128], bfloat16
        v: [total_kv, 8, 128], bfloat16
        qo_indptr: [len_indptr], int32
        kv_indptr: [len_indptr], int32
        sm_scale: float32 scalar
        Returns (output: [total_q, 32, 128], lse: [total_q, 32], both float32)
        """
        assert q.is_cuda and k.is_cuda and v.is_cuda, "Triton kernels require CUDA tensors"
        device = q.device

        total_q = int(qo_indptr[-1].item())
        total_kv = int(kv_indptr[-1].item())
        len_indptr = qo_indptr.shape[0]
        num_qo_heads = 32
        num_kv_heads = 8
        head_dim = 128
        gqa_ratio = num_qo_heads // num_kv_heads  # 4

        # Output and LSE tensors (float32 for numerics)
        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Iterate segments
        for b in range(len_indptr - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            if q_start >= q_end or kv_start >= kv_end:
                continue

            q_batch = q[q_start:q_end].contiguous()  # [num_q_tokens, 32, 128]
            k_batch = k[kv_start:kv_end].contiguous()  # [num_kv_tokens, 8, 128]
            v_batch = v[kv_start:kv_end].contiguous()  # [num_kv_tokens, 8, 128]

            num_q_tokens = q_batch.shape[0]
            num_kv_tokens = k_batch.shape[0]
            delta = num_kv_tokens - num_q_tokens

            # Expand K and V by GQA ratio to 32 heads
            k_expanded = k_batch.repeat_interleave(gqa_ratio, dim=1)  # [num_kv_tokens, 32, 128]
            v_expanded = v_batch.repeat_interleave(gqa_ratio, dim=1)  # [num_kv_tokens, 32, 128]

            # Prepare intermediate buffers
            logits = torch.empty((num_q_tokens, num_qo_heads, num_kv_tokens), dtype=torch.float32, device=device)
            softmax = torch.empty_like(logits)

            # Launch qk_matmul: [Nq, 32, Nk]
            BLOCK_M, BLOCK_N = 32, 32
            grid_qk = (triton.cdiv(num_q_tokens, BLOCK_M), triton.cdiv(num_kv_tokens, BLOCK_N), num_qo_heads)
            qk_matmul_kernel[grid_qk](
                q_batch, k_expanded, logits,
                num_q_tokens, num_kv_tokens,
                32, 128,
                BLOCK_M, BLOCK_N,
                sm_scale=sm_scale
            )

            # Apply causal mask in-place on logits
            grid_mask = (triton.cdiv(num_q_tokens, BLOCK_M), triton.cdiv(num_kv_tokens, BLOCK_N))
            apply_causal_mask_kernel[grid_mask](
                logits,
                num_q_tokens, num_kv_tokens,
                delta,
                sm_scale
            )

            # Softmax along Nk (dim=2) per (q, head)
            grid_softmax = (triton.cdiv(num_q_tokens, BLOCK_M), triton.cdiv(num_kv_tokens, BLOCK_N), num_qo_heads)
            masked_softmax_dimN_kernel[grid_softmax](
                logits, softmax,
                num_q_tokens, num_kv_tokens,
                32,
                sm_scale
            )

            # Compute output Y = softmax @ v_expanded
            # V_expanded is [Nk, 32, 128]; we pass as such. Kernel expects V expanded to 32 heads.
            Y = torch.empty((num_q_tokens, num_qo_heads, head_dim), dtype=torch.float32, device=device)
            BLOCK_M2, BLOCK_N2, BLOCK_D = 32, 32, 128
            grid_attn = (triton.cdiv(num_q_tokens, BLOCK_M2), triton.cdiv(num_kv_tokens, BLOCK_N2), num_qo_heads)
            attn_dot_v_kernel[grid_attn](
                softmax, v_expanded, Y,
                num_q_tokens, num_kv_tokens,
                32, 128,
                BLOCK_M2, BLOCK_N2, BLOCK_D
            )

            # Store results for this segment
            output[q_start:q_end] = Y
            lse[q_start:q_end] = self._compute_lse(logits * (1.0 / sm_scale), num_q_tokens, num_kv_tokens)

        return output, lse

    @staticmethod
    def _compute_lse(logits, Nq, Nk):
        # Compute LSE per (q, head): logsumexp over Nk after removing scaling
        # logits: [Nq, Hq, Nk], float32
        # LSE per head h: log(sum_j exp(logits[n,h,j])) / log(2)
        # Use PyTorch for robustness
        max_vals = torch.amax(logits, dim=-1, keepdim=True)  # [Nq, Hq, 1]
        sum_vals = torch.sum(torch.exp(logits - max_vals), dim=-1)  # [Nq, Hq]
        lse = torch.log(sum_vals)  # [Nq, Hq]
        return (lse / math.log(2.0)).detach().cpu().numpy()  # detach and return as float32 tensor on device

import torch
import triton
import triton.language as tl


@triton.jit
def qk_matmul_kernel(
    Q_ptr, K_ptr, L_ptr,
    Nq: tl.int32, Nk: tl.int32,
    Hq: tl.constexpr, D: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Compute logits L = Q @ K_expanded^T, where:
      - Q: [Nq, Hq, D], Hq=32, D=128
      - K_expanded: [Nk, Hq, D] (we pass K with 8 heads and expand in host)
      - L: [Nq, Hq, Nk]
    """
    pid0 = tl.program_id(0)  # tile over Nq
    pid1 = tl.program_id(1)  # tile over Nk
    pid2 = tl.program_id(2)  # head index 0..Hq-1

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    j_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_j = j_offsets < Nk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over D (compile-time 128) to compute Q @ K^T
    for d in tl.static_range(0, D):
        q_vec = tl.load(
            Q_ptr + q_offsets[:, None] * (Hq * D) + pid2 * D + d,
            mask=mask_q[:, None],
            other=0.0  # [BLOCK_M, 1]
        )
        k_vec = tl.load(
            K_ptr + j_offsets[:, None] * (Hq * D) + pid2 * D + d,
            mask=mask_j[:, None],
            other=0.0  # [BLOCK_N, 1]
        )
        # acc += q @ k^T -> [BLOCK_M, 1] @ [1, BLOCK_N] = [BLOCK_M, BLOCK_N]

    # Apply scaling
    acc = acc * sm_scale

    tl.store(
        L_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + j_offsets[None, :],
        acc,
        mask=mask_q[:, None] & mask_j[None, :]
    )


@triton.jit
def attn_dot_v_kernel(
    Soft_ptr, V_ptr, Y_ptr,
    Nq: tl.int32, Nk: tl.int32,
    Hq: tl.constexpr, D: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Compute Y = Soft @ V_expanded^T, where:
      - Soft: [Nq, Hq, Nk] float32 (attention weights)
      - V_expanded: [Nk, Hq, D] (we pass expanded to 32 heads; here D=128)
      - Y: [Nq, Hq, D] float32
    Grid: (Nq tiles, Nk tiles, Hq)
    """
    pid0 = tl.program_id(0)  # tile over Nq
    pid1 = tl.program_id(1)  # tile over Nk
    pid2 = tl.program_id(2)  # head index 0..Hq-1

    q_offsets = pid0 * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    j_offsets = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_q = q_offsets < Nq
    mask_j = j_offsets < Nk

    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # D=128; single iteration
    d = 0
    Soft_tile = tl.load(
        Soft_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + j_offsets[None, :],
        mask=mask_q[:, None] & mask_j[None, :],
        other=0.0
    )  # [BLOCK_M, BLOCK_N]
    V_tile = tl.load(
        V_ptr + j_offsets[:, None] * (Hq * D) + pid2 * D + d,
        mask=mask_j[:, None],
        other=0.0  # [BLOCK_N, 1]
    )  # [BLOCK_N, 1]
    acc += Soft_tile @ tl.trans(V_tile)

    # For completeness, if D > 1, we would iterate; here D=128 and we handle all in next loop below.
    # However, we need a proper loop; Triton requires static unrolling over D dimension.
    # Since D is 128, we can do a single iteration using tiles:
    for d_start in tl.static_range(0, 128, 128):
        d_offsets = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_offsets < 128
        # Load Soft tile [BLOCK_M, BLOCK_N]
        Soft_tile = tl.load(
            Soft_ptr + q_offsets[:, None] * (Hq * Nk) + pid2 * Nk + j_offsets[None, :],
            mask=mask_q[:, None] & mask_j[None, :],
            other=0.0
        )
        # Load V tile as [BLOCK_N, BLOCK_D]
        V_tile = tl.load(
            V_ptr + j_offsets[:, None] * (Hq * D) + pid2 * D + d_offsets[None, :],
            mask=mask_j[:, None] & mask_d[None, :],
            other=0.0
        )
        acc += Soft_tile @ V_tile  # [BLOCK_M, BLOCK_D]

    tl.store(
        Y_ptr + q_offsets[:, None] * (Hq * D) + pid2 * D + d_offsets[None, :],
        acc,
        mask=mask_q[:, None] & mask_d[None, :]
    )

        super().__init__()

    def forward(self, q, k, v, qo_indptr, kv_indptr, sm_scale):
        """
        q: [total_q, 32, 128], bfloat16
        k: [total_kv, 8, 128], bfloat16
        v: [total_kv, 8, 128], bfloat16
        qo_indptr: [len_indptr], int32
        kv_indptr: [len_indptr], int32
        sm_scale: float32 scalar
        Returns (output: [total_q, 32, 128], lse: [total_q, 32], both float32)
        """
        assert q.is_cuda and k.is_cuda and v.is_cuda, "Triton kernels require CUDA tensors"
        device = q.device

        # Ensure contiguity
        q = q.contiguous()
        k = k.contiguous()
        v = v.contiguous()

        total_q = int(qo_indptr[-1].item())
        total_kv = int(kv_indptr[-1].item())

        num_qo_heads = 32
        num_kv_heads = 8
        head_dim = 128
        gqa_ratio = num_qo_heads // num_kv_heads  # 4

        output = torch.empty((total_q, num_qo_heads, head_dim), dtype=torch.float32, device=device)
        lse = torch.empty((total_q, num_qo_heads), dtype=torch.float32, device=device)

        # Iterate segments
        for b in range(len_indptr - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            if q_start >= q_end or kv_start >= kv_end:
                continue

            k_batch = k[kv_start:kv_end]  # [num_kv_tokens, 8, 128]
            v_batch = v[kv_start:kv_end]  # [num_kv_tokens, 8, 128]

            num_q_tokens = q_batch.shape[0]
            num_kv_tokens = k_batch.shape[0]
            delta = num_kv_tokens - num_q_tokens

            # Expand K and V
