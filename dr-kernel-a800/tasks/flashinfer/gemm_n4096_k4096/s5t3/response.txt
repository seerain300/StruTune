    configs=[
        triton.Config({'BLOCK_N': 128, 'BLOCK_K': 256}, num_warps=4, num_stages=2),

        triton.Config({'BLOCK_N': 512, 'BLOCK_K': 512}, num_warps=8, num_stages=2),
    ],
    key=['N', 'K'],
)
@triton.jit
def _row_matmul_bt_kernel(
    A_ptr, B_ptr, Y_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_yn,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Since M == 1, we only need to index along rows of A by 0
    pid_n = tl.program_id(0)
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Accumulator for this tile of columns
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)

        # A is [M, K], but M==1; A[0, k]
        A_ptrs = A_ptr + 0 * stride_am + k_offsets * stride_ak
        a_mask = (k_offsets < K)
        a = tl.load(A_ptrs, mask=a_mask, other=0.0)  # shape [BLOCK_K], fp16

        # B_T[n, k] = B[k, n] with strides: (stride_bk along k, stride_bn along n)
        B_ptrs = B_ptr + n_offsets[:, None] * stride_bn + k_offsets[None, :] * stride_bk
        b_mask = (n_offsets[:, None] < N) & (k_offsets[None, :] < K)
        b = tl.load(B_ptrs, mask=b_mask, other=0.0)  # shape [BLOCK_N, BLOCK_K], fp16

        # Accumulate: acc[n] += sum_k a[k] * b[n, k]
        acc += tl.sum(b * a[None, :], axis=1)

    Y_ptrs = Y_ptr + n_offsets * stride_yn
