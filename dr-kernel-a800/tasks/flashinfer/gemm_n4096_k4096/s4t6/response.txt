import torch
import triton
import triton.language as tl


@triton.jit
def matmul_transpose_row_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,   # strides for A: A is [M, K]
    stride_bk, stride_bn,   # strides for B: B is [K, N]
    stride_cm, stride_cn,   # strides for C: C is [M, N]
    BLOCK_N: tl.constexpr,
):
    # Each program handles one row m and a block of columns n
    pid_m = tl.program_id(0)   # row index
    pid_n = tl.program_id(1)   # block index along columns

    # Column offsets for this block
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    n_mask = n_offsets < N

    # Initialize accumulator for this row block
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K dimension
    # We load A[m, k] as a scalar and B[n, k] as a vector, accumulate
    for k in range(0, K):
        # Load A[m, k] (scalar). Mask is not needed for m < M; grid ensures pid_m < M.
        a_val = tl.load(A_ptr + pid_m * stride_am + k * stride_ak)
        # Load B[n, k] as a vector
        b_vec = tl.load(B_ptr + k * stride_bk + n_offsets * stride_bn, mask=n_mask, other=0.0)
        # Accumulate
        acc += a_val.to(tl.float32) * b_vec.to(tl.float32)

    # Store result to C[m, n_offsets]
    tl.store(C_ptr + pid_m * stride_cm + n_offsets * stride_cn, acc, mask=n_mask)


def _matmul_transpose_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B.T using Triton. A: [M, K], B: [K, N], C: [M, N].
    Returns C as float16 tensor (to match typical inputs).
    """
    assert A.is_cuda and B.is_cuda, "Triton kernel requires CUDA tensors"
    # Make tensors contiguous and in fp16 to match original get_inputs behavior
    A = A.contiguous().to(torch.float16)
    B = B.contiguous().to(torch.float16)

    M, K = A.shape
    Kb, N = B.shape
    assert K == Kb, "A's second dim must match B's first dim"

    C = torch.empty((M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements, Triton uses element-wise pointer arithmetic)
    stride_am, stride_ak = A.stride(0), A.stride(1)
    stride_bk, stride_bn = B.stride(0), B.stride(1)
    stride_cm, stride_cn = C.stride(0), C.stride(1)

    # Launch configuration: one program per row and per column block
    BLOCK_N = 128
    grid = (M, triton.cdiv(N, BLOCK_N))

    matmul_transpose_row_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_N=BLOCK_N,
        num_warps=4, num_stages=2,
    )
    return C
