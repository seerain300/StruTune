import torch
import triton
import triton.language as tl

# Autotuned Triton kernel for M == 1:
# Computes Y[0, n] = sum_k A[0, k] * B[k, n], with B shape [K, N] and arbitrary strides.
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_N': 128,  'BLOCK_K': 128}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_N': 256,  'BLOCK_K': 128}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_N': 256,  'BLOCK_K': 256}, num_warps=8, num_stages=3),
        triton.Config({'BLOCK_N': 512,  'BLOCK_K': 128}, num_warps=8, num_stages=3),
        triton.Config({'BLOCK_N': 512,  'BLOCK_K': 256}, num_warps=8, num_stages=3),
        triton.Config({'BLOCK_N': 1024, 'BLOCK_K': 128}, num_warps=8, num_stages=3),
        triton.Config({'BLOCK_N': 1024, 'BLOCK_K': 256}, num_warps=8, num_stages=3),
    ],
    key=['N', 'K'],
)
@triton.jit
def _row_matmul_bt_kernel_1xN(
    A_ptr,  # *f16, A is [1, K], we only use row 0
    B_ptr,  # *f16, B is [K, N] with arbitrary strides
    Y_ptr,  # *f16, Y is [1, N], row 0
    M, N, K,
    am, ak,            # strides for A: A has shape [M, K], but M == 1 so am unused in practice
    bn, bk,            # strides for B: [K, N], bn along N, bk along K
    # OUTPUT strides for Y (row-major): Y has shape [M, N], we only use row 0
    stride_ym, stride_yn,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 1D grid over N tiles
    pid_n = tl.program_id(0)
    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    n_mask = n_offsets < N

    # Accumulator for the BLOCK_N outputs: compute in fp32 for stability
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K in chunks
    for k_start in range(0, K, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < K

        # Load A row 0 over k_offsets (A is [1, K])
        A_row_ptr = A_ptr + 0 * am + k_offsets * ak
        a_vec = tl.load(A_row_ptr, mask=k_mask, other=0.0)  # [BLOCK_K], f16

        # Load B tile [k_offsets, n_offsets]: shape [BLOCK_K, BLOCK_N], with strides (bk, bn)
        B_ptrs = B_ptr + k_offsets[:, None] * bk + n_offsets[None, :] * bn
        b_tile = tl.load(B_ptrs, mask=(k_mask[:, None] & n_mask[None, :]), other=0.0)  # [BLOCK_K, BLOCK_N], f16

        # Accumulate dot products: sum over k for each n
        # Cast to fp32 for robust accumulation
        a_vec_f32 = a_vec.to(tl.float32)                     # [BLOCK_K]
        b_tile_f32 = b_tile.to(tl.float32)                  # [BLOCK_K, BLOCK_N]
        partial = tl.sum(b_tile_f32 * a_vec_f32[:, None], axis=0)  # [BLOCK_N]
        acc += partial

    # Store results to Y row 0
    Y_row_ptrs = Y_ptr + 0 * stride_ym + n_offsets * stride_yn
    tl.store(Y_row_ptrs, acc.to(tl.float16), mask=n_mask)

    def forward(self, A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:

        if B.dtype != torch.float16:
            B = B.to(torch.float16)

        M, K = A.shape
        Kb, N = B.shape
        if K != Kb:
            raise ValueError(f"Incompatible shapes: A is (M={M}, K={K}), B is (K={Kb}, N={N}). B's K must equal A's K.")

        # Allocate output Y [M, N], fp16, contiguous

        if M == 1:

            grid = (triton.cdiv(N, 256),)  # initial grid; autotuner overrides effective BLOCK_N
            _row_matmul_bt_kernel_1xN[grid](
                A, B, Y,
                M, N, K,
                A.stride(0), A.stride(1),           # strides for A [1, K]
                B.stride(0), B.stride(1),           # strides for B [K, N]
                Y.stride(0), Y.stride(1),           # strides for Y [1, N]
            )
            return Y
        else:

            # If needed later, a 2D Triton GEMM can be added, but correctness failures were in M==1.

# Computes Y[0, n] = sum_k A[0, k] * B[k, n] using B's strides directly.

)

    B_ptr,   # *f16, B is [K, N] with arbitrary strides
    Y_ptr,   # *f16, Y is [1, N]
    M, N, K,
    am, ak,               # strides for A [M, K] (M==1, but we pass them generically)
    bn, bk,               # strides for B [K, N]
    ym, yn,               # strides for Y [M, N]
    BLOCK_N: tl.constexpr,
):
    # 1D grid over N tiles
    pid_n = tl.program_id(0)
    n_start = pid_n * BLOCK_N
    n_offsets = n_start + tl.arange(0, BLOCK_N)
    n_mask = n_offsets < N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)

    # Loop over K and accumulate A[0, k] * B[k, n]
    for k in range(0, K):
        # Load A[0, k] as scalar
        a_val = tl.load(A_ptr + 0 * am + k * ak)  # A[0, k]
        a_val_f32 = a_val.to(tl.float32)

        b_vec_f32 = b_vec.to(tl.float32)

        acc += b_vec_f32 * a_val_f32

    # Store results to Y[0, n]

    tl.store(Y_row_ptrs, acc.to(tl.float16), mask=n_mask)
