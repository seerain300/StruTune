import torch
import triton
import triton.language as tl


@triton.jit
def matmul_at_bT_kernel(
    A_ptr,  # *ptr to A (M x K), dtype: f16
    BT_ptr,  # *ptr to B.T (K x N), dtype: f16 (i.e., original B transposed)
    C_ptr,  # *ptr to C (M x N), dtype: f16
    M, N, K,
    stride_am, stride_ak,  # strides for A: row-major assumed but we use given strides
    stride_btk, stride_btn,  # strides for BT: (K, N)
    stride_cm, stride_cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program ids for 2D grid
    pid_m = tl.program_id(axis=0)  # along M dimension
    pid_n = tl.program_id(axis=1)  # along N dimension

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator in f16
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over K dimension in chunks
    # Triton requires Python range for runtime K; this avoids static_range issues.
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)  # current K chunk

        # Pointers for A and BT tiles
        A_tile_ptr = A_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        BT_tile_ptr = BT_ptr + (offs_k[:, None] * stride_btk + offs_n[None, :] * stride_btn)

        # Masks for boundary handling
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load tiles as f16
        a = tl.load(A_tile_ptr, mask=a_mask, other=0.0)
        b = tl.load(BT_tile_ptr, mask=b_mask, other=0.0)

        # Accumulate in f16 (matching original dtype to avoid upcast-related discrepancies)
        acc += a @ b

    # Write back results
    C_tile_ptr = C_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(C_tile_ptr, acc, mask=c_mask)
