import torch
import triton
import triton.language as tl


@triton.jit
def b_transpose_kernel(
    B_ptr, BT_ptr,
    K, N,
    stride_bk, stride_bn,   # strides for B: B is [K, N] (contiguous => stride_bn=1, stride_bk=N)
    stride_btk, stride_btn, # strides for BT: BT is [K, N] (we'll allocate contiguous, stride_btn=1, stride_btk=N)
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 2D grid: tiles over N and K
    pid_n = tl.program_id(0)  # along N (columns of B, become rows of BT)
    pid_k = tl.program_id(1)  # along K (rows of B, become columns of BT)

    n_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # N tile indices
    k_offsets = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)  # K tile indices

    # Compute pointers for source B[n, k] and destination BT[k, n]
    b_ptrs = B_ptr + n_offsets[None, :] * stride_bn + k_offsets[:, None] * stride_bk
    bt_ptrs = BT_ptr + k_offsets[:, None] * stride_btk + n_offsets[None, :] * stride_btn

    # Masks for boundaries
    mask = (n_offsets[None, :] < N) & (k_offsets[:, None] < K)

    # Load and store: since we are copying, no math needed
    vals = tl.load(b_ptrs, mask=mask, other=0.0)
    tl.store(bt_ptrs, vals, mask=mask)


def _transpose_b_triton(B: torch.Tensor) -> torch.Tensor:
    """
    Compute BT = B.T using Triton. B is [K, N], BT is [K, N] with BT[k, n] = B[n, k].
    """
    assert B.ndim == 2, "B must be 2D"
    K, N = B.shape
    # Allocate output BT (we'll use [K, N] layout, but logically it's the transpose)
    BT = torch.empty((K, N), device=B.device, dtype=B.dtype)
    # Make sure we use contiguous strides; Triton expects row-major addressing.
    # For a contiguous BT of shape [K, N], stride_btn = 1, stride_btk = N.
    stride_btn = BT.stride(1)  # typically 1
    stride_btk = BT.stride(0)  # typically N
    stride_bk = B.stride(1)    # for B [K, N], typically 1
    stride_bn = B.stride(0)    # typically N

    # Choose conservative block sizes to avoid resource issues and ensure correctness.
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(K, BLOCK_K))

    b_transpose_kernel[grid](
        B, BT,
        K, N,
        stride_bk, stride_bn,
        stride_btk, stride_btn,
        BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=2, num_stages=1,
    )
    return BT


class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # Original Model.forward calls run(A, B) where C = A @ B.T
        if len(args) != 2:
            raise ValueError("ModelNew.forward expects two tensors: A and B.")
        A, B = args

        if not A.is_cuda:
            A = A.cuda(non_blocking=True)
        if not B.is_cuda:
            B = B.cuda(non_blocking=True)
        A = A.contiguous()
        B = B.contiguous()

        # Compute B_T using Triton (correct and simple)
        BT = _transpose_b_triton(B)

        # Compute C = A @ B_T using PyTorch matmul (guarantees correctness)
        C = torch.matmul(A, BT)
        return C
