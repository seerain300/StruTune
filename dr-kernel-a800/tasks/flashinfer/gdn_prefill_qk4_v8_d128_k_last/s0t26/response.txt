import torch
import triton
import triton.language as tl


@triton.jit
def compute_g_and_beta(a_ptr, dt_bias_ptr, A_log_ptr, b_ptr, g_ptr, beta_ptr, T, V):
    """
    Compute g[t, v] = exp(-exp(A_log[v]) * softplus(a[t, v] + dt_bias[v])) and
    beta[t, v] = sigmoid(b[t, v]) for all t in [0, T) and v in [0, V), writing to g_ptr and beta_ptr.
    g_ptr, beta_ptr: [T*V] float32
    """
    # One program per token t
    t = tl.program_id(0)
    for v in range(0, V):
        a_val = tl.load(a_ptr + t * V + v).to(tl.float32)
        dt_val = tl.load(dt_bias_ptr + v).to(tl.float32)
        A_val = tl.load(A_log_ptr + v).to(tl.float32)
        # softplus(x) = log(1 + exp(x))
        sp = tl.log(1.0 + tl.exp(a_val + dt_val))
        # g = exp(-exp(A_log[v]) * softplus(a + dt))
        g_val = tl.exp(-tl.exp(A_val) * sp)
        tl.store(g_ptr + t * V + v, g_val)
        # beta = sigmoid(b)
        b_val = tl.load(b_ptr + t * V + v).to(tl.float32)
        beta_val = 1.0 / (1.0 + tl.exp(-b_val))
        tl.store(beta_ptr + t * V + v, beta_val)


@triton.jit
def update_state(q_ptr, k_ptr, v_ptr, state_old_ptr, g_ptr, beta_ptr,
                 new_state_ptr,
                 T, H, V, K, t, seq_idx):
    """
    Update state per token t within sequence block seq_idx:
    For each h in [0, H), v in [0, V):
      old_v[h, :] = sum_j k[t, h, j] * state_old[h, v, j]
      new_v[h, :] = beta[v] * v[t, v, :] + (1 - beta[v]) * old_v[h, :]
      state_remove[h, :] = sum_j k[t, h, j] * old_v[h, j]
      state_update[h, :] = sum_j k[t, h, j] * new_v[h, j]
      state_new[h, v, :] = g[h, v] * state_old[h, v, :] - state_remove[h, :] + state_update[h, :]
    new_state_ptr points to a [H, V, K] contiguous buffer for this seq_idx.
    """
    # Fixed loop bounds to avoid dynamic indexing
    for h in range(0, H):
        for v_i in range(0, V):
            # Load g and beta scalars
            g_idx = h * V + v_i
            g_val = tl.load(g_ptr + g_idx).to(tl.float32)
            beta_val = tl.load(beta_ptr + g_idx).to(tl.float32)

            # Initialize vectors
            old_v = tl.zeros((K,), dtype=tl.float32)
            state_new = tl.zeros((K,), dtype=tl.float32)

            # Compute old_v = k @ state_old[h, v_i, :]
            # state_old layout: contiguous [H, V, K], item (h, v, k) offset = h*V*K + v*K + k
            state_old_offset = h * V * K + v_i * K
            for j in range(0, K):
                # k[t, h, j] offset: t*H*K + h*K + j
                k_offset = t * H * K + h * K + j
                kj = tl.load(k_ptr + k_offset).to(tl.float32)
                # state_old[h, v_i, j] at offset state_old_offset + j
                svj = tl.load(state_old_ptr + state_old_offset + j).to(tl.float32)
                old_v[j] = kj * svj

            # v[t, v_i, :] layout: contiguous [T, V, K], offset = t*V*K + v_i*K + k
            v_offset = t * V * K + v_i * K
            new_v = beta_val * old_v  # will be overwritten; keep vector for later use

            # new_v[h, :] = beta * v[t, v_i, :] + (1 - beta) * old_v[h, :]
            # since old_v was beta*old_v above, but we need actual v vector
            # Compute v vector explicitly
            for j in range(0, K):
                vj = tl.load(v_ptr + v_offset + j).to(tl.float32)
                # new_v[j] = beta * vj + (1 - beta) * old_v[j]
                new_v[j] = beta_val * vj + (1.0 - beta_val) * old_v[j]

            # Compute state_remove[h, :] = sum_j k[t, h, j] * old_v[j]
            state_remove = tl.zeros((K,), dtype=tl.float32)
            for j in range(0, K):
                kj = tl.load(k_ptr + k_offset).to(tl.float32)
                state_remove[j] = kj * old_v[j]

            # Compute state_update[h, :] = sum_j k[t, h, j] * new_v[j]
            state_update = tl.zeros((K,), dtype=tl.float32)
            for j in range(0, K):
                kj = tl.load(k_ptr + k_offset).to(tl.float32)
                state_update[j] = kj * new_v[j]

            # Compute state_new[h, v_i, :] = g * state_old[h, v_i, :] - state_remove + state_update
            # Load state_old[h, v_i, :] again explicitly
            state_old_vec = tl.zeros((K,), dtype=tl.float32)
            for j in range(0, K):
                svj = tl.load(state_old_ptr + state_old_offset + j).to(tl.float32)
                state_old_vec[j] = svj

            state_new_vec = g_val * state_old_vec - state_remove + state_update

            # Store state_new to new_state[h, v_i, :]
            # new_state layout: contiguous [H, V, K], item (h, v, k) offset = h*V*K + v*K + k
            new_state_offset = h * V * K + v_i * K
            for j in range(0, K):
                tl.store(new_state_ptr + new_state_offset + j, state_new_vec[j])

def compute_output(q_ptr, new_state_ptr, out_ptr, scale, T, H, V, K):
    """
    For each token t in [0, T) and each head h in [0, H),
    compute o[t, h, :] = scale * q[t, h, :] @ new_state[H, V, K] via explicit K-loop reduction.
    new_state_ptr points to a [H, V, K] contiguous block; we assume it is the last sequence block.
    out_ptr: [T, H, K] float32
    """
    t = tl.program_id(0)
    for h in range(0, H):
        # q[t, h, :] layout: contiguous [T, H, K], offset = t*H*K + h*K + k
        q_vec = tl.zeros((K,), dtype=tl.float32)
        for j in range(0, K):
            q_offset = t * H * K + h * K + j
            qj = tl.load(q_ptr + q_offset).to(tl.float32)
            q_vec[j] = qj

        # new_state[h, v, :] for all v via loop V, but we only need sum over v of scale * q_vec @ new_state[h, v, :]
        # Implement: iterate over v and accumulate into out[t, h, :]

            new_state_offset = h * V * K + v_i * K
            state_vec = tl.zeros((K,), dtype=tl.float32)

                svj = tl.load(new_state_ptr + new_state_offset + j).to(tl.float32)
