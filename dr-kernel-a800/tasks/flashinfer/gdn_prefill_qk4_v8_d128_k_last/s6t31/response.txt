    # Implement Triton per_n_kT_vec kernel with 2D grid:
    @triton.jit
    def per_n_kT_vec_2d(k_row_ptr, vec_ptr, per_n_ptr, N: tl.constexpr, K: tl.constexpr):
        """
        2D grid: axis0 = t_h in [0, T*H), axis1 = n in [0, N).
        For each (t_h, n), compute scalar = sum_{kk=0..K-1} k_row[t_h, kk] * vec[t_h, kk]
        and store into per_n_ptr[n].
        """
        t_h = tl.program_id(0)
        n = tl.program_id(1)
        acc = 0.0
        for kk in range(0, K):
            k_val = tl.load(k_row_ptr + t_h * K + kk)
            vec_val = tl.load(vec_ptr + t_h * K + kk)
            acc += k_val * vec_val
        tl.store(per_n_ptr + n, acc)
