import torch
import math
import triton
import triton.language as tl


# Triton kernel: compute g and beta for all (t, hv)
# Inputs:
#   a_ptr: [T, HV] bfloat16
#   dt_bias_ptr: [HV] float32
#   A_log_ptr: [HV] float32
#   b_ptr: [T, HV] bfloat16
# Outputs:
#   g_ptr: [T, HV] float32
#   beta_ptr: [T, HV] float32
@triton.jit
def _compute_g_beta_kernel(
    a_ptr, dt_bias_ptr, A_log_ptr, b_ptr,
    g_ptr, beta_ptr,
    T: tl.int32, HV: tl.int32
):
    pid = tl.program_id(0)  # over T*HV
    t = pid // HV
    hv = pid % HV
    if t >= T or hv >= HV:
        return
    # Load scalars
    a_val = tl.load(a_ptr + t * HV + hv)
    dt_bias_val = tl.load(dt_bias_ptr + hv)
    A_log_val = tl.load(A_log_ptr + hv)
    b_val = tl.load(b_ptr + t * HV + hv)

    # Compute x = a + dt_bias
    x_val = a_val.to(tl.float32) + dt_bias_val
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x_val))
    # g = exp(-exp(A_log) * softplus(x))
    g_val = tl.exp(-tl.exp(A_log_val) * sp)
    tl.store(g_ptr + t * HV + hv, g_val)

    # beta = sigmoid(b)
    beta_val = 1.0 / (1.0 + tl.exp(-b_val.to(tl.float32)))
    tl.store(beta_ptr + t * HV + hv, beta_val)


# Triton kernel: compute per-time-step output for each v across all seqs
# For each (seq, t, v), compute o_vec = scale * q_exp[t, v, :] @ state_new[:, v, :]
# Outputs: output_ptr shaped as (num_seqs, V, K), but in host we will reshape to (T, V, K).
# Here, we store per seq; in host we allocate output as (T, V, K) and copy per seq.
@triton.jit
def _output_per_seq_kernel(
    q_exp_ptr, state_new_ptr, output_ptr,
    T: tl.int32, V: tl.int32, K: tl.int32, scale: tl.float32
):
    pid_seq = tl.program_id(0)  # over num_seqs
    pid_t = tl.program_id(1)    # over T
    pid_v = tl.program_id(2)    # over V
    if pid_seq >= 1 or pid_t >= T or pid_v >= V:
        return
    # For each v, output is a vector of length K. We compute it here.
    # q_exp_ptr: [T, V, K], contiguous: linear index = t*(V*K) + v*K + kk
    # state_new_ptr: [H, V, K] = [H, K, V] in original, but here we store as [H, V, K].
    # We need to compute dot between q_row (length K) and state_new_row (length K), per v.
    dot = tl.zeros((), dtype=tl.float32)
    for kk in range(0, K):
        q_index = pid_t * (V * K) + pid_v * K + kk
        q_elem = tl.load(q_exp_ptr + q_index)
        # Load state_new row for this v: state_new[h, v, k] across h
        # We need to sum over h: state_new_ptr[pid_h * (V*K) + pid_v * K + kk]
        # Since we only need per-v row across h, we sum over h for each k:
        # Initialize sum for this kk
        sum_state = tl.zeros((), dtype=tl.float32)
        for pid_h in range(0, 4):  # H = 4
            state_index = pid_h * (V * K) + pid_v * K + kk
            sum_state += tl.load(state_new_ptr + state_index)
        dot += q_elem * sum_state
    o_elem = scale * dot
    # Store output: [num_seqs, V, K], contiguous -> index = pid_seq*(V*K) + pid_v*K + kk
    for kk in range(0, K):
        out_index = pid_seq * (V * K) + pid_v * K + kk
        tl.store(output_ptr + out_index, o_elem.to(tl.bfloat16))

        output = torch.empty((T, V, K), dtype=torch.bfloat16, device=q.device)
