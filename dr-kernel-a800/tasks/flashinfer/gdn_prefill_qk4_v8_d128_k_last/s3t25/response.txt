        acc += b_k * a_ki

# Triton elementwise kernels
@triton.jit
def _softplus_vector(x_ptr, out_ptr, N: tl.constexpr):
    # Computes softplus(x) = log(1 + exp(x)) elementwise for N elements
    i = tl.arange(0, N)
    x = tl.load(x_ptr + i, mask=i < N, other=0.0)
    y = tl.log(1.0 + tl.exp(x))
    tl.store(out_ptr + i, y, mask=i < N)

def _sigmoid_vector(x_ptr, out_ptr, N: tl.constexpr):
    # Computes sigmoid(x) = 1 / (1 + exp(-x)) elementwise for N elements
    i = tl.arange(0, N)
    x = tl.load(x_ptr + i, mask=i < N, other=0.0)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(out_ptr + i, y, mask=i < N)

# B_ptr: [K] vector

# C_ptr: [V] output vector

# Triton dot: dot(x, y) -> scalar
@triton.jit
def _dot_scalar(x_ptr, y_ptr, out_ptr, K: tl.constexpr):
    s = 0.0
    for k in range(0, K):
        xv = tl.load(x_ptr + k)
        yv = tl.load(y_ptr + k)
        s += xv * yv
    tl.store(out_ptr, s)


# Triton add scalar to matrix: adds alpha to all elements of A[K, V] and writes to out[K, V]
@triton.jit
def _add_scalar_to_matrix(A_ptr, out_ptr, K: tl.constexpr, V: tl.constexpr,
                          stride_Ak: tl.constexpr, stride_Av: tl.constexpr,
                          alpha: tl.float32):
    k_offsets = tl.arange(0, K)[:, None]  # [K, 1]
    v_offsets = tl.arange(0, V)[None, :]  # [1, V]
    idx = k_offsets * stride_Ak + v_offsets * stride_Av  # [K, V]
    vals = tl.load(A_ptr + idx)
    vals = vals + alpha
    tl.store(out_ptr + idx, vals)


# Triton GEMV: 1xV x VxK -> 1xK (matrix-vector multiply)
# B_ptr: [V] vector
# A_ptr: [V, K] matrix, row-major contiguous
# C_ptr: [K] output vector
@triton.jit
def _gemv_1xVxK_into_1xK(B_ptr, A_ptr, C_ptr,
                         V: tl.constexpr, K: tl.constexpr,
                         stride_Av: tl.constexpr, stride_Ak: tl.constexpr):
    i = tl.arange(0, K)  # output indices in K
    acc = tl.zeros([K], dtype=tl.float32)

    for vv in range(0, V):
        b_v = tl.load(B_ptr + vv)  # scalar
        a_v = tl.load(A_ptr + vv * stride_Av + i * stride_Ak, mask=i < K, other=0.0)
        acc += b_v * a_v

    tl.store(C_ptr + i, acc, mask=i < K)

        super().__init__()
        self.head_size = 128
        self.num_q_heads = 4
        self.num_k_heads = 4
        self.num_v_heads = 8

    def forward(self, q, k, v, state, A_log, a, dt_bias, b, cu_seqlens, scale):
        # Ensure inputs are on CUDA and consistent
        assert q.is_cuda and k.is_cuda and v.is_cuda and state is not None, "Tensors must be CUDA and state provided"
        assert cu_seqlens.is_cuda, "cu_seqlens must be CUDA tensor"

        device = q.device
        total_seq_len, num_q_heads, head_size_q = q.shape
        num_v_heads = v.shape[1]
        num_k_heads = k.shape[1]
        num_sab_heads = max(num_q_heads, num_v_heads)
        num_seqs = cu_seqlens.numel() - 1

        # Compute repeated expansions for q/k (k-last) semantics
        q_exp = q.float().contiguous().repeat_interleave(num_v_heads // num_q_heads, dim=1)
        k_exp = k.float().contiguous().repeat_interleave(num_v_heads // num_k_heads, dim=1)

        # Precompute g and beta scalars (host-side torch ops, minimal)
        A_log_dev = A_log.to(device).float()
        a_dev = a.to(device).float()
        dt_bias_dev = dt_bias.to(device).float()
        b_dev = b.to(device).float()

        softplus_a_dt = torch.log1p(torch.exp(a_dev + dt_bias_dev))  # [T, H]
        g_vals = torch.exp(-torch.exp(A_log_dev) * softplus_a_dt)   # [T, H]
        beta_vals = torch.sigmoid(b_dev)                             # [T, H]

        # Process per segment
        for seq_idx in range(num_seqs):
            seq_start = int(cu_seqlens[seq_idx].item())
            seq_end = int(cu_seqlens[seq_idx + 1].item())
            seq_len = seq_end - seq_start

            if seq_len <= 0:
                continue

            # state_curr is [num_v_heads, K, K] in original; we need per-head [K, K]
            # Use the given state tensor (shape [num_seqs, num_v_heads, K, K])
            state_curr = state[seq_idx].float().contiguous()  # [num_v_heads, K, K]

                    k_vec = k_exp[abs_t, h].contiguous()  # [128]
                    v_vec = v[abs_t, h].contiguous()      # [128]

                    state_h = state_curr[h].contiguous()    # [K, K]

                    V = self.head_size

                                           K=K, V=V,
                                           stride_Ak=stride_Ak, stride_Av=stride_Av](num_warps=4)

                    beta_scalar = float(beta_vals[t, h].item())

                    # Compute state_remove and state_update scalars

                    state_update_val = torch.empty((1,), dtype=torch.float32, device=device)
                    _dot_scalar[k_vec, old_v, state_remove, K=K](num_warps=1)

                    new_v_vec = beta_scalar * v
