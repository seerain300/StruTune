import math

import triton
import triton.language as tl

@triton.jit
def _softplus_vector(x_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Computes softplus(x) = log(1 + exp(x)) elementwise over N elements.
    x_ptr: input pointer (float32)
    out_ptr: output pointer (float32)
    N: number of elements (must be 128 in our use)
    BLOCK: tile size (e.g., 128)
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.log(1.0 + tl.exp(x))
    tl.store(out_ptr + offsets, y, mask=mask)


@triton.jit
def _sigmoid_vector(x_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Computes sigmoid(x) = 1 / (1 + exp(-x)) elementwise over N elements.
    x_ptr: input pointer (float32)
    out_ptr: output pointer (float32)
    N: number of elements (must be 128 in our use)
    BLOCK: tile size (e.g., 128)
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(out_ptr + offsets, y, mask=mask)


@triton.jit
def _gemv_1xKxKxV_into_1xV(q_ptr, A_ptr, out_ptr, K: tl.constexpr, V: tl.constexpr, BLOCK: tl.constexpr):
    """
    Matrix-vector product: out[i] = sum_{k=0..K-1} q[k] * A[k, i], where A is [K, V] contiguous, q is [K] contiguous.
    N = K * V, but we iterate over K and V to avoid large N. We’ll use BLOCK tiling on V dimension.
    """
    pid = tl.program_id(axis=0)  # 1D grid; we can set axis=0 to number of V tiles, but since V is 128, we use a single program.
    # For simplicity and given V=128, we implement a single-program kernel. If V were larger, we would need a 2D loop.
    # Here we assume V=128; K=128 also assumed. We'll keep it general by looping over K and accumulating.
    acc = tl.zeros((128,), dtype=tl.float32)
    for k in range(0, K):
        qk = tl.load(q_ptr + k)
        # load row k across V
        offs_v = tl.arange(0, BLOCK)  # BLOCK=128, so V fits in one vector
        mask_v = offs_v < V
        A_row = tl.load(A_ptr + k * V + offs_v, mask=mask_v, other=0.0)
        acc += qk * A_row
    tl.store(out_ptr + tl.arange(0, 128), acc, mask=tl.arange(0, 128) < 128)


@triton.jit
def _gemv_1xVxK_into_1xK(q_ptr, A_ptr, out_ptr, V: tl.constexpr, K: tl.constexpr, BLOCK: tl.constexpr):
    """
    Matrix-vector product: out[k] = sum_{v=0..V-1} q[v] * A[v, k], where A is [V, K] contiguous, q is [V] contiguous.
    Similar to above, assume V=128, K=128. We accumulate across V and store out[k].
    """
    pid = tl.program_id(axis=0)
    acc = tl.zeros((K,), dtype=tl.float32)
    for v in range(0, V):
        qv = tl.load(q_ptr + v)
        offs_k = tl.arange(0, BLOCK)  # BLOCK=128, so K fits in one vector
        mask_k = offs_k < K
        A_col = tl.load(A_ptr + v * K + offs_k, mask=mask_k, other=0.0)
        acc += qv * A_col
    tl.store(out_ptr + tl.arange(0, K), acc, mask=tl.arange(0, K) < K)


@triton.jit
def _elementwise_mul_add_scalar(v_ptr, old_ptr, out_ptr, alpha: tl.constexpr, beta: tl.constexpr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Elementwise: out[i] = alpha * v[i] + beta * old[i], over N elements.
    v_ptr, old_ptr: input pointers (float32)
    out_ptr: output pointer (float32)
    N: number of elements (must be 128)
    alpha, beta: scalars (float32)
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N
    v = tl.load(v_ptr + offsets, mask=mask, other=0.0)
    old = tl.load(old_ptr + offsets, mask=mask, other=0.0)
    out = alpha * v + beta * old
    tl.store(out_ptr + offsets, out, mask=mask)


@triton.jit
def _dot_scalar(x_ptr, y_ptr, out_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Computes scalar = sum_i x[i] * y[i] over N elements.
    x_ptr, y_ptr: input pointers (float32)
    out_ptr: output pointer (float32), single element
    N: number of elements (must be 128 for our use)
    """
    pid = tl.program_id(axis=0)  # single program
    offsets = tl.arange(0, BLOCK)
    mask = offsets < N
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0)
    y = tl.load(y_ptr + offsets, mask=mask, other=0.0)
    prod = x * y
    # reduce within vector
    s = tl.sum(prod, axis=0)
    tl.store(out_ptr, s)
