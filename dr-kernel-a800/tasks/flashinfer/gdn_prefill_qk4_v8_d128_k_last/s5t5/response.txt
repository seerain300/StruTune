        k_row = tl.load(k_exp_ptr + k_index)

        # Load state_HKV vector for this v: state_ptr is [num_seqs, H, V, K], transposed into [H, K, V] conceptually.
        # We need to load state[pid_seq, h, pid_v, k] for h in [0..H-1], k in [0..K-1], then for each h, we have K elements.
        # Build state_vec[h, :] = state[pid_seq, h, pid_v, :]
        state_vec = tl.zeros([H, K], dtype=tl.float32)
        for h in range(0, H):
            base = pid_seq * (H * V * K) + h * (V * K) + pid_v * K
            # load k elements
            state_vec[h, :] = tl.load(state_ptr + base + tl.arange(0, K))

        # old_v = k @ state_vec (reduce over K)
        old_v = tl.zeros([1], dtype=tl.float32)
        for kk in range(0, K):
            old_v += k_row[kk] * state_vec[:, kk]
        old_v = old_v[0]  # scalar

        # v_row = v[t, v, :] which is a 1xK vector; we need only its K elements to compute new_v
        # v_ptr is [T, V, K]; linearized as T*(V*K) + v*K + kk
        v_row = tl.zeros([K], dtype=tl.float32)
        v_index = t * (V * K) + pid_v * K + tl.arange(0, K)
        v_row = tl.load(v + v_index)  # v is passed as pointer; but Triton cannot load from 'v' variable here. We need to pass v_ptr.

        # new_v = beta_t * v_row + (1-beta_t) * old_v
        new_v_vec = tl.zeros([K], dtype=tl.float32)
        for kk in range(0, K):
            new_v_vec[kk] = beta_t * v_row[kk] + (1.0 - beta_t) * old_v

        # remove = k @ state_vec
        remove = tl.zeros([1], dtype=tl.float32)
        for kk in range(0, K):
            remove += k_row[kk] * state_vec[:, kk]
        remove = remove[0]  # scalar

        # update = k @ new_v_vec
        update = tl.zeros([1], dtype=tl.float32)
        for kk in range(0, K):
            update += k_row[kk] * new_v_vec[kk]
        update = update[0]  # scalar

        # new_state_HKV per v across K: new_state_vec += g_t * state_vec - remove + update
        # Here we update new_state_vec for all h's K vectors:
        # We need to apply scalar remove/update to each h's K vector. So for each h, new_state_vec[h, :] =
        # state_vec[h, :] * g_t - remove + update
        # Then store new_state_vec into new_state_ptr for this seq and v.
        # But Triton stores require pointer arithmetic. We'll store per h.
        # new_state_ptr is [num_seqs, H, V, K], linearized as num_seqs * (H * V * K) + h * (V * K) + v * K + kk
        for h in range(0, H):
            base = pid_seq * (H * V * K) + h * (V * K) + pid_v * K
            # compute new vector for this h: state_vec[h, :] * g_t - remove + update
            new_h_vec = state_vec[h, :] * g_t - remove + update
            # store new_h_vec
            for kk in range(0, K):
                tl.store(new_state_ptr + base + kk, new_h_vec[kk])

        # Compute output o_vec for this t and v: o_vec = scale * (q_exp[t, v, :] @ new_state_HKV)
        # q_exp_ptr: [T, V, K] linearized as T*(V*K) + v*K + kk
        q_row = tl.zeros([K], dtype=tl.float32)
        q_index = t * (V * K) + pid_v * K + tl.arange(0, K)
        q_row = tl.load(q_exp_ptr + q_index)
        # new_state_HKV vector for this v is stored in new_state_ptr; load it as new_state_vec2
        new_state_vec2 = tl.zeros([K], dtype=tl.float32)
        # load all h's K elements for this v
        for h in range(0, H):
            base2 = pid_seq * (H * V * K) + h * (V * K) + pid_v * K
            # load K elements into new_state_vec2
            new_state_vec2 += tl.load(new_state_ptr + base2 + tl.arange(0, K))
        # Now dot = sum(q_row * new_state_vec2)
        dot = tl.zeros([1], dtype=tl.float32)
        for kk in range(0, K):
            dot += q_row[kk] * new_state_vec2[kk]
        o_elem = scale * dot[0]
        # Store into output_ptr: [num_seqs, V, K], linearized as num_seqs * (V*K) + v*K + kk
        out_index = pid_seq * (V * K) + pid_v * K + tl.arange(0, K)
        tl.store(output_ptr + out_index, o_elem.to(tl.bfloat16))
