Root cause and fix plan

- The evaluator reported RUNTIME_ERROR: “must be real number, not NoneType” and this happened in multiple workloads. In my previous Triton kernels, I used tl.exp(tensor) for gating. In Triton, tl.exp only accepts floating-point tensors; passing a None-like or non-float can cause this error. In one case, A_log was a 0-dim float tensor, but in another, it could have become None or a non-float due to incorrect dtype handling. The safest fix is:
  - Ensure all Triton inputs are explicitly cast to float32 before calling tl.exp, tl.log, tl.sigmoid, etc.
  - Avoid any chance of None or non-float values entering Triton kernels. This includes A_log, a, dt_bias, b, q, k, v, state. If the input is None (unlikely for these tensors), handle it; otherwise cast to float32 and feed to Triton.
- Secondary correctness issue:
  - My previous version computed q@state_new using a Triton GEMM, but in the original, q is [T, 4, 128], not repeated. To exactly match the original behavior, forward should not repeat q. I will remove q_exp and compute q@state using the original q[h] vector for each v head. The heavy cost per time step is the q@state per head, which we can do in Triton. The update terms (old_v_j[h], update_j[h]) are dot products over 128, best done in Triton as well. The original code updates state in-place; Triton cannot return updated state easily, so I will update state using torch indexing with scalars computed by Triton kernels. This avoids torch.mm/einsum in forward while keeping state updates exact.
- Performance strategy:
  - Heavy per-step compute is q@state (4 heads) and dot products (4 per head, 8 v heads). Implement these in Triton with simple 1D tiling over K=128 and N=128. For q@state, accumulate across K tiles into an N-length vector; for dot products, accumulate across K into a scalar. We’ll call these Triton kernels per t, j, h. This is not as fast as PyTorch’s optimized mm, but it satisfies the “TRITON ONLY” requirement and avoids mm/einsum in forward.

import torch

# Triton kernel to compute softplus(x) and exp(...) for gating: writes g, beta
@triton.jit
def _compute_g_beta_kernel(A_log_ptr, a_ptr, dt_bias_ptr, b_ptr,
                           g_ptr, beta_ptr,
                           T: tl.constexpr, H: tl.constexpr):
    # 2D grid over (t, j)
    t = tl.program_id(0)
    j = tl.program_id(1)
    if t >= T or j >= H:
        return

    # Load scalars as float32
    a_val = tl.load(a_ptr + t * H + j, mask=True, other=0.0)
    dt = tl.load(dt_bias_ptr + j, mask=True, other=0.0)
    b_val = tl.load(b_ptr + t * H + j, mask=True, other=0.0)
    A_log_val = tl.load(A_log_ptr + j, mask=True, other=0.0)

    a_val = tl.cast(a_val, tl.float32)
    dt = tl.cast(dt, tl.float32)
    b_val = tl.cast(b_val, tl.float32)
    A_log_val = tl.cast(A_log_val, tl.float32)

    x = a_val + dt
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x))
    # g = exp(-exp(A_log) * softplus(a + dt_bias))
    g_val = tl.exp(-tl.exp(A_log_val) * sp)
    # beta = sigmoid(b) = 1 / (1 + exp(-b))
    beta_val = 1.0 / (1.0 + tl.exp(-b_val))

    tl.store(g_ptr + t * H + j, g_val)
    tl.store(beta_ptr + t * H + j, beta_val)


# Triton kernel: compute q_row @ B where q_row is [1, K], B is [K, N], output is [1, N] (store as 1D length N)
@triton.jit
def _q_mm_row_kernel(q_ptr, B_ptr, out_ptr,
                     T: tl.constexpr, H_q: tl.constexpr, K: tl.constexpr, N: tl.constexpr):
    # One program per (t, j, h)
    t = tl.program_id(0)
    j = tl.program_id(1)
    h = tl.program_id(2)
    if t >= T or j >= H_q or h >= 1:
        return

    # q_row is [K], state is [N, N], output [N]
    # Build K index
    offs_k = tl.arange(0, K)  # K=128
    # Load q_row as float32
    q_row = tl.load(q_ptr + t * (H_q * K) + h * K + offs_k)
    q_row = tl.cast(q_row, tl.float32)

    # Load B as [K, N] contiguous
    # For B, indexing: row k, col n => B_ptr + k*N + n
    offs_n = tl.arange(0, N)
    B_mat = tl.load(B_ptr + offs_k[:, None] * N + offs_n[None, :])
    B_mat = tl.cast(B_mat, tl.float32)

    # Accumulate: out = sum_k q_row[k] * B_mat[k, :]
    out_vec = tl.zeros((N,), dtype=tl.float32)
    for k in range(K):
        out_vec += q_row[k] * B_mat[k, :]

    # Store out_vec as contiguous 1D length N
    tl.store(out_ptr + t * (H_q * N) + j * N + offs_n, out_vec)


# Triton kernel: compute dot(q_row[h], state[h, :]) -> scalar
@triton.jit
def _dot_q_state_kernel(q_ptr, state_ptr, out_ptr,
                         T: tl.constexpr, H_q: tl.constexpr, K: tl.constexpr):
    # One program per (t, j, h) computing dot(q_row[h], state[h, :])
    t = tl.program_id(0)
    j = tl.program_id(1)
    h = tl.program_id(2)
    if t >= T or j >= H_q or h >= 1:
        return

    offs_k = tl.arange(0, K)
    q_row = tl.load(q_ptr + t * (H_q * K) + h * K + offs_k)
    q_row = tl.cast(q_row, tl.float32)

    # state[h, :] is [K], indexing: state_ptr + h*K + k
    state_vec = tl.load(state_ptr + h * K + offs_k, mask=True, other=0.0)
    state_vec = tl.cast(state_vec, tl.float32)

    dot_val = tl.zeros((), dtype=tl.float32)
    for k in range(K):
        dot_val += q_row[k] * state_vec[k]

    tl.store(out_ptr + t * (H_q * 1) + j * 1 + h * 1, dot_val)  # store scalar per (t, j, h)

@triton.jit
def _dot_k_state_kernel(k_ptr, state_ptr, out_ptr,

    k_row = tl.cast(k_row, tl.float32)

    state_vec = tl.load(state_ptr + h * K + offs_k_vec, mask=True, other=0.0)

        dot_val += k_row[kk] * state_vec[kk]

    # Store scalar per (t, k, h)
    tl.store(out_ptr + t * (H_k * H_q) + k * H_q + h, dot_val)
