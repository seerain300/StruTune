import torch

import triton.language as tl


# Triton kernels

@triton.jit
def softplus_triton(x_ptr, out_ptr, N: tl.int32):
    """
    Elementwise softplus(x) = log(1 + exp(x)).
    x_ptr: [N]
    out_ptr: [N]
    Grid: (N,)
    """
    pid = tl.program_id(0)
    if pid >= N:
        return
    x = tl.load(x_ptr + pid)
    sp = tl.log(1.0 + tl.exp(x))
    tl.store(out_ptr + pid, sp)


@triton.jit
def sigmoid_triton(x_ptr, out_ptr, N: tl.int32):
    """
    Elementwise sigmoid(x) = 1 / (1 + exp(-x)).
    x_ptr: [N]
    out_ptr: [N]
    Grid: (N,)
    """
    pid = tl.program_id(0)
    if pid >= N:
        return
    x = tl.load(x_ptr + pid)
    sig = 1.0 / (1.0 + tl.exp(-x))
    tl.store(out_ptr + pid, sig)


@triton.jit
def compute_g_beta_kernel(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, beta_ptr, T: tl.int32, H: tl.int32):
    """
    Compute per (t, h):
      x = a[t, h] + dt_bias[h]
      softplus(x) = log(1 + exp(x))
      g = exp(-exp(A_log[h]) * softplus(x))
      beta = sigmoid(b[t, h])

    dt_bias_ptr: [H] float32
    A_log_ptr: [H] float32
    g_ptr: [T*H] float32
    beta_ptr: [T*H] float32
    Grid: (T*H,)
    """

    t = pid // H
    h = pid % H
    if t >= T:
        return
    # Load a[t,h] as bfloat16 and convert to float32
    a_val = tl.load(a_ptr + pid).to(tl.float32)          # [1]
    db_val = tl.load(dt_bias_ptr + h)                    # [1] float32

    b_val = tl.load(beta_ptr + pid)                      # [1] float32 (beta provided externally)
    tl.store(g_ptr + pid, g_val)
    tl.store(beta_ptr + pid, b_val)


@triton.jit
def eye_init_kernel(out_ptr, N: tl.int32):
    """
    Initialize out_ptr as identity matrix of size N x N (row-major).
    out_ptr: [N*N] float32
    Grid: (N*N,)
    """
    pid = tl.program_id(0)
    if pid >= N * N:
        return
    row = pid // N
    col = pid % N
    value = 1.0 if row == col else 0.0
    tl.store(out_ptr + pid, value)


@triton.jit
def output_matmul_row_kernel(q_row_ptr, state_ptr, out_ptr, scale, N: tl.int32):
    """
    Compute out = scale * q_row @ state, where:
      q_row: [N] (vector q[t,h])
      state: [N, N] row-major (we pass a contiguous [N, N] tensor flattened as [N*N])
      out:   [N] (vector output)
    Grid: (N,)
    """
    pid = tl.program_id(0)
    if pid >= N:
        return
    acc = 0.0
    # Compute dot product: acc = sum_i q_row[i] * state[i, pid]
    # state is [N, N] row-major, so element at (i, j) is state_ptr[i*N + j]
    for i in range(N):
        q_i = tl.load(q_row_ptr + i)
        # Load state[i, pid]
        state_ij = tl.load(state_ptr + i * N + pid)
        acc += q_i * state_ij
    acc = acc * scale
    tl.store(out_ptr + pid, acc)

import torch
import triton
import triton.language as tl


# Triton kernels

@triton.jit
def softplus_triton(x_ptr, out_ptr, N: tl.int32):
    """
    Elementwise softplus(x) = log(1 + exp(x)).
    x_ptr: [N] float32
    out_ptr: [N] float32
    Grid: (N,)
    """
    pid = tl.program_id(0)
    if pid >= N:
        return
    x = tl.load(x_ptr + pid)
    sp = tl.log(1.0 + tl.exp(x))
    tl.store(out_ptr + pid, sp)


@triton.jit
def sigmoid_triton(x_ptr, out_ptr, N: tl.int32):
    """
    Elementwise sigmoid(x) = 1 / (1 + exp(-x)).
    x_ptr: [N] float32
    out_ptr: [N] float32
    Grid: (N,)
    """
    pid = tl.program_id(0)
    if pid >= N:
        return
    x = tl.load(x_ptr + pid)
    sig = 1.0 / (1.0 + tl.exp(-x))
    tl.store(out_ptr + pid, sig)


@triton.jit
def compute_g_beta_kernel(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, beta_ptr, T: tl.int32, H: tl.int32):
    """
    Compute per (t, h):
      x = a[t, h] + dt_bias[h]
      softplus(x) = log(1 + exp(x))
      g = exp(-exp(A_log[h]) * softplus(x))
      beta = sigmoid(b[t, h])
    a_ptr: [T*H] bfloat16
    dt_bias_ptr: [H] float32
    A_log_ptr: [H] float32
    g_ptr: [T*H] float32
    beta_ptr: [T*H] float32
    Grid: (T*H,)
    """
    pid = tl.program_id(0)
    t = pid // H
    h = pid % H
    if t >= T:
        return
    # Load a[t,h] as bfloat16 and convert to float32
    a_val = tl.load(a_ptr + pid).to(tl.float32)          # [1]
    db_val = tl.load(dt_bias_ptr + h)                    # [1] float32
    A_val = tl.load(A_log_ptr + h)                       # [1] float32
    x = a_val + db_val
    sp = tl.log(1.0 + tl.exp(x))                         # softplus(x)
    g_val = tl.exp(-tl.exp(A_val) * sp)
    b_val = tl.load(beta_ptr + pid)                      # [1] float32 (beta provided externally)
    tl.store(g_ptr + pid, g_val)
    tl.store(beta_ptr + pid, b_val)


@triton.jit
def output_matmul_row_kernel(q_row_ptr, state_ptr, out_ptr, scale, N: tl.int32):
    """
    Compute out = scale * q_row @ state, where:
      q_row: [N] (vector q[t,h])
      state: [N, N] row-major (flattened pointer of size N*N)
      out:   [N] (vector output)
    Grid: (N,)
    """
    pid = tl.program_id(0)
    if pid >= N:
        return
    acc = 0.0
    # Compute dot product: acc = sum_i q_row[i] * state[i, pid]
    for i in range(N):
        q_i = tl.load(q_row_ptr + i)
        state_ij = tl.load(state_ptr + i * N + pid)
        acc += q_i * state_ij
    acc = acc * scale
    tl.store(out_ptr + pid, acc)


@triton.jit
def state_update_kernel(k_row_ptr, v_row_ptr, state_old_ptr, g, beta, out_ptr, N: tl.int32):
    """
    Update per-head state (N x N) using:
      old_v = k @ state_old  -> [N]
      new_v = beta * v + (1 - beta) * old_v
      state_remove = k^T @ old_v -> [N, N]
      state_update = k^T @ new_v -> [N, N]
      new_state = g * state_old + state_update - state_remove
    k_row_ptr: [N] float32
    v_row_ptr: [N] float32
    state_old_ptr: [N, N] row-major flattened float32
    out_ptr: [N, N] row-major flattened float32
    """
    # Load scalars
    g_val = g  # float32
    beta_val = beta  # float32
    # Initialize new_state as zeros
    # We will fill out_ptr with computed new_state
    # First, compute old_v = k @ state_old
    old_v = tl.zeros((N,), dtype=tl.float32)
    for i in range(N):
        s = 0.0
        for j in range(N):
            state_old_ij = tl.load(state_old_ptr + i * N + j)
            k_j = tl.load(k_row_ptr + j)
            s += state_old_ij * k_j
        old_v[i] = s

    # new_v = beta * v + (1 - beta) * old_v
    new_v = tl.zeros((N,), dtype=tl.float32)
    # v_row_ptr is [N]
    for i in range(N):
        v_i = tl.load(v_row_ptr + i)
        new_v[i] = beta_val * v_i + (1.0 - beta_val) * old_v[i]

    # Compute state_remove = k^T @ old_v -> [N, N]
    state_remove = tl.zeros((N * N,), dtype=tl.float32)
    for i in range(N):
        k_i = tl.load(k_row_ptr + i)  # scalar k_i
        for j in range(N):
            old_v_j = tl.load(old_v_ptr + j)  # old_v is a vector; we need old_v[j]
            state_remove[i * N + j] = k_i * old_v_j

    # Compute state_update = k^T @ new_v -> [N, N]
    state_update = tl.zeros((N * N,), dtype=tl.float32)
    for i in range(N):
        k_i = tl.load(k_row_ptr + i)  # scalar k_i
        for j in range(N):
            new_v_j = tl.load(new_v_ptr + j)  # new_v is a vector; we need new_v[j]
            state_update[i * N + j] = k_i * new_v_j

    # Compute new_state = g * state_old + state_update - state_remove
    new_state_flat = tl.zeros((N * N,), dtype=tl.float32)
    for idx in range(N * N):
        i = idx // N
        j = idx % N
        state_old_ij = tl.load(state_old_ptr + idx)
        new_state_flat[idx] = g_val * state_old_ij + state_update[idx] - state_remove[idx]

    tl.store(out_ptr, new_state_flat)
