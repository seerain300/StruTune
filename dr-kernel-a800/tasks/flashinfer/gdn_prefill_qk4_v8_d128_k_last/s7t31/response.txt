import torch
import math
import triton
import triton.language as tl


@triton.jit
def softplus_exp_and_g(A_log_ptr, a_ptr, dt_bias_ptr, g_ptr, T: tl.constexpr, H_v: tl.constexpr):
    """
    Compute g[t, j] = exp(-exp(A_log[j]) * softplus(a[t, j] + dt_bias[j])) for t in [0..T-1], j in [0..H_v-1]
    Stores results into g_ptr as [T, H_v].
    """
    for t in range(T):
        for j in range(H_v):
            A = tl.load(A_log_ptr + j)
            x = tl.load(a_ptr + t * H_v + j) + tl.load(dt_bias_ptr + j)
            sp = tl.log(1.0 + tl.exp(x))  # softplus(x) = log(1 + exp(x))
            g = tl.exp(-tl.exp(A) * sp)
            tl.store(g_ptr + t * H_v + j, g)


@triton.jit
def sigmoid_and_beta(b_ptr, beta_ptr, T: tl.constexpr, H_v: tl.constexpr):
    """
    Compute beta[t, j] = sigmoid(b[t, j]) = 1 / (1 + exp(-b[t, j])) for t in [0..T-1], j in [0..H_v-1]
    Stores results into beta_ptr as [T, H_v].
    """
    for t in range(T):
        for j in range(H_v):
            b = tl.load(b_ptr + t * H_v + j)
            sig = 1.0 / (1.0 + tl.exp(-b))
            tl.store(beta_ptr + t * H_v + j, sig)


@triton.jit
def mm_row_row(A_ptr, B_ptr, C_ptr, K: tl.constexpr):
    """
    Compute C = A @ B where:
      A is [1, K], B is [K, N], C is [1, N]
    Here N=K=128. This kernel multiplies a single row A by a matrix B.
    """
    # A row vector A[1, K], B matrix B[K, N], result vector C[1, N]
    offs_n = tl.arange(0, 128)  # N=128
    acc = tl.zeros((128,), dtype=tl.float32)
    for k in range(0, 128, 32):
        offs_k = k + tl.arange(0, 32)
        a = tl.load(A_ptr + offs_k)  # [32]
        b = tl.load(B_ptr + offs_k[:, None] * 128 + offs_n[None, :])  # [32, 128]
        # b now has shape [32, 128]
        # Accumulate: acc += sum(a[:, None] * b, axis=0)
        prod = a[:, None] * b
        acc += tl.sum(prod, axis=0)
    tl.store(C_ptr + offs_n, acc)
