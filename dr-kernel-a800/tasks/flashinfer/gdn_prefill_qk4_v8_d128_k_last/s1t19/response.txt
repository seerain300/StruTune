```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def _compute_g_beta_kernel(
    A_log_ptr,   # [H_v] float32
    a_ptr,       # [B, H_v] float32
    dt_bias_ptr, # [H_v] float32
    b_ptr,       # [B, H_v] float32
    g_ptr,       # [B, H_v] float32
    beta_ptr,    # [B, H_v] float32
    B: tl.constexpr,
    H_v: tl.constexpr,
):
    # 2D grid over (token b, head hv)
    b_idx = tl.program_id(0)
    hv_idx = tl.program_id(1)

    # Compute g and beta
    x = tl.load(a_ptr + b_idx * H_v + hv_idx) + tl.load(dt_bias_ptr + hv_idx)
    softplus_x = tl.log(1.0 + tl.exp(x))  # softplus(x) = log(1 + exp(x))
    g_val = tl.exp(-tl.exp(tl.load(A_log_ptr + hv_idx)) * softplus_x)
    beta_val = 1.0 / (1.0 + tl.exp(-tl.load(b_ptr + b_idx * H_v + hv_idx)))

    # Store results
    tl.store(g_ptr + b_idx * H_v + hv_idx, g_val)
    tl.store(beta_ptr + b_idx * H_v + hv_idx, beta_val)


@triton.jit
def _state_update_kernel(
    k_ptr,   # [B, H_v, D] float32
    v_ptr,   # [B, H_v, D] float32
    beta_ptr, # [B, H_v] float32
    g_ptr,   # [B, H_v] float32
    state_ptr,  # [H_v, D, D] float32
    B: tl.constexpr,
    H_v: tl.constexpr,
    D: tl.constexpr,
):
    t_idx = tl.program_id(0)
    hv_idx = tl.program_id(1)

    # Load k vector and v vector
    k_vec = tl.zeros((D,), dtype=tl.float32)
    v_vec = tl.zeros((D,), dtype=tl.float32)
    for i in range(0, D):
        k_vec[i] = tl.load(k_ptr + t_idx * H_v * D + hv_idx * D + i)
        v_vec[i] = tl.load(v_ptr + t_idx * H_v * D + hv_idx * D + i)

    # Load state slice
    state_HKV = tl.zeros((D, D), dtype=tl.float32)
    for i in range(0, D):
        for j in range(0, D):
            state_HKV[i, j] = tl.load(state_ptr + hv_idx * D * D + i * D + j)

    # Compute old_v = k @ state
    old_v = tl.zeros((D,), dtype=tl.float32)
    for i in range(0, D):
        row = state_HKV[i, :]  # [D]
        old_v += k_vec[i] * row

    # Load beta and g for this token/head
    beta_val = tl.load(beta_ptr + t_idx * H_v + hv_idx)
    g_val = tl.load(g_ptr + t_idx * H_v + hv_idx)

    # new_v = beta * v + (1 - beta) * old_v
    new_v = beta_val * v_vec + (1.0 - beta_val) * old_v

    # kT_old = sum_i k[i] * old_v[i], kT_newv = sum_i k[i] * new_v[i]
    kT_old = 0.0
    kT_newv = 0.0
    for i in range(0, D):
        kT_old += k_vec[i] * old_v[i]
        kT_newv += k_vec[i] * new_v[i]

    # Update state
    # state[hv, :, :] = g * state - kT_old + kT_newv
    for i in range(0, D):
        for j in range(0, D):
            state_HKV[i, j] = g_val * state_HKV[i, j] - kT_old + kT_newv

    # Store updated state
    for i in range(0, D):
        for j in range(0, D):
            tl.store(state_ptr + hv_idx * D * D + i * D + j, state_HKV[i, j])


@triton.jit
def _output_kernel(
    q_ptr,        # [B, H_q, D] float32 (we'll form q_exp by concatenation in forward)
    state_ptr,    # [H_v, D, D] float32
    out_ptr,      # [D] bfloat16
    scale,        # float32 scalar
    H_v: tl.constexpr,
    H_q: tl.constexpr,
    D: tl.constexpr,
):
    t_idx = tl.program_id(0)
    hv_idx = tl.program_id(1)

    # Form q_exp: in provided setup H_v // H_q == 2, so q_exp = concatenate(q[t, 0, :], q[t, 1, :])
    # We will load q0 and q1 vectors as [D] and concatenate.
    q0 = tl.zeros((D,), dtype=tl.float32)
    q1 = tl.zeros((D,), dtype=tl.float32)
    for i in range(0, D):
        q0[i] = tl.load(q_ptr + t_idx * H_q * D + 0 * D + i)
        q1[i] = tl.load(q_ptr + t_idx * H_q * D + 1 * D + i)
    q_exp = tl.concatenate([q0, q1])  # [2D]

    # Load state slice
    state_HKV = tl.zeros((D, D), dtype=tl.float32)
    for i in range(0, D):
        for j in range(0, D):
            state_HKV[i, j] = tl.load(state_ptr + hv_idx * D * D + i * D + j)

    # output[t, hv, :] = scale * q_exp @ state[hv, :, :]
    # We implement q_exp[:, None] @ state.HKV and take first D elements (the rest are from q1, but q_exp is length 2D and we sum across 2D).
    # Since q_exp has length 2D, we will compute the D-length output by summing over the two parts:
    # q0 @ state + q1 @ state
    out_vec = tl.zeros((D,), dtype=tl.float32)
    for i in range(0, D):
        row = state_HKV[i, :]  # [D]
        out_vec += q0[i] * row
    for i in range(0, D):
        row = state_HKV[i, :]
        out_vec += q1[i] * row

    out_vec = scale * out_vec

    # Store output (cast to bfloat16 if needed; Triton will store as fp32, but we can create bf16 buffer in host)
    # out_ptr is bfloat16; cast explicitly
    for i in range(0, D):
        tl.store(out_ptr + i, tl.astype(out_vec[i], tl.bfloat16))


class ModelNew(torch.nn.Module):
    def forward(self, q, k, v, state, A_log, a, dt_bias, b, cu_seqlens, scale):
        # Enforce the fixed setup as in the original code
        assert q.dim() == 3 and k.dim() == 3 and v.dim() == 3
        B = q.shape[0]
        H_q = q.shape[1]
        D = q.shape[2]
        H_k = k.shape[1]
        V = v.shape[1]
        assert H_k == H_q, "num_q_heads must equal num_k_heads"
        assert D == 128, "head_size must be 128"
        assert H_q == 4, "num_q_heads must be 4"
        assert H_k == 4, "num_k_heads must be 4"
        assert V == 8, "num_v_heads must be 8"
        assert k.shape == (B, H_k, D) and v.shape == (B, V, D)

        # Ensure dtypes for compute
        q_fp32 = q.to(torch.float32)
        k_fp32 = k.to(torch.float32)
        v_fp32 = v.to(torch.float32)
        a_fp32 = a.to(torch.float32)
        dt_bias_fp32 = dt_bias.to(torch.float32)
        b_fp32 = b.to(torch.float32)
        A_log_fp32 = A_log.to(torch.float32)

        # Enforce state shape as [1, 8, 128, 128] (original Model.run expects num_seqs=1)
        if state is not None:
            assert state.shape[1] == V and state.shape[2] == D and state.shape[3] == D, "state must be [*, 8, 128, 128]"
            # If num_seqs != 1, reshape to [1, 8, 128, 128]
            if state.dim() == 3:
                state = state.unsqueeze(0)
            else:
                state = state[:, :1, :, :]  # force num_seqs=1
        else:
            state = torch.zeros((1, V, D, D), device=q.device, dtype=torch.float32)

        # Compute g and beta via Triton kernel
        g = torch.empty((B, V), device=q.device, dtype=torch.float32)
        beta = torch.empty((B, V), device=q.device, dtype=torch.float32)
        _compute_g_beta_kernel[(B, V)](
            A_log_fp32, a_fp32, dt_bias_fp32, b_fp32, g, beta,
            B=B, H_v=V,
        )

        # Output buffer [B, V, D] bfloat16
        out = torch.empty((B, V, D), device=q.device, dtype=torch.bfloat16)

        # If scale is None or 0, default to 1/sqrt(D)
        if scale is None or scale == 0.0:
            scale = 1.0 / math.sqrt(D)

        # Process tokens sequentially; Triton kernels handle per-token updates
        new_state = torch.empty((V, D, D), device=q.device, dtype=torch.float32)  # we will update in-place each token
        for t in range(B):
            # Launch state_update_kernel for token t over all hv
            _state_update_kernel[(1, V)](
                k_fp32[t], v_fp32[t], beta[t], g[t], state[0],
                B=B, H_v=V, D=D,
            )
            # Write updated state
            new_state.copy_(state[0])

            # Launch output_kernel for token t over all hv
            _output_kernel[(1, V)](
                q_fp32[t], new_state, out[t],
                float(scale),
                H_v=V, H_q=H_q, D=D,
            )

        return out, new_state
```
