import torch
import math
import triton
import triton.language as tl


@triton.jit
def _q_mm(A_ptr, B_ptr, C_ptr, K: tl.constexpr):
    """
    Compute C = A @ B where:
      A is [1, K], B is [K, K], C is [1, K]
    All dims are 128 for this use case.
    """
    # A_ptr points to [1, K] contiguous row, B_ptr points to [K, K] contiguous matrix
    offs_k = tl.arange(0, K)  # columns of A and rows of B
    A = tl.load(A_ptr + offs_k)  # [K]
    B_rows = tl.load(B_ptr + offs_k * K + offs_k)  # [K, K] as list of rows
    acc = tl.zeros((K,), dtype=tl.float32)
    # Manual FMA over K
    for kk in range(K):
        acc += A[kk] * B_rows[kk]
    tl.store(C_ptr + offs_k, acc)


@triton.jit
def _reduce_dot_k(A_ptr, B_ptr, out_ptr, K: tl.constexpr):
    """
    Compute dot = sum_k A[k] * B[k] where:
      A is [K], B is [K], store result to out_ptr[0]
    """
    offs_k = tl.arange(0, K)
    A = tl.load(A_ptr + offs_k)
    B = tl.load(B_ptr + offs_k)
    acc = tl.zeros((), dtype=tl.float32)
    for kk in range(K):
        acc += A[kk] * B[kk]
    tl.store(out_ptr, acc)

import triton
import triton.language as tl


@triton.jit
def _softplus_triton(x_ptr, out_ptr, N: tl.constexpr):
    """
    Compute softplus(x) = log(1 + exp(x)) elementwise for vector of length N.
    """
    offs = tl.arange(0, N)
    x = tl.load(x_ptr + offs)
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x))
    tl.store(out_ptr + offs, sp)


@triton.jit
def _sigmoid_triton(y_ptr, out_ptr, N: tl.constexpr):
    """
    Compute sigmoid(y) = 1 / (1 + exp(-y)) elementwise for vector of length N.
    """
    offs = tl.arange(0, N)
    y = tl.load(y_ptr + offs)
    sig = 1.0 / (1.0 + tl.exp(-y))
    tl.store(out_ptr + offs, sig)


@triton.jit
def _exp_triton(x_ptr, out_ptr, N: tl.constexpr):
    """
    Compute exp(x) elementwise for vector of length N.
    """
    offs = tl.arange(0, N)
    x = tl.load(x_ptr + offs)
    ex = tl.exp(x)
    tl.store(out_ptr + offs, ex)


@triton.jit
def _reduce_dot_k_triton(A_ptr, B_ptr, out_ptr, K: tl.constexpr):
    """
    Compute dot = sum_k A[k] * B[k] where A is [K], B is [K], store result to out_ptr[0].
    """
    offs_k = tl.arange(0, K)
    A = tl.load(A_ptr + offs_k)
    B = tl.load(B_ptr + offs_k)
    acc = tl.zeros((), dtype=tl.float32)
    for kk in range(K):
        acc += A[kk] * B[kk]
    tl.store(out_ptr, acc)


@triton.jit
def _update_vec_triton(beta_ptr, v_ptr, old_v_ptr, new_v_ptr, K: tl.constexpr):
    """
    Compute new_v = beta * v + (1 - beta) * old_v for a single vector of length K.
    beta_ptr: scalar tensor [1]
    v_ptr, old_v_ptr: [K]
    new_v_ptr: [K]
    """
    offs_k = tl.arange(0, K)
    beta = tl.load(beta_ptr)
    v = tl.load(v_ptr + offs_k)
    old_v = tl.load(old_v_ptr + offs_k)
    new_v = beta * v + (1.0 - beta) * old_v
    tl.store(new_v_ptr + offs_k, new_v)


@triton.jit
def _mm_q_triton(A_ptr, B_ptr, C_ptr, K: tl.constexpr):
    """
    Compute C = A @ B where:
      A is [1, K], B is [K, K], C is [1, K]
    A_ptr points to a contiguous [K]-vector (row of A), B_ptr points to [K, K], C_ptr points to [K].
    """
    offs_k = tl.arange(0, K)
    A = tl.load(A_ptr + offs_k)  # [K]
    B_rows = tl.load(B_ptr + offs_k * K + offs_k)  # [K, K] as list of rows
    acc = tl.zeros((K,), dtype=tl.float32)
    for kk in range(K):
        acc += A[kk] * B_rows[kk]
    tl.store(C_ptr + offs_k, acc)
