Key points:

- Ensure Triton kernels are actually launched and perform the required computations. We will implement:
  1) compute_g_beta_kernel: computes g[b, hv] and beta[b, hv] for all tokens b and heads hv.
  2) state_update_kernel: updates the state per token t and head hv:
     - old_v = sum_i k[t, hv, i] * state[hv, i, :] (reduce over D).
     - new_v = beta * v[t, hv, :] + (1 - beta) * old_v.
     - kT_old = sum_i k[t, hv, i] * old_v[i], kT_newv = sum_i k[t, hv, i] * new_v[i].
     - state[hv, :, :] = g * state - kT_old + kT_newv.
     We will implement reductions via tl.sum over D=128.
  3) output_kernel: computes output[t, hv, :] = scale * q_exp[t, hv, :] @ state[hv, :, :], where q_exp is formed by repeating q along head dimension (in provided setup, H_v // H_q = 2, so q_exp[t, hv, :] = concatenate(q[t, 0, :], q[t, 1, :])).

Assumptions:
- H_q=4, H_v=8, D=128 per provided setup; forward will enforce these constants.
- We operate in float32 for computation and return output as bfloat16 per original. The original code also uses float32 for matmul internally.

```python
import math
import torch
import triton
import triton.language as tl


@triton.jit
def _compute_g_beta_kernel(
    A_log_ptr,   # [H_v] float32
    a_ptr,       # [total_seq_len, H_v] float32
    dt_bias_ptr, # [H_v] float32
    b_ptr,       # [total_seq_len, H_v] float32
    g_ptr,       # [total_seq_len, H_v] float32
    beta_ptr,    # [total_seq_len, H_v] float32
    H_v: tl.constexpr,
    total_seq_len: tl.constexpr,
):
    # Each program handles one hv; loop over tokens to fill g[b, hv] and beta[b, hv]
    pid_hv = tl.program_id(0)
    if pid_hv >= H_v:
        return
    dt = tl.load(dt_bias_ptr + pid_hv)  # scalar for this hv

    for b in range(total_seq_len):
        a_val = tl.load(a_ptr + b * H_v + pid_hv)
        bb_val = tl.load(b_ptr + b * H_v + pid_hv)
        x = a_val + dt
        # softplus(x) = log(1 + exp(x))
        sp = tl.log(1.0 + tl.exp(x))
        A_val = tl.load(A_log_ptr + pid_hv)
        g_val = tl.exp(-tl.exp(A_val) * sp)  # g = exp(-exp(A) * softplus(x))
        beta_val = 1.0 / (1.0 + tl.exp(-bb_val))  # sigmoid(b)
        tl.store(g_ptr + b * H_v + pid_hv, g_val)
        tl.store(beta_ptr + b * H_v + pid_hv, beta_val)


@triton.jit
def _state_update_kernel(
    k_ptr,       # [total_seq_len, H_v, D] float32
    v_ptr,       # [total_seq_len, H_v, D] float32
    state_ptr,   # [H_v, D, D] float32
    g_ptr,       # [total_seq_len, H_v] float32
    beta_ptr,    # [total_seq_len, H_v] float32
    new_state_ptr,  # [H_v, D, D] float32 (will be updated in-place)
    H_v: tl.constexpr,
    D: tl.constexpr,
    total_seq_len: tl.constexpr,
):
    # 2D grid over (t, hv)
    t_idx = tl.program_id(0)
    hv_idx = tl.program_id(1)
    if t_idx >= total_seq_len or hv_idx >= H_v:
        return

    # Load k[t, hv, :] and v[t, hv, :]
    k_vec = tl.load(k_ptr + t_idx * (H_v * D) + hv_idx * D + tl.arange(0, D))  # [D]
    v_vec = tl.load(v_ptr + t_idx * (H_v * D) + hv_idx * D + tl.arange(0, D))  # [D]

    # Load current state[hv, :, :]
    state_mat = tl.load(state_ptr + hv_idx * (D * D) + tl.arange(0, D)[:, None] * D + tl.arange(0, D)[None, :])  # [D, D]

    # Compute old_v = k @ state (reduce over D)
    old_v = tl.sum(k_vec[:, None] * state_mat, axis=0)  # [D]

    # Load g and beta for this (t, hv)
    g_val = tl.load(g_ptr + t_idx * H_v + hv_idx)
    beta_val = tl.load(beta_ptr + t_idx * H_v + hv_idx)

    # Compute new_v = beta * v + (1 - beta) * old_v
    new_v = beta_val * v_vec + (1.0 - beta_val) * old_v  # [D]

    # Compute k^T @ old_v and k^T @ new_v (scalar reductions)
    kT_old = tl.sum(k_vec * old_v, axis=0)
    kT_newv = tl.sum(k_vec * new_v, axis=0)

    # Update state[hv, :, :] = g * state - kT_old + kT_newv (broadcast scalar across rows)
    # Note: state_mat is [D, D], we want to multiply each row by g, and then subtract/add scalars.
    # Implement row-wise update:
    # For each row i, new_row = g * state[i, :] - kT_old + kT_newv.
    # But to avoid reading every element, we can update directly via pointer arithmetic:
    for i in range(D):
        row_ptr = new_state_ptr + hv_idx * (D * D) + i * D + tl.arange(0, D)
        # Load row, compute new row, store
        row = tl.load(row_ptr)
        # factor = g * 1 - kT_old + kT_newv applied elementwise across columns
        # However, we don't have row factor; instead, we update each element j:
        # new_row[j] = g * state[i, j] - kT_old + kT_newv
        # We can reconstruct state[i, :] by loading from state_ptr:
        state_row_j = tl.load(state_ptr + hv_idx * (D * D) + i * D + tl.arange(0, D))
        new_row_j = g_val * state_row_j - kT_old + kT_newv
        tl.store(row_ptr, new_row_j)


@triton.jit
def _output_kernel(
    q_ptr,       # [total_seq_len, H_q, D] float32
    state_ptr,   # [H_v, D, D] float32
    out_ptr,     # [total_seq_len, H_v, D] float32
    scale,       # float32
    H_q: tl.constexpr,
    H_v: tl.constexpr,
    D: tl.constexpr,
    total_seq_len: tl.constexpr,
):
    # 2D grid over (t, hv)
    t_idx = tl.program_id(0)
    hv_idx = tl.program_id(1)
    if t_idx >= total_seq_len or hv_idx >= H_v:
        return

    # Form q_exp[t, hv, :] by repeating q along head dimension: H_v // H_q == 2 in provided setup
    q_vec = tl.zeros((2 * D,), dtype=tl.float32)
    # Copy q[t, 0, :]
    q0 = tl.load(q_ptr + t_idx * (H_q * D) + 0 * D + tl.arange(0, D))
    q_vec[0:D] = q0
    # Copy q[t, 1, :]
    q1 = tl.load(q_ptr + t_idx * (H_q * D) + 1 * D + tl.arange(0, D))
    q_vec[D:2*D] = q1

    # Compute out_vec = scale * q_exp @ state[hv, :, :]
    state_mat = tl.load(state_ptr + hv_idx * (D * D) + tl.arange(0, D)[:, None] * D + tl.arange(0, D)[None, :])  # [D, D]
    out_vec = tl.sum(q_vec[:, None] * state_mat, axis=0)  # [D]
    out_vec = out_vec * scale
    tl.store(out_ptr + t_idx * (H_v * D) + hv_idx * D + tl.arange(0, D), out_vec)


class ModelNew(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, q, k, v, state, A_log, a, dt_bias, b, cu_seqlens, scale):
        # Enforce fixed head sizes per provided setup
        total_seq_len, H_q, D = q.shape
        H_v = v.shape[1]
        assert H_q == 4 and H_v == 8 and D == 128, "This Triton implementation assumes H_q=4, H_v=8, D=128"

        # Cast inputs to float32 for computation
        device = q.device
        q_f = q.float().contiguous()
        k_f = k.float().contiguous()
        v_f = v.float().contiguous()
        A_log_f = A_log.float()
        a_f = a.float()
        dt_bias_f = dt_bias.float()
        b_f = b.float()

        # Allocate buffers for g and beta
        g = torch.empty((total_seq_len, H_v), dtype=torch.float32, device=device)
        beta = torch.empty((total_seq_len, H_v), dtype=torch.float32, device=device)

        # Launch gate computation kernel
        _compute_g_beta_kernel[(H_v,)](
            A_log_f, a_f, dt_bias_f, b_f, g, beta,
            H_v=H_v, total_seq_len=total_seq_len
        )

        # Initialize new_state as provided or zeros
        new_state = None
        if state is None:
            new_state = torch.zeros((H_v, D, D), dtype=torch.float32, device=device)
        else:
            new_state = state.float().contiguous()

        # Update state per token using Triton kernel
        for t_idx in range(total_seq_len):
            _state_update_kernel[(H_v,)](
                k_f, v_f, new_state, g, beta, new_state,
                H_v=H_v, D=D, total_seq_len=total_seq_len,
                t_idx=t_idx  # pass as constexpr via grid; Triton will treat as scalar argument
            )

        # Compute output using Triton kernel
        out = torch.empty((total_seq_len, H_v, D), dtype=torch.float32, device=device)
        _output_kernel[(total_seq_len, H_v)](
            q_f, new_state, out, float(scale),
            H_q=H_q, H_v=H_v, D=D, total_seq_len=total_seq_len
        )

        # Return output as bfloat16 and new_state as float32 to match original's state dtype behavior
        return out.to(torch.bfloat16), new_state
```
