@triton.jit
def _softplus_and_gate_kernel(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, beta_ptr, T, H):
    """
    Triton kernel to compute per (t, j):
      x = a[t, j] + dt_bias[j]
      softplus(x) = log(1 + exp(x))
      g = exp(-exp(A_log[j]) * softplus(x))
      beta = sigmoid(b[t, j]) = 1 / (1 + exp(-b[t, j]))
    Writes g_ptr[t, j] and beta_ptr[t, j].
    Assumes a_ptr, dt_bias_ptr, A_log_ptr, b_ptr are valid 2D tensors [T, H].
    """
    # 2D grid: (T, H)
    t = tl.program_id(0)
    j = tl.program_id(1)
    if t >= T or j >= H:
        return

    a_tj = tl.load(a_ptr + t * H + j)
    dtbj = tl.load(dt_bias_ptr + j)
    x = a_tj + dtbj
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x))

    g_tj = tl.exp(-tl.exp(a_log_j) * sp)
    b_tj = tl.load(beta_ptr + t * H + j)
    beta_tj = 1.0 / (1.0 + tl.exp(-b_tj))

    # store results
    tl.store(g_ptr + t * H + j, g_tj)
    tl.store(beta_ptr + t * H + j, beta_tj)


@triton.jit
def _matmul_row_small(A_ptr, B_ptr, C_ptr, K, N):
    """
    Triton kernel to compute C = A @ B where:
      A is [1, K], B is [K, N], C is [1, N]
    K and N are compile-time constants here. We tile over K using BLOCK_K=64.
    """
    # Offsets
    offs_n = tl.arange(0, N)  # output columns
    offs_k = tl.arange(0, 64)  # tile size for reduction
    acc = tl.zeros((N,), dtype=tl.float32)

    # Load A row (1 x K) and iterate over K in chunks
    # A_ptr is contiguous [1, K], so base offset = 0
    for k_start in range(0, K, 64):
        k_idx = k_start + offs_k
        mask_k = k_idx < K
        a_vec = tl.load(A_ptr + k_idx, mask=mask_k, other=0.0)  # [64]
        # Load B[k, :] for all k in the tile -> [64, N]
        b_tile = tl.load(B_ptr + k_idx[:, None] * N + offs_n[None, :], mask=mask_k[:, None], other=0.0)
        acc += tl.sum(a_vec[:, None] * b_tile, axis=0)

    # Store result C[0, :]
    tl.store(C_ptr + offs_n, acc)
