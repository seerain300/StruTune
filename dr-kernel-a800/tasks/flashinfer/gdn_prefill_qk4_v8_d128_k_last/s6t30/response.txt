def compute_g_and_beta(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, beta_ptr, T: tl.int32, H: tl.int32):
    """
    Compute per (t, h):
      g = exp(-exp(A_log[h]) * softplus(a[t, h] + dt_bias[h]))
      beta = sigmoid(b[t, h])
    a_ptr: [T*H] bfloat16
    dt_bias_ptr: [H] float32
    A_log_ptr: [H] float32
    g_ptr: [T*H] float32
    beta_ptr: [T*H] float32
    """
    pid = tl.program_id(0)
    t = pid // H
    h = pid % H
    if t >= T:
        return
    a_val = tl.load(a_ptr + pid).to(tl.float32)  # [1]
    db_val = tl.load(dt_bias_ptr + h)            # [1]
    A_val = tl.load(A_log_ptr + h)               # [1]
    x = a_val + db_val
    sp = softplus_triton(x)                      # scalar softplus(x)
    g_val = tl.exp(-tl.exp(A_val) * sp)
    tl.store(g_ptr + pid, g_val)
    b_val = tl.load(beta_ptr + pid)              # beta provided via beta_ptr
    beta_val = sigmoid_triton(b_val)
    tl.store(beta_ptr + pid, beta_val)


@triton.jit
def softplus_triton(x: tl.tensor) -> tl.tensor:
    """
    Elementwise softplus(x) = log(1 + exp(x))
    """
    return tl.log(1.0 + tl.exp(x))


@triton.jit
def sigmoid_triton(x: tl.tensor) -> tl.tensor:
    """
    Elementwise sigmoid(x) = 1 / (1 + exp(-x))
    """
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def matmul_row(a_ptr, B_ptr, C_ptr, N1: tl.int32, N2: tl.int32, L: tl.int32):
    """
    Compute C[n] = a[K] @ B[n, K], where B is laid out as [N1, N2] contiguous.
    a_ptr: [K], B_ptr: [N1, N2], C_ptr: [N1].
    Grid: one program per n in [0, N1).
    """
    n = tl.program_id(0)
    if n >= N1:
        return
    K = N2
    acc = 0.0
    for k in range(0, K):
        a_k = tl.load(a_ptr + k)
        b_k = tl.load(B_ptr + n * K + k)
        acc += a_k * b_k
    tl.store(C_ptr + n, acc)


@triton.jit
def mm_k_state(k_row_ptr, state_ptr, out_ptr, T_H: tl.int32, H: tl.int32, N: tl.int32):
    """
    For each (t,h), compute out_vec = k[t,h] @ state_old[h, N, N] -> [N].
    k_row_ptr: flattened [T*H] vector, element at t_h = k[t,h,:] as [N]
    state_ptr: flattened [H*N*N] (per seq per h), element at h*N*N + offset over N*N
    out_ptr: flattened [T*H*N], each element is [N]
    Grid: (N, T*H). We derive (t,h) from program_id(1).
    """
    n = tl.program_id(0)
    t_h = tl.program_id(1)
    h = t_h % H
    t = t_h // H
    if n >= N or t >= T or h >= H:
        return
    # Load k_row vector of length N (K=N)
    k_row = tl.zeros([N], dtype=tl.float32)
    for kk in range(0, N):
        k_row[kk] = tl.load(k_row_ptr + t * H * N + h * N + kk)  # k[t,h,kk]
    # Load state row across N columns: state[h, n, :] for each n
    # state is laid out as [H, N, N] contiguous; for a fixed h, row index across N is h*(N*N) + n*N + col
    # We need to sum over col in 0..N-1
    acc = tl.zeros((), dtype=tl.float32)
    for col in range(0, N):
        state_elem = tl.load(state_ptr + h * (N * N) + n * N + col)
        acc += k_row[col] * state_elem
    tl.store(out_ptr + t * H * N + h * N + n, acc)


@triton.jit
def mm_kT_vec(k_row_ptr, vec_ptr, out_ptr, T_H: tl.int32, H: tl.int32, N: tl.int32, K: tl.int32):
    """
    Compute scalar = k[t,h]^T @ vec (length K). vec is [T*H*K] contiguous, index by t_h and kk.
    Grid: (T*H,) one program per (t,h).
    """
    pid = tl.program_id(0)
    t_h = pid
    h = t_h % H
    t = t_h // H
    acc = 0.0
    for kk in range(0, K):
        k_val = tl.load(k_row_ptr + t * H * K + h * K + kk)
        vec_val = tl.load(vec_ptr + t_h * K + kk)
        acc += k_val * vec_val
    tl.store(out_ptr + pid, acc)
