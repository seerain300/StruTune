# Triton kernel: compute g and beta for all (t, hv)
# Inputs:
#   a_ptr: [T, HV] bfloat16
#   dt_bias_ptr: [HV] float32
#   A_log_ptr: [HV] float32
#   b_ptr: [T, HV] bfloat16
# Outputs:
#   g_ptr: [T, HV] float32
#   beta_ptr: [T, HV] float32
@triton.jit
def _compute_g_beta_kernel(
    a_ptr, dt_bias_ptr, A_log_ptr, b_ptr,
    g_ptr, beta_ptr,
    T: tl.int32, HV: tl.int32
):
    # We need to map pid to (t, hv). Launch grid=(T*HV,) so each program handles one element.
    pid = tl.program_id(0)
    if pid >= T * HV:
        return
    t = pid // HV
    hv = pid % HV
    a_val = tl.load(a_ptr + t * HV + hv)
    dt_bias_val = tl.load(dt_bias_ptr + hv)
    A_log_val = tl.load(A_log_ptr + hv)

    # x = a + dt_bias
    x_val = a_val.to(tl.float32) + dt_bias_val
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x_val))
    # g = exp(-exp(A_log) * softplus(x))
    g_val = tl.exp(-tl.exp(A_log_val) * sp)
    tl.store(g_ptr + t * HV + hv, g_val)

    b_val = tl.load(b_ptr + t * HV + hv)
    beta_val = 1.0 / (1.0 + tl.exp(-b_val.to(tl.float32)))
    tl.store(beta_ptr + t * HV + hv, beta_val)

# state_new_ptr: [H, V*K] float32, laid out similarly
# output_ptr: [V*K] float32 (we cast to bfloat16 on host after kernel)

    # We assume q_exp_ptr and state_new_ptr are indexed such that rows correspond to heads h in [0, H).
    # For each column col in [0, V*K), output[col] = sum_h q_exp[h, col] * state_new[h, col].
    # Grid: (1,) specialized per v-vector; host will launch once per (t, v).

            st = tl.load(state_new_ptr + h * (V * K) + col)
            dot += qh * st
        o_elem = dot * scale
        tl.store(output_ptr + col, o_elem)


# Triton kernel: update state per v: new_state[h, v, k] = g * state_old[h, v, k] + (beta * v_row + (1-beta) * old_v) @ k_row - (k_row @ old_v)
# We implement elementwise per (h, v, k) using loaded vectors for k_row, v_row, and old_v, and g, beta scalars for this (t, h, v).
# state_old_ptr: [H, V, K] float32 (k-last: last dim K)
# k_row_ptr: [K] float32, laid out as t*(H*K) + h*K + kk
# v_row_ptr: [K] float32, laid out similarly
# old_v_ptr: [K] float32, same as k_row layout
# new_state_ptr: [H, V, K] float32
@triton.jit
def _update_state_per_v_kernel(
    state_old_ptr, k_row_ptr, v_row_ptr, old_v_ptr, new_state_ptr,
    g_val: tl.float32, beta_val: tl.float32,

    v = tl.program_id(1)

        return
    # Compute sums needed:
    # remove = sum_k k_row[k] * old_v[k]

        k_val = tl.load(k_row_ptr + kk)
        oldvk = tl.load(old_v_ptr + kk)
        remove += k_val * oldvk

    # state_update = sum_k k_row[k] * (beta * v_row[k] + (1-beta) * old_v[k])
    state_update = tl.zeros((), dtype=tl.float32)
    for kk in range(0, K):
        k_val = tl.load(k_row_ptr + kk)
        vvk = tl.load(v_row_ptr + kk)
        oldvk = tl.load(old_v_ptr + kk)
        term = k_val * (beta_val * vvk + (1.0 - beta_val) * oldvk)
        state_update += term

    # new_state[h, v, k] = g * state_old[h, v, k] - remove + state_update

        oldk = tl.load(state_old_ptr + h * (V * K) + v * K + kk)

        tl.store(new_state_ptr + h * (V * K) + v * K + kk, newk)
