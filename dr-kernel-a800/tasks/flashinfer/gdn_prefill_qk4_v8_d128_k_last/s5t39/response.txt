# Triton kernel: compute g and beta for all (t, hv)
# Inputs:
#   a_ptr: [T, HV] bfloat16
#   dt_bias_ptr: [HV] float32
#   A_log_ptr: [HV] float32
#   b_ptr: [T, HV] bfloat16
# Outputs:
#   g_ptr: [T, HV] float32
#   beta_ptr: [T, HV] float32
@triton.jit
def _compute_g_beta_kernel(
    a_ptr, dt_bias_ptr, A_log_ptr, b_ptr,
    g_ptr, beta_ptr,
    T: tl.int32, HV: tl.int32
):
    pid = tl.program_id(0)  # over T*HV
    t = pid // HV
    hv = pid % HV
    if t >= T or hv >= HV:
        return
    # Load values
    a_val = tl.load(a_ptr + t * HV + hv)
    dt_bias_val = tl.load(dt_bias_ptr + hv)
    A_log_val = tl.load(A_log_ptr + hv)
    b_val = tl.load(b_ptr + t * HV + hv)

    # Compute x = a + dt_bias
    x_val = a_val.to(tl.float32) + dt_bias_val
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x_val))
    # g = exp(-exp(A_log) * softplus(x))
    g_val = tl.exp(-tl.exp(A_log_val) * sp)
    tl.store(g_ptr + t * HV + hv, g_val)

    # beta = sigmoid(b)
    beta_val = 1.0 / (1.0 + tl.exp(-b_val.to(tl.float32)))
    tl.store(beta_ptr + t * HV + hv, beta_val)


# Triton kernel: repeat q heads to [T, Hv, K] using factor = Hv // H
# Input q_ptr: [T, H, K], bfloat16
# Output q_exp_ptr: [T, Hv, K], bfloat16
@triton.jit
def _repeat_q_heads_kernel(
    q_ptr, q_exp_ptr,
    T: tl.int32, H: tl.int32, K: tl.int32, factor: tl.int32
):
    pid_t = tl.program_id(0)  # over T
    pid_h = tl.program_id(1)  # over H * factor
    if pid_t >= T or pid_h >= (H * factor):
        return
    hv = pid_h % factor
    h_src = pid_h // factor
    # src index: t * (H*K) + h_src * K + kk
    for kk in range(0, K):
        src_index = pid_t * (H * K) + h_src * K + kk
        val = tl.load(q_ptr + src_index)
        dst_index = pid_t * (factor * K) + pid_h * K + kk
        tl.store(q_exp_ptr + dst_index, val)


# Triton kernel: repeat k heads to [T, Hv, K] using factor = Hv // H
# Input k_ptr: [T, H, K], bfloat16
# Output k_exp_ptr: [T, Hv, K], bfloat16
@triton.jit
def _repeat_k_heads_kernel(

        val = tl.load(k_ptr + src_index)
        dst_index = pid_t * (factor * K) + pid_h * K + kk
        tl.store(k_exp_ptr + dst_index, val)
