@triton.jit
def compute_g_and_beta(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, beta_ptr, T: tl.int32, H: tl.int32):
    """
    Compute per (t, h):
      x = a[t, h] + dt_bias[h]
      softplus(x) = log(1 + exp(x))
      g = exp(-exp(A_log[h]) * softplus(x))
      beta = sigmoid(b[t, h])
    a_ptr: [T*H] bfloat16
    dt_bias_ptr: [H] float32
    A_log_ptr: [H] float32
    g_ptr: [T*H] float32
    beta_ptr: [T*H] float32 (b is provided externally; we store beta)
    Grid: (T*H,)
    """
    pid = tl.program_id(0)
    t = pid // H
    h = pid % H
    if t >= T or h >= H:
        return
    a_val = tl.load(a_ptr + pid).to(tl.float32)
    db = tl.load(dt_bias_ptr + h)
    A = tl.load(A_log_ptr + h)
    x = a_val + db
    sp = tl.log(1.0 + tl.exp(x))  # softplus(x)
    g_val = tl.exp(-tl.exp(A) * sp)
    # beta provided externally and stored in beta_ptr
    beta_val = tl.load(beta_ptr + pid)
    tl.store(g_ptr + pid, g_val)
    tl.store(beta_ptr + pid, beta_val)


@triton.jit

    """

    x_ptr: [N]
    out_ptr: [N]
    Grid: (N,)

    if pid >= N:
        return
    x = tl.load(x_ptr + pid)
    sp = tl.log(1.0 + tl.exp(x))
    tl.store(out_ptr + pid, sp)


@triton.jit
def sigmoid_triton(x_ptr, out_ptr, N: tl.int32):

    tl.store(out_ptr + pid, sig)

            b = tl.load(b_ptr + i).to(tl.float32)
            tl.store(out_ptr + i, (a + b).to(tl.bfloat16))

    @triton.jit
    def sub_vec_bf16(a_ptr, b_ptr, out_ptr, N: tl.constexpr):
        """
        out[i] = a[i] - b[i]
        """
        for i in range(0, N):
            a = tl.load(a_ptr + i).to(tl.float32)
            b = tl.load(b_ptr + i).to(tl.float32)
            tl.store(out_ptr + i, (a - b).to(tl.bfloat16))

    # Triton matmul row-wise for output: q_exp[t,h] @ new_state[h,N,N] -> [N]
    @triton.jit
    def matmul_row_vec2d(q_ptr, new_ptr, out_ptr, K: tl.constexpr, N: tl.constexpr):
        """
        Compute out[n] = sum_k q[k] * new[n,k] for all n across the vector.
        q_ptr: [K], new_ptr: [N*N] (flattened), out_ptr: [N].
        Launch grid: (N,) with constexpr K,N. Host sets q_ptr and new_ptr base pointers per (t,h).
        Triton does not accept dynamic (t,h) inside kernel; we pass base pointers computed in host.
        """
        n = tl.program_id(0)
        if n >= N:
            return
        acc = 0.0
        for kk in range(0, K):
            qk = tl.load(q_ptr + kk)
            new_val = tl.load(new_ptr + n * K + kk)  # new[n,kk]
            acc += qk * new_val
        tl.store(out_ptr + n, acc)
