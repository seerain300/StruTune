# Kernel to fill a contiguous tensor with random values (normal distribution).
# We implement our own RNG using Triton and tl.rand per element.
@triton.jit
def fill_rand_kernel(out_ptr, n_elements: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * 1024 + tl.arange(0, 1024)
    mask = offsets < n_elements
    rand = tl.rand(offsets)
    tl.store(out_ptr + offsets, rand, mask=mask)


# Depthwise Conv2d (kernel 1x7x7, padding=3, groups=C): y[b,c,h,w] = sum_{i,j} x[b,c,h+3,w+3] * w[c,0,i,j]
@triton.jit
def conv2d_depthwise_grouped_kernel(
    x_ptr, w_ptr, y_ptr,
    B, C, H_in, W_in, H_out, W_out,
    x_stride_b, x_stride_c, x_stride_h, x_stride_w,
    w_stride_c, w_stride_kh, w_stride_kw,
    y_stride_b, y_stride_c, y_stride_h, y_stride_w,
    eps: tl.constexpr,  # not used, but kept for signature symmetry
    num_warps: tl.constexpr, num_stages: tl.constexpr
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    h = tl.program_id(2)
    w = tl.program_id(3)

    # Only compute for valid output positions
    # H_out = H_in and W_out = W_in for padding=3 with kernel 7x7 => dilation would differ; here padding 3, kernel 7 => output size equals input.
    # But to be general, compute only where h in [0,H_out) and w in [0,W_out)
    acc = 0.0
    # Sum over 7x7 kernel
    for kh in range(7):
        for kw in range(7):
            in_h = h + kh - 3  # padding
            in_w = w + kw - 3
            # bounds check
            if (in_h >= 0) and (in_h < H_in) and (in_w >= 0) and (in_w < W_in):
                x_off = b * x_stride_b + c * x_stride_c + in_h * x_stride_h + in_w * x_stride_w
                w_off = c * w_stride_c + 0 * w_stride_kh + kh * w_stride_kw + kw * w_stride_kw  # since k=0 only
                x_val = tl.load(x_ptr + x_off)
                w_val = tl.load(w_ptr + w_off)
                acc += x_val * w_val
    y_off = b * y_stride_b + c * y_stride_c + h * y_stride_h + w * y_stride_w
    tl.store(y_ptr + y_off, acc)

def layernorm_per_channel_kernel(

    B, H, W, C,
    x_stride_b, x_stride_h, x_stride_w, x_stride_c,
    y_stride_b, y_stride_h, y_stride_w, y_stride_c,
    eps: tl.constexpr,
    num_warps: tl.constexpr, num_stages: tl.constexpr
):

    pid = tl.program_id(0)
    # Map linear pid to (b,h,w)
    hw_total = H * W
    b = pid // hw_total
    rem = pid % hw_total
    h = rem // W
    w = rem % W

    sum_val = 0.0
    sum_sq = 0.0
    for c in range(C):
        x_off = b * x_stride_b + h * x_stride_h + w * x_stride_w + c * x_stride_c

        sum_val += x_val
        sum_sq += x_val * x_val

    y = 0.5 * x * (1.0 + tanh_inner)
    tl.store(y_ptr + offsets, y, mask=mask)
