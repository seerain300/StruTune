Here is the complete Triton-optimized implementation with ModelNew as the entry point. It moves all “compute” to Triton kernels (no torch.sort, torch.bmm, torch.silu, torch.bincount, torch.cumsum on tensors), while using torch for necessary data movement like scatter and index_add. The Triton kernels implement:

- Flatten + stable sort (via odd-even transposition sort)
- Per-expert bincount
- Cumulative starts for stable sort
- Within-group position computation
- Batched matrix multiplications for gate, up, and final down
- Elementwise silu
- Mask application and aggregation to final output

def flatten_and_sort_stable(
    selected_experts_ptr,    # int64* [T, K] (input and workspace)
    routing_weights_ptr,     # dtype* [T, K] (input and workspace)
    flat_experts_ptr,        # int64* [N]

    sorted_experts_ptr,      # int64* [N]

    K: tl.int32,             # num_experts_per_tok
    N: tl.int32,             # T*K
    BLOCK: tl.constexpr,     # next power of two >= N
):

    # 1) Compute flat vectors (already done in host by pre-flattening),
    #    but we need to write sorted_experts and sorted_indices via stable sort.
    #    Here, we perform odd-even transposition sort in global memory.
    #    We assume N <= BLOCK lanes; we pad with max-expert value for keys to simplify.

    # For simplicity, we assume host pre-filled flat buffers, and we only perform sort.

            # even pass: compare (0,1), (2,3), ...
            for i in range(0, BLOCK, 2):
                idx0 = sorted_experts_ptr + i
                idx1 = sorted_experts_ptr + i + 1
                v0 = tl.load(idx0)
                v1 = tl.load(idx1)

                new1 = tl.where(need_swap, v0, v1)
                tl.store(idx0, new0)
                tl.store(idx1, new1)

@triton.jit
def bincount_experts(flat_experts_ptr, counts_ptr, N: tl.int32, EXPERTS: tl.int32):
    # Simple Triton reduction per expert: counts[exp] += sum_{i in flat_experts==exp}
    for i in range(0, N):
        val = tl.load(flat_experts_ptr + i)
        # Accumulate with atomic add
        tl.atomic_add(counts_ptr + val, 1)


@triton.jit
def cumsum_starts(counts_ptr, starts_ptr, EXPERTS: tl.int32):
    # starts[0] = 0
    tl.store(starts_ptr + 0, 0)
    running = 0
    for i in range(1, EXPERTS):
        running += tl.load(counts_ptr + i - 1)
        tl.store(starts_ptr + i, running)


@triton.jit
def compute_within_pos(sorted_experts_ptr, counts_ptr, within_ptr, N: tl.int32, EXPERTS: tl.int32):
    # Compute starts
    starts = tl.zeros([EXPERTS], dtype=tl.int32)
    running = 0
    for i in range(1, EXPERTS):
        running += tl.load(counts_ptr + i - 1)
        tl.store(starts + i, running)

        start = tl.load(starts + exp)
        within = j - start
        tl.store(within_ptr + j, within)

@triton.jit
def bmm_row_triton(A_ptr, B_ptr, C_ptr, H: tl.constexpr, INTER: tl.constexpr):
    # Compute C = A @ B, where A is [H], B is [H, INTER], C is [INTER].
    # This kernel handles one row (N=1). In practice, call it per valid pair.
    for k in range(0, INTER):
        dot = 0.0
        for i in range(0, H):
            a = tl.load(A_ptr + i)
            b = tl.load(B_ptr + i * INTER + k)
            dot += a * b
        tl.store(C_ptr + k, dot)


@triton.jit
def silu_elementwise(X_ptr, Y_ptr, size: tl.int32, BLOCK: tl.constexpr):
    # Compute Y = X * sigmoid(X) elementwise over a flat buffer.
    for i in range(0, size, BLOCK):
        offs = i + tl.arange(0, BLOCK)
        mask = offs < size
        x = tl.load(X_ptr + offs, mask=mask, other=0.0)
        s = 1.0 / (1.0 + tl.exp(-x))
        y = x * s
        tl.store(Y_ptr + offs, y, mask=mask)
