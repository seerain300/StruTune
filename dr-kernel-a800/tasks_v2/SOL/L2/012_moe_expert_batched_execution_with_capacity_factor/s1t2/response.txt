                B, H, M,
                X_stride_b, X_stride_h,
                W_stride_h, W_stride_m,
                Y_stride_b, Y_stride_m,

    Batched matmul for each b:
    Y[b, m] = sum_k X[b, k] * W[k, m]
    Shapes:
      - X: [B, H]
      - W: [H, M]
      - Y: [B, M]
    Accumulate in fp32, output stored in Y (fp32). B is a runtime parameter (we launch grid over B).
    """
    b = tl.program_id(0)  # batch index (we launch grid with B)
    pid_m = tl.program_id(1)  # tile along M
    pid_h = tl.program_id(2)  # tile along H

    m_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Loop over H dimension in BLOCK_H chunks
    for k in range(0, H, BLOCK_H):
        k_offsets = k + h_offsets

        mask_h = k_offsets < H
        mask_m = m_offsets < M

        # Load X[b, k_offsets] as vector (BLOCK_H,)
        x_ptrs = X_ptr + b * X_stride_b + k_offsets * X_stride_h
        x = tl.load(x_ptrs, mask=mask_h, other=0.0).to(tl.float32)

    y_block = tl.sum(acc, axis=1)  # shape (BLOCK_M,)

    tl.store(y_ptrs, y_block, mask=mask_m)

    Elementwise: Out[b, m] = silu(Z[b, m]) * W[b, m]
    silu(x) = x * sigmoid(x)

    m_block = tl.program_id(1)
    m_offsets = m_block * BLOCK_M + tl.arange(0, BLOCK_M)

    w = tl.load(w_ptrs, mask=mask, other=0.0).to(tl.float32)

                W_stride_h, W_stride_m,
                Y_stride_b, Y_stride_m,

    """
    Triton batched matmul for each b:
    Y[b, m] = sum_k X[b, k] * W[k, m]
    X: [B, H], W: [H, M], Y: [B, M]
    Accumulate in fp32; output is fp32.
    """

    pid_m = tl.program_id(1)
    pid_h = tl.program_id(2)

    h_offsets = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
