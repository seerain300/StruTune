def triton_bmm_row(A_ptr, B_ptr, C_ptr,

                   K: tl.constexpr,            # A cols (hidden_size), B rows

                   B_stride_col: tl.int32,     # stride for B col (N_COLS dimension)
                   C_stride_row: tl.int32,     # stride for C row (M dimension)

    # We will launch once per valid pair. For generality, we implement a single-row kernel.
    # A[M, K] is provided as a contiguous row vector A_ptr (row 0). B[K, N_COLS] is 2D.

    # We'll implement a simple loop over K in blocks BLOCK_K and accumulate over N_COLS in blocks BLOCK_N.

            # Reduce along K-axis
            acc += tl.sum(b_block * a_vec[:, None], axis=0)
    # Store acc to C[0, :]
    tl.store(C_ptr + 0 * C_stride_row + n_offsets * C_stride_col, acc, mask=mask_n)

        sig = 1.0 / (1.0 + tl.exp(-x))
        y = x * sig
        tl.store(y_ptr + offsets, y, mask=mask)
