# Note: We cannot guarantee full correctness without a multi-dimensional grid for convs here. We will keep convs in PyTorch to ensure the overall forward works.
@triton.jit
def gelu_triton_1d_kernel(
    X_ptr, Y_ptr,
    total_elements: tl.constexpr,  # total number of elements to process
):
    pid = tl.program_id(axis=0)
    # Compute index
    idx = pid
    # Load, compute GELU, store
    x = tl.load(X_ptr + idx).to(tl.float32)
    c0 = 0.7978845608028654  # sqrt(2/pi)

# Batched GEMV Triton kernel: compute Y[n, t, k] = sum_j X[n, t, j] * W[k, j], for n in [0..N-1], t in [0..T-1], k in [0..K-1]
# X: [N, T, M] bf16, W: [K, M] bf16, Y: [N, T, K] bf16
@triton.jit
def linear_bmm_kernel(
    X_ptr, W_ptr, Y_ptr,
    N, T, M, K,
    x_strideN, x_strideT, x_strideM,
    w_strideK, w_strideM,
    y_strideN, y_strideT, y_strideK,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # one instance per (n, t)
    n = pid // T
    t = pid % T

    # Accumulator vector for k in tiles
    for ko in range(0, K, BLOCK_K):
        offs_k = ko + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        acc = tl.zeros([BLOCK_K], dtype=tl.float32)

        # Reduce over M in tiles
        for m0 in range(0, M, BLOCK_M):
            offs_m = m0 + tl.arange(0, BLOCK_M)
            mask_m = offs_m < M

            # Load X[n, t, offs_m] as bf16 -> float32

            x_vals = tl.load(x_ptrs, mask=mask_m, other=0.0).to(tl.float32)  # [BLOCK_M]

        # Store results to Y[n, t, offs_k]
        y_ptrs = Y_ptr + n * y_strideN + t * y_strideT + offs_k * y_strideK
        tl.store(y_ptrs, acc.to(tl.bfloat16), mask=mask_k)
