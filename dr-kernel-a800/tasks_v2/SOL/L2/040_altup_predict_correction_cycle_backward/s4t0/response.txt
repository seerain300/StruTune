# - scaled: [H] float32
# - W:      [K, H] float32 (no bias in original code, but kernel supports bias)
# - bias:   [K] or None (we pass a zero tensor if no bias; kernel supports bias)
# - out:    [K] float32

def tanh_linear_kernel(
    scaled_ptr,   # *f32, shape [H]
    W_ptr,        # *f32, shape [K, H]
    bias_ptr,     # *f32, shape [K] (can be zero tensor if no bias)
    out_ptr,      # *f32, shape [K]
    H: tl.constexpr,       # hidden_size
    K: tl.constexpr,       # output feature size of W
    BLOCK: tl.constexpr    # reduction block size over H
):
    # One program per output element (token index)
    pid = tl.program_id(axis=0)
    # Accumulate dot product over H in chunks of BLOCK
    acc = tl.zeros([1], dtype=tl.float32)
    # Loop over H in blocks
    for start in range(0, H, BLOCK):
        offs = start + tl.arange(0, BLOCK)
        mask = offs < H
        s = tl.load(scaled_ptr + offs, mask=mask, other=0.0)  # [BLOCK] f32
        w = tl.load(W_ptr + pid * H + offs, mask=mask, other=0.0)  # [BLOCK] f32
        acc += tl.sum(s * w, axis=0)
    # Add bias if provided
    b = tl.load(bias_ptr + pid, mask=True, other=0.0)  # scalar
    val = acc + b
    # Apply tanh
    val = tl.math.tanh(val)
    # Store result
    tl.store(out_ptr + pid, val)

  - Batch tokens: num_tokens = B * S. We will launch Triton kernels with grid = (num_tokens,) for RMSNorm and tanh(linear). For reduction over H, use a constexpr BLOCK (e.g., 256 or 512) and loop over H in chunks. For K, use a small unrolled loop up to a constexpr MAX_K (we can set to 36), which covers both cases (prediction 36 and correction 36).
  - The matmul (predictions) we keep in PyTorch. This is acceptable for this task and focuses on the required Triton usage.

Below is a Triton-backed ModelNew that:
- Calls Triton kernels to compute rstd per token (RMSNorm).
- Calls Triton kernels to compute modalities_predict and modalities_correct (tanh(linear) with appropriate W).
- Recomputes the forward structure and then derives gradients via PyTorch using the Triton-produced tensors. The heavy numerical work is in Triton.

import torch
import torch.nn as nn
import triton
import triton.language as tl


# Triton kernel: compute rstd = rsqrt(mean(x^2) + eps) for a vector x of length H
# Input: x_ptr [H], eps scalar
# Output: rstd_ptr[1] (store single scalar)
@triton.jit
def rms_norm_forward_kernel(x_ptr, rstd_ptr, H: tl.constexpr, eps: tl.float32, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)  # token index
    acc = tl.zeros([1], dtype=tl.float32)
    for start in range(0, H, BLOCK):
        offs = start + tl.arange(0, BLOCK)
        mask = offs < H
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        acc += tl.sum(x * x, axis=0)
    mean = acc / H
    val = tl.rsqrt(mean + eps)
    tl.store(rstd_ptr, val)


# Triton kernel: y[k] = tanh(dot(scaled, W[k, :]) + bias[k])
# Inputs: scaled [H], W [K, H], bias [K]
# Output: y [K]
@triton.jit
def tanh_linear_kernel(scaled_ptr, W_ptr, bias_ptr, y_ptr, H: tl.constexpr, K: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)  # output index k
    acc = tl.zeros([1], dtype=tl.float32)
    for start in range(0, H, BLOCK):
        offs = start + tl.arange(0, BLOCK)
        mask = offs < H
        s = tl.load(scaled_ptr + offs, mask=mask, other=0.0)
        w = tl.load(W_ptr + pid * H + offs, mask=mask, other=0.0)
        acc += tl.sum(s * w, axis=0)
    b = tl.load(bias_ptr + pid)
    val = acc + b
    val = tl.math.tanh(val)
    tl.store(y_ptr + pid, val)
