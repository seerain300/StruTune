import torch
import triton
import triton.language as tl


@triton.jit
def lower_tri_mask_kernel(out_ptr, chunk_size: tl.constexpr, diagonal: tl.constexpr):
    # Produce a 2D lower-triangular mask of shape [chunk_size, chunk_size] with given diagonal.
    # out_ptr points to a contiguous float32 tensor of size chunk_size*chunk_size.
    rows = tl.arange(0, chunk_size)
    cols = tl.arange(0, chunk_size)
    row_idx = rows[:, None]  # [chunk_size, 1]
    col_idx = cols[None, :]  # [1, chunk_size]
    # Keep lower-triangular including diagonal? For diagonal=-1, condition is i >= j.
    cond = (row_idx >= (col_idx + diagonal))
    out_val = tl.where(cond, 1.0, 0.0)
    offsets = row_idx * chunk_size + col_idx
    tl.store(out_ptr + offsets, out_val)


@triton.jit
def per_row_cumsum_kernel(in_ptr, out_ptr, rows_ptr, cols_ptr, chunk_size: tl.constexpr):
    # For each row r, compute inclusive cumsum of that row in the 2D matrix in_ptr -> out_ptr.
    # We assume in_ptr and out_ptr are contiguous [chunk_size, chunk_size].
    # This kernel is launched with grid=(chunk_size,) and uses per-program row index.
    r = tl.program_id(0)
    cols = tl.arange(0, chunk_size)
    base = r * chunk_size
    in_row = in_ptr + base + cols
    out_row = out_ptr + base + cols
    acc = 0.0
    for j in range(0, chunk_size):
        val = tl.load(in_row + j)
        acc = acc + val
        tl.store(out_row + j, acc)
