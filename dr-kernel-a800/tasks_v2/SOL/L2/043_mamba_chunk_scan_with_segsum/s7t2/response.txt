@triton.jit
def pad_last_dim_kernel(

    out_ptr,   # *float32, output tensor flattened as [n_out]

@triton.jit
def cumsum_1d_kernel(

    out_ptr,   # *float32, output flattened
    n_elements: tl.constexpr,
