import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import triton
import triton.language as tl


@triton.jit
def layernorm_stats_3d(
    X,            # *ptr* input [B, L, D]
    SUM,          # *ptr* per (b,l) sum [B*L]
    SUMSQ,        # *ptr* per (b,l) sumsq [B*L]
    B, L, D,
    BLOCK_D: tl.constexpr,
):
    # 2D grid over (B, L), reduce across D in tiles
    b = tl.program_id(0)
    l = tl.program_id(1)
    base = (b * L + l) * D

    sum_val = tl.zeros((), dtype=tl.float32)
    sumsq_val = tl.zeros((), dtype=tl.float32)

    for d0 in range(0, D, BLOCK_D):
        d_offsets = d0 + tl.arange(0, BLOCK_D)
        mask = d_offsets < D
        x = tl.load(X + base + d_offsets, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sumsq_val += tl.sum(x * x, axis=0)

    # Store stats for this (b,l)
    tl.store(SUM + b * L + l, sum_val)
    tl.store(SUMSQ + b * L + l, sumsq_val)


@triton.jit
def layernorm_apply_3d(
    X,            # *ptr* input [B, L, D]
    Y,            # *ptr* output [B, L, D]
    W,            # *ptr* weight [D]
    BIAS,         # *ptr* bias [D]
    SUM,          # *ptr* per (b,l) sum [B*L]
    SUMSQ,        # *ptr* per (b,l) sumsq [B*L]
    EPS: tl.constexpr,
    B, L, D,
    BLOCK_D: tl.constexpr,
):
    # 2D grid over (B, L), normalize and apply affine
    b = tl.program_id(0)
    l = tl.program_id(1)
    base = (b * L + l) * D

    sum_val = tl.load(SUM + b * L + l)
    sumsq_val = tl.load(SUMSQ + b * L + l)
    mean = sum_val / D
    var = sumsq_val / D - mean * mean

        tl.store(Y + base + d_offsets, y, mask=mask)

    """

    Returns y: [B, L, D] in float32.
    """
    assert x.ndim == 3, "Expected x to be [B, L, D]"
    B, L, D = x.shape
    x_fp32 = x.to(torch.float32)
    y = torch.empty_like(x_fp32)

    # Compute stats
    sum_buf = torch.empty(B * L, device=x.device, dtype=torch.float32)
    sumsq_buf = torch.empty(B * L, device=x.device, dtype=torch.float32)

    # Choose tile size along D

        sum_buf, sumsq_buf,
        EPS=eps,
        B=B, L=L, D=D,
        BLOCK_D=BLOCK_D,
    )
    return y
