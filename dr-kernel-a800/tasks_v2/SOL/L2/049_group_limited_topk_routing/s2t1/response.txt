import triton.language as tl

NUM_EXPERTS = 256

EXPERTS_PER_GROUP = NUM_EXPERTS // N_GROUPS  # 32
TOPK_GROUP = 4
TOPK_EXPERTS = 8

    weight_ptr,         # *f32, shape [NUM_EXPERTS, hidden_dim]
    expert_bias_ptr,    # *f32, shape [NUM_EXPERTS]

    num_tokens,         # int32
    hidden_dim,         # int32

    scores = tl.zeros((NUM_EXPERTS,), dtype=tl.float32)

        score = 0.0
        for j in range(0, hidden_dim):
            h = tl.load(hidden_ptr + token * hidden_dim + j)
            w = tl.load(weight_ptr + e * hidden_dim + j)
            score += h * w
        scores[e] = score

    # Sigmoid activation
    scores = 1.0 / (1.0 + tl.exp(-scores))

    # Add expert bias (broadcasted)
    bias = tl.load(expert_bias_ptr + tl.arange(0, NUM_EXPERTS))
    scores = scores + bias

            s = scores[idx]

                top1 = s

                top2 = s

    selected_groups = tl.full((TOPK_GROUP,), -1, dtype=tl.int32)

    top8_val = tl.full((TOPK_EXPERTS,), -1.0e30, dtype=tl.float32)

    for k in range(0, TOPK_EXPERTS):
        best_e = -1
        best_score = -1.0e30
        for e in range(0, NUM_EXPERTS):
            if scores[e] > best_score:
                best_score = scores[e]
                best_e = e
        # store best_e and its normalized weight

        # Note: normalization here means dividing by sum of selected top8 scores per token. But we don't have sum yet.

        top8_val[k] = best_score

        scores[best_e] = -1.0e30

    sum_top8 = 0.0
    for k in range(0, TOPK_EXPERTS):

  3) Adds expert_bias → scores_for_routing.
  4) Reshapes scores_for_routing into groups [num_tokens, 8, 32].
  5) Computes top-2 per group (within 32 experts) and sums → group_scores [num_tokens, 8].
  6) Selects top-4 groups based on group_scores → group_idx [num_tokens, 4].
  7) Creates group_mask [num_tokens, 8] with 1.0 for selected groups, 0.0 otherwise.
  8) Expands group_mask to per-expert mask → score_mask [num_tokens, 256] where non-selected groups are set to False (0), selected groups are True (1).
