```python
import torch
import triton
import triton.language as tl


@triton.jit
def _matmul_linear_fwd(
    hidden_ptr,      # *ptr to hidden_states [N, K]
    weight_ptr,      # *ptr to weight [E, K]
    out_ptr,         # *ptr to output [N, E]
    N, K, E,         # sizes: num_tokens, hidden_dim, num_experts
    stride_hn, stride_hk,  # strides for hidden: row (N), col (K)
    stride_we, stride_wk,  # strides for weight: row (E), col (K)
    stride_on, stride_oe,  # strides for out: row (N), col (E)
    BLOCK_K: tl.constexpr,  # tile size over K
):
    # Each program computes one output element: out[n, e] for given n, e
    pid_n = tl.program_id(0)  # token row
    pid_e = tl.program_id(1)  # expert column

    # Bounds check (if grid is larger than N/E, though we set grid to (N, E))
    if pid_n >= N or pid_e >= E:
        return

    # Accumulator in fp32
    acc = tl.zeros((), dtype=tl.float32)

    # Loop over K in chunks
    # K is a runtime integer; Triton supports Python-level loops for meta-parameters,
    # but since K can vary, we use a while-loop over range(0, K, BLOCK_K).
    # Note: range here is evaluated on host side, but Triton will handle it.
    for k0 in range(0, K, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load hidden row slice: hidden_states[pid_n, offs_k]
        h = tl.load(hidden_ptr + pid_n * stride_hn + offs_k * stride_hk,
                    mask=mask_k, other=0.0)

        # Load weight row slice: weight[pid_e, offs_k]
        w = tl.load(weight_ptr + pid_e * stride_we + offs_k * stride_wk,
                    mask=mask_k, other=0.0)

        # Accumulate dot product over this chunk
        acc += tl.sum(h * w, axis=0)

    # Store result
    tl.store(out_ptr + pid_n * stride_on + pid_e * stride_oe, acc)

    def forward(self, hidden_states: torch.Tensor, weight: torch.Tensor, expert_bias: torch.Tensor, routed_scaling_factor: float):

        Performs the linear projection using a Triton kernel, and keeps subsequent logic in PyTorch.

        assert hidden_states.dim() == 2, "hidden_states must be [num_tokens, hidden_dim]"
        assert weight.dim() == 2, "weight must be [num_experts, hidden_dim]"
        assert expert_bias.dim() == 1, "expert_bias must be [num_experts]"

        _, topk_idx = torch.topk(masked_scores, k=8, dim=-1, sorted=False)  # top_k = 8

        # 10) Gather selected expert scores from original (pre-bias) scores
        selected_scores = torch.gather(scores, dim=1, index=topk_idx)  # [num_tokens, 8]

        # 11) Normalize routing weights
        denom = selected_scores.sum(dim=-1, keepdim=True) + 1e-20
        topk_weight = selected_scores / denom  # [num_tokens, 8]

        # 12) Apply routing scaling factor
        topk_weight = topk_weight * routed_scaling_factor

        return topk_idx, topk_weight
```

```python
        selected_scores = masked_scores.gather(1, topk_idx)  # [num_tokens, 8]

        # 8) Normalize routing weights
        denom = selected_scores.sum(dim=-1, keepdim=True) + 1e-20
        topk_weight = selected_scores / denom  # [num_tokens, 8]

        # 9) Apply routing scaling factor
        topk_weight = topk_weight * routed_scaling_factor

        return topk_idx, topk_weight
```
