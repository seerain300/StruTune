import torch
import torch.nn as nn

import triton
import triton.language as tl

# Kernel 1: logits = hidden_states @ weight.T
# hidden_states: [M, K] where K=hidden_size=128
# weight: [N, K] where N=num_experts=256, row-major
# output logits: [M, N]
@triton.jit
def matmul_logits_kernel(
    a_ptr,        # *f32, [M, K]
    b_ptr,        # *f32, [N, K] (row-major)
    out_ptr,      # *f32, [M, N]
    M: tl.int32,
    N: tl.constexpr,             # 256
    K: tl.constexpr,             # 128
):
    row = tl.program_id(0)  # program id over tokens
    col = tl.program_id(1)  # program id over experts
    acc = 0.0
    for k in range(K):
        a_val = tl.load(a_ptr + row * K + k)
        b_val = tl.load(b_ptr + col * K + k)
        acc += a_val * b_val
    tl.store(out_ptr + row * N + col, acc)


# Kernel 2: scores = sigmoid(logits) + expert_bias
# logits: [M, N], bias: [N], scores: [M, N]
@triton.jit
def sigmoid_bias_kernel(
    logits_ptr,        # *f32, [M, N]
    bias_ptr,          # *f32, [N]
    scores_ptr,        # *f32, [M, N]
    M: tl.int32,
    N: tl.constexpr,             # 256
):
    row = tl.program_id(0)  # one program per row (token)
    cols = tl.arange(0, N)   # N=256
    mask = cols < N
    logits = tl.load(logits_ptr + row * N + cols, mask=mask, other=0.0)
    bias = tl.load(bias_ptr + cols, mask=mask, other=0.0)
    scores = 1.0 / (1.0 + tl.exp(-logits)) + bias
    tl.store(scores_ptr + row * N + cols, scores, mask=mask)


# Kernel 3: for each token, compute sum of top-2 scores within each group (32 experts) -> [M, 8]
# scores are expected in shape [M, group_count, experts_per_group] with last dim=32 (contiguous).
@triton.jit
def group_top2_sum_kernel(
    scores_ptr,        # *f32, [M, 8, 32]
    group_scores_ptr,  # *f32, [M, 8]
    M: tl.int32,
    group_count: tl.constexpr,      # 8
    experts_per_group: tl.constexpr,  # 32
):
    row = tl.program_id(0)
    for g in range(group_count):
        group_start = g * experts_per_group
        idx = tl.arange(0, 32)
        cols = group_start + idx
        mask = idx < experts_per_group
        vals = tl.load(scores_ptr + row * (group_count * experts_per_group) + g * experts_per_group + idx, mask=mask, other=-float('inf'))
        m1 = tl.max(vals, axis=0)
        vals2 = tl.where(vals == m1, -float('inf'), vals)
        m2 = tl.max(vals2, axis=0)
        tl.store(group_scores_ptr + row * group_count + g, m1 + m2)


# Kernel 4: iterative top-k selection (K is constexpr), returns indices in out_idx [M, K]
# inp_ptr: [M, M_dim] where M_dim can be N (for groups) or number of experts (256).
@triton.jit
def topk_select_kernel(
    inp_ptr,       # *f32, [M, M_dim]
    out_idx_ptr,   # *i32, [M, K]
    M: tl.int32,
    M_dim: tl.int32,
    K: tl.constexpr,             # e.g., 4 or 8
):
    row = tl.program_id(0)
    # Iterative argmax selection
    for k in range(K):
        best_val = -float('inf')
        best_idx = 0
        for j in range(M_dim):
            val = tl.load(inp_ptr + row * M_dim + j)
            better = val > best_val
            best_val = tl.where(better, val, best_val)
            best_idx = tl.where(better, j, best_idx)
        tl.store(out_idx_ptr + row * K + k, best_idx)
        # set selected position to -inf for next iterations
        tl.store(inp_ptr + row * M_dim + best_idx, -float('inf'))


# Kernel 5: build group_mask [M, group_count] with 1 for selected groups
# group_idx: [M, K], where K is 4 in our case. We scatter 1 into group_mask[:, g] for each g in 0..group_count-1.
@triton.jit
def build_group_mask_kernel(
    group_idx_ptr,     # *i32, [M, K]
    group_mask_ptr,    # *i32, [M, group_count], we'll store 0/1
    M: tl.int32,
    K: tl.constexpr,            # 4
    group_count: tl.constexpr,  # 8
):
    row = tl.program_id(0)
    for g in range(group_count):
        # count how many of the K indices equal g
        count = 0
        for k in range(K):
            idx = tl.load(group_idx_ptr + row * K + k)
            count += (idx == g).to(tl.int32)
        # If count == 1, then set group_mask[row, g] = 1
        if count == 1:
            tl.store(group_mask_ptr + row * group_count + g, 1)
        else:
            tl.store(group_mask_ptr + row * group_count + g, 0)


# Kernel 6: expand group_mask to scores and set non-selected groups' scores to -inf
# group_mask: [M, group_count], scores: [M, N], out_scores: [M, N]
@triton.jit
def expand_group_mask_and_set_ninf_kernel(
    group_mask_ptr,    # *i32, [M, group_count]
    scores_ptr,        # *f32, [M, N]
    out_ptr,           # *f32, [M, N]
    M: tl.int32,
    N: tl.constexpr,              # 256
    group_count: tl.constexpr,    # 8
    experts_per_group: tl.constexpr,  # 32
):
    row = tl.program_id(0)
    # Iterate over all experts and set to -inf if their group is not selected
    for col in range(N):
        group_index = col // experts_per_group  # 0..7
        selected = tl.load(group_mask_ptr + row * group_count + group_index)  # int32 0/1
        orig = tl.load(scores_ptr + row * N + col)
        val = tl.where(selected == 1, orig, -float('inf'))
        tl.store(out_ptr + row * N + col, val)


# Kernel 7: gather selected scores from original scores using expert indices (K=8)
@triton.jit
def gather_scores_kernel(
    scores_ptr,       # *f32, [M, N]
    idx_ptr,          # *i32, [M, K]
    out_scores_ptr,   # *f32, [M, K]
    M: tl.int32,
    N: tl.constexpr,              # 256
    K: tl.constexpr               # 8
):
    row = tl.program_id(0)
    for k in range(K):
        idx = tl.load(idx_ptr + row * K + k)
        val = tl.load(scores_ptr + row * N + idx)
        tl.store(out_scores_ptr + row * K + k, val)


# Kernel 8: normalize_and_scale for [M, K] vector, out is scaled normalized values
@triton.jit
def normalize_and_scale_kernel(
    inp_ptr,        # *f32, [M, K]
    out_ptr,        # *f32, [M, K]
    scale: tl.float32,
    M: tl.int32,
    K: tl.constexpr
):
    row = tl.program_id(0)
    total = 0.0
    for k in range(K):
        val = tl.load(inp_ptr + row * K + k)
        total += val
    for k in range(K):
        val = tl.load(inp_ptr + row * K + k)
        val = val / (total + 1e-20) * scale
        tl.store(out_ptr + row * K + k, val)

        Returns:

        - topk_weight: [num_tokens, 8], float32

        M = hidden_states.shape[0]
        assert weight.shape[0] == self.num_experts, "weight.num_experts must be 256"
        assert weight.shape[1] == self.hidden_size, "weight.hidden_size must be 128"
        assert hidden_states.shape[1] == self.hidden_size, "hidden_states.hidden_size must be 128"
        assert expert_bias.shape[0] == self.num_experts, "expert_bias length must be 256"

        grid = (M, self.num_experts)
        matmul_logits_kernel[grid](

            M=M,
            N=self.num_experts, K=self.hidden_size

        scores = torch.empty_like(logits)

        # 5) Build group_mask [M, 8], 1 where group selected, else 0
        group_mask = torch.empty((M, self.group_count), dtype=torch.int32, device=scores.device)
        grid5 = (M,)
        build_group_mask_kernel[grid5](group_idx, group_mask, M=M, K=4, group_count=self.group_count)

        # 6) Expand group_mask to scores and set non-selected groups' scores to -inf -> masked_scores [M, 256]
        masked_scores = torch.empty_like(scores)

        # 7) Triton top-8 expert selection: topk_idx [M, 8]
        topk_idx = torch.empty((M, self.final_k), dtype=torch.int32, device=scores.device)
        grid7 = (M,)
        topk_select_kernel[grid7](masked_scores, topk_idx, M=M, M_dim=self.num_experts, K=self.final_k)

        # 8) Gather selected scores from original scores (use original scores, not masked_scores)
        selected_scores = torch.empty((M, self.final_k), dtype=torch.float32, device=scores.device)
        grid8 = (M,)
        gather_scores_kernel[grid8](scores, topk_idx, selected_scores, M=M, N=self.num_experts, K=self.final_k)

        # 9) Normalize and scale topk_weight

        grid9 = (M,)
        normalize_and_scale_kernel[grid9](selected_scores, topk_weight, self.routed_scaling_factor, M=M, K=self.final_k)

        # Return as required: int64 indices and float32 weights
        return topk_idx.to(torch.int64), topk_weight

  - gather_scores_kernel: gather selected scores from original scores using expert indices
  - normalize_and_scale_kernel: normalize gathered selected scores and apply routing scaling

- Important: We must avoid any PyTorch ops in forward. Reshaping via .view or torch operations is not allowed. Triton kernels must produce the reshaped data we need. 3) Kernel details
- matmul_logits_kernel: grid = (M, N). For each (row, col), accumulate over K: out[row, col] = sum_i hidden_states[row, i] * weight[col, i].
- sigmoid_bias_kernel: grid = (M,). Each program processes one row, applying sigmoid and bias across N=256.
- group_top2_sum_kernel: grid = (M,). For each row, loop g in 0..7, load group’s 32 scores, compute top-2 sum, store into group_scores[row, g].
- topk_select_kernel: grid = (M,). For each row, iterative argmax K times, set chosen element to -inf for next iterations, store indices.
- build_group_mask_kernel: grid = (M,). For each row, check how many times each group g appears in group_idx; if count==1, set group_mask[row, g]=1 else 0. We’ll implement this by loading and comparing four indices.
- expand_group_mask_and_set_ninf_kernel: grid = (M,). For each row, iterate col in 0..N-1, determine group_index = col // 32, load selected flag, if not selected, load original score and store -inf; else store original score. Note: we need to keep the original score of selected group experts; therefore, we pass a pointer to original scores to load the value if selected. We’ll store into out_ptr (masked_scores).
- gather_scores_kernel: grid = (M,). For each row, iterate k in 0..7, load index, load original score, store to out.
- normalize_and_scale_kernel: grid = (M,). For each row, sum K values, then write each value / sum * routed_scaling_factor.

4) Correctness considerations
- Group mapping: group j covers experts in [j*32 .. (j+1)*32-1].
- Masking: Only non-selected groups’ experts are set to -inf; selected group’s experts keep original scores. Then we select top-8 from masked_scores, ensuring we only pick from selected groups.
- Final normalization: we normalize selected_scores (from original scores) per token, not masked_scores; masked_scores is only used to derive indices, and selected_scores are gathered afterward.

import torch
import torch.nn as nn

import triton
import triton.language as tl

# Kernel 1: logits = hidden_states @ weight.T
# hidden_states: [M, K] row-major
# weight: [N, K] row-major, N = num_experts
# out: [M, N] row-major
@triton.jit
def matmul_logits_kernel(
    a_ptr,        # *f32, [M, K]
    b_ptr,        # *f32, [N, K] (row-major)
    out_ptr,      # *f32, [M, N]
    M: tl.int32,
    N: tl.int32,
    K: tl.int32,
):
    row = tl.program_id(0)  # token id
    col = tl.program_id(1)  # expert id
    acc = 0.0
    for k in range(K):
        a_val = tl.load(a_ptr + row * K + k)
        b_val = tl.load(b_ptr + col * K + k)
        acc += a_val * b_val
    tl.store(out_ptr + row * N + col, acc)


# Kernel 2: scores = sigmoid(logits) + expert_bias
# logits: [M, N], bias: [N], scores: [M, N]
@triton.jit
def sigmoid_bias_kernel(
    logits_ptr,        # *f32, [M, N]
    bias_ptr,          # *f32, [N]
    scores_ptr,        # *f32, [M, N]
    M: tl.int32,
    N: tl.int32,
):
    row = tl.program_id(0)  # one program per token row
    cols = tl.arange(0, N)  # vector of N
    mask = cols < N
    logits = tl.load(logits_ptr + row * N + cols, mask=mask, other=0.0)
    bias = tl.load(bias_ptr + cols, mask=mask, other=0.0)
    scores = 1.0 / (1.0 + tl.exp(-logits)) + bias
    tl.store(scores_ptr + row * N + cols, scores, mask=mask)


# Kernel 3: for each token, compute sum of top-2 scores within each group (32 experts) -> [M, 8]
# We will pass a pointer to 'scores' in layout [M, group_count, experts_per_group], but we do not rely on PyTorch view.
# Here, group_count=8, experts_per_group=32.
@triton.jit
def group_top2_sum_kernel(
    scores_ptr,        # *f32, [M, 8, 32] where strides are provided by layout (we pass as row-major)
    group_scores_ptr,  # *f32, [M, 8]
    M: tl.int32,
    group_count: tl.int32,            # 8
    experts_per_group: tl.int32,      # 32
):
    row = tl.program_id(0)
    for g in range(group_count):
        group_start = g * experts_per_group
        idx = tl.arange(0, 32)
        cols = group_start + idx
        # scores_ptr is row-major with last dim=32, so pointer offset is row * (group_count * experts_per_group) + g * experts_per_group + idx
        ptr = scores_ptr + row * (group_count * experts_per_group) + g * experts_per_group + idx
        vals = tl.load(ptr)  # assume N is 256 and group layout is contiguous
        m1 = tl.max(vals, axis=0)
        vals2 = tl.where(vals == m1, -float('inf'), vals)
        m2 = tl.max(vals2, axis=0)
        tl.store(group_scores_ptr + row * group_count + g, m1 + m2)


# Kernel 4: iterative top-k selection (K is constexpr), returns indices in out_idx [M, K]
# inp_ptr: [M, M_dim] where M_dim can be N (for groups) or number of experts (256).
@triton.jit
def topk_select_kernel(
    inp_ptr,       # *f32, [M, M_dim]
    out_idx_ptr,   # *i32, [M, K]
    M: tl.int32,
    M_dim: tl.int32,
    K: tl.constexpr,             # e.g., 4 or 8
):
    row = tl.program_id(0)
    # Iterative argmax selection
    for k in range(K):
        best_val = -float('inf')
        best_idx = 0
        for j in range(M_dim):
            val = tl.load(inp_ptr + row * M_dim + j)
            better = val > best_val
            best_val = tl.where(better, val, best_val)
            best_idx = tl.where(better, j, best_idx)
        tl.store(out_idx_ptr + row * K + k, best_idx)
        # set selected position to -inf for next iterations
        tl.store(inp_ptr + row * M_dim + best_idx, -float('inf'))


# Kernel 5: build group_mask [M, group_count] with 1 for selected groups
# group_idx: [M, K], where K is 4 in our case. We scatter 1 into group_mask[:, g] for each g in 0..group_count-1.
@triton.jit
def build_group_mask_kernel(
    group_idx_ptr,     # *i32, [M, K]
    group_mask_ptr,    # *i32, [M, group_count], we'll store 0/1
    M: tl.int32,
    K: tl.constexpr,            # 4
    group_count: tl.constexpr,  # 8
):
    row = tl.program_id(0)
    for g in range(group_count):
        # count how many of the K indices equal g
        count = 0
        for k in range(K):
            idx = tl.load(group_idx_ptr + row * K + k)
            count += (idx == g).to(tl.int32)
        # If count == 1, then set group_mask[row, g] = 1
        if count == 1:
            tl.store(group_mask_ptr + row * group_count + g, 1)
        else:
            tl.store(group_mask_ptr + row * group_count + g, 0)


# Kernel 6: expand group_mask to scores and set non-selected groups' scores to -inf
# group_mask: [M, group_count], scores: [M, N], out_scores: [M, N]
@triton.jit
def expand_group_mask_and_set_ninf_kernel(
    group_mask_ptr,    # *i32, [M, group_count]
    scores_ptr,        # *f32, [M, N]
    out_ptr,           # *f32, [M, N]
    M: tl.int32,
    N: tl.int32,
    group_count: tl.int32,             # 8
    experts_per_group: tl.int32,       # 32
):
    row = tl.program_id(0)
    # Iterate over all experts and set to -inf if their group is not selected
    for col in range(N):
        group_index = col // experts_per_group  # 0..7
        selected = tl.load(group_mask_ptr + row * group_count + group_index)  # int32 0/1
        orig = tl.load(scores_ptr + row * N + col)
        val = tl.where(selected == 1, orig, -float('inf'))
        tl.store(out_ptr + row * N + col, val)


# Kernel 7: gather selected scores from original scores using expert indices (K=8)
@triton.jit
def gather_scores_kernel(
    scores_ptr,       # *f32, [M, N]
    idx_ptr,          # *i32, [M, K]
    out_scores_ptr,   # *f32, [M, K]
    M: tl.int32,
    N: tl.int32,
    K: tl.constexpr               # 8
):
    row = tl.program_id(0)
    for k in range(K):
        idx = tl.load(idx_ptr + row * K + k)
        val = tl.load(scores_ptr + row * N + idx)
        tl.store(out_scores_ptr + row * K + k, val)


# Kernel 8: normalize_and_scale for [M, K] vector, out is scaled normalized values
@triton.jit
def normalize_and_scale_kernel(
    inp_ptr,        # *f32, [M, K]
    out_ptr,        # *f32, [M, K]
    scale: tl.float32,
    M: tl.int32,
    K: tl.constexpr
):
    row = tl.program_id(0)
    total = 0.0
    for k in range(K):
        val = tl.load(inp_ptr + row * K + k)
        total += val
    for k in range(K):
        val = tl.load(inp_ptr + row * K + k)
        val = val / (total + 1e-20) * scale
        tl.store(out_ptr + row * K + k, val)
