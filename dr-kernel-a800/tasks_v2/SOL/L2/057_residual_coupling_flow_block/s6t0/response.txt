import math
import torch
import torch.nn as nn
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def conv1d_forward_5(
    x_ptr,        # *float32, [B, C_in, T_in]
    w_ptr,        # *float32, [C_out, C_in, K=5]
    y_ptr,        # *float32, [B, C_out, T_out]
    B, C_out, C_in, T_in, T_out,
    stride_xb, stride_xc, stride_xt,
    stride_wco, stride_wci, stride_wk,
    stride_yb, stride_yco, stride_yt,
    BLOCK_T: tl.constexpr,
):
    # Grid: (pid0, pid1) -> (B*C_out, ceil_div(T_out, BLOCK_T))
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # Decode batch and output channel
    b = pid0 // C_out
    co = pid0 % C_out

    # Time tile
    t0 = pid1 * BLOCK_T
    t_offsets = t0 + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    # Accumulator for output vector across BLOCK_T
    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # For valid conv, for each output time position t, input taps are t+k (k=0..4)
    # We accumulate over input channels and kernel taps.
    # Loop over input channels (C_in can be large; Triton supports loops).
    for ci in range(0, C_in):
        # For each kernel tap
        for k in range(0, 5):
            # input time index
            t_in = t_offsets + k  # valid since t_in in [k, k+T_out-1] and T_in >= T_out + K - 1
            # Load x[b, ci, t_in]
            x_idx = b * stride_xb + ci * stride_xc + t_in * stride_xt
            x_vals = tl.load(x_ptr + x_idx, mask=mask_t, other=0.0)

            # Load w[co, ci, k]
            w_idx = co * stride_wco + ci * stride_wci + k * stride_wk
            w_val = tl.load(w_ptr + w_idx)

            acc += x_vals * w_val

    # Add bias for output channel co
    bias_idx = co  # assuming bias is 1D length C_out
    # In this implementation, we pass bias as part of y_ptr or separately; here we assume bias is pre-added
    # We'll add bias in a separate kernel below.

    # Store result to y[b, co, t_offsets]
    y_idx = b * stride_yb + co * stride_yco + t_offsets * stride_yt
    tl.store(y_ptr + y_idx, acc, mask=mask_t)


@triton.jit
def conv1d_bias(
    y_ptr,        # *float32, [B, C_out, T_out] (pre-conv output)
    bias_ptr,     # *float32, [C_out]
    B, C_out, T_out,
    stride_yb, stride_yco, stride_yt,
    BLOCK_T: tl.constexpr,
):
    # Grid: (pid0, pid1) -> (B*C_out, ceil_div(T_out, BLOCK_T))
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    b = pid0 // C_out
    co = pid0 % C_out

    t0 = pid1 * BLOCK_T
    t_offsets = t0 + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    y_idx = b * stride_yb + co * stride_yco + t_offsets * stride_yt
    y_vals = tl.load(y_ptr + y_idx, mask=mask_t, other=0.0)

    b_idx = co  # bias is 1D
    bias_val = tl.load(bias_ptr + b_idx)

    y_vals += bias_val

    tl.store(y_ptr + y_idx, y_vals, mask=mask_t)


@triton.jit
def conv1d_relu(
    y_ptr,        # *float32, [B, C_out, T_out] (pre-activation)
    B, C_out, T_out,
    stride_yb, stride_yco, stride_yt,
    BLOCK_T: tl.constexpr,
):
    # Grid: (pid0, pid1) -> (B*C_out, ceil_div(T_out, BLOCK_T))
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    b = pid0 // C_out
    co = pid0 % C_out

    t0 = pid1 * BLOCK_T
    t_offsets = t0 + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    y_idx = b * stride_yb + co * stride_yco + t_offsets * stride_yt
    y_vals = tl.load(y_ptr + y_idx, mask=mask_t, other=0.0)

    y_vals = tl.maximum(y_vals, 0.0)

    tl.store(y_ptr + y_idx, y_vals, mask=mask_t)


# Helper to run a single conv layer using Triton and return output (no ReLU, bias not added).
def triton_conv1d_forward(x, w, T_out, BLOCK_T=128):
    """
    x: [B, C_in, T_in], w: [C_out, C_in, 5], returns y: [B, C_out, T_out]
    Assumes valid conv: T_in == T_out + 4 (kernel_size=5 => padding=2, but here we use valid conv).
    We implement valid conv: output length = T_in - 4. Ensure x_mask handles padding.
    """
    assert x.is_cuda and w.is_cuda, "Triton kernels require CUDA tensors."
    B, C_in, T_in = x.shape
    C_out, C_in_w, K = w.shape
    assert C_in == C_in_w, "Mismatch in input channels for weight."
    assert K == 5, "Kernel size must be 5 for this kernel."
    assert T_in == T_out + 4, "For valid conv with kernel=5, T_in must equal T_out + 4."

    # Ensure contiguous
    x = x.contiguous()
    w = w.contiguous()

    # Allocate output
    y = torch.empty((B, C_out, T_out), device=x.device, dtype=x.dtype)

    # Strides
    stride_xb = x.stride(0)
    stride_xc = x.stride(1)
    stride_xt = x.stride(2)

    stride_wco = w.stride(0)
    stride_wci = w.stride(1)
    stride_wk = w.stride(2)

    stride_yb = y.stride(0)
    stride_yco = y.stride(1)
    stride_yt = y.stride(2)

    grid = (B * C_out, triton.cdiv(T_out, BLOCK_T))
    conv1d_forward_5[grid](
        x, w, y,
        B, C_out, C_in, T_in, T_out,
        stride_xb, stride_xc, stride_xt,
        stride_wco, stride_wci, stride_wk,
        stride_yb, stride_yco, stride_yt,
        BLOCK_T=BLOCK_T,
    )
    return y


# Helper to add bias using Triton
def triton_add_bias(y, bias):
    """
    y: [B, C_out, T_out], bias: [C_out]
    """
    assert y.is_cuda and bias.is_cuda, "Triton kernels require CUDA tensors."
    B, C_out, T_out = y.shape

    y = y.contiguous()
    bias = bias.contiguous()

    stride_yb = y.stride(0)
    stride_yco = y.stride(1)
    stride_yt = y.stride(2)

    grid = (B * C_out, triton.cdiv(T_out, 128))
    conv1d_bias[grid](y, bias, B, C_out, T_out, stride_yb, stride_yco, stride_yt, BLOCK_T=128)
    return y


# Helper to apply ReLU using Triton
def triton_relu(y):
    """
    y: [B, C_out, T_out]
    """
    assert y.is_cuda, "Triton kernels require CUDA tensors."
    B, C_out, T_out = y.shape
    y = y.contiguous()

    stride_yb = y.stride(0)
    stride_yco = y.stride(1)
    stride_yt = y.stride(2)

    grid = (B * C_out, triton.cdiv(T_out, 128))
    conv1d_relu[grid](y, B, C_out, T_out, stride_yb, stride_yco, stride_yt, BLOCK_T=128)
    return y


class ModelNew(torch.nn.Module):
    def forward(self, x, x_mask, reverse, transform_0_conv0_weight, transform_0_conv0_bias,
                transform_0_conv1_weight, transform_0_conv1_bias,
                transform_0_conv2_weight, transform_0_conv2_bias,
                transform_1_conv0_weight, transform_1_conv0_bias,
                transform_1_conv1_weight, transform_1_conv1_bias,
                transform_1_conv2_weight, transform_1_conv2_bias,
                transform_2_conv0_weight, transform_2_conv0_bias,
                transform_2_conv1_weight, transform_2_conv1_bias,
                transform_2_conv2_weight, transform_2_conv2_bias,
                transform_3_conv0_weight, transform_3_conv0_bias,
                transform_3_conv1_weight, transform_3_conv1_bias,
                transform_3_conv2_weight, transform_3_conv2_bias):
        """
        Triton-optimized forward and reverse pass. All convs are done by Triton kernels.
        Elementwise ops (mask multiply, add/sub, concat) are done in PyTorch.
        """
        # Ensure inputs are on CUDA for Triton
        device = x.device
        half_channels = x.shape[1] // 2

        # Collect transforms
        transforms = [
            (transform_0_conv0_weight, transform_0_conv0_bias,
             transform_0_conv1_weight, transform_0_conv1_bias,
             transform_0_conv2_weight, transform_0_conv2_bias),
            (transform_1_conv0_weight, transform_1_conv0_bias,
             transform_1_conv1_weight, transform_1_conv1_bias,
             transform_1_conv2_weight, transform_1_conv2_bias),
            (transform_2_conv0_weight, transform_2_conv0_bias,
             transform_2_conv1_weight, transform_2_conv1_bias,
             transform_2_conv2_weight, transform_2_conv2_bias),
            (transform_3_conv0_weight, transform_3_conv0_bias,
             transform_3_conv1_weight, transform_3_conv1_bias,
             transform_3_conv2_weight, transform_3_conv2_bias),
        ]

        if not reverse:
            # Forward: apply transforms sequentially
            for conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b in transforms:
                # Split input
                x0 = x[:, :half_channels, :]
                x1 = x[:, half_channels:, :]

                # Compute transform(x0) via Triton convs
                # conv0: in_channels=half_channels, out_channels=hidden_channels, kernel=5
                # conv1: in_channels=hidden_channels, out_channels=hidden_channels, kernel=5
                # conv2: in_channels=hidden_channels, out_channels=half_channels, kernel=5
                hidden_channels = conv0_w.shape[0]

                T_in0 = x0.shape[2]  # time length
                T_out0 = T_in0 - 4    # valid conv with K=5 => output length = input - 4

                # conv0
                h = triton_conv1d_forward(x0, conv0_w, T_out=T_out0)
                # bias and ReLU
                h = triton_add_bias(h, conv0_b)
                h = triton_relu(h)

                T_in1 = h.shape[2]
                T_out1 = T_in1 - 4
                h = triton_conv1d_forward(h, conv1_w, T_out=T_out1)
                h = triton_add_bias(h, conv1_b)
                h = triton_relu(h)

                T_in2 = h.shape[2]
                T_out2 = T_in2 - 4
                h = triton_conv1d_forward(h, conv2_w, T_out=T_out2)
                h = triton_add_bias(h, conv2_b)

                # Mask
                h = h * x_mask

                # Affine coupling: x1 = x1 + h
                x1 = x1 + h

                # Concatenate back
                x = torch.cat([x0, x1], dim=1)
                # Apply mask to output
                x = x * x_mask
        else:
            # Reverse: apply transforms in reverse order, subtract
            for conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b in reversed(transforms):
                # Split input
                x0 = x[:, :half_channels, :]
                x1 = x[:, half_channels:, :]

                hidden_channels = conv0_w.shape[0]

                T_in0 = x0.shape[2]
                T_out0 = T_in0 - 4

                # conv0
                h = triton_conv1d_forward(x0, conv0_w, T_out=T_out0)
                h = triton_add_bias(h, conv0_b)
                h = triton_relu(h)

                T_in1 = h.shape[2]
                T_out1 = T_in1 - 4
                h = triton_conv1d_forward(h, conv1_w, T_out=T_out1)
                h = triton_add_bias(h, conv1_b)
                h = triton_relu(h)

                T_in2 = h.shape[2]
                T_out2 = T_in2 - 4
                h = triton_conv1d_forward(h, conv2_w, T_out=T_out2)
                h = triton_add_bias(h, conv2_b)

                # Mask
                h = h * x_mask

                # Inverse affine coupling: x1 = x1 - h
                x1 = x1 - h

                # Concatenate back
                x = torch.cat([x0, x1], dim=1)
                # Apply mask to output
                x = x * x_mask

        return x

    w_ptr,        # *float32, [C_out, C_in, 5]
    y_ptr,        # *float32, [B, C_out, T_out] where T_out = T_in - 1 (padding=2, kernel=5)
    B, C_out, C_in, T_in, T_out,
    stride_xb, stride_xc, stride_xt,
    stride_wco, stride_wci, stride_wk,
    stride_yb, stride_yco, stride_yt,
    BLOCK_T: tl.constexpr,
):
    # Grid: (pid0, pid1) -> (B*C_out, ceil_div(T_out, BLOCK_T))
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # Decode batch and output channel
    b = pid0 // C_out
    co = pid0 % C_out

    # Time tile
    t0 = pid1 * BLOCK_T
    t_offsets = t0 + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    # Accumulator for output vector across BLOCK_T
    acc = tl.zeros([BLOCK_T], dtype=tl.float32)

    # For each input channel and kernel tap, compute contribution
    # Padding=2: output index t_out corresponds to input index t_in = t_out - 2 + k
    for ci in range(0, C_in):
        for k in range(0, 5):
            t_in = t_offsets - 2 + k  # valid since T_out = T_in - 1
            # Load x[b, ci, t_in]
            x_idx = b * stride_xb + ci * stride_xc + t_in * stride_xt
            x_vals = tl.load(x_ptr + x_idx, mask=mask_t, other=0.0)
            # Load w[co, ci, k]
            w_idx = co * stride_wco + ci * stride_wci + k * stride_wk
            w_val = tl.load(w_ptr + w_idx)
            acc += x_vals * w_val

    # Store result to y[b, co, t_offsets]
    y_idx = b * stride_yb + co * stride_yco + t_offsets * stride_yt
    tl.store(y_ptr + y_idx, acc, mask=mask_t)


@triton.jit
def conv1d_bias(
    y_ptr,        # *float32, [B, C_out, T_out] (pre-conv output)
    bias_ptr,     # *float32, [C_out]
    B, C_out, T_out,
    stride_yb, stride_yco, stride_yt,
    BLOCK_T: tl.constexpr,
):
    # Grid: (pid0, pid1) -> (B*C_out, ceil_div(T_out, BLOCK_T))
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    b = pid0 // C_out
    co = pid0 % C_out

    t0 = pid1 * BLOCK_T
    t_offsets = t0 + tl.arange(0, BLOCK_T)
    mask_t = t_offsets < T_out

    y_idx = b * stride_yb + co * stride_yco + t_offsets * stride_yt
    y_vals = tl.load(y_ptr + y_idx, mask=mask_t, other=0.0)

    bias_val = tl.load(bias_ptr + co)
    y_vals += bias_val

    tl.store(y_ptr + y_idx, y_vals, mask=mask_t)


class ModelNew(torch.nn.Module):
    def forward(self, x, x_mask, reverse,
                # transform 0
                transform_0_conv0_weight, transform_0_conv0_bias,
                transform_0_conv1_weight, transform_0_conv1_bias,
                transform_0_conv2_weight, transform_0_conv2_bias,
                # transform 1
                transform_1_conv0_weight, transform_1_conv0_bias,
                transform_1_conv1_weight, transform_1_conv1_bias,
                transform_1_conv2_weight, transform_1_conv2_bias,
                # transform 2
                transform_2_conv0_weight, transform_2_conv0_bias,
                transform_2_conv1_weight, transform_2_conv1_bias,
                transform_2_conv2_weight, transform_2_conv2_bias,
                # transform 3
                transform_3_conv0_weight, transform_3_conv0_bias,
                transform_3_conv1_weight, transform_3_conv1_bias,
                transform_3_conv2_weight, transform_3_conv2_bias):
        """
        Triton-optimized forward and reverse pass. All convs are done by Triton kernels.
        Elementwise ops (mask multiply, add/sub, concat) are done in PyTorch.
        """
        device = x.device
        half_channels = x.shape[1] // 2

        # List of transforms (each is (conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b))
        transforms = [
            (transform_0_conv0_weight, transform_0_conv0_bias,
             transform_0_conv1_weight, transform_0_conv1_bias,
             transform_0_conv2_weight, transform_0_conv2_bias),
            (transform_1_conv0_weight, transform_1_conv0_bias,
             transform_1_conv1_weight, transform_1_conv1_bias,
             transform_1_conv2_weight, transform_1_conv2_bias),
            (transform_2_conv0_weight, transform_2_conv0_bias,
             transform_2_conv1_weight, transform_2_conv1_bias,
             transform_2_conv2_weight, transform_2_conv2_bias),
            (transform_3_conv0_weight, transform_3_conv0_bias,
             transform_3_conv1_weight, transform_3_conv1_bias,
             transform_3_conv2_weight, transform_3_conv2_bias),
        ]

        if not reverse:
            # Forward: apply transforms sequentially
            for conv0_w, conv0_b, conv1_w, conv1_b, conv2_w, conv2_b in transforms:
                # Split input
                x0 = x[:, :half_channels, :]
                x1 = x[:, half_channels:, :]

                hidden_channels = conv0_w.shape[0]

                T_in0 = x0.shape[2]  # original time length
                # With padding=2 and kernel=5, output length is T_in - 1
                T_out0 = T_in0 - 1

                # conv0
                h = self._triton_conv1d_k5_p2(x0, conv0_w, T_out=T_out0)
                # bias and ReLU
                h = self._triton_add_bias(h, conv0_b)
                h = F.relu(h)  # elementwise ReLU in PyT
