import torch
import triton
import triton.language as tl


# Kernel 1: Build concatenated input matrix X_cat[b] = [encoder_hidden_states[b], hidden_states[b]]
# It writes directly into the preallocated output Y_cat[b] with shape [(T+I), H].
# Grid: (1, T_plus_I, 1) -> each program handles one row (p) for batch b.
@triton.jit
def kernel_cat_write_X(
    e_ptr, i_ptr, out_ptr,
    B, T, I, H,
    stride_eb, stride_et, stride_eh,
    stride_ib, stride_it, stride_ih,
    stride_ob, stride_om, stride_on,
    T_plus_I: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_p = tl.program_id(1)  # sequence position in concatenated output
    # Bounds check (in case grid is larger than needed)
    if pid_b >= B or pid_p >= T_plus_I:
        return

    # Determine which tensor to read from: first T rows come from encoder_hidden_states, rest from hidden_states
    if pid_p < T:
        src = 0  # encoder
    else:
        src = 1  # image
        pid_p -= T  # local position within hidden_states

    # Base pointers for this batch
    if src == 0:
        e_base = e_ptr + pid_b * stride_eb
        # Row index within encoder_hidden_states is pid_p
        row_ptr = e_base + pid_p * stride_et
        # Column stride is stride_eh (dim=2 stride)
        # We will write this row into out at position (pid_b, pid_p)
        out_row_ptr = out_ptr + pid_b * stride_ob + pid_p * stride_om
    else:
        i_base = i_ptr + pid_b * stride_ib
        # Row index within hidden_states is pid_p (shifted by T)
        row_ptr = i_base + pid_p * stride_it
        out_row_ptr = out_ptr + pid_b * stride_ob + pid_p * stride_om

    # Copy row by iterating over H (last dimension)
    # We assume inputs are contiguous along the last dimension (stride_ih == 1). For safety, use stride for generality.
    for j in range(0, H):
        val = tl.load(row_ptr + j * stride_eh)
        tl.store(out_row_ptr + j * stride_on, val)


# Kernel 2: Batched matmul for each batch b: Y[b] = X_cat[b] @ W^T
# X_cat: [M, K] where M = T + I, K = H
# W: [K, N] where N = H
# Y: [M, N]
@triton.jit
def kernel_batched_matmul(
    X_ptr, W_ptr, Y_ptr,
    B, M, K, N,
    stride_xb, stride_xm, stride_xk,
    stride_wk, stride_wn,
    stride_yb, stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    if pid_b >= B:
        return

    m_start = pid_m * BLOCK_M
    n_start = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in chunks
    for k0 in range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        # Load A block: X[pid_m, k0:k0+BLOCK_K] -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = X_ptr + pid_b * stride_xb + m_start * stride_xm + k_offsets[None, :] * stride_xk
        a_mask = (m_start + tl.arange(0, BLOCK_M))[:, None] < M
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B block: W[k0:k0+BLOCK_K, n_start:n_start+BLOCK_N] -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = W_ptr + k_offsets[:, None] * stride_wk + (n_start + tl.arange(0, BLOCK_N))[None, :] * stride_wn
        b_mask = (n_start + tl.arange(0, BLOCK_N))[None, :] < N
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store results
    y_ptrs = Y_ptr + pid_b * stride_yb + (m_start + tl.arange(0, BLOCK_M))[:, None] * stride_ym + (n_start + tl.arange(0, BLOCK_N))[None, :] * stride_yn
    y_mask = (m_start + tl.arange(0, BLOCK_M))[:, None] < M
    tl.store(y_ptrs, acc, mask=y_mask)


class ModelNew(torch.nn.Module):
    def __init__(self, max_batch: int = 32):
        super().__init__()
        self.max_batch = max_batch

    def forward(self, hidden_states: torch.Tensor, encoder_hidden_states: torch.Tensor, process_weight: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        # Ensure CUDA tensors (Triton requires CUDA)
        assert hidden_states.is_cuda and encoder_hidden_states.is_cuda and process_weight.is_cuda, "Inputs must be CUDA tensors for Triton kernels."

        B = hidden_states.shape[0]
        assert B == encoder_hidden_states.shape[0], "Batch sizes of hidden_states and encoder_hidden_states must match."
        assert hidden_states.shape[2] == encoder_hidden_states.shape[2] == process_weight.shape[0] == process_weight.shape[1], "Hidden dim and weight shape must match."

        T = encoder_hidden_states.shape[1]
        I = hidden_states.shape[1]
        H = hidden_states.shape[2]

        # Allocate concatenated X_cat per batch as [T+I, H]
        # We will build X_cat[b] using Triton kernel
        X_cat = [torch.empty((T + I, H), device=hidden_states.device, dtype=hidden_states.dtype) for _ in range(B)]

        # Launch Triton kernel to build concatenated X_cat
        # grid: (B, T+I, 1)
        grid_cat = (B, T + I, 1)

        # Compute strides (assuming last dim is contiguous, but we'll use strides for generality)
        stride_eb, stride_et, stride_eh = encoder_hidden_states.stride(0), encoder_hidden_states.stride(1), encoder_hidden_states.stride(2)
        stride_ib, stride_it, stride_ih = hidden_states.stride(0), hidden_states.stride(1), hidden_states.stride(2)
        # For X_cat[b], out shape is (T+I, H), and we write row-by-row. Strides:
        # We allocate contiguous [T+I, H], so stride_om = H, stride_on = 1. But we use generic strides by passing (row, col) as (pid_b, pid_p).
        # We don't need specific strides for out_ptr since we compute addresses via pid_b and pid_p; Triton pointers are byte offsets. We'll pass stride_ob = H, stride_om = 1, stride_on = 1.
        stride_ob = H  # base offset for batch b; we will compute row pointer as out_ptr + pid_b*stride_ob + pid_p*stride_om
        stride_om = 1
        stride_on = 1

        kernel_cat_write_X[grid_cat](
            encoder_hidden_states, hidden_states, X_cat[0] if B == 1 else X_cat,  # trick: we pass a list; Triton will handle indexing via pid_b in kernel
            B, T, I, H,
            stride_eb, stride_et, stride_eh,
            stride_ib, stride_it, stride_ih,
            stride_ob, stride_om, stride_on,
            T + I,
            num_warps=1, num_stages=1,
        )

        # Now perform matmul for each batch. For simplicity and to avoid dynamic grid depending on B, we run kernels up to MAX_BATCH and ignore others.
        # We will only use B <= self.max_batch. For given test cases, B is small.
        # Prepare output Y_cat: [B, T+I, H]
        # However, Triton requires fixed shapes; we'll perform per-batch matmul into a separate output tensor Y_batch[b] of shape [(T+I), H].
        # Then slice rows to get processed_encoder and processed_hidden.
        Y_batch = [torch.empty((T + I, H), device=hidden_states.device, dtype=hidden_states.dtype) for _ in range(self.max_batch)]
        # Launch Triton matmul kernels per batch (up to MAX_BATCH). If B < MAX_BATCH, others will not be used.
        # We'll construct pointers for each batch b=0..self.max_batch-1, but only launch for b < B.
        W = process_weight.contiguous()  # [H, H]

        # BLOCK sizes for GEMM
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32

        for b in range(0, self.max_batch):
            if b >= B:
                break
            grid_mm = (1, triton.cdiv(T + I, BLOCK_M), triton.cdiv(H, BLOCK_N))
            X_b = X_cat[b]
            Y_b = Y_batch[b]
            # Strides for X_b and W
            stride_xb = X_b.stride(0)  # = H
            stride_xm = X_b.stride(1)  # = 1
            stride_xk = X_b.stride(2)  # = 1
            # W strides
            stride_wk = W.stride(0)  # = H
            stride_wn = W.stride(1)  # = 1
            # Y_b strides
            stride_yb = Y_b.stride(0)  # = H
            stride_ym = Y_b.stride(1)  # = 1
            stride_yn = Y_b.stride(2)  # = 1

            kernel_batched_matmul[grid_mm](
                X_b, W, Y_b,
                B, T + I, H, H,
                stride_xb, stride_xm, stride_xk,
                stride_wk, stride_wn,
                stride_yb, stride_ym, stride_yn,
                BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
                num_warps=4, num_stages=2,
            )

        # Now form outputs: processed_encoder [B, T, H] and processed_hidden [B, I, H]
        processed_encoder_list = []
        processed_hidden_list = []
        for b in range(0, self.max_batch):
            if b >= B:
                break
            Y = Y_batch[b]  # [T+I, H]
            # Split: first T rows for encoder, next I rows for image
            processed_encoder = Y[:T, :]  # [T, H]
            processed_hidden = Y[T:, :]   # [I, H]
            processed_encoder_list.append(processed_encoder)
            processed_hidden_list.append(processed_hidden)

        processed_encoder = torch.stack(processed_encoder_list, dim=0) if len(processed_encoder_list) > 0 else torch.empty((0, T, H), device=hidden_states.device, dtype=hidden_states.dtype)
        processed_hidden = torch.stack(processed_hidden_list, dim=0) if len(processed_hidden_list) > 0 else torch.empty((0, I, H), device=hidden_states.device, dtype=hidden_states.dtype)

        return processed_encoder, processed_hidden

        stride_ob, stride_om, stride_on = X_cat[0].stride(0), X_cat[0].stride(1), X_cat[0].stride(2)
        # Note: we use out_ptr as a pointer to X_cat[b], Triton will handle indexing via pid_b and pid_p.
        # To pass X_cat properly, we need to pass each tensor individually; Triton allows calling with separate arguments.
        # So we call the kernel once per batch by iterating over b and launching with appropriate out_ptr.
        for b in range(B):
            out = X_cat[b]
            kernel_cat_rowwise[(1, M)](
                encoder_hidden_states[b], hidden_states[b], out,
                1, T, I, H,
                stride_eb, stride_et, stride_eh,
                stride_ib, stride_it, stride_ih,
                stride_ob, stride_om, stride_on,
                num_warps=1, num_stages=1,
            )

        # Prepare output Y per batch
        Y = [torch.empty((M, H), device=hidden_states.device, dtype=hidden_states.dtype) for _ in range(B)]

        # Launch batched matmul kernel: grid = (B, tiles over M, tiles over N)
        # Choose block sizes based on runtime shapes (simple heuristic)
        BLOCK_M = 128 if M >= 128 else 64
        BLOCK_N = 64 if H >= 64 else 32
        BLOCK_K = 32 if H >= 32 else 16

        grid_mm = (B, triton.cdiv(M, BLOCK_M), triton.cdiv(H, BLOCK_N))
        for b in range(B):
            X_b = X_cat[b]
            Y_b = Y[b]
            W = process_weight  # [H, H], no bias

            # Strides for X_b
            stride_xb = X_b.stride(0)
            stride_xm = X_b.stride(1)
            stride_xk = X_b.stride(2)

            # Strides for W
            stride_wk = W.stride(0)
            stride_wn = W.stride(1)

            # Strides for Y_b
            stride_yb = Y_b.stride(0)
            stride_ym = Y_b.stride(1)
            stride_yn = Y_b.stride(2)

            kernel_batched_matmul[grid_mm](
                X_b, W, Y_b,
                B, M, H, H,
                stride_xb, stride_xm, stride_xk,
                stride_wk, stride_wn,
                stride_yb, stride_ym, stride_yn,
                BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
                num_warps=4, num_stages=2,
            )

        # Split outputs: processed_encoder [B, T, H], processed_hidden [B, I, H]
        processed_encoder_list = [Y_b[:T, :] for b, Y_b in enumerate(Y)]
        processed_hidden_list = [Y_b[T:, :] for b, Y_b in enumerate(Y)]
        processed_encoder = torch.stack(processed_encoder_list, dim=0) if len(processed_encoder_list) > 0 else torch.empty((0, T, H), device=hidden_states.device, dtype=hidden_states.dtype)
        processed_hidden = torch.stack(processed_hidden_list, dim=0) if len(processed_hidden_list) > 0 else torch.empty((0, I, H), device=hidden_states.device, dtype=hidden_states.dtype)

        return processed_encoder, processed_hidden
