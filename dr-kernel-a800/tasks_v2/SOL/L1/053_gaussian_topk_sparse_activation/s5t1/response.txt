    def reduce_sum_sumsq_kernel(
        x_ptr,                  # *const float32 (input flattened)
        sums_ptr,               # *float32, length B*S
        sums2_ptr,              # *float32, length B*S
        B, S, F,                # int32, dims
        BLOCK_SIZE: tl.constexpr,

        b = pid // S
        s = pid % S
        row_start = (b * S + s) * F

        local_sum = 0.0
        local_sumsq = 0.0

        for offs in range(0, F, BLOCK_SIZE):
            idx = offs + tl.arange(0, BLOCK_SIZE)
            mask = idx < F
            ptrs = x_ptr + row_start + idx

            local_sumsq += tl.sum(vals * vals, axis=0)

        tl.atomic_add(sums2_ptr + pid, local_sumsq)

        std_ptr,                # *float32, length B*S

        total2 = tl.load(sums2_ptr + pid)

    # Abramowitz and Stegun 7.1.26 (polynomial approximation)
    # Valid for p in (0,1). We mirror the original constants.
    a1 = -3.969683028665376e+01
    a2 = 2.209460984245205e+02
    a3 = -2.759285104469687e+02
    a4 = 1.383577518672690e+02
    a5 = -3.066479806614716e+01
    a6 = 2.506628277459239e+00

    b1 = -5.447609879822406e+01
    b2 = 1.615858368580409e+02
    b3 = -1.556989798598866e+02
    b4 = 6.680131188771972e+01
    b5 = -1.328068155288572e+01

    c1 = -7.784894002430293e-03
    c2 = -3.223964580411365e-01
    c3 = -2.400758277161838e+00
    c4 = -2.549732539343734e+00
    c5 = 4.374664141464968e+00
    c6 = 2.938163982698783e+00

    d1 = 7.784695709041462e-03
    d2 = 3.224671290700398e-01
    d3 = 2.445134137142996e+00
    d4 = 3.754408661907416e+00

    p_low = 0.02425
    p_high = 1.0 - p_low

    if sp < p_low:
        z = torch.sqrt(torch.tensor(-2.0 * torch.log(torch.tensor(sp))))  # scalar
        return (((((c1 * z + c2) * z + c3) * z + c4) * z + c5) * z + c6) / \
               ((((d1 * z + d2) * z + d3) * z + d4) * z + 1.0)
    elif sp > p_high:
        z = torch.sqrt(torch.tensor(-2.0 * torch.log(torch.tensor(1.0 - sp))))  # scalar
        return -(((((c1 * z + c2) * z + c3) * z + c4) * z + c5) * z + c6) / \
               ((((d1 * z + d2) * z + d3) * z + d4) * z + 1.0)
    else:
        q = sp - 0.5
        r = q * q
        return (((((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6) * q / \
               (((((b1 * r + b2) * r + b3) * r + b4) * r + b5) * r + 1.0)

            multiplier = float(z.item())
        except Exception:
            multiplier = 0.0
        cutoff = mean + std * multiplier  # [B, S]
        threshold = cutoff.unsqueeze(-1).contiguous()  # [B, S, 1]

        )
