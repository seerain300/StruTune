This satisfies the strict Triton-only requirement: no torch computation in ModelNew.forward, only Triton kernels and host-side allocations/grid setup.

Design details:
- Shapes: Input is [B, S, F]. We use a grid of B*S programs, each program processes one row across F.
- Reductions:
  - sum_rows_kernel: For each (b, s), loop over F in chunks, accumulate sum in registers, atomic_add into out[pid].
  - sumsq_rows_kernel: Same, accumulate sum of squares.
  - compute_stats_kernel: Load out_sum and out_sumsq, compute mean = sum/F, var = sumsq/F - mean^2, var = max(var, 0), std = sqrt(var), store to per-row mean and std outputs.
- ndtri scalar kernel: Given p (a single float), compute A&S approximation and write z. This is a single-program kernel.
- apply_threshold_kernel: Elementwise y = max(inp - threshold_row, 0). Each program processes one row across F in chunks.
- Dtype handling: Compute in float32 in Triton kernels for numerical stability, return bfloat16 to match the original code’s final cast. If the harness is strict, we can also write bfloat16 directly in the kernel; to avoid reliance on Triton’s out_ptr dtype, I’ll keep compute in float32 and cast at the end. import triton
import triton.language as tl


@triton.jit
def sum_rows_kernel(
    x_ptr,           # *const float32
    out_ptr,         # *float32 (length B*S)
    B: tl.constexpr, # int
    S: tl.constexpr, # int
    F: tl.constexpr, # int
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,   # int
):
    # One program per (b, s) row
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    total_sum = 0.0
    f_start = 0
    offs = tl.arange(0, BLOCK_F)
    while f_start < F:
        f = f_start + offs
        mask = f < F
        x_idx = b * stride_b + s * stride_s + f * stride_f
        vals = tl.load(x_ptr + x_idx, mask=mask, other=0.0)
        total_sum += tl.sum(vals, axis=0)
        f_start += BLOCK_F

    tl.atomic_add(out_ptr + pid, total_sum)


@triton.jit
def sumsq_rows_kernel(
    x_ptr,           # *const float32
    out_ptr,         # *float32 (length B*S)
    B: tl.constexpr, # int
    S: tl.constexpr, # int
    F: tl.constexpr, # int
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,   # int

        f_start += BLOCK_F

@triton.jit
def compute_stats_kernel(
    out_sum_ptr,          # *const float32 (length B*S)
    out_sumsq_ptr,        # *const float32 (length B*S)
    mean_out_ptr,         # *float32 (length B*S)
    std_out_ptr,          # *float32 (length B*S)

    S: tl.constexpr,      # int
    F: tl.constexpr,      # int
):

    pid = tl.program_id(axis=0)
    sum_val = tl.load(out_sum_ptr + pid)
    sumsq_val = tl.load(out_sumsq_ptr + pid)
    # mean and std (population std, unbiased=False)
    mean = sum_val / F
    var = sumsq_val / F - mean * mean
    var = tl.maximum(var, 0.0)
    std = tl.sqrt(var)
    tl.store(mean_out_ptr + pid, mean)
    tl.store(std_out_ptr + pid, std)


@triton.jit
def ndtri_scalar_kernel(
