import torch
import triton
import triton.language as tl


@triton.jit
def sum_rows_kernel(
    x_ptr,           # *const float32
    out_ptr,         # *float32 (length B*S)
    B: tl.constexpr, # int
    S: tl.constexpr, # int
    F: tl.constexpr, # int
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,   # int
):
    # One program per (b, s) row
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    # Accumulate sum across F in chunks
    total_sum = 0.0
    f_start = 0
    offs = tl.arange(0, BLOCK_F)
    while f_start < F:
        f = f_start + offs
        mask = f < F
        x_idx = b * stride_b + s * stride_s + f * stride_f
        vals = tl.load(x_ptr + x_idx, mask=mask, other=0.0)
        total_sum += tl.sum(vals, axis=0)
        f_start += BLOCK_F

    # Atomic add into out[pid]
    tl.atomic_add(out_ptr + pid, total_sum)


@triton.jit
def sumsq_rows_kernel(
    x_ptr,           # *const float32
    out_ptr,         # *float32 (length B*S)
    B: tl.constexpr, # int
    S: tl.constexpr, # int
    F: tl.constexpr, # int
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,   # int
):
    # One program per (b, s) row
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    # Accumulate sum of squares across F in chunks
    total_sumsq = 0.0
    f_start = 0
    offs = tl.arange(0, BLOCK_F)
    while f_start < F:
        f = f_start + offs
        mask = f < F
        x_idx = b * stride_b + s * stride_s + f * stride_f
        vals = tl.load(x_ptr + x_idx, mask=mask, other=0.0)
        total_sumsq += tl.sum(vals * vals, axis=0)
        f_start += BLOCK_F

    # Atomic add into out[pid]
    tl.atomic_add(out_ptr + pid, total_sumsq)


@triton.jit
def apply_threshold_kernel(
    inp_ptr,        # *const float32
    out_ptr,        # *float32
    thr_ptr,        # *const float32 (length B*S)
    B: tl.constexpr, # int
    S: tl.constexpr, # int
    F: tl.constexpr, # int
    stride_b: tl.constexpr,  # int
    stride_s: tl.constexpr,  # int
    stride_f: tl.constexpr,  # int
    BLOCK_F: tl.constexpr,   # int
):
    # One program per (b, s) row
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    # Load threshold for this row
    threshold = tl.load(thr_ptr + pid)

    f_start = 0
    offs = tl.arange(0, BLOCK_F)
    while f_start < F:
        f = f_start + offs
        mask = f < F
        inp_idx = b * stride_b + s * stride_s + f * stride_f
        inp_vals = tl.load(inp_ptr + inp_idx, mask=mask, other=0.0)
        # y = max(inp - threshold, 0)
        y = tl.maximum(inp_vals - threshold, 0.0)
        out_idx = b * stride_b + s * stride_s + f * stride_f  # same layout
        tl.store(out_ptr + out_idx, y, mask=mask)
        f_start += BLOCK_F

    def forward(self, input: torch.Tensor, target_sparsity: float):
        """
        Triton-optimized version of the original run.
        Computes per-(b,s) mean and std along last dim, forms threshold = mean + std * _ndtri(target_sparsity),
        then applies ReLU(input - threshold). Returns bfloat16, matching the original behavior.
        """
        # If no sparsity requested, return as-is (original returns input)
        if target_sparsity == 0.0:
            # Return in bfloat16 to match original behavior
            return input.to(torch.bfloat16)

        # Ensure we work in float32 for numerics; original casts to float32 and returns bfloat16
        input_f32 = input.to(torch.float32)
        B, S, F = input_f32.shape
        device = input_f32.device

        # Strides for general layout (inputs are expected to be contiguous)
        stride_b, stride_s, stride_f = input_f32.stride()

        # Output buffers for sums and sumsq (length B*S)
        out_sum = torch.zeros(B * S, dtype=torch.float32, device=device)
        out_sumsq = torch.zeros(B * S, dtype=torch.float32, device=device)

        # Launch reduction kernels over all rows
        BLOCK_F = 1024  # tuneable
        grid = (B * S,)
        triton.run(sum_rows_kernel, grid=grid,
                   x_ptr=input_f32,
                   out_ptr=out_sum,
                   B=B, S=S, F=F,
                   stride_b=stride_b, stride_s=stride_s, stride_f=stride_f,
                   BLOCK_F=BLOCK_F)

        triton.run(sumsq_rows_kernel, grid=grid,
                   x_ptr=input_f32,
                   out_ptr=out_sumsq,
                   B=B, S=S, F=F,
                   stride_b=stride_b, stride_s=stride_s, stride_f=stride_f,
                   BLOCK_F=BLOCK_F)

        # Compute mean and std per row on host (torch ops are fine here as they are not in the critical path)
        mean = out_sum / F
        var = out_sumsq / F - mean * mean
        # Clamp small negative due to numerical error
        var = torch.clamp(var, min=0.0)
        std = torch.sqrt(var)

        # Compute inverse normal CDF for target_sparsity using PyTorch (scalar). This matches original behavior.
        # The original function _ndtri uses A&S approximation; here we use torch.icdf for simplicity.
        # Note: torch.distributions.normal has icdf; alternatively, torch.erf/erfinv can be used to build it.
        # We use torch.erfinv to compute standard normal inverse CDF: z = erfinv(2p - 1) / sqrt(2)
        # Ensure scalar tensor on device
        p = torch.tensor(target_sparsity, dtype=torch.float32, device=device)
        z = torch.erfinv(2.0 * p - 1.0) * 0.7071067811865476  # 1/sqrt(2)

        # threshold per row: mean + std * z
        threshold = mean + std * z

        # Allocate output tensor (float32 during compute)
        output_f32 = torch.empty_like(input_f32)

        # Launch elementwise apply kernel
        triton.run(apply_threshold_kernel, grid=grid,
                   inp_ptr=input_f32,
                   out_ptr=output_f32,
                   thr_ptr=threshold,
                   B=B, S=S, F=F,
                   stride_b=stride_b, stride_s=stride_s, stride_f=stride_f,
                   BLOCK_F=BLOCK_F)

        # Return in bfloat16 to match original's final cast
        return output_f32.to(torch.bfloat16)
