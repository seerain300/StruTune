import torch
import triton
import triton.language as tl


# Triton kernel: Conv2d 3x3, stride=1, padding=1, NCHW layout, no bias
# Computes out[n, co, ho, wo] = sum_{ci=0..C-1} sum_{dh=-1..1} sum_{dw=-1..1} x[n, ci, ho+dh, wo+dw] * w[co, ci, 1+dh, 1+dw]
# x, w, out are contiguous NCHW.
@triton.jit
def conv3x3_nchw_kernel(
    x_ptr, w_ptr, out_ptr,
    B, C, H, W, C_OUT,
    KH: tl.constexpr, KW: tl.constexpr,  # 3x3
    PAD_H: tl.constexpr, PAD_W: tl.constexpr,  # 1
    BLOCK_W: tl.constexpr,                 # vectorization across W
):
    # Grid: (B, C_OUT, H, ceil_div(W, BLOCK_W))
    # Each program handles one (n, co, ho) row and a block of W positions.
    n = tl.program_id(0)
    co = tl.program_id(1)
    ho = tl.program_id(2)
    wb = tl.program_id(3)

    # compute W offsets for this block
    wo_start = wb * BLOCK_W
    wo = wo_start + tl.arange(0, BLOCK_W)
    mask_w = wo < W

    # base offsets for x and out for this (n, co, ho, wo)
    # x layout: ((n*C + ci)*H + hi)*W + wi
    # out layout: ((n*C_OUT + co)*H + ho)*W + wo
    # we will accumulate across input channels ci and 3x3 neighborhood
    acc = tl.zeros([BLOCK_W], dtype=tl.float32)

    # Loop over input channels
    for ci in range(0, C):
        # Loop over 3x3 neighborhood
        for dh in range(-PAD_H, PAD_H + 1):
            for dw in range(-PAD_W, PAD_W + 1):
                hi = ho + dh
                # clamp hi
                if (hi < 0) or (hi >= H):
                    continue
                # build vector wi
                wi = wo + dw
                # clamp wi
                valid = (wi >= 0) & (wi < W) & mask_w
                # compute input offsets for this (n, ci, hi, wi)
                x_offset = ((n * C + ci) * H + hi) * W + wi
                # load x values (masked)
                x_vals = tl.load(x_ptr + x_offset, mask=valid, other=0.0)
                # load corresponding weights (scalar for this (co, ci, dh+1, dw+1))
                # w layout: ((co * C) + ci) * (KH*KW) + (dh+1)*KW + (dw+1)
                w_idx = ((co * C) + ci) * (KH * KW) + (dh + 1) * KW + (dw + 1)
                w_val = tl.load(w_ptr + w_idx)
                # accumulate
                acc += x_vals * w_val

    # write out: ((n * C_OUT + co) * H + ho) * W + wo
    out_offset = ((n * C_OUT + co) * H + ho) * W + wo
    tl.store(out_ptr + out_offset, acc, mask=mask_w)


# Triton kernel: GroupNorm reduction per (n, group) -> sums and sumsq
# out: (B, C, H, W), num_groups divides C, G = C // num_groups
# Each program handles one (n, g); loops over channels in the group and all H*W positions for those channels.
@triton.jit
def groupnorm_sums_kernel(
    x_ptr, sums_ptr, sumsq_ptr,
    B, C, H, W, num_groups, eps,
    C_PER_GROUP: tl.constexpr,  # C // num_groups
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups

    # starting ci for this group
    start_ci = g * C_PER_GROUP

    # accumulator for sum and sumsq over this group
    s = tl.float32(0.0)
    s2 = tl.float32(0.0)

    # loop over channels in group
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        # loop over all H*W elements for this (n, ci)
        for h in range(0, H):
            for w in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w
                x_val = tl.load(x_ptr + idx)
                s += x_val
                s2 += x_val * x_val

    # write to global arrays at index (n * num_groups + g)
    out_idx = n * num_groups + g
    tl.store(sums_ptr + out_idx, s)
    tl.store(sumsq_ptr + out_idx, s2)


# Triton kernel: compute inverse std per (n, group) from sums and sumsq
@triton.jit
def groupnorm_invstd_kernel(
    sums_ptr, sumsq_ptr, invstd_ptr,
    B, C, H, W, num_groups, eps,
    C_PER_GROUP: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_idx = n * num_groups + g

    s = tl.load(sums_ptr + out_idx)
    s2 = tl.load(sumsq_ptr + out_idx)
    group_size = C_PER_GROUP * H * W
    mean = s / group_size
    var = s2 / group_size - mean * mean
    invstd = 1.0 / tl.sqrt(var + eps)
    tl.store(invstd_ptr + out_idx, invstd)


# Triton kernel: normalize + affine + SiLU per (n, group)
# out_ptr: normalized output (float32), x_ptr: original conv output, norm_w_ptr, norm_b_ptr: (C,), invstd_ptr: (B*num_groups,)
@triton.jit
def groupnorm_silu_kernel(
    x_ptr, norm_w_ptr, norm_b_ptr, out_ptr,
    invstd_ptr,
    B, C, H, W, num_groups,
    C_PER_GROUP: tl.constexpr,
):
    pid = tl.program_id(0)  # 0 .. (B * num_groups - 1)
    n = pid // num_groups
    g = pid % num_groups
    out_base = n * num_groups + g

    # load invstd for this (n, g)
    invstd = tl.load(invstd_ptr + out_base)

    start_ci = g * C_PER_GROUP
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        # per-channel scale and bias
        w = tl.load(norm_w_ptr + ci)
        b = tl.load(norm_b_ptr + ci)
        # loop over H*W
        for h in range(0, H):
            for w_idx in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w_idx
                x = tl.load(x_ptr + idx)
                y = (x - mean) * invstd * w + b
                # SiLU: y * sigmoid(y)
                z = y * tl.sigmoid(y)
                tl.store(out_ptr + idx, z)

    def __init__(self, num_groups: int = 32, eps: float = 1e-5):
        super().__init__()
        self.num_groups = num_groups

    def forward(self, x: torch.Tensor, conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,

        assert x.dim() == 4, "x must be (B, C, H, W)"
        B, C, H, W = x.shape
        assert C % self.num_groups == 0, "num_groups must divide C"
        # conv weights are (C_out, C_in, 3, 3)
        assert conv1_weight.shape[1] == C, "conv1_weight in_channels must equal C"
        assert conv1_weight.shape[0] == C, "conv1_weight out_channels must equal C"
        assert conv2_weight.shape[1] == C, "conv2_weight in_channels must equal C"

            B, C, H, W, n_groups,
            C_PER_GROUP=C_per_group

            sums2, sumsq2, invstd2,
            B, C, H, W, n_groups, self.eps,
            C_PER_GROUP=C_per_group
        )
        out2_norm = torch.empty_like(out2, dtype=torch.float32, device=x.device)
        groupnorm_silu_kernel[(B * n_groups,)](
            out2, norm2_weight.to(torch.float32), norm2_bias.to(torch.float32), out2_norm,
            invstd2,
            B, C, H, W, n_groups,
            C_PER_GROUP=C_per_group
        )

        out = out2_norm + x_f32

        return out.to(x.dtype)

import triton
import triton.language as tl


# Triton kernel: GroupNorm + SiLU for NCHW
# Performs per-(n, group) normalization across all channels in the group and all H*W elements,
# then applies affine (weight, bias) and SiLU activation.
# Assumptions:
# - Input x is (B, C, H, W) contiguous, float32.
# - num_groups divides C. We pass C_PER_GROUP = C // num_groups as tl.constexpr.
# - norm_weight and norm_bias are (C,) vectors, applied per channel (GroupNorm affine).
# - eps is a float for numerical stability.
@triton.jit
def groupnorm_silu_kernel(
    x_ptr, norm_w_ptr, norm_b_ptr, out_ptr,
    B, C, H, W, num_groups, eps,
    C_PER_GROUP: tl.constexpr,
):
    # One program per (n, group)
    pid = tl.program_id(0)  # range [0, B * num_groups)
    n = pid // num_groups
    g = pid % num_groups

    # Compute group statistics: sum and sum of squares
    s = tl.float32(0.0)
    s2 = tl.float32(0.0)

    start_ci = g * C_PER_GROUP
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        for h in range(0, H):
            for w in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w
                x_val = tl.load(x_ptr + idx)
                s += x_val
                s2 += x_val * x_val

    group_size = C_PER_GROUP * H * W
    mean = s / group_size
    var = s2 / group_size - mean * mean
    invstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize, affine, SiLU, store
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        w = tl.load(norm_w_ptr + ci)
        b = tl.load(norm_b_ptr + ci)
        for h in range(0, H):
            for w_idx in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w_idx
                x = tl.load(x_ptr + idx)
                y = (x - mean) * invstd * w + b
                # SiLU: y * sigmoid(y)
                z = y * tl.sigmoid(y)
                tl.store(out_ptr + idx, z)

        B, C, H, W = x.shape
        assert C % self.num_groups == 0, "num_groups must divide C"
        assert conv1_weight.shape[1] == C and conv1_weight.shape[0] == C, "conv1_weight must be (C, C, 3, 3)"
        assert conv2_weight.shape[1] == C and conv2_weight.shape[0] == C, "conv2_weight must be (C, C, 3, 3)"

        x_f32 = x.to(torch.float32)

        # First conv: F.conv2d handles stride=1, padding=1
        out1 = F.conv2d(x_f32, conv1_weight, bias=None, stride=1, padding=1)

    start_ci = g * C_PER_GROUP
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        for h in range(0, H):
            for w in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w
                x_val = tl.load(x_ptr + idx)
                s += x_val
                s2 += x_val * x_val

    group_size = C_PER_GROUP * H * W
    mean = s / group_size
    var = s2 / group_size - mean * mean
    invstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize, affine, SiLU, store
    for ci in range(start_ci, start_ci + C_PER_GROUP):
        w = tl.load(norm_w_ptr + ci)
        b = tl.load(norm_b_ptr + ci)
        for h in range(0, H):
            for w_idx in range(0, W):
                idx = ((n * C + ci) * H + h) * W + w_idx
                x = tl.load(x_ptr + idx)
                y = (x - mean) * invstd * w + b
                # SiLU: y * sigmoid(y)
                z = y * tl.sigmoid(y)
                tl.store(out_ptr + idx, z)


class ModelNew(torch.nn.Module):
    def __init__(self, num_groups: int = 32, eps: float = 1e-5):
        super().__init__()
        self.num_groups = num_groups
        self.eps = eps

    def forward(self, x: torch.Tensor, conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,

        assert x.dim() == 4, "x must be (B, C, H, W)"
        B, C, H, W = x.shape
        assert C % self.num_groups == 0, "num_groups must divide C"
        assert conv1_weight.shape[1] == C and conv1_weight.shape[0] == C, "conv1_weight must be (C, C, 3, 3)"
        assert conv2_weight.shape[1] == C and conv2_weight.shape[0] == C, "conv2_weight must be (C, C, 3, 3)"

        # Ensure contiguous and float32 for compute
        x = x.contiguous()
        x_f32 = x.to(torch.float32)

        # First conv: use cuDNN
        out1 = F.conv2d(x_f32, conv1_weight, bias=None, stride=1, padding=1)

        # GroupNorm + SiLU via Triton for out1
        out1_gn_silu = torch.empty_like(out1, dtype=torch.float32, device=x.device)
        C_per_group = C // self.num_groups
        grid = (B * self.num_groups,)
        groupnorm_silu_kernel[grid](
            out1, norm1_weight.to(torch.float32), norm1_bias.to(torch.float32), out1_gn_silu,
            B, C, H, W, self.num_groups, self.eps,
            C_PER_GROUP=C_per_group
        )

        # Second conv: use cuDNN
        out2 = F.conv2d(out1_gn_silu, conv2_weight, bias=None, stride=1, padding=1)

        # GroupNorm + SiLU via Triton for out2
        out2_gn_silu = torch.empty_like(out2, dtype=torch.float32, device=x.device)
        groupnorm_silu_kernel[grid](
            out2, norm2_weight.to(torch.float32), norm2_bias.to(torch.float32), out2_gn_silu,
            B, C, H, W, self.num_groups, self.eps,
            C_PER_GROUP=C_per_group
        )

        # Residual add
        out = out2_gn_silu + x_f32

        # Cast back to original dtype
        return out.to(x.dtype)
