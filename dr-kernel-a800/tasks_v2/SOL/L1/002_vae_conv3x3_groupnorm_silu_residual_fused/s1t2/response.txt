import torch
import triton
import triton.language as tl


@triton.jit
def conv2d_nchw_3x3_stride1_pad1_kernel(
    x_ptr,  # *float32, input: (B, C_in, H, W)
    w_ptr,  # *float32, weight: (C_out, C_in, 3, 3)
    y_ptr,  # *float32, output: (B, C_out, H, W)
    B, C_in, C_out, H, W, H_out, W_out,
    BLOCK_IN: tl.constexpr,
):
    # program id maps to (n, c_out, h_out, w_out)
    pid_n = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_h = tl.program_id(2)
    pid_w = tl.program_id(3)

    if pid_n >= B or pid_c >= C_out or pid_h >= H_out or pid_w >= W_out:
        return

    acc = 0.0  # scalar accumulator

    # loop over input channels in chunks
    for ic_start in range(0, C_in, BLOCK_IN):
        # sum over this chunk of input channels
        for ic in range(BLOCK_IN):
            c = ic_start + ic
            if c >= C_in:
                break
            # accumulate over 3x3 neighborhood
            sum_vec = 0.0
            for kh in range(3):
                for kw in range(3):
                    ih = pid_h + kh
                    iw = pid_w + kw
                    # padding=1, so valid when ih in [0,H-1], iw in [0,W-1]
                    valid = (ih >= 0) & (ih < H) & (iw >= 0) & (iw < W)
                    # input linear index: ((n*C_in + c)*H + ih)*W + iw
                    in_off = (pid_n * C_in * H * W) + c * H * W + ih * W + iw
                    x_val = tl.load(x_ptr + in_off, mask=valid, other=0.0)
                    # weight linear index: ((c_out*C_in + c)*9) + kh*3 + kw
                    w_off = (pid_c * C_in + c) * 9 + kh * 3 + kw
                    w_val = tl.load(w_ptr + w_off)
                    sum_vec += x_val * w_val
            acc += sum_vec

    # store output: y[n, c_out, h_out, w_out]
    y_off = (pid_n * C_out + pid_c) * H_out * W_out + pid_h * W_out + pid_w
    tl.store(y_ptr + y_off, acc)


@triton.jit
def groupnorm_affine_kernel(
    in_ptr, out_ptr,
    weight_ptr, bias_ptr,
    N, C, H, W, NUM_GROUPS, EPS,
    BLOCK_HW: tl.constexpr,
):
    n = tl.program_id(0)
    g = tl.program_id(1)

    GROUP_SIZE = C // NUM_GROUPS
    c0 = g * GROUP_SIZE

    # compute sum and sum of squares across channels in group and all spatial positions
    sum_val = 0.0
    sum_sq = 0.0

    for ic in range(GROUP_SIZE):
        c = c0 + ic
        hw = H * W
        for start in range(0, hw, BLOCK_HW):
            offs = start + tl.arange(0, BLOCK_HW)
            mask = offs < hw
            # map flat offs to (h, w)
            h = offs // W
            w = offs % W
            in_offs = (n * C + c) * hw + offs
            x = tl.load(in_ptr + in_offs, mask=mask, other=0.0)
            sum_val += tl.sum(x, axis=0)
            sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / (GROUP_SIZE * H * W)
    var = sum_sq / (GROUP_SIZE * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # normalize and apply affine for each element in the group
    for ic in range(GROUP_SIZE):
        c = c0 + ic
        hw = H * W
        for start in range(0, hw, BLOCK_HW):
            offs = start + tl.arange(0, BLOCK_HW)
            mask = offs < hw
            h = offs // W
            w = offs % W
            in_offs = (n * C + c) * hw + offs
            x = tl.load(in_ptr + in_offs, mask=mask, other=0.0)
            y = (x - mean) * inv_std
            scale = tl.load(weight_ptr + c)
            bias = tl.load(bias_ptr + c)
            y = y * scale + bias
            out_offs = (n * C + c) * hw + offs
            tl.store(out_ptr + out_offs, y, mask=mask)


@triton.jit
def silu_kernel(in_ptr, out_ptr, TOTAL_ELEMS, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(out_ptr + offs, y, mask=mask)


@triton.jit
def add_residual_kernel(a_ptr, b_ptr, out_ptr, TOTAL_ELEMS, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    y = a + b
    tl.store(out_ptr + offs, y, mask=mask)


class ModelNew(torch.nn.Module):
    def __init__(self, num_groups: int = 32, eps: float = 1e-5, block_in: int = 32, block_hw: int = 1024, silu_block: int = 1024, add_block: int = 1024):
        super().__init__()
        self.num_groups = num_groups
        self.eps = eps
        self.block_in = block_in
        self.block_hw = block_hw
        self.silu_block = silu_block
        self.add_block = add_block

    def forward(self, x: torch.Tensor, conv1_weight: torch.Tensor, norm1_weight: torch.Tensor, norm1_bias: torch.Tensor,
                conv2_weight: torch.Tensor, norm2_weight: torch.Tensor, norm2_bias: torch.Tensor):
        # Ensure float32 and contiguous
        B, C, H, W = x.shape
        assert C % self.num_groups == 0, f"Channels {C} must be divisible by num_groups {self.num_groups}"
        device = x.device
        dtype = torch.float32

        # Save residual
        residual = x.to(dtype).contiguous()

        # First conv: (B, C, H, W) -> (B, C, H, W)
        out = torch.empty((B, C, H, W), device=device, dtype=dtype)
        grid_conv = (B, C, H, W)
        conv2d_nchw_3x3_stride1_pad1_kernel[grid_conv](
            x.contiguous().to(dtype),
            conv1_weight.contiguous().to(dtype),
            out,
            B, C, C, H, W, H, W,
            BLOCK_IN=self.block_in,
        )

        # GroupNorm 1 with affine
        in_flat = out.view(B, C, H * W).contiguous().to(dtype)
        gn_out = torch.empty_like(in_flat, device=device, dtype=dtype)
        grid_gn1 = (B, self.num_groups)
        groupnorm_affine_kernel[grid_gn1](
            in_flat, gn_out,
            norm1_weight.to(dtype), norm1_bias.to(dtype),
            B, C, H, W, self.num_groups, self.eps,
            BLOCK_HW=self.block_hw,
        )

        # SiLU 1
        silu_out = torch.empty_like(gn_out, device=device, dtype=dtype)
        grid_silu1 = (triton.cdiv(C * H * W, self.silu_block),)
        silu_kernel[grid_silu1](gn_out, silu_out, C * H * W, BLOCK=self.silu_block)

        # Second conv: (B, C, H, W) -> (B, C, H, W)
        out2 = torch.empty((B, C, H, W), device=device, dtype=dtype)
        conv2d_nchw_3x3_stride1_pad1_kernel[grid_conv](
            silu_out.view(B, C, H, W),
            conv2_weight.contiguous().to(dtype),
            out2,
            B, C, C, H, W, H, W,
            BLOCK_IN=self.block_in,
        )

        # GroupNorm 2 with affine
        in_flat2 = out2.view(B, C, H * W).contiguous().to(dtype)
        gn_out2 = torch.empty_like(in_flat2, device=device, dtype=dtype)
        grid_gn2 = (B, self.num_groups)
        groupnorm_affine_kernel[grid_gn2](
            in_flat2, gn_out2,
            norm2_weight.to(dtype), norm2_bias.to(dtype),
            B, C, H, W, self.num_groups, self.eps,
            BLOCK_HW=self.block_hw,
        )

        # SiLU 2
        silu_out2 = torch.empty_like(gn_out2, device=device, dtype=dtype)
        grid_silu2 = (triton.cdiv(C * H * W, self.silu_block),)
        silu_kernel[grid_silu2](gn_out2, silu_out2, C * H * W, BLOCK=self.silu_block)

        # Residual add: out2 += residual
        add_out = torch.empty_like(silu_out2, device=device, dtype=dtype)
        grid_add = (triton.cdiv(C * H * W, self.add_block),)
        add_residual_kernel[grid_add](silu_out2, residual.view(-1).to(dtype), add_out, C * H * W, BLOCK=self.add_block)

        # Reshape back to (B, C, H, W)
        final = add_out.view(B, C, H, W)

        return final

This should provide good performance while ensuring correctness. If needed, SiLU could be done with PyTorch too, but the requirement is to use Triton; we do use Triton for SiLU and GroupNorm.

Final code: ModelNew with Triton GroupNorm and SiLU, and PyTorch convs

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def groupnorm_affine_kernel(
    in_ptr, out_ptr,
    weight_ptr, bias_ptr,
    N, C, H, W, NUM_GROUPS, EPS,
    BLOCK_HW: tl.constexpr,
):
    # Each program handles one (n, group)
    n = tl.program_id(0)
    g = tl.program_id(1)

    GROUP_SIZE = C // NUM_GROUPS
    c0 = g * GROUP_SIZE

    # Accumulate sum and sum of squares across channels in group and all spatial positions
    sum_val = 0.0
    sum_sq = 0.0

    for ic in range(GROUP_SIZE):
        c = c0 + ic
        hw = H * W
        for start in range(0, hw, BLOCK_HW):
            offs = start + tl.arange(0, BLOCK_HW)
            mask = offs < hw
            h = offs // W
            w = offs % W
            in_offs = (n * C + c) * hw + offs
            x = tl.load(in_ptr + in_offs, mask=mask, other=0.0)
            sum_val += tl.sum(x, axis=0)
            sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / (GROUP_SIZE * H * W)
    var = sum_sq / (GROUP_SIZE * H * W) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Normalize and apply affine for each element in the group
    for ic in range(GROUP_SIZE):
        c = c0 + ic
        hw = H * W
        for start in range(0, hw, BLOCK_HW):
            offs = start + tl.arange(0, BLOCK_HW)
            mask = offs < hw
            h = offs // W
            w = offs % W
            in_offs = (n * C + c) * hw + offs
            x = tl.load(in_ptr + in_offs, mask=mask, other=0.0)
            y = (x - mean) * inv_std
            scale = tl.load(weight_ptr + c)
            bias = tl.load(bias_ptr + c)
            y = y * scale + bias
            out_offs = (n * C + c) * hw + offs
            tl.store(out_ptr + out_offs, y, mask=mask)


@triton.jit
def silu_kernel(in_ptr, out_ptr, TOTAL_ELEMS, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    sig = 1.0 / (1.0 + tl.exp(-x))
    y = x * sig
    tl.store(out_ptr + offs, y, mask=mask)


@triton.jit
def add_residual_kernel(a_ptr, b_ptr, out_ptr, TOTAL_ELEMS, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    y = a + b
    tl.store(out_ptr + offs, y, mask=mask)
