import triton.language as tl

# Each (b, c) slice writes its L elements (x[b, c, :]) followed by zeros into the vector.
@triton.jit
def pad_kernel(
    x_ptr,              # *const float, input tensor pointer to x of shape (B, C, L), flattened
    out_ptr,            # *float, output pointer to a flattened vector of length M = B*C*2*L
    B: tl.constexpr,    # int
    C: tl.constexpr,    # int
    L: tl.constexpr,    # int
    two_L: tl.constexpr # int, 2*L
):

    pid_c = tl.program_id(1)

    # Compute base offsets
    start = (pid_b * C + pid_c) * two_L  # base position in the flattened output vector for this (b, c) slice

    # Copy x[b, c, :] into the first L positions
    # We assume x_ptr points to a flattened tensor of shape (B, C, L). The offset for x[b, c, t] is:
    # x_offset = (pid_b * C + pid_c) * L + t
    x_offset = pid_b * C * L + pid_c * L
    for t in range(0, L):
        val = tl.load(x_ptr + x_offset + t)
        tl.store(out_ptr + start + t, val)

    # Write zeros for the padded part (two_L - L) positions
    # We can use scalar 0.0 to fill; Triton will cast to float32.
    for t in range(L, two_L):
        tl.store(out_ptr + start + t, 0.0)

@triton.jit

        im = tl.sin(angle)
