import torch
import triton
import triton.language as tl


@triton.jit
def dft_kernel(x_ptr, y_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Compute y[k] = sum_{n=0}^{N-1} x[n] * exp(-2*pi*i*k*n / N) for k in [0, N-1].
    x_ptr: input real vector pointer (float32), length N.
    y_ptr: output complex vector pointer (complex64), length N.
    N: problem size (even).
    """
    k = tl.program_id(axis=0)
    # Only one program per k
    if k >= N:
        return

    # Accumulator for complex output
    acc = 0.0 + 0.0j

    # Loop over n
    for n in range(0, N):
        val_n = tl.load(x_ptr + n)  # load real value
        angle = -2.0 * 3.141592653589793 * k * n / N
        # exp(-i angle) = cos(angle) - i sin(angle)
        e = tl.cos(angle) - 1.0j * tl.sin(angle)
        acc += val_n * e

    # Store complex result
    # Triton stores complex64 by writing two floats (real, imag) to consecutive memory.
    real_k = tl.real(acc)
    imag_k = tl.imag(acc)
    # y_ptr is complex64; we store as two floats
    tl.store(y_ptr + k * 2 + 0, real_k)
    tl.store(y_ptr + k * 2 + 1, imag_k)


@triton.jit
def rfft_real_kernel(y_ptr, zhat_ptr, out_ptr, N: tl.constexpr, M: tl.constexpr):
    """
    Compute real part of rfft: out[j] = (y[j] + conj(zhat[j])) / N, for j in [0, M].
    y_ptr: complex64 input of length N
    zhat_ptr: complex64 input of length N (DFT of reversed x)
    out_ptr: float32 output of length M+1 (real part)
    N: total length (even), M = N//2
    """
    j = tl.program_id(axis=0)
    if j > M:
        return

    # Load y[j] and zhat[j] as complex
    # y_ptr stores real/imag interleaved: at index idx, real at 2*idx, imag at 2*idx+1
    real_y = tl.load(y_ptr + j * 2 + 0)
    imag_y = tl.load(y_ptr + j * 2 + 1)
    real_zh = tl.load(zhat_ptr + j * 2 + 0)
    imag_zh = tl.load(zhat_ptr + j * 2 + 1)

    yj = real_y + 1.0j * imag_y
    zhj = real_zh + 1.0j * imag_zh

    # out[j] = (y[j] + conj(zhat[j])) / N
    outj = (yj + tl.conj(zhj)) / (2.0 * M)

    # Store real part
    tl.store(out_ptr + j, tl.real(outj))


@triton.jit
def rfft_imag_kernel(y_ptr, zhat_ptr, out_ptr, N: tl.constexpr, M: tl.constexpr):
    """
    Compute general imaginary part for rfft: out[j] = (y[M - j].imag - zhat[M - j].imag) / (2j*N), j in [1, M-1].
    Here we compute imag = (y[M-j].imag - zhat[M-j].imag) / (2*N) because j factor is 1/(2j).
    We handle j=0 and j=M separately in Python.
    """
    j = tl.program_id(axis=0)
    # Only for j in [1, M-1]
    if (j < 1) or (j >= M):
        return

    m_idx = M - j
    imag_y_mj = tl.load(y_ptr + m_idx * 2 + 1)
    imag_zh_mj = tl.load(zhat_ptr + m_idx * 2 + 1)

    # imag contribution
    # Note: Python will handle j=0 and j=M separately
    imag_out = (imag_y_mj - imag_zh_mj) / (2.0 * N)

    tl.store(out_ptr + j, imag_out)

import triton.language as tl

def cosine_kernel(x_ptr, out_ptr, j, N: tl.constexpr):

    Compute y[j].real = sum_{k=0..N-1} x[k] * cos(2*pi*j*k/N) for a given j.
    Writes a single float to out_ptr.
    """
    acc = 0.0
    for k in range(0, N):
        xk = tl.load(x_ptr + k)
        angle = 2.0 * 3.141592653589793 * j * k / N
        ck = tl.cos(angle)
        acc += xk * ck
    tl.store(out_ptr, acc)
