import torch
import triton
import triton.language as tl


@triton.jit
def rfft_real_kernel(
    x_ptr,            # *f32, input tensor flattened as [M, 2L] with M=B*C
    real_out_ptr,     # *f32, output real part [M, L+1]
    imag_out_ptr,     # *f32, output imag part [M, L+1]
    B: tl.int32,      # batch size
    C: tl.int32,      # channels
    L: tl.int32,      # seqlen
    STRIDE_B: tl.int32,
    STRIDE_C: tl.int32,
    STRIDE_L: tl.int32,   # stride along seqlen dimension (normally 1)
    BLOCK_N: tl.constexpr  # tile size for j (time index)
):
    # program id: one per (batch, channel) slice
    pid = tl.program_id(axis=0)
    # map pid to b, c
    b = pid // C
    c = pid % C

    # base offsets for this slice
    base_in = b * STRIDE_B + c * STRIDE_C
    # We'll compute for k in [0, L]
    L_plus1 = L + 1
    twoL = 2 * L

    # Precompute inv_twoL = 1 / (2*L) for normalization
    inv_twoL = 1.0 / twoL

    # For each output frequency index k, compute sum over j in tiles
    for k in range(0, L_plus1):
        # Accumulators for real and imaginary parts
        sum_real = 0.0
        sum_imag = 0.0

        # Iterate over time index j in tiles
        for start in range(0, twoL, BLOCK_N):
            j = start + tl.arange(0, BLOCK_N)
            mask = j < twoL
            # Load x[j] for this slice
            x_vals = tl.load(x_ptr + base_in + j * STRIDE_L, mask=mask, other=0.0)  # shape: [BLOCK_N]

            # Compute cos and sin for this k and all j in the tile
            # For k == 0, use h0 = 0.5 / L (since 1 / (2L) * 2L = 1 / (2L), but for k=0, it's real)
            # Note: k is scalar; we broadcast to vector j
            # cos_term = cos(pi * k * j / (2L)), sin_term = sin(pi * k * j / (2L))
            # Handle k == 0 case: cos=1, sin=0; h0 = 0.5 / L
            if k == 0:
                cos_term = 1.0
                sin_term = 0.0
            else:
                angle = (tl.pi * k * j) / twoL
                cos_term = tl.cos(angle)
                sin_term = tl.sin(angle)

            # Multiply and reduce across the tile
            # We need to multiply x_vals (real) by (cos_term + i*sin_term). But since x_vals is real,
            # x_vals * cos_term contributes to real sum, and x_vals * sin_term contributes to imag sum.
            # Accumulate:
            sum_real += tl.sum(x_vals * cos_term, axis=0)
            sum_imag += tl.sum(x_vals * sin_term, axis=0)

        # Normalize by 2*L
        sum_real = sum_real * inv_twoL
        sum_imag = sum_imag * inv_twoL

        # Store results at frequency index k
        tl.store(real_out_ptr + b * STRIDE_B + c * STRIDE_C + k * (STRIDE_B * C), sum_real)
        tl.store(imag_out_ptr + b * STRIDE_B + c * STRIDE_C + k * (STRIDE_B * C), sum_imag)

            x = x.to(torch.float32)

        # Better: reshape to 2D [M, 2L] where M=B*C and last dim is time. We can do:
        # We need to create a 2D view; since x is (B,C,L), we can index as [b,c,l] in kernel via strides.
        # For Triton, we pass pointers and use strides. Let's create a 2D view: reshape to (M, 2L) by creating a contiguous copy with last dim 2L.
        # To do that, we need to expand to 2D: since x is (B,C,L), we can make it (B,C,2L) by concatenating L with zeros, but rfft expects length 2L.
        # Instead, construct a 2D tensor of shape (M, 2L) directly:
        # Create a 2D contiguous tensor by iterating: We can just use x as-is and pass strides; Triton supports 3D indexing via pointer arithmetic.
        # Simpler: keep x as (B,C,L) and index via strides inside the kernel as we do.

        real_out = torch.empty((batch, channels, L + 1), dtype=torch.float32, device=x.device)
        imag_out = torch.empty((batch, channels, L + 1), dtype=torch.float32, device=x.device)

        stride_b, stride_c, stride_l = x.stride()

        M = batch * channels

        BLOCK_N = 256

            x, real_out, imag_out,

            BLOCK_N=BLOCK_N,
            num_warps=4,  # tune as needed
            num_stages=2  # tune as needed
        )

import triton.language as tl

    real_out_ptr,           # *f32, output real part of shape (B, C, L+1)
    imag_out_ptr,           # *f32, output imag part of shape (B, C, L+1)
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,            # seqlen
    stride_b: tl.int32,     # stride for batch in input
    stride_c: tl.int32,     # stride for channel in input
    stride_l: tl.int32,     # stride for last dim in input (normally 1)
    BLOCK_N: tl.constexpr    # tile size along time dimension
):

    b = pid // channels
    c = pid % channels

    twoL = 2 * L
    inv_twoL = 1.0 / twoL
    L_plus1 = L + 1

    # Base offset for this (b, c) slice
    base_in = b * stride_b + c * stride_c

    # For each output frequency index k in [0, L], compute DFT
    for k in range(0, L_plus1):
        # Accumulators for real and imaginary parts
        sum_real = 0.0
        sum_imag = 0.0

            sum_real += tl.sum(x_vals * cos_term, axis=0)
            sum_imag += tl.sum(x_vals * sin_term, axis=0)

  - For each slice, loop k from 0 to L and j in tiles of size BLOCK_N (e.g., 256), loading x[j], computing cos/sin for the tile, multiplying and reducing, accumulating into per-k real/imag accumulators. After the loops, divide by 2*L (normalization) and store to output.
- This approach is correct and keeps all computation in Triton. It’s O(N^2) per slice, which may not be optimal for very large seqlen, but for many typical sizes (1024, 4096, 8192) and moderate batch and channel counts, it can be acceptable. If you need maximum performance for very large seqlen, a Cooley-Tukey FFT would be preferable, but is more involved.

Design considerations
- Data types: Operate in float32 for numerical stability. Ensure input is float32 and on CUDA before launching Triton.
- Output tensors: Create real_out and imag_out of shape (batch, channels, seqlen+1) as float32 and store per-k values.
- Strides: Use input strides to address elements and write outputs using the linearized index for contiguous tensors.

Plan summary
- Write a Triton kernel that computes the DFT per (batch, channel) slice, uses cos/sin per k and j tile, accumulates real and imaginary parts, normalizes by 2*seqlen, and stores them.
- Implement ModelNew.forward to:
  - Validate input shape.
  - Ensure x is float32 and on CUDA.
  - Allocate outputs real_out and imag_out.
  - Launch the Triton kernel with grid=(batch*channels,).
  - Return real_out and imag_out.

import torch
import triton
import triton.language as tl


@triton.jit
def rfft_real_kernel(
    x_ptr,                  # *f32, input tensor (B, C, L)
    real_out_ptr,           # *f32, output real part (B, C, L+1)
    imag_out_ptr,           # *f32, output imag part (B, C, L+1)
    batch: tl.int32,
    channels: tl.int32,
    L: tl.int32,            # seqlen
    stride_b: tl.int32,     # stride for batch in input
    stride_c: tl.int32,     # stride for channel in input
    stride_l: tl.int32,     # stride for last dim in input (usually 1)
    BLOCK_N: tl.constexpr    # tile size along time dimension
):
    # One program per (batch, channel) slice
    pid = tl.program_id(axis=0)
    b = pid // channels
    c = pid % channels

    twoL = 2 * L
    inv_twoL = 1.0 / twoL
    L_plus1 = L + 1

    # Base offset for input slice (b, c, :)
    base_in = b * stride_b + c * stride_c

    # For each output frequency index k in [0, L], compute DFT
    for k in range(0, L_plus1):
        sum_real = 0.0
        sum_imag = 0.0

        # Iterate over time index j in tiles
        for start in range(0, twoL, BLOCK_N):
            j = start + tl.arange(0, BLOCK_N)  # vector of time indices
            mask = j < twoL
            # Load x[b, c, j]
            x_vals = tl.load(x_ptr + base_in + j * stride_l, mask=mask, other=0.0)  # shape: [BLOCK_N], real

            # Compute cos and sin for this k and all j in the tile
            if k == 0:
                cos_term = 1.0
                sin_term = 0.0
            else:
                angle = (tl.pi * k * j) / twoL
                cos_term = tl.cos(angle)
                sin_term = tl.sin(angle)

            # Accumulate contributions
            sum_real += tl.sum(x_vals * cos_term, axis=0)
            sum_imag += tl.sum(x_vals * sin_term, axis=0)

        # Normalize by 2*L (same as dividing by fft_size in the original code)
        sum_real = sum_real * inv_twoL
        sum_imag = sum_imag * inv_twoL

        # Store results at frequency index k
