import torch
import triton
import triton.language as tl

@triton.jit
def real_fft_normalize_kernel(x_ptr, out_real_ptr, out_imag_ptr,
                               seqlen: tl.int32, N: tl.int32, M: tl.int32, B: tl.int32, C: tl.int32):
    """
    Triton kernel computing real FFT for each (batch, channel) slice and normalizing by 1/N (which
    corresponds to dividing by 2*seqlen in the original code). It writes both real and imaginary parts
    into out_real_ptr and out_imag_ptr of shape (B, C, M).
    Grid: one program per (b, c) slice.
    """
    pid = tl.program_id(0)  # 0..(B*C-1)
    b = pid // C
    c = pid % C

    # If pid exceeds B*C-1 (in case grid is larger), guard (though grid should match exactly)
    if b >= B:
        return

    # Base index for this (b, c) slice along the last dimension (seqlen)
    base = (b * C + c) * seqlen

    # Initialize output arrays for this slice: re and im of length M
    # Triton doesn't let us allocate local arrays of dynamic size; instead we compute and store directly.
    # We'll iterate k from 0 to M-1, accumulate per k, then store at [b, c, k].
    inv_n = 1.0 / (2.0 * seqlen)  # match original normalization: divide by 2*seqlen

    # Loop over k from 0 to M-1
    # Note: Triton requires compile-time constant loops. We use a while loop to handle dynamic M.
    k = 0
    while k < M:
        # Accumulators for real and imaginary parts for this k
        # Triton allows scalar variables; we maintain them across t loop
        re_k = 0.0
        im_k = 0.0

        # Loop over t from 0 to N-1
        t = 0
        while t < N:
            # Load x[t] for this (b, c) slice
            x_t = tl.load(x_ptr + base + t)

            # Compute angle = 2*pi*k*t/N
            angle = 2.0 * 3.141592653589793 * float(k) * float(t) / float(N)

            # Accumulate contributions
            re_k += x_t * tl.cos(angle)
            im_k += x_t * tl.sin(angle)

            t += 1

        # Apply normalization: original divides by 2*seqlen, so re_k *= 1/(2*seqlen), same for im_k
        re_k = re_k * inv_n
        im_k = im_k * inv_n

        # Store results to output tensors at [b, c, k]
        # We need to compute the linear index for out[b, c, k]: ((b*C + c) * M) + k
        out_base = (b * C + c) * M + k
        tl.store(out_real_ptr + out_base, re_k)
        tl.store(out_imag_ptr + out_base, im_k)

        k += 1
