import torch
import triton
import triton.language as tl

@triton.jit
def real_fft_sums_kernel(x_ptr, sum_cos_ptr, sum_sin_ptr, B: tl.int32, C: tl.int32, 
                          N: tl.int32, M: tl.int32, seqlen: tl.int32, 
                          BLOCK_T: tl.constexpr):
    """
    Triton kernel computing per-(batch,channel) slice:
    For each k in 0..M-1, compute:
      sum_cos[k] = sum_{t=0}^{N-1} x[t] * cos(2*pi*k*t/N)
      sum_sin[k] = sum_{t=0}^{N-1} x[t] * sin(2*pi*k*t/N)
    Writes sum_cos_ptr[k] and sum_sin_ptr[k] for linear index k in 0..M-1.
    Grid: (B, C), each program handles one (batch, channel) slice and iterates over k and t.
    """
    b = tl.program_id(0)
    c = tl.program_id(1)
    # Linear index for the output arrays
    k = 0
    while k < M:
        # Accumulators as scalars
        sum_cos = 0.0
        sum_sin = 0.0
        # Loop over t in chunks of BLOCK_T
        t = 0
        # Base offset for (b, c) slice: b*C + c is a fixed integer since grid is (B,C)
        # We need to compute the base offset for x slice. Since each program handles a (b,c),
        # x is laid out as B*C rows and seqlen columns if we view it as (B*C, seqlen).
        # Here, we assume x is (B, C, seqlen) contiguous, so linear index is ((b*C + c)*seqlen + t).
        base = (b * C + c) * seqlen
        while t < N:
            t_offsets = t + tl.arange(0, BLOCK_T)
            mask = t_offsets < N
            x_vals = tl.load(x_ptr + base + t_offsets, mask=mask, other=0.0)
            # Compute angle = 2*pi*k*t_offsets/N
            angle = (2.0 * tl.pi * k * t_offsets) / N
            # cos and sin vectors
            cos_vec = tl.cos(angle)
            sin_vec = tl.sin(angle)
            # Accumulate sums for this chunk
            sum_cos += tl.sum(x_vals * cos_vec, axis=0)
            sum_sin += tl.sum(x_vals * sin_vec, axis=0)
            t += BLOCK_T
        # Store results to sum_cos_ptr and sum_sin_ptr at linear index k
        tl.store(sum_cos_ptr + k, sum_cos)
        tl.store(sum_sin_ptr + k, sum_sin)
        k += 1

    1D copy kernel: copies 'size' elements from inp_ptr to out_ptr.

    vals = tl.load(inp_ptr + offsets, mask=mask)
    tl.store(out_ptr + offsets, vals, mask=mask)

class ModelNew(torch.nn.Module):
    def forward(self, *args):
        # Expect a single input tensor of shape (batch, channels, seqlen)
        x = args[0]
        assert x.dim() == 3, "Input must be a 3D tensor (batch, channels, seqlen)"
        batch, channels, seqlen = x.shape

        # Cast to float32 and ensure contiguous. Keep on the same device as input.
        x_f32 = x.to(torch.float32).contiguous()

        # N and M as in original
        N = 2 * seqlen
        M = seqlen + 1

        # Allocate 1D buffers for sums per (batch, channel) slice
        # We'll create two 1D arrays of length batch*channels*M
        total_pairs = batch * channels * M
        sum_cos = torch.empty(total_pairs, dtype=torch.float32, device=x.device)
        sum_sin = torch.empty(total_pairs, dtype=torch.float32, device=x.device)

        # Launch Triton kernel: grid over (batch, channels)
        BLOCK_T = 128  # chunk size for t-loop
        grid = (batch, channels)
        real_fft_sums_kernel[grid](
            x_f32.view(-1), sum_cos, sum_sin,
            B=batch, C=channels, N=N, M=M, seqlen=seqlen,
            BLOCK_T=BLOCK_T, num_warps=4, num_stages=2
        )

        # Now we need to expand these sums to shape (batch, channels, M)
        # We can do this with Triton copy kernels. First, we need a mapping:
        # For each (b, c, k), the linear index into sum_cos/sum_sin is (b*C + c)*M + k.
        # We'll allocate out_real and out_imag and copy into them.
        out_real = torch.empty((batch, channels, M), dtype=torch.float32, device=x.device)
        out_imag = torch.empty((batch, channels, M), dtype=torch.float32, device=x.device)

        # Flatten views for Triton copying
        out_real_flat = out_real.view(-1)    # length = batch*channels*M
        out_imag_flat = out_imag.view(-1)    # length = batch*channels*M

        # Copy sum_cos and sum_sin into their respective outputs. sum_cos maps to real, sum_sin to imag.
        num_items = total_pairs
        BLOCK_COPY = 1024
        grid_copy = (triton.cdiv(num_items, BLOCK_COPY),)

        # For sum_cos -> out_real
        # We need to read sum_cos[i] where i runs 0..batch*channels*M-1, and write it to out_real_flat[i].
        copy_flat_kernel[grid_copy](sum_cos, out_real_flat, num_items, BLOCK=BLOCK_COPY, num_warps=4)
        # For sum_sin -> out_imag
        copy_flat_kernel[grid_copy](sum_sin, out_imag_flat, num_items, BLOCK=BLOCK_COPY, num_warps=4)

        # Finally, apply normalization by n = 2 * seqlen (matching original). Since we didn't normalize in-kernel,
        # we do it here to match original semantics. This is a simple elementwise divide.
        # However, note that sum_cos and sum_sin are already computed with factor 1/N in-kernel, not 1/(2*seqlen).
        # The original divides the complex rfft output by n=2*seqlen; since rfft returns (sum_cos, sum_sin), we need:
        # re[k] = sum_cos[k] * (1/(2*seqlen)), im[k] = sum_sin[k] * (1/(2*seqlen)).
        # We can incorporate this by scaling before copying. But since we copied raw sums, we scale here.
        # The scaling factor is 1/(2*seqlen). Since we didn't divide in-kernel, we divide now.
        scale = 1.0 / (2.0 * seqlen)
        out_real.mul_(scale)
        out_imag.mul_(scale)

        # Return real and imaginary parts as per original function
        return out_real, out_imag

import triton
import triton.language as tl

@triton.jit
def copy_flat_kernel(inp_ptr, out_ptr, size: tl.int32, BLOCK: tl.constexpr):
    """
    1D copy kernel: copies 'size' elements from inp_ptr to out_ptr.
    Launch multiple programs; each handles a chunk of size BLOCK.
    """

    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask = offsets < size
    vals = tl.load(inp_ptr + offsets, mask=mask)
    tl.store(out_ptr + offsets, vals, mask=mask)
