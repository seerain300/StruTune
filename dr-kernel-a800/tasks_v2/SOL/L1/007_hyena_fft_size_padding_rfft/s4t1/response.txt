import torch
import triton
import triton.language as tl


# Kernel 1: Zero-pad the input along the last dimension to length n = 2 * seqlen
@triton.jit
def _pad_to_n_kernel(x_ptr, padded_ptr, L, n, BLOCK: tl.constexpr):
    """
    x_ptr: *float32, length L (original seqlen)
    padded_ptr: *float32, length n (2*seqlen), output buffer
    L: int, original seqlen
    n: int, 2*seqlen
    Each program writes one element of padded_ptr (assuming x is contiguous).
    """
    pid = tl.program_id(axis=0)
    if pid < L:
        tl.store(padded_ptr + pid, tl.load(x_ptr + pid))
    else:
        tl.store(padded_ptr + pid, 0.0)


# Kernel 2: Compute real and imaginary parts of rFFT for k in [0..seqlen], i.e., output length = seqlen + 1
# We implement y[k] = sum_{j=0}^{2*seqlen-1} x[j] * exp(-i * 2*pi*k*j / (2*seqlen))
# Separately compute cosine (real) and sine (imag) contributions. Input x is zero-padded to length n.
@triton.jit
def _rfft_simple_kernel(padded_ptr, out_real_ptr, out_imag_ptr, L, n, seqlen, BLOCK_K: tl.constexpr):
    """
    padded_ptr: *float32, length n (2*seqlen), contains input x padded with zeros
    out_real_ptr: *float32, length (seqlen+1)
    out_imag_ptr: *float32, length (seqlen+1)
    L: int, original seqlen
    n: int, 2*seqlen
    seqlen: int, original seqlen
    Each program handles a block of k indices and computes sums over j from 0 to n-1.
    """
    pid = tl.program_id(axis=0)
    k0 = pid * BLOCK_K
    ks = k0 + tl.arange(0, BLOCK_K)
    mask_k = ks <= seqlen  # we only produce up to Nyquist index

    # Initialize accumulators
    acc_real = tl.zeros([BLOCK_K], dtype=tl.float32)
    acc_imag = tl.zeros([BLOCK_K], dtype=tl.float32)

    # Sum over j from 0 to n-1
    # For j >= L, padded_ptr[j] is 0, so we don't need extra branching
    for j in range(0, n):
        xj = tl.load(padded_ptr + j)
        # angle = 2*pi*k*j / n
        angle = 2.0 * 3.141592653589793 * ks * j * (1.0 / n)
        cosv = tl.cos(angle)
        sinv = tl.sin(angle)
        acc_real += xj * cosv
        acc_imag += xj * sinv

    # Store results for ks within [0, seqlen]
    tl.store(out_real_ptr + ks, acc_real, mask=mask_k)
    tl.store(out_imag_ptr + ks, acc_imag, mask=mask_k)


# Kernel 3: Normalize by division with a scalar (elementwise): out = in / scale
@triton.jit
def _normalize_divide_kernel(in_ptr, out_ptr, numel, scale, BLOCK: tl.constexpr):
    """
    Elementwise division: out[i] = in[i] / scale
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < numel
    x = tl.load(in_ptr + offsets, mask=mask, other=0.0)
    y = x / scale
    tl.store(out_ptr + offsets, y, mask=mask)

    def forward(self, *args):
        # Expect a single input tensor of shape (batch, channels, seqlen)
        if len(args) != 1:

        x = args[0]

        # Launch kernel to copy first L elements and set the rest to zero

        # 2) Compute rFFT for k in [0..L], i.e., output length = L + 1
        # We'll allocate outputs (real and imag) of length L + 1
        out_real = torch.empty(L + 1, dtype=torch.float32, device=device)
        out_imag = torch.empty(L + 1, dtype=torch.float32, device=device)

        # Triton grid: we need to cover k in [0..L]. Choose BLOCK_K based on L.
        # For simplicity, use BLOCK_K = 128. Grid size = ceil((L+1)/BLOCK_K).
        BLOCK_K = 128
        grid_k = (triton.cdiv(L + 1, BLOCK_K),)
        _rfft_simple_kernel[grid_k](padded, out_real, out_imag, L, n, L, BLOCK_K=BLOCK_K)

        # We can use Triton elementwise division kernels to normalize both outputs

        grid_norm = (triton.cdiv(numel_r, 1024),)

        x_freq_real = out_real.unsqueeze(0).expand(batch, channels, L + 1).contiguous()
        x_freq_imag = out_imag.unsqueeze(0).expand(batch, channels, L + 1).contiguous()

Optimization approach (keeping it Triton-only)
- We cannot rely on torch.fft in host; we implement the rFFT directly in Triton:
  - Step 1: Zero-pad x along last dim to n = 2*seqlen. We allocate padded buffer and launch a simple Triton kernel to copy the first L elements and set the rest to zero.
  - Step 2: Compute rFFT y[k] for k in [0..L] via direct summation:
    y[k] = sum_{j=0}^{2L-1} x[j] * exp(-i * 2*pi*k*j / (2L))
    We separately accumulate real and imaginary parts:
    real += xj * cos(2*pi*k*j / (2L)), imag += xj * sin(2*pi*k*j / (2L))
    We launch a Triton kernel that iterates over k in blocks; inside the kernel we loop over j from 0 to 2L-1. Since we’re computing directly, we do not need to worry about handling padding: the padded buffer already contains zeros for j >= L.
  - Step 3: Normalize the results by 2*seqlen using a Triton elementwise division kernel (no torch division).
- Note: This direct DFT is O(n^2) and will be slow for large seqlen, but it ensures all computation is done in Triton kernels and avoids any torch ops in host code.

  - Input cast to float32 (permitted minimal conversion).
  - Tensor allocations (torch.empty for outputs, padded buffer).

import torch
import triton
import triton.language as tl


# Kernel 1: Zero-pad input of length L to a buffer of length n (n = 2*L)
@triton.jit
def _pad_to_n_kernel(x_ptr, padded_ptr, L, n, BLOCK: tl.constexpr):
    """
    x_ptr: *float32, length L (original seqlen)
    padded_ptr: *float32, length n (2*seqlen), output buffer to be zero-padded
    Each program writes one element of padded_ptr.
    """
    pid = tl.program_id(axis=0)
    if pid < L:
        tl.store(padded_ptr + pid, tl.load(x_ptr + pid))
    else:
        tl.store(padded_ptr + pid, 0.0)


# Kernel 2: Direct rFFT for k in [0..L] using explicit padding to n
# y[k] = sum_{j=0}^{n-1} x[j] * exp(-i * 2*pi*k*j / n)
# We compute real and imag parts separately: real += xj*cos, imag += xj*sin
# This kernel processes a block of k indices; it loops over j to accumulate.
@triton.jit
def _rfft_simple_kernel(padded_ptr, out_real_ptr, out_imag_ptr, L, n, BLOCK_K: tl.constexpr):
    """
    padded_ptr: *float32, length n, contains input x padded with zeros
    out_real_ptr: *float32, length (L+1)
    out_imag_ptr: *float32, length (L+1)
    """
    pid = tl.program_id(axis=0)
    k0 = pid * BLOCK_K
    ks = k0 + tl.arange(0, BLOCK_K)
    mask_k = ks <= L  # output length is L+1

    acc_real = tl.zeros([BLOCK_K], dtype=tl.float32)
    acc_imag = tl.zeros([BLOCK_K], dtype=tl.float32)

    # Sum over j from 0 to n-1; padded_ptr has zeros for j >= L so we don't need extra branching
    for j in range(0, n):
        xj = tl.load(padded_ptr + j)  # float32
        # angle = 2*pi*k*j / n
        angle = 2.0 * 3.141592653589793 * ks * j * (1.0 / n)
        cosv = tl.cos(angle)
        sinv = tl.sin(angle)
        acc_real += xj * cosv
        acc_imag += xj * sinv

    # Store results for ks up to L (Nyquist index)
    tl.store(out_real_ptr + ks, acc_real, mask=mask_k)
    tl.store(out_imag_ptr + ks, acc_imag, mask=mask_k)
