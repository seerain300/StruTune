import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel(X_ptr, Y_ptr, M, D, eps):
    """
    RMSNorm over last dimension of shape (M, D).
    X_ptr, Y_ptr point to 1D contiguous arrays of length M*D.
    eps: float32
    """
    row_id = tl.program_id(axis=0)
    if row_id >= M:
        return
    # Accumulate sum of squares over D
    sum_sq = 0.0
    for d in range(0, D):
        x = tl.load(X_ptr + row_id * D + d)
        # upcast to fp32 for accumulation
        x_f32 = x.to(tl.float32)
        sum_sq += x_f32 * x_f32
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)  # scalar r for this row
    # Scale and store
    for d in range(0, D):
        x = tl.load(X_ptr + row_id * D + d)
        y = x / r
        tl.store(Y_ptr + row_id * D + d, y)


@triton.jit
def rotate_and_scale_kernel(X_ptr, Y_ptr, B, N_HEADS, S, D, inv_ptr, POS_PTR, MAX_POS):
    """
    Per-token rotation and scaling (RoPE) applied to X_ptr -> Y_ptr.
    We operate on rows (b, head, pos) and compute cos/sin vectors for that token position.
    X_ptr: input pointer to 1D contiguous array of length (B*N_HEADS*S)*D
    Y_ptr: output pointer to 1D contiguous array of length (B*N_HEADS*S)*D
    B, N_HEADS, S: int32
    D: int32
    inv_ptr: float32 pointer to length D, where inv = [inv_freq, inv_freq]
    POS_PTR: int64 pointer to position_ids (shape B, S), we load pos from it using b and s.
    MAX_POS: ignored (for signature consistency); we derive pos from POS_PTR directly.
    """
    pid = tl.program_id(axis=0)
    # pid indexes rows: rows = B * N_HEADS * S
    total_rows = B * N_HEADS * S
    if pid >= total_rows:
        return
    # Compute b, head, s from pid
    S_i32 = tl.full((), S, tl.int32)
    N_heads = tl.full((), N_HEADS, tl.int32)
    b = pid // (N_heads * S_i32)
    rem = pid % (N_heads * S_i32)
    s = rem // N_heads
    head = rem % N_heads

    # Load position for this row
    # POS_PTR has shape (B, S); we load pos[b, s]
    # Note: POS_PTR dtype is int64; Triton supports int32/int64 indexing arithmetic.
    pos_val = tl.load(POS_PTR + b * S + s)  # scalar int64 position
    pos_f32 = pos_val.to(tl.float32)

    # Build cos and sin vectors of length D (float32)
    cos_vec = tl.zeros((D,), dtype=tl.float32)
    sin_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        inv_d = tl.load(inv_ptr + d)  # inv[d], d in [0..D-1] where inv = [inv_freq, inv_freq]
        angle = pos_f32 * inv_d
        cos_vec[d] = tl.cos(angle)
        sin_vec[d] = tl.sin(angle)

    # Load original row X_ptr[pid, :]
    base = pid * D
    x = [tl.load(X_ptr + base + d) for d in range(0, D)]
    x_f32 = [x_i.to(tl.float32) for x_i in x]

    # Split halves
    x1 = x_f32[:D // 2]
    x2 = x_f32[D // 2:]

    # rotate_half(x) = [-x2, x1]
    rotated_half = [-x2[0], *x1]

    # Compute y = x1 * cos + (rotate_half(x)) * sin
    y1 = [x1_i * cos_vec[i] for i, x1_i in enumerate(x1)]
    y2 = [rotated_half[i] * sin_vec[i] for i in range(D // 2)]
    y = y1 + y2

    # Store results
    for d in range(0, D):
        tl.store(Y_ptr + base + d, y[d])

        key_rot = torch.empty_like(key_norm, dtype=key_norm.dtype)
        total_rows_key = M_key
        # Note: position_ids has shape (B, S); we can reuse it for key since each token position is independent.
        rotate_and_scale_kernel[(total_rows_key,)](
            key_norm, key_rot, B, N_kv, S, D, inv_freq, position_ids, MAX_POS=self.max_position_embeddings

        # 5) Cache updates: write rotated key and raw value into caches at cache_position indices.
        # Use PyTorch index_put to preserve original behavior exactly.
        # Ensure cache_position is int64
        cache_pos_i64 = cache_position.to(torch.int64)
        # key_cache shape: (B, N_kv, max_pos, D)
        # Assign key_rot[b, nk, s, :] to key_cache[b, nk, cache_pos_i64[s], :]
        for b in range(B):

import torch
import triton
import triton.language as tl


@triton.jit
def rmsnorm_rows_kernel(X_ptr, Y_ptr, M, D, eps, BLOCK_SIZE: tl.constexpr):
    """
    RMSNorm over last dimension of shape (M, D).
    Each program handles one row. Computes r = sqrt(mean(x^2) + eps) and writes y = x / r.
    X_ptr, Y_ptr are 1D contiguous arrays of length M*D. eps: float32.
    BLOCK_SIZE is a compile-time constant (e.g., 128).
    """
    row_id = tl.program_id(axis=0)
    if row_id >= M:
        return
    sum_sq = 0.0
    # Accumulate sum of squares over D in fp32
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)
    mean = sum_sq / D
    r = tl.sqrt(mean + eps)

    # Scale and store
    for d in range(0, D, BLOCK_SIZE):
        offs = d + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(X_ptr + row_id * D + offs, mask=mask, other=0.0)
        y = x / r
        tl.store(Y_ptr + row_id * D + offs, y, mask=mask)
