  4) Grouped causal 1D conv with kernel_size=4, groups=H. Input is (B, H, S) post-transpose; conv weight is (H, 1, 4). This is groups=H depthwise conv on S dimension. It’s a small per-channel reduction over K=4, which is quite light compared to a dense linear.
  5) Output gating: y = C * conv_out. Elementwise multiply.
  6) Final out_proj: y ∈ (B, S, H) → F.linear(y, out_proj_weight, out_proj_bias) to produce output ∈ (B, S, H). This is another GEMV-style linear per (b, s).

What can/should be optimized with Triton
- The dense F.linear (in_proj) in step 1 performs a sizable tensor contraction. A Triton kernel can implement this contraction directly: for each (b, s), compute out_vec of length I = 3*H by iterating over H and accumulating in FP32. This reduces reliance on PyTorch and gives us control over tiling and dtype.
- The final F.linear (out_proj) in step 6 is also a good target: for each (b, s), compute H outputs by summing over H with weight (H, H).

Elementwise ops and conv should remain in PyTorch for correctness and simplicity:
- Chunks, transposes, elementwise multiplies, and conv1d are not heavy compute. Reimplementing them in Triton would either be overkill (elementwise) or require padding logic and grouping handling that adds complexity without clear gains. import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def in_proj_linear_kernel(
    X_ptr,         # *ptr to input x: shape (B, S, H), contiguous
    W_ptr,         # *ptr to in_proj_weight: shape (I, H), contiguous, I = 3*H
    Out_ptr,       # *ptr to output: shape (B, S, I), contiguous
    B: tl.constexpr,  # batch size (passed as int; used for bounds, but we use dynamic indexing)
    S: tl.constexpr,  # seq_len
    H: tl.constexpr,  # hidden size
    I: tl.constexpr,  # out channels, I = 3*H
    BLOCK_H: tl.constexpr,
):
    # Each program handles one (b, s) row and computes the entire output vector of length I
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    # Base pointers for this (b, s)
    x_base = X_ptr + b * (S * H) + s * H
    out_base = Out_ptr + b * (S * I) + s * I

    # Accumulator for output vector of length I (float32 for stability)
    acc = tl.zeros((I,), dtype=tl.float32)

    # Loop over H in chunks of BLOCK_H
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # For each i in I, accumulate sum_{h in chunk} X[b, s, h] * W[i, h]
        # We'll do this with a small loop over i-chunk (compile-time inner loop)
        # But to keep register usage low, we update acc[i] one by one.
        # Note: W[i, h] contiguous means pointer W_ptr + i*H + h.
        for i in range(0, I):
            # load W[i, offs_h]
            w_ptrs = W_ptr + i * H + offs_h
            w_vals = tl.load(w_ptrs, mask=mask_h, other=0.0).to(tl.float32)
            # load X[b, s, offs_h]
            x_vals = tl.load(x_base + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            # accumulate
            acc[i] += tl.sum(w_vals * x_vals, axis=0)

    # Write out acc to Out_ptr[b, s, :]
    out_ptrs = out_base + tl.arange(0, I)
    tl.store(out_ptrs, acc)


@triton.jit
def out_proj_linear_kernel(
    Y_ptr,         # *ptr to input y: shape (B, S, H), contiguous
    W_ptr,         # *ptr to out_proj_weight: shape (H, H), contiguous
    Out_ptr,       # *ptr to output: shape (B, S, H), contiguous
    B: tl.constexpr,
    S: tl.constexpr,
    H: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # Each program handles one (b, s) and computes all H outputs
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    y_base = Y_ptr + b * (S * H) + s * H
    out_base = Out_ptr + b * (S * H) + s * H

    acc = tl.zeros((H,), dtype=tl.float32)

    for h_out in range(0, H, BLOCK_H):
        offs_h = h_out + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # For each h_out, sum over input features h_in: acc[h_out] = sum_h y[b, s, h] * W[h_out, h]
        for h_in in range(0, H, BLOCK_H):
            offs_h_in = h_in + tl.arange(0, BLOCK_H)
            mask_h_in = offs_h_in < H

            y_vals = tl.load(y_base + offs_h_in, mask=mask_h_in, other=0.0).to(tl.float32)
            # load W[h_out, offs_h_in]
            # For each h_out, we iterate h_in chunk
            # We'll construct w_ptrs for vector h_in; since we use nested loops, we need to combine them.
            # Better: load a vector for each h_out chunk by broadcasting.
            # Implement via small inner loop over BLOCK_H:
            # Note: W[h, h_in] contiguous -> W_ptr + h*H + h_in
            # We need vector w_vals for current offs_h_in.
            # We'll do this by loading scalar per h_out per h_in chunk; that's okay for H up to a few thousands.
            for j in range(0, BLOCK_H):
                h = h_out + j
                h_valid = h < H
                if h_valid:
                    w_row = W_ptr + h * H
                    w_ptrs = w_row + offs_h_in
                    w_vals = tl.load(w_ptrs, mask=mask_h_in, other=0.0).to(tl.float32)
                    acc[h] += tl.sum(w_vals * y_vals, axis=0)

    # Store results
    out_ptrs = out_base + tl.arange(0, H)
    tl.store(out_ptrs, acc)


class ModelNew(torch.nn.Module):
    def forward(self, x: torch.Tensor,
                in_proj_weight: torch.Tensor,
                in_proj_bias: torch.Tensor,
                conv_weight: torch.Tensor,
                conv_bias: torch.Tensor,
                out_proj_weight: torch.Tensor,
                out_proj_bias: torch.Tensor):

        assert x.dtype == torch.float32, "This Triton implementation assumes float32 tensors"
        assert in_proj_weight.dtype == torch.float32 and out_proj_weight.dtype == torch.float32
        assert conv_weight.dtype == torch.float32 and conv_bias.dtype == torch.float32

        B, S, H = x.shape
        I = 3 * H

        # 1) Triton in_proj linear: compute BCx = X @ W_in^T, shape (B, S, I)
        x_contig = x.contiguous()
        in_proj_weight_contig = in_proj_weight.contiguous()
        BCx = torch.empty((B, S, I), device=x.device, dtype=torch.float32)

        grid_in = (B * S,)
        in_proj_linear_kernel[grid_in](
            x_contig, in_proj_weight_contig, BCx,
            B, S, H, I,
            BLOCK_H=128,
            num_warps=4,
        )

        # 2) Reconstruct B, C, x_proj using torch views/slices (no compute, correctness only):
        #    BCx shape (B,S,I) with I=3H. We need to split along last dim.
        #    PyTorch chunk along last dimension requires contiguous; ensure BCx is contiguous already.
        #    However, to keep Triton-only compute heavy, we will not reconstruct here in Triton (no tensor view ops in Triton).
        #    Instead, we keep BCx as is and perform gating with torch:
        #    This step is cheap and necessary to match original semantics.
        # Note: The original code uses chunk(3, dim=1) after transposing, but since we have BCx directly, we can index without transpose.
        #       We cannot use chunk here because BCx is (B,S,I); chunk operates on dim=1 which would be S. We need to split I into 3H.
        #       torch cannot do this in Triton; we use torch indexing:
        #       First, we create B, C, x_proj by slicing: B = BCx[:, :, :H], C = BCx[:, :, H:2*H], x_proj = BCx[:, :, 2*H:3*H]
        B_tensor = BCx[:, :, :H]   # (B, S, H)
        C_tensor = BCx[:, :, H:2*H]  # (B, S, H)
        x_proj_tensor = BCx[:, :, 2*H:]  # (B, S, H)

        # 3) Elementwise gating: Bx = B * x_proj
        Bx = B_tensor * x_proj_tensor  # elementwise multiply, torch

        # 4) Grouped causal conv: F.conv1d(Bx, conv_weight, conv_bias, groups=H)
        #    Bx shape: (B, H, S) because groups conv expects (N, C, L). Our tensor is (B, H, S).
        #    We must pad Bx along the last dimension to account for causal kernel_size=4:
        #    pad left by 3 (kernel_size-1). conv1d expects groups=H.
        padding = conv_weight.shape[2] - 1  # 4-1 = 3
        Bx_padded = F.pad(Bx, (padding, 0))  # pad left only
        # Now Bx_padded shape is (B, H, S + padding). conv1d with groups=H, kernel_size=4.
        conv_out = F.conv1d(Bx_padded, conv_weight, conv_bias, groups=H)

        # conv_out shape: (B, H, S), as expected

        # 5) Output gating: y = C * conv_out
        y = C_tensor * conv_out  # elementwise multiply

        # 6) Triton out_proj linear: y @ out_proj_weight^T -> output (B, S, H)
        output = torch.empty((B, S, H), device=x.device, dtype=torch.float32)
        y_contig = y.contiguous()
        out_proj_weight_contig = out_proj_weight.contiguous()

        grid_out = (B * S,)
        out_proj_linear_kernel[grid_out](
            y_contig, out_proj_weight_contig, output,
            B, S, H,
            BLOCK_H=128,
            num_warps=4,
        )

        return output

        w_ptrs = W_ptr + i * H + tl.arange(0, H)

        out_ptr_i = out_base + i

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def in_proj_linear_kernel(
    X_ptr,         # *ptr to input x: shape (B, S, H), contiguous
    W_ptr,         # *ptr to in_proj_weight: shape (I, H), contiguous, I = 3*H (constexpr)
    Out_ptr,       # *ptr to output: shape (B, S, I), contiguous
    B: tl.constexpr,
    S: tl.constexpr,
    H: tl.constexpr,   # hidden size, constexpr
    I: tl.constexpr,   # out channels, I = 3*H (constexpr)
):
    # Each program handles one (b, s)
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    x_base = X_ptr + b * (S * H) + s * H
    out_base = Out_ptr + b * (S * I) + s * I

    # For each output index i, compute acc[i] = sum_h x[b, s, h] * W[i, h]
    for i in range(0, I):
        # Load weights vector W[i, :]
        w_ptrs = W_ptr + i * H + tl.arange(0, H)
        w_vals = tl.load(w_ptrs).to(tl.float32)  # H is constexpr, so tl.arange(0, H) is valid

        # Load input vector x[b, s, :]
        x_vals = tl.load(x_base + tl.arange(0, H)).to(tl.float32)

        acc = tl.sum(w_vals * x_vals, axis=0)  # scalar

        # Store result at Out[b, s, i]
        out_ptr_i = out_base + i
        tl.store(out_ptr_i, acc)


@triton.jit
def out_proj_linear_kernel(
    Y_ptr,         # *ptr to input y: shape (B, S, H), contiguous
    W_ptr,         # *ptr to out_proj_weight: shape (H, H), contiguous
    Out_ptr,       # *ptr to output: shape (B, S, H), contiguous
    B: tl.constexpr,
    S: tl.constexpr,
    H: tl.constexpr,   # constexpr
):
    # Each program handles one (b, s)
    pid = tl.program_id(axis=0)
    b = pid // S
    s = pid % S

    y_base = Y_ptr + b * (S * H) + s * H
    out_base = Out_ptr + b * (S * H) + s * H

    # For each output index h_out in 0..H-1, compute sum over h_in
    for h_out in range(0, H):
        acc = 0.0
        for h_in in range(0, H):
            # y[b, s, h_in]
            y_val = tl.load(y_base + h_in).to(tl.float32)
            # W[h_out, h_in]
            w_val = tl.load(W_ptr + h_out * H + h_in).to(tl.float32)
            acc += y_val * w_val

        # Store result at Out[b, s, h_out]
        out_ptr = out_base + h_out
        tl.store(out_ptr, acc)
