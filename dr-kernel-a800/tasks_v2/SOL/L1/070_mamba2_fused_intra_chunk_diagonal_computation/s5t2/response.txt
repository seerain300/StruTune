import torch
import triton
import triton.language as tl

# Triton kernel to compute G[b, c, i, j, h] = sum_s C_expanded[b, c, i, h, s] * B_expanded[b, c, j, h, s]
# Grid: (B, C, L). Each program computes G for all j and h for a fixed (b, c, i).
@triton.jit
def g_outer_kernel(
    B_ptr,   # *float32, [B, C, L, H, S] after expand
    C_ptr,   # *float32, [B, C, L, H, S]
    G_ptr,   # *float32, [B, C, L, L, H]
    B_size, C_size, H, L, S,  # meta-parameters
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)

    # Loop over j and h, accumulate over s
    j = 0
    while j < L:
        h = 0
        while h < H:
            acc = 0.0
            s = 0
            while s < S:
                # Load B[b, c, j, h, s] and C[b, c, i, h, s]
                # Address for B:
                # offset_B = b*(C_size*L*H*S) + c*(L*H*S) + j*(H*S) + h*S + s
                offset_B = b * (C_size * L * H * S) + c * (L * H * S) + j * (H * S) + h * S + s
                B_val = tl.load(B_ptr + offset_B)

                offset_C = b * (C_size * L * H * S) + c * (L * H * S) + i * (H * S) + h * S + s
                C_val = tl.load(C_ptr + offset_C)

                acc += B_val * C_val
                s += 1
            # Store acc into G[b, c, i, j, h]
            # offset_G = b*(C_size*L*L*H) + c*(L*L*H) + i*(L*H) + j*H + h
            offset_G = b * (C_size * L * L * H) + c * (L * L * H) + i * (L * H) + j * H + h
            tl.store(G_ptr + offset_G, acc)
            h += 1
        j += 1

# Triton kernel to compute Y_diag[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden_states[b, c, j, h, d]
# Grid: (B, C, L, H)
@triton.jit
def y_diag_reduce_kernel(
    M_ptr,          # *float32, [B, C, L, L, H]
    HS_ptr,         # *float32, [B, C, L, H, D] = hidden_states
    Y_ptr,          # *float32, [B, C, L, H, D]
    B_size, C_size, L, H, D,
    BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    h = tl.program_id(3)

    # Accumulator vector over D
    d_offsets = tl.arange(0, BLOCK_D)
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # Loop over j to accumulate
    j = 0
    while j < L:
        # Load M[b, c, i, j, h]
        M_offset = b * (C_size * L * L * H) + c * (L * L * H) + i * (L * H) + j * H + h
        M_val = tl.load(M_ptr + M_offset)

        # Load hidden_states[b, c, j, h, d] vector for d in [0..BLOCK_D-1]
        HS_offset_base = b * (C_size * L * H * D) + c * (L * H * D) + j * (H * D) + h * D
        HS_vec = tl.load(HS_ptr + HS_offset_base + d_offsets, mask=d_offsets < D, other=0.0)

        acc += M_val * HS_vec
        j += 1

    # Store acc into Y[b, c, i, h, d]
    Y_offset_base = b * (C_size * L * H * D) + c * (L * H * D) + i * (H * D) + h * D
    # Store only valid d components
    tl.store(Y_ptr + Y_offset_base + d_offsets, acc, mask=d_offsets < D)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states: torch.Tensor,
                A_cumsum: torch.Tensor,
                B: torch.Tensor,
                C: torch.Tensor) -> torch.Tensor:
        """
        hidden_states: [B, C, L, H, D]
        A_cumsum:      [B, C, L, H]
        B:             [B, C, L, groups, S]
        C:             [B, C, L, groups, S]
        Returns:       [B, C, L, H, D] in bfloat16
        """

        B_size, C_size, L, H, D = hidden_states.shape
        groups = 8  # N_GROUPS
        GROUP_EXPAND = H // groups  # original uses NUM_HEADS=32, N_GROUPS=8 -> 4

        # Expand B and C from groups to H (num_heads)
        B_expanded = B.repeat_interleave(GROUP_EXPAND, dim=3)  # [B, C, L, H, S]
        C_expanded = C.repeat_interleave(GROUP_EXPAND, dim=3)  # [B, C, L, H, S]

        # Compute L (causal mask) using PyTorch to ensure correctness for any L:
        # Original code: expand A_cumsum to [B, H, C, L, L], apply lower-triangular mask (diagonal = -1),
        # then cumsum along -2 and exp. Since we can't rely on Triton for dynamic sizes, use torch.
        # We'll construct L as a [B, C, H, L, L] float32 tensor.
        L_mat = torch.empty((B_size, C_size, H, L, L), device=hidden_states.device, dtype=torch.float32)
        for b in range(B_size):
            for c in range(C_size):
                for h in range(H):
                    # Build lower-triangular mask over [L, L] with diagonal=-1
                    mask = torch.tril(torch.ones((L, L), device=hidden_states.device, dtype=torch.bool), diagonal=-1)
                    # Cumsum along L (internal chunk dimension) of A_cumsum[b, c, :, h]
                    # A_cumsum has shape [B, C, L, H], we need a vector of length L for fixed (b, c, h).
                    # addr for A[b, c, l, h] is A_ptr + b*(C*L*H) + c*(L*H) + l*H + h
                    # But torch indexing is simple: A_cumsum[b, c, :, h]
                    vec = A_cumsum[b, c, :, h].to(torch.float32)  # length L
                    cumsum = torch.cumsum(vec, dim=0)  # length L
                    # Broadcast cumsum to [L, L], then apply mask
                    cumsum_mat = cumsum[:, None]  # [L, 1] -> broadcast to [L, L]
                    L_mat[b, c, h, :, :] = torch.where(mask, cumsum_mat, torch.zeros((L, L), device=hidden_states.device, dtype=torch.float32))
                    # Apply exp to get causal decay
                    L_mat[b, c, h, :, :] = torch.exp(L_mat[b, c, h, :, :])

        # Compute G in Triton: G[b, c, i, j, h] = sum_s C_expanded[b, c, i, h, s] * B_expanded[b, c, j, h, s]
        G = torch.empty((B_size, C_size, L, L, H), device=hidden_states.device, dtype=torch.float32)

        # Launch Triton kernel
        g_outer_kernel[(B_size, C_size, L)](
            B_expanded, C_expanded, G,
            B_size, C_size, H, L, B_expanded.shape[-1],  # S = B_expanded's last dim (should equal C_expanded's last dim)
            num_warps=4
        )

        # Compute M = G * L (mask). L_mat is [B, C, H, L, L]; G is [B, C, L, L, H].
        # We need to broadcast M over H; simplest is to multiply elementwise with broadcasting.
        M = G * L_mat.permute(0, 1, 3, 4, 2)  # [B, C, L, L, H]

        # Compute Y_diag via Triton reduction over j
        Y = torch.empty((B_size, C_size, L, H, D), device=hidden_states.device, dtype=torch.float32)

        # Launch Triton reduction kernel
        # Choose BLOCK_D as min power-of-two up to 128; we can set 64 or 128. D is typically small.
        BLOCK_D = 64 if D <= 64 else 128
        y_diag_reduce_kernel[(B_size, C_size, L, H)](
            M, hidden_states.to(torch.float32), Y,
            B_size, C_size, L, H, D,
            BLOCK_D=BLOCK_D,
            num_warps=4
        )

        # Return in bfloat16
        return Y.to(torch.bfloat16)

- We keep tensors in float32 for numerical stability and cast to bfloat16 at the end to match the original function.

import torch
import triton
import triton.language as tl

# Triton kernel to compute Y_diag[b, c, i, h, d] = sum_j M[b, c, i, j, h] * hidden_states[b, c, j, h, d]
# Grid: (B, C, L, H)
@triton.jit
def y_diag_reduce_kernel(
    M_ptr,          # *float32, [B, C, L, L, H]
    HS_ptr,         # *float32, [B, C, L, H, D] = hidden_states
    Y_ptr,          # *float32, [B, C, L, H, D]
    B_size, C_size, L, H, D,
    BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    c = tl.program_id(1)
    i = tl.program_id(2)
    h = tl.program_id(3)

    # Vector of d offsets for this program (we only use contiguous D)
    d_offsets = tl.arange(0, BLOCK_D)
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # Loop over j to accumulate
    j = 0
    while j < L:
        # Load M[b, c, i, j, h]
        M_offset = b * (C_size * L * L * H) + c * (L * L * H) + i * (L * H) + j * H + h
        M_val = tl.load(M_ptr + M_offset)

        # Load hidden_states[b, c, j, h, d] vector for d in [0..BLOCK_D-1]
        HS_offset_base = b * (C_size * L * H * D) + c * (L * H * D) + j * (H * D) + h * D
        HS_vec = tl.load(HS_ptr + HS_offset_base + d_offsets, mask=d_offsets < D, other=0.0)

        acc += M_val * HS_vec
        j += 1

    # Store acc into Y[b, c, i, h, d]
    Y_offset_base = b * (C_size * L * H * D) + c * (L * H * D) + i * (H * D) + h * D
    tl.store(Y_ptr + Y_offset_base + d_offsets, acc, mask=d_offsets < D)

        A_cumsum:      [B, C, L, H]

        C:             [B, C, L, groups, S]
        Returns:       [B, C, L, H, D] in bfloat16

        # Original code expands A_cumsum to [B, H, C, L, L], applies lower-triangular mask (diagonal = -1),

        A_expanded = A_cumsum.unsqueeze(1).unsqueeze(2)  # [B, 1, C, L, L]
