import torch
import triton
import triton.language as tl


@triton.jit
def histogram_atomic_kernel(flat_ptr, counts_ptr, N, BLOCK: tl.constexpr):
    """
    Histogram of values in flat_ptr (int32), one pass with atomic adds.
    flat_ptr: *int32, shape [N]
    counts_ptr: *int32, shape [num_experts]
    N: int, total number of elements
    BLOCK: int, number of elements processed per program
    """
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # Load a block of indices; mask out-of-range with 0 (won't contribute to adds)
    ids = tl.load(flat_ptr + offsets, mask=mask, other=0)

    # For each valid id, atomically add to counts[id]
    for i in range(BLOCK):
        id_val = ids[i]
        # Only add if id is in range
        valid = mask[i]
        # id_val may be 0 due to mask; if so, we should skip (atomic on zero is fine).
        if valid:
            tl.atomic_add(counts_ptr + id_val, 1)


@triton.jit
def prefix_scan_kernel(counts_ptr, offsets_ptr, num_experts: tl.constexpr, BLOCK: tl.constexpr):
    """
    Compute prefix sum of counts_ptr (length num_experts) into offsets_ptr (length num_experts),
    then set offsets_ptr[0] = 0. We do a block-wise parallel scan using a single program
    that processes the entire num_experts range (BLOCK must be >= num_experts). For generality,
    we can launch multiple programs over blocks; here num_experts is small (256), so one program
    suffices. This kernel assumes we process the whole array. To keep it generic, we launch it
    once and rely on num_experts <= BLOCK. If num_experts > BLOCK, you would need a two-pass
    approach or a different kernel. Given the task and provided configs, num_experts is 256.

    counts_ptr: *int32, length num_experts
    offsets_ptr: *int32, length num_experts (we'll allocate length num_experts + 1 on host, and write
                  offsets into positions 1..num_experts, then host sets 0).
    num_experts: compile-time constant (we set at launch).
    BLOCK: compile-time constant >= num_experts.
    """
    # We assume one program processes all elements; offsets_ptr[0] is handled separately on host.
    running = 0
    for i in range(0, num_experts):
        ci = tl.load(counts_ptr + i)
        running += ci
        tl.store(offsets_ptr + i + 1, running)


class ModelNew(torch.nn.Module):
    def forward(self, topk_idx: torch.Tensor):
        # Ensure we are on CUDA; Triton requires CUDA tensors
        assert topk_idx.is_cuda, "topk_idx must be a CUDA tensor for Triton kernels."

        # Flatten to 1D
        flat = topk_idx.reshape(-1)

        # Compute sorted_token_indices: stable argsort of flattened values (matches .sort(stable=True) permutation)
        # Note: The original code sorts the values themselves, not by position; argsort on values gives the same permutation.
        sorted_token_indices = torch.argsort(flat, stable=True).to(torch.int32)

        # num_experts is fixed (256), per the original code
        num_experts = 256

        # Allocate counts for histogram
        counts = torch.zeros(num_experts, dtype=torch.int32, device=flat.device)

        # Launch histogram kernel
        N = flat.numel()
        # Choose a BLOCK size for histogram; 1024 works well across a range of N
        BLOCK = 1024
        grid = (triton.cdiv(N, BLOCK),)
        histogram_atomic_kernel[grid](flat, counts, N, BLOCK)

        # Compute expert_offsets (prefix sum of counts), including offset[0] = 0
        # We'll allocate offsets with length num_experts + 1
        expert_offsets = torch.empty(num_experts + 1, dtype=torch.int32, device=flat.device)

        # Run prefix scan kernel: since num_experts is 256, we can set BLOCK >= 256.
        # Using BLOCK=256 keeps the kernel simple; if you want, you can use 512 or 1024, but 256 is fine.
        BLOCK_EXPERTS = 256
        # We launch a single program that processes all 256 experts; this is sufficient and simple.
        prefix_scan_kernel[(1,)](counts, expert_offsets, num_experts, BLOCK_EXPERTS)

        # Return the same shapes/dtypes as original: sorted_token_indices (num_tokens,) and expert_offsets (num_experts+1,)
        return sorted_token_indices, expert_offsets

    Stable argsort of flat_ptr (int32, 1D) producing permutation in out_idx_ptr (int32, 1D).
    We implement a counting-based stable sort: for each chunk, we compute counts of values <= v,
    derive a local rank for each element considering original positions for ties, then scatter
    the indices to their stable destination. This is repeated for all chunks.

    flat_ptr: *int32, length N
    out_idx_ptr: *int32, length N (output permutation of indices)
    N: int32
    num_experts: int (compile-time constant, e.g., 256)
    BLOCK: int (number of elements processed per program instance)

    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < N

    # The kernel will be called multiple times (or we could run inside a loop, but Triton
    # doesn't support arbitrary loops over N; instead, we launch the kernel once and it
    # handles the whole range by walking through chunks from 0..N in BLOCK steps using

    # instance processes only one chunk, and host launches grid=(cdiv(N, BLOCK),). We
    # pass N and rely on multiple program instances. However, to compute per-element

    #   prev values, we cannot directly compute rank per element in-kernel. Therefore,
