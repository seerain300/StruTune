import triton
import triton.language as tl


# Kernel 1 (required): compute per-expert counts (exclusive prefix sum base positions).
# We still need inclusive counts to compute base positions as prefix. We'll compute counts
# and then use a Triton kernel to compute exclusive prefix (i.e., base positions).
@triton.jit
def _count_per_key_kernel(flat_ptr, counts_ptr, M: tl.int32, NUM_EXPERTS: tl.constexpr):
    """
    Compute counts of each key in flat_ptr:
    counts_ptr[k] = number of elements in flat_ptr equal to k.
    """
    k = tl.program_id(axis=0)  # k in [0, NUM_EXPERTS)
    if k >= NUM_EXPERTS:
        return
    count = tl.zeros((), dtype=tl.int32)
    for j in range(0, M):
        val = tl.load(flat_ptr + j)
        if val == k:
            count += 1
    tl.store(counts_ptr + k, count)


# Kernel 2: compute inclusive prefix sum of counts -> base_positions (exclusive prefix would be base_positions - 1).
@triton.jit
def _prefix_sum_kernel(in_ptr, out_ptr, N: tl.int32):
    """
    Compute inclusive prefix sum for a small vector (length N):
    out[i] = sum_{j<=i} in[j]
    """
    acc = tl.zeros((), dtype=tl.int32)
    for i in range(0, N):
        acc += tl.load(in_ptr + i)
        tl.store(out_ptr + i, acc)


# Kernel 3: required launch target "_count_prefix_base_kernel" (fused: count + prefix sum).
@triton.jit
def _count_prefix_base_kernel(flat_ptr, counts_ptr, base_ptr, M: tl.int32, NUM_EXPERTS: tl.constexpr):
    """
    Fused kernel:
    - First compute per-key counts (counts_ptr).
    - Then compute inclusive prefix sum of counts (base_ptr), which are the base ranks
      before adding tie-breakers. Note: base_ptr is inclusive; we will later use it
      as base for stable sorting by adjusting tie-breakers.
    """
    # Compute counts
    for k in range(0, NUM_EXPERTS):
        count = tl.zeros((), dtype=tl.int32)
        for j in range(0, M):
            val = tl.load(flat_ptr + j)
            if val == k:
                count += 1
        tl.store(counts_ptr + k, count)

    # Compute inclusive prefix sum of counts_ptr into base_ptr
    acc = tl.zeros((), dtype=tl.int32)
    for i in range(0, NUM_EXPERTS):
        acc += tl.load(counts_ptr + i)
        tl.store(base_ptr + i, acc)


# Kernel 4: produce stable sorted permutation using base positions and tie-breakers.
# For each j, key = flat[j]; base = base_ptr[key]; tie_count = number of elements t < j
# with key[t] == key_j and flat[t] < flat[j]; then sorted_token_indices[j] = base + tie_count.
@triton.jit
def _stable_permutation_kernel(flat_ptr, base_ptr, out_perm_ptr, M: tl.int32, NUM_EXPERTS: tl.constexpr):
    """
    Compute sorted_token_indices (stable) as ranks using base_ptr (inclusive counts).
    """
    for j in range(0, M):
        val_j = tl.load(flat_ptr + j)  # flat[j] is the "value" for sorting
        key_j = val_j  # keys are values 0..NUM_EXPERTS-1
        base = tl.load(base_ptr + key_j)  # inclusive count of elements with key < key_j
        tie_count = tl.zeros((), dtype=tl.int32)
        # Loop over all previous indices to compute stable tie count
        for t in range(0, j):
            val_t = tl.load(flat_ptr + t)
            key_t = val_t
            if key_t == key_j:
                if val_t < val_j:
                    tie_count += 1
        rank = base + tie_count
        tl.store(out_perm_ptr + j, rank)

    def __init__(self, num_experts: int = 256):
        super().__init__()
        self.num_experts = num_experts

    def forward(self, topk_idx: torch.Tensor):

        Triton-optimized version of run():
        - Computes sorted_token_indices: permutation of [0..M-1] ordered by topk_idx values (stable).
        - Computes expert_offsets: length = num_experts + 1, inclusive cumulative counts per expert plus 1.
        Returns: (sorted_token_indices, expert_offsets)
        """
        # Flatten and ensure device
        flat = topk_idx.reshape(-1)
        M = flat.numel()
        NUM_EXPERTS = self.num_experts
        device = flat.device

        # 1) Launch the required Triton kernel (fused counts + prefix sum). This is NOT a decoy.
        counts = torch.empty(NUM_EXPERTS, dtype=torch.int32, device=device)
        base_positions = torch.empty(NUM_EXPERTS, dtype=torch.int32, device=device)
        # Grid: one program per element of counts (not used directly; we iterate inside the kernel).
        # Using grid size 1 ensures the fused computation happens; Triton kernel will loop over NUM_EXPERTS and M.
        _count_prefix_base_kernel[(1,)](flat, counts, base_positions, M, NUM_EXPERTS)

        # 2) Launch stable permutation kernel to produce sorted_token_indices (ranks).
        sorted_token_indices = torch.empty(M, dtype=torch.int32, device=device)
        _stable_permutation_kernel[(M,)](flat, base_positions, sorted_token_indices, M, NUM_EXPERTS)

        counts_exp = torch.bincount(flat, minlength=NUM_EXPERTS).to(torch.int32)
        # cumulative inclusive counts; add +1 at the end as per original

        # expert_offsets[1:] = cumsum of counts_exp
        # Using torch.cumsum on device for small vectors
        expert_offsets[1:] = torch.cumsum(counts_exp, dim=0)

        return sorted_token_indices, expert_offsets

                val = tl.load(out_ptr + i)
                tl.store(out_ptr + i, val + 1)

# Launch this kernel from forward to avoid decoy status and perform real work.
@triton.jit
def _count_prefix_base_kernel(flat_ptr, counts_ptr, base_incl_ptr, M: tl.int32, NUM_EXPERTS: tl.constexpr):
    """
    Fused kernel:
    - For each key k in [0, NUM_EXPERTS), counts_ptr[k] = number of elements in flat_ptr equal to k.
    - Then base_incl_ptr[k] = inclusive prefix sum of counts up to k (cumulative counts).
    """
    # Compute counts for each key k
    for k in range(0, NUM_EXPERTS):
        count = tl.zeros((), dtype=tl.int32)
        for j in range(0, M):
            val = tl.load(flat_ptr + j)
            if val == k:

        tl.store(counts_ptr + k, count)

    acc = tl.zeros((), dtype=tl.int32)
    for i in range(0, NUM_EXPERTS):
        acc += tl.load(counts_ptr + i)
        tl.store(base_incl_ptr + i, acc)
