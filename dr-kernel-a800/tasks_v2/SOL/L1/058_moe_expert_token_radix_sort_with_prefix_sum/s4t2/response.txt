import torch
import triton
import triton.language as tl


@triton.jit
def _histogram_kernel(inp_ptr, N, out_ptr):
    """
    Triton histogram: one program per element.
    out_ptr has length num_experts (global), we'll pass num_experts via device-side indexing.
    Each program loads its index, casts to int32, and atomically increments out_ptr[idx].
    """
    pid = tl.program_id(axis=0)
    mask = pid < N
    # Load the index (assumed int32 from host), cast to int32 for atomic add
    idx_val = tl.load(inp_ptr + pid, mask=mask, other=0).to(tl.int32)
    # Atomically increment the bin for this index
    tl.atomic_add(out_ptr + idx_val, 1, mask=mask)

    def __init__(self):
        super().__init__()
        # No parameters; matches the evaluation harness requirement (NewModule).

    def forward(self, *args):
        """
        Triton-optimized version:
        - Flattens topk_idx, computes histogram in Triton, then creates expert_offsets via torch.cumsum.
        - Keeps the stable sort in PyTorch as in the original code.
        """
        if len(args) != 1:
            raise RuntimeError("ModelNew.forward expects a single tensor argument: topk_idx")
        topk_idx = args[0]

        # Ensure tensor is on CUDA and contiguous
