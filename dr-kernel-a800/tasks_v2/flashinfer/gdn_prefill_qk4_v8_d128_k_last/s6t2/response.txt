import triton

# Kernel: elementwise compute g = exp(-exp(A_log) * softplus(a + dt_bias)), beta = sigmoid(a + dt_bias)
# Inputs:
#   a_ptr: [T, H] bfloat16
#   dt_bias_ptr: [H] float32
#   A_log_ptr: [H] float32
#   g_ptr: [T, H] float32
#   beta_ptr: [T, H] float32
@triton.jit
def compute_g_beta_kernel(a_ptr, dt_bias_ptr, A_log_ptr, g_ptr, beta_ptr,
                          T: tl.int32, H: tl.int32):
    pid = tl.program_id(axis=0)
    t = pid // H
    h = pid % H
    if t >= T:
        return
    a = tl.load(a_ptr + t * H + h)  # bfloat16
    A_log = tl.load(A_log_ptr + h)  # float32
    dt_bias = tl.load(dt_bias_ptr + h)  # float32
    x = (a.float() + dt_bias)        # float32
    # softplus(x) = log(1 + exp(x))
    sp = tl.log(1.0 + tl.exp(x))     # float32
    g = tl.exp(-tl.exp(A_log) * sp)  # float32
    b = tl.sigmoid(x)                # float32
    tl.store(g_ptr + t * H + h, g)
    tl.store(beta_ptr + t * H + h, b)

    state_old_f = state_old  # float32

    # Compute old_v = k_vec @ state_old -> [N]
    old_v = tl.zeros((N,), dtype=tl.float32)
    for kk in tl.static_range(0, K):
        old_v += state_old_f[kk, :] * k_vec[kk]

    # Compute new_v = beta * v_vec + (1 - beta) * old_v
    new_v = beta_scalar * v_vec + (1.0 - beta_scalar) * old_v  # [N], float32

    # Compute state_remove[h, kk] = sum_n k_vec[kk] * old_v[n]
    state_remove = tl.zeros((K,), dtype=tl.float32)
    for kk in tl.static_range(0, K):
        state_remove[kk] = tl.sum(state_old_f[:, kk] * old_v, axis=0)

    # Compute state_update[h, kk] = sum_n k_vec[kk] * new_v[n]
    state_update = tl.zeros((K,), dtype=tl.float32)
    for kk in tl.static_range(0, K):
        state_update[kk] = tl.sum(state_old_f[:, kk] * new_v, axis=0)

    # Update new_state[h, :, :] = g * state_old + state_update - state_remove
    new_state = g_scalar * state_old_f
    # Add state_update and subtract state_remove
    for kk in tl.static_range(0, K):
        new_state[:, kk] += state_update[kk]
    for n in tl.static_range(0, N):
        new_state[n, :] -= state_remove[n]

    # Store new_state[h, :, :]
    tl.store(new_state_ptr + h * (N * N) + n_idx[:, None] * N + n_idx[None, :],
             new_state, mask=(n_idx[:, None] < N) & (n_idx[None, :] < N))

    # Compute output vector for this head: out_vec = scale * q_vec @ new_state[h, :, :]
    # We'll do it via row_matmul kernel using A=q_vec, B=new_state[h, :, :], C=out_vec
    # Prepare A_ptr as 1xK, B_ptr as [N, N], C_ptr as [N]
    # Note: Triton matmul kernels expect 2D A and B; but here A is 1xK. We can emulate by setting A[0, :] = q_vec and B as new_state, then multiply row by columns. Implement directly here:
    out_vec = tl.zeros((N,), dtype=tl.float32)
    for kk in tl.static_range(0, K):
        out_vec += new_state[:, kk] * q_vec[kk]
    out_vec = scale * out_vec
    tl.store(out_vec_ptr + n_idx, out_vec, mask=n_idx < N)


# Kernel: row-wise matmul for output: A[K] @ B[N, N] -> C[N] with scale
# Inputs:
#   A_ptr: [K], bfloat16 (q_exp[t, h, :])
#   B_ptr: [N, N], float32 (new_state[t, h, :])
#   C_ptr: [N], float32
@triton.jit
def row_matmul_kernel(A_ptr, B_ptr, C_ptr,
                      K: tl.int32, N: tl.int32, scale: tl.float32):
    k_idx = tl.arange(0, K)
    n_idx = tl.arange(0, N)
    A = tl.load(A_ptr + k_idx, mask=k_idx < K, other=0.0).float()  # [K]
    B = tl.load(B_ptr + n_idx[:, None] * N + n_idx[None, :], mask=(n_idx[:, None] < N) & (n_idx[None, :] < N), other=0.0)  # [N, N]
    C = tl.zeros((N,), dtype=tl.float32)
    for kk in tl.static_range(0, K):
        C += B[:, kk] * A[kk]
    C = scale * C
    tl.store(C_ptr + n_idx, C, mask=n_idx < N)
