    q_ptr, state_ptr, out_ptr,
    T, H, V, K,
    seq_start,  # int
    scale,      # float

    # out[h, k] = scale * sum_v (q[h, k] @ state[h, v, k])
    t = seq_start

        out_vec = tl.zeros([K], dtype=tl.float32)
        for v_i in range(0, V):
            # q[h, K]
            q_vec = tl.zeros([K], dtype=tl.float32)
            for k_i in range(0, K):
                q_off = t * (H * K) + h * K + k_i
                q_elem = tl.load(q_ptr + q_off).to(tl.float32)
                q_vec[k_i] = q_elem

            # state[h, v_i, K] as vector
            state_vec = tl.zeros([K], dtype=tl.float32)
            for k_i in range(0, K):
                state_off = (t * (H * V * K)) + h * (V * K) + v_i * K + k_i
                state_elem = tl.load(state_ptr + state_off).to(tl.float32)
                state_vec[k_i] = state_elem

            # out_vec += q_vec @ state_vec
            # Implement as sum
            for k_j in range(0, K):
                dot_j = 0.0
                for kk in range(0, K):
                    dot_j += q_vec[kk] * state_vec[k_j]
                out_vec[k_j] = dot_j

        # Store out[h, k] as bfloat16
        for k_i in range(0, K):
            out_off = t * (H * K) + h * K + k_i
            tl.store(out_ptr + out_off, out_vec[k_i].to(tl.bfloat16))
