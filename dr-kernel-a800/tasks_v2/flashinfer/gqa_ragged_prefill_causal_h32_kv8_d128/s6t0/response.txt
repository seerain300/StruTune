        q_offset = i * q_stride_b + h * q_stride_h  # head stride is head_dim
        # If tensors are contiguous [N, H, D], then q_stride_b = H*D, q_stride_h = D, q_stride_d = 1
        # We'll pass actual strides from torch.
        q_val = tl.load(q_ptr + q_offset)

        # Load k_expanded[j, h], v_expanded[j, h]
        k_offset = j * k_stride_b + h * k_stride_h
        v_offset = j * v_stride_b + h * v_stride_h

        k_val = tl.load(k_ptr + k_offset)
        v_val = tl.load(v_ptr + v_offset)

        score = q_val * k_val * scale

        # Causal mask: i can only attend to j < i + 1 + delta where delta = num_kv_tokens - num_q_tokens
        delta = num_kv_tokens - num_q_tokens
        causal = (j < (i + 1 + delta))
        if not causal:
            score = -float("inf")

        # Store logits[i, h, j]
        logits_offset = i * logits_stride_q + h * logits_stride_h + j * logits_stride_j
        tl.store(logits_ptr + logits_offset, score)
