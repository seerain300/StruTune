Analysis of the original code and optimization plan

- The original run function processes per batch b defined by qo_indptr and kv_indptr:

  - It computes lse = logsumexp(logits, dim=KV) / log(2) for each query token and head, and returns output and lse.

- We can implement this in Triton by:

  - Kernel 2 (compute_output_from_logits): Given the logits (with mask applied), compute softmax along KV dimension per (query, head), and perform the matvec with V_expanded to produce the final output. This kernel also writes the lse per (query, head).

try:
    import triton

    TRITON_AVAILABLE = True

# Triton kernel 1: compute logits (Q @ K^T) with scaling and causal mask, and store row-wise maxima for lse
@triton.jit
def compute_logits_masked_lse(
    q_ptr,  # *float32, [num_q_tokens, num_qo_heads, head_dim]
    k_ptr,  # *float32, [num_kv_tokens, num_qo_heads, head_dim] (this is K expanded to 32 heads)
    mask_ptr,  # *int8, [num_q_tokens, num_kv_tokens]
    logits_ptr,  # *float32, [num_q_tokens, num_qo_heads, num_kv_tokens]
    lse_row_ptr,  # *float32, [num_q_tokens, num_qo_heads]
    sm_scale,  # float32 scalar
    num_q_tokens: tl.constexpr,
    num_kv_tokens: tl.constexpr,
    num_qo_heads: tl.constexpr,
    head_dim: tl.constexpr,
    BLOCK_Q: tl.constexpr,
    BLOCK_KV: tl.constexpr,
    BH: tl.constexpr
):
    # Grid: (num_batches, ceil_div(num_qo_heads, BH), ceil_div(num_q_tokens, BLOCK_Q))
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_q = tl.program_id(2)

    qo_start = pid_b * num_q_tokens  # start token index for this batch
    qo_end = (pid_b + 1) * num_q_tokens
    # heads covered by this tile
    h_start = pid_h * BH

    # qo mask for this tile
    qo_mask = pid_q * BLOCK_Q + tl.arange(0, BLOCK_Q)
    valid_q = qo_mask < (qo_end - qo_start)

    # For each head h in tile, compute logits block
    for hh in range(BH):
        h = h_start + hh
        if h >= num_qo_heads:
            break

        # Initialize logits_acc for this head
        logits_acc = tl.zeros((BLOCK_Q, 1), dtype=tl.float32)  # we only store per-row max; mat itself not stored

        # Loop over KV tiles to accumulate matmul Qh @ Kh^T
        for kv_off in range(0, num_kv_tokens, BLOCK_KV):
            kv_mask = kv_off + tl.arange(0, BLOCK_KV)
            valid_k = kv_mask < num_kv_tokens

            # Load Q block: shape [BLOCK_Q, head_dim]
            q_block = tl.load(
                q_ptr + (qo_start + qo_mask[:, None]) * (num_qo_heads * head_dim) + h * head_dim,
                mask=valid_q[:, None],
                other=0.0
            )  # [BLOCK_Q, head_dim]

            # Load K block: shape [BLOCK_KV, head_dim]
            k_block = tl.load(
                k_ptr + (kv_mask[:, None]) * (num_qo_heads * head_dim) + h * head_dim,
                mask=valid_k[:, None],
                other=0.0
            )  # [BLOCK_KV, head_dim]

            # Dot product: [BLOCK_Q, head_dim] @ [head_dim, BLOCK_KV] => [BLOCK_Q, BLOCK_KV]
            acc = tl.dot(q_block, tl.trans(k_block))  # [BLOCK_Q, BLOCK_KV]
            acc = acc * sm_scale

            # Build mask for this tile
            mask_tile = tl.load(
                mask_ptr + qo_mask[:, None] * num_kv_tokens + (kv_off + tl.arange(0, BLOCK_KV))[None, :],
                mask=valid_q[:, None] & valid_k[None, :],
                other=0  # 0 means False; we'll convert to -inf
            )  # [BLOCK_Q, BLOCK_KV] as int8

            # Apply mask: invalid entries set to -inf
            acc = tl.where(mask_tile != 0, acc, -float('inf'))

            # Accumulate into logits_acc
            logits_acc += acc

        # Store per-row maxima to lse_row_ptr: shape [num_q_tokens, num_qo_heads]
        for i in range(BLOCK_Q):
            row_max = tl.max(logits_acc[i, 0], axis=0)  # logits_acc is [1] dim, but we need scalar
            # Note: Triton expects row_max as a scalar; since logits_acc is [BLOCK_Q, 1], use tl.max over axis=1 (columns).
            # However, acc was [BLOCK_Q, BLOCK_KV], so row_max should be computed from acc. We need to recompute per row.
            # Instead, we recompute row_max by taking max over BLOCK_KV columns of acc (we saved acc per iteration).
            # But we already summed acc into logits_acc. To get max, we need to track max. Let's do it by recomputing max per row.
            # We'll recompute row_max by taking max across BLOCK_KV entries of acc for each i.
            # Since acc is per iteration, we can compute row_max by taking max across the final logits_acc's second dim.
            # However, Triton requires explicit computation. We'll set row_max as max of acc across last dim.
            # For simplicity, we store max of acc across its second dim. Since acc has shape [BLOCK_Q, BLOCK_KV], we compute
            # row_max = max(acc[i, :]). But we must compute this per iteration? Triton supports reductions.
            # We'll compute row_max by reducing acc across its columns: tl.max(acc, axis=1) -> [BLOCK_Q].
            row_max = tl.max(acc, axis=1)[i]  # take i-th row max
            tl.store(
                lse_row_ptr + (qo_start + qo_mask[i]) * num_qo_heads + h,
                row_max
            )

        # Store masked logits to logits_ptr for potential later use in output computation
        # We store logits_acc for each head tile. Note: lse is computed from max of original logits before softmax, so storing logits_acc here would be redundant.
        # Instead, we'll store masked acc for each h tile if needed. But output kernel will recompute softmax from original Q @ K^T + mask.
        # Therefore, we need to recompute logits in output kernel. For now, we only compute and store lse.


# Triton kernel 2: compute output (attention + matvec) and final lse (logsumexp over KV) per (query, head)
@triton.jit
def compute_output_from_logits(
    q_ptr,  # *float32, [num_q_tokens, num_qo_heads, head_dim]
    k_ptr,  # *float32, [num_kv_tokens, num_qo_heads, head_dim] (K expanded)
    logits_ptr,  # *float32, [num_q_tokens, num_qo_heads, num_kv_tokens]
    v_ptr,  # *float32, [num_kv_tokens, num_qo_heads, head_dim] (V expanded)
    output_ptr,  # *float32, [num_q_tokens, num_qo_heads, head_dim]
    lse_row_ptr,  # *float32, [num_q_tokens, num_qo_heads]
    sm_scale,  # float32
    num_q_tokens: tl.constexpr,
    num_kv_tokens: tl.constexpr,
    num_qo_heads: tl.constexpr,
    head_dim: tl.constexpr,
    BLOCK_Q: tl.constexpr,
    BLOCK_KV: tl.constexpr,
    BH: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_q = tl.program_id(2)

    qo_start = pid_b * num_q_tokens
    kv_start = pid_b * num_kv_tokens

    h_start = pid_h * BH
    qo_mask = pid_q * BLOCK_Q + tl.arange(0, BLOCK_Q)
    valid_q = qo_mask < num_q_tokens

    for hh in range(BH):
        h = h_start + hh
        if h >= num_qo_heads:
            break

        # Load per-(query,head) row max for normalization (lse)
        lse_val = tl.load(lse_row_ptr + (qo_start + qo_mask) * num_qo_heads + h, mask=valid_q, other=0.0)  # [BLOCK_Q]

        # Initialize output accumulator for this head
        out_acc = tl.zeros((BLOCK_Q, head_dim), dtype=tl.float32)

        # Loop over KV tiles to compute softmax per row and matvec
        for kv_off in range(0, num_kv_tokens, BLOCK_KV):
            kv_mask = kv_off + tl.arange(0, BLOCK_KV)
            valid_k = kv_mask < num_kv_tokens

            # Load Q block
            q_block = tl.load(
                q_ptr + (qo_start + qo_mask[:, None]) * (num_qo_heads * head_dim) + h * head_dim,
                mask=valid_q[:, None],
                other=0.0
            )  # [BLOCK_Q, head_dim]

            # Load logits tile: [BLOCK_Q, BLOCK_KV]
            logits_tile = tl.load(
                logits_ptr + (qo_start + qo_mask[:, None]) * (num_qo_heads * num_kv_tokens) + h * num_kv_tokens + kv_mask[None, :],
                mask=valid_q[:, None] & valid_k[None, :],
                other=-float('inf')
            )  # [BLOCK_Q, BLOCK_KV]

            # Load K block: [BLOCK_KV, head_dim]
            k_block = tl.load(
                k_ptr + (kv_start + kv_mask[:, None]) * (num_qo_heads * head_dim) + h * head_dim,
                mask=valid_k[:, None],
                other=0.0
            )  # [BLOCK_KV, head_dim]

            # Softmax along KV dimension: subtract lse, exp, sum, normalize
            logits_tile = logits_tile - lse_val[:, None]  # broadcast over columns
            exp_tile = tl.exp(logits_tile)
            sum_tile = tl.sum(exp_tile, axis=1)  # [BLOCK_Q]
            soft_tile = exp_tile / sum_tile[:, None]  # [BLOCK_Q, BLOCK_KV]

            # Compute output: out_acc += q_block * sum_j soft_tile[:, j] * k_block[j, :]
            # We can do this by looping over kv tiles and accumulating:
            # out_acc += sum(soft_tile * k_block) per column. Since k_block has head_dim columns, we need to dot each column with q_block.
            # However, k_block is [BLOCK_KV, head_dim]. We want to multiply soft_tile[:, j] with k_block[j, :] and accumulate.
            # We'll do a per-column accumulation by looping j in tile:
            for jj in range(BLOCK_KV):
                valid_j = (kv_off + jj) < num_kv_tokens
                # Extract the j-th column soft and K
                soft_col = soft_tile[:, jj]  # [BLOCK_Q]
                k_col = k_block[jj, :]       # [head_dim]
                # Broadcast soft_col to [BLOCK_Q, head_dim] and multiply with k_col broadcast to [BLOCK_Q, head_dim]
                # Then accumulate: q_block * soft_col[:, None] * k_col[None, :]
                # However, Triton requires elementwise ops. We can compute q_block * soft_col[:, None] and then dot with k_col by broadcasting:
                # But since k_col is [head_dim], and q_block is [BLOCK_Q, head_dim], we can compute:
                # We need to multiply q_block by soft_col[:, None] and then multiply by k_col (broadcast), then reduce over head_dim.
                # This is tricky; better approach: compute per j, out_acc += sum(soft_tile[:, j] * k_block[j, :]) * q_block.
                # That is: out_acc += soft_tile[:, j][:, None] * k_block[jj, None, :] * q_block  -- but Triton doesn't support broadcasting like that.
                # Instead, we'll compute the contribution for each j and add it to out_acc:
                # We'll loop over head_dim dimension and accumulate. But we need a way to access individual columns of k_block.
                # Triton doesn't allow direct indexing like k_block[:, d]. We'll instead compute via tl.dot with q_block and soft_col*k_col vector.
                # To do that, we need to dot q_block (BLOCK_Q, head_dim) with a vector (head_dim), which Triton supports:
                # contribution_vec = soft_col * k_col  # [head_dim]
                contribution_vec = soft_col * k_col  # [head_dim]
                # Now out_acc += q_block * contribution_vec[:, None], where q_block * contribution_vec[:, None] yields [BLOCK_Q, head_dim]
                # This is the same as output for this j across all head_dim dims. But we must add this contribution to out_acc.
                # Triton allows elementwise multiply and tl.store; we'll add:
                out_acc += q_block * contribution_vec[:, None]  # [BLOCK_Q, head_dim]

        # Store output for this head tile
        tl.store(
            output_ptr + (qo_start + qo_mask[:, None]) * (num_qo_heads * head_dim) + h * head_dim,
            out_acc,
            mask=valid_q[:, None]
        )
