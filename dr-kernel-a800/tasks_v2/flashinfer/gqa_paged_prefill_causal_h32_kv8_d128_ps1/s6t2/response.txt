```python
# Inputs:
#   - q_vec_ptr: pointer to q[global_q_idx, h, :] of length head_dim (float32)
#   - k_seg_ptr: pointer to k segment [num_kv_tokens, head_dim] (float32)
#   - v_seg_ptr: pointer to v segment [num_kv_tokens, head_dim] (float32)
#   - out_ptr: pointer to output vector [head_dim] (float32)
#   - lse_ptr: pointer to a single scalar lse (float32)
#   - num_kv_tokens: int32, length of the segment
#   - sm_scale: float32 scaling factor
#   - head_dim: tl.constexpr, specialization to this value
#   - CHUNK: tl.constexpr, chunk size for vectorized iteration over the segment
@triton.jit
def head_kernel_2d_final(
    q_vec_ptr,        # *fp32, [head_dim]
    k_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    v_seg_ptr,        # *fp32, [num_kv_tokens, head_dim]
    out_ptr,          # *fp32, [head_dim]
    lse_ptr,          # *fp32, scalar
    num_kv_tokens: tl.int32,
    sm_scale: tl.float32,
    head_dim: tl.constexpr,
    CHUNK: tl.constexpr
):
    # Pass 1: compute m = max(logits_scaled) across the segment
    m = -float("inf")
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask = offs < num_kv_tokens
        # Load q vector (float32)
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim), mask=mask, other=0.0)
        # Compute logits_scaled for this chunk
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            valid_i = idx_i < num_kv_tokens
            # Load k_row and v_row (masked by valid_i)
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            v_row_ptr = v_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim), mask=valid_i, other=0.0)
            # Dot product over head_dim
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits[i] = dot * sm_scale
        # Update m
        m = tl.maximum(m, tl.max(tl.where(mask, logits, -float("inf")), axis=0))

    # Compute l = sum(exp(logits - m)) across all chunks
    l = 0.0
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim), mask=mask, other=0.0)
        logits = tl.zeros([CHUNK], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            valid_i = idx_i < num_kv_tokens
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            v_row_ptr = v_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim), mask=valid_i, other=0.0)
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits[i] = dot * sm_scale
        l += tl.sum(tl.exp(tl.where(mask, logits - m, -float("inf"))), axis=0)

    ln2 = 1.0 / math.log(2.0)
    lse_val = tl.log(l) / ln2
    # Store lse
    tl.store(lse_ptr, lse_val)

    # Pass 2: compute output vector
    for start in range(0, num_kv_tokens, CHUNK):
        offs = start + tl.arange(0, CHUNK)
        mask = offs < num_kv_tokens
        q_vec = tl.load(q_vec_ptr + tl.arange(0, head_dim), mask=mask, other=0.0)
        out_vec = tl.zeros([head_dim], dtype=tl.float32)
        for i in range(0, CHUNK):
            idx_i = offs[i]
            valid_i = idx_i < num_kv_tokens
            k_row_ptr = k_seg_ptr + idx_i * head_dim
            v_row_ptr = v_seg_ptr + idx_i * head_dim
            k_row = tl.load(k_row_ptr + tl.arange(0, head_dim), mask=valid_i, other=0.0)
            dot = 0.0
            for j in range(0, head_dim):
                dot += q_vec[j] * k_row[j]
            logits_i = dot * sm_scale
            attn_i = tl.exp(logits_i - lse_val)
            v_row = tl.load(v_row_ptr + tl.arange(0, head_dim), mask=valid_i, other=0.0)
            out_vec += attn_i * v_row
        # Store output vector
        tl.store(out_ptr + tl.arange(0, head_dim), out_vec, mask=mask)

    def forward(self, q, k_cache, v_cache, qo_indptr, kv_indptr, kv_indices, sm_scale):

        head_dim = q.shape[2]
        num_kv_heads = k_cache.shape[2]

        v_cache_flat = v_cache.squeeze(1)  # [num_pages, num_kv_heads, head_dim]

        assert total_q == qo_indptr[-1].item(), "Sum of qo_indptr must equal total_q."
        # Compute output and lse
        output = torch.zeros((total_q, num_qo_heads, head_dim), dtype=torch.bfloat16, device=device)

        gqa_ratio = num_qo_heads // num_kv_heads

        # Iterate over batch segments
        for b in range(0, qo_indptr.numel() - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            num_q_tokens = q_end - q_start
            num_kv_tokens = kv_end - kv_start

            if num_q_tokens <= 0 or num_kv_tokens <= 0:
                continue

            # Gather "page_ids" for this batch
            # Ensure kv_indices is on the same device
            if kv_indices.device != device:
                kv_indices = kv_indices.to(device)
            kv_indices_b = kv_indices[kv_start:kv_end]  # [num_kv_tokens]

            # For each query in the batch
            for q_idx in range(num_q_tokens):
                global_q_idx = q_start + q_idx

                # Causal window: max number of kv tokens this query sees
                delta = num_kv_tokens - num_q_tokens
                max_kv_idx = min(q_idx + 1 + delta, num_kv_tokens)
                if max_kv_idx <= 0:
                    continue

                # For each head
                for h in range(num_qo_heads):
                    kv_head = h // gqa_ratio  # 0..7

                    # Gather q vector for this head
                    q_vec = q[global_q_idx, h, :].to(torch.float32).contiguous()  # [head_dim]
                    # Gather k and v rows for this batch
                    # Note: k_cache_flat shape [num_pages, num_kv_heads, head_dim]
                    # We select rows for each page id in kv_indices_b[:max_kv_idx] and kv_head
                    k_rows = k_cache_flat[kv_indices_b[:max_kv_idx], kv_head, :].to(torch.float32).contiguous()
                    v_rows = v_cache_flat[kv_indices_b[:max_kv_idx], kv_head, :].to(torch.float32).contiguous()

                    # Allocate output vector and lse scalar
                    out_vec = torch.empty(head_dim, dtype=torch.float32, device=device)
                    lse_buf = torch.empty(1, dtype=torch.float32, device=device)

                    # Launch Triton kernel: one program per (b, q_idx, h)
                    CHUNK = 256  # chunk size for vectorized iteration over the segment
                    head_kernel_2d_final[(1,)](
                        q_vec,                 # *fp32 [head_dim]
                        k_rows,                # *fp32 [max_kv_idx, head_dim]
                        v_rows,                # *fp32 [max_kv_idx, head_dim]
                        out_vec,               # *fp32 [head_dim]
                        lse_buf,               # *fp32 scalar
                        num_kv_tokens=max_kv_idx,
                        sm_scale=float(sm_scale),
                        head_dim=head_dim,
                        CHUNK=CHUNK
                    )

                    # Store results
                    output[global_q_idx, h, :] = out_vec.to(torch.bfloat16)
                    lse[global_q_idx, h] = lse_buf[0]

        return output, lse
```

```python
        for b in range(0, qo_indptr.numel() - 1):
            q_start = int(qo_indptr[b].item())
            q_end = int(qo_indptr[b + 1].item())
            kv_start = int(kv_indptr[b].item())
            kv_end = int(kv_indptr[b + 1].item())

            num_q_tokens = q_end - q_start
            num_kv_tokens = kv_end - kv_start

            if num_q_tokens <= 0 or num_kv_tokens <= 0:
                continue

            # kv_indices must be on device
            if kv_indices.device != device:
                kv_indices = kv_indices.to(device)
            kv_indices_b = kv_indices[kv_start:kv_end]  # [num_kv_tokens]

            for q_idx in range(num_q_tokens):
                global_q_idx = q_start + q_idx

                # Causal window: max number of kv tokens this query sees
                delta = num_kv_tokens - num_q_tokens
                max_kv_idx = min(q_idx + 1 + delta, num_kv_tokens)
                if max_kv_idx <= 0:
                    continue

                # For each head
                for h in range(num_qo_heads):
                    kv_head = h // gqa_ratio  # 0..7

                    # Gather q vector for this head
                    q_vec = q[global_q_idx, h, :].to(torch.float32).contiguous()  # [head_dim]

                    # Gather k and v rows for this batch
                    k_rows = k_cache_flat[kv_indices_b[:max_kv_idx], kv_head, :].to(torch.float32).contiguous()  # [max_kv_idx, head_dim]
                    v_rows = v_cache_flat[kv_indices_b[:max_kv_idx], kv_head, :].to(torch.float32).contiguous()  # [max_kv_idx, head_dim]

                    # Compute lse using Triton
                    lse_buf = torch.empty(1, dtype=torch.float32, device=device)
                    CHUNK = 128  # tile size; 128 matches head_dim in the original
                    compute_lse_kernel[(1,)](
                        q_vec,                 # *fp32 [head_dim]
                        k_rows,                # *fp32 [max_kv_idx, head_dim]
                        lse_buf,               # *fp32 scalar
                        num_kv_tokens=max_kv_idx,
                        sm_scale=float(sm_scale),
                        head_dim=head_dim,
                        CHUNK=CHUNK
                    )
                    lse_val = lse_buf[0]

                    # Compute output vector using Triton
                    out_vec = torch.empty(head_dim, dtype=torch.float32, device=device)
                    compute_output_kernel[(1,)](
                        q_vec,                 # *fp32 [head_dim]
                        k_rows,                # *fp32 [max_kv_idx, head_dim]
                        v_rows,                # *fp32 [max_kv_idx, head_dim]
                        lse_val,               # float32
                        out_vec,               # *fp32 [head_dim]
                        num_kv_tokens=max_kv_idx,
                        sm_scale=float(sm_scale),
                        head_dim=head_dim,
                        CHUNK=CHUNK
                    )

                    # Store results
                    output[global_q_idx, h, :] = out_vec.to(torch.bfloat16)
                    lse[global_q_idx, h] = lse_val

        return output, lse
```

Plan to fix:
- Replace the 2D tile loads and reductions with a simpler, robust design that uses 1D vectors for k and v and loops over the segment in chunks. This avoids Triton’s complex broadcasting and reduces compile-time complexity. It should compile reliably and run in a predictable amount of time.
- Keep the exact original get_inputs() function so the evaluator can unpack axis values correctly and generate inputs as intended.
- Implement two Triton kernels:
  1) compute_lse_kernel: For a given (b, q_idx, h), iterate over the segment in chunks of size CHUNK, compute dot products q · k_row for each i in the chunk, and update m (max) and l (sum of exp(logits - m)). Store lse = log(l)/ln(2) in a scalar.
  2) compute_output_kernel: Same loop structure, recompute logits, compute attn = exp(logits - lse), and accumulate out += attn * v_row for each i. Store the output vector.
- Host code: For each (b, q_idx, h), gather q vector, k_rows, v_rows, launch reduction kernel to get lse, then launch output kernel to write the output. This ensures TRITON-ONLY compliance: all computation happens inside Triton, and ModelNew.forward only allocates, indexes, and launches kernels.

Why this should pass:
- The kernel uses only 1D vector operations (tl.load, tl.arange, reductions) and simple scalar loops. This avoids Triton’s problematic broadcasting and 2D tile semantics that previously caused compilation issues and timeouts.
- CHUNK is set to 128, which matches the original head_dim, but the loops are general and will work for any num_kv_tokens. We mask out-of-range indices.
- We use compute in float32 inside the kernel and store outputs as bfloat16 and lse as float32, matching the original code’s expected dtypes.
