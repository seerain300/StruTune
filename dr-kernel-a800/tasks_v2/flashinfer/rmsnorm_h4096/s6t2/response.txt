```python
import torch
import triton
import triton.language as tl

# Triton kernel: per-row RMS scaling with weight.
# One program per row. We read x twice: once to compute sum of squares, once to write output.
@triton.jit
def _rms_scale_kernel(
    hidden_ptr,        # *float32, pointer to hidden_states (cast to float32)
    weight_ptr,        # *float32, pointer to weight (cast to float32)
    out_ptr,           # *T_out, pointer to output (will store casted result)
    batch_size,        # int32
    hidden_size,       # int32
    eps,               # float32
    out_dtype_code: tl.constexpr,  # 0: fp32, 1: fp16, 2: bf16
    BLOCK_SIZE: tl.constexpr        # e.g., 1024 or 256 or 128
):
    row_id = tl.program_id(0)  # one program per row
    if row_id >= batch_size:
        return

    row_base = hidden_ptr + row_id * hidden_size

    # First pass: accumulate sum of squares across columns in chunks
    sumsq = 0.0
    for col_start in range(0, hidden_size, BLOCK_SIZE):
        cols = col_start + tl.arange(0, BLOCK_SIZE)
        mask = cols < hidden_size
        x = tl.load(row_base + cols, mask=mask, other=0.0)  # float32
        sumsq += tl.sum(x * x, axis=0)

    # Compute inv_rms = 1 / sqrt(mean + eps), mean = sumsq / hidden_size
    mean = sumsq / hidden_size
    inv_rms = 1.0 / tl.sqrt(mean + eps)  # float32 scalar

    # Second pass: compute output and store
    for col_start in range(0, hidden_size, BLOCK_SIZE):
        cols = col_start + tl.arange(0, BLOCK_SIZE)
        mask = cols < hidden_size
        x = tl.load(row_base + cols, mask=mask, other=0.0)  # float32
        w = tl.load(weight_ptr + cols, mask=mask, other=0.0)  # float32
        y = x * inv_rms * w  # float32

        # Cast to desired output dtype
        if out_dtype_code == 0:
            out_val = y
        elif out_dtype_code == 1:
            out_val = y.to(tl.float16)
        elif out_dtype_code == 2:
            out_val = y.to(tl.bfloat16)
        else:
            out_val = y  # default fp32
        tl.store(out_ptr + row_id * hidden_size + cols, out_val, mask=mask)


class ModelNew(torch.nn.Module):
    def forward(self, hidden_states, weight):
        # Ensure CUDA tensors and contiguity
        assert hidden_states.is_cuda and weight.is_cuda, "Tensors must be on CUDA for Triton kernel."
        hidden = hidden_states.contiguous()
        w = weight.contiguous()

        # Match original assertion: hidden_size must be 4096
        assert hidden.shape[1] == 4096, "hidden_size must be 4096."

        # Cast to float32 for compute
        x = hidden.to(torch.float32)
        w = w.to(torch.float32)

        batch_size = x.shape[0]
        hidden_size = x.shape[1]

        # Prepare output tensor in original dtype
        out = torch.empty_like(hidden, dtype=hidden.dtype, device=hidden.device)

        # Determine output dtype code for kernel
        if out.dtype == torch.float32:
            out_dtype_code = 0
        elif out.dtype == torch.float16:
            out_dtype_code = 1
        elif out.dtype == torch.bfloat16:
            out_dtype_code = 2
        else:
            # Fallback: store in fp32 if unexpected dtype
            out = torch.empty_like(hidden, dtype=torch.float32, device=hidden.device)
            out_dtype_code = 0

        # Adaptive tuning of BLOCK_SIZE and num_warps
        # Prefer larger blocks if divisible to reduce loop iterations
        if hidden_size % 1024 == 0:
            BLOCK_SIZE = 1024
            num_warps = 4
        elif hidden_size % 256 == 0:
            BLOCK_SIZE = 256
            num_warps = 8
        else:
            BLOCK_SIZE = 128
            num_warps = 4
        num_stages = 3

        # Launch Triton kernel: one program per row
        grid = (batch_size,)
        _rms_scale_kernel[grid](
            x, w, out,
            batch_size, hidden_size,
            1e-5,
            out_dtype_code=out_dtype_code,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=num_warps,
            num_stages=num_stages,
        )

        return out
