```python
import math
import torch

import triton
import triton.language as tl


@triton.jit
def _forward_b_kernel(
    q_ptr,             # *bf16, [B, H, D]
    k_ptr,             # *bf16, [N_max, num_kv_heads, D]  Note: in practice we use kv_indices to select rows
    v_ptr,             # *bf16, [N_max, num_kv_heads, D]
    kv_indices_b_ptr,  # *int32, [N_max] subset of kv_indices for this batch
    out_ptr,           # *float32, [B, H, D]
    lse_ptr,           # *float32, [B, H]
    D: tl.constexpr,   # head dim (128)
    H: tl.constexpr,   # num_qo_heads (32)
    gqa_ratio: tl.constexpr,  # H // num_kv_heads (4)
    num_kv_heads: tl.constexpr,  # 8
    sm_scale,          # float32 scalar
    N_max: tl.constexpr,       # max number of tokens per batch (>= actual_num_tokens)
):
    b = tl.program_id(0)  # one program per batch element

    # Precompute base q vector for this batch element
    q_base = q_ptr + b * H * D
    q_vec = tl.zeros((D,), dtype=tl.float32)
    for d in range(0, D):
        q_val = tl.load(q_base + d).to(tl.float32)
        q_vec[d] = q_val

    # Precompute constants
    ln2 = 0.6931471805599453  # math.log(2.0)

    # Process each query head h
    for h in range(0, H):
        # Find corresponding KV head for GQA
        kv_head = h // gqa_ratio  # since gqa_ratio == H // num_kv_heads

        # First pass: compute m and sumexp via streaming logsumexp over tokens
        m = -float('inf')
        sumexp = 0.0  # scalar float32
        for nn in range(0, N_max):
            # If nn >= actual tokens, we skip by masking (nn < actual_num_tokens not directly accessible here;
            # we emulate by loading q_vec which is valid, but logit_tmp becomes NaN if idx out of range;
            # in practice we guard by loading k_ptr when nn < N_max and mask otherwise.
            # Instead, we use an if-like pattern: Triton requires static loop; we can't break. So we assume
            # N_max equals actual_num_tokens at launch and rely on masks via condition nn < N.
            # However, Triton static loop requires no if. We need to pass actual_num_tokens to control loop.
            # To keep it Triton-static, we compute lse only for nn < N via masked loads. But Triton can't mask
            # scalar pointers; so we restructure: we'll not attempt to guard nn via pointer. Instead, we rely
            # on host-side to pass N_max == actual_num_tokens. In forward we set N_max=actual for each b.
            # Therefore, we keep this simple: nn is always < N_max and N_max == actual_num_tokens.
            # Compute dot product
            idx_tmp = tl.load(kv_indices_b_ptr + nn).to(tl.int32)
            # Build k and v base pointers for this token
            k_base = k_ptr + nn * num_kv_heads * D + kv_head * D
            v_base = v_ptr + nn * num_kv_heads * D + kv_head * D

            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_val = tl.load(k_base + d).to(tl.float32)
                k_vec[d] = k_val

            dot_tmp = 0.0
            for d in range(0, D):
                dot_tmp += q_vec[d] * k_vec[d]
            logit_tmp = dot_tmp * sm_scale

            # Streaming logsumexp
            new_m = tl.maximum(m, logit_tmp)
            sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit_tmp - new_m)
            m = new_m

        # lse for this head
        lse_val = m + tl.log(sumexp) / ln2
        # store lse
        lse_base = lse_ptr + b * H + h
        tl.store(lse_base, lse_val)

        # Second pass: compute softmax and accumulate output
        out_base = out_ptr + b * H * D + h * D
        for nn in range(0, N_max):
            idx_tmp = tl.load(kv_indices_b_ptr + nn).to(tl.int32)
            k_base = k_ptr + nn * num_kv_heads * D + kv_head * D
            v_base = v_ptr + nn * num_kv_heads * D + kv_head * D

            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_val = tl.load(k_base + d).to(tl.float32)
                k_vec[d] = k_val

            dot_tmp = 0.0
            for d in range(0, D):
                dot_tmp += q_vec[d] * k_vec[d]
            logit_tmp = dot_tmp * sm_scale

            attn = tl.exp(logit_tmp - m) / sumexp  # softmax
            v_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                v_val = tl.load(v_base + d).to(tl.float32)
                v_vec[d] = v_val

            # Accumulate output vector for this head
            for d in range(0, D):
                # out_ptr is float32
                tl.store(out_base + d, tl.load(out_base + d) + attn * v_vec[d])

    # Kernel ends
```

import torch
import triton
import triton.language as tl


@triton.jit
def _forward_b_kernel(
    q_ptr,            # *bf16, [B, H, D]
    k_ptr,            # *bf16, [N_max, num_kv_heads, D]
    v_ptr,            # *bf16, [N_max, num_kv_heads, D]
    kv_indices_b_ptr, # *int32, [N_max] token indices for this batch
    out_ptr,          # *bf16, [B, H, D]
    lse_ptr,          # *float32, [B, H]
    D: tl.constexpr,  # head dimension (128)
    H: tl.constexpr,  # num_qo_heads (32)
    gqa_ratio: tl.constexpr,  # H // num_kv_heads (4)
    num_kv_heads: tl.constexpr,  # 8
    sm_scale,         # float32 scalar
    N_max: tl.constexpr,        # number of tokens for this batch (actual_num_tokens)
):
    b = tl.program_id(0)  # one program per batch element

    # Preload q vectors for all heads: we need q[b, h] for each head h
    # We will compute q[b, h] on the fly; no need to preload an entire vector.
    # The kernel loops over h.

    # Constants
    ln2 = 0.6931471805599453  # log(2)

    for h in range(0, H):
        kv_head = h // gqa_ratio  # GQA mapping: 32/8 -> 4

        # First pass: compute m and sumexp of logits_scaled over tokens using streaming
        m = -float("inf")
        sumexp = 0.0
        for nn in range(0, N_max):
            idx_tmp = tl.load(kv_indices_b_ptr + nn).to(tl.int32)
            # Base for k and v at this token index
            k_base = k_ptr + nn * num_kv_heads * D + kv_head * D
            v_base = v_ptr + nn * num_kv_heads * D + kv_head * D

            # Load q[b, h] and compute dot with k_base
            q_base = q_ptr + b * H * D + h * D
            q_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                q_val = tl.load(q_base + d).to(tl.float32)
                q_vec[d] = q_val

            k_vec = tl.zeros((D,), dtype=tl.float32)
            for d in range(0, D):
                k_val = tl.load(k_base + d).to(tl.float32)
                k_vec[d] = k_val

            dot = 0.0
            for d in range(0, D):
                dot += q_vec[d] * k_vec[d]

            logit = dot * sm_scale

            new_m = tl.maximum(m, logit)
            sumexp = sumexp * tl.exp(m - new_m) + tl.exp(logit - new_m)
            m = new_m

        # Compute lse for this head
        lse_val = m + tl.log(sumexp) / ln2
        lse_base = lse_ptr + b * H + h
        tl.store(lse_base, lse_val)

        out_base = out_ptr + b * H * D + h * D
        for nn in range(0, N_max):
            idx_tmp = tl.load(kv_indices_b_ptr + nn).to(tl.int32)
            k_base = k_ptr + nn * num_kv_heads * D + kv_head * D
            v_base = v_ptr + nn * num_kv_heads * D + kv_head * D

            q_base = q_ptr + b * H * D + h * D
