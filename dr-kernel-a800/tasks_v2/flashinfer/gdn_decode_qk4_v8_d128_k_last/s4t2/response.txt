@triton.jit
def softplus_and_exp_kernel(
    A_log_ptr,      # float32 [H]
    x_ptr,          # float32 [B*H]
    g_out_ptr,      # float32 [B*H]
    H: tl.constexpr,
    K: tl.constexpr,  # kept for signature consistency; not used here
):
    pid = tl.program_id(axis=0)  # 0 .. B*H-1
    A_log = tl.load(A_log_ptr + (pid % H))

    e = tl.exp(A_log)
    g = tl.exp(-e * s)
    tl.store(g_out_ptr + pid, g)

# Kernel: compute g = exp(-exp(A_log) * softplus(x)) per (b,h); we compute softplus(x) = log(1 + exp(x))
@triton.jit
def softplus_and_exp_kernel(
    A_log_ptr,      # float32 [H]
    x_ptr,          # float32 [B*H]
    g_out_ptr,      # float32 [B*H]
    H: tl.constexpr,
    K: tl.constexpr,  # not used here; kept for signature consistency
):
    pid = tl.program_id(axis=0)  # 0 .. B*H-1
    A_log = tl.load(A_log_ptr + (pid % H))
    x_val = tl.load(x_ptr + pid)
    s = tl.log(1.0 + tl.exp(x_val))  # softplus(x)
    e = tl.exp(A_log)
    g = tl.exp(-e * s)
    tl.store(g_out_ptr + pid, g)


# Kernel: compute beta = sigmoid(b) per (b,h)
@triton.jit
def sigmoid_kernel(
    b_ptr,          # float32 [B*H]
    beta_out_ptr,   # float32 [B*H]
    H: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    b_val = tl.load(b_ptr + pid)
    beta = 1.0 / (1.0 + tl.exp(-b_val))
    tl.store(beta_out_ptr + pid, beta)


# Kernel: compute h_state_vec[i] = sum_j state[i,j] * g - state_remove + state_update
# Inputs:
#  - state_ptr: float32 [V*K], flattened
#  - g_scalar: float32
#  - state_remove: float32
#  - state_update: float32
#  - h_state_ptr: float32 [V] (output vector)

    state_remove,   # float32
    state_update,   # float32
    h_state_ptr,    # [V]

            acc += s_ij
        h_state_i = acc * g_scalar - state_remove + state_update
        tl.store(h_state_ptr + i, h_state_i)
