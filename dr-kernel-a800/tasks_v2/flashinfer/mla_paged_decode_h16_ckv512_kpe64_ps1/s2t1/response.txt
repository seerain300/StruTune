        # for each column by reusing Kc_sub and Kp_sub.

        # logits.

        # out_vec += softmax_t * Kc_row[t, :] for each t. We can compute softmax_t per token as follows:
        # We need to know max across tokens and sum exp across tokens. We already have lse_h = max + log(sum).
        # Let M = max(logits[t, :]) across tokens and S = sum exp(logits[t, :] - M). Then:
        # For each token t, exp_t = exp((dot1+dot2)*sm_scale - M)
        # softmax_t = exp_t / S
        # But we don't have per-token logits in memory. The only way to get M and S without storing all logits is to
        # recompute them and keep M and S updated.

        # Therefore, we will implement the proper computation: recompute logits per token and update M and S while
        # also accumulating out_vec.

        token_sum = tl.zeros((), dtype=tl.float32)

            token_sum += tl.exp(logits_t - token_max)

        # lse per head: lse_h = token_max + log(token_sum); we already stored lse_h
        # Compute attention weights per token
        for t in range(0, L_tokens):
            Kc_row = tl.load(Kc_sub_ptr + t * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
            Kp_row = tl.load(Kp_sub_ptr + t * D2 + tl.arange(0, D2)).to(tl.float32)  # [D2]

            dot2 = tl.zeros((), dtype=tl.float32)

    D1: tl.constexpr,      # head_dim_ckv

    L_tokens: tl.int32,    # number of tokens in this batch element
    sm_scale: tl.float32,  # scaling factor
):

        qp = tl.load(q_pe_rows_ptr + h * D2 + tl.arange(0, D2)).to(tl.float32)   # [D2]

        out_vec = tl.zeros((D1,), dtype=tl.float32)
        token_max = tl.full((), -float("inf"), dtype=tl.float32)
        token_sum = tl.zeros((), dtype=tl.float32)

        for t in range(0, L_tokens):
            Kc_row = tl.load(Kc_sub_ptr + t * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
            Kp_row = tl.load(Kp_sub_ptr + t * D2 + tl.arange(0, D2)).to(tl.float32)  # [D2]

            dot1 = tl.zeros((), dtype=tl.float32)
            dot2 = tl.zeros((), dtype=tl.float32)
            for i in range(0, D1):
                dot1 += qn[i] * Kc_row[i]
            for j in range(0, D2):
                dot2 += qp[j] * Kp_row[j]

            logits_t = (dot1 + dot2) * sm_scale
            token_max = tl.maximum(token_max, logits_t)
            token_sum += tl.exp(logits_t - token_max)

        # lse per head (compute again)
        # Note: we already stored lse per head. Here, we recompute for consistency if needed, but not used.
        # Compute attention weights per token and update out_vec
        for t in range(0, L_tokens):
            Kc_row = tl.load(Kc_sub_ptr + t * D1 + tl.arange(0, D1)).to(tl.float32)  # [D1]
