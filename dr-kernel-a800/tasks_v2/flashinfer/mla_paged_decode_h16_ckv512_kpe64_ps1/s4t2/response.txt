import torch
import math
import triton
import triton.language as tl


@triton.jit
def matvec_row_kernel(
    A_ptr,        # *float32, [1, K]  (qn or qp: single row)
    B_ptr,        # *float32, [M, K]  (Kc or Kp rows)
    C_ptr,        # *float32, [M]     (output vector)
    M,            # int               (number of tokens)
    K: tl.constexpr,  # 512 or 64, constexpr
    BLOCK_K: tl.constexpr = 64,
):
    # One program computes the entire output vector C of length M.
    # A is a single row vector of length K; B has M rows of length K.
    # We compute C[i] = sum_k A[k] * B[i, k] for i in 0..M-1.
    for i in range(0, M):
        acc = 0.0
        for k0 in range(0, K, BLOCK_K):
            k_offsets = k0 + tl.arange(0, BLOCK_K)  # [BLOCK_K]
            a_vec = tl.load(A_ptr + k_offsets)      # [BLOCK_K]
            b_row = tl.load(B_ptr + i * K + k_offsets)  # [BLOCK_K]
            acc += tl.sum(a_vec * b_row, axis=0)
        tl.store(C_ptr + i, acc)


@triton.jit
def softmax_lse_kernel(
    x_ptr,        # *float32, [M] (logits_scaled)
    out_lse_ptr,  # *float32, [1] (lse per head)
    attn_ptr,     # *float32, [M] (attn vector per head)
    M,            # int
    sm_scale,     # float32
):
    # Vectorized computation across M
    idx = tl.arange(0, M)
    x = tl.load(x_ptr + idx)
    max_val = tl.max(x, axis=0)
    x_shift = x - max_val
    x_scaled = x_shift * sm_scale
    exp_x = tl.exp(x_scaled)
    sum_exp = tl.sum(exp_x, axis=0)
    ln2 = 0.6931471805599453  # math.log(2)
    lse = tl.log(sum_exp) / ln2
    tl.store(out_lse_ptr + tl.arange(0, 1), lse)
    attn = exp_x / sum_exp
    tl.store(attn_ptr + idx, attn)


@triton.jit
def matvec_reduce_kernel(
    A_ptr,        # *float32, [M]     (attn vector)
    B_ptr,        # *float32, [M, Kc_dim] (Kc rows)
    C_ptr,        # *float32, [Kc_dim]    (output vector)
    M,            # int
    Kc_dim: tl.constexpr,  # 512
    BLOCK_M: tl.constexpr = 128,
):
    # Accumulate output vector of length Kc_dim
    out_vec = tl.zeros((Kc_dim,), dtype=tl.float32)
    for m0 in range(0, M, BLOCK_M):
        m_offsets = m0 + tl.arange(0, BLOCK_M)
        mask = m_offsets < M
        a_chunk = tl.load(A_ptr + m_offsets, mask=mask, other=0.0)  # [BLOCK_M]
        # We will load B chunk as a matrix [BLOCK_M, Kc_dim] and reduce over rows
        b_chunk = tl.zeros((BLOCK_M, Kc_dim), dtype=tl.float32)
        # Load rows sequentially within the chunk
        for mm in range(BLOCK_M):
            row_idx = m0 + mm
            row_mask = row_idx < M
            b_row = tl.load(B_ptr + row_idx * Kc_dim + tl.arange(0, Kc_dim),
                            mask=row_mask, other=0.0)
            b_chunk[mm, :] = b_row
        # Accumulate dot products across rows
        for rr in range(BLOCK_M):
            row_valid = (m0 + rr) < M
            if row_valid:
                out_vec += tl.sum(b_chunk[rr, :] * a_chunk[rr], axis=0)
    tl.store(C_ptr + tl.arange(0, Kc_dim), out_vec)

  3) matvec_reduce_d_kernel: computes out[d] = sum_i A[i] * B[i, d]. Grid = (Kc_dim,). We’ll loop over M in tiles (BLOCK_M) using tl.static_range and accumulate into out[d]. This is a standard reduction kernel.

- In ModelNew.forward:
  - Ensure inputs are on CUDA.
  - For each batch b, compute M = end - start from kv_indptr[b:b+1].
  - If M <= 0, output zeros and continue.
  - Gather Kc and Kp rows and cast to float32 for compute.
  - For each head h:
    - Run matvec_k_kernel twice to compute logits_qn[M] and logits_qp[M], sum them, scale by sm_scale, then run softmax_lse_kernel to get lse[b, h] and attn[M].
    - Run matvec_reduce_d_kernel to compute out_vec[Kc_dim].
    - Store out_vec into output[b, h, :].
  - Return output cast to bfloat16 and lse as float32.

This avoids any torch matmul/softmax/logsumexp in the host and uses only Triton kernels. The use of tl.static_range with constexpr K and Kc_dim ensures the kernels compile reliably in the evaluation environment.

import math
import triton
import triton.language as tl


@triton.jit
def matvec_k_kernel(
    A_ptr,        # *float32, [1, K]  (A is a single row vector of length K)
    B_ptr,        # *float32, [M, K]  (matrix of rows, we access by fixed i)
    C_ptr,        # *float32, [M]     (output vector)
    i,            # int               (output index i in 0..M-1)
    K: tl.constexpr,           # constexpr: Kc_dim or Kp_dim
    BLOCK_K: tl.constexpr = 64,  # tile size along K
):
    # Compute C[i] = sum_k A[k] * B[i, k]
    acc = 0.0
    for k0 in tl.static_range(0, K, BLOCK_K):
        k_offsets = k0 + tl.arange(0, BLOCK_K)  # [BLOCK_K]
        a_vec = tl.load(A_ptr + k_offsets)      # [BLOCK_K]
        b_row = tl.load(B_ptr + i * K + k_offsets)  # [BLOCK_K]
        acc += tl.sum(a_vec * b_row, axis=0)
    tl.store(C_ptr + i, acc)


@triton.jit
def softmax_lse_kernel(
    x_ptr,        # *float32, [M] (logits_scaled)
    out_lse_ptr,  # *float32, [1] (lse per head)
    attn_ptr,     # *float32, [M] (attn vector per head)
    M,            # int
    sm_scale,     # float32
):
    idx = tl.arange(0, M)
    x = tl.load(x_ptr + idx)
    max_val = tl.max(x, axis=0)
    x_shift = x - max_val
    x_scaled = x_shift * sm_scale
    exp_x = tl.exp(x_scaled)
    sum_exp = tl.sum(exp_x, axis=0)
    ln2 = 0.6931471805599453  # math.log(2)
    lse = tl.log(sum_exp) / ln2
    tl.store(out_lse_ptr + tl.arange(0, 1), lse)
    attn = exp_x / sum_exp
    tl.store(attn_ptr + idx, attn)
