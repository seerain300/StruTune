- Given:
  - q_nope[t]: [16, 512]
  - q_pe[t]: [16, 64]
  - ckv_cache: [num_pages, 64, 512], where num_pages=8462
  - kpe_cache: [num_pages, 64, 64]

- For each token, we build:
  - Kc = ckv_cache[valid_indices] → [num_valid, 512]
  - Kp = kpe_cache[valid_indices] → [num_valid, 64]
  - qn[h] = q_nope[t, h] → [512]
  - qp[h] = q_pe[t, h] → [64]
- Compute logits for head h:
  - logits = (qn[h] @ Kc.T) + (qp[h] @ Kp.T) → [num_valid]
  - logits_scaled = logits * sm_scale
  - lse[h] = logsumexp(logits_scaled, base-2)

Optimization with Triton:
- We cannot avoid the per-token loop, but we can offload all per-token math to Triton:
  1) Row-wise matmul kernel: compute C[j] = sum_i A[i] * B[j, i], where A is 1D (length M), B is 2D (shape N x M), output C is 1D (length N). We’ll use this twice per head: once for qn[h] @ Kc.T and once for qp[h] @ Kp.T.
  2) Softmax with base-2 logsumexp kernel: given a row vector X (length TOPK, padded with -inf), compute stable softmax into Out and the base-2 logsumexp into LSE. We will pass a validity mask of length TOPK: 1 for valid, 0 for padded. Invalid positions are set to -inf and won’t be stored.
  3) Reduction kernel: given attn vector [num_valid] and Kc matrix [num_valid, 512], compute out [512] = sum_i attn[i] * Kc[i, :]. We do this in chunks over num_valid.

Implementation details:
- We will not use any torch matmul, softmax, or gather in forward. All data selection (indices) and computation will be done by Triton kernels.
- For q_nope and q_pe, since we only need the h-th row, we can pass A pointers directly.
- We pad/topk for the softmax kernel as 2048 and use a validity mask to ignore padded entries. This ensures kernels handle dynamic num_valid up to 2048.

import math
import torch
import triton
import triton.language as tl


# Triton kernel: row-wise matmul C = A_row @ B, where:
# A_row is a 1D vector of length M.
# B is a 2D matrix of shape (N, M). Output C is a 1D vector of length N.
# We compute: for j in [0, N), C[j] = sum_i A_row[i] * B[j, i]
@triton.jit
def matmul_row(A_ptr, B_ptr, C_ptr,
               M: tl.constexpr, N: tl.constexpr, BLOCK_N: tl.constexpr):
    # Single-program reduction over N in chunks
    for j_start in range(0, N, BLOCK_N):
        j = j_start + tl.arange(0, BLOCK_N)
        mask_j = j < N
        # Load A row (length M), contiguous
        a = tl.load(A_ptr)
        # Load B block: shape [BLOCK_N, M]
        b_ptrs = B_ptr + j[:, None] * M + tl.arange(0, M)
        b = tl.load(b_ptrs, mask=mask_j[:, None], other=0.0)
        # Accumulate dot products over M
        acc = tl.zeros([BLOCK_N], dtype=tl.float32)
        for m in range(0, M):
            acc += b[:, m] * a[m]
        # Store results
        tl.store(C_ptr + j, acc, mask=mask_j)


# Triton kernel: row-wise softmax (numerically stable) and base-2 logsumexp.
# X is 1D vector of length TOPK (e.g., 2048).
# Valid is 1D mask (int32) of length TOPK: 1 for valid, 0 for padded.
# Outputs:
#   Out: softmax probabilities (float32) of length TOPK (only valid positions written via Valid mask).
#   LSE_ptr: scalar (float32) storing logsumexp base-2 for this row.
@triton.jit
def softmax_logsumexp2_row(X_ptr, Valid_ptr, Out_ptr, LSE_ptr,
                            TOPK: tl.constexpr, NUM_VALID: tl.constexpr):
    # Compute max over valid entries
    m = -float("inf")
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)  # int32 0/1
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        m = tl.maximum(m, xj)
    # Compute sum of exp over valid entries
    s = 0.0
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        s += tl.exp(xj - m)
    # Write softmax into Out (valid positions only; invalid entries already -inf)
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        prob = tl.exp(xj - m) / s
        tl.store(Out_ptr + j, prob)
    # Base-2 logsumexp: lse = m + log(s), then divide by ln(2)
    lse_val = m + tl.log(s)
    tl.store(LSE_ptr, lse_val / 1.4426950408889634)  # 1 / ln(2)


# Triton kernel: compute out = attn @ Kc_row
# attn is a 1D vector of length NUM_VALID.
# Kc is a 2D matrix of shape (NUM_VALID, OUT), OUT is a constexpr (e.g., 512).
# out is a 1D vector of length OUT, computed as sum_i attn[i] * Kc[i, :].
@triton.jit
def reduction_row(Attn_ptr, Kc_ptr, Out_ptr,
                  NUM_VALID: tl.constexpr, OUT: tl.constexpr, BLOCK_N: tl.constexpr):
    acc = tl.zeros([OUT], dtype=tl.float32)
    for i_start in range(0, NUM_VALID, BLOCK_N):
        i = i_start + tl.arange(0, BLOCK_N)
        mask_i = i < NUM_VALID
        attn_chunk = tl.load(Attn_ptr + i, mask=mask_i, other=0.0)
        kc_ptrs = Kc_ptr + i[:, None] * OUT + tl.arange(0, OUT)
        kc = tl.load(kc_ptrs, mask=mask_i[:, None], other=0.0)
        # Accumulate over the chunk
        for ii in range(0, BLOCK_N):
            ii_mask = mask_i[ii]
            # attn_chunk[ii] is zero when ii_mask is False
            acc += attn_chunk[ii] * kc[ii, :]
    tl.store(Out_ptr, acc)

                # The previous instruction attempted to compute logits_qp using matmul_row(qp[h], Kp). That is correct:
                # Kp is [num_valid, 64], A_row = qp[h] is [64] => output [num_valid].

                logits = logits_qn + logits_qp  # [num_valid]

                # 2) Softmax (base-2 logsumexp) on logits_scaled

                output[t, h] = out_vec

        return output, lse

import math
import torch
import triton
import triton.language as tl


# Triton kernel: row-wise matmul C = A_row @ B, where:
# A_row is a 1D vector of length M.
# B is a 2D matrix of shape (N, M). Output C is a 1D vector of length N.
# For j in [0, N), C[j] = sum_i A_row[i] * B[j, i]
@triton.jit
def matmul_row(A_ptr, B_ptr, C_ptr,
               M: tl.constexpr, N: tl.constexpr, BLOCK_N: tl.constexpr):
    for j_start in range(0, N, BLOCK_N):
        j = j_start + tl.arange(0, BLOCK_N)
        mask_j = j < N
        # Load A row (length M), contiguous
        a = tl.load(A_ptr)  # A_ptr points to a contiguous 1D vector
        # Load B block: shape [BLOCK_N, M]
        b_ptrs = B_ptr + j[:, None] * M + tl.arange(0, M)
        b = tl.load(b_ptrs, mask=mask_j[:, None], other=0.0)
        # Accumulate dot products over M
        acc = tl.zeros([BLOCK_N], dtype=tl.float32)
        for m in range(0, M):
            acc += b[:, m] * a[m]
        # Store results
        tl.store(C_ptr + j, acc, mask=mask_j)

# Valid is 1D mask (int32) of length TOPK: 1 for valid, 0 for padded.
# Outputs:
#   Out: softmax probabilities (float32) of length TOPK (only valid positions are meaningful).
