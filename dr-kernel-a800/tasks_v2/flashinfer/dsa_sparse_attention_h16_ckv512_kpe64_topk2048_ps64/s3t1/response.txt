  - Final output per head: attn @ Kc. We’ll implement this in Triton as a reduction across num_valid.

3) Kernel design:
- Triton matmul kernel for row-wise matmul: given A row vector [M] and B matrix [N, M], produce C [N] with C[j] = sum_i A[i] * B[j, i]. We’ll launch grid=(1,) and compute all N in a single program. - Triton softmax kernel per head: given input vector x (length up to MAX_BLOCK_SIZE=2048), compute:

Note: This code defines all required Triton kernels and launches them from ModelNew.forward. It removes all torch matmul/softmax usage in forward. The kernels are:
- matmul_row: computes a row-wise matmul C = A_row @ B^T, where A_row is 1D, B is 2D.
- softmax_logsumexp2_row_kernel: computes stable softmax per row (1D vector) and returns both softmax vector and logsumexp (base-2) scalar.

import math
import torch
import triton
import triton.language as tl


# Triton kernel: row-wise matmul.
# A is a 1D vector of length M (row to be multiplied).
# B is a 2D matrix of shape (N, M). Output C is a 1D vector of length N.
# This kernel computes: C[j] = sum_i A[i] * B[j, i] for all j in [0, N).
@triton.jit
def matmul_row(A_ptr, B_ptr, C_ptr,
               M: tl.constexpr, N: tl.constexpr,  # M is number of columns in B (and length of A)
               BLOCK_N: tl.constexpr):
    # Single-program reduction over N in chunks
    # For each j, compute dot(A, B[j, :])
    # We iterate j from 0 to N, in steps of BLOCK_N.
    for j_start in range(0, N, BLOCK_N):
        j = j_start + tl.arange(0, BLOCK_N)
        mask_j = j < N
        # Load A row
        a = tl.load(A_ptr, mask=True)  # always valid since A_ptr points to contiguous [M]
        # Load B block: shape [BLOCK_N, M]
        b_ptrs = B_ptr + j[:, None] * M + tl.arange(0, M)
        b = tl.load(b_ptrs, mask=mask_j[:, None], other=0.0)
        # Accumulate dot products: sum over M
        acc = tl.zeros([BLOCK_N], dtype=tl.float32)
        # We assume M is known at launch (constexpr). Triton will unroll.
        for m in range(0, M):
            acc += b[:, m] * a[m]
        # Store results
        out = acc
        tl.store(C_ptr + j, out, mask=mask_j)


# Triton kernel: row-wise softmax with logsumexp (base-2) per head.
# X is 1D vector of length TOPK (up to MAX_BLOCK_SIZE).
# Valid is 1D mask (int32) of length TOPK: 1 for valid, 0 for padded.
# We mask invalid entries by setting them to -inf before softmax.
# Outputs:
#   out: softmax probabilities (float32) of length TOPK
#   lse: logsumexp base-2 scalar (float32)
@triton.jit
def softmax_logsumexp2_row_kernel(X_ptr, Valid_ptr, Out_ptr, LSE_ptr,
                                   TOPK: tl.constexpr, NUM_VALID: tl.constexpr):
    # Only one program handles this row
    # Compute max
    m = -float("inf")
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)  # int32 0/1
        xj = tl.load(X_ptr + j)
        # If invalid, set to -inf
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        m = tl.maximum(m, xj)
    # Compute sum of exp
    s = 0.0
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        s += tl.exp(xj - m)
    # Write softmax and lse
    # Note: we store only valid positions; invalid positions won't be written via mask.
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        prob = tl.exp(xj - m) / s
        tl.store(Out_ptr + j, prob, mask=(is_valid != 0))
    # LSE in base-2: lse = m + log(s); base2 = / log(2)
    lse_val = m + tl.log(s)
    # Triton doesn't have direct store of scalar, but we can write via pointer
    tl.store(LSE_ptr, lse_val / 1.4426950408889634)  # 1 / ln(2)


# Triton kernel: compute out = attn @ Kc_row
# attn is a 1D vector of length N (NUM_VALID).
# Kc is a 2D matrix of shape (N, OUT), OUT=512 (constexpr).
# out is a 1D vector of length OUT, computed as sum_i attn[i] * Kc[i, :].
@triton.jit
def reduction_row(Attn_ptr, Kc_ptr, Out_ptr,
                  NUM_VALID: tl.constexpr, OUT: tl.constexpr, BLOCK_N: tl.constexpr):
    # Single-program reduction across NUM_VALID in chunks
    acc = tl.zeros([OUT], dtype=tl.float32)
    for i_start in range(0, NUM_VALID, BLOCK_N):
        i = i_start + tl.arange(0, BLOCK_N)
        mask_i = i < NUM_VALID
        # Load attn chunk
        attn_chunk = tl.load(Attn_ptr + i, mask=mask_i, other=0.0)
        # Load Kc rows for this chunk: shape [BLOCK_N, OUT]
        kc_ptrs = Kc_ptr + i[:, None] * OUT + tl.arange(0, OUT)
        kc = tl.load(kc_ptrs, mask=mask_i[:, None], other=0.0)
        # Accumulate over the chunk: acc += sum_i attn_chunk[i] * kc[i, :]
        for ii in range(0, BLOCK_N):
            ii_mask = mask_i[ii]
            # Skip if masked
            # Triton supports per-element masking via tl.where; here we just ensure attn_chunk[ii] is zero when masked.
            acc += attn_chunk[ii] * kc[ii, :]
    tl.store(Out_ptr, acc)

        assert num_qo_heads == self.num_qo_heads
        assert head_dim_ckv == self.head_dim_ckv
        assert q_pe.shape[-1] == self.head_dim_kpe
        assert ckv_cache.shape[1] == 64
        assert kpe_cache.shape[1] == 64

        # Flatten paged KV cache to token-level: [num_pages, 64, 512] -> [num_pages*64, 512]
        Kc_all = ckv_cache.reshape(-1, self.head_dim_ckv).to(torch.float32)  # [total_kv_tokens, 512]
        Kp_all = kpe_cache.reshape(-1, self.head_dim_kpe).to(torch.float32)  # [total_kv_tokens, 64]
        device = q_nope.device

        output = torch.empty((num_tokens, num_qo_heads, self.head_dim_ckv), dtype=torch.float32, device=device)
        lse = torch.empty((num_tokens, num_qo_heads), dtype=torch.float32, device=device)

        for t in range(num_tokens):

            valid_indices = indices[valid_mask].to(torch.long)
            num_valid = valid_indices.numel()

            # Triton wrote to lse_elem (allocated on device). We can do: lse[t, h] = lse_elem.clone()

            # But since we're in Triton-kernel driven code, ensure we assign using .item() conversion.
            # We can just compute lse[t, h] = lse_elem.clone().detach() after kernel returns.
            # To avoid .item(), we can keep scalar in tensor form by using out_vec as temporary; but out_vec is overwritten.

        output_bf16 = output.to(torch.bfloat16)

```python
import math
import torch
import triton
import triton.language as tl


# Triton kernel: row-wise matmul C = A_row @ B, where:
# A_row is a 1D vector of length M.
# B is a 2D matrix of shape (N, M). Output C is a 1D vector of length N.
@triton.jit
def matmul_row(A_ptr, B_ptr, C_ptr,
               M: tl.constexpr, N: tl.constexpr,
               BLOCK_N: tl.constexpr):
    # Single-program reduction over N in chunks
    # For each chunk [j_start, j_start+BLOCK_N), compute dot(A, B[j, :]) for j in chunk
    for j_start in range(0, N, BLOCK_N):
        j = j_start + tl.arange(0, BLOCK_N)
        mask_j = j < N
        # Load A vector
        a = tl.load(A_ptr)  # A_ptr points to [M], contiguous
        # Load B block: shape [BLOCK_N, M]
        b_ptrs = B_ptr + j[:, None] * M + tl.arange(0, M)
        b = tl.load(b_ptrs, mask=mask_j[:, None], other=0.0)
        # Accumulate dot products over M
        acc = tl.zeros([BLOCK_N], dtype=tl.float32)
        for m in range(0, M):
            acc += b[:, m] * a[m]
        # Store results
        tl.store(C_ptr + j, acc, mask=mask_j)


# Triton kernel: row-wise softmax (numerically stable) and base-2 logsumexp.
# X is 1D vector of length TOPK (up to MAX_BLOCK_SIZE).
# Valid is 1D mask (int32) of length TOPK: 1 for valid, 0 for padded.
# Outputs:
#   Out: softmax probabilities (float32) of length TOPK (only valid positions written via mask logic).
#   LSE_ptr: scalar (float32) storing logsumexp base-2 for this row.
@triton.jit
def softmax_logsumexp2_row(X_ptr, Valid_ptr, Out_ptr, LSE_ptr,
                            TOPK: tl.constexpr, NUM_VALID: tl.constexpr):
    # Compute max
    m = -float("inf")
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)  # int32 0/1
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        m = tl.maximum(m, xj)
    # Compute sum of exp
    s = 0.0
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        s += tl.exp(xj - m)
    # Write softmax into Out (valid positions only)
    for j in range(0, TOPK):
        is_valid = tl.load(Valid_ptr + j)
        xj = tl.load(X_ptr + j)
        xj = tl.where(is_valid != 0, xj, -float("inf"))
        prob = tl.exp(xj - m) / s
        # We store all entries but invalid positions will be zeroed on host side. Here we can skip store if invalid.
        # Triton doesn't support conditional store via mask in for loop easily; we rely on host to zero invalid after.
        tl.store(Out_ptr + j, prob)
    # Compute base-2 logsumexp: lse = m + log(s) then divide by ln(2)
    lse_val = m + tl.log(s)
    tl.store(LSE_ptr, lse_val / 1.4426950408889634)  # 1 / ln(2)


# Triton kernel: compute out = attn @ Kc_row
# attn is a 1D vector of length N (NUM_VALID).
# Kc is a 2D matrix of shape (N, OUT), OUT is a constexpr (512).
# out is a 1D vector of length OUT, computed as sum_i attn[i] * Kc[i, :].
@triton.jit
def reduction_row(Attn_ptr, Kc_ptr, Out_ptr,
                  NUM_VALID: tl.constexpr, OUT: tl.constexpr, BLOCK_N: tl.constexpr):
    acc = tl.zeros([OUT], dtype=tl.float32)
    for i_start in range(0, NUM_VALID, BLOCK_N):
        i = i_start + tl.arange(0, BLOCK_N)
        mask_i = i < NUM_VALID
        attn_chunk = tl.load(Attn_ptr + i, mask=mask_i, other=0.0)
        kc_ptrs = Kc_ptr + i[:, None] * OUT + tl.arange(0, OUT)
        kc = tl.load(kc_ptrs, mask=mask_i[:, None], other=0.0)
        # Accumulate over the chunk
        for ii in range(0, BLOCK_N):
            ii_mask = mask_i[ii]
            # attn_chunk[ii] is zero when ii_mask is False
            acc += attn_chunk[ii] * kc[ii, :]
    tl.store(Out_ptr, acc)
