- Calls them from ModelNew.forward

Analysis and plan

- Original behavior per token t:
  - Gather Kc and Kp rows from flattened caches using sparse_indices[t, :].
  - For each head h, compute logits[h] as qn[h] @ Kc.T + qp[h] @ Kp.T over the num_valid candidates.
  - Scale by sm_scale, compute lse per head: logsumexp(logits_scaled) / log(2).
  - Compute softmax(logits_scaled) and output[h] = softmax @ Kc, then store in output[t, h, :].
  - If no valid indices, output[t] = 0.

- Optimization approach:
  - We will implement a single Triton kernel that:
    - Processes one token t at a time.
    - For each head h, loops over candidates in chunks, computes logits, applies scaling, computes softmax in a numerically stable online fashion, and finally computes the output vector via a weighted reduction over Kc rows.
  - This avoids launching multiple kernels and keeps all math in Triton. The kernel will:
    1) Load indices for this token.
    2) For each head h:
       a) Initialize lse_acc and sum_exp (for online logsumexp).
       b) Loop over candidates:
          - Load Kc_row and Kp_row for the candidate.
          - Compute qn[h] @ Kc_row + qp[h] @ Kp_row and store in logits buffer for that head.
          - Update lse_acc and sum_exp using the scaled logit (online logsumexp).
       c) After computing lse_acc, loop again to compute attention weights and output vector:
          - For each candidate i, attn_i = exp((logits[i] - lse_acc) * sm_scale) / sum_exp.
          - Load Kc_row[i], accumulate out_vec += attn_i * Kc_row[i].
       d) Store out_vec to output[t, h, :].
  - This design ensures a single Triton kernel is invoked from ModelNew.forward and performs the heavy work. It fuses per-token computation and avoids torch matmul/softmax in the hot path.

- Implementation details:
  - Inputs to the kernel: q_nope, q_pe, Kc_all, Kp_all, sparse_indices, output, lse, along with shape parameters N, H, Dk, Dp, topk, and sm_scale.
  - Compute in float32; output is stored as float32 in the kernel, then we cast to bfloat16 in the host before returning (to match the original’s output dtype). LSE is stored as float32.
  - We handle padding indices (-1) by masking out invalid candidates; for no-valid case, we return zeros. We guard this in the host code as a fallback; the kernel can also write zeros, but for simplicity, host fallback is used for t with empty valid indices.

import math
import torch
import triton
import triton.language as tl


@triton.jit
def attention_token_kernel(
    q_nope_ptr, q_pe_ptr, Kc_all_ptr, Kp_all_ptr, sparse_idx_ptr,
    out_ptr, lse_ptr,
    N, H, Dk, Dp, topk,
    sm_scale, log2,
    BLOCK_H: tl.constexpr  # e.g., 16
):
    # One program per token
    t = tl.program_id(0)

    # Base offsets for q_nope and q_pe
    # q_nope: [N, H, Dk], q_pe: [N, H, Dp]
    base_nope = t * H * Dk
    base_pe = t * H * Dp

    # Load all indices for this token; sparse_idx_ptr is flattened [N*topk]
    base_idx = t * topk
    idx_vals = tl.load(sparse_idx_ptr + base_idx + tl.arange(0, topk), mask=tl.arange(0, topk) < topk, other=-1)

    # Determine number of valid entries
    valid_mask = idx_vals != -1
    num_valid = tl.sum(valid_mask, axis=0)

    # Process each head
    for h in range(0, BLOCK_H):
        # Initialize variables for online logsumexp
        # We'll compute lse per head
        # Note: BLOCK_H is expected to be H=16 here; we assume H==BLOCK_H for simplicity.
        lse_acc = tl.full((), -1e30, dtype=tl.float32)
        sum_exp = tl.full((), 0.0, dtype=tl.float32)

        # Compute logits and update lse in a numerically stable way
        for i in range(0, topk):
            if valid_mask[i]:
                tok_idx = idx_vals[i].to(tl.int32)
                Kc_row_ptr = Kc_all_ptr + tok_idx * Dk
                Kp_row_ptr = Kp_all_ptr + tok_idx * Dp

                # Load qn[h] and qp[h]
                qn_vec = tl.load(q_nope_ptr + base_nope + h * Dk + tl.arange(0, Dk), mask=tl.arange(0, Dk) < Dk, other=0.0)
                qp_vec = tl.load(q_pe_ptr + base_pe + h * Dp + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0)

                # Load K rows
                Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dk), mask=tl.arange(0, Dk) < Dk, other=0.0)
                Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0)

                # Dot products
                dot_qn = tl.sum(qn_vec * Kc_row, axis=0)
                dot_qp = tl.sum(qp_vec * Kp_row, axis=0)
                logit = dot_qn + dot_qp  # scalar float32

                # Online logsumexp update (base-2 scaling later)
                logit_scaled = logit * sm_scale
                m = tl.maximum(lse_acc, logit_scaled)
                sum_exp = sum_exp * tl.exp(lse_acc - m) + tl.exp(logit_scaled - m)
                lse_acc = m

        # Now compute attention weights and output vector
        out_vec = tl.zeros((Dk,), dtype=tl.float32)

        for i in range(0, topk):
            if valid_mask[i]:
                tok_idx = idx_vals[i].to(tl.int32)
                Kc_row_ptr = Kc_all_ptr + tok_idx * Dk
                Kp_row_ptr = Kp_all_ptr + tok_idx * Dp

                qn_vec = tl.load(q_nope_ptr + base_nope + h * Dk + tl.arange(0, Dk), mask=tl.arange(0, Dk) < Dk, other=0.0)
                qp_vec = tl.load(q_pe_ptr + base_pe + h * Dp + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0)

                Kc_row = tl.load(Kc_row_ptr + tl.arange(0, Dk), mask=tl.arange(0, Dk) < Dk, other=0.0)
                Kp_row = tl.load(Kp_row_ptr + tl.arange(0, Dp), mask=tl.arange(0, Dp) < Dp, other=0.0)

                dot_qn = tl.sum(qn_vec * Kc_row, axis=0)
                dot_qp = tl.sum(qp_vec * Kp_row, axis=0)
                logit = dot_qn + dot_qp

                attn_i = tl.exp((logit * sm_scale) - lse_acc) / sum_exp  # softmax weight for this candidate
                out_vec = out_vec + attn_i * Kc_row

        # Store output[t, h, :]
        out_row_base = t * H * Dk + h * Dk
        tl.store(out_ptr + out_row_base + tl.arange(0, Dk), out_vec, mask=tl.arange(0, Dk) < Dk)

        # Store lse[t, h] = lse_acc / log(2)
        tl.store(lse_ptr + t * H + h, lse_acc / log2)


class ModelNew(torch.nn.Module):
    def forward(self, q_nope, q_pe, ckv_cache, kpe_cache, sparse_indices, sm_scale):
        # Ensure CUDA device
        device = q_nope.device
        assert device.type == "cuda", "Triton kernels require CUDA tensors."

        # Shapes
        N, H, Dk = q_nope.shape
        Dp = q_pe.shape[2]
        num_pages, _, _ = ckv_cache.shape
        assert q_pe.shape[1] == H, "num_qo_heads must match q_pe's head dimension."
        assert Dk == ckv_cache.shape[2], "head_dim_ckv must match ckv_cache last dim."
        assert kpe_cache.shape[2] == Dp, "head_dim_kpe must match kpe_cache last dim."
        assert sparse_indices.shape[0] == N, "sparse_indices first dim must match num_tokens."

        # Cast inputs to float32 for compute
        q_nope = q_nope.contiguous().to(torch.float32)  # [N, H, Dk]
        q_pe = q_pe.contiguous().to(torch.float32)     # [N, H, Dp]

        # Flatten paged KV cache
        Kc_all = ckv_cache.reshape(-1, Dk).contiguous().to(torch.float32)  # [(num_pages*64), Dk]
        Kp_all = kpe_cache.reshape(-1, Dp).contiguous().to(torch.float32)  # [(num_pages*64), Dp]

        # Flatten sparse_indices to [N*topk]
        topk = sparse_indices.shape[1]
        # Ensure indices are on device
        sparse_indices = sparse_indices.contiguous().to(torch.int32)

        # Allocate outputs
        out = torch.empty((N, H, Dk), dtype=torch.float32, device=device)  # per-triton float32
        lse = torch.empty((N, H), dtype=torch.float32, device=device)

        # Launch Triton kernel: one program per token
        grid = (N,)
        attention_token_kernel[grid](
            q_nope, q_pe, Kc_all, Kp_all, sparse_indices,
            out, lse,
            N, H, Dk, Dp, topk,
            sm_scale, math.log(2.0),
            BLOCK_H=H,
        )

        # Return in desired dtypes (output bfloat16, lse float32)
        output = out.to(torch.bfloat16)
        return output, lse

    out_ptr, lse_ptr,
    N, H, Dk, Dp, topk,
    sm_scale, inv_log2,
    BLOCK_K: tl.constexpr  # number of candidates processed per iteration; e.g., 256
):

    base_nope = t * H * Dk + h * Dk
    base_pe = t * H * Dp + h * Dp

    base_idx = t * topk
    offs = tl.arange(0, topk)
    idx_vals = tl.load(sparse_idx_ptr + base_idx + offs, mask=offs < topk, other=-1)

    # Count valid entries
    valid_mask = idx_vals != -1
    num_valid = tl.sum(valid_mask, axis=0)

    # Online logsumexp initialization
    m = tl.full((), -1.0e30, dtype=tl.float32)  # running max
    sum_exp = tl.full((), 0.0, dtype=tl.float32)  # running sum in the max reference

    # First pass: compute max and sum for logsumexp over valid candidates
    # We will store all logits for reuse in second pass to compute exact attention
    # Create a small logits vector in registers (use BLOCK_K sized chunks)
    # Store each logit into out_ptr for this head (we can write to out as intermediate)
    out_row_base = t * H * Dk + h * Dk  # used for storing final output
    lse_val = tl.full((), 0.0, dtype=tl.float32)

    for chunk in range(0, topk, BLOCK_K):
        k = chunk + tl.arange(0, BLOCK_K)
        mask_k = k < topk

        for i in range(0, BLOCK_K):

                    tok_idx = idx_vals[i].to(tl.int32)
                    Kc_row_ptr = Kc_all_ptr + tok_idx * Dk
                    Kp_row_ptr = Kp_all_ptr + tok_idx * Dp
